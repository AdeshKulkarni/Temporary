import React, { useState, useMemo, useCallback, useEffect } from "react";
import {
  Box,
  Button,
  Snackbar,
  Alert,
  Typography,
  Stack,
  Divider,
  Paper,
  DialogContent,
  Dialog,
  CircularProgress,
} from "@mui/material";
import { DataGrid } from "@mui/x-data-grid";
import axios from "axios"; // 👈 Import axios
import { Close, Done, DoNotDisturb } from "@mui/icons-material";
import { useNavigate } from "react-router-dom";

export default function AssignRequests() {
  const user = useMemo(() => JSON.parse(localStorage.getItem("user")), []);

  // State for pending requests
  const [rows, setRows] = useState([]);
  const [displayText, setDisplayText] = useState("Loading ...");

  // State for cancelled/rejected requests
  const [cancelledRows, setCancelledRows] = useState([]);
  const [cancelledDisplayText, setCancelledDisplayText] =
    useState("Loading ...");

  // State to manage which view is active
  const [viewMode, setViewMode] = useState("pending"); // 'pending' or 'cancelledRejected'

  const [selectionModel, setSelectionModel] = useState([]);
  const [snackbar, setSnackbar] = useState({
    open: false,
    message: "",
    severity: "success",
  });
  const [loadOpen, setLoadOpen] = useState(false);

  const navigate = useNavigate();

  const handleCloseSnackbar = () => {
    setSnackbar({ ...snackbar, open: false });
  };

  // Columns for the Pending Requests DataGrid
  const columns = [
    { field: "SNO", headerName: "S.No", flex: 0.5, align: "center", headerAlign: "center" },
    { field: "FRN_NO", headerName: "FRN", flex: 2, align: "center", headerAlign: "center" },
    { field: "ACCOUNT_NUMBER", headerName: "Account Number", flex: 2, align: "center", headerAlign: "center" },
    { field: "ACCOUNT_NAME", headerName: "Account Name", flex: 3 },
    { field: "NAME", headerName: "Assignment Type", flex: 1.5, align: "center", headerAlign: "center" },
    { field: "REQUESTED_BY", headerName: "Requested by", flex: 1.5, align: "center", headerAlign: "center" },
    { field: "REQUESTED_TIME", headerName: "Requested at", flex: 2, align: "center", headerAlign: "center" },
  ];

  // Columns for the Cancelled/Rejected Requests DataGrid
  const cancelledColumns = [
    { field: "SNO", headerName: "S.No", flex: 0.5, align: "center", headerAlign: "center" },
    { field: "FRN_NO", headerName: "FRN", flex: 2, align: "center", headerAlign: "center" },
    { field: "ACCOUNT_NUMBER", headerName: "Account Number", flex: 2, align: "center", headerAlign: "center" },
    { field: "ACCOUNT_NAME", headerName: "Account Name", flex: 3 },
    { field: "NAME", headerName: "Assignment Type", flex: 1.5, align: "center", headerAlign: "center" },
    { field: "REQUESTED_BY", headerName: "Requested by", flex: 1.5, align: "center", headerAlign: "center" },
    { field: "REQUESTED_TIME", headerName: "Requested at", flex: 2, align: "center", headerAlign: "center" },
  ];

  const getSelectedRequests = useCallback(
    (selectionModel) => {
      return rows.filter((row) => selectionModel.includes(row.index));
    },
    [rows]
  );

  const fetchPendingData = useCallback(async () => {
    try {
      const { data } = await axios.post("/Server/assignAudit/getPendingRequests", {});
      if (data?.result?.requests) {
        setRows(
          data.result.requests.map((row, index) => ({ ...row, index })) || []
        );
        setDisplayText("No pending requests.");
      } else {
        setRows([]);
        setDisplayText("No pending requests.");
      }
    } catch (error) {
      console.error("Failed to load pending requests!", error);
      setRows([]);
      setDisplayText("Error loading data.");
      setSnackbar({
        open: true,
        message: error.response?.data?.message || "Could not fetch pending requests.",
        severity: "error",
      });
    }
  }, []);

  const fetchCancelledData = useCallback(async () => {
    try {
      const { data } = await axios.post("/Server/assignAudit/getCancelledRequests", {});
      if (data?.result?.requests) {
        setCancelledRows(
          data.result.requests.map((row, index) => ({ ...row, index })) || []
        );
        setCancelledDisplayText("No cancelled or rejected requests found.");
      } else {
        setCancelledRows([]);
        setCancelledDisplayText("No cancelled or rejected requests found.");
      }
    } catch (error) {
      console.error("Failed to load cancelled/rejected requests!", error);
      setCancelledRows([]);
      setCancelledDisplayText("Error loading data.");
      setSnackbar({
        open: true,
        message: error.response?.data?.message || "Could not fetch cancelled requests.",
        severity: "error",
      });
    }
  }, []);

  const handleAction = useCallback(
    async (selectionModel, type) => {
      setLoadOpen(true);
      const requests = getSelectedRequests(selectionModel);
      const payload = { requests, type };
      const actionVerb = type === "2" ? "approve" : type === "3" ? "reject" : "cancel";
      const actionedVerb = type === "2" ? "approved" : type === "3" ? "rejected" : "cancelled";

      try {
        const { data } = await axios.post("/Server/assignAudit/actionOnRequests", payload);
        if (data?.result?.status === "true") {
          setSelectionModel([]);
          setSnackbar({
            open: true,
            message: `Request${selectionModel.length !== 1 ? "s" : ""} ${actionedVerb}.`,
            severity: type === "2" ? "success" : "warning",
          });
          await fetchPendingData();
        } else {
          // Handle cases where API returns success but action failed logically
          throw new Error(data?.result?.message || `Failed to ${actionVerb} requests.`);
        }
      } catch (error) {
        console.error(`Failed to ${actionVerb} requests!`, error);
        setSnackbar({
          open: true,
          message: error.response?.data?.message || `Failed to ${actionVerb} requests. Please try again.`,
          severity: "error",
        });
      } finally {
        setLoadOpen(false);
      }
    },
    [getSelectedRequests, fetchPendingData]
  );

  useEffect(() => {
    if (
      (user.user_role !== "1" && user.user_role !== "9") ||
      user.module !== "IAM"
    ) {
      navigate("/");
    }
    fetchPendingData();
  }, [user, navigate, fetchPendingData]);

  useEffect(() => {
    if (viewMode === "cancelledRejected") {
      fetchCancelledData();
    }
  }, [viewMode, fetchCancelledData]);

  return (
    <Box>
      <Box sx={{ mb: 2 }}>
        <Typography variant="h5" gutterBottom>
          Audit Requests
        </Typography>
        <Divider />
      </Box>

      {/* View Mode Toggle Buttons */}
      <Stack direction="row" spacing={1} sx={{ mb: 2 }}>
        <Button
          variant={viewMode === "pending" ? "contained" : "outlined"}
          onClick={() => setViewMode("pending")}
        >
          Pending Requests
        </Button>
        <Button
          variant={viewMode === "cancelledRejected" ? "contained" : "outlined"}
          onClick={() => setViewMode("cancelledRejected")}
        >
          Cancelled/Rejected Requests
        </Button>
      </Stack>

      {/* Action buttons are only visible in 'pending' mode */}
      {viewMode === "pending" && (
        <>
          {user.user_role === "9" ? (
            <Stack direction="row" spacing={2} sx={{ mb: 2 }}>
              <Button
                startIcon={<Done />}
                variant="contained"
                color="success"
                onClick={() => handleAction(selectionModel, "2")}
                disabled={selectionModel.length === 0}
              >
                Approve Selected
              </Button>
              <Button
                startIcon={<DoNotDisturb />}
                variant="contained"
                color="error"
                onClick={() => handleAction(selectionModel, "3")}
                disabled={selectionModel.length === 0}
              >
                Reject Selected
              </Button>
            </Stack>
          ) : user.user_role === "1" ? (
            <Stack direction="row" spacing={2} sx={{ mb: 2 }}>
              <Button
                startIcon={<Close />}
                variant="contained"
                color="warning"
                onClick={() => handleAction(selectionModel, "4")}
                disabled={selectionModel.length === 0}
              >
                Cancel Selected
              </Button>
            </Stack>
          ) : null}
        </>
      )}

      <Paper>
        {viewMode === "pending" ? (
          // DataGrid for Pending Requests (with checkboxes)
          <DataGrid
            rows={rows}
            columns={columns}
            getRowId={(row) => row.index}
            checkboxSelection
            disableRowSelectionOnClick
            onRowSelectionModelChange={(newSelection) => {
              setSelectionModel(newSelection);
            }}
            rowSelectionModel={selectionModel}
            localeText={{
              noRowsLabel: displayText,
            }}
            pageSizeOptions={[10, 25, 50]}
            sx={{
              "& .MuiDataGrid-row.Mui-selected": {
                backgroundColor: "#E6E6FA",
                color: "#4a4a4a",
                fontWeight: "Bold",
              },
              "& .MuiDataGrid-cell:focus": { outline: "none" },
              "& .MuiDataGrid-row:hover": { backgroundColor: "#E6E6FA" },
              "& .MuiDataGrid-columnHeader": {
                backgroundColor: "#826dfc",
                color: "#FFFFFF",
              },
              height: 620,
            }}
          />
        ) : (
          // DataGrid for Cancelled/Rejected Requests (no checkboxes)
          <DataGrid
            rows={cancelledRows}
            columns={cancelledColumns}
            getRowId={(row) => row.index}
            localeText={{
              noRowsLabel: cancelledDisplayText,
            }}
            pageSizeOptions={[10, 25, 50]}
            sx={{
              "& .MuiDataGrid-cell:focus": { outline: "none" },
              "& .MuiDataGrid-row:hover": { backgroundColor: "#f5f5f5" },
              "& .MuiDataGrid-columnHeader": {
                backgroundColor: "#826dfc",
                color: "#FFFFFF",
              },
              height: 620,
            }}
          />
        )}
      </Paper>

      <Snackbar
        open={snackbar.open}
        autoHideDuration={3000}
        onClose={handleCloseSnackbar}
        anchorOrigin={{
          vertical: "bottom",
          horizontal: "right",
        }}
      >
        <Alert
          variant="filled"
          onClose={handleCloseSnackbar}
          severity={snackbar.severity}
          sx={{ width: '100%' }}
        >
          {snackbar.message}
        </Alert>
      </Snackbar>

      <Dialog
        PaperProps={{
          style: { backgroundColor: "transparent", boxShadow: "none" },
        }}
        open={loadOpen}
      >
        <DialogContent sx={{ display: "Grid", placeItems: "center" }}>
          <CircularProgress />
        </DialogContent>
      </Dialog>
    </Box>
  );
}
