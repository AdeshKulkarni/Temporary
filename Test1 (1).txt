
class Solution {
    public int minSumOfLengths(int[] arr, int target) {
        int n = arr.length;
        
        // Map to store (runningSum, index). Initialize with the "ghost" anchor.
        HashMap<Integer, Integer> map = new HashMap<>();
        map.put(0, -1);
        
        // The "Notebook" to track the shortest subarray length found up to index i
        int[] best = new int[n];
        
        int currentBestLength = Integer.MAX_VALUE;
        int minSum = Integer.MAX_VALUE;
        int runningSum = 0;
        
        for (int i = 0; i < n; i++) {
            runningSum += arr[i];
            
            // If we found a valid subarray ending at i
            if (map.containsKey(runningSum - target)) {
                int pastIndex = map.get(runningSum - target);
                int currentLength = i - pastIndex;
                
                // 1. THE NON-OVERLAPPING CHECK
                // Look at the notebook strictly before our current subarray started (pastIndex).
                if (pastIndex > -1 && best[pastIndex] != Integer.MAX_VALUE) {
                    minSum = Math.min(minSum, currentLength + best[pastIndex]);
                }
                
                // 2. Update our running record for the shortest subarray seen so far
                currentBestLength = Math.min(currentBestLength, currentLength);
            }
            
            // 3. Write today's best record into the notebook for future iterations
            best[i] = currentBestLength;
            
            // 4. Always memorize the current running sum and its index
            map.put(runningSum, i);
        }
        
        return minSum == Integer.MAX_VALUE ? -1 : minSum;
    }
}
