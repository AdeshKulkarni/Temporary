package com.fincore.VoucherService;

import jakarta.annotation.PostConstruct;
import lombok.extern.slf4j.Slf4j; // CORRECTED LOGGER
import org.springframework.boot.SpringApplication;
import org.springframework.boot.autoconfigure.SpringBootApplication;
import org.springframework.cache.annotation.EnableCaching;
import org.springframework.scheduling.annotation.EnableScheduling;

import java.util.TimeZone;

@SpringBootApplication
@EnableCaching
@EnableScheduling
@Slf4j // ADDED THIS
public class VoucherServiceApplication {

    /**
     * This method runs immediately after Spring starts up.
     * It forces the entire JVM, including the Oracle JDBC Driver,
     * to operate strictly in Indian Standard Time (IST).
     */
    @PostConstruct
    public void init() {
        TimeZone.setDefault(TimeZone.getTimeZone("Asia/Kolkata"));
        // FIXED LOGGER SYNTAX (Removed the weird HBase TestChoreService import)
        log.info("SYSTEM_TIMEZONE_FORCED | Application running globally in: Asia/Kolkata"); 
    }

    public static void main(String[] args) {
        SpringApplication.run(VoucherServiceApplication.class, args);
    }
}
