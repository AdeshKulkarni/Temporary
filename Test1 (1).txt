package com.tcs.fincore.CommonRequestService.dto.payload;

import jakarta.validation.constraints.NotBlank;
import jakarta.validation.constraints.NotNull;
import lombok.Data;
import java.math.BigDecimal;
import java.time.LocalDate;

@Data
public class MocPayloadDto implements KeyablePayload {

    @NotNull(message = "Post date cannot be null")
    private LocalDate tmmocPdate;

    @NotBlank(message = "Branch cannot be blank")
    private String tmmocBranch;

    @NotBlank(message = "Currency cannot be blank")
    private String currency;

    @NotBlank(message = "CGL cannot be blank")
    private String tmmocCgl;

    @NotNull(message = "Amount cannot be null")
    private BigDecimal tmmocAmt;

    @NotBlank(message = "Transaction type cannot be blank")
    private String txnType;

    @NotBlank(message = "Product type cannot be blank")
    private String productType;

    private String remarks;

    // This method creates the composite business key used for duplicate checks.
    // The returned string will be stored in the TARGET_ID column.
    @Override
    public String getKey() {
        return tmmocBranch + "-" + currency + "-" + tmmocCgl;
    }
}
