class Solution {
    public int MaxElement(int[] nums , int low , int high) {
        int maxi = 0;

        for(int i=low; i<=high; i++) {
           
            if(nums[i] > maxi) {
                maxi = nums[i];
            }
        }

        return maxi;
    }


    public int maxFrequency(int[] nums, int k) {

        int low = 0;
        int n = nums.length;      
           Arrays.sort(nums);       

        int currentSum = 0;
        int ans = 1;

        for(int high=0; high<n; high++) {  
                 
            
            currentSum += nums[high];
            int maxiEle = MaxElement(nums , low , high);  
           
            int len = Math.abs(high - low) + 1;
            
            while(Math.abs(maxiEle - currentSum) > k) {
                if(len > ans) {
                    ans = len;
                }
            }

            if(Math.abs(maxiEle - currentSum) > k) {
                low++;
            }
                
        }

        return ans;
        
    }
}
class Solution {
    public int maxFrequency(int[] nums, int k) {
        Arrays.sort(nums);

        int low = 0;
        int n = nums.length;
        long currentSum = 0;
        int ans = 1;

        for(int high = 0; high < n; high++) {
            currentSum += nums[high];

            while(nums[high] * (high - low + 1) - currentSum > k) {
                currentSum -= nums[low];
                low++;
            }

            ans = Math.max(ans, high - low + 1);
        }

        return ans;
    }
}
