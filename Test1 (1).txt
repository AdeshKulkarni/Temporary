import React, { useEffect, useRef, useState } from "react";
import axios from "axios";

  Root,

  LeftCol,
  RightCol,

  SubHeadingText,
} from "./LoginStyles";
import { resolveConfig } from "../../../config/EnvironmentConfig";

import { loginSuccess } from "../../../store/slices/authSlice";
import { setMenus, setSelectedMenuItem } from "../../../store/slices/menuSlice";
import { setTabId } from "../../../store/slices/tabSlice";

export default function LoginPort({ appVersion }) {
  const [content, setContent] = useState(null);

  const dispatch = useDispatch();
  const navigate = useNavigate();
  const menus = useSelector((state) => state.menus.menus);

  const [userId, setUserId] = useState("");
  const [password, setPassword] = useState("");
  const [showPassword, setShowPassword] = useState(false);

  const [error, setError] = useState("");
  const [passwordError, setPasswordError] = useState("");

  const [page, setPage] = useState("HOME"); // HOME | UPDATE | OTP
  const [dialogHeading, setDialogHeading] = useState("");
  const [dialogContent, setDialogContent] = useState("");
  const [showErrorDialog, setShowErrorDialog] = useState(false);
  const [loading, setLoading] = useState(false);
  const [userIdDisabled, setUserIdDisabled] = useState(false);

  const [publicKey, setPublicKey] = useState(null);
  const [otpData, setOtpData] = useState(null);

  const userIdInputRef = useRef(null);
  const passwordInputRef = useRef(null);

  // Load static content JSON
  useEffect(() => {
    const loadContent = async () => {
      try {
        const res = await axios.get("/loginContent.json");
        setContent(res.data);
      } catch (err) {
        console.error("Failed to load content JSON:", err);
      }
    };
    loadContent();
  }, []);

  // Fetch Encryption Key On Load
  useEffect(() => {
    const fetchEncryptionKey = async () => {
      console.log("Fetching Encryption key");

      try {
        const userId = "1015699";
        const res = await axios.post(
          `${resolveConfig("/LS")}/auth/check-user`,
          {
            userId,
          },
        );
        const result = res?.data?.data || {};
        setPublicKey(result?.publicKey);
        // const res = await api.get(
        //   `${resolveConfig("/LS")}/auth/encryption-key`,
        // );

        // const key = res?.data?.data?.publicKey || res?.data?.publicKey;
        // if (key) setPublicKey(key);
      } catch (err) {
        console.error("Failed to fetch encryption key:", err);
      }
    };
    fetchEncryptionKey();
  }, []);

  // Focus User ID initially
  useEffect(() => {
    const timer = setTimeout(() => {
      if (userIdInputRef.current && page === "HOME") {
        userIdInputRef.current.focus();
      }
    }, 100);
    return () => clearTimeout(timer);
  }, [page]);

  // Post-authentication navigation handler
  useEffect(() => {
    if (menus && menus.length > 0 && menus[0]?.route) {
      if (menus[0].hasChildren) {
        navigate("/choose-option" + menus[0]?.route);
      } else {
        navigate(menus[0]?.route);
      }
    }
  }, [menus, navigate]);

  const handleDialogClose = () => {
    setShowErrorDialog(false);
    setDialogHeading("");
    setDialogContent("");
  };

  const openLoginErrorDialog = (heading, content) => {
    setDialogHeading(heading);
    setDialogContent(content);
    setShowErrorDialog(true);
  };

  const validateId = (id) => {
    if (!id) return false;
    const v = id.trim().toLowerCase();
    const regex = /^(?:\d{7}|v\d{7}|tcs\d{7}|tcsv\d{7})$/;
    return regex.test(v);
  };

  const handleUserIdChange = (raw) => {
    if (userIdDisabled) return;
    let val = (raw || "").replace(/\s+/g, "");
    const m = val.match(/^([A-Za-z]*)(.*)$/);
    let prefixCandidate = (m && m[1]) || "";
    let rest = (m && m[2]) || "";

    const prefixLow = prefixCandidate.toLowerCase();
    const allowedPrefixList = ["", "t", "tc", "tcs", "tcsv", "v"];
    if (!allowedPrefixList.some((p) => prefixLow?.startsWith(p))) {
      prefixCandidate = "";
    }

    const prefix = prefixCandidate;
    let digitsPart = rest.replace(/\D/g, "");
    if (digitsPart.length > 7) digitsPart = digitsPart.slice(0, 7);

    const newVal = prefix + digitsPart;
    if (
      digitsPart.length < 8 &&
      allowedPrefixList.some((p) => p.startsWith(prefix?.toLowerCase()))
    ) {
      setUserId(newVal);
    }
    if (error) setError("");
  };

  const handleUserIdPaste = (e) => {
    e.preventDefault();
    const paste = (e.clipboardData || window.clipboardData).getData("text");
    handleUserIdChange(paste);
  };

  // Unified Login & Validation Execution
  const handleLogin = async (e) => {
    if (e) e.preventDefault();

    setError("");
    setPasswordError("");

    if (!validateId(userId)) {
      setError("Invalid User ID");
      userIdInputRef.current?.focus();
      return;
    }

    if (!password || password.trim() === "") {
      setPasswordError("Password cannot be empty.");
      passwordInputRef.current?.focus();
      return;
    }

    if (!publicKey) {
      openLoginErrorDialog(
        "Security Loading",
        "Encryption key is still loading or failed. Please refresh the page.",
      );
      return;
    }

    try {
      setLoading(true);
      const payloadUserId = userId.trim();

      // Encrypt the password using the fetched key
      const encryptor = new JSEncrypt();
      encryptor.setPublicKey(publicKey);
      const encryptedPassword = encryptor.encrypt(password);

      if (!encryptedPassword) {
        throw new Error("Encryption failed.");
      }

      const res = await api.post(`${resolveConfig("/LS")}/auth/login`, {
        userId: payloadUserId,
        password: encryptedPassword,
      });

      const result = res?.data?.data || {};

      // 1. Check for legacy checkUser conditions included in login payload response
      if (result?.userStatus === "RESTRICTED") {
        openLoginErrorDialog(
          "Access Restricted",
          "This system is restricted to use FinCore, Please contact the administrator.",
        );
        return;
      }
      if (result?.userStatus === "INVALID") {
        setError(result?.message || "User ID does not exist");
        return;
      }
      if (result?.passwordLoginStatus === "INACTIVE") {
        openLoginErrorDialog(
          "Password Locked!",
          "You cannot login using password currently. Please contact the administrator.",
        );
        return;
      }
      if (result?.updatePassword === true) {
        setPage("UPDATE");
        setUserIdDisabled(true);
        return;
      }

      // 2. Validate Headers & Route to Success/OTP
      if (!result.payloadEncryptionKey) {
        result.payloadEncryptionKey = res.headers["x-payload-key"];
      }

      if (res.status === 202 && res?.data?.success) {
        setOtpData(result);
        setPage("OTP");
      } else if (res.status === 200 && res?.data?.success) {
        handleDirectSessionEstablish(result);
      } else {
        openLoginErrorDialog(
          "Login Failed",
          result.message || res?.data?.message || "Invalid credentials.",
        );
      }
    } catch (err) {
      console.error("Login error:", err);

      const errResult = err.response?.data?.data || {};

      if (errResult?.userStatus === "RESTRICTED") {
        openLoginErrorDialog(
          "Access Restricted",
          "This system is restricted to use FinCore, Please contact the administrator.",
        );
        return;
      }
      if (errResult?.passwordLoginStatus === "INACTIVE") {
        openLoginErrorDialog(
          "Password Locked!",
          "You cannot login using password currently. Please contact the administrator.",
        );
        return;
      }

      const msg =
        err.response?.data?.message ||
        err.message ||
        "Something went wrong, please try again later.";
      openLoginErrorDialog("Login Failed", msg);
    } finally {
      setLoading(false);
    }
  };

  const handleKeyDown = (e) => {
    if (e.key === "Enter") {
      e.preventDefault();
      if (!loading) handleLogin();
    }
  };

  const handlePasswordUpdateSuccess = () => {
    setPage("HOME");
    setUserIdDisabled(false);
    setPassword(""); // Clear password so they log in fresh
  };

  const enableEditingBackToHome = () => {
    setPage("HOME");
    setUserIdDisabled(false);
    setError("");
    setPasswordError("");
    setPassword("");
    setOtpData(null);
  };

  // Shared token payload & redux session hydration
  const handleDirectSessionEstablish = (result) => {
    if (result?.roleData?.root_menus?.length === 0) {
      openLoginErrorDialog(
        "Access Denied",
        "You don't have enough permissions to access this application.",
      );
      return;
    }

    const user = parseJwt(result.accessToken);
    const tabId = crypto.randomUUID();

    dispatch(loginSuccess({ token: result.accessToken, user }));
    localStorage.setItem("payloadEncryptionKey", result.payloadEncryptionKey);

    dispatch(setTabId(tabId));
    dispatch(setMenus(result.roleData.root_menus));
    dispatch(setSelectedMenuItem(result.roleData.root_menus[0]));

    const bc = new BroadcastChannel("auth");
    setTimeout(() => {
      bc.postMessage({
        type: "USER_LOGGED_IN",
        tabId,
        userId: user.userId,
      });
    }, 1000);
  };

  if (!content) return null;

  return (
    <Root sx={{ justifyContent: "space-between", padding: 0 }}>
      <Box
        sx={{
          flexGrow: 1,
          display: "flex",
          alignItems: "center",
          width: "100%",
          py: { xs: 4, md: 6 },
          px: { xs: 2.5, sm: 4, md: 6 },
        }}
      >
        <ContainerInner>
          <Content
            sx={{
              display: "flex",
              flexDirection: { xs: "column", md: "row" },
              gap: { xs: 5, md: 4 },
            }}
          >
            <LeftCol>
              <LogoWrap>
                <Box
                  component="img"
                  src={LogoImg}
                  alt="Logo Icon"
                  sx={{
                    width: { xs: 100, sm: 130, md: 150 },
                    height: "auto",
                  }}
                />
                <Box sx={{ textAlign: { xs: "center", md: "left" } }}>
                  <Box
                    component="img"
                    src={fTitle}
                    alt="FinCore"
                    sx={{
                      height: { xs: 28, md: 35 },
                      width: "auto",
                      mx: { xs: "auto", md: 0 },
                      mt: { xs: 2, md: 5 },
                      mb: 1,
                    }}
                  />
                  <BrandSub>{content?.brand?.subtitle}</BrandSub>
                </Box>
              </LogoWrap>

              <Box
                sx={{
                  display: { xs: "none", md: "block" },
                  mt: 2,
                  mb: 1,
                  textAlign: { xs: "center", md: "left" },
                }}
              >
                <BigHeading>
                  {(content?.hero?.welcomeTitle || "").split(" - ")[0]}
                </BigHeading>
                {content?.hero?.welcomeTitle?.includes(" - ") && (
                  <SubHeadingText>
                    {content?.hero?.welcomeTitle.split(" - ")[1]}
                  </SubHeadingText>
                )}
              </Box>

              <SubText
                sx={{
                  display: { xs: "none", md: "block" },
                  textAlign: { xs: "center", md: "left" },
                }}
              >
                {content?.hero?.subText}
              </SubText>

              <Box sx={{ display: { xs: "none", md: "block" }, mt: 3 }}>
                <BigHeading>{content?.legal?.title}</BigHeading>
                <FeatureList sx={{ mt: 2 }}>
                  {content?.legal?.items.map((item, idx) => (
                    <FeatureItem key={idx}>
                      <Bullet /> {item}
                    </FeatureItem>
                  ))}
                </FeatureList>
              </Box>
            </LeftCol>

            <RightCol
              sx={{
                display: "flex",
                flexDirection: "column",
                alignItems: "center",
                width: "100%",
              }}
            >
              <GlassCard elevation={3} sx={{ width: "100%" }}>
                <FormTitle align="center">
                  {page === "OTP"
                    ? "Two-Factor Verification"
                    : content?.form?.signIn}
                </FormTitle>

                {page === "HOME" && (
                  <Box>
                    <StyledTextField
                      fullWidth
                      required
                      inputRef={userIdInputRef}
                      label="User ID"
                      placeholder={content?.form?.userIdPlaceholder}
                      margin="dense"
                      value={userId}
                      onChange={(e) => handleUserIdChange(e.target.value)}
                      onPaste={handleUserIdPaste}
                      onKeyDown={handleKeyDown}
                      error={Boolean(error)}
                      helperText={error || " "}
                      disabled={userIdDisabled || loading}
                      InputProps={{
                        startAdornment: (
                          <InputAdornment position="start">
                            <PersonIcon />
                          </InputAdornment>
                        ),
                      }}
                    />

                    <StyledTextField
                      fullWidth
                      required
                      inputRef={passwordInputRef}
                      label="Password"
                      placeholder="Enter your Password"
                      type={showPassword ? "text" : "password"}
                      margin="dense"
                      value={password}
                      onChange={(e) => {
                        setPassword(e.target.value);
                        if (passwordError) setPasswordError("");
                      }}
                      onKeyDown={handleKeyDown}
                      error={Boolean(passwordError)}
                      helperText={passwordError || " "}
                      disabled={loading}
                      autoComplete="current-password"
                      InputProps={{
                        startAdornment: (
                          <InputAdornment position="start">
                            <LockIcon />
                          </InputAdornment>
                        ),
                        endAdornment: password.trim() && (
                          <InputAdornment position="end">
                            <IconButton
                              onClick={() => setShowPassword(!showPassword)}
                              edge="end"
                              disabled={loading}
                            >
                              {showPassword ? (
                                <Visibility />
                              ) : (
                                <VisibilityOff />
                              )}
                            </IconButton>
                          </InputAdornment>
                        ),
                      }}
                    />

                    <PrimaryButton
                      type="button"
                      variant="contained"
                      onClick={handleLogin}
                      disabled={
                        loading || !userId || password.trim().length < 8
                      }
                    >
                      {loading
                        ? content?.form?.checking || "Verifying..."
                        : content?.form?.signInButton}
                    </PrimaryButton>
                  </Box>
                )}

                {page === "OTP" && (
                  <Box>
                    <LoginWithOTP
                      userId={userId}
                      otpData={otpData}
                      openLoginErrorDialog={openLoginErrorDialog}
                      onBackToHome={enableEditingBackToHome}
                      onSessionEstablished={handleDirectSessionEstablish}
                    />
                  </Box>
                )}

                {page === "UPDATE" && (
                  <Box>
                    <UpdatePassword
                      userId={userId}
                      publicKey={publicKey}
                      openLoginErrorDialog={openLoginErrorDialog}
                      onUpdateSuccess={handlePasswordUpdateSuccess}
                    />
                  </Box>
                )}
              </GlassCard>

              <Box
                sx={{
                  display: { xs: "block", md: "none" },
                  mt: 5,
                  width: "100%",
                }}
              >
                <BigHeading
                  sx={{
                    fontSize: { xs: "1.2rem", md: "1.5rem" },
                    textAlign: "center",
                    mb: 2,
                  }}
                >
                  {content?.legal?.title}
                </BigHeading>
                <FeatureList>
                  {content?.legal?.items.map((item, idx) => (
                    <FeatureItem
                      key={idx}
                      sx={{
                        fontSize: { xs: "0.875rem", sm: "1rem" },
                      }}
                    >
                      <Bullet /> <span style={{ flex: 1 }}>{item}</span>
                    </FeatureItem>
                  ))}
                </FeatureList>
              </Box>
            </RightCol>
          </Content>
        </ContainerInner>
      </Box>

      <Box sx={{ width: "100%", zIndex: 2 }}>
        <Copyright appVersion={appVersion} />
      </Box>

      <LoginErrorDialog
        showErrorDialog={showErrorDialog}
        title={dialogHeading}
        content={dialogContent}
        onDialogClose={handleDialogClose}
      />
    </Root>
  );
}

export const RequirementText = styled(Typography)(({ theme }) => ({
  fontSize: "0.85rem",
  color: theme.palette.text.secondary,

import React, { useState, useEffect, useRef } from "react";
import {
  Typography,
  Button,
  Box,
  InputBase,
  Stack,
  CircularProgress,
} from "@mui/material";
import ArrowBackIcon from "@mui/icons-material/ArrowBack";
import RefreshIcon from "@mui/icons-material/Refresh";
import EmailIcon from "@mui/icons-material/Email";
import PhoneAndroidIcon from "@mui/icons-material/PhoneAndroid";
import AccessTimeIcon from "@mui/icons-material/AccessTime";
import WarningAmberIcon from "@mui/icons-material/WarningAmber";
import api from "../../../config/axiosConfig";
import { resolveConfig } from "../../../config/EnvironmentConfig";
import { PrimaryButton } from "./LoginStyles";
import useCustomSnackbar from "../../../utils/useCustomSnackbar";

const LoginWithOTP = ({
  userId,
  otpData,
  openLoginErrorDialog,
  onBackToHome,
  onSessionEstablished,
}) => {
  const showSnackbar = useCustomSnackbar();

  const [otpValues, setOtpValues] = useState(["", "", "", "", "", ""]);
  const [isLoading, setIsLoading] = useState(false);
  const [wrongAttempts, setWrongAttempts] = useState(0);
  const [isSessionUnauthorized, setIsSessionUnauthorized] = useState(false);

  // --- API Driven Timers & Limits ---
  const [currentTxnId, setCurrentTxnId] = useState(
    otpData?.transactionId || "",
  );
  const [expiresIn, setExpiresIn] = useState(otpData?.expiresInSeconds || 300);
  const [cooldown, setCooldown] = useState(
    otpData?.resendCooldownSeconds || 30,
  );
  const [resendsLeft, setResendsLeft] = useState(
    otpData?.resendsRemaining ?? 5,
  );

  const inputRefs = useRef([]);

  const isExpired = expiresIn <= 0 || isSessionUnauthorized;
  const isExhausted = resendsLeft <= 0;
  const otpString = otpValues.join("");

  // Focus management
  useEffect(() => {
    if (!isExpired && inputRefs.current[0]) {
      inputRefs.current[0].focus();
    }
  }, [isExpired]);

  // Global Tick Timer
  useEffect(() => {
    const timer = setInterval(() => {
      setExpiresIn((prev) => (prev > 0 ? prev - 1 : 0));
      setCooldown((prev) => (prev > 0 ? prev - 1 : 0));
    }, 1000);
    return () => clearInterval(timer);
  }, []);

  const formatTime = (seconds) => {
    const m = Math.floor(seconds / 60)
      .toString()
      .padStart(2, "0");
    const s = (seconds % 60).toString().padStart(2, "0");
    return `${m}:${s}`;
  };

  const updateTimersFromResponse = (data) => {
    if (data?.transactionId) setCurrentTxnId(data.transactionId);
    if (data?.expiresInSeconds) setExpiresIn(data.expiresInSeconds);
    if (data?.resendCooldownSeconds) setCooldown(data.resendCooldownSeconds);
    if (data?.resendsRemaining !== undefined)
      setResendsLeft(data.resendsRemaining);
  };

  const forceResendOTP = async () => {
    try {
      const res = await api.post(
        `${resolveConfig("/LS") || ""}/auth/resend-otp`,
        {
          userId,
          transactionId: currentTxnId,
        },
      );

      updateTimersFromResponse(res?.data?.data);
      setWrongAttempts(0);
      setOtpValues(["", "", "", "", "", ""]);
      showSnackbar(
        "3 invalid attempts. A fresh verification code was sent.",
        "warning",
      );
      inputRefs.current[0]?.focus();
    } catch (err) {
      console.error("Forced OTP Generation Error:", err);
      openLoginErrorDialog(
        "Session Terminated",
        "Could not generate a new code. Please restart login.",
      );
    }
  };

  const handleResendClick = async () => {
    if (isExhausted || cooldown > 0 || isExpired) return;

    setIsLoading(true);
    try {
      const res = await api.post(
        `${resolveConfig("/LS") || ""}/auth/resend-otp`,
        {
          userId,
          transactionId: currentTxnId,
        },
      );

      updateTimersFromResponse(res?.data?.data);
      setWrongAttempts(0);
      setOtpValues(["", "", "", "", "", ""]);
      showSnackbar(
        res?.data?.message || "A new code has been sent successfully.",
        "success",
      );
      inputRefs.current[0]?.focus();
    } catch (err) {
      const status = err.response?.status;
      if (status === 401) {
        setIsSessionUnauthorized(true);
        setExpiresIn(0);
      } else {
        const msg =
          err.response?.data?.message || "Failed to resend verification code.";
        showSnackbar(msg, "error");
      }
    } finally {
      setIsLoading(false);
    }
  };

  const handleVerifyOTP = async () => {
    if (otpString.length !== 6 || isExpired) return;

    setIsLoading(true);

    try {
      const response = await api.post(
        `${resolveConfig("/LS") || ""}/auth/verify-otp`,
        {
          userId,
          transactionId: currentTxnId,
          otp: otpString,
        },
      );

      const isSuccess = response?.data?.success === true;
      const result = response?.data?.data || {};

      if (isSuccess) {
        if (onSessionEstablished) {
          onSessionEstablished(result);
        }
      } else {
        const targetFailures = wrongAttempts + 1;
        setWrongAttempts(targetFailures);

        if (targetFailures >= 3) {
          await forceResendOTP();
        } else {
          showSnackbar(
            response?.data?.message ||
              `Invalid code. Attempt ${targetFailures} of 3.`,
            "error",
          );
        }
      }
    } catch (e) {
      const status = e.response?.status;

      // 401: Unauthorized / Expired -> Render expired state
      if (status === 401) {
        setIsSessionUnauthorized(true);
        setExpiresIn(0);
      }
      // 400: Invalid OTP
      else if (status === 400) {
        const targetFailures = wrongAttempts + 1;
        setWrongAttempts(targetFailures);

        if (targetFailures >= 3) {
          await forceResendOTP();
        } else {
          showSnackbar(
            e.response?.data?.message ||
              `Invalid OTP. Attempt ${targetFailures} of 3.`,
            "error",
          );
        }
      } else {
        console.error("OTP Verification Exception:", e);
        const msg =
          e.response?.data?.message ||
          "Verification request failed. Try again.";
        showSnackbar(msg, "error");
      }
    } finally {
      setIsLoading(false);
    }
  };

  const handleKeyDown = (index, e) => {
    if (e.key === "Backspace") {
      e.preventDefault();
      const updated = [...otpValues];
      if (updated[index]) {
        updated[index] = "";
        setOtpValues(updated);
      } else if (index > 0) {
        updated[index - 1] = "";
        setOtpValues(updated);
        inputRefs.current[index - 1]?.focus();
      }
    } else if (e.key === "ArrowLeft" && index > 0) {
      e.preventDefault();
      inputRefs.current[index - 1]?.focus();
    } else if (e.key === "ArrowRight" && index < 5) {
      e.preventDefault();
      inputRefs.current[index + 1]?.focus();
    } else if (e.key === "Enter" && otpString.length === 6 && !isLoading) {
      e.preventDefault();
      handleVerifyOTP();
    }
  };

  const handleChange = (index, e) => {
    const rawVal = e.target.value;
    const digitsOnly = rawVal.replace(/\D/g, "");

    if (!digitsOnly) {
      const updated = [...otpValues];
      updated[index] = "";
      setOtpValues(updated);
      return;
    }

    const updated = [...otpValues];

    if (digitsOnly.length === 1) {
      updated[index] = digitsOnly;
      setOtpValues(updated);
      if (index < 5) {
        inputRefs.current[index + 1]?.focus();
      }
    } else {
      const chars = digitsOnly.split("").slice(0, 6);
      chars.forEach((char, idx) => {
        if (index + idx < 6) {
          updated[index + idx] = char;
        }
      });
      setOtpValues(updated);
      const nextTarget = Math.min(index + chars.length, 5);
      inputRefs.current[nextTarget]?.focus();
    }
  };

  const handlePaste = (e) => {
    e.preventDefault();
    const pasted = e.clipboardData
      .getData("text/plain")
      .replace(/\D/g, "")
      .slice(0, 6);
    if (!pasted) return;

    const updated = [...otpValues];
    for (let i = 0; i < pasted.length; i++) {
      updated[i] = pasted[i];
    }
    setOtpValues(updated);

    const targetIdx = pasted.length < 6 ? pasted.length : 5;
    inputRefs.current[targetIdx]?.focus();
  };

  return (
    <Box sx={{ width: "100%", maxWidth: "420px", mx: "auto" }}>
      {/* Information Header */}
      <Box
        sx={{
          mb: 3,
          p: { xs: 1.5, sm: 2 },
          borderRadius: 2,
          bgcolor: "rgba(0,0,0,0.03)",
          border: "1px solid rgba(0,0,0,0.06)",
        }}
      >
        <Box
          sx={{
            display: "flex",
            justifyContent: "space-between",
            alignItems: "center",
            mb: 1.5,
          }}
        >
          <Typography
            variant="body2"
            color="text.secondary"
            sx={{ fontWeight: 500, fontSize: { xs: "0.8rem", sm: "0.875rem" } }}
          >
            {isExpired ? "Status" : "Security Code Sent To"}
          </Typography>

          <Box
            sx={{
              display: "inline-flex",
              alignItems: "center",
              justifyContent: "center",
              gap: 0.5,
              width: "76px",
              py: 0.3,
              px: 0.8,
              borderRadius: 1.5,
              bgcolor: isExpired
                ? "error.50"
                : expiresIn > 60
                  ? "primary.50"
                  : "warning.50",
              color: isExpired
                ? "error.main"
                : expiresIn > 60
                  ? "primary.main"
                  : "error.main",
              border: "1px solid",
              borderColor: isExpired
                ? "error.100"
                : expiresIn > 60
                  ? "primary.100"
                  : "warning.100",
            }}
          >
            <AccessTimeIcon sx={{ fontSize: 14 }} />
            <Typography
              component="span"
              sx={{
                fontFamily: "monospace",
                fontWeight: 700,
                fontSize: "0.75rem",
                width: "36px",
                textAlign: "center",
              }}
            >
              {formatTime(expiresIn)}
            </Typography>
          </Box>
        </Box>

        {!isExpired && (
          <Stack spacing={0.75}>
            {otpData?.maskedMobile && (
              <Box sx={{ display: "flex", alignItems: "center", gap: 1.25 }}>
                <PhoneAndroidIcon
                  sx={{ fontSize: 18, color: "text.secondary" }}
                />
                <Typography
                  variant="subtitle2"
                  color="text.primary"
                  sx={{ fontSize: { xs: "0.8rem", sm: "0.875rem" } }}
                >
                  {otpData.maskedMobile}
                </Typography>
              </Box>
            )}

            {otpData?.maskedEmail && (
              <Box sx={{ display: "flex", alignItems: "center", gap: 1.25 }}>
                <EmailIcon sx={{ fontSize: 18, color: "text.secondary" }} />
                <Typography
                  variant="subtitle2"
                  color="text.primary"
                  sx={{
                    fontSize: { xs: "0.8rem", sm: "0.875rem" },
                    wordBreak: "break-all",
                  }}
                >
                  {otpData.maskedEmail}
                </Typography>
              </Box>
            )}
          </Stack>
        )}

        {isExpired && (
          <Typography
            variant="body2"
            color="error.main"
            sx={{
              fontWeight: 600,
              fontSize: { xs: "0.825rem", sm: "0.875rem" },
            }}
          >
            {isSessionUnauthorized
              ? "Your validation session has expired."
              : "Your verification token has timed out."}
          </Typography>
        )}
      </Box>

      {/* STATE 1: Active Entry */}
      {!isExpired ? (
        <>
          <Box
            sx={{
              display: "grid",
              gridTemplateColumns: "repeat(6, 1fr)",
              gap: { xs: 0.75, sm: 1.25 },
              mb: 3,
            }}
          >
            {otpValues.map((val, idx) => (
              <Box
                key={idx}
                sx={{
                  display: "flex",
                  alignItems: "center",
                  justifyContent: "center",
                  borderRadius: 2,
                  bgcolor: "background.paper",
                  border: "1.5px solid",
                  borderColor: val ? "primary.main" : "rgba(0, 0, 0, 0.15)",
                  boxShadow: val
                    ? "0 0 0 3px rgba(25, 118, 210, 0.12)"
                    : "none",
                  transition: "all 0.15s ease-in-out",
                  height: { xs: 48, sm: 54, md: 58 },
                  "&:focus-within": {
                    borderColor: "primary.main",
                    boxShadow: "0 0 0 3px rgba(25, 118, 210, 0.2)",
                  },
                }}
              >
                <InputBase
                  inputRef={(el) => (inputRefs.current[idx] = el)}
                  value={val}
                  onChange={(e) => handleChange(idx, e)}
                  onKeyDown={(e) => handleKeyDown(idx, e)}
                  onPaste={handlePaste}
                  disabled={isLoading}
                  inputProps={{
                    maxLength: 1,
                    inputMode: "numeric",
                    pattern: "[0-9]*",
                    style: {
                      textAlign: "center",
                      fontSize: "1.25rem",
                      fontWeight: 700,
                      padding: 0,
                      width: "100%",
                      height: "100%",
                    },
                  }}
                />
              </Box>
            ))}
          </Box>

          <PrimaryButton
            fullWidth
            variant="contained"
            onClick={handleVerifyOTP}
            disabled={isLoading || otpString.length !== 6}
            sx={{ py: 1.25, fontSize: "0.95rem" }}
          >
            {isLoading ? (
              <CircularProgress size={22} color="inherit" />
            ) : (
              "Verify OTP & Sign In"
            )}
          </PrimaryButton>

          <Box
            sx={{
              display: "flex",
              justifyContent: "space-between",
              alignItems: "center",
              mt: 2.5,
            }}
          >
            <Button
              startIcon={<ArrowBackIcon />}
              onClick={onBackToHome}
              sx={{
                textTransform: "none",
                color: "text.secondary",
                fontSize: { xs: "0.75rem", sm: "0.85rem" },
                p: 0.5,
              }}
              size="small"
            >
              Back To Sign In
            </Button>

            {!isExhausted && (
              <Button
                onClick={handleResendClick}
                disabled={cooldown > 0 || isLoading}
                loading={isLoading}
                startIcon={<RefreshIcon />}
                size="small"
                sx={{
                  fontWeight: 600,
                  px: 1.5, // Consistent padding
                  py: 0.5,
                  minWidth: 100, // Prevent layout shift
                }}
              >
                {cooldown ? `Resend in ${cooldown}s` : "Resend Code"}
              </Button>
            )}
          </Box>
        </>
      ) : (
        /* STATE 2: Expired Clean View */
        <Box
          sx={{
            display: "flex",
            flexDirection: "column",
            alignItems: "center",
            textAlign: "center",
            py: 2,
            px: 1,
          }}
        >
          <Box
            sx={{
              width: 56,
              height: 56,
              borderRadius: "50%",
              bgcolor: "error.50",
              display: "flex",
              alignItems: "center",
              justifyContent: "center",
              color: "error.main",
              mb: 2,
            }}
          >
            <WarningAmberIcon sx={{ fontSize: 32 }} />
          </Box>

          <Typography variant="h6" fontWeight={700} gutterBottom>
            Session Expired
          </Typography>

          <Typography
            variant="body2"
            color="text.secondary"
            sx={{ mb: 3, maxWidth: "320px", lineHeight: 1.5 }}
          >
            {isSessionUnauthorized
              ? "Your authentication session is no longer active. Please return to the login screen."
              : "The OTP validity period has expired for security reasons. Please return to the login page to authenticate again."}
          </Typography>

          <PrimaryButton
            fullWidth
            variant="contained"
            startIcon={<ArrowBackIcon />}
            onClick={onBackToHome}
            sx={{ py: 1.25 }}
          >
            Back to User ID
          </PrimaryButton>
        </Box>
      )}
    </Box>
  );
};

export default LoginWithOTP;

const UpdatePassword = ({
  userId,
  publicKey,
  openLoginErrorDialog,
  onUpdateSuccess,
}) => {
  const [showPassword, setShowPassword] = useState(false);
  const [password, setPassword] = useState("");
  const [content, setContent] = useState(null);
  const [changingInput, setChangingInput] = useState(false);
  const [requirementsMet, setRequirementsMet] = useState({
    length: false,
    uppercase: false,
    lowercase: false,
    number: false,
    specialChar: false,
    // notPrevious: false,
  });

  useEffect(() => {
    const loadContent = async () => {
      try {
        const res = await axios.get("/loginContent.json");
        setContent(res.data);
      } catch (err) {
        console.error("Failed to load content JSON:", err);
      }
    };
    loadContent();
  }, []);
  const validatePassword = (password) => {
    if (password !== "") {
      const length = password.length >= 8 && password.length <= 16;
      const uppercase = /[A-Z]/.test(password);
      const lowercase = /[a-z]/.test(password);
      const number = /[0-9]/.test(password);
      const specialChar = /[^A-Za-z0-9]/.test(password);
      // const notPrevious = !prev.includes(hashedPass);
      setRequirementsMet({
        length,
        uppercase,
        lowercase,
        number,
        specialChar,
        //notPrevious,
      });
    }
  };
  const RequirementItem = ({ valid, typing, children }) => (
    <Box>
      <RequirementRow>
        <IconButton size="small" disableRipple>
          {typing ? (
            valid ? (
              <CheckCircleIcon color={"success"} />
            ) : (
              <CancelIcon color={"error"} />
            )
          ) : (
            <RadioButtonCheckedIcon color="linear-gradient(to right,#0f172a,#1e3a8a,#1e293b)" />
          )}
        </IconButton>
        <RequirementText>
          <Typography>{children}</Typography>
        </RequirementText>
      </RequirementRow>
    </Box>
  );
  const passwordValid = Object.values(requirementsMet).every(Boolean);
  const handlePassword = (e) => {
    let password = e.target.value;
    setPassword(e.target.value);
    if (password !== "") {
      setChangingInput(true);
      validatePassword(password);
    } else {
      setChangingInput(false);
    }
  };
  const updatePassword = async () => {
    // 1. Fetching the Public Key from the backend
    if (!publicKey) {
      throw new Error("Could not retrieve encryption key.");
    }

    // 2. Encrypting the password
    const encryptor = new JSEncrypt();
    encryptor.setPublicKey(publicKey);
    const encryptedPassword = encryptor.encrypt(password);

    if (!encryptedPassword) {
      throw new Error("Encryption failed.");
    }

    const payload = {
      userId,
      password: encryptedPassword,
    };
    try {
      const response = await axios.post(
        `${resolveConfig("/LS") ?? ""}/auth/update-password`,
        payload,
      );

      if (response?.data?.data) {
        if (response.data?.data?.updateFlag === true) {
          openLoginErrorDialog(
            "Updated Successfully",
            "Your password has been updated.Please try logging in now.",
          );
          setTimeout(() => {
            onUpdateSuccess();
          }, 1500);
        }
      } else {
        console.error("Failed to update password");
        openLoginErrorDialog(
          "Update Failed",
          "Failed to update your password. Please try again later",
        );
      }
    } catch (e) {
      openLoginErrorDialog(
        "Update Failed",
        "Failed to update your password. Please try again later",
      );
    }
  };
  const toggleShowPassword = () => setShowPassword((s) => !s);

  return (
    <>
      <FormTitle>{content?.form?.updatePassword?.title}</FormTitle>
      <StyledTextField
        fullWidth
        label={content?.form?.updatePassword?.newPasswordLable}
        type={showPassword ? "text" : "password"}
        placeholder={content?.form?.updatePassword?.newPasswordPlaceholder}
        value={password}
        onChange={handlePassword}
        variant="outlined"
        margin="normal"
        InputProps={{
          startAdornment: (
            <InputAdornment position="start">
              <LockIcon />
            </InputAdornment>
          ),
          endAdornment: (
            <InputAdornment position="end">
              <IconButton
                aria-label={showPassword ? "Hide password" : "Show password"}
                onClick={toggleShowPassword}
                edge="end"
              >
                {showPassword ? <Visibility /> : <VisibilityOff />}
              </IconButton>
            </InputAdornment>
          ),
        }}
      />
      <PrimaryButton
        fullWidth
        variant="contained"
        onClick={() => {
          updatePassword();
        }}
        disabled={!passwordValid}
      >
        {content?.form?.updatePassword?.updateButton}
      </PrimaryButton>

      <Box>
        <Box mt={2}>
          {(content?.form?.updatePassword?.requirements || []).map((r) => (
            <RequirementItem
              key={r?.key}
              valid={requirementsMet[r?.key]}
              typing={changingInput}
            >
              {r.text}
            </RequirementItem>
          ))}
        </Box>
      </Box>
    </>
  );
};

const Login = ({ appVersion }) => {
  const dispatch = useDispatch();

  useEffect(() => {
    const handleLogout = async () => {
      dispatch(clearMenus());
      dispatch(logout());
      try {
        //This clears redux persists local or session storage
        await persistor.purge();
      } catch (error) {
        console.error("Error while purging persisted store:", error);
      }

      // sessionStorage.clear();
      // localStorage.clear();
    };
    handleLogout();
  }, [dispatch]);

  return <LoginPort appVersion={appVersion} />;
};
