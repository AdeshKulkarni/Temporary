package com.fincore.JournalService.Models;

import jakarta.persistence.*;
import lombok.Data;
import lombok.NoArgsConstructor;
import lombok.AllArgsConstructor;
import java.time.LocalDateTime;

@Entity
@Table(name = "JOURNAL_ACTIVITY_LOGS") // Ensure this matches your DB table name
@Data
@NoArgsConstructor
@AllArgsConstructor
public class JournalLog {

    @Id
    // --- FIX: Added GeneratedValue so Hibernate handles the ID creation ---
    @GeneratedValue(strategy = GenerationType.SEQUENCE, generator = "JOURNAL_LOG_SEQ_GEN")
    @SequenceGenerator(name = "JOURNAL_LOG_SEQ_GEN", sequenceName = "JOURNAL_LOG_SEQ", allocationSize = 1)
    @Column(name = "LOG_ID")
    private Long id;
    // ---------------------------------------------------------------------

    @Column(name = "USER_ID")
    private String userId;

    @Column(name = "ACTION_TYPE")
    private String actionType; // CREATE, UPDATE, DELETE

    @Column(name = "CHANGE_TYPE")
    private String changeType; // BATCH_CREATION, STATUS_UPDATE

    @Column(name = "ACTION_TIME")
    private LocalDateTime actionTime;

    @Column(name = "REQUEST_ID")
    private Long requestId; // Can be null for Batch operations

    @Column(name = "IP_ADDRESS")
    private String ipAddress;

    @Lob // Use @Lob for potentially large text fields
    @Column(name = "NEW_VALUE", length = 4000)
    private String newValue;

    @Lob
    @Column(name = "OLD_VALUE", length = 4000)
    private String oldValue;
}
