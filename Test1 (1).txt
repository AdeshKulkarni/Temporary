const handleEditorDrop = (e) => {
  e.preventDefault();
  const text = e.dataTransfer.getData("text");
  if (!text) return;
  const quill = bodyEditorRef.current?.getEditor();
  if (!quill) return;
  quill.focus();
  quill.history.cutoff(); // End previous undo stack, start new group
  const range = quill.getSelection(true);
  quill.insertEmbed(range.index, "variable", text, Quill.sources.USER);
  quill.setSelection(range.index + 1, Quill.sources.SILENT);
  quill.history.cutoff(); // End this op as a single undo step
};
