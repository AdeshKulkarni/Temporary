class Solution {
     public boolean Valid(int i, int j, int n, int m) {
        if (i < 0 || i >= n || j < 0 || j >= m) {
            return false;
        }

        return true;
    }

    public int dfs(int[][] grid , int i, int j, int m, int n, boolean[][] visited) {

        visited[i][j] = true;
        int count = 1;
    
        int[] x = {-1, 1, 0, 0};
        int[] y = {0, 0, -1, 1};

        for (int k = 0; k < 4; k++) {

            int row = i + x[k];
            int col = j + y[k];

            if (Valid(row, col, n, m)
                    && visited[row][col] == false
                    && grid[row][col] == 1) {

               count += dfs(grid, row, col, m, n, visited);
            }
        }
        return count;
    }
    public int numEnclaves(int[][] grid) {
        
        int n = grid.length;
        int m = grid[0].length;
        int res = 0;

        boolean[][] visited = new boolean[n][m];

         // Start DFS from top and bottom boundaries
        for (int j = 0; j < m; j++) {

            if (grid[0][j] == 1 && visited[0][j] == false) {
                res += dfs(grid, 0, j, m, n, visited);
            }

            if (grid[n - 1][j] == 1 && visited[n - 1][j] == false) {
                 res += dfs(grid, n - 1, j, m, n, visited);
            }
        }

        // Start DFS from left and right boundaries
        for (int i = 0; i < n; i++) {

            if (grid[i][0] == 1 && visited[i][0] == false) {
                 res += dfs(grid, i, 0, m, n, visited);
            }

            if (grid[i][m - 1] == 1 && visited[i][m - 1] == false) {
                 res += dfs(grid, i, m - 1, m, n, visited);
            }
        }
        return res;
    }
}
