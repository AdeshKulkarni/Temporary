package com.tcs.fincore.CommonRequestService.controller;

import com.tcs.fincore.CommonRequestService.dto.CglDto;
import com.tcs.fincore.CommonRequestService.service.CglService;
import lombok.RequiredArgsConstructor;
import org.springframework.web.bind.annotation.GetMapping;
import org.springframework.web.bind.annotation.RequestMapping;
import org.springframework.web.bind.annotation.RequestParam;
import org.springframework.web.bind.annotation.RestController;

import java.util.List;
import java.util.Map;

@RestController
@RequestMapping("/api/cgls")
@RequiredArgsConstructor
public class CglController {

    private final CglService cglService;

    /**
     * GET /api/cgls/classifications
     * Returns all possible account classifications for a UI dropdown.
     * @return A map e.g., { "A": "Asset", "L": "Liability", ... }
     */
    @GetMapping("/classifications")
    public Map<Character, String> getAccountClassifications() {
        return cglService.getAccountClassifications();
    }

    /**
     * GET /api/cgls/filter?classification=A
     * Filters the CGL list by the provided account classification code.
     * @param classification The code ('A', 'L', 'I', 'E', or 'M').
     * @return A list of CGLs matching the classification.
     */
    @GetMapping("/filter")
    public List<CglDto> getCglsByClassification(@RequestParam String classification) {
        return cglService.findCglsByClassification(classification);
    }

    /**
     * GET /api/cgls/search?query=1234
     * Searches for CGLs where the CGL number contains the query string.
     * @param query The partial or full CGL number.
     * @return A list of matching CGLs.
     */
    @GetMapping("/search")
    public List<CglDto> searchCgls(@RequestParam String query) {
        return cglService.searchCglsByNumber(query);
    }
}
