import random
import csv

# =====================================================================
# 1. PASTE YOUR RAW DATA HERE (NO QUOTES OR COMMAS NEEDED!)
# =====================================================================

RAW_BRANCH_LIST = """

10001
10002
"""

RAW_OPPOSITE_CGL_LIST = """
5051030601
5051240801
1102003001
"""

# =====================================================================
# 2. FIXED CURRENCY LIST
# =====================================================================
FIXED_CURRENCIES = [
    "USD", "AUD", "CAD", "CHF", "SGD", "HKD", 
    "NZD", "EUR", "GBP", "JPY", "DKK", "SEK", "INR"
]

# =====================================================================
# 3. FIXED CGL LIST (I = Income/Cr | E = Expense/Dr)
# =====================================================================
FIXED_CGLS = [
    {"cgl": "7016500301", "type": "I"}, {"cgl": "7016500401", "type": "I"},
    {"cgl": "7016500501", "type": "I"}, {"cgl": "7016501101", "type": "I"},
    {"cgl": "7016500101", "type": "I"}, {"cgl": "7049500101", "type": "I"},
    {"cgl": "7049500301", "type": "I"}, {"cgl": "7049500501", "type": "I"},
    {"cgl": "7049501101", "type": "I"}, {"cgl": "7049501501", "type": "I"},
    {"cgl": "7913505001", "type": "I"}, {"cgl": "7914505001", "type": "I"},
    
    {"cgl": "8041080670", "type": "E"}, {"cgl": "8041080680", "type": "E"},
    {"cgl": "8051080670", "type": "E"}, {"cgl": "8051080690", "type": "E"},
    {"cgl": "8051500170", "type": "E"}, {"cgl": "8051501170", "type": "E"},
    {"cgl": "8001500160", "type": "E"}, {"cgl": "8001500560", "type": "E"},
    {"cgl": "8041070660", "type": "E"}, {"cgl": "8057070601", "type": "E"},
    {"cgl": "8972505001", "type": "E"}, {"cgl": "8972030601", "type": "E"},
    {"cgl": "8972070601", "type": "E"}
]

# =====================================================================
# 4. CONFIGURATION
# =====================================================================
PRODUCT_CODE = "12345678"
REMARKS = "Bulk Script Entry"

# =====================================================================
# 5. DATA GENERATION LOGIC
# =====================================================================
def generate_data():
    print("\n--- Voucher Bulk Data Generator ---")
    
    # Ask the user exactly how many rows they want
    try:
        user_input = input("Enter the total number of rows you want to generate (e.g., 100): ")
        total_rows = int(user_input.strip())
    except ValueError:
        print("Invalid number. Defaulting to 100 rows.")
        total_rows = 100
        
    # Since we need balanced pairs, divide by 2
    total_pairs = total_rows // 2
    if total_pairs == 0:
        total_pairs = 1

    print("\nParsing raw input lists...")
    branch_list = [b.strip() for b in RAW_BRANCH_LIST.splitlines() if b.strip()]
    opposite_cgl_list = [c.strip() for c in RAW_OPPOSITE_CGL_LIST.splitlines() if c.strip()]
    
    if not branch_list:
        print("ERROR: Please paste at least one Branch into the RAW_BRANCH_LIST.")
        return
    if not opposite_cgl_list:
        print("ERROR: Please paste at least one CGL into the RAW_OPPOSITE_CGL_LIST.")
        return

    print(f"Generating {total_pairs * 2} rows of perfectly balanced entries...")
    rows = []

    for i in range(total_pairs):
        branch = random.choice(branch_list)
        currency = random.choice(FIXED_CURRENCIES)
        amount = random.randint(10, 999) * 100  
        
        fixed_item = random.choice(FIXED_CGLS)
        opposite_cgl = random.choice(opposite_cgl_list)
        
        if fixed_item["type"] == "I":
            fixed_leg_type = "Cr"
            opposite_leg_type = "Dr"
        else:
            fixed_leg_type = "Dr"
            opposite_leg_type = "Cr"
            
        row1 = {
            "Branch*": branch, "Currency*": currency, "CGL*": fixed_item["cgl"],
            "Amount*": amount, "Type*": fixed_leg_type, "Remarks*": f"{REMARKS} {i+1}",
            "Product Code": PRODUCT_CODE
        }
        
        row2 = {
            "Branch*": branch, "Currency*": currency, "CGL*": opposite_cgl,
            "Amount*": amount, "Type*": opposite_leg_type, "Remarks*": f"{REMARKS} {i+1}",
            "Product Code": PRODUCT_CODE
        }
        
        rows.append(row1)
        rows.append(row2)

    # =====================================================================
    # 6. WRITE TO CSV (Native Python, No external libraries needed)
    # =====================================================================
    output_filename = f"Voucher_Upload_{total_pairs * 2}_Rows.csv"
    print(f"Writing to CSV ({output_filename})...")
    
    # Extract headers from the first dictionary keys
    headers = ["Branch*", "Currency*", "CGL*", "Amount*", "Type*", "Remarks*", "Product Code"]
    
    with open(output_filename, mode='w', newline='', encoding='utf-8') as file:
        writer = csv.DictWriter(file, fieldnames=headers)
        writer.writeheader()
        writer.writerows(rows)
        
    print(f"\nSUCCESS! Created {output_filename} with {len(rows)} rows.")
    print("NOTE: Download this CSV, open it in Excel, and 'Save As' an .xlsx file before uploading to the backend!")

if __name__ == "__main__":
    generate_data()
