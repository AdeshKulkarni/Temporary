hivejdbcTemplate1.batchUpdate(datalakesql, new BatchPreparedStatementSetter() {

    @Override
    public void setValues(PreparedStatement ps, int i) throws SQLException {
        Object[] args = batchArgs.get(i);

        ps.setLong(1, (Long) args[0]);                      // REQ_ID
        ps.setString(2, (String) args[1]);                  // REQ_STATUS
        ps.setString(3, (String) args[2]);                  // CHANGE_TYPE
        ps.setDate(4, (java.sql.Date) args[3]);             // ✅ REQ_DATE
        ps.setString(5, (String) args[4]);                  // CREATOR_ID
        ps.setInt(6, (Integer) args[5]);                    // CREATOR_ROLE
        ps.setString(7, (String) args[6]);                  // BATCH_ID
        ps.setString(8, (String) args[7]);                  // JOURNAL_ID
        ps.setString(9, (String) args[8]);                  // COMMON_BATCH_REMARKS
        ps.setString(10, (String) args[9]);                 // PAYLOAD
    }

    @Override
    public int getBatchSize() {
        return batchArgs.size();
    }
});
