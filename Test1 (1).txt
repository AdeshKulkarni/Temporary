class Solution {
    int maxi1 = 0;
    int maxi2 = 0;


    public boolean IsAnns(TreeNode u, TreeNode v) {

    if (u == null) return false;

    if (u == v) return true;

    return IsAnns(u.left, v) || IsAnns(u.right, v);

    }


    public int maxAncestorDiff(TreeNode root) {

        if(root == null) {
            return 0;
        }

        TreeNode passRoot = root;

        int Left = maxAncestorDiff(root.left);
        int right = maxAncestorDiff(root.right);

        if(IsAnns(root , root.left)) {

            D.p("root" + root.val);
               D.p("root left" + root.left.val);

            maxi1 = Math.max(Math.abs(passRoot.val - Left), maxi1);
        }


        if(IsAnns(root , root.right)) {
             maxi2 = Math.max(Math.abs(passRoot.val - right), maxi2);
        }       


        return Math.max(maxi1 , maxi2);


        
    }
}
