const validateAndFormatNumber = (value) => {
  let val = String(value).trim();
  if (val === '') return { isvalid: true, value: '0.00', message: '' };

  if (val.match(/\s/g)) return { isvalid: false, value: val, message: 'Space is not allowed.' };
  if (isNaN(parseFloat(val)) && val !== '-' && val !== '-.')
    return { isvalid: false, value: val, message: 'Please enter Numeric value only.' };

  if (!numberRegex.test(val)) {
    if (/^\\-?\d+$/.test(val) && val.length <= 16) {
      /* valid int, needs formatting */
    } else if (/^\\-?\d+\.\d?$/.test(val) && val.split('.')[0].length <= 16) {
      /* valid float, needs formatting */
    } else if (val !== '-' && val !== '-.' && !val.endsWith('.')) {
      return {
        isvalid: false,
        value: val,
        message: 'Amount not allowed more than 16 digits before decimal or 2 after.',
      };
    }
  }

  const idot = val.indexOf('.');
  if (idot === -1 && val !== '-' && val !== '' && !val.endsWith('.')) {
    if (val.length > 16 || (val.startsWith('-') && val.length > 17)) {
      return { isvalid: false, value: val, message: 'Amount not allowed more than 16 digits.' };
    }
    val = `${val}.00`;
  } else if (idot !== -1) {
    const parts = val.split('.');
    if (parts[0].length > 16 || (parts[0].startsWith('-') && parts[0].length > 17)) {
      return { isvalid: false, value: val, message: 'Amount not allowed more than 16 digits before decimal.' };
    }
    if (parts[1].length === 1) val = `${val}0`;
    else if (parts[1].length > 2) val = `${parts[0]}.${parts[1].substring(0, 2)}`;
  }
  return { isvalid: true, value: val, message: '' };
};

const validateAbsoluteNumber = (value) => {
  let val = String(value).trim();
  if (val === '') return { isvalid: true, value: '0', message: '' };

  if (val.match(/\s/g)) return { isvalid: false, value: val, message: 'Space is not allowed.' };
  if (val.includes('.'))
    return { isvalid: false, value: val, message: 'Please enter whole Numeric value only (no decimals).' };
  // Allow '-' for negative numbers during typing
  if (isNaN(parseInt(val, 10)) && val !== '-')
    return { isvalid: false, value: val, message: 'Please enter whole Numeric value only.' };

  // Max 16 digits (17 with sign)
  if ((val.startsWith('-') && val.length > 17) || (!val.startsWith('-') && val.length > 16)) {
    return { isvalid: false, value: val, message: 'Amount not to be more than 16 digits.' };
  }
  return { isvalid: true, value: val, message: '' };
};

  const formatApiValue = (valueFromApi, validationType) => {
    if (valueFromApi === undefined || valueFromApi === null) return validationType === 'numInput' ? '0' : '0.00'; // Default for missing values
    const strValue = String(valueFromApi);
    if (validationType === 'numInput') {
      return strValue === '0' || strValue === '' ? '0' : strValue;
    }
    if (strValue === '0' || parseFloat(strValue) === 0) {
      return '0.00';
    }

    if (typeof valueFromApi === 'number') return valueFromApi.toFixed(2);

    const parts = strValue.split('.');
    if (parts.length === 1 && !isNaN(parseFloat(strValue))) return `${strValue}.00`;
    if (parts.length === 2 && parts[1].length === 1 && !isNaN(parseFloat(strValue))) return `${strValue}0`;

    return strValue;
