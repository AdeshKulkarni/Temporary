@Component("defaultRequestCreation") // Give it a name
@RequiredArgsConstructor // Lombok for constructor injection
@Slf4j
public class DefaultRequestCreationStrategy implements RequestCreationStrategy {

    private final CommonRequestRepository commonRequestRepository;
    private final ObjectMapper objectMapper;

    @Override
    public Object create(CreateRequestDto dto, String creatorId) throws JsonProcessingException {
        KeyablePayload payload = (KeyablePayload) dto.getPayload();
        String targetId = payload.getKey();

        if (targetId == null || targetId.isBlank()) {
            throw new IllegalArgumentException("Could not determine targetId from payload.");
        }

        // The duplicate check, specific to this repository
        List<CommonReq> existingRequests = commonRequestRepository.findByTargetIdAndReqStatus(targetId, RequestStatus.PENDING);
        if (!existingRequests.isEmpty()) {
            log.warn("Attempted to create a duplicate request for targetId {}", targetId);
            throw new DataIntegrityViolationException("An active or pending request for '" + targetId + "' already exists.");
        }

        CommonReq request = new CommonReq();
        request.setCreatorId(creatorId);
        request.setReqType(dto.getRequestType());
        request.setChangeType(dto.getChangeType());
        request.setPayload(objectMapper.writeValueAsString(payload));
        request.setTargetId(targetId);
        request.setReqStatus(RequestStatus.PENDING);

        log.info("Saving new, validated DEFAULT request for targetId {}", targetId);
        return commonRequestRepository.save(request);
    }
}

@Component
public class RequestCreationStrategyFactory {

    private final Map<RequestType, RequestCreationStrategy> strategyMap = new EnumMap<>(RequestType.class);
    
    // Spring will inject the two named beans we created
    public RequestCreationStrategyFactory(
        @Qualifier("defaultRequestCreation") RequestCreationStrategy defaultStrategy,
        @Qualifier("mocRequestCreation") RequestCreationStrategy mocStrategy
    ) {
        // Initialize the map for all default types
        for (RequestType type : RequestType.values()) {
            strategyMap.put(type, defaultStrategy);
        }
        
        // Override the MOC type to use its specific strategy
        strategyMap.put(RequestType.MOC, mocStrategy);
    }

    public RequestCreationStrategy getStrategy(RequestType requestType) {
        RequestCreationStrategy strategy = strategyMap.get(requestType);
        if (strategy == null) {
            // Fallback to default just in case
            return strategyMap.get(RequestType.STATE); // Or any default key
        }
        return strategy;
    }
}
