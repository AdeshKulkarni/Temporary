package com.uber.pricing;

public class Main {
    public static void main(String[] args) {

        // 1. Create the Ride Details (The input data)
        rideDetails ride = new rideDetails();
        ride.setRouteType(routeType.AIRPORT);
        ride.setDisInKm(25.0);
        ride.setRaining(true);
        ride.setNearStadium(false);
        ride.setAccountType(accountType.CORPORATE);

        // 2. Create the empty Engine (The Machine)
        PricingEngine engine = new PricingEngine();

        // ==========================================
        // 3. THE CHOOSING (Routing logic lives here!)
        // ==========================================

        // --- Pick ONE Base Fare ---
        if (ride.getRouteType() == routeType.AIRPORT) {
            engine.setBaseFareStrategy(new FlatRateFare()); // We know it returns 200!
        } else {
            engine.setBaseFareStrategy(new DistanceBasedFare());
        }

        // --- Pick Multiple Surges (Using Composite) ---
        CompositeSurge activeSurges = new CompositeSurge();
        
        if (ride.isRaining()) {
            activeSurges.addSurge(new WeatherSurge()); // Just returns 1.3
        }
        if (ride.isNearStadium()) {
            activeSurges.addSurge(new EventSurge(2.0)); 
        }
        
        engine.setSurgeStrategy(activeSurges);

        // --- Pick the Best Discount ---
        // We will assume you built a BestDiscountStrategy that loops through eligible ones
        BestDiscountStrategy eligibleDiscounts = new BestDiscountStrategy();
        
        if (ride.getAccountType() == accountType.CORPORATE) {
            eligibleDiscounts.addDiscount(new CorporateDiscount());
        }
        if (ride.getUserTotalRides() > 50) {
            eligibleDiscounts.addDiscount(new LoyaltyDiscount());
        }

        engine.setDiscountStrategy(eligibleDiscounts);

        // ==========================================
        // 4. THE DOING (Execute the calculation)
        // ==========================================
        double finalFare = engine.calculateFinalPrice(ride);
        System.out.println("The final calculated fare is: ₹" + finalFare);
    }
}
