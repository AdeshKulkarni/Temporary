class Solution {

    // FIX 3: Global variable to track the maximum streak safely
    int maxCount = 0;

    // FIX 1: Pass the direction we came from (cameFromLeft) as a parameter
    public void Helper(TreeNode root, boolean cameFromLeft, int currentCount) {
        
        if (root == null) {
            return;
        }

        // Update the global record if our current streak is the biggest we've seen
        maxCount = Math.max(maxCount, currentCount);

        // FIX 2: We ALWAYS explore both left and right children, 
        // but we change the 'currentCount' based on the direction!
        
        if (cameFromLeft) {
            // If we came from the left, going RIGHT continues the ZigZag (streak + 1)
            Helper(root.right, false, currentCount + 1);
            
            // We still explore going left, but it breaks the ZigZag, so streak resets to 1
            Helper(root.left, true, 1);
            
        } else {
            // If we came from the right, going LEFT continues the ZigZag (streak + 1)
            Helper(root.left, true, currentCount + 1);
            
            // We still explore going right, but it breaks the ZigZag, so streak resets to 1
            Helper(root.right, false, 1);
        }
    }

    public int longestZigZag(TreeNode root) {
        if (root == null) {
            return 0;
        }
        
        // Kick off the recursion from the root.
        // We try both possible starting directions.
        Helper(root.left, true, 1);
        Helper(root.right, false, 1);
        
        return maxCount;
    }
}
