class Solution {
    public int widthOfBinaryTree(TreeNode root) {

        if (root == null) {
            return 0;
        }

        Queue<TreeNode> q = new LinkedList<>();

        int MaxCount = 0;

        int DoubleCounter = 1;

        q.offer(root);

        while (!q.isEmpty()) {

            int Qsize = q.size();
            D.p("before double" + DoubleCounter);
              if (Qsize > 1) {


                    DoubleCounter = DoubleCounter * 2;
                    D.p("after double" + DoubleCounter);

                    int diff = DoubleCounter - Qsize;

                      D.p("diff" + diff);

                    MaxCount = diff + Qsize;

                    D.p("MaxCount" + MaxCount);

                }

            for (int i = 0; i < Qsize; i++) {

                TreeNode firstnode = q.poll();

                             

                if (firstnode.left != null) {
                    q.offer(firstnode.left);
                }

                if (firstnode.right != null) {
                    q.offer(firstnode.right);
                }

            }

        }

        return MaxCount;

    }
}
