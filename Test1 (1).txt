void solve(int i) {

    if (i == 2) {
        System.out.println("Base: " + i);
        return;
    }

    System.out.println("Before: " + i);

    solve(i + 1);                       // First recursive call

    System.out.println("Middle: " + i);

    solve(i + 1);                       // Second recursive call

    System.out.println("After: " + i);
}

solve(0);
