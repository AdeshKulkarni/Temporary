import { createSlice } from "@reduxjs/toolkit";

const initialState = {
  user: null, // The decoded user object
  token: null, // The Access Token string
  payloadEncryptionKey: null, // Encryption key used for request/response payloads
  isAuthenticated: false,
  loading: false,
  error: null,
};

const authSlice = createSlice({
  name: "auth",
  initialState,
  reducers: {
    // Existing
    setToken: (state, action) => {
      state.token = action.payload;
      // localStorage.setItem("token", action.payload);
    },

    setUser: (state, action) => {
      state.user = action.payload;
      // localStorage.setItem("user", JSON.stringify(action.payload));
    },

    logout: (state) => {
      state.token = null;
      state.user = null;
      state.payloadEncryptionKey = null;
      state.isAuthenticated = false;

      localStorage.removeItem("token");
      localStorage.removeItem("user");
      localStorage.removeItem("payloadEncryptionKey");
    },

    // Called immediately after Login API success
    loginSuccess: (state, action) => {
      const { user, token, payloadEncryptionKey } = action.payload;

      state.user = user;
      state.token = token;
      state.payloadEncryptionKey = payloadEncryptionKey || null;
      state.isAuthenticated = true;
      state.error = null;
      state.loading = false;

      // Persist to local storage for page reloads
      // localStorage.setItem("token", token);
      // localStorage.setItem("user", JSON.stringify(user));

      if (payloadEncryptionKey) {
        localStorage.setItem("payloadEncryptionKey", payloadEncryptionKey);
      }
    },

    // Called when Refresh Token API returns a new Access Token
    tokenRefreshed: (state, action) => {
      const { token, user, payloadEncryptionKey } = action.payload;

      state.token = token;

      if (user) {
        state.user = user;
        // localStorage.setItem("user", JSON.stringify(user));
      }

      if (payloadEncryptionKey) {
        state.payloadEncryptionKey = payloadEncryptionKey;

        // localStorage.setItem("payloadEncryptionKey", payloadEncryptionKey);
      }

      state.isAuthenticated = true;

      // localStorage.setItem("token", token);
    },

    // Called on Logout or Session Expiry
    logoutSuccess: (state) => {
      state.user = null;
      state.token = null;
      state.payloadEncryptionKey = null;
      state.isAuthenticated = false;
      state.error = null;
      state.loading = false;

      localStorage.removeItem("token");
      localStorage.removeItem("user");
      localStorage.removeItem("payloadEncryptionKey");
    },

    // Optional: Load state from LocalStorage on app startup
    initializeAuth: (state) => {
      // const token = localStorage.getItem("token");
      // const userStr = localStorage.getItem("user");
      const payloadEncryptionKey = localStorage.getItem("payloadEncryptionKey");

      if (token && userStr) {
        state.token = token;
        state.user = JSON.parse(userStr);
        state.payloadEncryptionKey = payloadEncryptionKey;
        state.isAuthenticated = true;
      }
    },
  },
});

export const {
  setToken,
  setUser,
  logout,
  loginSuccess,
  tokenRefreshed,
  logoutSuccess,
  initializeAuth,
} = authSlice.actions;

export default authSlice.reducer;

import { configureStore, combineReducers } from "@reduxjs/toolkit";
import {
  persistStore,
  persistReducer,
  FLUSH,
  REHYDRATE,
  PAUSE,
  PERSIST,
  PURGE,
  REGISTER,
} from "redux-persist";
import storageSession from "redux-persist/lib/storage/session";
import undoable from "redux-undo";

import menuReducer from "./slices/menuSlice";
import authReducer from "./slices/authSlice";
import tabReducer from "./slices/tabSlice";
import uiReducer from "./slices/uiSlice";
import templateReducer from "../features/reportTemplateBuilder/store/templateSlice";
import configReducer from "../features/reportTemplateBuilder/store/configSlice";

/* Persist Configs */

const authPersistConfig = {
  key: "auth",
  storage: storageSession,
  whitelist: ["token", "user"],
};

const sessionPersistConfig = {
  key: "session",
  storage: storageSession,
  whitelist: ["menus", "tabId"],
};

/* Undoable Template Reducer */

const undoableTemplateReducer = undoable(templateReducer, {
  limit: 50,
  filter: (action) => {
    const undoableActions = [
      "template/updateCell",
      "template/updateCellRender",
      "template/updateCellFormat",
      "template/updateCellSource",
      "template/addColumn",
      "template/removeColumn",
      "template/updateColumn",
      "template/updateColumnFormat",
      "template/addRow",
      "template/removeRow",
      "template/updateRow",
      "template/reorderRows",
      "template/updateDynamicConfig",
      "template/setRowHeight",
    ];
    return undoableActions.includes(action.type);
  },
});

/*  Root Reducer */

const rootReducer = combineReducers({
  auth: persistReducer(authPersistConfig, authReducer),
  menus: persistReducer(sessionPersistConfig, menuReducer),
  tab: persistReducer(sessionPersistConfig, tabReducer),
  ui: uiReducer,

  template: undoableTemplateReducer,
  config: configReducer,
});


/**
 * Redux store
 * 
 * @author v1018404
 */
export const store = configureStore({
  reducer: rootReducer,
  middleware: (getDefaultMiddleware) =>
    getDefaultMiddleware({
      serializableCheck: {
        ignoredActions: [FLUSH, REHYDRATE, PAUSE, PERSIST, PURGE, REGISTER],
      },
      immutableCheck: false,
    }),
});

export const persistor = persistStore(store);
