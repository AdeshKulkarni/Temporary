import React, { useState, useMemo, useEffect } from "react";
import {
  Box,
  Button,
  IconButton,
  Stack,
  Tooltip,
  Dialog,
  DialogTitle,
  DialogContent,
  DialogActions,
  TextField,
  Typography,
} from "@mui/material";
import { DataGrid } from "@mui/x-data-grid";
import {
  Edit as EditIcon,
  Delete as DeleteIcon,
  Add as AddIcon,
} from "@mui/icons-material";
import useApi from "../../../hooks/useApi"; 
import useCustomSnackbar from "../../../utils/useCustomSnackbar"; 

const initialCurrencyData = {
  currencyCode: "",
  currencyName: "",
  currencyRate: "",
};

export default function CurrencyMaster() {
  const { callApi } = useApi();
  const showSnackBar = useCustomSnackbar();
  const [rows, setRows] = useState([]);
  const [isLoading, setIsLoading] = useState(true);

  const [formDialog, setFormDialog] = useState({
    open: false,
    mode: "add", 
  });
  const [currencyData, setCurrencyData] = useState(initialCurrencyData);

  const [deleteDialog, setDeleteDialog] = useState({
    open: false,
    currencyCode: null,
  });

  const fetchData = async () => {
    setIsLoading(true);
    try {
        const payload = {};

      const response = await callApi("/currency-master", {}, "GET");
      console.log("response" , response); 


      

      if (response && response.data && response.data.length > 0) {
        setRows(response.data);
      } else {
        showSnackBar("No currency data found.", "info");
        setRows([]);
      }
    } catch (error) {
      console.error("Error fetching currency data:", error);
      showSnackBar(
        "Something went wrong while fetching data. Please try again later.",
        "error"
      );
      setRows([]);
    } finally {
      setIsLoading(false);
    }
  };

  useEffect(() => {
    fetchData();
  }, []);

  const handleOpenFormDialog = (mode, data = initialCurrencyData) => {
    setCurrencyData(data);
    setFormDialog({ open: true, mode });
  };

  const handleCloseFormDialog = () => {
    setFormDialog({ open: false, mode: "add" });
    setCurrencyData(initialCurrencyData);
  };

  const handleFormChange = (event) => {
    const { name, value } = event.target;
    setCurrencyData((prevData) => ({
      ...prevData,
      [name]: value,
    }));
  };

  const handleFormSubmit = async () => {

    if (!currencyData.currencyCode || !currencyData.currencyName || !currencyData.currencyRate) {
        showSnackBar("Please fill in all required fields.", "warning");
        return;
    }

    try {
        let response;
        const payload = {
            ...currencyData,
            currencyRate: parseFloat(currencyData.currencyRate) 
        };

        if (formDialog.mode === 'add') {
            response = await callApi("/master/currency", payload, "POST");
            showSnackBar("Currency added successfully!", "success");
        } else {
            response = await callApi(`/master/currency/${currencyData.currencyCode}`, payload, "PUT");
            showSnackBar("Currency updated successfully!", "success");
        }
        
        handleCloseFormDialog();
        fetchData(); 
    } catch (error) {
        console.error(`Error during currency ${formDialog.mode}:`, error);
        showSnackBar(`Failed to ${formDialog.mode} currency. Please try again.`, "error");
    }
  };

  const handleOpenDeleteDialog = (currencyCode) => {
    setDeleteDialog({ open: true, currencyCode });
  };

  const handleCloseDeleteDialog = () => {
    setDeleteDialog({ open: false, currencyCode: null });
  };

  const handleDeleteConfirm = async () => {
    try {
      await callApi(`currency-master/${deleteDialog.currencyCode}`, {}, "DELETE");
      showSnackBar("Currency deleted successfully!", "success");
      handleCloseDeleteDialog();
      fetchData(); // Refresh the grid
    } catch (error) {
      console.error("Error deleting currency:", error);
      showSnackBar("Failed to delete currency. Please try again.", "error");
    }
  };

  const columns = useMemo(
    () => [
      {
        field: "currencyCode",
        headerName: "Currency Code",
        flex: 1,
        sortable: true,
      },
      {
        field: "currencyName",
        headerName: "Currency Name",
        flex: 2,
        sortable: true,
      },
      {
        field: "currencyRate",
        headerName: "Currency Rate",
        flex: 1,
        sortable: true,
        align: "right",
        headerAlign: "right",
      },
      {
        field: "createdAt",
        headerName: "Created At",
        flex: 1,
        sortable: true,
        align: "center",
        headerAlign: "center",
        valueGetter: (params) => new Date(params.value).toLocaleString(),
      },
      {
        field: "actions",
        headerName: "Actions",
        sortable: false,
        flex: 1,
        align: "center",
        headerAlign: "center",
        renderCell: (params) => (
          <Stack direction="row" spacing={1}>
            <Tooltip title="Edit">
              <IconButton
                size="small"
                onClick={() => handleOpenFormDialog('edit', params.row)}
                color="primary"
              >
                <EditIcon fontSize="inherit" />
              </IconButton>
            </Tooltip>
            <Tooltip title="Delete">
              <IconButton
                size="small"
                onClick={() => handleOpenDeleteDialog(params.row.currencyCode)}
                color="error"
              >
                <DeleteIcon fontSize="inherit" />
              </IconButton>
            </Tooltip>
          </Stack>
        ),
      },
    ],
    []
  );

  return (
    <Box>
      <Box sx={{ mb: 2, display: 'flex', justifyContent: 'flex-end' }}>
        <Button
            variant="contained"
            startIcon={<AddIcon />}
            onClick={() => handleOpenFormDialog('add')}
        >
            Add Currency
        </Button>
      </Box>
      <DataGrid
        rows={rows}
        columns={columns}
        disableRowSelectionOnClick
        pageSizeOptions={[5, 10, 25, 50]}
        initialState={{
          pagination: { paginationModel: { pageSize: 10, page: 0 } },
          sorting: { sortModel: [{ field: "createdAt", sort: "desc" }] },
        }}
        getRowId={(row) => row.currencyCode} 
        loading={isLoading}
        sx={{
          height: 600,
          width: "100%",
          "& .MuiDataGrid-cell:focus": {
            outline: "none",
          },
        }}
      />

      {/* Add/Edit Form Dialog */}
      <Dialog open={formDialog.open} onClose={handleCloseFormDialog} maxWidth="sm" fullWidth>
        <DialogTitle>{formDialog.mode === 'add' ? 'Add New Currency' : 'Edit Currency'}</DialogTitle>
        <DialogContent>
          <TextField
            autoFocus
            margin="dense"
            name="currencyCode"
            label="Currency Code"
            type="text"
            fullWidth
            variant="outlined"
            value={currencyData.currencyCode}
            onChange={handleFormChange}
            disabled={formDialog.mode === 'edit'} 
            inputProps={{ maxLength: 3 }}
          />
          <TextField
            margin="dense"
            name="currencyName"
            label="Currency Name"
            type="text"
            fullWidth
            variant="outlined"
            value={currencyData.currencyName}
            onChange={handleFormChange}
          />
          <TextField
            margin="dense"
            name="currencyRate"
            label="Currency Rate"
            type="number"
            fullWidth
            variant="outlined"
            value={currencyData.currencyRate}
            onChange={handleFormChange}
          />
        </DialogContent>
        <DialogActions>
          <Button onClick={handleCloseFormDialog} color="inherit" variant="contained">Cancel</Button>
          <Button onClick={handleFormSubmit} color="primary" variant="contained">Save</Button>
        </DialogActions>
      </Dialog>

      {/* Delete Confirmation Dialog */}
      <Dialog open={deleteDialog.open} onClose={handleCloseDeleteDialog} maxWidth="xs" fullWidth>
        <DialogTitle>Confirm Deletion</DialogTitle>
        <DialogContent>
          <Typography>
            Are you sure you want to delete the currency with code '{deleteDialog.currencyCode}'?
          </Typography>
        </DialogContent>
        <DialogActions>
          <Button onClick={handleCloseDeleteDialog} color="inherit" variant="contained">Cancel</Button>
          <Button onClick={handleDeleteConfirm} color="error" variant="contained">Delete</Button>
        </DialogActions>
      </Dialog>
    </Box>
  );
}
