package com.fincore.VoucherService.Models.enums;

import com.fasterxml.jackson.annotation.JsonCreator; // CORRECTED IMPORT (Was org.codehaus.jackson)
import lombok.Getter;

@Getter
public enum RequestStatus {
    PENDING("P"),
    ACCEPTED("A"),
    REJECTED("R");

    private final String code;

    RequestStatus(String code) {
        this.code = code;
    }

    // Tells Jackson: If the frontend sends "A", map it to ACCEPTED.
    @JsonCreator
    public static RequestStatus fromCode(String code) {
        for (RequestStatus status : values()) {
            if (status.getCode().equalsIgnoreCase(code) || status.name().equalsIgnoreCase(code)) {
                return status;
            }
        }
        throw new IllegalArgumentException("Invalid status code: " + code);
    }
}
