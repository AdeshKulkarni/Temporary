
import React, { useState, useRef } from "react";
import {
  Box,
  Typography,
  Paper,
  Grid,
  TextField,
  Button,
  Alert,
  Table,
  TableBody,
  TableCell,
  TableContainer,
  TableHead,
  TableRow,
  Stack,
  CircularProgress,
  Divider,
  IconButton,
} from "@mui/material";
import {
  CloudUpload as CloudUploadIcon,
  Download as DownloadIcon,
  RestartAlt as RestartAltIcon,
  Description as DescriptionIcon,
  CheckCircle as CheckCircleIcon,
} from "@mui/icons-material";

// Import consistent styles
import {
  mainContainerStyle,
  tableHeaderCellStyle,
  cellBorderStyle,
  tableContainerStyle,
  uploadBoxStyle,
  controlPanelStyle,
} from "./JournalBulkStyle";

const JournalBulkUpload = () => {
  const [selectedDate, setSelectedDate] = useState("");
  const [file, setFile] = useState(null);
  const [processedRows, setProcessedRows] = useState([]);
  const [loading, setLoading] = useState(false);
  const [error, setError] = useState("");
  const [successMsg, setSuccessMsg] = useState("");

  const fileInputRef = useRef(null);



  const padZeros = (str, length) => {
    const s = String(str || "");
    return s.padStart(length, "0");
  };

  const formatAmount = (val) => {
    const num = parseFloat(val);
    if (isNaN(num)) return "0000000000000000.0000";
    const fixed = num.toFixed(4);
    const [integerPart, decimalPart] = fixed.split(".");
    const paddedInt = integerPart.padStart(16, "0");
    return `${paddedInt}.${decimalPart}`;
  };

  const getFinancialInfo = (dateString) => {
    if (!dateString) return { year: "", month: "", formattedDate: "" };

    const date = new Date(dateString);
    const day = date.getDate();
    const monthIndex = date.getMonth() + 1;
    const fullYear = date.getFullYear();

    const dd = String(day).padStart(2, "0");
    const mm = String(monthIndex).padStart(2, "0");
    const formattedDate = `${dd}${mm}${fullYear}`;

    
    let finYear, finMonth;
    if (monthIndex >= 4) {
      finYear = fullYear;
      finMonth = monthIndex - 3;
    } else {
      finYear = fullYear - 1;
      finMonth = monthIndex + 9;
    }

    return {
      year: String(finYear),
      month: String(finMonth).padStart(2, "0"),
      formattedDate,
    };
  };

  // --- HANDLERS ---

  const handleDateChange = (e) => {
    setSelectedDate(e.target.value);
    setProcessedRows([]);
    setSuccessMsg("");
    setError("");
  };

  const handleFileChange = (e) => {
    const selectedFile = e.target.files[0];
    if (selectedFile) {
      const fileType = selectedFile.name.split(".").pop().toLowerCase();
      if (fileType !== "xlsx" && fileType !== "xls") {
        setError("Invalid file type. Please upload .xlsx or .xls");
        setFile(null);
        return;
      }
      setFile(selectedFile);
      setError("");
      setSuccessMsg("");
      setProcessedRows([]);
    }
  };

  const handleUploadBoxClick = () => {
    fileInputRef.current.click();
  };

  const resetForm = () => {
    setFile(null);
    setSelectedDate("");
    setProcessedRows([]);
    setError("");
    setSuccessMsg("");
    if (fileInputRef.current) fileInputRef.current.value = "";
  };

  const processFile = async () => {
    if (!selectedDate || !file) {
      setError("Please select a Date and upload a File.");
      return;
    }

    setLoading(true);
    setError("");

    // Load XLSX via CDN if not present
    if (!window.XLSX) {
      const script = document.createElement("script");
      script.src =
        "https://cdn.sheetjs.com/xlsx-latest/package/dist/xlsx.full.min.js";
      document.body.appendChild(script);
      await new Promise((resolve) => (script.onload = resolve));
    }

    const reader = new FileReader();
    reader.onload = (e) => {
      try {
        const data = new Uint8Array(e.target.result);
        const workbook = window.XLSX.read(data, { type: "array" });
        const worksheet = workbook.Sheets[workbook.SheetNames[0]];
        const jsonData = window.XLSX.utils.sheet_to_json(worksheet);

        if (jsonData.length === 0) {
          setError("File is empty.");
          setLoading(false);
          return;
        }

        const {
          year: finYear,
          month: finMonth,
          formattedDate,
        } = getFinancialInfo(selectedDate);

        const mappedData = jsonData.map((row, index) => {
          const getVal = (key) =>
            row[key] || row[key.toLowerCase()] || row[key.toUpperCase()] || "";
          console.log(getVal);

          const col1 = "01";
          const col2 = formattedDate;
          const col3 = finYear;
          const col4 = finMonth;
          const col5 = padZeros(getVal("branch") || getVal("Branch"), 5) || "Value is not getting";
          const col6 = getVal("currency") || "INR";
          const col7 = getVal("cgl") || getVal("code") || "";
          const col8 = formatAmount(getVal("amount"));
          const col9 = getVal("txntype") || getVal("TxnType");
          const col10 = getVal("Remarks") || getVal("remarks");

          const csvLine = [
            col1,
            col2,
            col3,
            col4,
            col5,
            col6,
            col7,
            col8,
            col9,
            col10,
            "000000000",
            "00000",
            "0000000000",
            "X",
          ].join(",");

          return {
            id: index,
            csvLine: csvLine,
            preview: {
              date: col2,
              finYear: col3,
              finMonth: col4,
              branch: col5,
              currency: col6,
              cgl: col7,
              amount: col8,
              remarks: col10,
            },
          };
        });

        setProcessedRows(mappedData);
        setSuccessMsg(`Processed ${mappedData.length} records successfully.`);
      } catch (err) {
        console.error(err);
        setError("Failed to process file. Check format.");
      } finally {
        setLoading(false);
      }
    };
    reader.readAsArrayBuffer(file);
  };

  const downloadCSV = () => {
    if (processedRows.length === 0) return;
    const csvContent = processedRows.map((r) => r.csvLine).join("\n");
    const blob = new Blob([csvContent], { type: "text/csv;charset=utf-8;" });
    const link = document.createElement("a");
    const url = URL.createObjectURL(blob);
    const fileName = `Journal_Upload_${selectedDate}.csv`;
    link.setAttribute("href", url);
    link.setAttribute("download", fileName);
    document.body.appendChild(link);
    link.click();
    document.body.removeChild(link);
  };

  return (
    <Box sx={mainContainerStyle}>
      {/* Controls Section */}
      <Paper elevation={3} sx={controlPanelStyle}>
        

        <Grid container spacing={3}>
          {/* Date Picker */}
          <Grid item xs={12} md={4}>
            <Typography variant="subtitle2" gutterBottom>
              1. Select Posting Date
            </Typography>
            <TextField
              type="date"
              fullWidth
              size="small"
              value={selectedDate}
              onChange={handleDateChange}
              InputLabelProps={{ shrink: true }}
            />
            {selectedDate && (
              <Stack direction="row" spacing={1} sx={{ mt: 1 }}>
                <Typography variant="caption" color="textSecondary">
                  FY: {getFinancialInfo(selectedDate).year}
                </Typography>
                <Typography variant="caption" color="textSecondary">
                  |
                </Typography>
                <Typography variant="caption" color="textSecondary">
                  Month: {getFinancialInfo(selectedDate).month}
                </Typography>
              </Stack>
            )}
          </Grid>

          {/* File Upload Area */}
          <Grid item xs={12} md={4}>
            <Typography variant="subtitle2" gutterBottom>
              2. Upload Excel File
            </Typography>
            <Box sx={uploadBoxStyle} onClick={handleUploadBoxClick}>
              <input
                type="file"
                hidden
                accept=".xlsx, .xls"
                onChange={handleFileChange}
                ref={fileInputRef}
              />
              {file ? (
                <>
                  <DescriptionIcon color="success" fontSize="large" />
                  <Typography variant="body2" sx={{ mt: 1, fontWeight: "bold" }}>
                    {file.name}
                  </Typography>
                  <Typography variant="caption" color="textSecondary">
                    Click to change
                  </Typography>
                </>
              ) : (
                <>
                  <CloudUploadIcon color="action" fontSize="large" />
                  <Typography variant="body2" sx={{ mt: 1 }}>
                    Click to Upload Excel
                  </Typography>
                  <Typography variant="caption" color="textSecondary">
                    .xlsx or .xls
                  </Typography>
                </>
              )}
            </Box>
          </Grid>

          {/* Actions */}
          <Grid
            item
            xs={12}
            md={4}
            sx={{
              display: "flex",
              flexDirection: "column",
              justifyContent: "center",
              gap: 2,
            }}
          >
            <Button
              variant="contained"
              color="primary"
              onClick={processFile}
              disabled={loading || !file || !selectedDate}
              startIcon={
                loading ? <CircularProgress size={20} color="inherit" /> : <CheckCircleIcon />
              }
              fullWidth
            >
              {loading ? "Processing..." : "Process File"}
            </Button>

            <Button
              variant="outlined"
              color="error"
              onClick={resetForm}
              startIcon={<RestartAltIcon />}
              fullWidth
            >
              Reset
            </Button>
          </Grid>
        </Grid>

        {/* Alerts */}
        <Box sx={{ mt: 2 }}>
          {error && (
            <Alert severity="error" sx={{ mt: 1 }}>
              {error}
            </Alert>
          )}
          {successMsg && (
            <Alert severity="success" sx={{ mt: 1 }}>
              {successMsg}
            </Alert>
          )}
        </Box>
      </Paper>

      {/* Data Preview Table */}
      {processedRows.length > 0 && (
        <Paper elevation={3} sx={{ mt: 3, p: 0, overflow: "hidden" }}>
          <Box
            sx={{
              p: 2,
              display: "flex",
              justifyContent: "space-between",
              alignItems: "center",
              bgcolor: "#f5f5f5",
            }}
          >
            <Typography variant="subtitle1" fontWeight="bold">
              Preview & Download ({processedRows.length} records)
            </Typography>
            <Button
              variant="contained"
              color="success"
              startIcon={<DownloadIcon />}
              onClick={downloadCSV}
            >
              Download CSV
            </Button>
          </Box>
          <Divider />
          
          <TableContainer sx={tableContainerStyle}>
            <Table stickyHeader size="small">
              <TableHead>
                <TableRow>
                  <TableCell sx={tableHeaderCellStyle}>Date</TableCell>
                  <TableCell sx={tableHeaderCellStyle}>FY</TableCell>
                  <TableCell sx={tableHeaderCellStyle}>FM</TableCell>
                  <TableCell sx={tableHeaderCellStyle}>Branch</TableCell>
                  <TableCell sx={tableHeaderCellStyle}>Currency</TableCell>
                  <TableCell sx={tableHeaderCellStyle}>CGL</TableCell>
                  <TableCell sx={tableHeaderCellStyle} align="right">Amount</TableCell>
                  <TableCell sx={tableHeaderCellStyle}>Remarks</TableCell>
                  <TableCell sx={tableHeaderCellStyle}>CSV Output (Preview)</TableCell>
                </TableRow>
              </TableHead>
              <TableBody>
                {processedRows.slice(0, 100).map((row) => (
                  <TableRow hover key={row.id}>
                    <TableCell sx={cellBorderStyle}>{row.preview.date}</TableCell>
                    <TableCell sx={cellBorderStyle}>{row.preview.finYear}</TableCell>
                    <TableCell sx={cellBorderStyle}>{row.preview.finMonth}</TableCell>
                    <TableCell sx={{ ...cellBorderStyle, color: "primary.main", fontWeight: "bold" }}>
                      {row.preview.branch}
                    </TableCell>
                    <TableCell sx={cellBorderStyle}>{row.preview.currency}</TableCell>
                    <TableCell sx={cellBorderStyle}>{row.preview.cgl}</TableCell>
                    <TableCell sx={{ ...cellBorderStyle, fontFamily: "monospace" }} align="right">
                      {row.preview.amount}
                    </TableCell>
                    <TableCell sx={cellBorderStyle}>{row.preview.remarks}</TableCell>
                    <TableCell
                      sx={{
                        ...cellBorderStyle,
                        fontFamily: "monospace",
                        fontSize: "0.75rem",
                        color: "text.secondary",
                        maxWidth: "300px",
                        whiteSpace: "nowrap",
                        overflow: "hidden",
                        textOverflow: "ellipsis",
                      }}
                    >
                      {row.csvLine}
                    </TableCell>
                  </TableRow>
                ))}
              </TableBody>
            </Table>
          </TableContainer>
          {processedRows.length > 100 && (
            <Box sx={{ p: 1, textAlign: "center", bgcolor: "#fafafa" }}>
              <Typography variant="caption" color="textSecondary">
                Preview limited to first 100 rows for performance. Download to see all.
              </Typography>
            </Box>
          )}
        </Paper>
      )}
    </Box>
  );
};
