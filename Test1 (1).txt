import csv
import random

# --- CONFIGURATION ---
FILENAME = "clean_test_upload.csv"
NUM_ROWS = 1500  # Generates 1500 valid rows
SELECTED_DATE = "17122025"  # Format: ddMMyyyy (Match this in Frontend)

# --- CONSTANTS ---
BATCH_ID = "01"
YEAR = "2025"
MONTH = "09"
VALID_BRANCH = "1001"
VALID_CURRENCY = "USD"
VALID_CGL = "1234567890"

# --- GENERATION LOGIC ---
def generate_valid_row(index):
    # Alternating Debit/Credit to keep it somewhat balanced
    txn_type = "Dr" if index % 2 == 0 else "Cr"
    
    return [
        BATCH_ID,           # Col 1: BatchID
        SELECTED_DATE,      # Col 2: Date
        YEAR,               # Col 3: Year
        MONTH,              # Col 4: Month
        VALID_BRANCH,       # Col 5: Branch
        VALID_CURRENCY,     # Col 6: Currency
        VALID_CGL,          # Col 7: CGL
        f"{random.randint(10, 10000)}.{random.randint(10, 99)}", # Col 8: Amount
        txn_type,           # Col 9: Type
        f"Valid_Txn_Row_{index}", # Col 10: Remarks
        "12345678",         # Col 11: Product Code (8 digits)
        "B",                # Col 12: Static
        "C",                # Col 13: Static
        "D"                 # Col 14: Static
    ]

# --- WRITE FILE ---
with open(FILENAME, mode='w', newline='') as file:
    writer = csv.writer(file)
    # System files usually don't have headers, but if you need one, uncomment:
    # writer.writerow(["BatchID","Date","Year","Month","Branch","Currency","CGL","Amount","Type","Remarks","Product","C12","C13","C14"])
    
    print(f"Generating {NUM_ROWS} VALID rows...")
    for i in range(1, NUM_ROWS + 1):
        writer.writerow(generate_valid_row(i))

print(f"Successfully created '{FILENAME}'. Upload this file with date {SELECTED_DATE}.")
