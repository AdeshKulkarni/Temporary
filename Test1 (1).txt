const generateQuarterEndDates = (previousYearDate) => {
  // previousYearDate is expected in "DD/MM/YYYY" format, e.g., "30/06/2024"
  if (!previousYearDate) {
    return []; // Return empty if no date is provided
  }

  // --- Step 1: Parse the input date to get the month and year ---
  const parts = previousYearDate.split('/');
  const month = parseInt(parts[1], 10); // Get the month (e.g., 6 for June)
  const year = parseInt(parts[2], 10);  // Get the year (e.g., 2024)

  let targetYear;

  // --- Step 2: Apply the core logic ---
  // The financial year in India ends in March.
  // If the last quarter was June (6), September (9), or December (12) of a given year,
  // it means we are in the next financial year, which will end in the NEXT calendar year.
  if ([6, 9, 12].includes(month)) {
    targetYear = year + 1;
  } else {
    // If the last quarter was March (3), we are at the start of a financial year.
    // The target end date is still within the SAME calendar year.
    targetYear = year;
  }

  // --- Step 3: Create the new date string ---
  const newDate = `31-03-${targetYear}`;

  // Return in the required array format
  return [{ DATE: newDate }];
};
