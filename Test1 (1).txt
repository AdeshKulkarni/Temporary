import React, { useState, useMemo, useEffect } from "react";
import { Box } from "@mui/material";
import { DataGrid } from "@mui/x-data-grid";
import useApi from "../../../hooks/useApi";
import useCustomSnackbar from "../../../utils/useCustomSnackbar";

export default function CurrencyMaster() {
  const { callApi } = useApi();
  const showSnackBar = useCustomSnackbar();
  const [rows, setRows] = useState([]);
  const [isLoading, setIsLoading] = useState(true);

  const columns = useMemo(
    () => [
      {
        field: "currencyCode",
        headerName: "Currency Code",
        flex: 1,
        sortable: true,
        // ADDED: Centered alignment
        align: "center",
        headerAlign: "center",
      },
      {
        field: "currencyName",
        headerName: "Currency Name",
        flex: 2,
        sortable: true,
        // ADDED: Centered alignment
        align: "center",
        headerAlign: "center",
      },
      {
        field: "currencyRate",
        headerName: "Currency Rate",
        flex: 1,
        sortable: true,
        // CHANGED: Alignment from 'right' to 'center'
        align: "center",
        headerAlign: "center",
      },
      {
        // CHANGED: Using 'updatedAt' field from your API response
        field: "updatedAt",
        headerName: "Last Updated",
        flex: 1,
        sortable: true,
        align: "center",
        headerAlign: "center",
        // This formatter correctly handles the date string
        valueFormatter: (params) =>
          params.value ? new Date(params.value).toLocaleString() : "N/A",
      },
    ],
    []
  );

  const fetchData = async () => {
    setIsLoading(true);
    try {
      const response = await callApi("CM/currency-master", null, "GET");

      if (response && response.data && Array.isArray(response.data)) {
        // DataGrid requires a unique 'id' field for each row
        const dataWithIds = response.data.map((row) => ({
          ...row,
          id: row.currencyCode, // Using currencyCode as the unique ID
        }));

        setRows(dataWithIds);
        if (dataWithIds.length === 0) {
          showSnackBar("No currency data found.", "info");
        }
      } else {
        showSnackBar("Received invalid data format.", "error");
        setRows([]);
      }
    } catch (error) {
      console.error("Error fetching currency data:", error);
      showSnackBar(
        "Something went wrong while fetching data. Please try again later.",
        "error"
      );
      setRows([]);
    } finally {
      setIsLoading(false);
    }
  };

  useEffect(() => {
    fetchData();
  }, []);

  return (
    <Box>
      <DataGrid
        rows={rows}
        columns={columns}
        disableRowSelectionOnClick
        pageSizeOptions={[5, 10, 25, 50]}
        initialState={{
          pagination: { paginationModel: { pageSize: 10, page: 0 } },
          // Sorting by the updated date field
          sorting: { sortModel: [{ field: "updatedAt", sort: "desc" }] },
        }}
        loading={isLoading}
        sx={{
          height: 600,
          width: "100%",
          "& .MuiDataGrid-cell:focus": {
            outline: "none",
          },
        }}
      />
    </Box>
  );
}
