public class getChronologicalPosts implements InstagramIterator<post>{

    int index = 0;

    List<post> PostList = new ArrayList<>();

    public getChronologicalPosts(List<post> postList) {
        PostList = postList;
    }

    @Override
    public boolean hasNext() {
        return index < PostList.size();
    }

    ///  for now we are assuming this  will return the data by filtering the Chronologically
    ///  as our goal is not to learn Chronological we need to learn how we are attracting

    @Override
    public post next() {

        ///  in real we will throw Exception
        if(!hasNext()) {
            System.out.println("no more post");
        }

        return PostList.get(index++);
    }
}


public interface InstagramIterator<T> {

    boolean hasNext();
    T next();


}
