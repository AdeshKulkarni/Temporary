package com.tcs.fincore.CommonRequestService.service;

import com.tcs.fincore.CommonRequestService.dto.CglDto;
import com.tcs.fincore.CommonRequestService.model.CglMaster;
import com.tcs.fincore.CommonRequestService.model.enums.AcClassification;
import com.tcs.fincore.CommonRequestService.repository.CglMasterRepository;
import lombok.RequiredArgsConstructor;
import lombok.extern.slf4j.Slf4j;
import org.springframework.stereotype.Service;

import java.util.List;
import java.util.Map;
import java.util.stream.Collectors;

@Service
@RequiredArgsConstructor
@Slf4j
public class CglService {

    private final CglMasterRepository cglMasterRepository;

    /**
     * Filters CGLs based on the provided account classification code.
     * @param classification The single character code ('A', 'L', etc.).
     * @return A list of CglDto.
     */
    public List<CglDto> findCglsByClassification(String classification) {
        log.info("Fetching CGLs for classification: {}", classification);
        List<CglMaster> cgls = cglMasterRepository.findByAcClassification(classification.toUpperCase());
        return mapToDto(cgls);
    }

    /**
     * Searches for CGLs where the CGL number contains the query string.
     * @param query The search query.
     * @return A list of CglDto.
     */
    public List<CglDto> searchCglsByNumber(String query) {
        log.info("Searching for CGLs with query: '{}'", query);
        List<CglMaster> cgls = cglMasterRepository.findByCglNumberContainingIgnoreCase(query);
        return mapToDto(cgls);
    }

    /**
     * Retrieves all possible account classifications to populate a dropdown on the front-end.
     * @return A map of { "code": "description" }.
     */
    public Map<Character, String> getAccountClassifications() {
        log.info("Fetching all account classifications.");
        return AcClassification.getClassificationMap();
    }

    /**
     * Helper method to map a list of CglMaster entities to a list of CglDto.
     */
    private List<CglDto> mapToDto(List<CglMaster> cgls) {
        return cgls.stream()
                .map(cgl -> new CglDto(cgl.getCglNumber(), cgl.getDescription()))
                .collect(Collectors.toList());
    }
}
