package com.fincore.JournalService.Service;

import com.fincore.JournalService.Models.BranchMaster;
import com.fincore.JournalService.Models.CglMaster;
import com.fincore.JournalService.Models.CurrencyMaster;
import com.fincore.JournalService.Repository.BranchMasterRepository;
import com.fincore.JournalService.Repository.CglMasterRepository;
import com.fincore.JournalService.Repository.CurrencyMasterRepository;
import lombok.RequiredArgsConstructor;
import lombok.extern.slf4j.Slf4j;

import org.apache.poi.ss.usermodel.Workbook;
import org.apache.poi.xssf.usermodel.XSSFWorkbook;
import org.apache.poi.ss.usermodel.Sheet;
import org.apache.poi.ss.usermodel.Row;
import org.apache.poi.ss.usermodel.Cell;
import org.apache.poi.ss.usermodel.CellStyle;
import org.apache.poi.ss.usermodel.Font;
import org.apache.poi.ss.usermodel.IndexedColors;
import org.apache.poi.ss.usermodel.DataFormatter;
import org.apache.poi.ss.usermodel.DateUtil;
import org.apache.poi.ss.usermodel.CellType; 

import org.springframework.stereotype.Service;
import org.springframework.web.multipart.MultipartFile;

import java.io.ByteArrayOutputStream;
import java.io.IOException;
import java.math.BigDecimal;
import java.math.RoundingMode;
import java.util.*;
import java.util.stream.Collectors;

@Service
@RequiredArgsConstructor
@Slf4j
public class JournalBulkValidationService {

    private final BranchMasterRepository branchRepo;
    private final CglMasterRepository cglRepo;
    private final CurrencyMasterRepository currencyRepo;

    private static class ExcelRowData {
        int rowIndex;
        String branch;
        String currency;
        String cgl;
        BigDecimal amount; 
        String txnType;
        String remarks;
        List<String> errors = new ArrayList<>();
    }

    public Map<String, Object> validateAndProcessExcel(MultipartFile file) throws IOException {
        
        // 1. Fetch Master Data
        Set<String> validBranches = branchRepo.findAll().stream()
                .map(BranchMaster::getCode).map(String::trim).collect(Collectors.toSet());
        
        Set<String> validCgls = cglRepo.findAll().stream()
                .map(CglMaster::getCglNumber).map(String::trim).collect(Collectors.toSet());
        
        Set<String> validCurrencies = currencyRepo.findAll().stream()
                .map(CurrencyMaster::getCurrencyCode).map(String::trim).collect(Collectors.toSet());

        List<ExcelRowData> parsedRows = new ArrayList<>();
        Workbook workbook = new XSSFWorkbook(file.getInputStream());
        Sheet sheet = workbook.getSheetAt(0);

        Map<String, BigDecimal> balanceMap = new HashMap<>();

        for (Row row : sheet) {
            if (row.getRowNum() == 0) continue; 
            if (isRowEmpty(row)) continue;

            ExcelRowData data = new ExcelRowData();
            data.rowIndex = row.getRowNum();

            // --- READ DATA ---
            data.branch = parseCode(row.getCell(0));     
            data.currency = getCellValueAsString(row.getCell(1)).toUpperCase().trim(); 
            data.cgl = parseCode(row.getCell(2));        
            
            String amountRaw = getCellValueAsString(row.getCell(3)).trim(); // Col D (Amount)
            String typeRaw = getCellValueAsString(row.getCell(4)).trim().toUpperCase();   // Col E (Type)
            data.remarks = getCellValueAsString(row.getCell(5));

            data.txnType = typeRaw; 

            // --- 1. MASTER VALIDATION ---
            if (!validBranches.contains(data.branch)) data.errors.add("Invalid Branch Code (" + data.branch + ")");
            if (!validCurrencies.contains(data.currency)) data.errors.add("Invalid Currency (" + data.currency + ")");
            if (!validCgls.contains(data.cgl)) data.errors.add("Invalid CGL (" + data.cgl + ")");

            // --- 2. LOGIC FOR CR / DR ---
            try {
                boolean isCredit = false;

                // STRICT CHECK: Does Column E contain "CR"?
                if (typeRaw.equals("CR") || typeRaw.startsWith("C")) {
                    isCredit = true;
                }
                
                // Cleanup Amount (Remove commas, spaces, letters if any)
                String cleanAmountStr = amountRaw.replaceAll("[^0-9.]", ""); 

                if(cleanAmountStr.isEmpty()) throw new NumberFormatException("Empty Amount");

                BigDecimal amt = new BigDecimal(cleanAmountStr);

                if (amt.compareTo(BigDecimal.ZERO) == 0) {
                    data.errors.add("Amount cannot be Zero");
                }
                
                // ASSIGN SIGN:
                // Credit (CR) = Negative Number
                // Debit (DR)  = Positive Number
                data.amount = isCredit ? amt.abs().negate() : amt.abs();

            } catch (Exception e) {
                data.errors.add("Invalid Amount Format");
                data.amount = BigDecimal.ZERO;
            }

            // --- 3. REMARK VALIDATION ---
            if (data.remarks != null && data.remarks.length() > 30) {
                data.errors.add("Remarks > 30 chars");
            }

            // --- 4. ACCUMULATE TOTALS ---
            if (validBranches.contains(data.branch) && validCurrencies.contains(data.currency)) {
                String key = data.branch + "_" + data.currency;
                balanceMap.put(key, balanceMap.getOrDefault(key, BigDecimal.ZERO).add(data.amount));
            }

            parsedRows.add(data);
        }

        // --- 5. CHECK BALANCING ---
        for (ExcelRowData data : parsedRows) {
            if (!validBranches.contains(data.branch) || !validCurrencies.contains(data.currency)) continue;

            String key = data.branch + "_" + data.currency;
            BigDecimal netTotal = balanceMap.getOrDefault(key, BigDecimal.ZERO);

            // Compare with 4 decimal precision
            if (netTotal.setScale(4, RoundingMode.HALF_UP).compareTo(BigDecimal.ZERO) != 0) {
                data.errors.add("Branch Total Mismatch (Diff: " + netTotal.toPlainString() + ")");
            }
        }

        long errorCount = parsedRows.stream().filter(r -> !r.errors.isEmpty()).count();

        Map<String, Object> result = new HashMap<>();
        
        if (errorCount > 0) {
            byte[] errorFile = generateErrorExcel(parsedRows);
            result.put("status", "ERROR");
            result.put("errorCount", errorCount);
            result.put("file", errorFile);
            result.put("message", "Validation failed. Found " + errorCount + " invalid rows.");
        } else {
            result.put("status", "SUCCESS");
            result.put("message", "Validation Successful! " + parsedRows.size() + " rows ready.");
        }

        workbook.close();
        return result;
    }

    private String parseCode(Cell cell) {
        if (cell == null) return "";
        String val = getCellValueAsString(cell).trim();
        if (val.endsWith(".0")) return val.substring(0, val.length() - 2);
        return val;
    }

    private byte[] generateErrorExcel(List<ExcelRowData> rows) throws IOException {
        Workbook workbook = new XSSFWorkbook();
        Sheet sheet = workbook.createSheet("Error Report");
        
        CellStyle headerStyle = workbook.createCellStyle();
        Font font = workbook.createFont();
        font.setBold(true);
        font.setColor(IndexedColors.RED.getIndex());
        headerStyle.setFont(font);

        Row header = sheet.createRow(0);
        String[] headers = {"Branch", "Currency", "CGL", "Amount", "TxnType", "Remarks", "ERRORS"};
        for (int i = 0; i < headers.length; i++) {
            Cell cell = header.createCell(i);
            cell.setCellValue(headers[i]);
            cell.setCellStyle(headerStyle);
        }

        int rowIdx = 1;
        for (ExcelRowData data : rows) {
            Row row = sheet.createRow(rowIdx++);
            row.createCell(0).setCellValue(data.branch);
            row.createCell(1).setCellValue(data.currency);
            row.createCell(2).setCellValue(data.cgl);
            row.createCell(3).setCellValue(data.amount.abs().toPlainString());
            
            // Display Type
            String typeDisplay = (data.amount.compareTo(BigDecimal.ZERO) < 0) ? "Credit" : "Debit";
            row.createCell(4).setCellValue(typeDisplay);
            row.createCell(5).setCellValue(data.remarks);
            
            if (!data.errors.isEmpty()) {
                Cell errorCell = row.createCell(6);
                errorCell.setCellValue(String.join("; ", data.errors));
                errorCell.setCellStyle(headerStyle);
            }
        }
        for(int i=0; i<7; i++) sheet.autoSizeColumn(i);
        ByteArrayOutputStream out = new ByteArrayOutputStream();
        workbook.write(out);
        workbook.close();
        return out.toByteArray();
    }

    private String getCellValueAsString(Cell cell) {
        if (cell == null) return "";
        if (cell.getCellType() == CellType.STRING) return cell.getStringCellValue().trim();
        if (cell.getCellType() == CellType.NUMERIC) {
            if (DateUtil.isCellDateFormatted(cell)) return cell.getDateCellValue().toString();
            return new DataFormatter().formatCellValue(cell).trim();
        }
        return "";
    }

    private boolean isRowEmpty(Row row) {
        if (row == null) return true;
        for (int c = row.getFirstCellNum(); c < row.getLastCellNum(); c++) {
            Cell cell = row.getCell(c);
            if (cell != null && cell.getCellType() != CellType.BLANK && !getCellValueAsString(cell).trim().isEmpty())
                return false;
        }
        return true;
    }
}
