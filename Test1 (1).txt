package com.tcs.fincore.MocRequestService.controller;

import com.fasterxml.jackson.core.JsonProcessingException;
import com.fincore.commonutilities.jwt.JwtUtil;
import com.tcs.fincore.MocRequestService.dto.CreateMocRequestDto;
import com.tcs.fincore.MocRequestService.dto.ProcessMocRequestDto;
import com.tcs.fincore.MocRequestService.model.MocRequest;
import com.tcs.fincore.MocRequestService.service.MocRequestService;
import jakarta.validation.Valid;
import lombok.RequiredArgsConstructor;
import lombok.extern.slf4j.Slf4j;
import org.springframework.http.HttpStatus;
import org.springframework.web.bind.annotation.*;
import java.util.List;

@RestController
@RequestMapping("/api/moc") // Good practice to group related endpoints
@RequiredArgsConstructor
@Slf4j
public class MocRequestController {

    private final MocRequestService mocRequestService;

    @PostMapping("/create-request")
    @ResponseStatus(HttpStatus.CREATED)
    public MocRequest createRequest(
            @Valid @RequestBody CreateMocRequestDto createRequestDto,
            @RequestHeader("Authorization") String token) throws JsonProcessingException {
        String userId = JwtUtil.getUserIdFromToken(token);
        log.info("Received MOC request creation call from user: {}", userId);
        return mocRequestService.createRequest(createRequestDto, userId);
    }

    @PatchMapping("/update-request")
    public MocRequest updateRequestStatus(
            @RequestHeader("Authorization") String token,
            @Valid @RequestBody ProcessMocRequestDto processRequestDto) throws JsonProcessingException {
        String executorId = JwtUtil.getUserIdFromToken(token);
        return mocRequestService.updateRequestStatus(processRequestDto, executorId).get();
    }

    @GetMapping("/my-requests")
    public List<MocRequest> getMyRequests(@RequestHeader("Authorization") String token) {
        String userId = JwtUtil.getUserIdFromToken(token);
        return mocRequestService.getMyRequests(userId);
    }

    @GetMapping("/pending-requests")
    public List<MocRequest> getPendingRequests(@RequestHeader("Authorization") String token) {
        return mocRequestService.getPendingRequests();
    }
}
