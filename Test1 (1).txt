const handleBodyChange = (content, delta, source, editor) => {
  if (source === "api") return;

  // 1. Block more than 4 consecutive enters
  const consecutiveEnterPattern = /(<p><br\s*\/?><\/p>){5,}/gi;
  if (consecutiveEnterPattern.test(content)) {
    setSbar({
      isOpen: true,
      Msg: "❌ You cannot add more than 4 consecutive line breaks.",
    });
    // Remove the last added <p><br></p>
    const cleaned = content.replace(/(<p><br\s*\/?><\/p>){5,}/gi, "<p><br></p><p><br></p><p><br></p><p><br></p>");
    setBodyContent(cleaned);
    return;
  }

  // 2. Block special characters (except allow enter, space, braces, and alphanum)
  // Only check the last character entered
  const textOnly = editor.getText();
  const lastDelta = delta.ops && delta.ops.length > 0 ? delta.ops[delta.ops.length - 1] : null;
  if (lastDelta && typeof lastDelta.insert === "string") {
    const inserted = lastDelta.insert;
    // Ignore if it's only newline(s)
    if (inserted.replace(/\n/g, '') !== '') {
      // If any char other than whitespace, word chars, or {}
      if (SPECIAL_CHAR_REGEX.test(inserted)) {
        setSbar({
          isOpen: true,
          Msg: "❌ Special characters are not allowed.",
        });
        return;
      }
    }
  }

  setBodyContent(editor.getHTML());
  setWordCount(countWord(editor.getHTML()));
};
