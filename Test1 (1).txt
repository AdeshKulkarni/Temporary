class Solution {

    public int SizeList(int n, List<List<Integer>> adj,
                         int node, List<Boolean> visit, int count) {

        visit.set(node, true);
        count = 1;

        for (int i = 0; i < adj.get(node).size(); i++) {

            int nabar = adj.get(node).get(i);

            if (!visit.get(nabar)) {
                count += SizeList(n, adj, nabar, visit, 0);
            }
        }

        return count;
    }

    public long countPairs(int n, int[][] edges) {

        // Build adjacency list
        List<List<Integer>> adj = new ArrayList<>(n);

        for (int i = 0; i < n; i++) {
            adj.add(new ArrayList<>());
        }

        for (int i = 0; i < edges.length; i++) {

            int src = edges[i][0];
            int dis = edges[i][1];

            adj.get(src).add(dis);
            adj.get(dis).add(src);
        }

        // Find component sizes
        List<Boolean> visit =
                new ArrayList<>(Collections.nCopies(n, false));

        List<Integer> sizes = new ArrayList<>();

        for (int i = 0; i < n; i++) {

            if (!visit.get(i)) {

                int size = SizeList(n, adj, i, visit, 0);
                sizes.add(size);
            }
        }

        // Count unreachable pairs
        long ans = 0;
        long remaining = n;

        for (int size : sizes) {

            remaining -= size;

            ans += (long) size * remaining;
        }

        return ans;
    }
}
