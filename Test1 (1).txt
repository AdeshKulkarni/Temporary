    @Override
    @Transactional
    public void cancelMyRequest(Long requestId, String userId) {
        // This is the original method for single-cancellation
        MocRequest request = mocRequestRepository.findById(requestId)
                .orElseThrow(() -> new ResourceNotFoundException("MOC Request with ID " + requestId + " not found."));

        if (!userId.equals(request.getCreatorId())) {
            log.warn("User {} attempted to cancel request {} owned by {}", userId, requestId, request.getCreatorId());
            throw new IllegalStateException("You are not authorized to cancel this request.");
        }

        if (request.getRequestStatus() != RequestStatus.PENDING) {
            log.warn("User {} attempted to cancel request {} which is already in status {}", userId, requestId, request.getRequestStatus());
            throw new IllegalStateException("Cannot cancel a request that has already been processed.");
        }

        mocRequestRepository.delete(request);
        log.info("User {} successfully canceled MOC request {}", userId, requestId);
    }

    // --- NEW METHOD ---
    @Override
    @Transactional
    public void cancelMyRequestsByJournalPrefix(String journalIdPrefix, String userId) {
        log.info("User {} attempting to cancel all requests with journal prefix: {}", userId, journalIdPrefix);

        // Find ALL requests with this prefix
        List<MocRequest> requestsToCancel = mocRequestRepository.findByJournalIdStartingWith(journalIdPrefix);

        if (requestsToCancel.isEmpty()) {
            throw new ResourceNotFoundException("No requests found with journal prefix: " + journalIdPrefix);
        }

        // Check permissions and status for ALL of them before deleting
        for (MocRequest request : requestsToCancel) {
            // Check 1: User must be the creator
            if (!userId.equals(request.getCreatorId())) {
                log.warn("User {} attempted to bulk-cancel requests owned by {}", userId, request.getCreatorId());
                throw new IllegalStateException("You are not authorized to cancel requests in this journal batch.");
            }
            // Check 2: All requests must be PENDING
            if (request.getRequestStatus() != RequestStatus.PENDING) {
                log.warn("User {} attempted to bulk-cancel request {} which is already in status {}", userId, request.getId(), request.getRequestStatus());
                throw new IllegalStateException("Cannot cancel requests that are not in PENDING status. Batch is already processed.");
            }
        }

        // All checks passed, delete them all
        mocRequestRepository.deleteAll(requestsToCancel);
        log.info("User {} successfully bulk-canceled {} requests for journal prefix {}", userId, requestsToCancel.size(), journalIdPrefix);
    }
