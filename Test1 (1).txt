import React, { useEffect, useMemo, useState, useCallback } from "react";
import { EditorContent, useEditor } from "@tiptap/react";
import StarterKit from "@tiptap/starter-kit";
import Placeholder from "@tiptap/extension-placeholder";
import TextStyle from "@tiptap/extension-text-style";
import Color from "@tiptap/extension-color";
import FontSize from "@tiptap/extension-font-size";
import {
  Button, Typography, Divider, Grid, Card, CardHeader, CardContent, Chip,
  Snackbar, Paper, List, ListItem, ListItemIcon, ListItemText, Box, Dialog, DialogTitle, DialogContent, IconButton
} from "@mui/material";
import EditIcon from "@mui/icons-material/Edit";
import PreviewIcon from "@mui/icons-material/Preview";
import RestartAltIcon from "@mui/icons-material/RestartAlt";
import SaveIcon from "@mui/icons-material/Save";
import CancelIcon from "@mui/icons-material/Cancel";
import CheckCircleOutlineIcon from "@mui/icons-material/CheckCircleOutline";
import CloseIcon from "@mui/icons-material/Close";

// ----- Variable List -----
const VARIABLE_LIST = [
  "{{DATE}}",
  "{{REF_NO}}",
  "{{FIRM_NAME}}",
  "{{FRN_NO}}",
  "{{FIRM_ADDR}}",
  "{{ASSIGNMENT_TYPE}}",
  "{{GSTN}}"
];

// Sample variable replacements for preview
const PREVIEW_MAP = {
  DATE: "xx/xx/xxxx", REF_NO: "9999xxx", FIRM_NAME: "xxxxxxxxxx",
  FRN_NO: "999999999", GSTN: "99xx99xx", FIRM_ADDR: "xxxxxxxx,xxxxxxxx", ASSIGNMENT_TYPE: "xxxxxxx"
};

const initialHeader = `<h4 style="margin:0;">Non-editable header will appear here</h4>`;
const initialBody = `<p>Editable body will appear here</p>`;

const MAX_WORDS = 2000;
const ALLOWED_CHARS_REGEX = /^[a-zA-Z0-9_\s.+|\\/*&,;!?'"()\[\]%{}\-\u0900-\u097F]*$/; // Adjust as needed

function EmailTemplateTiptap() {
  const [headerContent] = useState(initialHeader);
  const [bodyHtml, setBodyHtml] = useState(initialBody);
  const [isEditing, setIsEditing] = useState(false);
  const [snackbar, setSnackbar] = useState({ open: false, msg: "" });
  const [previewOpen, setPreviewOpen] = useState(false);
  const [previewHtml, setPreviewHtml] = useState("");
  const [wordCount, setWordCount] = useState(0);

  // ---- Tiptap editor ----
  const editor = useEditor({
    editable: isEditing,
    content: bodyHtml,
    extensions: [
      StarterKit,
      TextStyle,
      Color,
      FontSize,
      Placeholder.configure({
        placeholder: "Type the email body here...",
        emptyEditorClass: "is-editor-empty"
      }),
      // Custom Variable Extension Below
      VariableChipExtension({ onDuplicate: () => setSnackbar({ open: true, msg: "❌ Cannot add the same variable consecutively." }) }),
    ],
    onUpdate: ({ editor }) => {
      setBodyHtml(editor.getHTML());
      // Count words
      const text = editor.getText();
      const words = text.trim().split(/\s+/).filter(Boolean);
      setWordCount(words.length);
    }
  });

  // ---- Drag and drop (sidebar to editor) ----
  const handleDragStart = (e, label) => {
    e.dataTransfer.setData("variable", label);
  };

  useEffect(() => {
    if (!editor) return;
    // Handle drop event
    const dom = editor.view.dom;
    const handleDrop = (event) => {
      const variable = event.dataTransfer.getData("variable");
      if (isEditing && VARIABLE_LIST.includes(variable)) {
        let { from } = editor.state.selection;
        // Prevent consecutive duplicates
        const before = editor.state.doc.textBetween(Math.max(0, from - variable.length), from);
        if (before === variable) {
          setSnackbar({ open: true, msg: `❌ Cannot add the same variable consecutively.` });
          return;
        }
        editor.commands.insertContent(`<variable-chip>${variable}</variable-chip>`);
      }
    };
    dom.addEventListener("drop", handleDrop);
    return () => dom.removeEventListener("drop", handleDrop);
  }, [editor, isEditing]);

  // ---- Paste validation & blank line check ----
  useEffect(() => {
    if (!editor) return;
    editor.setOptions({
      handlePaste(view, event, slice) {
        const text = event.clipboardData.getData("text/plain");
        if (!ALLOWED_CHARS_REGEX.test(text)) {
          setSnackbar({ open: true, msg: "❌ Special characters are not allowed!" });
          return true; // block paste
        }
        // Prevent >3 consecutive blank lines
        if ((editor.getText().match(/(\n{4,})/) || []).length) {
          setSnackbar({ open: true, msg: "❌ You cannot enter more than 3 consecutive blank lines." });
          return true;
        }
        return false;
      }
    });
  }, [editor]);

  // ---- Validation on editor update ----
  useEffect(() => {
    if (!editor) return;
    // Validate word count
    if (wordCount > MAX_WORDS) {
      setSnackbar({ open: true, msg: "Word limit exceeded. Please reduce content." });
    }
  }, [editor, wordCount]);

  // ---- Preview logic ----
  const handlePreview = () => {
    // Replace <variable-chip> with PREVIEW_MAP values
    let html = bodyHtml.replace(/<variable-chip>({{[A-Z_]+}})<\/variable-chip>/g, (m, v) => {
      const key = v.replace(/{{|}}/g, "");
      return `<span style="background:#e3f2fd;padding:2px 4px;border-radius:2px;">${PREVIEW_MAP[key] || v}</span>`;
    });
    setPreviewHtml(`${headerContent}<hr/>${html}`);
    setPreviewOpen(true);
  };

  // ---- Save handler ----
  const handleSave = () => {
    setIsEditing(false);
    setSnackbar({ open: true, msg: "✅ Template saved successfully!" });
    setPreviewOpen(false);
    // Store or send : { headerContent, bodyHtml }
  };

  // ---- Reset handler ----
  const handleReset = () => {
    editor.commands.setContent(initialBody);
    setSnackbar({ open: true, msg: "✅ Template reset to default." });
    setIsEditing(false);
    setWordCount(0);
  };

  // ---- Copy protection for header ----
  const handleHeaderCopy = (e) => {
    e.preventDefault();
    setSnackbar({ open: true, msg: "❌ Copying from the header is not allowed!" });
  };

  return (
    <>
      <Typography gutterBottom variant="h5" sx={{ mb: 1 }}>
        Empanelment Intimation Letter Template (Tiptap)
      </Typography>
      <Divider sx={{ mb: 1 }} />

      <Box sx={{ p: 1, minHeight: "80%" }}>
        <Grid container spacing={3}>
          {/* Sidebar */}
          <Grid item xs={12} md={4}>
            <Card sx={{ mb: 3 }}>
              <CardHeader title="Variables" />
              <CardContent sx={{ display: "flex", flexWrap: "wrap", gap: 1 }}>
                {VARIABLE_LIST.map(label => (
                  <Chip
                    key={label}
                    label={label}
                    color="primary"
                    draggable={isEditing}
                    onDragStart={e => handleDragStart(e, label)}
                    sx={{
                      borderRadius: "0",
                      cursor: isEditing ? "move" : "not-allowed",
                      fontWeight: "medium",
                    }}
                  />
                ))}
              </CardContent>
            </Card>
            <Card>
              <CardHeader title="Instructions" />
              <CardContent>
                <List dense>
                  {[
                    "Special characters are not allowed.",
                    "Only use variables from the list above.",
                    "Header section is non-editable.",
                    "Click 'Edit' to modify the template body.",
                    "Maximum word limit is 2000 words.",
                  ].map(text => (
                    <ListItem key={text} disablePadding>
                      <ListItemIcon sx={{ minWidth: 32 }}>
                        <CheckCircleOutlineIcon color="primary" fontSize="small" />
                      </ListItemIcon>
                      <ListItemText primary={text} />
                    </ListItem>
                  ))}
                </List>
              </CardContent>
            </Card>
          </Grid>
          {/* Editor */}
          <Grid item xs={12} md={8}>
            <Card>
              <CardHeader title="Empanelment Intimation Letter Draft" />
              <CardContent>
                <Typography variant="subtitle1" sx={{ color: "text.secondary" }}>
                  Non Configurable
                </Typography>
                <Paper
                  variant="outlined"
                  sx={{
                    bgcolor: "#f5f5f5",
                    p: 1,
                    mb: 3,
                    border: "1px solid #e0e0e0",
                    borderRadius: 1,
                  }}
                  onCopy={handleHeaderCopy}
                >
                  <div
                    dangerouslySetInnerHTML={{ __html: headerContent }}
                    style={{ minHeight: 48, color: "#888", fontWeight: 500, fontSize: 16 }}
                  />
                </Paper>
                <Typography variant="subtitle1" sx={{ color: "text.secondary" }}>Configurable</Typography>
                <Paper variant="outlined" sx={{ minHeight: 180, p: 1, position: "relative" }}>
                  {/* Toolbar */}
                  <Box sx={{ display: "flex", gap: "4px", mb: 1 }}>
                    <Button
                      size="small"
                      variant={editor?.isActive("bold") ? "contained" : "outlined"}
                      onClick={() => editor?.chain().focus().toggleBold().run()}
                    >B</Button>
                    <Button
                      size="small"
                      variant={editor?.isActive("italic") ? "contained" : "outlined"}
                      onClick={() => editor?.chain().focus().toggleItalic().run()}
                    >I</Button>
                    <Button
                      size="small"
                      variant={editor?.isActive("underline") ? "contained" : "outlined"}
                      onClick={() => editor?.chain().focus().toggleUnderline().run()}
                    >U</Button>
                    {/* Add more toolbar buttons as needed */}
                  </Box>
                  <EditorContent editor={editor} />
                  <Box
                    sx={{
                      display: "flex",
                      justifyContent: "space-between",
                      alignItems: "center",
                      mt: 2,
                      mb: 1,
                    }}
                  >
                    <Typography variant="body2" color="textSecondary">
                      Word Count: {wordCount} / {MAX_WORDS}
                    </Typography>
                    <Box sx={{ display: "flex", gap: 1 }}>
                      {!isEditing ? (
                        <Button
                          variant="contained"
                          startIcon={<EditIcon />}
                          onClick={() => setIsEditing(true)}
                        >Edit</Button>
                      ) : (
                        <Button
                          variant="outlined"
                          color="error"
                          startIcon={<CancelIcon />}
                          onClick={() => setIsEditing(false)}
                        >Cancel</Button>
                      )}
                      <Button
                        variant="outlined"
                        color="info"
                        startIcon={<PreviewIcon />}
                        onClick={handlePreview}
                      >Preview</Button>
                      <Button
                        variant="outlined"
                        color="secondary"
                        startIcon={<RestartAltIcon />}
                        onClick={handleReset}
                      >Reset</Button>
                    </Box>
                  </Box>
                </Paper>
              </CardContent>
            </Card>
          </Grid>
        </Grid>
      </Box>
      {/* Preview Dialog */}
      <Dialog open={previewOpen} onClose={() => setPreviewOpen(false)} fullWidth maxWidth="md">
        <DialogTitle>
          Intimation Letter Draft Preview
          <IconButton
            aria-label="close"
            onClick={() => setPreviewOpen(false)}
            sx={{
              position: "absolute",
              right: 8,
              top: 8,
              color: (theme) => theme.palette.grey[500],
            }}
          >
            <CloseIcon />
          </IconButton>
        </DialogTitle>
        <DialogContent>
          <Paper
            elevation={3}
            dangerouslySetInnerHTML={{ __html: previewHtml }}
            sx={{
              width: "210mm",
              minHeight: "297mm",
              margin: "2rem auto",
              padding: "20mm",
              boxSizing: "border-box",
              backgroundColor: "#fff",
              overflow: "auto",
              overflowWrap: "break-word",
            }}
          />
        </DialogContent>
        <Box sx={{ p: 2, borderTop: "1px solid #eee", display: "flex", gap: 2 }}>
          <Button onClick={() => setPreviewOpen(false)} color="secondary">
            Cancel
          </Button>
          <Button
            autoFocus
            onClick={handleSave}
            variant="contained"
            startIcon={<SaveIcon />}
          >
            Save Draft
          </Button>
        </Box>
      </Dialog>
      <Snackbar
        open={snackbar.open}
        autoHideDuration={6000}
        onClose={() => setSnackbar({ open: false, msg: "" })}
        message={snackbar.msg}
        anchorOrigin={{ vertical: "bottom", horizontal: "right" }}
      />
    </>
  );
}

export default EmailTemplateTiptap;

// --- Custom Tiptap Extension for Variable Chips ---
import { Node, mergeAttributes } from "@tiptap/core";

function VariableChipExtension({ onDuplicate }) {
  return Node.create({
    name: "variableChip",
    group: "inline",
    inline: true,
    selectable: true,
    atom: true,
    parseHTML: () => [{ tag: "variable-chip" }],
    renderHTML: ({ node, HTMLAttributes }) => [
      "variable-chip",
      mergeAttributes(HTMLAttributes, { style: "background:#A9A9A9;color:black;padding:4px 18px 4px 4px;border-radius:3px;display:inline-block;position:relative;margin:0 4px;" }),
      node.attrs.value
        ? [
            "span",
            { style: "pointer-events:none;" },
            node.attrs.value,
          ]
        : "",
      [
        "span",
        {
          style: "position:absolute;right:4px;top:50%;transform:translateY(-50%);font-weight:bold;font-family:monospace;font-size:14px;cursor:pointer;",
          onclick: "this.parentElement.remove()" // Simple way to add 'X' remove for proof of concept — in real app, use ProseMirror transactions!
        },
        "X",
      ],
    ],
    addAttributes() {
      return { value: { default: "" } };
    },
    addKeyboardShortcuts() {
      return {
        Backspace: () => {
          // Custom logic for deleting variable chip if selection is on it (optional)
          return false;
        }
      };
    },
    addCommands() {
      return {
        insertVariable:
          (value) =>
          ({ chain }) => {
            // Prevent consecutive duplicates
            // (Handle outside in handleDrop for simplicity)
            return chain()
              .insertContent(`<variable-chip>${value}</variable-chip>`)
              .run();
          }
      };
    }
  });
}
