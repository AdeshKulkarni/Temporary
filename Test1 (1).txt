public class JournalRequestServiceImpl implements JournalRequestService {

    private final JournalRequestRepository journalRequestRepository;
    private final GlTransactionRepository glTransactionRepository; 
    private final ObjectMapper objectMapper;
    private final SequenceService sequenceService;
    
    private final NotificationWriterService notificationWriterService;
    private final PermissionConfigService permissionConfigService;
    
    private static final String EVENT_SOURCE = "JournalService";
    private static final String JOURNAL_AUTH_TYPE = "JOURNAL_AUTH"; 
    private static final String JOURNAL_POSTING_URL = "/Journal-Posting-Status"; 

    // ... (Keep existing createBatchRequest, updateRequestStatus, etc.) ...

    // --- OPTIMIZED BULK CREATION METHOD ---
    @Override
    @Transactional
    public String createBulkBatchRequest(BatchRequestDto batchDto, String creatorId, Integer creatorRole) throws JsonProcessingException {
        
        String batchId = sequenceService.getNextBatchId();
        log.info("Processing BULK Batch ID: {}", batchId);
       
        LocalDate postingDate = LocalDate.now();
        List<JournalRequest> requestsToSave = new ArrayList<>();
        
        String commonRemarks = (batchDto.getCommonBatchRemarks() != null && !batchDto.getCommonBatchRemarks().isEmpty()) 
                                ? batchDto.getCommonBatchRemarks() 
                                : "Bulk Upload Batch " + batchId;

        // Group rows by Branch+Currency to ensure sequential Journal IDs
        Map<String, List<BatchRequestDto.JournalRequestRow>> groupedRows = batchDto.getRows().stream()
                .collect(Collectors.groupingBy(row -> row.getBranch() + ":" + row.getCurrency()));

        for (Map.Entry<String, List<BatchRequestDto.JournalRequestRow>> entry : groupedRows.entrySet()) {
            
            String journalIdPrefix = sequenceService.getNextJournalIdPrefix();
            int journalSequence = 1;
            List<BatchRequestDto.JournalRequestRow> rowsInGroup = entry.getValue();
            
            for (BatchRequestDto.JournalRequestRow rowDto : rowsInGroup) {
                
                String journalId = String.format("%s-%03d", journalIdPrefix, journalSequence);
                
                CreateJournalRequestDto payloadDto = new CreateJournalRequestDto();
                payloadDto.setPDate(postingDate); 
                payloadDto.setChangeType(rowDto.getChangeType());
                payloadDto.setMasterJournalId(rowDto.getMasterJournalId());
                payloadDto.setBranch(rowDto.getBranch());
                payloadDto.setCurrency(rowDto.getCurrency());
                payloadDto.setCgl(rowDto.getCgl());
                payloadDto.setAmount(rowDto.getAmount()); 
                payloadDto.setProductType(rowDto.getProductType());
                payloadDto.setRemarks(rowDto.getRemarks());
                payloadDto.setArFlag(rowDto.getArFlag() != null ? rowDto.getArFlag() : "A");
                payloadDto.setAcClassification(rowDto.getAcClassification() != null ? rowDto.getAcClassification() : "A");
                
                payloadDto.setBatchId(batchId); 
                payloadDto.setJournalId(journalId); 
                payloadDto.setCommonBatchRemarks(commonRemarks);
                payloadDto.setTransactionCount(journalSequence); 
                
                JournalRequest request = new JournalRequest();
                request.setCreatorId(creatorId);
                request.setCreatorRole(creatorRole);
                request.setChangeType(rowDto.getChangeType()); 
                request.setRequestStatus(RequestStatus.PENDING);
                request.setBatchId(batchId);
                request.setJournalId(journalId);
                request.setCommonBatchRemarks(commonRemarks);
                request.setPayload(objectMapper.writeValueAsString(payloadDto));

                requestsToSave.add(request);
                journalSequence++; 
            }
        }

        // 1. SAVE ALL (Database Operation)
        journalRequestRepository.saveAll(requestsToSave);
        
        // 2. SEND NOTIFICATION (Single Notification for the whole batch)
        try {
            NotificationConfigDto config = permissionConfigService.getConfig(JOURNAL_AUTH_TYPE);
            String message = String.format("Bulk Journal Batch %s (%d rows) by %s pending approval.", 
                                           batchId, requestsToSave.size(), creatorId);

            notificationWriterService.createNotification(
                null,                   
                config.getTargetRoles(),
                message,
                config.getTargetUrl(),  
                batchId,                
                EVENT_SOURCE
            );
            log.info("Sent single notification for Bulk Batch ID: {}", batchId);
        } catch (Exception e) {
            log.error("Failed to create notification for Bulk Batch: {}", batchId, e);
        }
   
        return batchId; 
    }

    // ... (Keep the rest of your methods: updateRequestStatus, getMyRequests, etc.) ...
}

import React, { useState, useRef } from "react";
import {
  Box,
  Typography,
  Paper,
  Grid,
  TextField,
  Button,
  CircularProgress,
  Chip,
  Divider,
  Alert,
  Stack,
} from "@mui/material";
import {
  CloudUpload as CloudUploadIcon,
  RestartAlt as RestartAltIcon,
  Description as DescriptionIcon,
  CheckCircle as CheckCircleIcon,
  Error as ErrorIcon,
  Download as DownloadIcon,
  Send as SendIcon,
} from "@mui/icons-material";

// Import DataGrid
import { DataGrid } from "@mui/x-data-grid";

import { 
  mainContainerStyle, 
  uploadBoxStyle, 
  controlPanelStyle, 
  iconStyle, 
  headerTextStyle, 
  subTextStyle 
} from "./JournalBulkStyle";

import useApi from "../../hooks/useApi";
import useCustomSnackbar from "../../utils/useCustomSnackbar";

const JournalBulkUpload = () => {
  const { callApi, loading } = useApi(); 
  const showSnackBar = useCustomSnackbar();
  
  const [selectedDate, setSelectedDate] = useState("");
  const [file, setFile] = useState(null);
  const [uiError, setUiError] = useState("");
  const [errorCount, setErrorCount] = useState(0);
  
  const [successFile, setSuccessFile] = useState(null);
  const [previewRows, setPreviewRows] = useState([]);
  const [isSubmitting, setIsSubmitting] = useState(false);

  const fileInputRef = useRef(null);

  const downloadBase64File = (base64String, fileName, mimeType) => {
    try {
      const byteCharacters = atob(base64String);
      const byteNumbers = new Array(byteCharacters.length);
      for (let i = 0; i < byteCharacters.length; i++) {
        byteNumbers[i] = byteCharacters.charCodeAt(i);
      }
      const byteArray = new Uint8Array(byteNumbers);
      const blob = new Blob([byteArray], { type: mimeType });
      const url = window.URL.createObjectURL(blob);
      const link = document.createElement("a");
      link.href = url;
      link.setAttribute("download", fileName);
      document.body.appendChild(link);
      link.click();
      link.parentNode.removeChild(link);
      window.URL.revokeObjectURL(url);
    } catch (e) {
      console.error("Failed to download", e);
      showSnackBar("Failed to download file.", "error");
    }
  };

  const handleDateChange = (e) => {
    setSelectedDate(e.target.value);
    setUiError("");
    setSuccessFile(null);
    setPreviewRows([]);
  };

  const handleFileChange = (e) => {
    const selectedFile = e.target.files[0];
    if (selectedFile) {
      const fileType = selectedFile.name.split(".").pop().toLowerCase();
      if (fileType !== "xlsx" && fileType !== "xls") {
        const msg = "Invalid file type. Please upload .xlsx or .xls";
        setUiError(msg);
        showSnackBar(msg, "error");
        setFile(null);
        return;
      }
      setFile(selectedFile);
      setUiError("");
      setErrorCount(0);
      setSuccessFile(null);
      setPreviewRows([]);
    }
  };

  const resetForm = () => {
    setFile(null);
    setSelectedDate("");
    setUiError("");
    setErrorCount(0);
    setSuccessFile(null);
    setPreviewRows([]);
    setIsSubmitting(false);
    if (fileInputRef.current) fileInputRef.current.value = "";
  };

  const handleValidation = async () => {
    if (!selectedDate || !file) {
      const msg = "Please select a Date and upload a File.";
      setUiError(msg);
      showSnackBar(msg, "warning");
      return;
    }

    setUiError("");
    setErrorCount(0);
    setSuccessFile(null);
    setPreviewRows([]);

    const formData = new FormData();
    formData.append("file", file);
    formData.append("postingDate", selectedDate);

    try {
      const response = await callApi("JS/journals/bulk-validate", formData, "POST", "json", "multipart/form-data");

      if (response && response.status === "ERROR") {
        const count = response.errorCount || 0;
        setErrorCount(count);
        const msg = `Validation Failed with ${count} errors. Downloading report...`;
        setUiError(msg);
        showSnackBar(msg, "error");

        if (response.errorReportBase64) {
          downloadBase64File(response.errorReportBase64, `Error_Report.xlsx`, "application/vnd.openxmlformats-officedocument.spreadsheetml.sheet");
        }
      } else if (response && response.status === "SUCCESS") {
        showSnackBar(response.message, "success");
        if (response.successFileBase64) setSuccessFile(response.successFileBase64);
        if (response.previewData) setPreviewRows(response.previewData); 
      } 
    } catch (err) {
      console.error("Validation Error:", err);
    }
  };

  const handleSubmitBatch = async () => {
    if (previewRows.length === 0) {
      showSnackBar("No validated data to submit.", "warning");
      return;
    }

    setIsSubmitting(true);

    try {
      const payloadRows = previewRows.map((row) => {
        let amt = parseFloat(row.amount.replace(/,/g, '')); 
        if (row.txnType === "Credit") {
          amt = -Math.abs(amt);
        } else {
          amt = Math.abs(amt);
        }

        return {
          changeType: "ADD",
          branch: row.branch,
          currency: row.currency,
          cgl: row.cgl,
          amount: amt,
          productType: row.productCode || "", 
          remarks: row.remarks || "",
          arFlag: "A", 
          acClassification: "A" 
        };
      });

      const finalPayload = {
        commonBatchRemarks: `Bulk Upload - ${selectedDate}`, 
        rows: payloadRows
      };

      const response = await callApi("JS/journals/create-bulk-batch", finalPayload, "POST");

      if (response && response.status === "SUCCESS") {
        showSnackBar(`Batch Created Successfully! Batch ID: ${response.batchId}`, "success");
        resetForm();
      } 

    } catch (error) {
      console.error("Batch Creation Failed:", error);
      showSnackBar("Failed to create batch.", "error");
    } finally {
      setIsSubmitting(false);
    }
  };

  // --- Grid Columns ---
  const columns = [
    { field: "branch", headerName: "Branch", flex: 1, align: "center", headerAlign: "center" },
    { field: "currency", headerName: "Currency", width: 100, align: "center", headerAlign: "center" },
    { field: "cgl", headerName: "CGL", flex: 1.5, align: "center", headerAlign: "center" },
    { field: "productCode", headerName: "Product", width: 120, align: "center", headerAlign: "center", renderCell: (p) => p.value || "-" },
    { 
      field: "amount", headerName: "Amount", flex: 1.5, minWidth: 150, align: "right", headerAlign: "center",
      valueFormatter: (params) => {
        const val = (params && typeof params === 'object' && 'value' in params) ? params.value : params;
        if (!val) return "0.00";
        const parts = String(val).split(".");
        parts[0] = parts[0].replace(/\B(?=(\d{3})+(?!\d))/g, ",");
        return parts.join(".");
      }
    },
    { 
      field: "txnType", headerName: "Type", width: 100, align: "center", headerAlign: "center",
      renderCell: (params) => (
        <Chip 
          label={params.value} 
          color={params.value === "Credit" ? "error" : "success"} 
          size="small" 
          variant="filled"
          sx={{ fontWeight: "bold", width: "80px" }}
        />
      )
    },
    { field: "remarks", headerName: "Remarks", flex: 2, headerAlign: "center" },
  ];

  return (
    <Box sx={mainContainerStyle}>
      <Paper elevation={0} sx={controlPanelStyle}>
        
        {/* Header Section */}
        <Box sx={{ mb: 4 }}>
            <Typography variant="h5" sx={{ fontWeight: 700, color: "#1a237e" }}>
                Bulk Journal Upload
            </Typography>
            <Typography variant="body2" sx={{ color: "#666", mt: 1 }}>
                Upload your journal entries via Excel. Validate data, preview, and submit in bulk.
            </Typography>
        </Box>

        <Grid container spacing={4}>
          {/* LEFT SIDE: CONTROLS */}
          <Grid item xs={12} md={4}>
            <Stack spacing={3}>
                <Box>
                    <Typography variant="subtitle2" sx={{ mb: 1, fontWeight: 600 }}>1. Posting Date</Typography>
                    <TextField 
                        type="date" 
                        fullWidth 
                        size="medium" 
                        value={selectedDate} 
                        onChange={handleDateChange} 
                        InputLabelProps={{ shrink: true }} 
                    />
                </Box>

                <Box>
                    <Typography variant="subtitle2" sx={{ mb: 1, fontWeight: 600 }}>2. Actions</Typography>
                    <Stack spacing={2}>
                        <Button 
                            variant="contained" 
                            color="primary" 
                            size="large"
                            onClick={handleValidation} 
                            disabled={loading || !file || !selectedDate || isSubmitting} 
                            startIcon={loading ? <CircularProgress size={20} color="inherit" /> : <CheckCircleIcon />}
                            fullWidth
                            sx={{ py: 1.5, textTransform: "none", fontSize: "1rem" }}
                        >
                            {loading ? "Validating File..." : "Validate & Process"}
                        </Button>
                        <Button 
                            variant="outlined" 
                            color="error" 
                            onClick={resetForm} 
                            startIcon={<RestartAltIcon />} 
                            disabled={isSubmitting}
                            fullWidth
                            sx={{ textTransform: "none" }}
                        >
                            Reset Form
                        </Button>
                    </Stack>
                </Box>
            </Stack>
          </Grid>

          {/* RIGHT SIDE: BIG UPLOAD BOX */}
          <Grid item xs={12} md={8}>
            <Typography variant="subtitle2" sx={{ mb: 1, fontWeight: 600 }}>3. Upload File</Typography>
            <Box sx={uploadBoxStyle} onClick={() => fileInputRef.current.click()}>
              <input type="file" hidden accept=".xlsx, .xls" onChange={handleFileChange} ref={fileInputRef} />
              
              {file ? (
                <>
                  <DescriptionIcon sx={{ fontSize: 80, color: "#2e7d32", mb: 2 }} />
                  <Typography variant="h6" sx={headerTextStyle}>{file.name}</Typography>
                  <Chip label="Ready to Validate" color="success" variant="outlined" size="small" />
                  <Typography variant="caption" sx={{ mt: 2, color: "#999" }}>Click to replace file</Typography>
                </>
              ) : (
                <>
                  <CloudUploadIcon sx={iconStyle} />
                  <Typography variant="h6" sx={headerTextStyle}>Drag & Drop or Click to Upload</Typography>
                  <Typography sx={subTextStyle}>Supported formats: .xlsx, .xls</Typography>
                  <Typography variant="caption" sx={{ mt: 2, color: "#aaa" }}>Max file size: 10MB</Typography>
                </>
              )}
            </Box>
          </Grid>
        </Grid>

        {/* FEEDBACK & ACTIONS */}
        <Box sx={{ mt: 4 }}>
          {uiError && (
            <Alert severity="error" sx={{ mb: 2 }} action={errorCount > 0 && <Chip label={`${errorCount} Errors`} color="error" size="small"/>}>
              <Typography variant="subtitle2" fontWeight="bold">Validation Failed</Typography>
              <Typography variant="body2">{uiError}</Typography>
            </Alert>
          )}

          {successFile && !uiError && (
             <Paper elevation={0} sx={{ p: 3, backgroundColor: "#e8f5e9", border: "1px solid #c8e6c9", borderRadius: 2 }}>
                <Stack direction={{ xs: 'column', md: 'row' }} justifyContent="space-between" alignItems="center" spacing={2}>
                    <Box>
                        <Stack direction="row" alignItems="center" spacing={1} sx={{mb: 0.5}}>
                            <CheckCircleIcon color="success" />
                            <Typography variant="h6" fontWeight="bold" color="success.main">Validation Successful!</Typography>
                        </Stack>
                        <Typography variant="body2" color="text.secondary">
                            {previewRows.length} rows are valid and ready for submission.
                        </Typography>
                    </Box>
                    <Stack direction="row" spacing={2}>
                        <Button 
                            variant="outlined" 
                            color="success" 
                            startIcon={<DownloadIcon />}
                            disabled={isSubmitting}
                            onClick={() => downloadBase64File(successFile, `Processed_Journal_${selectedDate}.csv`, "text/csv")}
                        >
                            Download CSV
                        </Button>
                        
                        <Button 
                            variant="contained" 
                            color="success" 
                            size="large"
                            startIcon={isSubmitting ? <CircularProgress size={20} color="inherit" /> : <SendIcon />}
                            onClick={handleSubmitBatch}
                            disabled={isSubmitting}
                            sx={{ px: 4, fontWeight: "bold" }}
                        >
                            {isSubmitting ? "Creating Batch..." : "Create Batch"}
                        </Button>
                    </Stack>
                </Stack>
             </Paper>
          )}
        </Box>
      </Paper>
      
      {/* DATA GRID PREVIEW */}
      {previewRows.length > 0 && !uiError && (
        <Paper elevation={2} sx={{ p: 0, mt: 3, height: 600, width: '100%', borderRadius: 3, overflow: "hidden" }}>
            <Box sx={{ p: 2, borderBottom: "1px solid #eee", backgroundColor: "#fafafa" }}>
                <Typography variant="h6" sx={{ fontWeight: 600, color: "#333" }}>Data Preview</Typography>
            </Box>
            <DataGrid
                rows={previewRows}
                columns={columns}
                pageSize={10}
                rowsPerPageOptions={[10, 25, 50, 100]}
                disableSelectionOnClick
                density="comfortable"
