class Solution {

    public int helper(int[] nums , int index , int sum , int[] memo) {

        if(index >= nums.length) {
            return sum;
        }

        if(memo[index] != -1) {
            return memo[index];
        }

        sum += nums[index];

        int ans1 = helper(nums , index + 2 , sum , memo);
        
        int ans2 = helper(nums , index + 1, sum - nums[index] , memo);

        memo[index] = Math.max(ans1 , ans2);

        return memo[index];

        
    }
    public int rob(int[] nums) {

        int[] memo = new int[nums.length + 1];

        Arrays.fill(memo , -1);


        return Math.max(helper(nums , 0 , 0 , memo) , helper(nums , 1 , 0 , memo));
        
    }
}
