class Solution {

    TreeNode ans = null;

    public int Helper(TreeNode node, TreeNode p, TreeNode q) {
        if (node == null) {
            return 0;
        }

        int left = Helper(node.left, p, q);
        int right = Helper(node.right, p, q);

        int self = 0;
        if (node == p || node == q) {
            self = 1;
        }

        int total = left + right + self;

        if (total == 2 && ans == null) {
            ans = node;
        }

        return total;
    }

    public TreeNode lcaDeepestLeaves(TreeNode root) {
        // FIX 1: Always reset your global/instance variable at the start!
        ans = null; 

        Queue<TreeNode> queue = new LinkedList<>();
        queue.offer(root);
        List<TreeNode> lastLevel = new ArrayList<>();

        while (!queue.isEmpty()) {
            int levelSize = queue.size();
            lastLevel = new ArrayList<>(); 

            for (int i = 0; i < levelSize; i++) {
                TreeNode node = queue.poll();
                lastLevel.add(node);

                if (node.left != null)
                    queue.offer(node.left);
                if (node.right != null)
                    queue.offer(node.right);
            }
        }

        // FIX 2: Handle the edge case where there is only 1 deepest leaf
        if (lastLevel.size() == 1) {
            return lastLevel.get(0);
        }

        // FIX 3: Grab the FIRST node and the LAST node of the level
        if (lastLevel.size() >= 2) {
            TreeNode p = lastLevel.get(0); 
            TreeNode q = lastLevel.get(lastLevel.size() - 1); 
            
            Helper(root, p, q);
        }

        return ans;
    }
}
