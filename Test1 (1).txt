package com.tcs.fincore.CommonRequestService.model.master;

import jakarta.persistence.*;
import lombok.Data;
import java.math.BigDecimal;
import java.time.LocalDate;

@Entity
@Table(name = "FN_MOC")
@Data
public class FnMoc {

    @Id
    @GeneratedValue(strategy = GenerationType.IDENTITY)
    private Long id; // Master table should have its own primary key

    @Column(name = "TMMOC_PDATE")
    private LocalDate tmmocPdate;

    @Column(name = "TMMOC_BRANCH", length = 50)
    private String tmmocBranch;

    @Column(name = "CURRENCY", length = 3)
    private String currency;

    @Column(name = "TMMOC_CGL", length = 50)
    private String tmmocCgl;

    @Column(name = "TMMOC_AMT")
    private BigDecimal tmmocAmt;

    @Column(name = "TXN_TYPE", length = 20)
    private String txnType;

    @Column(name = "PRODUCT_TYPE", length = 20)
    private String productType;
    
    @Column(name = "REMARKS", length = 40)
    private String remarks;

    @Column(name = "AR_FLAG", length = 1)
    private String arFlag;
}
