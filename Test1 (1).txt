import { Box, Grid, Typography, keyframes } from '@mui/material';

// Define keyframes to sweep a "spotlight" and then switch to a full gradient
const sweepAndFinalize = keyframes`
  /* The sweep animation happens from 0% to 99% */
  0% {
    transform: rotate(0deg);
    /* Start with the "spotlight" gradient */
    background: conic-gradient(
      from 180deg at 50% 50%,
      transparent 0%,
      #00aaff 30%, /* A nice starting blue */
      transparent 60%
    );
  }
  99% {
    transform: rotate(360deg);
    /* End the rotation with the same "spotlight" */
    background: conic-gradient(
      from 180deg at 50% 50%,
      transparent 0%,
      #00aaff 30%, 
      transparent 60%
    );
  }
  /* At 100%, instantly switch to the final, static, multi-colored border */
  100% {
    transform: rotate(360deg);
    background: conic-gradient(
      from 180deg at 50% 50%,
      rgba(0, 170, 255, 1),   /* Blue */
      rgba(255, 100, 100, 1), /* Pink/Red */
      rgba(255, 200, 50, 1),  /* Yellow/Gold */
      rgba(0, 170, 255, 1)    /* Looping back to Blue */
    );
  }
`;

function AnimatedBorderBox() {
  return (
    <Grid item xs={12} sm={5} md={4}>
      <Box
        sx={{
          position: 'relative',
          borderRadius: 1,
          p: 1,
          overflow: 'hidden',

          '&::before': {
            content: '""',
            position: 'absolute',
            inset: -20,
            zIndex: 0,
            
            // --- MODIFICATION START ---
            // Use the new animation. It runs once, feels smooth, and 'forwards'
            // makes it hold the style from the 100% keyframe when it's done.
            animation: `${sweepAndFinalize} 2.5s ease-out forwards`,
            // --- MODIFICATION END ---
          },
          
          '&::after': {
            content: '""',
            position: 'absolute',
            inset: '1px',
            background: (theme) => theme.palette.background.paper,
            borderRadius: 'inherit',
            zIndex: 1,
          },
        }}
      >
        <Box
          className="inner-content"
          sx={{
            position: 'relative',
            zIndex: 2,
            borderRadius: 'inherit',
            display: 'flex',
            alignItems: 'center',
            p: 1,
            backgroundColor: 'background.paper',
          }}
        >
          <Typography variant="body1" color="text.secondary">
            Please Note: All fields are required for successful user creation.
          </Typography>
        </Box>
      </Box>
    </Grid>
  );
}

export default AnimatedBorderBox;
