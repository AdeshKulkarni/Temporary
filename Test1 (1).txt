import static com.tcs.fincore.LLD.Drone.PayloadType.*;

public class DroneFactory {

    public FlightPath GetFlightObject(PayloadType payload , Wheather wheather) {

        if(payload == MEDICAL) {
            return new Medical();
        }
        else if(payload == FRAGILE) {
            return new Fragile();
        }

        // we are not giving imp to what Algo what whether just that if wheather is not good then in that case we cant go with the standard one
        else if(payload == STANDARD && (wheather == Wheather.RAIN || wheather == Wheather.WINDY)) {
            return new Fragile();
        }
        else {
            return new Standard();
        }
    }
}
