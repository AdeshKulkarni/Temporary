class Solution {
    boolean CanOutOfBox = false;
    public boolean Valid(int i, int j, int n, int m) {
    if (i < 0 || i >= n || j < 0 || j >= m) {
        return false;
    }   

    return true;
  }   

  public boolean Valid2(int i, int j, int n, int m) {
     if (i == 0 || i == n - 1 || j == 0 || j == m - 1) {
        return false;
    }

    return true;
  }

    public int dfs(int[][] grid , int i, int j, int m, int n, boolean[][] visited) {

        visited[i][j] = true;
        int count = 1;
        
    
        int[] x = {-1, 1, 0, 0};
        int[] y = {0, 0, -1, 1};

        for (int k = 0; k < 4; k++) {

            int row = i + x[k];
            int col = j + y[k];

            if(Valid2(row , col , n , m)) {
                hasOutOfBox = true;
            }           

           
            if (Valid(row, col, n, m)
                    && visited[row][col] == false
                    && grid[row][col] == 1) {

               count += dfs(grid, row, col, m, n, visited);
            }
        }
        D.p(count);
        return count;
    }
    public int numEnclaves(int[][] grid) {
        
        int n = grid.length;
        int m = grid[0].length;
        int res = 0;

        boolean[][] visited = new boolean[n][m];

        for(int i=0; i<n; i++) {
            for(int j=0; j<m; j++) {

               

               if(grid[i][j] == 1 && visited[i][j] == false &&Valid2(i , j , n , m)) {
                 if(hasOutOfBox == false) {
                     res += dfs(grid , i , j , m , n , visited);
                 }
               }
            }
        }

        return res;

        
    }
}
