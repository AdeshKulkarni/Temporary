class Solution {

    public void Helper(TreeNode root, int TotalSum, int Currentsum, List<Integer> ans) {

        if (root == null) {
            return;
        }

        Currentsum = Currentsum * 10 + root.val;

        if (root.left == null && root.right == null) {
            TotalSum += Currentsum;
            ans.add(TotalSum);
        }

        Helper(root.left, TotalSum, Currentsum, ans);
        Helper(root.right, TotalSum, Currentsum, ans);

    }

    public int sumNumbers(TreeNode root) {

        List<Integer> ans = new ArrayList<>();

        int sum = 0;
        Helper(root, 0, 0, ans);

        for(int i = 0; i < ans.size(); i++) {

            sum += ans.get(i);

        }

        return sum;

    }
}
