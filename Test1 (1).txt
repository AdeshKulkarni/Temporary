class Solution {

    public long Helper(int[] nums, int x, boolean isPrevEven, int index, Long[][] memo) {
        if (index >= nums.length) {
            return 0;
        }

        int p = isPrevEven ? 1 : 0; // map boolean to column 0 or 1
        if (memo[index][p] != null) {
            return memo[index][p];
        }

        boolean isCurrEven = (nums[index] % 2 == 0);
        long value = (isCurrEven == isPrevEven) ? nums[index] : nums[index] - x;

        long ans1 = value + Helper(nums, x, isCurrEven, index + 1, memo); // take
        long ans2 = Helper(nums, x, isPrevEven, index + 1, memo);         // skip

        return memo[index][p] = Math.max(ans1, ans2);
    }

    public long maxScore(int[] nums, int x) {
        Long[][] memo = new Long[nums.length][2];
        return nums[0] + Helper(nums, x, nums[0] % 2 == 0, 1, memo);
    }
}
