import csv
import random

# --- CONFIGURATION ---
FILENAME = "large_test_upload.csv"
NUM_ROWS = 1500  # Change this to generate more rows
SELECTED_DATE = "17122025"  # ddMMyyyy (Match this in your frontend)

# --- CONSTANTS ---
BATCH_ID = "01"
YEAR = "2025"
MONTH = "09"
VALID_BRANCH = "1001"
VALID_CURRENCY = "USD"
VALID_CGL = "1234567890"
STATIC_COLS = ["B", "C", "D"]

# --- GENERATION LOGIC ---
def generate_row(index):
    # Default Valid Data
    row = {
        "batch": BATCH_ID,
        "date": SELECTED_DATE,
        "year": YEAR,
        "month": MONTH,
        "branch": VALID_BRANCH,
        "curr": VALID_CURRENCY,
        "cgl": VALID_CGL,
        "amt": f"{random.randint(10, 10000)}.{random.randint(10, 99)}",
        "type": "Dr" if index % 2 == 0 else "Cr",
        "rem": f"Bulk_Txn_{index}",
        "prod": "12345678", # Valid Product Code
        "c12": "B", "c13": "C", "c14": "D"
    }

    # Inject Errors occasionally (every 50th row) to test validation
    if index % 50 == 0:
        row["rem"] = "" # Error: Empty Remark
    elif index % 50 == 1:
        row["branch"] = "" # Error: Empty Branch
    elif index % 50 == 2:
        row["amt"] = "" # Error: Empty Amount
    elif index % 50 == 3:
        row["date"] = "01012020" # Error: Date Mismatch
    elif index % 50 == 4:
        row["prod"] = "123" # Error: Invalid Product Code length

    return [
        row["batch"], row["date"], row["year"], row["month"],
        row["branch"], row["curr"], row["cgl"], row["amt"],
        row["type"], row["rem"], row["prod"],
        row["c12"], row["c13"], row["c14"]
    ]

# --- WRITE FILE ---
with open(FILENAME, mode='w', newline='') as file:
    writer = csv.writer(file)
    # No header needed for system file, but if you need one, uncomment below:
    # writer.writerow(["BatchID","Date","Year","Month","Branch","Currency","CGL","Amount","Type","Remarks","Product","C12","C13","C14"])
    
    print(f"Generating {NUM_ROWS} rows...")
    for i in range(1, NUM_ROWS + 1):
        writer.writerow(generate_row(i))

print(f"Successfully created '{FILENAME}' with {NUM_ROWS} rows.")
