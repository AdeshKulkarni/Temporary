    public Map<String, Object> getAllRequestsSummary(String userId, Integer roleId, int page, int size, String batchIdFilter, String vidFilter) {

        // =========================================================================
        // PERFECTLY ISOLATED VISIBILITY RULES
        // 1. You see it if you created it.
        // 2. You see it if you acted on it previously.
        // 3. EXACT STATE CHECK: You see it if it is CURRENTLY pending at your role 
        //    (If PENDING_AT is 28, w.ALLOWED_ROLE_ID is 52, so Role 52 sees it!).
        // 4. Tracking Rule: Explicitly prevents the Execution role from seeing all drafts.
        // =========================================================================
        StringBuilder whereClause = new StringBuilder("""
            WHERE (
                p.CREATED_BY = ? 
                OR w.ALLOWED_ROLE_ID = ? 
                OR EXISTS (
                    SELECT 1 FROM VOUCHER_WORKFLOW vw 
                    WHERE vw.BATCH_ID_FK = p.BATCH_ID 
                    AND vw.LAST_ACTION_BY = ?
                )
                OR (
                    -- Allows normal Checkers to track all batches in their workflow
                    EXISTS (
                        SELECT 1 FROM WORKFLOW_TRANSITIONS wt 
                        WHERE wt.WORKFLOW_CODE = p.WORKFLOW_CODE 
                        AND wt.ALLOWED_ROLE_ID = ?
                    )
                    -- IMPORTANT: This absolutely prevents the Execution Role (Role 52) 
                    -- from seeing all drafts. It forces them to rely ONLY on w.ALLOWED_ROLE_ID = ? (Step 28).
                    AND NOT EXISTS (
                        SELECT 1 FROM WORKFLOW_TRANSITIONS wt_exec 
                        WHERE wt_exec.WORKFLOW_CODE = p.WORKFLOW_CODE 
                        AND wt_exec.ALLOWED_ROLE_ID = ? 
                        AND wt_exec.WORKFLOW_STAGE = 'Execution'
                    )
                )
                OR (
                    -- Also let the Execution Role see Completed/Posted batches (After their step finishes)
                    p.BATCH_STATUS IN ('C', 'A')
                    AND EXISTS (
                        SELECT 1 FROM WORKFLOW_TRANSITIONS wt_exec 
                        WHERE wt_exec.WORKFLOW_CODE = p.WORKFLOW_CODE 
                        AND wt_exec.ALLOWED_ROLE_ID = ? 
                        AND wt_exec.WORKFLOW_STAGE = 'Execution'
                    )
                )
            )
        """);

        List<Object> paramsList = new java.util.ArrayList<>();
        paramsList.add(userId);    // 1. CREATED_BY
        paramsList.add(roleId);    // 2. w.ALLOWED_ROLE_ID (Checks if PENDING_AT matches their role)
        paramsList.add(userId);    // 3. LAST_ACTION_BY
        paramsList.add(roleId);    // 4. Tracking EXISTS
        paramsList.add(roleId);    // 5. Tracking NOT EXISTS (Blocks Role 52 from seeing drafts)
        paramsList.add(roleId);    // 6. Completed Batches EXISTS

        // Append Batch ID search if provided
        if (batchIdFilter != null && !batchIdFilter.trim().isEmpty()) {
            whereClause.append(" AND p.BATCH_ID LIKE ? ");
            paramsList.add("%" + batchIdFilter.trim() + "%");
        }

        // Append VID search if provided
        if (vidFilter != null && !vidFilter.trim().isEmpty()) {
            whereClause.append(" AND LOWER(p.VID_FK) LIKE LOWER(?) ");
            paramsList.add("%" + vidFilter.trim() + "%");
        }

        // FAST COUNT QUERY
        String countSql = "SELECT COUNT(*) FROM VOUCHER_POSTING p LEFT JOIN WORKFLOW_TRANSITIONS w ON p.PENDING_AT = w.WORKFLOW_ID " + whereClause.toString();
        Long totalElements = jdbcTemplate.queryForObject(countSql, Long.class, paramsList.toArray());
        if (totalElements == null) totalElements = 0L;

        int offset = page * size;
        
        String dataSql = """
            SELECT 
                p.BATCH_ID, 
                p.VID_FK,
                p.CREATED_BY AS CREATOR_ID, 
                u.FIRST_NAME,
                u.LAST_NAME,
                p.CREATED_AT AS REQ_DATE, 
                p.RECORDS_COUNT AS TOTAL_ROWS, 
                p.BATCH_STATUS, 
                p.IS_EXECUTION_SCHEDULED,
                w.WORKFLOW_STAGE, 
                w.WORKFLOW_CODE,
                w.WORKFLOW_ORDER,
                (SELECT LAST_ACTION FROM VOUCHER_WORKFLOW WHERE BATCH_ID_FK = p.BATCH_ID ORDER BY LAST_ACTION_AT DESC FETCH FIRST 1 ROWS ONLY) AS LATEST_ACTION
            FROM VOUCHER_POSTING p
            LEFT JOIN WORKFLOW_TRANSITIONS w ON p.PENDING_AT = w.WORKFLOW_ID
            LEFT JOIN USERS u ON p.CREATED_BY = u.USER_ID
            """ + whereClause.toString() + """
            ORDER BY p.CREATED_AT DESC, p.BATCH_ID DESC
            OFFSET ? ROWS FETCH NEXT ? ROWS ONLY
        """;

        List<Object> dataParams = new java.util.ArrayList<>(paramsList);
        dataParams.add(offset);
        dataParams.add(size);

        List<Map<String, Object>> content = jdbcTemplate.query(dataSql, (rs, rowNum) -> {
            Map<String, Object> map = new java.util.HashMap<>();
            map.put("batchId", rs.getString("BATCH_ID"));
            map.put("vid", rs.getString("VID_FK"));
            map.put("creatorId", rs.getString("CREATOR_ID"));

            String fName = rs.getString("FIRST_NAME");
            String lName = rs.getString("LAST_NAME");
            String fullName = (fName != null || lName != null)
                    ? ((fName != null ? fName : "") + " " + (lName != null ? lName : "")).trim()
                    : rs.getString("CREATOR_ID");
            map.put("creatorName", fullName);

            map.put("requestDate", rs.getTimestamp("REQ_DATE"));
            map.put("requestCount", rs.getLong("TOTAL_ROWS"));
            map.put("approvalWorkflow", rs.getString("WORKFLOW_CODE"));
            
            String bStatus = rs.getString("BATCH_STATUS");
            String isSched = rs.getString("IS_EXECUTION_SCHEDULED");
            String wfStage = rs.getString("WORKFLOW_STAGE");
            
            map.put("workflowState", wfStage);
            map.put("currentOrder", rs.getObject("WORKFLOW_ORDER"));
            map.put("batchStatus", bStatus);

            boolean isFailed = "Execution Failed".equals(rs.getString("LATEST_ACTION"));
            map.put("isExecutionFailed", isFailed);

            if ("R".equals(bStatus)) {
                map.put("currentStatus", "REJECTED");
            } else if ("A".equals(bStatus) || "C".equals(bStatus)) {
                map.put("currentStatus", "POSTED");
            } else if ("Q".equals(bStatus)) {
                map.put("currentStatus", "QUEUED FOR RETRY");
            } else if (isFailed) {
                map.put("currentStatus", "EXECUTION FAILED");
            } else if ("P".equals(bStatus) && "Execution".equalsIgnoreCase(wfStage)) {
                map.put("currentStatus", "YES".equalsIgnoreCase(isSched) ? "SCHEDULED" : "WAITING FOR EXECUTION");
            } else {
                map.put("currentStatus", wfStage != null ? wfStage : "PENDING");
            }

            map.put("timeSinceLastAction", calculateTimeSinceLastAction(rs.getString("BATCH_ID")));

            return map;
        }, dataParams.toArray());

        Map<String, Object> response = new java.util.HashMap<>();
        response.put("content", content);
        response.put("totalElements", totalElements);
        response.put("totalPages", (int) Math.ceil((double) totalElements / size));

        return response;
    }
