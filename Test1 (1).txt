import React, { useState, useEffect, useCallback } from "react";
import {
  Box,
  Typography,
  Grid,
  TextField,
  Button,
  Paper,
  IconButton,
  MenuItem,
  CircularProgress,
  Autocomplete,
} from "@mui/material";
import AddIcon from "@mui/icons-material/Add";
import DeleteIcon from "@mui/icons-material/Delete";
import { debounce } from "lodash";

// --- PERMANENT API SERVICE ---
const API_BASE_URL = "http://10.0.19.101:9000"; // Your backend URL

/**
 * A simple fetch wrapper to handle requests and errors
 */
const apiFetch = async (endpoint, options = {}) => {
  const response = await fetch(`${API_BASE_URL}${endpoint}`, {
    ...options,
    headers: {
      "Content-Type": "application/json",
      // TODO: Add Authorization header when auth is ready
      // "Authorization": `Bearer ${your_jwt_token}`,
      ...options.headers,
    },
  });

  if (!response.ok) {
    const errorBody = await response.text();
    console.error("API Error Response:", errorBody);
    throw new Error(
      `API request failed with status ${response.status}: ${errorBody}`
    );
  }
  
  // Handle successful but empty responses (e.g., from a DELETE)
  if (response.status === 204 || response.headers.get("content-length") === "0") {
      return null; 
  }

  // Handle responses that are JSON
  const data = await response.json();
  if (data.success === false) {
    // Handle specific API-level errors if your API sends them
    throw new Error(data.message || "An unknown API error occurred");
  }

  return data;
};

/**
 * This object contains all your "permanent" API calls.
 */
const apiService = {
  // GET /api/branches/search?query=...
  searchBranches: (query) => {
    return apiFetch(`/api/branches/search?query=${encodeURIComponent(query)}`);
  },

  // GET /currency (Using the path from your mock API)
  getCurrencies: async () => {
    const response = await apiFetch(`/currency`);
    // Assuming the response is { success: true, data: [...] }
    if (response?.success && Array.isArray(response.data)) {
        return [...new Set(response.data)]; // Return deduplicated list
    }
    return [];
  },

  // GET /api/cgls/classifications
  getHeads: () => {
    return apiFetch(`/api/cgls/classifications`);
  },

  // GET /api/cgls/search?query=...&classification=...
  searchCgls: (query, classification) => {
    return apiFetch(
      `/api/cgls/search?query=${encodeURIComponent(
        query
      )}&classification=${encodeURIComponent(classification)}`
    );
  },

  // POST /moc/create-request
  submitMocEntry: (mocEntryPayload) => {
    // This payload must match your Java `CreateMocRequestDto`
    return apiFetch(`/moc/create-request`, {
      method: "POST",
      body: JSON.stringify(mocEntryPayload),
    });
  },
};
// --- END OF PERMANENT API SERVICE ---


/**
 * Snackbar replacement
 */
const showSnackBar = (message, severity) => {
  console.log(`[${severity.toUpperCase()}]: ${message}`);
  alert(`[${severity.toUpperCase()}]: ${message}`);
};

/**
 * Initial Row State
 */
const initialRowState = {
  id: Date.now(),
  branch: null,
  postDate: new Date().toISOString().split("T")[0],
  currency: "",
  head: "",
  cgl: null,
  amount: "",
  txnType: "Debit", // Defaulted to Debit
  remarks: "",
  productCode: "",
};

/**
 * MOC Posting Component
 */
export default function MOCPosting() {
  const [rows, setRows] = useState([initialRowState]);
  const [currencyOptions, setCurrencyOptions] = useState([]);
  const [headOptions, setHeadOptions] = useState([]);
  const [isSubmitting, setIsSubmitting] = useState(false);
  
  const [branchSearch, setBranchSearch] = useState({
    loading: false,
    options: [],
  });
  const [cglSearch, setCglSearch] = useState({ loading: false, options: [] });

  // --- Initial Data Load ---
  useEffect(() => {
    const fetchInitialData = async () => {
      try {
        // Fetch Currencies
        const uniqueCurrencies = await apiService.getCurrencies();
        setCurrencyOptions(uniqueCurrencies);

        // Fetch Heads
        const headResponse = await apiService.getHeads();
        if (headResponse) {
          const formattedHeads = Object.entries(headResponse).map(
            ([key, value]) => ({ key, value })
          );
          setHeadOptions(formattedHeads);
        } else {
          showSnackBar("Failed to load head classifications.", "error");
        }
      } catch (error) {
        console.error("Error fetching initial data:", error);
        showSnackBar("An error occurred while loading page data.", "error");
      }
    };

    fetchInitialData();
  }, []); // Runs once on component mount

  const handleAddRow = () => {
    setRows([...rows, { ...initialRowState, id: Date.now() }]);
  };

  const handleDeleteRow = (id) => {
    if (rows.length > 1) {
      setRows(rows.filter((row) => row.id !== id));
    } else {
      showSnackBar("Cannot delete the last row.", "warning");
    }
  };

  const handleRowChange = (id, field, value) => {
    const newRows = rows.map((row) => {
      if (row.id === id) {
        // If head changes, clear the CGL selection
        if (field === "head" && row.head !== value) {
          setCglSearch({ loading: false, options: [] }); // Clear old CGL options
          return { ...row, [field]: value, cgl: null };
        }
        return { ...row, [field]: value };
      }
      return row;
    });
    setRows(newRows);
  };

  // --- API Search Functions ---
  const searchBranches = useCallback(
    debounce(async (query) => {
      if (!query) {
        setBranchSearch({ loading: false, options: [] });
        return;
      }
      setBranchSearch((prev) => ({ ...prev, loading: true }));
      try {
        // Using permanent apiService
        const response = await apiService.searchBranches(query);
        setBranchSearch({ loading: false, options: response || [] });
      } catch (error) {
        console.error("Branch search failed:", error);
        showSnackBar("Branch search failed. See console.", "error");
        setBranchSearch({ loading: false, options: [] });
      }
    }, 300),
    [] // No dependencies, apiService is a stable object
  );

  const searchCgls = useCallback(
    debounce(async (query, headKey) => {
      if (!query || !headKey) {
        setCglSearch({ loading: false, options: [] });
        return;
      }
      setCglSearch((prev) => ({ ...prev, loading: true }));
      try {
        // Using permanent apiService with both params
        const response = await apiService.searchCgls(query, headKey);
        setCglSearch({ loading: false, options: response || [] });
      } catch (error) {
        console.error("CGL search failed:", error);
        showSnackBar("CGL search failed. See console.", "error");
        setCglSearch({ loading: false, options: [] });
      }
    }, 300),
    [] // No dependencies
  );

  const handleSubmit = async () => {
    setIsSubmitting(true);
    
    let successfulSubmissions = 0;
    let failedSubmissions = 0;

    for (const [index, row] of rows.entries()) {
      // 1. Validate each row
      if (
        !row.branch || !row.currency || !row.head || !row.cgl ||
        !row.amount || !row.txnType || !row.productCode
      ) {
        showSnackBar(
          `Row ${index + 1} is incomplete. Please fill all fields.`,
          "error"
        );
        failedSubmissions++;
        continue; // Skip this row
      }

      // 2. Format payload to match Java DTO
      const payload = {
        changeType: "ADD", // Assuming all are new
        masterMocId: null, // Not updating an existing one
        pDate: row.postDate,
        branch: row.branch.code, // Send the code, not the object
        currency: row.currency,
        cgl: row.cgl.cglNumber, // Send the number, not the object
        amount: parseFloat(row.amount),
        transactionType: row.txnType,
        productType: row.productCode, // Map React state to DTO field
        remarks: row.remarks,
        arFlag: "Y", // Or logic to determine this
        acClassification: row.head, // Pass the 'A', 'L', 'I' key
      };

      // 3. Submit one request
      try {
        await apiService.submitMocEntry(payload);
        successfulSubmissions++;
      } catch (error) {
        console.error(`Failed to submit row ${index + 1}:`, error);
        showSnackBar(`Failed to submit row ${index + 1}: ${error.message}`, "error");
        failedSubmissions++;
      }
    }
    
    // 4. Report results
    if (successfulSubmissions > 0 && failedSubmissions === 0) {
        showSnackBar("All MOC entries submitted successfully!", "success");
        setRows([{ ...initialRowState, id: Date.now() }]); // Reset form
    } else if (successfulSubmissions > 0 && failedSubmissions > 0) {
        showSnackBar(`Partially successful: ${successfulSubmissions} submitted, ${failedSubmissions} failed.`, "warning");
    } else if (successfulSubmissions === 0 && failedSubmissions > 0) {
        showSnackBar("Submission failed. Please check errors and try again.", "error");
    }

    setIsSubmitting(false);
  };


  return (
    <Box sx={{ p: 3, maxWidth: "95%", margin: "auto" }}>
      <Paper elevation={3} sx={{ p: 3 }}>
        <Typography
          variant="h4"
          gutterBottom
          sx={{ fontWeight: "bold", color: "primary.main", mb: 3 }}
        >
          MOC Posting
        </Typography>

        {rows.map((row) => (
          <Grid
            container
            spacing={2}
            key={row.id}
            alignItems="center"
            sx={{ mb: 2 }}
          >
            {/* Branch */}
            <Grid item xs={12} md={2}>
              <Autocomplete
                options={branchSearch.options}
                getOptionLabel={(option) => `${option.code} - ${option.name}`}
                value={row.branch}
                onChange={(event, newValue) =>
                  handleRowChange(row.id, "branch", newValue)
                }
                onInputChange={(event, newInputValue) =>
                  searchBranches(newInputValue)
                }
                loading={branchSearch.loading}
                filterOptions={(x) => x}
                isOptionEqualToValue={(option, value) => option.code === value.code}
                renderInput={(params) => (
                  <TextField
                    {...params}
                    label="Branch"
                    placeholder="Search by name or code"
                    InputProps={{
                      ...params.InputProps,
                      endAdornment: (
                        <>
                          {branchSearch.loading ? (
                            <CircularProgress color="inherit" size={20} />
                          ) : null}
                          {params.InputProps.endAdornment}
                        </>
                      ),
                    }}
                  />
                )}
              />
            </Grid>

            {/* Post Date */}
            <Grid item xs={12} md={1.5}>
              <TextField
                label="Post Date"
                value={row.postDate}
                fullWidth
                InputProps={{
                  readOnly: true,
                }}
                sx={{ backgroundColor: "#f4f4f4" }}
              />
            </Grid>

            {/* Currency */}
            <Grid item xs={12} md={1}>
              <TextField
                select
                label="Currency"
                value={row.currency}
                onChange={(e) =>
                  handleRowChange(row.id, "currency", e.target.value)
                }
                fullWidth
              >
                {currencyOptions.map((c) => (
                  <MenuItem key={c} value={c}>
                    {c}
                  </MenuItem>
                ))}
              </TextField>
            </Grid>

            {/* Head */}
            <Grid item xs={12} md={1.5}>
              <TextField
                select
                label="Head"
                value={row.head}
                onChange={(e) =>
                  handleRowChange(row.id, "head", e.target.value)
                }
                fullWidth
              >
                {headOptions.map((h) => (
                  <MenuItem key={h.key} value={h.key}>
                    {h.value}
                  </MenuItem>
                ))}
              </TextField>
            </Grid>

            {/* CGL */}
            <Grid item xs={12} md={2}>
              <Autocomplete
                options={cglSearch.options}
                getOptionLabel={(option) =>
                  `${option.cglNumber} - ${option.description}`
                }
                value={row.cgl}
                onChange={(event, newValue) =>
                  handleRowChange(row.id, "cgl", newValue)
                }
                onInputChange={(event, newInputValue) =>
                  searchCgls(newInputValue, row.head)
                }
                loading={cglSearch.loading}
                disabled={!row.head}
                filterOptions={(x) => x}
                isOptionEqualToValue={(option, value) =>
                  option.cglNumber === value.cglNumber
                }
                renderInput={(params) => (
                  <TextField
                    {...params}
                    label="CGL"
                    placeholder={row.head ? "Search CGL" : "Select Head first"}
                    InputProps={{
                      ...params.InputProps,
                      endAdornment: (
                        <>
                          {cglSearch.loading ? (
                            <CircularProgress color="inherit" size={20} />
                          ) : null}
                          {params.InputProps.endAdornment}
                        </>
                      ),
                    }}
                  />
                )}
              />
            </Grid>

            {/* Amount */}
            <Grid item xs={12} md={1.5}>
              <TextField
                label="Amount"
                type="text"
                value={row.amount}
                onChange={(e) => {
                  const val = e.target.value;
                  if (/^\d{0,16}(\.\d{0,6})?$/.test(val)) {
                    handleRowChange(row.id, "amount", val);
                  }
                }}
                fullWidth
                placeholder="0.00"
              />
            </Grid>

            {/* Txn Type */}
            <Grid item xs={12} md={1}>
              <TextField
                select
                label="Txn Type"
                value={row.txnType}
                onChange={(e) =>
                  handleRowChange(row.id, "txnType", e.target.value)
                }
                fullWidth
              >
                <MenuItem value="Debit">Debit</MenuItem>
                <MenuItem value="Credit">Credit</MenuItem>
              </TextField>
            </Grid>

            {/* Remarks */}
            <Grid item xs={12} md={2}>
              <TextField
                label="Remarks"
                value={row.remarks}
                onChange={(e) => {
                  const val = e.target.value;
                  if (val.length <= 40) {
                    handleRowChange(row.id, "remarks", val);
                  }
                }}
                fullWidth
                inputProps={{ maxLength: 40 }}
              />
            </Grid>

            {/* Product Code */}
            <Grid item xs={12} md={1.5}>
              <TextField
                label="Product Code"
                value={row.productCode}
                onChange={(e) => {
                  const val = e.target.value;
                  if (/^\d{0,8}$/.test(val)) {
                    handleRowChange(row.id, "productCode", val);
                  }
                }}
                fullWidth
                inputProps={{ maxLength: 8 }}
              />
            </Grid>

            {/* Actions */}
            <Grid item xs={12} md={0.5} sx={{ textAlign: "center" }}>
              <IconButton
                onClick={() => handleDeleteRow(row.id)}
                color="error"
                aria-label="delete row"
              >
                <DeleteIcon />
              </IconButton>
            </Grid>
          </Grid>
        ))}

        <Box sx={{ mt: 3, display: "flex", gap: 2 }}>
          <Button
            variant="contained"
            startIcon={<AddIcon />}
            onClick={handleAddRow}
          >
            Add Row
          </Button>
          <Button
            variant="contained"
            color="success"
            onClick={handleSubmit}
            disabled={isSubmitting}
          >
            {isSubmitting ? <CircularProgress size={24} /> : "Submit"}
          </Button>
        </Box>
      </Paper>
    </Box>
  );
}
