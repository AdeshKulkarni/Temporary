            Map<String, Object> data = (Map<String, Object>) map.get("data");

            String quarterEndDate = user.get("quarterEndDate");

            List<Map<String, Object>> questionsData = lfarDataRepository.getQuestionsData(reportId, submissionId,

                Map<String, Object> res = new HashMap<>();

                // Fetch formData Here
                String lfarTable = String.valueOf(questionsData.get(i).get("LFAR_TABLE"));
                if (!lfarTable.equalsIgnoreCase("null")) {
                    Map<String, Object> formData = new HashMap<>();

                    String tableId = sanitizeSegment(String.valueOf(questionsData.get(i).get("LFAR_SEQ")), "tableId");

                    // Add Column Data
                    List<Map<String, Object>> columnData = crsLfarTableDataRepository.getColumnData(tableId, reportId);
                    formData.put("COLUMN_DATA", columnData);

                    // getting column names
                    List<String> columns = new ArrayList<>();
                    for (int j = 0; j < columnData.size(); j++)
                        columns.add(String.valueOf(columnData.get(j).get("TABLE_PARAM")));

                    // Row Data
                    List<Map<String, Object>> rowData = crsLfarTableDataRepository.getRowData(submissionId, tableId);
                    formData.put("ROW_DATA", getUpdatedRowData(rowData, columns));

                    res.put("lfarTableFlag", true);
                    res.put("formData", formData);

                } else {
                    res.put("lfarTableFlag", false);
                }

                result.add(res);
            }

            responseVo.setResult(result);

            String userRole = user.get("user_role").toString();

            Map<String, Object> questionData = (Map<String, Object>) data.get("questionData");

            if (userRole.equalsIgnoreCase("11")) {
                log.info("Inside auditor save question");
                // Save CRS_LFAR_DATA
                CrsLfarData crsLfarData = new CrsLfarData();
                crsLfarData.setBranch(branch_code);
                crsLfarData.setDate(formatter.parse(quarterEndDate));
                crsLfarData.setReportMasterListFk(submissionId);

                crsLfarData.setDataOptions(questionData.get("LFAR_DATA_OPTIONS").toString());
                crsLfarData.setDataComment(questionData.get("LFAR_DATA_COMMENT").toString());
                crsLfarData.setDataSeq(questionData.get("LFAR_SEQ").toString());

                lfarDataRepository.save(crsLfarData);
                log.info("Data Saved : " + crsLfarData);

            } else if (userRole.equalsIgnoreCase("1")) {
                log.info("Inside maker save question");

                //check if exist by id or not for maker and update the compliance comment only
                String cleanSeq = sanitizeSegment(String.valueOf(questionData.get("LFAR_SEQ")), "LFAR_SEQ");
                CrsLfarDataId dataId = new CrsLfarDataId(branch_code, formatter.parse(quarterEndDate), reportId, cleanSeq);

                if (lfarDataRepository.existsById(dataId)) {
                    CrsLfarData crs = lfarDataRepository.findById(dataId).get();
                    if (questionData.get("LFAR_COMPLIANCE_COMMENT") != null) {
                        crs.setCompComment(questionData.get("LFAR_COMPLIANCE_COMMENT").toString());
                        lfarDataRepository.save(crs);
                        log.info("Data Saved : " + crs);
                    }
                }
            }

            responseVo.setResult(true);
            responseVo.setStatusCode(HttpStatus.OK.value());
            responseVo.setMessage("Data Saved Successfully");

        } catch (Exception e) {
            log.info("Error Occurred:- " + e.getMessage());
            e.printStackTrace();
            responseVo.setResult(null);
            responseVo.setStatusCode(HttpStatus.INTERNAL_SERVER_ERROR.value());
            responseVo.setMessage("Error Fetching Data");
        }

        return responseVo;
    }

    private List<Map<String, String>> getUpdatedRowData(List<Map<String, Object>> rowData, List<String> columns) {
        List<Map<String, String>> result = new ArrayList<>();
        // columns.add("rowId");columns.add("rowNum");

            Map<String, String> record = new HashMap<>();
            for (String column : columns) {
                if (rowData.get(i).get(column.toUpperCase()) != null) {
                    record.put(column, rowData.get(i).get(column.toUpperCase()).toString());
                } else {
                    record.put(column, null);
                }

            }
            record.put(ROW_ID, rowData.get(i).get("DATA_ID").toString());
            result.add(record);
        }
        return result;
    }

    private String sanitizeSegment(String value, String fieldName) {
        if (value == null || value.isEmpty() || value.contains("..")) {
            throw new IllegalArgumentException("Invalid characters in " + fieldName);
        }
        StringBuilder clean = new StringBuilder(value.length());
        for (int i = 0; i < value.length(); i++) {
            char c = value.charAt(i);
            int idx = CommonConstants.ALLOWED_CHARS.indexOf(c);
            if (idx < 0) {
                throw new IllegalArgumentException("Invalid character in " + fieldName);
            }
            clean.append(CommonConstants.ALLOWED_CHARS.charAt(idx));
        }
        return clean.toString();
    }

    private int sanitizeInt(Object rawValue, String fieldName) {
        if (rawValue == null) {
            throw new IllegalArgumentException(fieldName + " cannot be null");
        }
        String str = rawValue.toString().trim();
        if (str.isEmpty()) {
            throw new IllegalArgumentException(fieldName + " cannot be empty");
        }
        StringBuilder clean = new StringBuilder(str.length());
        for (int i = 0; i < str.length(); i++) {
            char c = str.charAt(i);
            int idx = CommonConstants.ALLOWED_DIGITS.indexOf(c);
            if (idx < 0) {
                throw new IllegalArgumentException("Invalid digit in " + fieldName);
            }
            clean.append(CommonConstants.ALLOWED_DIGITS.charAt(idx));
        }
        return Integer.parseInt(clean.toString());
    }
