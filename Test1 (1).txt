const BASE_URL = '/RB/template-config';

const handleResponse = async (response) => {
  const result = await response.json().catch(() => ({ 
    message: "Failed to parse server response", 
    status: "ERROR" 
  }));

  if (!response.ok) {
    throw new Error(result.message || `Error ${response.status}: ${response.statusText}`);
  }
  return result;
};

export const templateApi = {
  getAllTemplates: async () => {
    try {
      const response = await fetch(`${BASE_URL}/get-templates`, {
        method: 'GET',
        headers: { 'Content-Type': 'application/json' },
      });
      return await handleResponse(response);
    } catch (error) {
      console.error("API Error [getAllTemplates]:", error.message);
      throw error;
    }
  },

  getTemplateById: async (templateId) => {
    try {
      const response = await fetch(`${BASE_URL}/${templateId}`, {
        method: 'GET',
        headers: { 'Content-Type': 'application/json' },
      });
      return await handleResponse(response);
    } catch (error) {
      console.error(`API Error [getTemplateById] ID: ${templateId}:`, error.message);
      throw error;
    }
  },

  // New method specifically for updating status
  updateTemplateStatus: async (templateId, status) => {
    try {
        // Assuming your backend accepts a PATCH or PUT for status
        // Adjust body structure based on your actual backend requirement
      const response = await fetch(`${BASE_URL}/update-status/${templateId}`, {
        method: 'PUT', 
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ status: status }), 
      });
      return await handleResponse(response);
    } catch (error) {
      console.error(`API Error [updateStatus] ID: ${templateId}:`, error.message);
      throw error;
    }
  },

  deleteTemplate: async (templateId) => {
    try {
      const response = await fetch(`${BASE_URL}/${templateId}`, {
        method: 'DELETE',
        headers: { 'Content-Type': 'application/json' },
      });
      return await handleResponse(response);
    } catch (error) {
      console.error(`API Error [deleteTemplate] ID: ${templateId}:`, error.message);
      throw error;
    }
  },

  updateTemplate: async (templateId, data) => {
    try {
      const response = await fetch(`${BASE_URL}/save`, {
        method: 'PUT',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify(data),
      });
      return await handleResponse(response);
    } catch (error) {
      console.error(`API Error [updateTemplate] ID: ${templateId}:`, error.message);
      throw error;
    }
  }
};

2. src/components/TemplateConfiguration.js
I refactored the fetch logic so it can be passed down to the table. This allows the table to refresh the data after a status change or deletion.
import React, { useState, useEffect, useCallback } from 'react';
import { Typography, Container, Paper } from '@mui/material';
import TemplateTable from './TemplateTable';
import { templateApi } from '../api/templateApi';

const TemplateConfiguration = () => {
  const [templates, setTemplates] = useState([]);

  // Defined as a function so we can pass it to the child to trigger a refresh
  const fetchTemplates = useCallback(() => {
    templateApi.getAllTemplates()
      .then(res => setTemplates(res.data))
      .catch(err => console.error("Failed to load templates", err));
  }, []);

  useEffect(() => {
    fetchTemplates();
  }, [fetchTemplates]);

  return (
    <Container maxWidth={false} disableGutters sx={{ mt: 4 }}>
      <Paper sx={{ width: "100%", p: 3 }}>
        <Typography variant="h5" gutterBottom sx={{ fontWeight: 'bold' }}>
          Report Management
        </Typography>
        {/* Pass fetchTemplates as onRefresh prop */}
        <TemplateTable data={templates} onRefresh={fetchTemplates} />
      </Paper>
    </Container>
  );
};

export default TemplateConfiguration;

import React, { useState } from 'react';
import { DataGrid } from '@mui/x-data-grid';
import Box from '@mui/material/Box';
import Paper from '@mui/material/Paper';
import IconButton from '@mui/material/IconButton';
import Tooltip from '@mui/material/Tooltip';
import Dialog from '@mui/material/Dialog';
import DialogTitle from '@mui/material/DialogTitle';
import DialogContent from '@mui/material/DialogContent';
import DialogContentText from '@mui/material/DialogContentText';
import DialogActions from '@mui/material/DialogActions';
import Button from '@mui/material/Button';
import Chip from '@mui/material/Chip';
import MenuItem from '@mui/material/MenuItem';
import TextField from '@mui/material/TextField';

// Icons
import Visibility from '@mui/icons-material/Visibility';
import Delete from '@mui/icons-material/Delete';
import Edit from '@mui/icons-material/Edit'; // Added Edit Icon
import ReportProblemOutlined from '@mui/icons-material/ReportProblemOutlined';

// Internal Dialogs
import TemplateViewDialog from './TemplateViewDialog';
import { templateApi } from '../api/templateApi';

const TemplateTable = ({ data, onRefresh }) => {
  const [selectedTemplate, setSelectedTemplate] = useState(null);
  
  // Dialog States
  const [openView, setOpenView] = useState(false);
  const [openDelete, setOpenDelete] = useState(false);
  const [openStatusEdit, setOpenStatusEdit] = useState(false);
  const [openGeneralEdit, setOpenGeneralEdit] = useState(false); // Placeholder for general edit

  // Status State
  const [newStatus, setNewStatus] = useState('');

  // --- Handlers ---
  const handleAction = (template, actionType) => {
    setSelectedTemplate(template);
    
    if (actionType === 'view') setOpenView(true);
    if (actionType === 'delete') setOpenDelete(true);
    
    if (actionType === 'status') {
      setNewStatus(template.status || 'INACTIVE'); // Default if null
      setOpenStatusEdit(true);
    }

    if (actionType === 'edit') {
        // Placeholder for full edit logic
        console.log("Edit full template:", template);
        setOpenGeneralEdit(true); 
    }
  };

  const closeAllDialogs = () => {
    setOpenView(false);
    setOpenDelete(false);
    setOpenStatusEdit(false);
    setOpenGeneralEdit(false);
    setSelectedTemplate(null);
  };

  const confirmDelete = async () => {
    try {
      await templateApi.deleteTemplate(selectedTemplate.templateId);
      onRefresh(); 
      closeAllDialogs();
    } catch (error) {
      console.error("Delete failed:", error.message);
    }
  };

  const confirmStatusUpdate = async () => {
    if (!selectedTemplate) return;
    try {
      await templateApi.updateTemplateStatus(selectedTemplate.templateId, newStatus);
      onRefresh(); // Refresh table data
      closeAllDialogs();
    } catch (error) {
      console.error("Status update failed:", error.message);
    }
  };

  // --- Data Grid Column Definitions ---
  const columns = [
    { 
      field: 'templateId', 
      headerName: 'ID', 
      flex: 0.5 
    },
    { 
      field: 'templateName', 
      headerName: 'Report Name', 
      flex: 1.5,
      renderCell: (params) => (
        <Box sx={{ fontWeight: 500 }}>{params.value}</Box>
      )
    },
    { 
      field: 'description', 
      headerName: 'Description', 
      flex: 2 
    },
    // --- New Status Column ---
    {
      field: 'status',
      headerName: 'Status',
      flex: 1.2,
      renderCell: (params) => {
        const status = params.value || 'INACTIVE';
        const isSuccess = status === 'ACTIVE';
        
        return (
            <Box sx={{ display: 'flex', alignItems: 'center', gap: 1 }}>
                <Chip 
                    label={status} 
                    color={isSuccess ? "success" : "default"} 
                    size="small" 
                    variant="outlined"
                />
                <Tooltip title="Change Status">
                    <IconButton 
                        size="small" 
                        onClick={() => handleAction(params.row, 'status')}
                    >
                        <Edit fontSize="inherit" sx={{ fontSize: '1rem', color: 'text.secondary' }} />
                    </IconButton>
                </Tooltip>
            </Box>
        );
      }
    },
    {
      field: 'actions',
      headerName: 'Actions',
      flex: 1.2,
      sortable: false,
      filterable: false,
      renderCell: (params) => (
        <Box sx={{ display: 'flex', gap: 0.5, width: '100%' }}>
          <Tooltip title="View & Run">
            <IconButton 
              size="small" 
              color="primary" 
              onClick={() => handleAction(params.row, 'view')}
            >
              <Visibility fontSize="small" />
            </IconButton>
          </Tooltip>

          {/* Added General Edit Button */}
          <Tooltip title="Edit Template Details">
            <IconButton 
              size="small" 
              color="info" 
              onClick={() => handleAction(params.row, 'edit')}
            >
              <Edit fontSize="small" />
            </IconButton>
          </Tooltip>

          <Tooltip title="Delete Template">
            <IconButton 
              size="small" 
              color="error" 
              onClick={() => handleAction(params.row, 'delete')}
            >
              <Delete fontSize="small" />
            </IconButton>
          </Tooltip>
        </Box>
      )
    }
  ];

  return (
    <Paper sx={{ width: "100%" }} variant="outlined">
      <DataGrid
        rows={data}
        columns={columns}
        getRowId={(row) => row.templateId}
        pageSizeOptions={[5, 10, 25]}
        initialState={{
          pagination: { paginationModel: { pageSize: 10 } },
        }}
        disableRowSelectionOnClick
        sx={{
          border: 'none',
          '& .MuiDataGrid-columnHeader': {
            backgroundColor: (theme) => theme.palette.grey[50],
            fontWeight: 'bold'
          },
        }}
      />

      {/* View Dialog */}
      {selectedTemplate && (
        <TemplateViewDialog
          open={openView}
          handleClose={closeAllDialogs}
          template={selectedTemplate}
        />
      )}

      {/* Status Change Dialog */}
      <Dialog open={openStatusEdit} onClose={closeAllDialogs} maxWidth="xs" fullWidth>
        <DialogTitle>Update Status</DialogTitle>
        <DialogContent>
            <Box sx={{ mt: 1 }}>
                <Typography variant="subtitle2" gutterBottom>
                    Current: <strong>{selectedTemplate?.status || 'N/A'}</strong>
                </Typography>
                <TextField
                    select
                    fullWidth
                    label="Select New Status"
                    value={newStatus}
                    onChange={(e) => setNewStatus(e.target.value)}
                    size="small"
                    sx={{ mt: 2 }}
                >
                    <MenuItem value="ACTIVE">ACTIVE</MenuItem>
                    <MenuItem value="INACTIVE">INACTIVE</MenuItem>
                    <MenuItem value="DRAFT">DRAFT</MenuItem>
                </TextField>
            </Box>
        </DialogContent>
        <DialogActions>
            <Button onClick={closeAllDialogs}>Cancel</Button>
            <Button onClick={confirmStatusUpdate} variant="contained" size="small">Update</Button>
        </DialogActions>
      </Dialog>

      {/* General Edit Placeholder Dialog */}
       <Dialog open={openGeneralEdit} onClose={closeAllDialogs}>
        <DialogTitle>Edit Template</DialogTitle>
        <DialogContent>
            <DialogContentText>
                Functionality to edit template details (Name, Description, Params) goes here.
            </DialogContentText>
        </DialogContent>
        <DialogActions>
            <Button onClick={closeAllDialogs}>Close</Button>
        </DialogActions>
      </Dialog>


      {/* Delete Confirmation */}
      <Dialog open={openDelete} onClose={closeAllDialogs}>
        <DialogTitle sx={{ display: 'flex', alignItems: 'center', gap: 1 }}>
          <ReportProblemOutlined color="error" /> Confirm Delete
        </DialogTitle>
        <DialogContent>
          <DialogContentText>
            Are you sure you want to delete <strong>{selectedTemplate?.templateName}</strong>?
          </DialogContentText>
        </DialogContent>
        <DialogActions sx={{ p: 2 }}>
          <Button onClick={closeAllDialogs}>Cancel</Button>
          <Button onClick={confirmDelete} color="error" variant="contained">
            Delete
          </Button>
        </DialogActions>
      </Dialog>
    </Paper>
  );
};

export default TemplateTable;
