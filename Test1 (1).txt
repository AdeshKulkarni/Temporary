package com.fincore.JournalService.Controllers;

import com.fasterxml.jackson.core.JsonProcessingException;
import com.fasterxml.jackson.databind.JsonNode;

import com.fincore.JournalService.Dto.*;
import com.fincore.JournalService.Models.JournalRequest;
import com.fincore.JournalService.Service.JournalBulkValidationService;
import com.fincore.JournalService.Service.JournalRequestService;
import com.fincore.commonutilities.jwt.JwtUtil;

import jakarta.validation.Valid;
import lombok.RequiredArgsConstructor;
import lombok.extern.slf4j.Slf4j;

// Standard Spring Imports
import org.springframework.http.HttpStatus;
import org.springframework.http.ResponseEntity;
import org.springframework.http.MediaType;

import org.springframework.web.bind.annotation.*;
import org.springframework.web.multipart.MultipartFile;

import java.io.IOException;
import java.time.LocalDate;
import java.time.format.DateTimeFormatter;
import java.util.Base64;
import java.util.HashMap;
import java.util.List;
import java.util.Map;

@RestController
@RequestMapping("journals")
@RequiredArgsConstructor
@Slf4j
public class JournalRequestController {

	private final JournalRequestService journalRequestService;
	private final JournalBulkValidationService journalBulkValidationService;

	@GetMapping("/current-posting-date")
	public String getCurrentPostingDate() {
		return LocalDate.now().format(DateTimeFormatter.ISO_LOCAL_DATE);
	}

	 @PostMapping(value = "/bulk-validate", consumes = MediaType.MULTIPART_FORM_DATA_VALUE)
	    public ResponseEntity<?> validateBulkUpload(
	            @RequestParam("file") MultipartFile file,
	            @RequestParam("postingDate") String postingDateStr
	    ) throws IOException {
	        
	        LocalDate postingDate = LocalDate.parse(postingDateStr);

	        Map<String, Object> result = journalBulkValidationService.validateAndProcessFile(file, postingDate);

	        byte[] fileContent = (byte[]) result.get("file");
	        String base64File = (fileContent != null) ? Base64.getEncoder().encodeToString(fileContent) : null;

	        if ("ERROR".equals(result.get("status"))) {
	            long count = (long) result.get("errorCount");
	            
	            // Use HashMap to allow potential nulls (though unlikely here)
	            Map<String, Object> response = new HashMap<>();
	            response.put("status", "ERROR");
	            response.put("message", "Validation failed. Found " + count + " errors.");
	            response.put("errorCount", count);
	            response.put("errorReportBase64", base64File);
	            
	            return ResponseEntity.ok(response);
	        }

	        // SUCCESS CASE
	        // FIX: Map.of() throws NPE if value is null. base64File is NULL for CSVs.
	        Map<String, Object> response = new HashMap<>();
	        response.put("status", "SUCCESS");
	        response.put("message", "Validation Successful!");
	        response.put("successFileBase64", base64File); // Safe to put null in HashMap
	        response.put("previewData", result.get("previewData"));

	        return ResponseEntity.ok(response);
	    }

	// ------------------------------------------

	@PostMapping("/create-batch")
	@ResponseStatus(HttpStatus.CREATED)
	public List<JournalRequest> createBatchRequest(@Valid @RequestBody BatchRequestDto batchDto,
			@RequestHeader("Authorization") String token) throws JsonProcessingException {

		String userId = JwtUtil.getUserIdFromToken(token);
		JsonNode claims = JwtUtil.getClaimsFromToken(token);
		Integer userRole = claims.get("role").asInt();

		log.info("Received Journal batch request creation call from user: {}, role: {}", userId, userRole);

		return journalRequestService.createBatchRequest(batchDto, userId, userRole);
	}

	@PatchMapping("/update-request")
	public JournalRequest updateRequestStatus(@RequestHeader("Authorization") String token,
			@RequestBody ProcessJournalRequestDto processRequestDto) throws JsonProcessingException {

		String executorId = JwtUtil.getUserIdFromToken(token);
		JsonNode claims = JwtUtil.getClaimsFromToken(token);
		Integer executorRole = claims.get("role").asInt();

		log.info("Received Journal update request call from user: {}, role: {}", executorId, executorRole);

		return journalRequestService.updateRequestStatus(processRequestDto, executorId, executorRole).get();
	}

	@GetMapping("/my-requests")
	public List<JournalRequest> getMyRequests() {
		return journalRequestService.getMyRequests();
	}

	@GetMapping("/pending-requests")
	public List<JournalRequest> getPendingRequests(@RequestHeader("Authorization") String token) {
		String userId = JwtUtil.getUserIdFromToken(token);
		JsonNode claims = JwtUtil.getClaimsFromToken(token);
		Integer userRole = claims.get("role").asInt();
		return journalRequestService.getPendingRequests(userId, userRole);
	}

	@PostMapping("/process-bulk")
	public List<JournalRequest> processBulkRequests(@RequestHeader("Authorization") String token,
			@Valid @RequestBody BulkProcessJournalRequestDto processRequestDto) throws JsonProcessingException {

		String executorId = JwtUtil.getUserIdFromToken(token);
		JsonNode claims = JwtUtil.getClaimsFromToken(token);
		Integer executorRole = claims.get("role").asInt();
		return journalRequestService.processBulkRequests(processRequestDto, executorId, executorRole);
	}

	@GetMapping("/by-batch/{batchId}")
	public ResponseEntity<List<JournalRequest>> getRequestsByBatchId(@PathVariable String batchId) {
		List<JournalRequest> requests = journalRequestService.getRequestsByBatchId(batchId);
		return ResponseEntity.ok(requests);
	}

	@DeleteMapping("/my-request/{requestId}")
	public ResponseEntity<Void> cancelMyRequest(@RequestHeader("Authorization") String token,
			@PathVariable Long requestId) {
		String userId = JwtUtil.getUserIdFromToken(token);
		journalRequestService.cancelMyRequest(requestId, userId);
		return ResponseEntity.noContent().build();
	}

	@DeleteMapping("/my-requests/by-journal/{journalIdPrefix}")
	public ResponseEntity<Void> cancelMyRequestsByJournal(@RequestHeader("Authorization") String token,
			@PathVariable String journalIdPrefix) {
		String userId = JwtUtil.getUserIdFromToken(token);
		journalRequestService.cancelMyRequestsByJournalPrefix(journalIdPrefix, userId);
		return ResponseEntity.noContent().build();
	}

	@DeleteMapping("/my-requests/by-batch/{batchId}")
	public ResponseEntity<Void> cancelMyRequestsByBatch(@RequestHeader("Authorization") String token,
			@PathVariable String batchId) throws JsonProcessingException {
		String userId = JwtUtil.getUserIdFromToken(token);
		journalRequestService.cancelMyRequestsByBatchId(batchId, userId);
		return ResponseEntity.noContent().build();
	}

	@DeleteMapping("/my-requests/by-journal-list")
	public ResponseEntity<Void> cancelMyRequestsByJournalPrefixes(@RequestHeader("Authorization") String token,
			@RequestBody List<String> journalIdPrefixes) throws JsonProcessingException {
		String userId = JwtUtil.getUserIdFromToken(token);
		journalRequestService.cancelMyRequestsByJournalPrefixes(journalIdPrefixes, userId);
		return ResponseEntity.noContent().build();
	}

	@GetMapping("/status")
	public ResponseEntity<List<JournalRequestStatusDto>> getJournalStatusList() {
		return ResponseEntity.ok(journalRequestService.getJournalRequestStatusList());
	}

	@PostMapping("/create-bulk-batch")
	@ResponseStatus(HttpStatus.CREATED)
	public ResponseEntity<Map<String, Object>> createBulkBatchRequest(@RequestBody BatchRequestDto batchDto,
			@RequestHeader("Authorization") String token) throws JsonProcessingException {

		String userId = JwtUtil.getUserIdFromToken(token);
		JsonNode claims = JwtUtil.getClaimsFromToken(token);
		Integer userRole = claims.get("role").asInt();

		log.info("Received BULK Journal batch request from user: {}. Rows: {}", userId, batchDto.getRows().size());

		String batchId = journalRequestService.createBulkBatchRequest(batchDto, userId, userRole);

		return ResponseEntity.ok(Map.of("status", "SUCCESS", "message", "Batch created successfully", "batchId",
				batchId, "totalRows", batchDto.getRows().size()));
	}
}
