    private void parseAmountAndType(ExcelRowData data, String amountRaw, String typeRaw) {
        try {
            if (amountRaw == null) amountRaw = "";
            if (typeRaw == null) typeRaw = "";

            // 1. Clean the number: Remove everything except digits and dots.
            // This explicitly IGNORES the '-' sign for the value parsing.
            String cleanAmountStr = CLEAN_AMOUNT_REGEX.matcher(amountRaw).replaceAll("");
            
            if (cleanAmountStr.isEmpty()) {
                 data.amount = BigDecimal.ZERO;
                 return;
            }
            
            // 2. Get Absolute Value
            BigDecimal amt = new BigDecimal(cleanAmountStr);

            // 3. Determine Debit or Credit based ONLY on Indicators
            boolean isCredit = false;
            String typeUpper = typeRaw.toUpperCase();
            String amtUpper = amountRaw.toUpperCase();

            // Check TxnType Column
            if (typeUpper.contains("C") || typeUpper.contains("CR") || typeUpper.contains("CREDIT")) {
                isCredit = true;
            }
            // Check Amount Column Text (e.g. "500 Cr")
            else if (amtUpper.contains("CR")) {
                isCredit = true;
            }
            // Note: We intentionally DO NOT check for "-" sign here as requested.

            // 4. Set Direction
            // In system: Credit is Negative, Debit is Positive
            data.amount = isCredit ? amt.abs().negate() : amt.abs();
            
            // 5. Store standardized Type for display
            data.txnType = isCredit ? "Credit" : "Debit";

        } catch (Exception e) {
            data.amount = BigDecimal.ZERO; 
            data.errors.add("Invalid Amount Format");
        }
    }

    // ... (Keep existing CSV/Excel parsing loops, just call parseAmountAndType) ...

        if (payloadDto.getChangeType() == ChangeType.ADD) {
            GlTransaction newGlTxn = new GlTransaction();

            newGlTxn.setJournalId(payloadDto.getJournalId());
            
            if (payloadDto.getPDate() != null) {

            newGlTxn.setCurrency(payloadDto.getCurrency());
            newGlTxn.setCgl(payloadDto.getCgl());
            newGlTxn.setNarration(payloadDto.getRemarks());
            
            // --- FIX: DEBIT / CREDIT CALCULATION ---
            BigDecimal amount = payloadDto.getAmount();
            
            // If negative, it is CREDIT. Store Absolute Value in Credit Column.
            if (amount.signum() < 0) { 
                newGlTxn.setCreditAmount(amount.abs()); // Corrected: Store positive magnitude
                newGlTxn.setDebitAmount(BigDecimal.ZERO);

            // If positive, it is DEBIT.

                newGlTxn.setDebitAmount(amount);
                newGlTxn.setCreditAmount(BigDecimal.ZERO);
            }
            // ----------------------------------------
            
            newGlTxn.setTransactionCount(payloadDto.getTransactionCount());
            newGlTxn.setSourceFlag("J"); 
            
            glTransactionRepository.save(newGlTxn);
        }
