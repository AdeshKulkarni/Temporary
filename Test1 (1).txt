import React, { useState, useEffect, useCallback } from "react";
import {
  Box,
  Typography,
  Grid,
  TextField,
  Button,
  Paper,
  IconButton,
  MenuItem,
  CircularProgress,
  Autocomplete,
} from "@mui/material";
import AddIcon from "@mui/icons-material/Add";
import DeleteIcon from "@mui/icons-material/Delete";
import { debounce } from "lodash";

/**
 * MOCK API
 * I've updated the CGL search to *actually* filter by classification
 * so you can see your logic working correctly.
 */
const useApi = () => {
  const callApi = async (endpoint, payload, method = "GET") => {
    console.log(`Calling API: ${method} ${endpoint}`, payload);
    await new Promise((res) => setTimeout(res, 500)); // Simulate network delay

    if (endpoint.startsWith("/branches/search")) {
      const query = new URLSearchParams(endpoint.split("?")[1]).get("query");
      if (!query) return [];
      const items = [
        { code: "1001", name: "Main Branch" },
        { code: "1002", name: "City Center Branch" },
        { code: "2005", name: "Industrial Estate" },
      ];
      return items.filter(
        (item) =>
          item.name.toLowerCase().includes(query.toLowerCase()) ||
          item.code.includes(query)
      );
    }
    if (endpoint === "/currency") {
      return {
        success: true,
        data: ["USD", "EUR", "INR", "GBP", "USD", "INR"], // Duplicates are fine
      };
    }
    if (endpoint === "/cgls/classification") {
      return {
        A: "Asset",
        E: "Expense",
        I: "Income",
        L: "Liability",
        M: "Memo A/c",
      };
    }
    if (endpoint.startsWith("/cgls/search")) {
      const params = new URLSearchParams(endpoint.split("?")[1]);
      const query = params.get("query");
      const classification = params.get("classification"); // Get the classification
      
      if (!query) return [];

      const mockCgls = {
        A: [
          { cglNumber: "11001", description: "Asset - Cash" },
          { cglNumber: "11002", description: "Asset - Buildings" },
        ],
        L: [
          { cglNumber: "21001", description: "Liability - Loans" },
          { cglNumber: "22002", description: "Liability - Deposits" },
        ],
        I: [
          { cglNumber: "31001", description: "Income - Interest" },
          { cglNumber: "32002", description: "Income - Fees" },
        ],
        E: [
          { cglNumber: "41001", description: "Expense - Salaries" },
          { cglNumber: "42002", description: "Expense - Rent" },
        ],
        M: [
          { cglNumber: "91001", description: "Memo - Guarantees" },
        ],
      };

      const items = mockCgls[classification] || [];
      
      return items.filter(
        (item) =>
          item.description.toLowerCase().includes(query.toLowerCase()) ||
          item.cglNumber.includes(query)
      );
    }
    if (endpoint === "/submit-moc") {
      console.log("Submitting MOC Data:", payload);
      return { success: true, message: "MOC Posting submitted successfully." };
    }

    return null;
  };
  return { callApi };
};

/**
 * Mock Snackbar
 */
const useCustomSnackbar = () => {
  return (message, severity) => {
    // This is a placeholder. In a real app, you'd use a Snackbar Context.
    console.log(`[${severity.toUpperCase()}]: ${message}`);
    alert(`[${severity.toUpperCase()}]: ${message}`);
  };
};

/**
 * Initial Row State
 */
const initialRowState = {
  id: Date.now(),
  branch: null,
  postDate: new Date().toISOString().split("T")[0],
  currency: "",
  head: "",
  cgl: null,
  amount: "",
  txnType: "Debit",
  remarks: "",
  productCode: "",
};

/**
 * MOC Posting Component
 */
export default function MOCPosting() {
  const { callApi } = useApi();
  const showSnackBar = useCustomSnackbar();

  const [rows, setRows] = useState([initialRowState]);
  const [currencyOptions, setCurrencyOptions] = useState([]);
  const [headOptions, setHeadOptions] = useState([]);
  const [isSubmitting, setIsSubmitting] = useState(false);

  // Separate search state for each Autocomplete
  const [branchSearch, setBranchSearch] = useState({
    loading: false,
    options: [],
  });
  
  // We need separate CGL search states for *each row*
  // But for this example, we'll keep one global one.
  // A better pattern would be to move row state into its own component.
  const [cglSearch, setCglSearch] = useState({ loading: false, options: [] });

  // --- Initial Data Load ---
  useEffect(() => {
    const fetchInitialData = async () => {
      try {
        // Currency: Your logic was already correct!
        const currencyResponse = await callApi("/currency", null, "GET");
        if (currencyResponse?.success && Array.isArray(currencyResponse.data)) {
          const uniqueCurrencies = [...new Set(currencyResponse.data)];
          setCurrencyOptions(uniqueCurrencies);
        } else {
          showSnackBar("Failed to load currencies.", "error");
        }

        // Head
        const headResponse = await callApi("/cgls/classification", null, "GET");
        if (headResponse) {
          const formattedHeads = Object.entries(headResponse).map(
            ([key, value]) => ({ key, value })
          );
          setHeadOptions(formattedHeads);
        } else {
          showSnackBar("Failed to load head classifications.", "error");
        }
      } catch (error) {
        console.error("Error fetching initial data:", error);
        showSnackBar("An error occurred while loading page data.", "error");
      }
    };

    fetchInitialData();
  }, [callApi, showSnackBar]); // Dependencies added

  const handleAddRow = () => {
    setRows([...rows, { ...initialRowState, id: Date.now() }]);
  };

  const handleDeleteRow = (id) => {
    if (rows.length > 1) {
      setRows(rows.filter((row) => row.id !== id));
    } else {
      showSnackBar("Cannot delete the last row.", "warning");
    }
  };

  const handleRowChange = (id, field, value) => {
    const newRows = rows.map((row) => {
      if (row.id === id) {
        // If head changes, clear the CGL selection
        if (field === "head" && row.head !== value) {
          setCglSearch({ loading: false, options: [] }); // Clear old CGL options
          return { ...row, [field]: value, cgl: null };
        }
        return { ...row, [field]: value };
      }
      return row;
    });
    setRows(newRows);
  };

  // --- API Search Functions ---
  const searchBranches = useCallback(
    debounce(async (query) => {
      if (!query) {
        setBranchSearch({ loading: false, options: [] });
        return;
      }
      setBranchSearch((prev) => ({ ...prev, loading: true }));
      try {
        const response = await callApi(
          `/branches/search?query=${query}`,
          null,
          "GET"
        );
        setBranchSearch({ loading: false, options: response || [] });
      } catch (error) {
        console.error("Branch search failed:", error);
        setBranchSearch({ loading: false, options: [] });
      }
    }, 300),
    [callApi] // Dependency added
  );

  const searchCgls = useCallback(
    debounce(async (query, headKey) => {
      if (!query || !headKey) {
        setCglSearch({ loading: false, options: [] });
        return;
      }
      setCglSearch((prev) => ({ ...prev, loading: true }));
      try {
        // --- FIX ---
        // Now we correctly pass BOTH the query and the classification to the API
        const response = await callApi(
          `/cgls/search?query=${query}&classification=${headKey}`,
          null,
          "GET"
        );
        setCglSearch({ loading: false, options: response || [] });
      } catch (error) {
        console.error("CGL search failed:", error);
        setCglSearch({ loading: false, options: [] });
      }
    }, 300),
    [callApi] // Dependency added
  );

  const handleSubmit = async () => {
    setIsSubmitting(true);
    // Validation
    for (const row of rows) {
      if (
        !row.branch ||
        !row.currency ||
        !row.head ||
        !row.cgl ||
        !row.amount ||
        !row.txnType ||
        !row.productCode
      ) {
        showSnackBar(
          "Please fill all fields in all rows before submitting.",
          "error"
        );
        setIsSubmitting(false);
        return;
      }
    }

    try {
      await callApi("/submit-moc", { transactions: rows }, "POST");
      showSnackBar("MOC Posting submitted successfully!", "success");
      setRows([{ ...initialRowState, id: Date.now() }]); // Reset form
    } catch (error) {
      showSnackBar("Submission failed. Please try again.", "error");
    } finally {
      setIsSubmitting(false);
    }
  };

  return (
    <Box sx={{ p: 3, maxWidth: "95%", margin: "auto" }}>
      <Paper elevation={3} sx={{ p: 3 }}>
        <Typography
          variant="h4"
          gutterBottom
          sx={{ fontWeight: "bold", color: "primary.main", mb: 3 }}
        >
          MOC Posting
        </Typography>

        {rows.map((row) => (
          <Grid
            container
            spacing={2}
            key={row.id}
            alignItems="center"
            sx={{ mb: 2 }}
          >
            {/* Branch */}
            <Grid item xs={12} md={2}>
              <Autocomplete
                options={branchSearch.options}
                getOptionLabel={(option) => `${option.code} - ${option.name}`}
                value={row.branch}
                onChange={(event, newValue) =>
                  handleRowChange(row.id, "branch", newValue)
                }
                onInputChange={(event, newInputValue) =>
                  searchBranches(newInputValue)
                }
                loading={branchSearch.loading}
                // --- FIX ---
                // Disables built-in filtering, shows only API results
                filterOptions={(x) => x} 
                // --- BEST PRACTICE ---
                // Helps Autocomplete understand your object
                isOptionEqualToValue={(option, value) => option.code === value.code} 
                renderInput={(params) => (
                  <TextField
                    {...params}
                    label="Branch"
                    placeholder="Search by name or code"
                    InputProps={{
                      ...params.InputProps,
                      endAdornment: (
                        <>
                          {branchSearch.loading ? (
                            <CircularProgress color="inherit" size={20} />
                          ) : null}
                          {params.InputProps.endAdornment}
                        </>
                      ),
                    }}
                  />
                )}
              />
            </Grid>

            {/* Post Date */}
            <Grid item xs={12} md={1.5}>
              <TextField
                label="Post Date"
                value={row.postDate}
                fullWidth
                InputProps={{
                    readOnly: true, // Use readOnly instead of disabled for styling
                }}
                sx={{ backgroundColor: "#f4f4f4" }} // Lighter gray
              />
            </Grid>

            {/* Currency */}
            <Grid item xs={12} md={1}>
              <TextField
                select
                label="Currency"
                value={row.currency}
                onChange={(e) =>
                  handleRowChange(row.id, "currency", e.target.value)
                }
                fullWidth
              >
                {currencyOptions.map((c) => (
                  <MenuItem key={c} value={c}>
                    {c}
                  </MenuItem>
                ))}
              </TextField>
            </Grid>

            {/* Head */}
            <Grid item xs={12} md={1.5}>
              <TextField
                select
                label="Head"
                value={row.head}
                onChange={(e) =>
                  handleRowChange(row.id, "head", e.target.value)
                }
                fullWidth
              >
                {headOptions.map((h) => (
                  <MenuItem key={h.key} value={h.key}>
                    {h.value}
                  </MenuItem>
                ))}
              </TextField>
            </Grid>

            {/* CGL */}
            <Grid item xs={12} md={2}>
              <Autocomplete
                options={cglSearch.options}
                getOptionLabel={(option) =>
                  `${option.cglNumber} - ${option.description}`
                }
                value={row.cgl}
                onChange={(event, newValue) =>
                  handleRowChange(row.id, "cgl", newValue)
                }
                onInputChange={(event, newInputValue) =>
                  searchCgls(newInputValue, row.head)
                }
                loading={cglSearch.loading}
                disabled={!row.head}
                // --- FIX ---
                filterOptions={(x) => x}
                // --- BEST PRACTICE ---
                isOptionEqualToValue={(option, value) => option.cglNumber === value.cglNumber}
                renderInput={(params) => (
                  <TextField
                    {...params}
                    label="CGL"
                    placeholder={row.head ? "Search CGL" : "Select Head first"}
                    InputProps={{
                      ...params.InputProps,
                      endAdornment: (
                        <>
                          {cglSearch.loading ? (
                            <CircularProgress color="inherit" size={20} />
                          ) : null}
                          {params.InputProps.endAdornment}
                        </>
                      ),
                    }}
                  />
                )}
              />
            </Grid>

            {/* Amount */}
            <Grid item xs={12} md={1.5}>
              <TextField
                label="Amount"
                type="text"
                value={row.amount}
                onChange={(e) => {
                  const val = e.target.value;
                  if (/^\d{0,16}(\.\d{0,6})?$/.test(val)) {
                    handleRowChange(row.id, "amount", val);
                  }
                }}
                fullWidth
                placeholder="0.00"
              />
            </Grid>

            {/* Txn Type */}
            <Grid item xs={12} md={1}>
              <TextField
                select
                label="Txn Type"
                value={row.txnType}
                onChange={(e) =>
                  handleRowChange(row.id, "txnType", e.target.value)
                }
                fullWidth
              >
                <MenuItem value="Debit">Debit</MenuItem>
                <MenuItem value="Credit">Credit</MenuItem>
              </TextField>
            </Grid>

            {/* Remarks */}
            <Grid item xs={12} md={2}>
              <TextField
                label="Remarks"
                value={row.remarks}
                onChange={(e) => {
                  const val = e.target.value;
                  // Allow alphanumeric and special chars
                  if (val.length <= 40) {
                     handleRowChange(row.id, "remarks", val);
                  }
                }}
                fullWidth
                inputProps={{ maxLength: 40 }}
              />
            </Grid>

            {/* Product Code */}
            <Grid item xs={12} md={1.5}>
              <TextField
                label="Product Code"
                value={row.productCode}
                onChange={(e) => {
                  const val = e.target.value;
                  if (/^\d{0,8}$/.test(val)) {
                    handleRowChange(row.id, "productCode", val);
                  }
                }}
                fullWidth
                inputProps={{ maxLength: 8 }}
              />
            </Grid>

            {/* Actions */}
            <Grid item xs={12} md={0.5} sx={{ textAlign: "center" }}>
              <IconButton
                onClick={() => handleDeleteRow(row.id)}
                color="error"
                aria-label="delete row"
              >
                <DeleteIcon />
              </IconButton>
            </Grid>
          </Grid>
        ))}

        <Box sx={{ mt: 3, display: "flex", gap: 2 }}>
          <Button
            variant="contained"
            startIcon={<AddIcon />}
            onClick={handleAddRow}
          >
            Add Row
          </Button>
          <Button
            variant="contained"
            color="success"
            onClick={handleSubmit}
            disabled={isSubmitting}
          >
            {isSubmitting ? <CircularProgress size={24} /> : "Submit"}
          </Button>
        </Box>
      </Paper>
    </Box>
  );
}
