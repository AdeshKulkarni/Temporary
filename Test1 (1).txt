import { Box, FormControl, Grid, InputLabel, MenuItem, Paper, Select, Typography, TextField, Button } from '@mui/material';
import { useEffect, useState } from 'react';
import useCustomSnackbar from '../../../../common/hooks/useCustomSnackbar';
import useApi from '../../../../common/hooks/useApi';
import LoadingOverlay from '../../../../common/components/ui/loadingOverlay';
import downloadFile from '../../../../common/components/viewer/DownloadFile';
import ReportViewer from '../../../../common/components/viewer/ReportViewer';

// Note: The original CustomButton is replaced with a styled MUI Button for a more integrated look.
// If you need to use your specific CustomButton, you can replace the <Button> component.

const emptyMap = { report: '', qed: '', circle: '' };

const Dashboard = () => {
  const snackbar = useCustomSnackbar();
  const { callApi } = useApi();
  const loginUser = JSON.parse(localStorage.getItem('user'));

  const [reportList, setReportList] = useState([]);
  const [qedList, setQedList] = useState([]);
  const [circleList, setCircleList] = useState([]);
  const [valueMap, setValueMap] = useState(emptyMap);
  const [loadOpen, setLoadOpen] = useState(false);
  const [viewData, setViewData] = useState(false);
  const [openViewDialog, setOpenViewDialog] = useState(false);

  useEffect(() => {
    fetchData().then(() => {
      console.log('data fetched.');
    });
  }, []);

  const handleChange = (key, value) => {
    const newValue = { ...valueMap };
    newValue[key] = value;
    setValueMap(newValue);
  };

  const fetchData = async () => {
    try {
      const responseData = await callApi('/dashboard/getDashboard', 62, 'POST', 'json', 'text/plain');
      if (
        responseData &&
        responseData.list.length > 0 &&
        responseData.list1.length > 0 &&
        responseData.list2.length > 0
      ) {
        setReportList(responseData.list);
        setQedList(responseData.list1);
        setCircleList(responseData.list2);
      } else {
        snackbar('Failed to fetch data, Kindly try again.', 'error');
      }
    } catch (error) {
      console.error('An error occured while fetching option values, Kindly try again after sometime ', error);
      snackbar('An error occured while fetching option values, Kindly try again after sometime. ', 'error');
    }
  };

  const handleAction = async (type) => {
    if (valueMap.report === '' || valueMap.qed === '' || valueMap.circle === '') {
      snackbar('Kindly select all the fields.', 'error');
      return;
    }

    setLoadOpen(true);

    const payload = {
      action: type,
      circleCode: valueMap.circle,
      jrxmlName: valueMap.report,
      quarterEndDate: valueMap.qed,
      user: loginUser,
    };

    const fileName = `${valueMap.circle}_${valueMap.qed.replace(/-/g, '_')}_${valueMap.report}`;
    try {
      const response = await callApi('/dashboard/downloadReports', payload, 'POST', 'arraybuffer');
      if (response) {
        switch (type) {
          case 'View':
            setViewData(response);
            setOpenViewDialog(true);
            break;
          case 'PDF':
            downloadFile(response, 'pdf', fileName);
            break;
          case 'Excel':
            downloadFile(response, 'excel', fileName);
            break;
          default:
            console.warn('Unhandled report action :: ', type);
        }
      } else {
        console.error('Failed to fetch data, kindly try again.');
        snackbar('Failed to fetch data, kindly try again.', 'error');
      }
    } catch (error) {
      console.error('An error occured, Kindly try again after sometime. ', error);
      snackbar('An error occured , Kindly try again after sometime. ', 'error');
    } finally {
        setLoadOpen(false);
    }
  };

  return (
    <Box sx={{ width: '100%', p: { xs: 2, sm: 3 } }}>
      <Paper elevation={4} sx={{ p: { xs: 2, sm: 3, md: 4 }, borderRadius: '12px' }}>
        <Typography variant="h4" component="h1" sx={{ fontWeight: 600, mb: 1 }}>
          GSTIN Wise Reports
        </Typography>
        <Typography variant="subtitle1" color="text.secondary" sx={{ mb: 5 }}>
          Select parameters to generate and view your report.
        </Typography>
        <Grid container spacing={4}>
          {/* Reports Dropdown */}
          <Grid item xs={12} md={4}>
            <TextField
              select
              fullWidth
              label="Select Report"
              value={valueMap.report}
              onChange={(e) => handleChange('report', e.target.value)}
              variant="outlined"
              SelectProps={{
                MenuProps: { PaperProps: { sx: { maxHeight: 300 } } }
              }}
            >
              <MenuItem value=""><em>---Select---</em></MenuItem>
              {reportList.map((report, index) => (
                <MenuItem key={`${report.dash_jrxml}-${index}`} value={report.dash_jrxml}>
                  {report.dash_name}
                </MenuItem>
              ))}
            </TextField>
          </Grid>

          {/* Quarter End Date Dropdown */}
          <Grid item xs={12} md={4}>
            <TextField
              select
              fullWidth
              label="Select Quarter End Date"
              value={valueMap.qed}
              onChange={(e) => handleChange('qed', e.target.value)}
              variant="outlined"
               SelectProps={{
                MenuProps: { PaperProps: { sx: { maxHeight: 300 } } }
              }}
            >
              <MenuItem value=""><em>---Select---</em></MenuItem>
              {qedList.map((qed, index) => (
                <MenuItem key={`${qed.paramValue}-${index}`} value={qed.paramValue}>
                  {qed.paramValue}
                </MenuItem>
              ))}
            </TextField>
          </Grid>

          {/* Circle Dropdown */}
          <Grid item xs={12} md={4}>
            <TextField
              select
              fullWidth
              label="Select Circle"
              value={valueMap.circle}
              onChange={(e) => handleChange('circle', e.target.value)}
              variant="outlined"
               SelectProps={{
                MenuProps: { PaperProps: { sx: { maxHeight: 300 } } }
              }}
            >
              <MenuItem value=""><em>---Select---</em></MenuItem>
              {circleList.map((circle, index) => (
                <MenuItem key={`${circle.paramValue}-${index}`} value={circle.paramValue}>
                  {circle.paramName}
                </MenuItem>
              ))}
            </TextField>
          </Grid>
        </Grid>

        {/* Action Button */}
        <Box sx={{ display: 'flex', justifyContent: 'center', mt: 6 }}>
          <Button
            variant="contained"
            color="primary"
            onClick={() => handleAction('View')}
            sx={{
              py: 1.5,
              px: 6,
              fontSize: '1rem',
              fontWeight: 600,
              textTransform: 'none',
              borderRadius: '8px',
              boxShadow: '0 4px 14px 0 rgba(0, 118, 255, 0.39)',
              '&:hover': {
                backgroundColor: 'primary.dark',
                boxShadow: '0 6px 20px 0 rgba(0, 118, 255, 0.23)'
              }
            }}
          >
            Submit
          </Button>
        </Box>
      </Paper>

      <LoadingOverlay loading={loadOpen} />

      {viewData && (
        <ReportViewer
          open={openViewDialog}
          onClose={() => setOpenViewDialog(false)}
          bufferData={viewData}
          type="html"
          onDownloadPDF={() => handleAction('PDF')}
          onDownloadExcel={() => handleAction('Excel')}
        />
      )}
    </Box>
  );
};

export default Dashboard;
