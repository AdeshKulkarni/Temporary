public class RuleRegistry {

   public static  Map<Event, List<Rules>> ruleRegistry;

   static  {
       ruleRegistry.put(Event.ENROLLMENT , List.of(new IsSameCource()));
       ruleRegistry.put(Event.PREREQUISITES , List.of(new MeetPrerequisites()));
       ruleRegistry.put(Event.RESTRICTIONS , List.of(new coursesRestricted() , new CourceCapacity()));
   }

  public static List<Rules> getRules(Event event) {
       return ruleRegistry.getOrDefault(event , List.of());
  }


}
