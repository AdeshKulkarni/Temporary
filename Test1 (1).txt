// Existing constants
const iv = crypto.getRandomValues(new Uint8Array(12));
const ivBase64 = btoa(String.fromCharCode.apply(null, iv));
const salt = crypto.getRandomValues(new Uint8Array(16));
const saltBase64 = btoa(String.fromCharCode.apply(null, salt));

// ✅ ADD THIS FUNCTION inside your component EmpanelmentIntimation
const handleDownloadZip = async () => {
  try {
    const selectedData = selectedRows.map((id) => rows.find((row) => row.id === id));
    const payload = {
      user: JSON.parse(localStorage.getItem("user")),
      data: selectedData,
    };

    const jsonFormData = JSON.stringify(payload);
    const encryptedData = await encrypt(iv, salt, jsonFormData);
    const requestData = {
      iv: ivBase64,
      salt: saltBase64,
      data: encryptedData,
    };

    const response = await axios.post("Server/IAMServices/downloadZip", requestData, {
      headers: {
        Authorization: `Bearer ${localStorage.getItem("token")}`,
      },
    });

    if (response.data?.statusCode === 200 && response.data.result) {
      downloadFile(response.data.result, "zip", "Selected_Letters");
    } else {
      console.error("Download failed");
    }
  } catch (error) {
    console.error("Zip download error", error);
  }
};
