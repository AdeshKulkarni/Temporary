class Solution {
    public int minSumOfLengths(int[] arr, int target) {
        int n = arr.length;
        HashMap<Integer, Integer> map = new HashMap<>();
        
        // 1. Keep your list, but upgrade it to hold [length, past_index, current_index]
        ArrayList<int[]> list = new ArrayList<>();
        
        map.put(0, -1);
        
        // YOUR EXACT LOGIC: Prefix array calculation
        int[] prefix = new int[n];
        prefix[0] = arr[0];
        for(int i = 1; i < n; i++) {
            prefix[i] = prefix[i-1] + arr[i];
        }
        
        // YOUR EXACT LOGIC: Main loop
        for(int i = 0; i < n; i++) {
            if(map.containsKey(prefix[i] - target)) {
                int val = map.get(prefix[i] - target);
                
                // 2. THE FIX: Instead of just saving the length, save the coordinates too
                list.add(new int[]{i - val, val, i});
            }
            // (Bug fix from earlier: this must be saved for every index)
            map.put(prefix[i], i);
        }
        
        // YOUR EXACT LOGIC: Sort the list (adapted to sort by the length at index 0)
        list.sort((a, b) -> Integer.compare(a[0], b[0]));
        
        // YOUR EXACT LOGIC: if(list.size() > 1) check
        if(list.size() > 1) {
            int minSum = Integer.MAX_VALUE;
            
            // 3. THE OVERLAP CHECK: Find the two smallest that do not cross paths
            for(int i = 0; i < list.size(); i++) {
                for(int j = i + 1; j < list.size(); j++) {
                    int[] sub1 = list.get(i);
                    int[] sub2 = list.get(j);
                    
                    // If sub1 ends before or when sub2 starts (sub1[2] <= sub2[1]) 
                    // OR sub2 ends before or when sub1 starts (sub2[2] <= sub1[1])
                    if(sub1[2] <= sub2[1] || sub2[2] <= sub1[1]) {
                        minSum = Math.min(minSum, sub1[0] + sub2[0]);
                    }
                }
            }
            return minSum == Integer.MAX_VALUE ? -1 : minSum;
        }
        
        return -1; // YOUR EXACT LOGIC: Return -1 if list is too small
    }
}
