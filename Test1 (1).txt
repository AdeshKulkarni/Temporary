{
    "statusCode": 200,
    "message": "Success fetched Assignment Data",
    "result": [
        {
            "IA": "Stock Audit,Tax Audit,Concurrent Audit,Risk Audit,Internal Audit",
            "SA": "Statutory Audit,Branch Audit"
        }
    ]
}
import React, { useState, useRef, useEffect, useMemo } from "react";
import {
  Button,
  Paper,
  IconButton,
  Typography,
  Dialog,
  DialogTitle,
  DialogContent,
  Divider,
  Checkbox,
  Box,
  TextField,
  MenuItem,
  Stack,
} from "@mui/material";
import { DataGrid, GridToolbar } from "@mui/x-data-grid";
import { Close } from "@mui/icons-material";
import PreviewIcon from "@mui/icons-material/Preview";
import { styled } from "@mui/system";
import axios from "axios";
import { encrypt } from "../Security/AES-GCM256";
import PdfViewer from "../PdfViewer/pdfViewer";
import downloadFile from "../DownloadFile/DownloadFile";
import DownloadForOfflineIcon from "@mui/icons-material/DownloadForOffline";

const iv = crypto.getRandomValues(new Uint8Array(12));
const ivBase64 = btoa(String.fromCharCode.apply(null, iv));
const salt = crypto.getRandomValues(new Uint8Array(16));
const saltBase64 = btoa(String.fromCharCode.apply(null, salt));

const BootstrapDialog = styled(Dialog)(({ theme }) => ({
  "& .MuiDialogContent-root": {
    padding: theme.spacing(2),
  },
  "& .MuiDialogActions-root": {
    padding: theme.spacing(1),
  },
}));

const WorkFlow = () => {
  const dataRef = useRef();
  const [selectedRows, setSelectedRows] = useState([]);
  const [previewData, setPreviewData] = useState(null);
  const [rows, setRows] = useState([]);
  const [loading, setLoading] = useState(true);
  const [templateContent, setTemplateContent] = useState("");
  const [openView, setOpenView] = useState(false);
  const [templateLoading, setTemplateLoading] = useState(false);
  const [selectionOptions, setSelectionOptions] = useState([]);
  const [userAccess, setUserAccess] = useState("");
  const [selectedSubType, setSelectedSubType] = useState("");

  console.log(selectionOptions);

  // Get user access from localStorage
  useEffect(() => {
    const user = JSON.parse(localStorage.getItem("user"));
    if (user && user.access) {
      const accessType = user.access.startsWith("S") ? "SA" : "IA";
      setUserAccess(accessType);
    }
  }, []);

  // Fetch subtype options
  const fetchOptions = async () => {
    try {
      const response = await axios.post(
        "Server/IAMServices/getAssignmentType",
        {},
        {
          headers: {
            Authorization: `Bearer ${localStorage.getItem("token")}`,
          },
        }
      );
      const result = response.data.result[0] || {};
      let user = "";
      if (localStorage.getItem("user").access == "SAC") {
        user = "SA";
      } else if (localStorage.getItem("user").access == "IAC") {
        {
          user = "IA";
        }
      }
      // Object.keys(result).forEach((type) => {
      //   options[type] = result[type]
      //     ? result[type].split(",").map((item) => item.trim())
      //     : [];
      // });

      setSelectionOptions(result[user].split(","));
    } catch (e) {
      console.error(e);
    }
  };

  useEffect(() => {
    fetchOptions();
  }, []);

  // Fetch data based on user access and selected subtype
  useEffect(() => {
    const fetchData = async () => {
      if (!userAccess) return;

      let dataToSend = {
        Type: "Active",
        EMPANELMENT_TYPE: userAccess,
        EMPANELMENT_SUB_TYPE: selectedSubType || "",
      };

      try {
        setLoading(true);
        let jsonFormData = JSON.stringify(dataToSend);
        const encryptedData = await encrypt(iv, salt, jsonFormData);
        const payload = { iv: ivBase64, salt: saltBase64, data: encryptedData };

        const response = await axios.post(
          "Server/IAMServices/getActiveFRN",
          payload,
          {
            headers: {
              Authorization: `Bearer ${localStorage.getItem("token")}`,
            },
          }
        );

        const resData = response.data;
        if (resData.statusCode === 200) {
          const formattedData = resData.result.map((item, index) => ({
            id: index + 1,
            serial: index + 1,
            frn: item.FRN,
            firmName: item.FIRM_NAME,
            empanelmentType: item.EMPANELMENT_TYPE || "NA",
            empanelmentSubType: item.EMPANELMENT_SUB_TYPE || "NA",
            empaneledStatus: item.EMPANELED_STATUS || "NA",
            empaneledBy: item.EMPANELED_BY || "NA",
          }));
          setRows(formattedData);
        } else {
          console.error("API Error:", resData.message);
        }
      } catch (e) {
        console.error("Fetch Error:", e);
      } finally {
        setLoading(false);
      }
    };

    fetchData();
  }, [userAccess, selectedSubType]);

  const handleRowPreview = async (rowData) => {
    setPreviewData(rowData);
    setTemplateLoading(true);

    try {
      const templateData = {
        user: { template: "Hi " },
        data: {
          DATE: new Date().toLocaleDateString(),
          REF_NO: "1001",
          FIRM_NAME: rowData.firmName,
          FRN_NO: rowData.frn,
          GSTN: "27AAHCK850D1ZK",
          FIRM_ADDR: "NAVI MUMBAI, BELPAUR",
          ASSIGNMENT_TYPE: `${rowData.empanelmentType}, ${rowData.empanelmentSubType}`,
        },
      };

      let jsonFormData = JSON.stringify(templateData);
      const encryptedData = await encrypt(iv, salt, jsonFormData);
      const payload = { iv: ivBase64, salt: saltBase64, data: encryptedData };

      const response = await axios.post(
        "Server/IAMServices/generateTemplate",
        payload,
        {
          headers: {
            Authorization: `Bearer ${localStorage.getItem("token")}`,
          },
        }
      );

      if (response.data.statusCode === 200) {
        if (response.data.result) {
          setTemplateContent(response.data.result);
        } else {
          setTemplateContent("");
        }
      } else {
        console.error("Template API Error:", response.data.message);
      }
    } catch (e) {
      console.error("Template Fetch Error:", e);
    } finally {
      setTemplateLoading(false);
    }
  };

  // Define all the missing functions
  const handleClosePreview = () => {
    setPreviewData(null);
    setTemplateContent("");
  };

  const handleSelectRow = (id) => {
    setSelectedRows((prev) =>
      prev.includes(id) ? prev.filter((rowId) => rowId !== id) : [...prev, id]
    );
  };

  const handleSelectAll = () => {
    if (selectedRows.length === rows.length) {
      setSelectedRows([]);
    } else {
      setSelectedRows(rows.map((row) => row.id));
    }
  };

  const handleEmail = async () => {
    try {
      let checked = selectedRows.map((id) => rows[id - 1]);
      console.log(checked);
      const selected = {
        user: localStorage.getItem("user"),
        data: checked,
      };
      let jsonFormData = JSON.stringify(selected);
      const encryptedData = await encrypt(iv, salt, jsonFormData);
      const payload = { iv: ivBase64, salt: saltBase64, data: encryptedData };

      const response = await axios.post(
        "Server/IAMServices/sendEmail",
        payload,
        {
          headers: {
            Authorization: `Bearer ${localStorage.getItem("token")}`,
          },
        }
      );

      if (response.data.statusCode !== 200) {
        console.error("Template API Error:", response.data.message);
      }
    } catch (e) {
      console.error("Template Fetch Error:", e);
    } finally {
      setTemplateLoading(false);
    }
  };

  const handleDownloadLetter = (base64) => {
    downloadFile(base64, "pdf", "letter");
  };

  // Grid columns
  const columns = [
    {
      field: "selected",
      headerName: "Select",
      width: 80,
      renderHeader: () => (
        <Checkbox
          checked={selectedRows.length === rows.length && rows.length !== 0}
          indeterminate={
            selectedRows.length > 0 && selectedRows.length < rows.length
          }
          onChange={handleSelectAll}
        />
      ),
      renderCell: (params) => (
        <Checkbox
          checked={selectedRows.includes(params.row.id)}
          onChange={() => handleSelectRow(params.row.id)}
        />
      ),
      sortable: false,
      filterable: false,
      disableColumnMenu: true,
    },
    { field: "serial", headerName: "Sr. No", width: 80 },
    { field: "frn", headerName: "FRN", width: 200 },
    { field: "firmName", headerName: "Firm Name", width: 250 },
    { field: "empanelmentType", headerName: "Empanelment Type", width: 180 },
    {
      field: "empanelmentSubType",
      headerName: "Empanelment Sub Type",
      width: 250,
    },
    { field: "empaneledStatus", headerName: "Status", width: 130 },
    {
      field: "actions",
      headerName: "Actions",
      width: 200,
      sortable: false,
      filterable: false,
      disableColumnMenu: true,
      renderCell: (params) => (
        <Button
          variant="outlined"
          color="primary"
          size="small"
          startIcon={<PreviewIcon />}
          onClick={() => handleRowPreview(params.row)}
          sx={{
            textTransform: "none",
            "&:hover": { backgroundColor: "primary.light" },
          }}
        >
          Letter Preview
        </Button>
      ),
    },
  ];

  return (
    <Box sx={{ p: { xs: 1, sm: 3 }, overflow: "hidden" }}>
      <Typography variant="h5" sx={{ mb: 2 }}>
        Empanelment Intimation
      </Typography>
      <Divider sx={{ mb: 2 }} />

      {userAccess && (
        <Stack direction="row" spacing={2} sx={{ mb: 3 }}>
          <TextField
            select
            label="Empanelment Sub Type"
            value={selectedSubType}
            onChange={(e) => setSelectedSubType(e.target.value)}
            sx={{ minWidth: 250 }}
            size="small"
          >
            <MenuItem value="">All Sub Types</MenuItem>
            {selectionOptions?.map((option) => (
              <MenuItem key={option} value={option}>
                {option}
              </MenuItem>
            ))}
          </TextField>
        </Stack>
      )}

      <Box sx={{ width: "100%", mb: 3 }}>
        <Paper
          elevation={2}
          sx={{
            width: "100%",
            p: 2,
            bgcolor: "#fafbfc",
            borderRadius: 2,
            boxShadow: 3,
          }}
        >
          <DataGrid
            ref={dataRef}
            rows={rows}
            columns={columns}
            loading={loading}
            getRowId={(row) => row.id}
            disableColumnMenu
            hideFooterSelectedRowCount
            components={{ Toolbar: GridToolbar }}
            componentsProps={{
              toolbar: {
                showQuickFilter: true,
                quickFilterProps: { debounceMs: 500 },
              },
            }}
            sx={{
              minHeight: 400,
              "& .MuiDataGrid-cell": { fontSize: "0.85rem" },
              "& .MuiDataGrid-columnHeader": {
                fontSize: "1rem",
                fontWeight: "bold",
              },
              bgcolor: "#fff",
            }}
            autoHeight
          />
        </Paper>
      </Box>
      <Box sx={{ textAlign: "right" }}>
        <Button variant="contained" sx={{ mr: 2 }} onClick={handleEmail}>
          Send Email
        </Button>
        <Button variant="contained">Download Zip</Button>
      </Box>
      <BootstrapDialog
        onClose={handleClosePreview}
        aria-labelledby="customized-dialog-title"
        open={previewData && templateContent}
        fullWidth
        maxWidth="lg"
      >
        <DialogTitle sx={{ m: 0, p: 2 }}>
          Empanelment Letter
          <Button
            variant="outlined"
            size="small"
            sx={{ float: "right", ml: 2, mr: 4 }}
            onClick={() => handleDownloadLetter(templateContent)}
          >
            <DownloadForOfflineIcon />
          </Button>
        </DialogTitle>
        <IconButton
          aria-label="close"
          onClick={handleClosePreview}
          sx={(theme) => ({
            position: "absolute",
            right: 8,
            top: 8,
            color: theme.palette.grey[500],
          })}
        >
          <Close />
        </IconButton>
        <DialogContent dividers>
          <PdfViewer pdfData={templateContent} sx={{ width: "100%" }} />
        </DialogContent>
      </BootstrapDialog>
    </Box>
  );
};

export default WorkFlow;
