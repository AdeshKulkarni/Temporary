package com.tcs.fincore.MOCService.Config;

import com.fasterxml.jackson.databind.ObjectMapper;
import com.fasterxml.jackson.databind.SerializationFeature;
import com.fasterxml.jackson.datatype.jsr310.JavaTimeModule;
import org.springframework.context.annotation.Bean;
import org.springframework.context.annotation.Configuration;
import org.springframework.context.annotation.Primary;

@Configuration
public class AppConfig {

    /**
     * Creates a primary ObjectMapper bean that is configured to handle
     * Java 8 Date/Time objects (LocalDate, LocalDateTime) correctly.
     * This module is required for serializing/deserializing date fields.
     */
    @Bean
    @Primary // Ensures this ObjectMapper bean is used by default in the application
    public ObjectMapper objectMapper() {
        ObjectMapper mapper = new ObjectMapper();
        
        // This is the critical line:
        // It registers the module that teaches Jackson how to handle Java 8 Time APIs.
        mapper.registerModule(new JavaTimeModule());
        
        // This is good practice:
        // It prevents Jackson from writing dates as numeric timestamps (e.g., 1678886400)
        // and instead uses the standard ISO-8601 string format (e.g., "2025-11-03").
        mapper.disable(SerializationFeature.WRITE_DATES_AS_TIMESTAMPS);
        
        return mapper;
    }
}
