    public byte[] generateBatchExcel(String batchId) {
        String safeBatchId = (batchId != null) ? batchId.replaceAll("[^0-9]", "") : "";
        
        try (SXSSFWorkbook workbook = new SXSSFWorkbook(100);
             java.io.ByteArrayOutputStream out = new java.io.ByteArrayOutputStream()) {

            org.apache.poi.ss.usermodel.Sheet sheet = workbook.createSheet("Batch_" + safeBatchId);
            org.apache.poi.ss.usermodel.Row headerRow = sheet.createRow(0);
            String[] columns = {"Branch", "Currency", "CGL", "Amount", "Dr/Cr", "Remarks", "Product Code"};

            for (int i = 0; i < columns.length; i++) {
                org.apache.poi.ss.usermodel.Cell cell = headerRow.createCell(i);
                cell.setCellValue(columns[i]);
            }

            String sql = "SELECT REQ_BRANCH, REQ_CURRENCY, REQ_CGL, REQ_AMOUNT, REQ_NARRATION, REQ_PRODUCT " +
                         "FROM voucher_requests WHERE batch_id = '" + safeBatchId + "'";

            final int[] rowIdx = {1};
            jdbcTemplate.query(
                    sql,
                    rs -> {
                        org.apache.poi.ss.usermodel.Row row = sheet.createRow(rowIdx[0]++);
                        row.createCell(0).setCellValue(rs.getString("REQ_BRANCH") != null ? rs.getString("REQ_BRANCH") : "");
                        row.createCell(1).setCellValue(rs.getString("REQ_CURRENCY") != null ? rs.getString("REQ_CURRENCY") : "");
                        row.createCell(2).setCellValue(rs.getString("REQ_CGL") != null ? rs.getString("REQ_CGL") : "");

                        BigDecimal amt = rs.getBigDecimal("REQ_AMOUNT");
                        if (amt != null) {
                            row.createCell(3).setCellValue(amt.abs().stripTrailingZeros().toPlainString());
                            row.createCell(4).setCellValue(amt.compareTo(BigDecimal.ZERO) < 0 ? "Credit" : "Debit");
                        }

                        row.createCell(5).setCellValue(rs.getString("REQ_NARRATION") != null ? rs.getString("REQ_NARRATION") : "");
                        row.createCell(6).setCellValue(rs.getString("REQ_PRODUCT") != null ? rs.getString("REQ_PRODUCT") : "");
                    }
            );

            if (rowIdx[0] == 1) {
                throw new CustomExceptions.InvalidDataException("No records found for batch.");
            }

            workbook.write(out);
            workbook.dispose();
            return out.toByteArray();

        } catch (java.io.IOException e) {
            throw new CustomExceptions.DatabaseOperationException("Failed to generate Excel file", e);
        }
    }
