const fetchData = async () => {
    try {
      const response = await axios.post(
        "Server/IAMServices/EmailTemplate",
        {},
        {
          headers: { Authorization: `Bearer ${localStorage.getItem("token")}` },
        }
      );
      let htmlResponse = response.data.result;

      // Create a temporary, in-memory element to parse the HTML string
      const container = document.createElement("div");
      container.innerHTML = htmlResponse;

      // Find our unique marker element
      const bodyStartNode = container.querySelector("#email-body-start");

      if (bodyStartNode) {
        let headerHtml = "";
        let bodyHtml = "";
        let bodyHasStarted = false;

        // Loop through all top-level nodes in the response
        for (const node of container.childNodes) {
          // Once we find our marker node, we start collecting body HTML
          if (node === bodyStartNode) {
            bodyHasStarted = true;
          }

          if (bodyHasStarted) {
            // node.outerHTML gets the full tag (e.g., "<p>...</p>")
            // node.textContent is a fallback for plain text nodes
            bodyHtml += node.outerHTML || node.textContent;
          } else {
            headerHtml += node.outerHTML || node.textContent;
          }
        }
        setHeaderContent(headerHtml);
        setBodyContent(bodyHtml);
      } else {
        // Fallback if the ID is not found in the template
        console.error("Could not find marker with id='email-body-start' in the template.");
        setHeaderContent(htmlResponse); // Show the full content as a fallback
        setBodyContent("");
      }
    } catch (e) {
      console.error("Failed to fetch email template:", e);
    }
  };
