// --- Quill Blots and Registrations (Unchanged) ---
const Embed = Quill.import("blots/embed");
class VariableBlot extends Embed {
  static create(value) {
    const node = super.create(value);
    node.setAttribute("contenteditable", "false");
    node.classList.add("variable-blot");
    const text = document.createElement("span");
    text.innerText = value;
    const closer = document.createElement("span");
    closer.innerText = "X";
    closer.classList.add("variable-delete-icon");
    node.appendChild(closer);
    node.appendChild(text);
    Object.assign(node.style, {
      position: "relative",
      color: "black",
      padding: "5px 20px 5px 8px",
      margin: "0 4px",
      cursor: "default",
      userSelect: "none",
      display: "inline-block",
    });
    return node;
  }
  static value(node) {
    // Ensure we return the variable text itself, not the delete icon or wrapper.
    // The first span child of the blot contains the variable text.
    return node.querySelector(".variable-blot > span:last-child")?.innerText;
  }
}

VariableBlot.blotName = "variable";
VariableBlot.tagName = "span";
Quill.register(VariableBlot);

/**
 * Transforms variable blots back into their raw text representation (e.g., {{VARIABLE}}).
 * This should be called *before* stripping classes or delete icons for saving.
 * @param {string} html - The HTML content from the editor.
 * @returns {string} The HTML content with variable blots converted to text.
 */
function convertVariablesToText(html) {
  const container = document.createElement("div");
  container.innerHTML = html;
  container.querySelectorAll(".variable-blot").forEach((el) => {
    const variableTextNode = el.querySelector(".variable-blot > span:last-child"); // Get the span with the actual variable text
    if (variableTextNode) {
      const text = document.createTextNode(variableTextNode.innerText);
      el.replaceWith(text);
    }
  });
  return container.innerHTML;
}

/**
 * Strips all class attributes from HTML elements for backend storage.
 * @param {string} html - The HTML content from the editor.
 * @returns {string} The HTML content without class attributes.
 */
function stripStyling(html) {
  const container = document.createElement("div");
  container.innerHTML = html;
  container.querySelectorAll("*").forEach((el) => {
    el.removeAttribute("class");
    el.removeAttribute("style"); // Also remove inline styles for cleaner backend storage
  });
  return container.innerHTML;
}

const iv = crypto.getRandomValues(new Uint8Array(12));
const ivBase64 = btoa(String.fromCharCode.apply(null, iv));
const salt = crypto.getRandomValues(new Uint8Array(16));
const saltBase64 = btoa(String.fromCharCode.apply(null, salt));

/**
 * Renders the email template editor page with drag-and-drop variables.
 */
const EmailTemplate = () => {
  const [headerContent, setHeaderContent] = useState("");
  const [bodyContent, setBodyContent] = useState("");
  const [sbar, setSbar] = useState({ isOpen: false, Msg: "" });
  const [wordCount, setWordCount] = useState(0);
  const [isEditing, setIsEditing] = useState(false);
  const [isPreviewing, setIsPreviewing] = useState(false);
  const [templateContent, settemplateContent] = useState(null);
  const draftVar = [
    "{{DATE}}",
    "{{REF_NO}}",
    "{{FIRM_NAME}}",
    "{{FRN_NO}}",
    "{{FIRM_ADDR}}",
    "{{ASSIGNMENT_TYPE}}",
    "{{GSTN}}",
    "{{ASSIGNMENT_SUBTYPE}}",
  ];
  const bodyEditorRef = useRef(null);

  const modules = {
    toolbar: [
      [{ font: Font.whitelist }],
      [{ size: Size.whitelist }],
      [
        {
          color: [
            "red",
            "green",
            "blue",
            "black",
            "orange",
            "#ffffff",
            "#000000",
          ],
        },
        { background: ["red", "green", "blue", "yellow", "gray"] },
      ],
      [
        { align: [] },
        { align: "center" },
        { align: "right" },
        { align: "justify" },
      ],
      [{ script: "sub" }, { script: "super" }],
      [{ header: [1, 2, 3, false] }],
      [{ list: "ordered" }, { list: "bullet" }],
      ["bold", "italic", "underline", "strike"],
      ["link", "clean"],
      [{ indent: "-1" }, { indent: "+1" }],
    ],
    clipboard: { matchVisual: false },
  };

  const formats = [
    "header",
    "font",
    "size",
    "bold",
    "italic",
    "underline",
    "strike",
    "blockquote",
    "list",
    "bullet",
    "indent",
    "link",
    "color",
    "background",
    "align",
    "script",
    "variable",
  ];

  /**
   * Effect to handle drag-and-drop of variables into the editor.
   * Validates that only allowed variables can be dropped.
   */
  useEffect(() => {
    if (!isEditing) return;
    const quill = bodyEditorRef.current?.getEditor();
    if (!quill) return;
    const editorRoot = quill.root;

    const handleEditorDrop = (e) => {
      e.preventDefault();
      const text = e.dataTransfer.getData("text");

      if (!draftVar.includes(text)) {
        setSbar({
          isOpen: true,
          Msg: "⚠️ Only variables from the list can be dropped.",
        });
        return;
      }

      const range = quill.getSelection(true);
      if (!range) return;

      quill.insertEmbed(range.index, "variable", text, Quill.sources.USER);
      quill.setSelection(range.index + 1, Quill.sources.USER);
    };

    const handleEditorDragOver = (e) => e.preventDefault();
    editorRoot.addEventListener("drop", handleEditorDrop);
    editorRoot.addEventListener("dragover", handleEditorDragOver);
    return () => {
      editorRoot.removeEventListener("drop", handleEditorDrop);
      editorRoot.removeEventListener("dragover", handleEditorDragOver);
    };
  }, [isEditing, draftVar]);

  /**
   * Effect to handle clicking the 'X' icon to delete a variable blot.
   */
  useEffect(() => {
    const editor = bodyEditorRef.current?.getEditor();
    if (!editor || !isEditing) return;
    const handleClick = (e) => {
      if (e.target.classList.contains("variable-delete-icon")) {
        const blotNode = e.target.closest(".variable-blot");
        if (blotNode) {
          const blot = Quill.find(blotNode, true);
          if (blot) {
            const index = editor.getIndex(blot);
            editor.deleteText(index, blot.length(), "user");
          }
        }
      }
    };
    editor.root.addEventListener("click", handleClick);
    return () => editor.root.removeEventListener("click", handleClick);
  }, [isEditing]);

  /**
   * Effect to fetch the initial template data on component mount.
   */
  useEffect(() => {
    fetchData();
  }, []);

  /**
   * Fetches the email template from the backend and splits it into header and body.
   */
  const fetchData = async () => {
    try {
      const response = await axios.post(
        "Server/IAMServices/EmailTemplate",
        {},
        {
          headers: { Authorization: `Bearer ${localStorage.getItem("token")}` },
        }
      );
      let res = response.data.result;
      let deliminator = "INTIMATION</h4>";
      const splitIndex = res.indexOf(deliminator) + deliminator.length;
      if (splitIndex > deliminator.length) {
        setHeaderContent(res.substring(0, splitIndex));
        setBodyContent(res.substring(splitIndex));
      } else {
        setHeaderContent("");
        setBodyContent(res);
      }
      setWordCount(countWord(res));
    } catch (e) {
      console.error("Failed to fetch template:", e);
      setSbar({ isOpen: true, Msg: "❌ Could not load email template." });
    }
  };

  /**
   * Sets the dragged variable data in the dataTransfer object.
   * @param {React.DragEvent} e - The drag event.
   */
  const handleDragStart = (e) => {
    e.dataTransfer.setData("text", e.target.innerText);
  };

  /**
   * Counts the words in an HTML string by stripping tags.
   * @param {string} str - The HTML string.
   * @returns {number} The number of words.
   */
  function countWord(str) {
    const text = str.replace(/<[^>]*>/g, "");
    return text.trim().split(/\s+/).filter(Boolean).length;
  }

  /**
   * Handles content changes in the Quill editor, including validation.
   * @param {string} content - The new content.
   * @param {object} delta - The change delta.
   * @param {string} source - The source of the change.
   * @param {object} editor - The Quill editor instance.
   */
  const handleBodyChange = (content, delta, source, editor) => {
    if (source === "api") return;
    const htmlContent = editor.getHTML();
    const fullText = headerContent + htmlContent;
    const length = countWord(fullText);

    // This block for preventing illegal characters might be problematic with custom blots.
    // It's trying to delete text based on insertion, which might interfere with blot insertion.
    // Consider if this validation is truly necessary or can be handled differently for variables.
    // For now, I'm commenting it out as it's a likely source of issues with blot handling.
    /*
    const insertedOp = delta.ops?.find((op) => typeof op.insert === "string");
    if (insertedOp) {
      const inserted = insertedOp.insert;
      const len = inserted.length;
      const illegalChar = inserted.match(
        /[^a-zA-Z0-9_\s.+|\\/*&,;!?'"()\[\]%\{\}\-]/
      );
      if (illegalChar) {
        setTimeout(() => {
          const quillEditor = bodyEditorRef.current?.getEditor();
          if (quillEditor) {
            const selection = quillEditor.getSelection();
            let position = selection?.index || quillEditor.getLength();
            quillEditor.deleteText(position - len, len, "silent");
          }
        }, 10);
        setSbar({
          isOpen: true,
          Msg: "❌ Special characters are not allowed!",
        });
      }
    }
    */

    if (length > 2000) {
      const quillEditor = bodyEditorRef.current?.getEditor();
      if (quillEditor) {
        const selection = quillEditor.getSelection();
        let position = selection?.index || quillEditor.getLength();
        quillEditor.deleteText(position - 1, 1, "silent");
        setSbar({ isOpen: true, Msg: "Word limit exceeded." });
        return;
      }
    }
    setBodyContent(htmlContent);
    setWordCount(length);
  };

  /**
   * Saves the complete template to the backend.
   */
  const handleSave = async () => {
    setIsEditing(false);
    let i = bodyContent.search(/<p><br><\/p>/g);
    let fullContent =
      i === 0
        ? headerContent + bodyContent
        : headerContent + "<p><br/></p>" + bodyContent;

    // 1. Convert variable blots back to raw text ({{VARIABLE_NAME}})
    let contentWithVariablesAsText = convertVariablesToText(fullContent);

    // 2. Strip all styling and class attributes
    let backendContent = stripStyling(contentWithVariablesAsText);

    // Validate variables
    let str = backendContent.match(/\{{2}[A-Z_]+\}{2}/g);
    let checked = str?.filter((w) => draftVar.some((s) => s === w));
    if (str != null && str.length > checked.length) {
      setSbar({
        isOpen: true,
        Msg: "Invalid Variable Names added! Ensure variables are correct.",
      });
      handleCloseDialog(); // Close preview if open
      return;
    }

    // Sanitize string for non-ASCII characters
    let SanStr = backendContent.replace(/[^\x00-\x7F]/g, "");

    let dataToSend = { MASTER_TEMPLATE: `<html>${SanStr}</html>` };
    try {
      let jsonFormData = JSON.stringify(dataToSend);
      await encrypt(iv, salt, jsonFormData).then(
        (result) => (jsonFormData = result)
      );
      let payload = { iv: ivBase64, salt: saltBase64, data: jsonFormData };
      const response = await axios.post(
        "Server/IAMServices/saveMasterTemplate",
        payload,
        {
          headers: { Authorization: `Bearer ${localStorage.getItem("token")}` },
        }
      );
      if (response.data.statusCode === 200 && response.data.result !== "") {
        setSbar({ isOpen: true, Msg: "✅ Template saved successfully!" });
        // After successful save, re-fetch data to reflect the saved state accurately
        fetchData();
      } else {
        setSbar({ isOpen: true, Msg: "⚠️ Template saved, but response was empty or non-200." });
      }
    } catch (error) {
      console.error("Error saving template:", error);
      setSbar({ isOpen: true, Msg: "❌ Error saving template." });
    }
    setIsPreviewing(false);
    handleCloseDialog();
  };

  /**
   * Generates a preview of the template with dummy data and opens the preview dialog.
   */
  const handlePreview = async () => {
    let i = bodyContent.search(/<p><br><\/p>/g);
    let fullContent =
      i === 0
        ? headerContent + bodyContent
        : headerContent + "<p><br/></p>" + bodyContent;

    // The preview should still use the blot for replacement
    const tempDiv = document.createElement("div");
    tempDiv.innerHTML = fullContent; // Use fullContent, which can still have variable blots

    let data = {
      DATE: "xx/xx/xxxx",
      REF_NO: "9999xxx",
      FIRM_NAME: "xxxxxxxxxx",
      FRN_NO: "999999999",
      GSTN: "99xx99xx",
      FIRM_ADDR: "xxxxxxxx,xxxxxxxx",
      ASSIGNMENT_TYPE: "xxxxxxx",
      ASSIGNMENT_SUBTYPE: "xxxxxx",
    };

    // Replace blot variables first
    tempDiv.querySelectorAll(".variable-blot").forEach((blot) => {
      // Ensure you're getting the text from the correct span inside the blot
      const variableTextNode = blot.querySelector(".variable-blot > span:last-child");
      if (variableTextNode) {
          let variableName = variableTextNode.innerText.trim().replace(/^{{|}}$/g, "");
          if (data[variableName]) {
              blot.replaceWith(document.createTextNode(data[variableName]));
          } else {
              // If data not found, replace blot with its original variable text
              blot.replaceWith(document.createTextNode(variableTextNode.innerText));
          }
      }
    });

    // Then replace any remaining text-based variables
    let htmlString = tempDiv.innerHTML.replace(
      /{{([A-Z_]+)}}/g,
      (match, p1) => {
        if (data[p1]) return data[p1];
        return match;
      }
    );

    // Strip styling for preview as well, as you don't want editor-specific classes in the final preview
    htmlString = stripStyling(htmlString);
    htmlString = htmlString.replace(/<br\s*\/?>/gi, "<br></br>"); // Consistent <br>
    settemplateContent(htmlString);
    setIsPreviewing(true);
  };


  /**
   * Closes the preview dialog.
   */
  const handleCloseDialog = () => setIsPreviewing(false);

  /**
   * Resets the template to its default version from the server.
   */
  const handleReset = async () => {
    try {
      const response = await axios.post(
        "Server/IAMServices/resetTemplate",
        {},
        {
          headers: { Authorization: `Bearer ${localStorage.getItem("token")}` },
        }
      );
      let res = response.data.result;
      const deliminator = "INTIMATION</h4>";
      const splitIndex = res.indexOf(deliminator) + deliminator.length;
      if (splitIndex > deliminator.length) {
        setHeaderContent(res.substring(0, splitIndex));
        setBodyContent(res.substring(splitIndex));
      } else {
        setHeaderContent("");
        setBodyContent(res);
      }
      setSbar({ isOpen: true, Msg: "✅ Template has been reset to default." });
      setIsEditing(false); // Exit editing mode after reset
    } catch (e) {
      setSbar({ isOpen: true, Msg: "❌ Error resetting template." });
    }
  };

  return (
    <>
      <Typography gutterBottom variant="h5" component="div" sx={{ mb: 1 }}>
        Letter Template : Empanelment Intimation
      </Typography>
      <Divider sx={{ mb: 1 }} />
      <Box sx={{ p: 1, minHeight: "80%" }}>
        {/* Layout reverted to top-aligned two-column grid */}
        <Grid container spacing={3}>
          <Grid item xs={12} md={isEditing ? 9 : 12}>
            <Card>
              <CardHeader
                title="Configurable"
                titleTypographyProps={{ variant: "h6" }}
              />
              <CardContent>
                <Paper variant="outlined">
                  <ReactQuill
                    theme="snow"
                    style={{ height: "50vh" }}
                    modules={modules}
                    formats={formats}
                    value={bodyContent}
                    ref={bodyEditorRef}
                    onChange={handleBodyChange}
                    readOnly={!isEditing}
                  />
                  <Box
                    sx={{
                      mt: "50px",
                      display: "flex",
                      justifyContent: "space-between",
                      alignItems: "center",
                      p: 2,
                      borderTop: "1px solid #e0e0e0",
                    }}
                  >
                    <Typography variant="body2" color="textSecondary">
                      Word Count: {wordCount} / 2000
                    </Typography>
                    <Box sx={{ display: "flex", gap: 1 }}>
                      {!isEditing ? (
                        <Button
                          variant="contained"
                          startIcon={<EditIcon />}
                          onClick={() => setIsEditing(true)}
                        >
                          Edit
                        </Button>
                      ) : (
                        <Button
                          variant="outlined"
                          color="error"
                          startIcon={<CancelIcon />}
                          onClick={() => {
                            setIsEditing(false);
                            fetchData(); // Revert changes by re-fetching
                          }}
                        >
                          Cancel
                        </Button>
                      )}
                      <Button
                        variant="outlined"
                        color="info"
                        startIcon={<PreviewIcon />}
                        onClick={handlePreview}
                      >
                        Preview
                      </Button>
                      <Button
                        variant="outlined"
                        color="secondary"
                        startIcon={<RestartAltIcon />}
                        onClick={handleReset}
                      >
                        Reset
                      </Button>
                    </Box>
                  </Box>
                </Paper>
              </CardContent>
            </Card>
          </Grid>

          {isEditing && (
            <Grid item xs={12} md={3}>
              <Card>
                <CardHeader
                  title="Variables to Drag and Drop to Letter Template"
                  titleTypographyProps={{ variant: "h6" }}
                />
                <CardContent sx={{ display: "flex", flexWrap: "wrap", gap: 1 }}>
                  {draftVar.map((label) => (
                    <Chip
                      key={label}
                      label={label}
                      color="primary"
                      draggable
                      onDragStart={handleDragStart}
                      sx={{
                        borderRadius: "0",
                        cursor: "move",
                        fontWeight: "medium",
                      }}
                    />
                  ))}
                </CardContent>
              </Card>
            </Grid>
          )}
        </Grid>

        <Dialogue
          open={isPreviewing}
          close={handleCloseDialog}
          save={handleSave}
          content={templateContent}
        />
        <Snackbar
          open={sbar.isOpen}
          autoHideDuration={6000}
          onClose={() => setSbar({ isOpen: false, Msg: "" })}
          message={sbar.Msg}
          anchorOrigin={{ vertical: "bottom", horizontal: "right" }}
        />
      </Box>
    </>
  );
};

  <BootstrapDialog onClose={close} open={open} fullWidth maxWidth="md">

      Intimation Letter Draft Preview

      </Button>

  </BootstrapDialog>
);

export default EmailTemplate;
