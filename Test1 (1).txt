import java.util.*;



public class Main {
    
    public static Boolean isDuplicate(int[] arr , int index) {
        
        
        if(index == arr.length) {
            return false;
        }
        
        
        boolean ans =  Dupli(arr , index , index+1);
        
        isDuplicate(arr , index+1);
        
         
        return ans;
       
    
    }
    
    public static Boolean Dupli(int[] arr , int i , int j) {
        
        if(j == arr.length ) {
            
            
            return false;
        }
        
        
        
        if(arr[i] == arr[j]) {
            return true;
        }
        
      
        
        return Dupli(arr , i , j+1);
        
      
        
       
        
    }
    
   
    
    public static void main(String[] args) {
        
        int[] arr = {1,2 ,3 , 3};
        
        System.out.println(isDuplicate(arr , 0));
        
      
        
      
    }
}
