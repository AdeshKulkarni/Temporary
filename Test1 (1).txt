Since you are using Avi Load Balancer (VMware NSX ALB), the standard behavior is to pass the URL path exactly as is.
Because the Frontend is sending requests starting with /JS/... (e.g., /JS/journals/bulk-validate-init), but your Backend was listening on /journals/..., Avi was forwarding the request, but the Backend (Tomcat/Spring) was rejecting it as 404 Not Found because it didn't recognize the /JS prefix.
The Solution (No Avi Config Required):
We will simply update the Backend Controller to listen for the /JS path. This is the safest and fastest fix.

    private List<Map<String, Object>> mapToSummaryDto(List<Object[]> rawData, boolean isStatusScreen) {
        List<Map<String, Object>> result = new ArrayList<>();
        for (Object[] row : rawData) {
            Map<String, Object> map = new HashMap<>();
            map.put("batchId", row[0]);
            map.put("creatorId", row[1]); // Always Actual Creator ID
            map.put("requestDate", row[2]);
            map.put("commonBatchRemarks", row[3]);
            map.put("requestCount", row[4]);
            map.put("totalDebit", row[5]);
            map.put("totalCredit", row[6]);
            
            // Status Mapping
            String fullStatus = "PENDING";
            if (isStatusScreen && row.length > 7 && row[7] != null) {
                String raw = (String) row[7];
                if ("A".equalsIgnoreCase(raw) || "ACCEPTED".equalsIgnoreCase(raw)) fullStatus = "ACCEPTED";
                else if ("R".equalsIgnoreCase(raw) || "REJECTED".equalsIgnoreCase(raw)) fullStatus = "REJECTED";
            }
            map.put("requestStatus", fullStatus);
            
            if (isStatusScreen && row.length > 9) {
                map.put("executorId", row[8]);
                map.put("executorRemarks", row[9]);
            }
            result.add(map);
        }
        return result;
    }

    @Override public List<JournalRequestStatusDto> getJournalRequestStatusList() { return journalRequestRepository.findAll().stream().map(JournalRequestStatusDto::new).collect(Collectors.toList()); }

    @Override public List<Map<String, Object>> getMyBatchSummaries(String userId) { List<Object[]> rawData = journalRequestRepository.findMyBatchSummariesNative(userId); return mapToSummaryDto(rawData, true); }
