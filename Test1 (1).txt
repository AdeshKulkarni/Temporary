    @Override
    public List<MocRequest> getMyRequests() {
        log.info("Fetching ALL MOC requests for 'My Requests' screen.");
        // We now return ALL requests, sorted by requestDate descending (newest first)
        return mocRequestRepository.findAll(Sort.by(Sort.Direction.DESC, "requestDate"));
    }

    @Override
    public List<MocRequest> getPendingRequests() {
        // This is what you wanted, but for the "my-requests" endpoint
        return mocRequestRepository.findByRequestStatus(RequestStatus.PENDING);
