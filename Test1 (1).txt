const columns = [
  { key: 'standard', label: 'Standard', editable: true },
  { key: 'subStandard', label: 'Sub-standard', editable: true },
  { key: 'doubtful', label: 'Doubtful', editable: true },

const sectionMap = {
  facTotal: ['fac1', 'fac2', 'fac3'],
  secTotal: ['sec1', 'sec2', 'sec3'],
  inTotal: ['in1', 'in2', 'in3', 'in4'],
  outTotal: ['out1', 'out2', 'out3', 'out4'],
  advA3: ['inTotal', 'outTotal'],
};

const getFieldName = (id, key) => {
    let base = id.startsWith('fac') ? 'facility' : id.startsWith('sec') ? 'security' : 'sector';
    const suffix = {
      standard: 'Standard', subStandard: 'SubStandard', doubtful: 'Doubtful',
      loss: 'Loss', adjustment: 'Adj',
    }[key] || 'Total';
    let idx;

    // ## CHANGE 1: Corrected the index mapping for sector rows to match the old system's data model exactly.
    if (id.startsWith('fac') || id.startsWith('sec')) {
        idx = id.replace(/\D/g, '');
    } else if (id.startsWith('in')) {
        idx = id.endsWith('Total') ? 'a_Total' : `a${id.replace('in', '')}`;
    } else if (id.startsWith('out')) {
        if (id === 'out1') idx = 'b1';
        else if (id === 'out2') idx = '1'; // Maps to sector_..._1
        else if (id === 'out3') idx = '2'; // Maps to sector_..._2
        else if (id === 'out4') idx = '3'; // Maps to sector_..._3
        else if (id === 'outTotal') idx = 'b_Total';
    } else if (id === 'advA3') {
        idx = 'ab_Total';
    }

    if (id.endsWith('Total') && (id.startsWith('fac') || id.startsWith('sec'))) {
        idx = 'Total';
    }
    return `${base}_${suffix}_${idx}`;
};

// buildSavePayload (no changes)
const buildSavePayload = ( values, userId, circleCode, quarterEndDate, role, reportName, save, status, reportMasterId, reportId, areMocPending) => {
    const payload = { userId, circleCode, quarterEndDate, role, reportName, save, status, reportMasterId, reportId, areMocPending };
    Object.entries(values).forEach(([id, fields]) => {
        Object.entries(fields).forEach(([key, val]) => {
            if (['standard', 'subStandard', 'doubtful', 'loss', 'adjustment'].includes(key)) {
                if (!(id.includes('header') || id === 'out5')) {
                    payload[getFieldName(id, key)] = val || '0';
                }
            }
        });
    });
    return payload;
};

export default function Schedule9Table({ showAdjustment = false }) {
  const { state } = useLocation();
  const showSnackbar = useCustomSnackbar();
  const user = JSON.parse(localStorage.getItem('user'));
  const [reportObject, setReportObject] = useState(state?.report || null);
  const navigate = useNavigate();
  const { callApi } = useApi();
  
  // Add a loading state
  const [isLoading, setIsLoading] = useState(true);

  const [values, setValues] = useState(
    rowDefinitions.reduce((acc, { id }) => {
      acc[id] = { standard: '', subStandard: '', doubtful: '', loss: '', adjustment: '', ysa: '' };
      return acc;
    }, {})
  );

  // Fetch data
  useEffect(() => {
    let isMounted = true;
    const loadAllData = async () => {
        try {
            setIsLoading(true);
            const reportPayload = { circleCode: user.circleCode, quarterEndDate: user.quarterEndDate, role: user.capacity, reportName: reportObject.name };
            const validationPayload = { circleCode: user.circleCode, quarterEndDate: user.quarterEndDate, role: user.capacity };

            // ## CHANGE 2: Use Promise.all to fetch both datasets before processing.
            // This prevents race conditions and temporary validation errors.
            const [reportData, validationData] = await Promise.all([
                callApi('/Maker/getSC09ReportData', reportPayload, 'POST'),
                callApi('/Maker/getSC09ValiadationData', validationPayload, 'POST')
            ]);
            
            if (!isMounted) return;

            // Process both data responses at once and update state a single time.
            const newValues = { ...values };

            // Process Report Data
            const backendFieldMap = { standard: 'Standard', subStandard: 'SubStandard', doubtful: 'Doubtful', loss: 'Loss', adjustment: 'Adj' };
            rowDefinitions.forEach(({ id, type }) => {
                if (type !== 'entry') return;
                Object.keys(backendFieldMap).forEach(fieldKey => {
                    const backendKey = getFieldName(id, fieldKey);
                    if (reportData[backendKey] !== undefined) {
                        newValues[id][fieldKey] = reportData[backendKey];
                    }
                });
            });

            // Process Validation Data
            if (user.capacity === '51' || user.capacity === '52') {
                const normalize = (s) => s.replace(/\s|[^a-zA-Z0-9]/g, '').toLowerCase();
                ['fac1', 'fac2', 'fac3'].forEach((id) => {
                    const label = rowDefinitions.find((r) => r.id === id).label;
                    const key = Object.keys(validationData).find((k) => normalize(k) === normalize(label));
                    newValues[id].ysa = validationData[key] || '0';
                });
            } else if (user.capacity === '61' || user.capacity === '62') {
                const codeMap = { fac1: '090101001051', fac2: '090101001052', fac3: '090101001053' };
                Object.keys(codeMap).forEach((id) => {
                    const val = validationData[codeMap[id]] || "0~0";
                    const parts = val.split('~');
                    newValues[id].adjustment = parts[0] || '0';
                    newValues[id].ysa = parts[1] || '0';
                });
            }

            setValues(newValues);

        } catch (error) {
            if (error.code !== 'ERR_CANCELED') {
              showSnackbar('Failed to load report data', 'error');
            }
        } finally {
            if(isMounted) setIsLoading(false);
        }
    };
    
    loadAllData();

    return () => { isMounted = false; };
  }, []);

  // handleChange, getRowData, computedRowData, totalMismatchFlags, validateForSubmit, onSave, onSubmit...
  // ... (No changes to the logic of these functions from the previous answer)
  const handleChange = (id, field, val) => {
    if (!/^-?\d*\.?\d{0,2}$/.test(val)) return;
    setValues((prev) => ({ ...prev, [id]: { ...prev[id], [field]: val } }));
  };

  const getRowData = (id) => {
    if (!sectionMap[id]) {
        const rowValues = values[id];
        const sum = ['standard', 'subStandard', 'doubtful', 'loss'].reduce( (acc, field) => acc + (parseFloat(rowValues[field]) || 0), 0);
        const adj = showAdjustment ? parseFloat(rowValues.adjustment) || 0 : 0;
        const total = sum + adj;
        const ysa = parseFloat(rowValues.ysa);
        const isMismatch = id.startsWith('fac') && !isNaN(ysa) ? ysa.toFixed(2) !== total.toFixed(2) : false;
        return { ...rowValues, total: total.toFixed(2), isMismatch };
    }
    const agg = sectionMap[id].reduce( (acc, childId) => {
        const childRowData = getRowData(childId);
        Object.keys(acc).forEach((field) => { acc[field] += parseFloat(childRowData[field]) || 0 });
        return acc;
      }, { standard: 0, subStandard: 0, doubtful: 0, loss: 0, adjustment: 0 });
    const total = Object.values(agg).reduce((sum, val) => sum + val, 0);
    return {
      standard: agg.standard.toFixed(2), subStandard: agg.subStandard.toFixed(2),
      doubtful: agg.doubtful.toFixed(2), loss: agg.loss.toFixed(2),
      adjustment: showAdjustment ? agg.adjustment.toFixed(2) : '',
      total: total.toFixed(2), ysa: values[id].ysa, isMismatch: false,
    };
  };

  const computedRowData = useMemo(() => {
    const result = {};
    rowDefinitions.forEach((row) => { result[row.id] = getRowData(row.id); });
    return result;
  }, [values, showAdjustment]);

  const totalMismatchFlags = useMemo(() => {
    const a = computedRowData['facTotal']; const b = computedRowData['secTotal']; const c = computedRowData['advA3'];
    const mismatches = {};
    const keysToCompare = ['standard', 'subStandard', 'doubtful', 'loss'];
    if (showAdjustment) keysToCompare.push('adjustment');
    keysToCompare.forEach((key) => { mismatches[key] = !(a[key] === b[key] && b[key] === c[key]); });
    return mismatches;
  }, [computedRowData, showAdjustment]);

  const validateForSubmit = useMemo(() => {
    if (isLoading) return []; // Don't validate while loading
    const errs = [];
    ['fac1', 'fac2', 'fac3'].forEach((id) => {
      const { total, ysa } = computedRowData[id];
      if (parseFloat(ysa || 0).toFixed(2) !== total) {
        errs.push(`Total not matching for ${rowDefinitions.find((r) => r.id === id).label}`);
      }
    });
    const aTot = computedRowData['facTotal'], bTot = computedRowData['secTotal'], cTot = computedRowData['advA3'];
    [['standard', 'Standard'], ['subStandard', 'Sub-standard'], ['doubtful', 'Doubtful'], ['loss', 'Loss']].forEach(([key, label]) => {
      if (!(aTot[key] === bTot[key] && bTot[key] === cTot[key])) {
        errs.push(`A-1, A-2, A-3 totals differ for ${label}`);
      }
    });
    if (showAdjustment) {
      if (!(aTot.adjustment === bTot.adjustment && bTot.adjustment === cTot.adjustment)) {
        errs.push('A-1, A-2, A-3 totals differ for Adjustment');
      }
    }
    return errs;
  }, [computedRowData, showAdjustment, isLoading]);

  const onSave = async () => { /* ... no changes ... */ };
  const onSubmit = async () => { /* ... no changes ... */ };


  if (isLoading) {
    return <Box sx={{ display: 'flex', justifyContent: 'center', p: 4 }}><CircularProgress /></Box>;
  }

  return (
    <Box>
      <TableRow>
        <TableCell colSpan={columns.length + 2}>
            <Stack direction={'row'} spacing={2} sx={{ my: 2 }}>
                <CustomButton label={'Save'} buttonType={'save'} onClickHandler={onSave} />
                <CustomButton label={'Submit'} buttonType={'submit'} onClickHandler={onSubmit} disabled={!!validateForSubmit.length} />
            </Stack>
        </TableCell>
      </TableRow>

      {validateForSubmit.length > 0 && (
        <Alert severity="error" sx={{ mb: 2 }}>
          <ul style={{ margin: 0, paddingLeft: '1.2em' }}>
            {validateForSubmit.map((e, i) => ( <li key={i}>{e}</li> ))}
          </ul>
        </Alert>
      )}

      <TableContainer component={Paper} sx={{ width: '100%' }}>
        <Table sx={{ tableLayout: 'fixed' }}>
          <TableHead>
            <TableRow>
              <StyledTableCell>Classification of Advances (...) (A1 = A2 = A3)</StyledTableCell>
              {columns.map((col) => (!col.conditional || showAdjustment) && <StyledTableCell key={col.key} align="center">{col.label}</StyledTableCell>)}
            </TableRow>
          </TableHead>
          <TableBody>
            {rowDefinitions.map((row) => {
              const data = computedRowData[row.id];
              if (row.type !== 'entry') {
                return (
                  <StyledTableRow key={row.id}>
                    <StyledTableCell colSpan={columns.filter((c) => !c.conditional || showAdjustment).length + 1} sx={{ fontWeight: 'bold' }}>{row.label}</StyledTableCell>
                  </StyledTableRow>
                );
              }
              return (
                <StyledTableRow key={row.id}>
                  <StyledTableCell>{row.label}</StyledTableCell>
                  {columns.map((col) => (!col.conditional || showAdjustment) && (
                      <StyledTableCell key={col.key} align="right">
                        <FormInput
                          value={data[col.key] || ''}
                          onChange={(e) => handleChange(row.id, col.key, e.target.value)}
                          readOnly={
                            !col.editable || Boolean(sectionMap[row.id]) ||
                            (user.capacity === '61' && col.key !== 'adjustment') ||
                            (user.capacity === '61' && ['fac1', 'fac2', 'fac3'].includes(row.id) && col.key === 'adjustment')
                          }
                          inputType="wholeAmountDecimal"
                          sx={{
                            ...(data.isMismatch && col.key === 'total' && { border: '1px solid red', borderRadius: 1 }),
                            ...(['facTotal', 'secTotal', 'advA3'].includes(row.id) && col.key !== 'total' && totalMismatchFlags[col.key] && { border: '1px solid red', borderRadius: 1 }),
                          }}
                        />
                      </StyledTableCell>
                    )
                  )}
                </StyledTableRow>
              );
            })}
          </TableBody>
        </Table>
      </TableContainer>
    </Box>
  );
}
