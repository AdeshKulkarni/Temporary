package com.fincore.VoucherService.Service;

import com.fincore.VoucherService.Dto.BranchDto;
import com.fincore.VoucherService.Models.BranchMaster;
import com.fincore.VoucherService.Repository.BranchMasterRepository;
import lombok.extern.slf4j.Slf4j;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.beans.factory.annotation.Qualifier;
import org.springframework.dao.DataAccessException;
import org.springframework.data.domain.PageRequest;
import org.springframework.data.domain.Pageable;
import org.springframework.jdbc.core.JdbcTemplate;
import org.springframework.stereotype.Service;

import java.util.Collections;
import java.util.List;
import java.util.Map;
import java.util.stream.Collectors;

/**
 * Provides functionality for retrieving branch mapping and performing targeted searches.
 * Implements specific exception handling for database access and secure logging practices.
 */
@Service
@Slf4j
public class BranchService {

    private final BranchMasterRepository branchMasterRepository;
    private final JdbcTemplate jdbcTemplate;

    @Autowired
    public BranchService(BranchMasterRepository branchMasterRepository, 
                         @Qualifier("oracleJdbcTemplate") JdbcTemplate jdbcTemplate) {
        this.branchMasterRepository = branchMasterRepository;
        this.jdbcTemplate = jdbcTemplate;
    }

    /**
     * RESTORED METHOD: Retrieves a mapping of branch codes to branch names.
     * Useful for dropdown population or lookup optimization.
     */
    public Map<String, String> getBranchCodeAndNameMap() {
        try {
            List<BranchMaster> branches = branchMasterRepository.findAll();
            log.info("BRANCH_MAP_LOADED | Count: {}", branches.size());

            return branches.stream()
                    .collect(Collectors.toMap(BranchMaster::getCode, BranchMaster::getName, (existing, replacement) -> existing));

        } catch (DataAccessException e) {
            log.error("BRANCH_MAP_FETCH_FAIL | Type: {}", e.getClass().getSimpleName());
            return Collections.emptyMap();
        }
    }

    /**
     * NEW METHOD: Performs a paginated search for branches using the BackDate filter.
     */
    public List<BranchDto> searchBranches(String query, java.time.LocalDate backDate) {
        if (query == null || query.trim().isEmpty()) return Collections.emptyList();

        try {
            String sql = "SELECT CODE, NAME FROM BRANCH_MASTER WHERE (LOWER(CODE) LIKE LOWER(?) OR LOWER(NAME) LIKE LOWER(?))";
            List<Object> params = new java.util.ArrayList<>();
            params.add("%" + query.trim() + "%");
            params.add("%" + query.trim() + "%");

            if (backDate != null) {
                sql += " AND TRUNC(OPEN_DATE) <= ? AND (CLOSE_DATE IS NULL OR TRUNC(CLOSE_DATE) >= ?)";
                params.add(java.sql.Date.valueOf(backDate));
                params.add(java.sql.Date.valueOf(backDate));
            } else {
                sql += " AND STATUS = 1";
            }
            sql += " FETCH FIRST 50 ROWS ONLY";

            return jdbcTemplate.query(sql, (rs, rowNum) -> new BranchDto(rs.getString("CODE"), rs.getString("NAME")), params.toArray());
        } catch (DataAccessException e) {
            log.error("BRANCH_SEARCH_FAIL | Backdate logic error: {}", e.getMessage());
            return Collections.emptyList();
        }
    }

    /**
     * ORIGINAL METHOD: Performs a paginated search for branches by name or code.
     */
    public List<BranchDto> searchBranches(String query) {
        log.info("BRANCH_SEARCH_INIT | QueryLength: {}", (query != null ? query.length() : 0));

        if (query == null || query.trim().isEmpty()) {
            return Collections.emptyList();
        }

        try {
            Pageable limit = PageRequest.of(0, 50);
            List<BranchMaster> results = branchMasterRepository
                    .findByNameContainingIgnoreCaseOrCodeContainingIgnoreCase(query, query, limit);

            log.info("BRANCH_SEARCH_SUCCESS | ResultsFound: {}", results.size());

            return results.stream()
                    .map(branch -> new BranchDto(branch.getCode(), branch.getName()))
                    .collect(Collectors.toList());

        } catch (DataAccessException e) {
            log.error("BRANCH_SEARCH_FAIL | Query: [REDACTED] | Error: {}", e.getMessage());
            return Collections.emptyList();
        }
    }
}
