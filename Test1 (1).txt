public enum accountType {

    CORPORATE , NORMAL
}



public interface BaseFareStrategy {

    double baseFare(rideDetails ride);
}

public interface DiscountStrategy {

    double ApplyDiscount(rideDetails ride);
}

public interface SurgeStrategy {

    double CalculateSurge(rideDetails ride);
}
import java.util.List;

public class AuctionFare implements BaseFareStrategy  {
    @Override
    public double baseFare(rideDetails ride) {

        List<driver> bid = ride.getListOfDriverBid();

        ///  for now we assume that we calculate the lowest from this and return lowest
        /// for now we will return the first one

        return bid.getFirst().getBidAmount();


    }
}


public class DistanceBasedFare implements BaseFareStrategy {

    @Override
    public double baseFare(rideDetails ride) {
      return (ride.getDisInKm() * 10) + 50;
    }
}



public class FlatRateFare implements BaseFareStrategy  {
    @Override
    public double baseFare(rideDetails ride) {

        if(ride.getRouteType() == routeType.AIRPORT) {
            return 200;
        }
        return 0;
    }
}
public class HybridFare implements BaseFareStrategy {
    @Override
    public double baseFare(rideDetails ride) {
        return 0;
    }
}
public class TimeBasedFare implements BaseFareStrategy {
    @Override
    public double baseFare(rideDetails ride) {
        return (ride.getDurationInMin() * 2) + 30;
    }
}


public class ConditionalDiscount  implements DiscountStrategy {

    @Override
    public double ApplyDiscount(rideDetails ride) {
        return 0;
    }
}

public class CorporateDiscount implements DiscountStrategy {
    @Override
    public double ApplyDiscount(rideDetails ride) {

        if(ride.getAccountType() == accountType.CORPORATE) {
            return 15;
        }

        return 0;
    }
}

public class LoyaltyDiscount implements DiscountStrategy {
    @Override
    public double ApplyDiscount(rideDetails ride) {

        if(ride.getUserTotalRides() >= 50) {
            return 10;
        }

        return 0;
    }
}
public class OffPeakDiscount implements SurgeStrategy {
    @Override
    public double CalculateSurge(rideDetails ride) {

        int hour = ride.getHourOfDay().getHour();

        if(hour >= 11 && hour < 17) {
            return 20;
        }

        return 0;
    }
}

public class PromoCodeDiscount implements DiscountStrategy {
    @Override
    public double ApplyDiscount(rideDetails ride) {
          if(ride.getPromoCode() == promoCode.ZERO) {
              return 0;
          }
          else  if(ride.getPromoCode() == promoCode.FIFTY) {
              return 50;
          }
          else if(ride.getPromoCode() == promoCode.TEN) {
              return 10;
          }
          else if(ride.getPromoCode() == promoCode.SEVENTY) {
              return 70;
          }

          return 0;


    }
}
public class CompositeSurge implements SurgeStrategy {
    @Override
    public double CalculateSurge(rideDetails ride) {

        return 0;

    }
}
public class DemandSurge implements SurgeStrategy {
    @Override
    public double CalculateSurge(rideDetails ride) {

        ///  directly returning the Demand supply chain ration for now

        return ride.getDemandSupplyRation();


    }
}
public class EventSurge implements SurgeStrategy {
    @Override
    public double CalculateSurge(rideDetails ride) {

        if(ride.isNearStadium()) {
            return 2.0;
        }

        return 0;
    }
}
public class TimeOfDaySurge implements SurgeStrategy {
    @Override
    public double CalculateSurge(rideDetails ride) {
       int hour = ride.getHourOfDay().getHour();

        if (hour >= 17 && hour < 20 || hour >= 8 && hour < 10) {
            return 1.5;
        }

        return 0;
    }
}
public class WeatherSurge implements SurgeStrategy {
    @Override
    public double CalculateSurge(rideDetails ride) {
       if(ride.isRaining()) {
           return 1.3;
       }

       return 0;
    }
}
