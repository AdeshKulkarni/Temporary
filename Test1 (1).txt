class Solution {

    public boolean valid(int i, int j, int n, int m) {
        if (i < 0 || i >= n || j < 0 || j >= m) {
            return false;
        }
        return true;
    }

    public void dfs(char[][] board, int i, int j, int m, int n, boolean[][] visited) {

        visited[i][j] = true;
        board[i][j] = '#';

        int[] x = {-1, 1, 0, 0};
        int[] y = {0, 0, -1, 1};

        for (int k = 0; k < 4; k++) {

            int row = i + x[k];
            int col = j + y[k];

            if (valid(row, col, n, m)
                    && visited[row][col] == false
                    && board[row][col] == 'O') {

                dfs(board, row, col, m, n, visited);
            }
        }
    }

    public void solve(char[][] board) {

        int n = board.length;
        int m = board[0].length;

        boolean[][] visited = new boolean[n][m];

        // Start DFS from top and bottom boundaries
        for (int j = 0; j < m; j++) {

            if (board[0][j] == 'O' && visited[0][j] == false) {
                dfs(board, 0, j, m, n, visited);
            }

            if (board[n - 1][j] == 'O' && visited[n - 1][j] == false) {
                dfs(board, n - 1, j, m, n, visited);
            }
        }

        // Start DFS from left and right boundaries
        for (int i = 0; i < n; i++) {

            if (board[i][0] == 'O' && visited[i][0] == false) {
                dfs(board, i, 0, m, n, visited);
            }

            if (board[i][m - 1] == 'O' && visited[i][m - 1] == false) {
                dfs(board, i, m - 1, m, n, visited);
            }
        }

        // Convert surrounded O to X and restore safe cells
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {

                if (board[i][j] == 'O') {
                    board[i][j] = 'X';
                } else if (board[i][j] == '#') {
                    board[i][j] = 'O';
                }
            }
        }
    }
}
