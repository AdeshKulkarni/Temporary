import React, { useState, useCallback, useMemo, useRef } from "react";
import {
  Button,
  Card,
  CardContent,
  CardHeader,
  Chip,
  Divider,
  Grid,
  List,
  ListItem,
  ListItemIcon,
  ListItemText,
  Paper,
  Snackbar,
  Typography,
  Dialog,
  DialogTitle,
  DialogContent,
  IconButton,
  Box,
} from "@mui/material";
import EditIcon from "@mui/icons-material/Edit";
import PreviewIcon from "@mui/icons-material/Preview";
import RestartAltIcon from "@mui/icons-material/RestartAlt";
import CancelIcon from "@mui/icons-material/Cancel";
import SaveIcon from "@mui/icons-material/Save";
import CheckCircleOutlineIcon from "@mui/icons-material/CheckCircleOutline";
import CloseIcon from "@mui/icons-material/Close";

import { createEditor, Transforms, Editor, Element as SlateElement, Text, Node } from "slate";
import { Slate, Editable, withReact, useSlate, ReactEditor } from "slate-react";
import { withHistory } from "slate-history";

// ----- Variable List -----
const VARIABLE_LIST = [
  "{{DATE}}",
  "{{REF_NO}}",
  "{{FIRM_NAME}}",
  "{{FRN_NO}}",
  "{{FIRM_ADDR}}",
  "{{ASSIGNMENT_TYPE}}",
  "{{GSTN}}",
];

// Dummy data for preview
const PREVIEW_MAP = {
  DATE: "xx/xx/xxxx",
  REF_NO: "9999xxx",
  FIRM_NAME: "xxxxxxxxxx",
  FRN_NO: "999999999",
  GSTN: "99xx99xx",
  FIRM_ADDR: "xxxxxxxx,xxxxxxxx",
  ASSIGNMENT_TYPE: "xxxxxxx",
};

function sanitizeNodeText(text) {
  // Only certain special chars allowed
  return text.replace(
    /[^a-zA-Z0-9_\s\.\+\|\\\/\*\&,\;\!\?'"()\[\]%\{\}\-]/g,
    ""
  );
}

// ------ Custom Slate Element for Variable ------
const VariableElement = ({ attributes, children, element }) => (
  <span
    {...attributes}
    contentEditable={false}
    className="variable-blot"
    style={{
      backgroundColor: "#A9A9A9",
      color: "black",
      padding: "5px 20px 5px 8px",
      margin: "0 4px",
      borderRadius: "3px",
      // "display: inline-block" is default for span
      userSelect: "none",
      position: "relative",
      cursor: "default",
    }}
    data-variable-name={element.value}
  >
    <span>{element.value}</span>
    <span
      className="variable-delete-icon"
      onClick={(e) => {
        e.preventDefault();
        e.stopPropagation();
        const editor = window.currentSlateEditor;
        if (editor) {
          const path = ReactEditor.findPath(editor, element);
          Transforms.removeNodes(editor, { at: path });
        }
      }}
      style={{
        display: "inline-block",
        position: "absolute",
        right: 5,
        top: "50%",
        transform: "translateY(-50%)",
        fontSize: 12,
        fontWeight: "bold",
        fontFamily: "monospace",
        cursor: "pointer",
        marginLeft: 3,
      }}
    >
      X
    </span>
    {children}
  </span>
);

const DefaultElement = (props) => <p {...props.attributes}>{props.children}</p>;

// Renders elements
const renderElement = (props) => {
  switch (props.element.type) {
    case "variable":
      return <VariableElement {...props} />;
    default:
      return <DefaultElement {...props} />;
  }
};

// ----- Initial Content ------
const initialHeader = [
  {
    type: "paragraph",
    children: [{ text: "Non-editable header will appear here" }],
  },
];

const initialBody = [
  {
    type: "paragraph",
    children: [{ text: "Editable body will appear here" }],
  },
];

// -------- Main Component --------
function EmailTemplateWithSlate() {
  // Simulate backend: header always as static string; body is editable in Slate
  const [header] = useState(initialHeader);
  const [body, setBody] = useState(initialBody);
  const [isEditing, setIsEditing] = useState(false);
  const [isPreviewing, setIsPreviewing] = useState(false);
  const [snackbar, setSnackbar] = useState({ open: false, msg: "" });
  const [wordCount, setWordCount] = useState(() => countWords(body));

  // Slate setup
  const editorRef = useRef();
  const editor = useMemo(
    () =>
      withVariables(
        withHistory(
          withReact(createEditor())
        )
      ),
    []
  );
  // Window variable for handling delete
  window.currentSlateEditor = editor;

  // ---------- Drag & Drop support ----------
  const handleDragStart = (e, label) => {
    e.dataTransfer.setData("application/x-variable", label);
  };

  // ----- Insert variable at drop -----
  const handleDrop = useCallback(
    (event) => {
      event.preventDefault();
      if (!isEditing) return;
      let variableLabel = event.dataTransfer.getData("application/x-variable");
      if (!VARIABLE_LIST.includes(variableLabel)) return;
      // Insert at caret
      const { selection } = editor;
      if (selection) {
        // Check: prevent consecutive repeated variable
        const [node, path] = Editor.node(editor, selection.anchor);
        let prev = Editor.previous(editor, { at: selection })?.[0];
        if (prev && prev.type === "variable" && prev.value === variableLabel) {
          setSnackbar({ open: true, msg: `❌ Cannot add the same variable consecutively.` });
          return;
        }
        Transforms.insertNodes(
          editor,
          {
            type: "variable",
            value: variableLabel,
            children: [{ text: "" }],
          },
          { at: selection }
        );
        Editor.normalize(editor, { force: true });
      }
    },
    [editor, isEditing]
  );

  // ----- Sanitize & Validate input (onType) -----
  const handleChange = (newValue) => {
    // Block too many blank lines and special chars
    let flat = Node.string({ children: newValue });
    // No >3 consecutive newlines
    if ((flat.match(/(\n{4,})/) || []).length) {
      setSnackbar({ open: true, msg: "❌ You cannot enter more than 3 consecutive blank lines." });
      return;
    }
    setWordCount(countWords(newValue));
    setBody(newValue);
  };

  // --------- Formatting: Toolbar demo (Bold, Italic, etc) ---------
  const markButton = (format, icon, tip) => (
    <Button
      key={format}
      size="small"
      variant={isMarkActive(editor, format) ? "contained" : "outlined"}
      color="primary"
      sx={{ minWidth: 32, fontWeight: "bold" }}
      onMouseDown={e => {
        e.preventDefault();
        toggleMark(editor, format);
      }}
    >
      {icon}
    </Button>
  );

  // --------- Save Handler --------
  const handleSave = () => {
    setIsEditing(false);
    // TODO: Call your backend with `body` value as serialized HTML or JSON.
    setSnackbar({ open: true, msg: "✅ Template saved successfully!" });
    setIsPreviewing(false);
  };
  // ------ Preview -----
  const handlePreview = () => {
    setIsPreviewing(true);
  };
  // ------ Reset ------
  const handleReset = () => {
    // Reset to initial state
    setBody(initialBody);
    setSnackbar({ open: true, msg: "✅ Template reset to default." });
    setIsEditing(false);
    setWordCount(countWords(initialBody));
  };

  // ----- Block copy from header ----
  const handleHeaderCopy = (e) => {
    e.preventDefault();
    setSnackbar({ open: true, msg: "❌ Copying from the header is not allowed!" });
  };

  // ----- Close Snackbar -----
  const handleSnackbarClose = () => setSnackbar({ open: false, msg: "" });

  // ------ Render ------
  return (
    <>
      <Typography gutterBottom variant="h5" sx={{ mb: 1 }}>
        Empanelment Intimation Letter Template (Slate)
      </Typography>
      <Divider sx={{ mb: 1 }} />

      <Box sx={{ p: 1, minHeight: "80%" }}>
        <Grid container spacing={3}>
          {/* ------ Sidebar ------- */}
          <Grid item xs={12} md={4}>
            <Card sx={{ mb: 3 }}>
              <CardHeader title="Variables" />
              <CardContent sx={{ display: "flex", flexWrap: "wrap", gap: 1 }}>
                {VARIABLE_LIST.map((label) => (
                  <Chip
                    key={label}
                    label={label}
                    color="primary"
                    draggable={isEditing}
                    onDragStart={e => handleDragStart(e, label)}
                    sx={{
                      borderRadius: "0",
                      cursor: isEditing ? "move" : "not-allowed",
                      fontWeight: "medium",
                    }}
                  />
                ))}
              </CardContent>
            </Card>
            <Card>
              <CardHeader title="Instructions" />
              <CardContent>
                <List dense>
                  {[
                    "Special characters are not allowed.",
                    "Only use variables from the list above.",
                    "Header section is non-editable.",
                    "Click 'Edit' to modify the template body.",
                    "Maximum word limit is 2000 words.",
                  ].map((text) => (
                    <ListItem key={text} disablePadding>
                      <ListItemIcon sx={{ minWidth: 32 }}>
                        <CheckCircleOutlineIcon color="primary" fontSize="small" />
                      </ListItemIcon>
                      <ListItemText primary={text} />
                    </ListItem>
                  ))}
                </List>
              </CardContent>
            </Card>
          </Grid>

          {/* ------- Main Editor ------ */}
          <Grid item xs={12} md={8}>
            <Card>
              <CardHeader title="Empanelment Intimation Letter Draft" />
              <CardContent>
                <Typography variant="subtitle1" sx={{ color: "text.secondary" }}>
                  Non Configurable
                </Typography>
                <Paper
                  variant="outlined"
                  sx={{
                    bgcolor: "#f5f5f5",
                    p: 1,
                    mb: 3,
                    border: "1px solid #e0e0e0",
                    borderRadius: 1,
                  }}
                  onCopy={handleHeaderCopy}
                >
                  <Slate editor={editor} value={header} onChange={() => {}}>
                    <Editable
                      readOnly
                      renderElement={renderElement}
                      style={{ minHeight: 48, color: "grey" }}
                    />
                  </Slate>
                </Paper>
                <Typography variant="subtitle1" sx={{ color: "text.secondary" }}>
                  Configurable
                </Typography>
                <Paper variant="outlined" sx={{ minHeight: 180, p: 1 }}>
                  <Slate
                    editor={editor}
                    value={body}
                    onChange={handleChange}
                  >
                    <Box
                      sx={{ display: "flex", gap: "4px", mb: 1 }}
                      // Minimal formatting
                    >
                      {markButton("bold", <b>B</b>, "Bold")}
                      {markButton("italic", <i>I</i>, "Italic")}
                      {markButton("underline", <u>U</u>, "Underline")}
                    </Box>
                    <Editable
                      renderElement={renderElement}
                      placeholder="Type the body here..."
                      readOnly={!isEditing}
                      style={{ minHeight: 96, outline: "none", background: "#fff" }}
                      spellCheck
                      onDrop={handleDrop}
                      onPaste={event => {
                        // Block special chars on paste
                        event.preventDefault();
                        const text = event.clipboardData.getData("text/plain");
                        const cleanText = sanitizeNodeText(text);
                        Editor.insertText(editor, cleanText);
                      }}
                    />
                  </Slate>
                  {/* Actions */}
                  <Box
                    sx={{
                      display: "flex",
                      justifyContent: "space-between",
                      alignItems: "center",
                      mt: 2,
                      mb: 1,
                    }}
                  >
                    <Typography variant="body2" color="textSecondary">
                      Word Count: {wordCount} / 2000
                    </Typography>
                    <Box sx={{ display: "flex", gap: 1 }}>
                      {!isEditing ? (
                        <Button
                          variant="contained"
                          startIcon={<EditIcon />}
                          onClick={() => setIsEditing(true)}
                        >
                          Edit
                        </Button>
                      ) : (
                        <Button
                          variant="outlined"
                          color="error"
                          startIcon={<CancelIcon />}
                          onClick={() => setIsEditing(false)}
                        >
                          Cancel
                        </Button>
                      )}
                      <Button
                        variant="outlined"
                        color="info"
                        startIcon={<PreviewIcon />}
                        onClick={handlePreview}
                      >
                        Preview
                      </Button>
                      <Button
                        variant="outlined"
                        color="secondary"
                        startIcon={<RestartAltIcon />}
                        onClick={handleReset}
                      >
                        Reset
                      </Button>
                    </Box>
                  </Box>
                </Paper>
              </CardContent>
            </Card>
          </Grid>
        </Grid>
      </Box>

      {/* -------- Preview Dialog -------- */}
      <Dialog
        open={isPreviewing}
        onClose={() => setIsPreviewing(false)}
        fullWidth
        maxWidth="md"
      >
        <DialogTitle>
          Intimation Letter Draft Preview
          <IconButton
            aria-label="close"
            onClick={() => setIsPreviewing(false)}
            sx={{
              position: "absolute",
              right: 8,
              top: 8,
              color: (theme) => theme.palette.grey[500],
            }}
          >
            <CloseIcon />
          </IconButton>
        </DialogTitle>
        <DialogContent>
          <Paper
            elevation={3}
            sx={{
              width: "210mm",
              minHeight: "297mm",
              margin: "2rem auto",
              padding: "20mm",
              boxSizing: "border-box",
              backgroundColor: "#fff",
              overflow: "auto",
              overflowWrap: "break-word",
            }}
          >
            {renderPreview(body)}
          </Paper>
        </DialogContent>
        <Box sx={{ p: 2, borderTop: "1px solid #eee", display: "flex", gap: 2 }}>
          <Button onClick={() => setIsPreviewing(false)} color="secondary">
            Cancel
          </Button>
          <Button
            autoFocus
            onClick={handleSave}
            variant="contained"
            startIcon={<SaveIcon />}
          >
            Save Draft
          </Button>
        </Box>
      </Dialog>

      {/* --- Snackbar for user feedback --- */}
      <Snackbar
        open={snackbar.open}
        autoHideDuration={6000}
        onClose={handleSnackbarClose}
        message={snackbar.msg}
        anchorOrigin={{ vertical: "bottom", horizontal: "right" }}
      />
    </>
  );
}

export default EmailTemplateWithSlate;


// ---------- Slate Utility Functions ---------
function countWords(nodes) {
  let text = Node.string({ children: nodes || [] });
  const words = text.trim().split(/\s+/).filter(Boolean);
  return words.length;
}

// ----- Mark Functions (Bold/Italic/etc) -----
const isMarkActive = (editor, format) => {
  const marks = Editor.marks(editor);
  return marks ? marks[format] === true : false;
};
const toggleMark = (editor, format) => {
  const isActive = isMarkActive(editor, format);
  if (isActive) Editor.removeMark(editor, format);
  else Editor.addMark(editor, format, true);
};

// ----- withVariables (for custom VariableElement insertion) -----
function withVariables(editor) {
  const { isInline } = editor;
  editor.isInline = (element) =>
    element.type === "variable" ? true : isInline(element);
  return editor;
}

// --------- Render body as Preview ---------
function renderPreview(nodes) {
  // Render as plain React nodes, but substitute dummy data for variables.
  return (
    <Slate
      editor={createEditor()}
      value={nodes}
      onChange={() => { }}
    >
      <Editable
        readOnly
        renderElement={props => {
          if (props.element.type === "variable") {
            const label = props.element.value;
            const key = label.slice(2, -2);
            return <span style={{
              background: "#e3f2fd",
              padding: "4px 8px",
              margin: "0 2px",
              borderRadius: 3,
            }}>{PREVIEW_MAP[key] || label}</span>;
          }
          return <p>{props.children}</p>;
        }}
      />
    </Slate>
  );
}
