Here are the changes required to implement the logic to fetch USERS_DATE from the FINCORE_DATE table instead of using the system date.
I will structure this by adding the new necessary files (Entity and Repository) and then modifying the existing Service and Controller.
1. Create New Entity: FincoreDate.java
Create this file in com.fincore.JournalService.Models.
Note: Since JPA requires an @Id, we will mark usersDate as the ID. Since there is only one row, this is safe.
package com.fincore.JournalService.Models;

import jakarta.persistence.Column;
import jakarta.persistence.Entity;
import jakarta.persistence.Id;
import jakarta.persistence.Table;
import lombok.Data;
import java.time.LocalDate;

@Entity
@Table(name = "FINCORE_DATE")
@Data
public class FincoreDate {

    // JPA requires an ID. Using usersDate as ID since table has no specific PK
    // and contains only one row.
    @Id
    @Column(name = "USERS_DATE")
    private LocalDate usersDate;

    @Column(name = "ETL_DATE")
    private LocalDate etlDate;
}

2. Create New Repository: FincoreDateRepository.java
Create this file in com.fincore.JournalService.Repository.
package com.fincore.JournalService.Repository;

import com.fincore.JournalService.Models.FincoreDate;
import org.springframework.data.jpa.repository.JpaRepository;
import org.springframework.data.jpa.repository.Query;
import org.springframework.stereotype.Repository;

import java.time.LocalDate;

@Repository
public interface FincoreDateRepository extends JpaRepository<FincoreDate, LocalDate> {

    // Custom query to fetch the single date efficiently
    @Query(value = "SELECT USERS_DATE FROM FINCORE_DATE FETCH FIRST 1 ROWS ONLY", nativeQuery = true)
    LocalDate findCurrentPostingDate();
}

4. Modify Implementation: JournalRequestServiceImpl.java
Update the class in com.fincore.JournalService.Service. Inject the new repository and implement the logic.
package com.fincore.JournalService.Service;

// ... existing imports ...
import com.fincore.JournalService.Repository.FincoreDateRepository; // Import the new repo

@Service
@RequiredArgsConstructor
@Slf4j
public class JournalRequestServiceImpl implements JournalRequestService {

    // ... existing repositories ...
    private final FincoreDateRepository fincoreDateRepository; // Inject new repo

    // ... existing methods ...

    @Override
    public String getCurrentPostingDate() {
        try {
            // Fetch date from DB
            LocalDate dbDate = fincoreDateRepository.findCurrentPostingDate();
            
            if (dbDate != null) {
                return dbDate.format(DateTimeFormatter.ISO_LOCAL_DATE);
            }
            
            // Fallback to system date if table is empty
            log.warn("FINCORE_DATE table is empty, falling back to System Date.");
            return LocalDate.now().format(DateTimeFormatter.ISO_LOCAL_DATE);
            
        } catch (Exception e) {
            log.error("Error fetching USERS_DATE from FINCORE_DATE", e);
            // Safety Fallback
            return LocalDate.now().format(DateTimeFormatter.ISO_LOCAL_DATE);
        }
    }
}
