import java.util.*;
import java.util.logging.Logger;



public class Main {

   
public static Boolean  rec(int[] arr , int left , int right , int target) {
    
    if(left == arr.length || right == arr.length) {
        return false;
    }
    
    
    if(arr[left] + arr[right] == target) {
        
        return true;
    }
    
    Boolean ans = rec(arr , left , right+1 , target);
    
    Boolean finalAns =  rec(arr , left + 1 , right , target);
    
    
    return ans || finalAns;
  
    
}

    

    
    
    
    public static void main(String[] args) {
        
    
      
      int arr[]=  {1,2,3,4,5};
      
      System.out.print(rec(arr , 0 , 1 , 2449));
      
    }
}
