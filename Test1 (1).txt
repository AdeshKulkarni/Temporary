This happens because FormData is tricky—you cannot simply console.log(formData) to see its contents. It will always look empty in the console unless you iterator over it.
Here is the complete solution with Debug Logging on both Frontend and Backend to prove exactly what is being sent and received.
1. Frontend: JournalBulkUpload.js (Updated)
I have added a loop to log the FormData entries so you can verify the file is actually attached before sending.

      setIsProcessing(false);
      setResultStatus("ERROR");
      setResultMsg("Init Failed: " + (err.message || "Unknown Error"));
    }

    // --- FILE UPLOAD (WITH DEBUG LOGS) ---
    @PostMapping(value = "/bulk-validate-init", consumes = MediaType.MULTIPART_FORM_DATA_VALUE)
    public ResponseEntity<?> initiateValidation(
            @RequestParam("file") MultipartFile file, 
            @RequestParam("postingDate") String date,
            HttpServletRequest request) { // Added Request to log headers
        
        try {
            // 1. LOG HEADER DETAILS
            log.info(">>> INCOMING UPLOAD REQUEST <<<");
            log.info("Content-Type Header: {}", request.getContentType());
            log.info("Content-Length Header: {}", request.getContentLength());

            // 2. LOG FILE DETAILS
            if (file == null) {
                log.error("File is NULL");
                return ResponseEntity.badRequest().body(Map.of("error", "File parameter is missing"));
            }
            
            log.info("File Name: {}", file.getOriginalFilename());
            log.info("File Size: {} bytes", file.getSize());
            log.info("File ContentType: {}", file.getContentType());
            log.info("Posting Date Param: {}", date);

            if (file.isEmpty()) {
                log.error("File is EMPTY (0 bytes)");
                return ResponseEntity.badRequest().body(Map.of("error", "File is empty"));
            }

            // 3. PROCESS
            String reqId = journalBulkValidationService.initiateValidation(
                file.getBytes(), 
                file.getOriginalFilename(), 
                LocalDate.parse(date)
            );
            
            log.info("Validation Queued. ReqID: {}", reqId);
            return ResponseEntity.ok(Map.of("status", "QUEUED", "requestId", reqId));

        } catch (Exception e) {
            log.error("File Upload Exception", e);
            return ResponseEntity.badRequest().body(Map.of("error", e.getMessage()));
        }
    }

        try {

            return ResponseEntity.ok().header(HttpHeaders.CONTENT_DISPOSITION, "attachment; filename=\"" + name + "\"").contentType(MediaType.parseMediaType(mime)).body(new ByteArrayResource(data));
