import React, { useState, useEffect, useCallback, useRef } from "react";
import {
  Box,
  Chip,
  IconButton,
  Tooltip,
  Stack,
  Typography,
  alpha
} from "@mui/material";
import { DataGrid } from "@mui/x-data-grid";
import { Visibility as VisibilityIcon, Info as InfoIcon } from "@mui/icons-material";

// Adjust these paths based on your folder structure
import useApi from "../../hooks/useApi"; 
import useCustomSnackbar from "../../utils/useCustomSnackbar";

// Import the Dialog Component
import TemplateViewDialog from "./TemplateViewDialog"; 

const styles = {

function CustomNoRowsOverlay() {
  return (
    <Stack height="100%" alignItems="center" justifyContent="center">
      <InfoIcon sx={{ fontSize: 48, color: "text.secondary", mb: 2 }} />
      <Typography variant="h6">No Templates Found</Typography>
    </Stack>
  );
}

export default function ReportViewScreen() {
  const { callApi } = useApi();
  const showSnackBar = useCustomSnackbar();
  const isMounted = useRef(true);
  
  const [loading, setLoading] = useState(true);
  const [rows, setRows] = useState([]);
  
  // Dialog State
  const [isViewDialogOpen, setIsViewDialogOpen] = useState(false);
  const [selectedTemplate, setSelectedTemplate] = useState(null);

  useEffect(() => {
    isMounted.current = true;
    return () => { isMounted.current = false; };
  }, []);

  const fetchTemplates = useCallback(async () => {
    setLoading(true);
    try {
      const response = await callApi("/template-config/get-templates", null, "GET");

      if (isMounted.current) {
        // Handling { message: "...", data: [...] } structure
        const dataList = response && Array.isArray(response.data) 
          ? response.data 
          : [];
        setRows(dataList);
      }
    } catch (err) {
      if (isMounted.current && err.name !== "CanceledError") {
        showSnackBar("Failed to load templates", "error");
      }
    } finally {
      if (isMounted.current) setLoading(false);
    }
  }, [callApi, showSnackBar]);

  useEffect(() => {
    fetchTemplates();
  }, [fetchTemplates]);

  const handleView = (row) => {
    setSelectedTemplate(row);
    setIsViewDialogOpen(true);
  };

  const columns = [
    {
      field: "templateName",
      headerName: "Template Name",
      flex: 1,
      minWidth: 180,
    },
    {
      field: "description",
      headerName: "Description",
      flex: 1.5,
      minWidth: 220,
    },
    {
      field: "status",
      headerName: "Status",
      width: 150, 
      align: "center",
      headerAlign: "center",
      renderCell: (params) => (
        // Just the Chip, no Edit Button
        <Chip 
            label={params.value} 
            color={params.value === "ACTIVE" ? "success" : "default"} 
            size="small" 
            variant="outlined"
            sx={{ fontWeight: 'bold' }}
        />
      ),
    },
    {
      field: "versionNo",
      headerName: "Ver.",
      width: 80,
      align: "center",
      headerAlign: "center",
    },
    {
      field: "actions",
      headerName: "Action",
      width: 100,
      sortable: false,
      align: "center",
      headerAlign: "center",
      renderCell: (params) => (
        <Tooltip title="View Report">
          <IconButton 
            color="primary" 
            onClick={() => handleView(params.row)}
          >
            <VisibilityIcon />
          </IconButton>
        </Tooltip>
      ),
    },
  ];

  return (
    <Box sx={styles.mainContainer}>
      {/* Title removed as requested */}
      
      <DataGrid
        rows={rows}
        columns={columns}
        loading={loading}
        getRowId={(row) => row.templateId}
        disableRowSelectionOnClick
        pageSizeOptions={[5, 10, 25]}
        rowHeight={60} 
        initialState={{
          pagination: { paginationModel: { page: 0, pageSize: 10 } },
        }}
        slots={{ noRowsOverlay: CustomNoRowsOverlay }}
        sx={styles.dataGridContainer}
      />

      <TemplateViewDialog 
        open={isViewDialogOpen} 
        handleClose={() => setIsViewDialogOpen(false)} 
        template={selectedTemplate} 
      />
    </Box>
  );
}

2. TemplateViewDialog.js
(The Pop-up Modal - Fully Styled)
import React, { useState, useEffect } from 'react';
import { 
  Dialog, DialogTitle, DialogContent, DialogActions, 
  Button, MenuItem, TextField, Divider, Grid, Typography, Box, IconButton, Chip 
} from '@mui/material';
import CloseIcon from '@mui/icons-material/Close';
import DescriptionIcon from '@mui/icons-material/Description';
import { alpha } from "@mui/material/styles";

const styles = {
    dialogTitle: {
        display: "flex",
        justifyContent: "space-between",
        alignItems: "center",
        backgroundColor: (theme) => alpha(theme.palette.primary.main, 0.08),
        pb: 2,
    },
};

const TemplateViewDialog = ({ open, handleClose, template }) => {
  const [selectedVariant, setSelectedVariant] = useState('');
  const [formData, setFormData] = useState({});

  useEffect(() => {
    if (open && template) {
        setSelectedVariant('');
        setFormData({});
    }
  }, [open, template]);

  if (!template) return null;

  const activeVariant = template.variants?.find(v => v.variantCode === selectedVariant);

  const handleInputChange = (name, value) => {
    setFormData(prev => ({ ...prev, [name]: value }));
  };

  const handleGenerate = () => {
    console.log("Generating Report for:", {
        template: template.templateName,
        variant: selectedVariant,
        data: formData
    });
    handleClose();
  };

  return (
    <Dialog 
        open={open} 
        onClose={handleClose} 
        maxWidth="sm" 
        fullWidth
        PaperProps={{ sx: { borderRadius: 2 } }}
    >
      {/* Styled Header */}
      <DialogTitle sx={styles.dialogTitle}>
        <Box display="flex" alignItems="center" gap={1}>
            <DescriptionIcon color="primary" />
            <Typography variant="h6" component="span" fontWeight="bold">
                {template.templateName}
            </Typography>
        </Box>
        <IconButton onClick={handleClose} size="small">
            <CloseIcon />
        </IconButton>
      </DialogTitle>

      <DialogContent dividers sx={{ p: 3 }}>
        <Typography variant="body2" color="text.secondary" sx={{ mb: 2, fontStyle: 'italic' }}>
            {template.description}
        </Typography>

        <Grid container spacing={3}>
          {/* Variant Selection */}
          <Grid item xs={12}>
            <TextField
              select
              fullWidth
              label="Select Report Variant"
              value={selectedVariant}
              onChange={(e) => setSelectedVariant(e.target.value)}
              helperText={
                !template.variants?.length 
                    ? "No variants available" 
                    : !activeVariant 
                        ? "Please select a variant to proceed" 
                        : activeVariant.description
              }
            >
              {template.variants?.map((v) => (
                <MenuItem key={v.variantCode} value={v.variantCode}>
                  {v.variantName}
                </MenuItem>
              ))}
            </TextField>
          </Grid>

          {/* Dynamic Inputs based on Params */}
          {activeVariant && (
              <>
                <Grid item xs={12}>
                    <Divider><Chip label="Parameters" size="small" /></Divider>
                </Grid>
                
                {activeVariant.params?.map((param) => (
                    <Grid item xs={12} key={param.paramName}>
                    <TextField
                        fullWidth
                        label={param.label}
                        type={param.paramType === 'DATE' ? 'date' : 'text'}
                        required={param.required}
                        InputLabelProps={{ shrink: true }} 
                        onChange={(e) => handleInputChange(param.paramName, e.target.value)}
                        placeholder={param.uiHint || ''}
                    />
                    </Grid>
                ))}

                {/* Export Format */}
                <Grid item xs={12}>
                    <Divider><Chip label="Export Options" size="small" /></Divider>
                </Grid>
                <Grid item xs={12}>
                    <TextField select fullWidth label="Export Format" defaultValue="pdf">
                    <MenuItem value="pdf">PDF Document</MenuItem>
                    <MenuItem value="excel">Excel Spreadsheet</MenuItem>
                    <MenuItem value="text">Plain Text</MenuItem>
                    </TextField>
                </Grid>
              </>
          )}
        </Grid>
      </DialogContent>
      
      <DialogActions sx={{ p: 2.5, backgroundColor: 'background.default' }}>
        <Button onClick={handleClose} color="inherit">Cancel</Button>
        <Button 
            variant="contained" 
            color="primary" 
            onClick={handleGenerate}
            disabled={!selectedVariant}
            sx={{ px: 4 }}
        >
          Generate
        </Button>
      </DialogActions>
    </Dialog>
  );
};

export default TemplateViewDialog;
