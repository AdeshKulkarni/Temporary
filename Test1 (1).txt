import React, { useState, useEffect, useCallback } from 'react';
import {
  Box,
  Typography,
  Grid,
  TextField,
  Button,
  Paper,
  IconButton,
  MenuItem,
  CircularProgress,
  Autocomplete,
} from '@mui/material';
import AddIcon from '@mui/icons-material/Add';
import DeleteIcon from '@mui/icons-material/Delete';
import { debounce } from 'lodash';

// --- Mock API Functions (Replace with your actual API calls) ---

// Mock useApi hook
const useApi = () => {
    // In a real app, this would handle base URLs, headers, auth, etc.
    const callApi = async (endpoint, payload, method = 'GET') => {
        console.log(`Calling API: ${method} ${endpoint}`, payload);
        // Simulate network delay
        await new Promise(res => setTimeout(res, 500));

        // Mock different endpoints
        if (endpoint.startsWith('/branches/search')) {
            const query = endpoint.split('=')[1];
            if (!query) return [];
            const items = [
                { code: "1001", name: "Main Branch" },
                { code: "1002", name: "City Center Branch" },
                { code: "2005", name: "Industrial Estate" },
            ];
            return items.filter(item => item.name.toLowerCase().includes(query.toLowerCase()) || item.code.includes(query));
        }
        if (endpoint === '/currency') {
            return {
                success: true,
                data: ["USD", "EUR", "INR", "GBP", "USD", "INR"],
            };
        }
        if (endpoint === '/cgls/classification') {
            return {
                "A": "Asset",
                "E": "Expense",
                "I": "Income",
                "L": "Liability",
                "M": "Memo A/c"
            };
        }
        if (endpoint.startsWith('/cgls/search')) {
             const query = endpoint.split('=')[1];
             if (!query) return [];
             const items = [
                 { cglNumber: "1423123411", description: "Office Supplies" },
                 { cglNumber: "1423123412", description: "Computer Equipment" },
                 { cglNumber: "2100001001", description: "Salaries and Wages" },
             ];
             return items.filter(item => item.description.toLowerCase().includes(query.toLowerCase()) || item.cglNumber.includes(query));
        }
        if(endpoint === '/submit-moc') {
            console.log("Submitting MOC Data:", payload);
            return { success: true, message: "MOC Posting submitted successfully." };
        }

        return null;
    };
    return { callApi };
};


// Mock useCustomSnackbar hook
const useCustomSnackbar = () => {
    return (message, severity) => {
        // In a real app, this would trigger a Snackbar/Toast notification
        console.log(`[${severity.toUpperCase()}]: ${message}`);
        alert(`[${severity.toUpperCase()}]: ${message}`);
    };
};


// --- Main Component ---

const initialRowState = {
  id: Date.now(),
  branch: null,
  postDate: new Date().toISOString().split('T')[0],
  currency: '',
  head: '',
  cgl: null,
  amount: '',
  txnType: 'Debit',
  remarks: '',
  productCode: '',
};

export default function MOCPosting() {
  const { callApi } = useApi();
  const showSnackBar = useCustomSnackbar();
  
  const [rows, setRows] = useState([initialRowState]);
  const [currencyOptions, setCurrencyOptions] = useState([]);
  const [headOptions, setHeadOptions] = useState([]);
  const [isSubmitting, setIsSubmitting] = useState(false);

  // States for Autocomplete fields
  const [branchSearch, setBranchSearch] = useState({ loading: false, options: [] });
  const [cglSearch, setCglSearch] = useState({ loading: false, options: [] });

  // Fetch initial dropdown data
  useEffect(() => {
    const fetchInitialData = async () => {
      try {
        // Fetch Currencies
        const currencyResponse = await callApi('/currency', null, 'GET');
        if (currencyResponse?.success && Array.isArray(currencyResponse.data)) {
          const uniqueCurrencies = [...new Set(currencyResponse.data)];
          setCurrencyOptions(uniqueCurrencies);
        } else {
            showSnackBar("Failed to load currencies.", "error");
        }

        // Fetch Head Classifications
        const headResponse = await callApi('/cgls/classification', null, 'GET');
        if (headResponse) {
            const formattedHeads = Object.entries(headResponse).map(([key, value]) => ({ key, value }));
            setHeadOptions(formattedHeads);
        } else {
            showSnackBar("Failed to load head classifications.", "error");
        }
      } catch (error) {
        console.error("Error fetching initial data:", error);
        showSnackBar("An error occurred while loading page data.", "error");
      }
    };

    fetchInitialData();
  }, []);
  
  const handleAddRow = () => {
    setRows([...rows, { ...initialRowState, id: Date.now() }]);
  };

  const handleDeleteRow = (id) => {
    if (rows.length > 1) {
      setRows(rows.filter((row) => row.id !== id));
    } else {
      showSnackBar("Cannot delete the last row.", "warning");
    }
  };

  const handleRowChange = (id, field, value) => {
    const newRows = rows.map((row) => {
      if (row.id === id) {
        // Reset CGL if head changes
        if (field === 'head' && row.head !== value) {
            return { ...row, [field]: value, cgl: null };
        }
        return { ...row, [field]: value };
      }
      return row;
    });
    setRows(newRows);
  };

  // --- Autocomplete Handlers with Debouncing ---

  const searchBranches = useCallback(
    debounce(async (query) => {
      if (!query) {
        setBranchSearch({ loading: false, options: [] });
        return;
      }
      setBranchSearch(prev => ({ ...prev, loading: true }));
      try {
        const response = await callApi(`/branches/search?query=${query}`, null, 'GET');
        setBranchSearch({ loading: false, options: response || [] });
      } catch (error) {
        console.error("Branch search failed:", error);
        setBranchSearch({ loading: false, options: [] });
      }
    }, 300),
    []
  );

  const searchCgls = useCallback(
    debounce(async (query, headKey) => {
      if (!query || !headKey) {
        setCglSearch({ loading: false, options: [] });
        return;
      }
      setCglSearch(prev => ({ ...prev, loading: true }));
      try {
        // In a real API, you might pass the headKey: `/cgls/search?query=${query}&classification=${headKey}`
        const response = await callApi(`/cgls/search?query=${query}`, null, 'GET');
        setCglSearch({ loading: false, options: response || [] });
      } catch (error) {
        console.error("CGL search failed:", error);
        setCglSearch({ loading: false, options: [] });
      }
    }, 300),
    []
  );

  const handleSubmit = async () => {
      setIsSubmitting(true);
      // Basic Validation
      for (const row of rows) {
          if (!row.branch || !row.currency || !row.head || !row.cgl || !row.amount || !row.txnType || !row.productCode) {
              showSnackBar("Please fill all fields in all rows before submitting.", "error");
              setIsSubmitting(false);
              return;
          }
      }

      try {
        await callApi('/submit-moc', { transactions: rows }, 'POST');
        showSnackBar("MOC Posting submitted successfully!", "success");
        setRows([{ ...initialRowState, id: Date.now() }]); // Reset form
      } catch (error) {
        showSnackBar("Submission failed. Please try again.", "error");
      } finally {
        setIsSubmitting(false);
      }
  };

  return (
    <Box sx={{ p: 3, maxWidth: '95%', margin: 'auto' }}>
      <Paper elevation={3} sx={{ p: 3 }}>
        <Typography variant="h4" gutterBottom sx={{ fontWeight: 'bold', color: 'primary.main', mb: 3 }}>
          MOC Posting
        </Typography>

        {rows.map((row, index) => (
          <Grid container spacing={2} key={row.id} alignItems="center" sx={{ mb: 2 }}>
            {/* Branch */}
            <Grid item xs={12} md={2}>
              <Autocomplete
                options={branchSearch.options}
                getOptionLabel={(option) => `${option.code} - ${option.name}`}
                value={row.branch}
                onChange={(event, newValue) => handleRowChange(row.id, 'branch', newValue)}
                onInputChange={(event, newInputValue) => searchBranches(newInputValue)}
                loading={branchSearch.loading}
                renderInput={(params) => (
                  <TextField
                    {...params}
                    label="Branch"
                    placeholder="Search by name or code"
                    InputProps={{
                      ...params.InputProps,
                      endAdornment: (
                        <>
                          {branchSearch.loading ? <CircularProgress color="inherit" size={20} /> : null}
                          {params.InputProps.endAdornment}
                        </>
                      ),
                    }}
                  />
                )}
              />
            </Grid>
            
            {/* Post Date */}
            <Grid item xs={12} md={1.5}>
                <TextField label="Post Date" value={row.postDate} fullWidth disabled />
            </Grid>
            
            {/* Currency */}
            <Grid item xs={12} md={1}>
                <TextField select label="Currency" value={row.currency} onChange={(e) => handleRowChange(row.id, 'currency', e.target.value)} fullWidth>
                    {currencyOptions.map((c) => (<MenuItem key={c} value={c}>{c}</MenuItem>))}
                </TextField>
            </Grid>
            
            {/* Head */}
            <Grid item xs={12} md={1.5}>
                <TextField select label="Head" value={row.head} onChange={(e) => handleRowChange(row.id, 'head', e.target.value)} fullWidth>
                    {headOptions.map((h) => (<MenuItem key={h.key} value={h.key}>{h.value}</MenuItem>))}
                </TextField>
            </Grid>

            {/* CGL */}
            <Grid item xs={12} md={2}>
                 <Autocomplete
                    options={cglSearch.options}
                    getOptionLabel={(option) => `${option.cglNumber} - ${option.description}`}
                    value={row.cgl}
                    onChange={(event, newValue) => handleRowChange(row.id, 'cgl', newValue)}
                    onInputChange={(event, newInputValue) => searchCgls(newInputValue, row.head)}
                    loading={cglSearch.loading}
                    disabled={!row.head}
                    renderInput={(params) => (
                        <TextField
                            {...params}
                            label="CGL"
                            placeholder={row.head ? "Search CGL" : "Select Head first"}
                            InputProps={{
                                ...params.InputProps,
                                endAdornment: (
                                    <>
                                        {cglSearch.loading ? <CircularProgress color="inherit" size={20} /> : null}
                                        {params.InputProps.endAdornment}
                                    </>
                                ),
                            }}
                        />
                    )}
                />
            </Grid>

            {/* Amount */}
            <Grid item xs={12} md={1.5}>
                <TextField
                    label="Amount"
                    type="text"
                    value={row.amount}
                    onChange={(e) => {
                        const val = e.target.value;
                        if (/^\d{0,16}(\.\d{0,6})?$/.test(val)) {
                           handleRowChange(row.id, 'amount', val);
                        }
                    }}
                    fullWidth
                    placeholder="0.00"
                />
            </Grid>

            {/* Txn Type */}
            <Grid item xs={12} md={1}>
                <TextField select label="Txn Type" value={row.txnType} onChange={(e) => handleRowChange(row.id, 'txnType', e.target.value)} fullWidth>
                    <MenuItem value="Debit">Debit</MenuItem>
                    <MenuItem value="Credit">Credit</MenuItem>
                </TextField>
            </Grid>
            
            {/* Remarks */}
            <Grid item xs={12} md={2}>
                 <TextField
                    label="Remarks"
                    value={row.remarks}
                    onChange={(e) => {
                         const val = e.target.value;
                         if (/^[a-zA-Z0-9\s.,?/-]*$/.test(val)) {
                            handleRowChange(row.id, 'remarks', val);
                         }
                    }}
                    fullWidth
                    inputProps={{ maxLength: 40 }}
                />
            </Grid>

            {/* Product Code */}
            <Grid item xs={12} md={1.5}>
                 <TextField
                    label="Product Code"
                    value={row.productCode}
                    onChange={(e) => {
                         const val = e.target.value;
                         if (/^\d{0,8}$/.test(val)) {
                            handleRowChange(row.id, 'productCode', val);
                         }
                    }}
                    fullWidth
                    inputProps={{ maxLength: 8 }}
                />
            </Grid>
            
            {/* Actions */}
            <Grid item xs={12} md={0.5} sx={{ textAlign: 'center' }}>
              <IconButton onClick={() => handleDeleteRow(row.id)} color="error" aria-label="delete row">
                <DeleteIcon />
              </IconButton>
            </Grid>
          </Grid>
        ))}
        
        <Box sx={{ mt: 3, display: 'flex', gap: 2 }}>
            <Button variant="contained" startIcon={<AddIcon />} onClick={handleAddRow}>
                Add Row
            </Button>
            <Button variant="contained" color="success" onClick={handleSubmit} disabled={isSubmitting}>
                {isSubmitting ? <CircularProgress size={24} /> : 'Submit'}
            </Button>
        </Box>
      </Paper>
    </Box>
  );
}
