package com.fincore.VoucherService.Service.validation;

import com.fincore.VoucherService.Dto.ExcelRowData;
import com.fincore.VoucherService.Service.ValidationMasterService;
import org.junit.jupiter.api.Test;

import java.math.BigDecimal;
import java.util.ArrayList;
import java.util.Arrays;
import java.util.Collections;
import java.util.HashSet;
import java.util.Set;

import static org.junit.jupiter.api.Assertions.assertEquals;
import static org.junit.jupiter.api.Assertions.assertTrue;

public class SimpleValidatorTest {

    // 1. Create a "Fake" Master Service purely in Java (No Mockito needed)
    private ValidationMasterService createFakeMasterService() {
        return new ValidationMasterService(null) {
            @Override
            public Set<String> getAllActiveBranches() {
                return new HashSet<>(Arrays.asList("10301", "10302"));
            }
            @Override
            public Set<String> getAllActiveCurrencies() {
                return new HashSet<>(Collections.singletonList("INR"));
            }
            @Override
            public Set<String> getAllActiveCgls() {
                return new HashSet<>(Arrays.asList("1102003001", "1102003002"));
            }
        };
    }

    // 2. Helper method to easily create rows for testing
    private ExcelRowData createRow(String branch, String currency, String cgl, String amountStr, String type) {
        ExcelRowData row = new ExcelRowData();
        row.branch = branch;
        row.currency = currency;
        row.cgl = cgl;
        row.amount = new BigDecimal(amountStr);
        row.txnType = type;
        row.remarks = "Valid Remarks";
        
        // Initialize the error lists so we don't get NullPointerExceptions
        row.formatErrors = Collections.synchronizedList(new ArrayList<>());
        row.dbErrors = Collections.synchronizedList(new ArrayList<>());
        row.balErrors = Collections.synchronizedList(new ArrayList<>());
        return row;
    }

    @Test
    public void testFormatCheck_ConvertsCrToCredit() {
        // Setup
        VoucherRuleValidator validator = new VoucherRuleValidator(createFakeMasterService());
        ExcelRowData row = createRow("10301", "INR", "1102003001", "500.00", "CR");

        // Execute
        validator.runFormatCheck(Collections.singletonList(row));

        // Verify
        assertEquals("Credit", row.txnType, "The validator should convert 'CR' to 'Credit'");
        assertTrue(row.formatErrors.isEmpty(), "There should be no format errors");
    }

    @Test
    public void testDatabaseCheck_InvalidBranchFails() {
        // Setup
        VoucherRuleValidator validator = new VoucherRuleValidator(createFakeMasterService());
        // 99999 is NOT in our fake master service
        ExcelRowData row = createRow("99999", "INR", "1102003001", "500.00", "Debit");

        // Execute
        validator.runDbCheck(Collections.singletonList(row));

        // Verify
        assertTrue(row.dbErrors.contains("Branch is inactive or not found: 99999"));
    }

    @Test
    public void testBalanceCheck_PerfectlyBalancedPasses() {
        // Setup
        VoucherRuleValidator validator = new VoucherRuleValidator(createFakeMasterService());
        
        // Total Debit: 1000.00 | Total Credit: 1000.00 -> Balanced!
        ExcelRowData creditRow = createRow("10301", "INR", "1102003001", "1000.00", "Credit");
        ExcelRowData debitRow1 = createRow("10301", "INR", "1102003002", "700.00", "Debit");
        ExcelRowData debitRow2 = createRow("10301", "INR", "1102003002", "300.00", "Debit");

        // Execute
        validator.runBalanceCheck(Arrays.asList(creditRow, debitRow1, debitRow2));

        // Verify
        assertTrue(creditRow.balErrors.isEmpty(), "Credit row should have no balance errors");
        assertTrue(debitRow1.balErrors.isEmpty(), "Debit row 1 should have no balance errors");
    }

    @Test
    public void testBalanceCheck_UnbalancedFails() {
        // Setup
        VoucherRuleValidator validator = new VoucherRuleValidator(createFakeMasterService());
        
        // Total Debit: 900.00 | Total Credit: 1000.00 -> Unbalanced!
        ExcelRowData creditRow = createRow("10301", "INR", "1102003001", "1000.00", "Credit");
        ExcelRowData debitRow = createRow("10301", "INR", "1102003002", "900.00", "Debit");

        // Execute
        validator.runBalanceCheck(Arrays.asList(creditRow, debitRow));

        // Verify
        assertTrue(creditRow.balErrors.contains("Branch, Currency, and CGL-wise balance mismatch."));
        assertTrue(debitRow.balErrors.contains("Branch, Currency, and CGL-wise balance mismatch."));
    }
}
