class Solution {

    public int SizeList(int n ,  List<List<Integer>> adj , int node , List<Boolean> visit , int count) {

        List<Integer> ans = new ArrayList<>();

        visit.set(node, true); 
        visit.get(node); 
        count = 1;

        for(int i=0; i<adj.get(node).size(); i++) {

            int nabar = adj.get(node).get(i);
            D.p(count);

            if(visit.get(nabar) == false) {
               
               count += SizeList(n , adj , nabar , visit , count);
               

            }          
          
                      
        }
        return count;

    }
    public long countPairs(int n, int[][] edges) {

        List<List<Integer>> adj = new ArrayList<>(n);

        for (int i = 0; i < n; i++) {
            adj.add(new ArrayList<>());
        }

        for(int i=0; i<edges.length; i++) {

            int[] temp = edges[i];

            int src = temp[0];
            int dis = temp[1];

            adj.get(src).add(dis);
            adj.get(dis).add(src);
                

        }

        List<Boolean> visit = new ArrayList<>(Collections.nCopies(n, false));

        List<Integer> sizes = new ArrayList<>();

        for(int i=0; i<n; i++) {

            if(visit.get(i)== false) {
                int size = SizeList(n , adj , i , visit , 0);
                sizes.add(size);
            }
        }     

        long ans = 0;

        for(int i=0; i<sizes.size(); i++) {

            for(int j=0; j<sizes.size(); j++) {

                ans += sizes.get(i) * sizes.get(j);
                D.p(sizes.get(i) , sizes.get(j));
            }
        }

        return ans;

        
    }
}
