import React, { useState, useCallback } from 'react';
import axios from 'axios';
import { Button, CircularProgress, Box, Typography } from '@mui/material';
import PdfView from './PdfView'; // Assuming your pdfview component is here

const Workflow = () => {
  // State to manage the loading spinner's visibility
  const [isLoading, setIsLoading] = useState(false);

  // State to hold the fetched PDF data (e.g., a blob URL)
  const [pdfFileUrl, setPdfFileUrl] = useState(null);

  // State to display any errors that occur during the fetch
  const [error, setError] = useState('');

  /**
   * Fetches the PDF from the server when the user clicks the button.
   */
  const handlePreviewClick = useCallback(async () => {
    setIsLoading(true);   // 1. Show loading spinner
    setPdfFileUrl(null);  // Clear any previous PDF
    setError('');         // Clear any previous error

    try {
      // NOTE: Replace with your actual API endpoint and payload
      const response = await axios.post(
        "Server/IAMServices/generatePdfPreview", // Your API endpoint
        { /* Your request data */ },
        {
          headers: { Authorization: `Bearer ${localStorage.getItem("token")}` },
          // This is crucial: it tells axios to handle the response as a binary file
          responseType: 'blob',
        }
      );

      // Create a local URL for the fetched blob object
      const file = new Blob([response.data], { type: 'application/pdf' });
      const fileURL = URL.createObjectURL(file);

      // 3. Set the PDF data, which will trigger the PdfView to render
      setPdfFileUrl(fileURL);

    } catch (err) {
      console.error("Failed to fetch PDF:", err);
      setError("Could not generate the letter preview. Please try again later.");
    } finally {
      setIsLoading(false); // 4. Hide loading spinner
    }
  }, []); // useCallback ensures this function isn't recreated on every render

  return (
    <Box sx={{ padding: 2 }}>
      <Typography variant="h5" gutterBottom>
        Document Workflow
      </Typography>
      
      <Button
        variant="contained"
        onClick={handlePreviewClick}
        disabled={isLoading} // Disable button while loading
      >
        {isLoading ? 'Generating Preview...' : 'Preview Letter'}
      </Button>

      <Box sx={{ marginTop: 3, padding: 2, border: '1px dashed grey', minHeight: '500px' }}>
        {/* Case 1: Show a loading spinner while fetching data */}
        {isLoading && (
          <Box sx={{ display: 'flex', justifyContent: 'center', alignItems: 'center', height: '100%' }}>
            <CircularProgress />
            <Typography sx={{ marginLeft: 2 }}>Loading PDF...</Typography>
          </Box>
        )}

        {/* Case 2: Show an error message if the API call failed */}
        {!isLoading && error && (
          <Typography color="error" align="center">
            {error}
          </Typography>
        )}

        {/* Case 3: ONLY render PdfView when loading is false and we have a valid URL */}
        {!isLoading && pdfFileUrl && (
          <PdfView fileUrl={pdfFileUrl} />
        )}

        {/* Case 4: Initial state before the button is clicked */}
        {!isLoading && !pdfFileUrl && !error && (
            <Typography align="center" color="text.secondary">
                Click "Preview Letter" to generate the document.
            </Typography>
        )}
      </Box>
    </Box>
  );
};

export default Workflow;
