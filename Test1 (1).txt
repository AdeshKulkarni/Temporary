    @Override
    @Transactional
    public String createBulkBatchRequest(BatchRequestDto batchDto, String creatorId, Integer creatorRole) throws JsonProcessingException {
        long startTime = System.currentTimeMillis();

        String batchId = sequenceService.getNextBatchId();
        LocalDate postingDate = LocalDate.now();
        String commonRemarks = (batchDto.getCommonBatchRemarks() != null && !batchDto.getCommonBatchRemarks().isEmpty())
                ? batchDto.getCommonBatchRemarks()
                : "Bulk Upload Batch " + batchId;

        // Group rows by Branch + Currency
        Map<String, List<BatchRequestDto.JournalRequestRow>> groupedRows = batchDto.getRows().stream()
                .collect(Collectors.groupingBy(row -> row.getBranch() + ":" + row.getCurrency()));

        int totalJournals = groupedRows.size();
        log.info("Processing {} rows into {} journals for Batch {}", batchDto.getRows().size(), totalJournals, batchId);

        List<String> journalPrefixes = fetchJournalSequences(totalJournals);
        Iterator<String> prefixIterator = journalPrefixes.iterator();
        
        DateTimeFormatter dateFmt = DateTimeFormatter.ISO_DATE;
        
        String sql = "INSERT INTO JOURNAL_REQUEST " +
                     "(REQ_ID, REQ_STATUS, CHANGE_TYPE, REQ_DATE, CREATOR_ID, CREATOR_ROLE, BATCH_ID, JOURNAL_ID, COMMON_BATCH_REMARKS, PAYLOAD) " +
                     "VALUES (JOURNAL_REQUEST_SEQ.nextval, ?, ?, ?, ?, ?, ?, ?, ?, ?)";

        final int INSERT_BATCH_SIZE = 1000;
        List<Object[]> batchArgs = new ArrayList<>(INSERT_BATCH_SIZE);
        int processedCount = 0;

        for (Map.Entry<String, List<BatchRequestDto.JournalRequestRow>> entry : groupedRows.entrySet()) {
            String journalIdPrefix = prefixIterator.next();
            int journalSequence = 1;

            for (BatchRequestDto.JournalRequestRow rowDto : entry.getValue()) {
                String journalId = String.format("%s-%03d", journalIdPrefix, journalSequence);
                
                // Manual JSON builder (Faster than Jackson)
                String payload = buildJsonPayloadFast(rowDto, postingDate, batchId, journalId, commonRemarks, journalSequence, dateFmt);

                batchArgs.add(new Object[]{
                    RequestStatus.PENDING.getCode(), // Inserts "P"
                    rowDto.getChangeType().name(), 
                    LocalDateTime.now(), 
                    creatorId, 
                    creatorRole,
                    batchId, 
                    journalId, 
                    commonRemarks, 
                    payload
                });
                
                // Flush to DB if batch is full
                if (batchArgs.size() >= INSERT_BATCH_SIZE) {
                    jdbcTemplate.batchUpdate(sql, batchArgs);
                    batchArgs.clear();
                }
                
                journalSequence++;
                processedCount++;
            }
        }

        // Final Flush
        if (!batchArgs.isEmpty()) {
            jdbcTemplate.batchUpdate(sql, batchArgs);
        }

        log.info("Batch Insert Complete: {} rows in {} ms", processedCount, (System.currentTimeMillis() - startTime));
        createNotification(batchId, creatorId, processedCount);

        return batchId;
    }

    private List<String> fetchJournalSequences(int count) {
        if (count == 0) return Collections.emptyList();
        String sql = "SELECT JRNL_SEQ.nextval FROM dual CONNECT BY LEVEL <= ?";
        List<Long> seqs = jdbcTemplate.queryForList(sql, Long.class, count);
        return seqs.stream().map(id -> String.format("%07d", id)).collect(Collectors.toList());
    }

    private String buildJsonPayloadFast(BatchRequestDto.JournalRequestRow row, LocalDate pDate, String batchId, String jId, String rem, int count, DateTimeFormatter fmt) {
        StringBuilder sb = new StringBuilder(400); 
        sb.append("{");
        sb.append("\"changeType\":\"").append(row.getChangeType()).append("\",");
        sb.append("\"masterJournalId\":").append(row.getMasterJournalId() == null ? "null" : row.getMasterJournalId()).append(",");
        sb.append("\"pDate\":\"").append(pDate.format(fmt)).append("\",");
        sb.append("\"branch\":\"").append(row.getBranch()).append("\",");
        sb.append("\"currency\":\"").append(row.getCurrency()).append("\",");
        sb.append("\"cgl\":\"").append(row.getCgl()).append("\",");
        sb.append("\"amount\":").append(row.getAmount()).append(","); 
        sb.append("\"productType\":").append(row.getProductType() == null ? "null" : "\"" + row.getProductType() + "\"").append(",");
        sb.append("\"remarks\":").append(row.getRemarks() == null ? "null" : "\"" + escapeJson(row.getRemarks()) + "\"").append(",");
        sb.append("\"arFlag\":\"").append(row.getArFlag() == null ? "A" : row.getArFlag()).append("\",");
        sb.append("\"acClassification\":\"").append(row.getAcClassification() == null ? "A" : row.getAcClassification()).append("\",");
        sb.append("\"batchId\":\"").append(batchId).append("\",");
        sb.append("\"journalId\":\"").append(jId).append("\",");
        sb.append("\"commonBatchRemarks\":\"").append(escapeJson(rem)).append("\",");
        sb.append("\"transactionCount\":").append(count);
        sb.append("}");
        return sb.toString();
    }
    
    private String escapeJson(String s) {
        if(s == null) return "";
        return s.replace("\"", "\\\"").replace("\\", "\\\\");
    }

    @Override
    @Transactional
    public List<JournalRequest> processBulkRequests(BulkProcessJournalRequestDto dto, String executorId, Integer executorRole) {
        long startTime = System.currentTimeMillis();
        
        if (!StringUtils.hasText(dto.getBatchId()) && (dto.getJournalIdPrefixes() == null || dto.getJournalIdPrefixes().isEmpty())) {
            throw new IllegalArgumentException("Either Batch ID or Journal ID list is required.");
        }

        String targetClause;
        List<Object> args = new ArrayList<>();
        
        if (StringUtils.hasText(dto.getBatchId())) {
            targetClause = "BATCH_ID = ?";
            args.add(dto.getBatchId());
        } else {
            List<String> conditions = new ArrayList<>();
            for (String prefix : dto.getJournalIdPrefixes()) {
                conditions.add("JOURNAL_ID LIKE ?");
                args.add(prefix + "%");
            }
            targetClause = "(" + String.join(" OR ", conditions) + ")";
        }

        // IF APPROVED: Insert using Native SQL (0 Java Overhead)
        if (RequestStatus.ACCEPTED.equals(dto.getStatus())) {
            log.info("Starting Bulk Approval Insert for executor: {}", executorId);
            
            String insertSql = """
                INSERT INTO GL_TRANSACTIONS (
                    TRANSACTION_ID, BATCH_ID, JOURNAL_ID, TRANSACTION_DATE, POST_DATE,
                    BRANCH_CODE, CURRENCY, CGL, NARRATION,
                    DEBIT_AMOUNT, CREDIT_AMOUNT, TRANSACTION_COUNT, SOURCE_FLAG
                )
                SELECT
                    GL_TRANSACTIONS_SEQ.nextval,
                    jr.BATCH_ID,
                    jr.JOURNAL_ID,
                    TO_DATE(JSON_VALUE(jr.PAYLOAD, '$.pDate'), 'YYYY-MM-DD'),
                    SYSTIMESTAMP,
                    JSON_VALUE(jr.PAYLOAD, '$.branch'),
                    JSON_VALUE(jr.PAYLOAD, '$.currency'),
                    JSON_VALUE(jr.PAYLOAD, '$.cgl'),
                    JSON_VALUE(jr.PAYLOAD, '$.remarks'),
                    
                    -- DEBIT Logic: If > 0, it's Debit. Else 0.
                    CASE WHEN TO_NUMBER(JSON_VALUE(jr.PAYLOAD, '$.amount')) > 0 
                         THEN TO_NUMBER(JSON_VALUE(jr.PAYLOAD, '$.amount')) 
                         ELSE 0 END,
                         
                    -- CREDIT Logic: If < 0, insert the NEGATIVE value (no ABS).
                    CASE WHEN TO_NUMBER(JSON_VALUE(jr.PAYLOAD, '$.amount')) < 0 
                         THEN TO_NUMBER(JSON_VALUE(jr.PAYLOAD, '$.amount')) 
                         ELSE 0 END,
                         
                    TO_NUMBER(JSON_VALUE(jr.PAYLOAD, '$.transactionCount')),
                    'J'
                FROM JOURNAL_REQUEST jr
                WHERE jr.REQ_STATUS = 'P' -- Look for 'P' Code
                  AND jr.CREATOR_ID != ?  -- Security: Prevent Self-Approval
                  AND 
            """ + targetClause;

            List<Object> insertArgs = new ArrayList<>();
            insertArgs.add(executorId);
            insertArgs.addAll(args);

            int rowsInserted = jdbcTemplate.update(insertSql, insertArgs.toArray());
            log.info("Bulk Approval: Inserted {} rows into GL_TRANSACTIONS", rowsInserted);
        }

        // Update Status in JOURNAL_REQUEST (Bulk Update)
        String updateSql = "UPDATE JOURNAL_REQUEST SET REQ_STATUS = ?, EXECUTOR_ID = ?, EXECUTOR_REMARKS = ?, EXECUTION_DATE = SYSDATE " +
                           "WHERE REQ_STATUS = 'P' AND CREATOR_ID != ? AND " + targetClause;

        List<Object> updateArgs = new ArrayList<>();
        updateArgs.add(dto.getStatus().getCode()); // Fix: Use "A" or "R" Code
        updateArgs.add(executorId);
        updateArgs.add(dto.getRemarks());
        updateArgs.add(executorId); // Security Check
        updateArgs.addAll(args);

        int rowsUpdated = jdbcTemplate.update(updateSql, updateArgs.toArray());
        log.info("Bulk Status Update: Updated {} rows to {}", rowsUpdated, dto.getStatus());
        log.info("Total Processing Time: {} ms", (System.currentTimeMillis() - startTime));
        
        return Collections.emptyList(); 
    }

    public Optional<JournalRequest> updateRequestStatus(ProcessJournalRequestDto dto, String executorId, Integer executorRole) {
        JournalRequest request = journalRequestRepository.findById(dto.getRequestId())
                .orElseThrow(() -> new ResourceNotFoundException("Journal Request not found."));

        if (request.getRequestStatus() != RequestStatus.PENDING) throw new IllegalStateException("Already processed.");
        if (executorId.equals(request.getCreatorId())) throw new IllegalStateException("Self-approval not allowed.");

        if (RequestStatus.ACCEPTED.equals(dto.getStatus())) {
             String insertSql = """
                INSERT INTO GL_TRANSACTIONS (
                    TRANSACTION_ID, BATCH_ID, JOURNAL_ID, TRANSACTION_DATE, POST_DATE,
                    BRANCH_CODE, CURRENCY, CGL, NARRATION,
                    DEBIT_AMOUNT, CREDIT_AMOUNT, TRANSACTION_COUNT, SOURCE_FLAG
                )
                SELECT
                    GL_TRANSACTIONS_SEQ.nextval,
                    BATCH_ID, JOURNAL_ID, TO_DATE(JSON_VALUE(PAYLOAD, '$.pDate'), 'YYYY-MM-DD'), SYSTIMESTAMP,
                    JSON_VALUE(PAYLOAD, '$.branch'), JSON_VALUE(PAYLOAD, '$.currency'), JSON_VALUE(PAYLOAD, '$.cgl'), JSON_VALUE(PAYLOAD, '$.remarks'),
                    CASE WHEN TO_NUMBER(JSON_VALUE(PAYLOAD, '$.amount')) > 0 THEN TO_NUMBER(JSON_VALUE(PAYLOAD, '$.amount')) ELSE 0 END,
                    CASE WHEN TO_NUMBER(JSON_VALUE(PAYLOAD, '$.amount')) < 0 THEN TO_NUMBER(JSON_VALUE(PAYLOAD, '$.amount')) ELSE 0 END,
                    TO_NUMBER(JSON_VALUE(PAYLOAD, '$.transactionCount')), 'J'
                FROM JOURNAL_REQUEST WHERE REQ_ID = ?
            """;
            jdbcTemplate.update(insertSql, request.getId());
        }

        request.setExecutorId(executorId);
        request.setExecutorRemarks(dto.getRemarks());
        request.setExecutionDate(LocalDateTime.now());
        request.setRequestStatus(dto.getStatus());
        return Optional.of(journalRequestRepository.save(request));
    }

    @Override
    @Transactional
    public void cancelMyRequestsByJournalPrefixes(List<String> prefixes, String userId) {
         List<JournalRequest> requestsToCancel = journalRequestRepository.findAll((root, query, cb) -> {
            List<Predicate> predicates = new ArrayList<>();
            for (String prefix : prefixes) {
                predicates.add(cb.like(root.get("journalId"), prefix + "%"));
            }
            Predicate journalIdOr = cb.or(predicates.toArray(new Predicate[0]));
            Predicate status = cb.equal(root.get("requestStatus"), RequestStatus.PENDING);
            Predicate creator = cb.equal(root.get("creatorId"), userId);
            return cb.and(journalIdOr, status, creator);
        });
        if (requestsToCancel.isEmpty()) throw new ResourceNotFoundException("No pending requests found.");
        journalRequestRepository.deleteAll(requestsToCancel);
    }

                const payload = JSON.parse(req.payload);
                const prefix = req.journalId.split("-")[0];
                const amount = parseFloat(payload.amount) || 0;
                return {
                  ...req, payload, journalPrefix: prefix, amount: Math.abs(amount),
                  transactionType: amount < 0 ? "Credit" : "Debit",
                };
              } catch (e) { return null; }
            }).filter(Boolean);

          processedData.sort((a, b) => a.journalId.localeCompare(b.journalId));
          for (const row of processedData) {
            const prefix = row.journalPrefix;
            if (!journalGroups.has(prefix)) journalGroups.set(prefix, []);
            journalGroups.get(prefix).push(row);
          }
          setBatchDetails(Array.from(journalGroups.values()));
        }
        setIsViewLoading(false);
      }, 100);
    } catch (error) {
      showSnackBar("Failed to load details", "error");
      setIsViewLoading(false);
    }
  };

  const handleCloseViewModal = () => {
    setIsViewModalOpen(false);
    setBatchDetails([]);
    setViewingBatchId(null);
    setViewingBatchCreatorId(null);
    setSelectedJournalPrefixes([]);
    setPartialExecutorRemarks("");
    setIsPartialSubmitting(false);
    // Refresh main grid on close to reflect updates
    fetchPendingBatches();
  };

  const handleSubmitPartialAction = async (mode) => {
    if (mode === "REJECTED" && !partialExecutorRemarks.trim()) {
      showSnackBar("Remarks required for rejection.", "warning"); return;
    }
    if (selectedJournalPrefixes.length === 0) {
      showSnackBar("Select at least one entry.", "warning"); return;
    }
    setIsPartialSubmitting(true);
    try {
      await callApi("/JS/journals/process-bulk", {
        journalIdPrefixes: selectedJournalPrefixes,
        status: mode,
        remarks: partialExecutorRemarks.trim() || (mode === "ACCEPTED" ? "Accepted" : "Rejected"),
      }, "POST");
      showSnackBar(`Selection ${mode === "ACCEPTED" ? "Approved" : "Rejected"}`, "success");
      handleCloseViewModal();
    } catch (error) {
      showSnackBar("Failed to process selection", "error");
      setIsPartialSubmitting(false);
    }
  };

  const handleSelectAllClick = (event) => {
    if (event.target.checked) {
      setSelectedJournalPrefixes(batchDetails.map(g => g[0].journalPrefix));
    } else {
      setSelectedJournalPrefixes([]);
    }
  };

  const handleRowCheckboxClick = (event, prefix) => {
    if (event) event.stopPropagation();
    const index = selectedJournalPrefixes.indexOf(prefix);
    let newSelected = [];
    if (index === -1) newSelected = [...selectedJournalPrefixes, prefix];
    else newSelected = selectedJournalPrefixes.filter(p => p !== prefix);
    setSelectedJournalPrefixes(newSelected);
  };

  const isViewingSelfCreated = currentUserId && viewingBatchCreatorId && 
                               String(currentUserId) === String(viewingBatchCreatorId);

  const columns = [
    { field: "batchId", headerName: "Batch ID", width: 125, align: "center", headerAlign: "center" },
    {
      field: "requestStatus", headerName: "Status", width: 125, align: "center", headerAlign: "center",
      renderCell: (params) => <Chip label={params.value} color="warning" size="small" variant="outlined" />
    },
    {
      field: "creatorId", headerName: "Creator", width: 150, align: "center", headerAlign: "center",
      renderCell: (params) => <Chip icon={<AccountCircleIcon />} label={params.value} size="small" />
    },
    {
      field: "requestDate", headerName: "Submitted On", width: 200, align: "center", headerAlign: "center",
      renderCell: (params) => new Date(params.value).toLocaleString("en-IN", { dateStyle: "medium", timeStyle: "short" })
    },
    { field: "commonBatchRemarks", headerName: "Creator Remarks", width: 200, align: "center", headerAlign: "center" },
    { field: "requestCount", headerName: "Entries", width: 100, align: "center", headerAlign: "center" },
    {
      field: "totalDebit", headerName: "Total Debit", width: 150, align: "right", headerAlign: "right",
      renderCell: (params) => <Typography variant="body2" sx={{ mt: 1.5, fontWeight: 500 }}>{formatCurrency(params.value)}</Typography>
    },
    {
      field: "totalCredit", headerName: "Total Credit", width: 150, align: "right", headerAlign: "right",
      renderCell: (params) => <Typography variant="body2" sx={{ mt: 1.5, fontWeight: 500 }}>{formatCurrency(params.value)}</Typography>
    },
    {
      field: "actions", headerName: "Actions", flex: 0.8, minWidth: 150, align: "center", headerAlign: "center", sortable: false,
      renderCell: (params) => {
        const isSelfCreated = currentUserId && params.row.creatorId && 
                              String(currentUserId) === String(params.row.creatorId);
        const tooltipMsg = "You cannot approve your own request";

        return (
          <Stack direction="row" spacing={1} justifyContent="center">
            <Tooltip title="View Details">
              <IconButton onClick={() => handleOpenViewModal(params.row.batchId, params.row.creatorId)} color="primary">
                <VisibilityIcon />
              </IconButton>
            </Tooltip>
            <Tooltip title={isSelfCreated ? tooltipMsg : "Accept Batch"}>
              <span>
                <IconButton onClick={() => handleOpenModal(params.row, "ACCEPTED")} color="success" disabled={isSelfCreated}>
                  <CheckIcon />
                </IconButton>
              </span>
            </Tooltip>
            <Tooltip title={isSelfCreated ? tooltipMsg : "Reject Batch"}>
              <span>
                <IconButton onClick={() => handleOpenModal(params.row, "REJECTED")} color="error" disabled={isSelfCreated}>
                  <ClearIcon />
                </IconButton>
              </span>
            </Tooltip>
          </Stack>
        );
      },
    },
  ];

  return (
    <Box sx={styles.mainContainer}>
      <DataGrid
        rows={rows}
        columns={columns}
        loading={loading}
        getRowId={(row) => row.batchId}
        disableRowSelectionOnClick
        pageSizeOptions={[5, 10, 25]}
        disableColumnResize={true}
        rowHeight={52}
        initialState={{
          pagination: { paginationModel: { page: 0, pageSize: 10 } },
          sorting: { sortModel: [{ field: "requestDate", sort: "desc" }] },
        }}
        slots={{ noRowsOverlay: CustomNoRowsOverlay }}
        sx={styles.dataGridContainer}
      />

      <Dialog open={isModalOpen} onClose={handleCloseModal} fullWidth maxWidth="sm">
        <DialogTitle>{modalMode === "ACCEPTED" ? "Approve" : "Reject"} Batch</DialogTitle>
        <DialogContent>
          <DialogContentText>
            Are you sure you want to {modalMode.toLowerCase()} batch <strong>{selectedBatch?.batchId}</strong>?
            {modalMode === "REJECTED" && " Remarks required."}
          </DialogContentText>
          <TextField
            autoFocus={modalMode === "REJECTED"}
            required={modalMode === "REJECTED"}
            margin="dense"
            label="Approver Remarks"
            fullWidth
            variant="standard"
            value={executorRemarks}
            onChange={(e) => setExecutorRemarks(e.target.value)}
            disabled={isSubmitting}
            InputProps={{ inputProps: { maxLength: 200 } }}
          />
        </DialogContent>
        <DialogActions sx={styles.dialogActions}>
          <Stack direction="row" spacing={1}>
            <Button onClick={handleCloseModal} disabled={isSubmitting}>Cancel</Button>
            <Button onClick={handleSubmitAction} variant="contained" color={modalMode === "ACCEPTED" ? "success" : "error"} disabled={isSubmitting}>
              {isSubmitting ? <CircularProgress size={24} color="inherit" /> : "Confirm"}
            </Button>
          </Stack>
        </DialogActions>
      </Dialog>

      <Dialog open={isViewModalOpen} onClose={handleCloseViewModal} fullWidth maxWidth="lg">
        <DialogTitle sx={styles.dialogTitle}>
          <Typography variant="h6">Batch Details: {viewingBatchId}</Typography>
          <IconButton onClick={handleCloseViewModal}><CloseIcon /></IconButton>
        </DialogTitle>
        <DialogContent dividers sx={{ p: 0 }}>
          {isViewLoading ? (
            <Box sx={styles.loadingContainer}><CircularProgress /></Box>
          ) : (
            batchDetails.length > 0 ? (
              <TableContainer component={Paper} sx={styles.tableContainer}>
                <Table sx={{ minWidth: 650 }} size="small" stickyHeader>
                  <TableHead>
                    <TableRow>
                      <TableCell padding="checkbox" sx={styles.stickyHeaderColumn(0)}>
                        <Checkbox
                          color="primary"
                          indeterminate={selectedJournalPrefixes.length > 0 && selectedJournalPrefixes.length < batchDetails.length}
                          checked={batchDetails.length > 0 && selectedJournalPrefixes.length === batchDetails.length}
                          onChange={handleSelectAllClick}
                          disabled={isViewingSelfCreated}
                        />
                      </TableCell>
                      <TableCell align="center" sx={styles.stickyHeaderColumn(48)}>Journal ID</TableCell>
                      <TableCell align="center" sx={styles.tableHeaderCell}>Branch</TableCell>
                      <TableCell align="center" sx={styles.tableHeaderCell}>Currency</TableCell>
                      <TableCell align="center" sx={styles.tableHeaderCell}>CGL</TableCell>
                      <TableCell align="center" sx={styles.tableHeaderCell}>Amount</TableCell>
                      <TableCell align="center" sx={styles.tableHeaderCell}>Txn Type</TableCell>
                    </TableRow>
                  </TableHead>
                  <TableBody>
                    {batchDetails.map((group, groupIndex) => {
                      const isSelected = selectedJournalPrefixes.includes(group[0].journalPrefix);
                      return group.map((row, rowIndex) => (
                        <TableRow
                          key={row.id}
                          hover={!isViewingSelfCreated}
                          onClick={() => !isViewingSelfCreated && handleRowCheckboxClick(null, row.journalPrefix)}
                          selected={isSelected}
                          sx={{
                            cursor: isViewingSelfCreated ? "not-allowed" : "pointer",
                            ...(rowIndex === 0 && groupIndex > 0 && { borderTop: 1, borderColor: "divider" }),
                          }}
                        >
                          {rowIndex === 0 && (
                            <>
                              <TableCell padding="checkbox" rowSpan={group.length} sx={styles.stickyBodyColumn(0)}>
                                <Checkbox
                                  color="primary"
                                  checked={isSelected}
                                  disabled={isViewingSelfCreated}
                                  onClick={(e) => !isViewingSelfCreated && handleRowCheckboxClick(e, row.journalPrefix)}
                                />
                              </TableCell>
                              <TableCell align="center" rowSpan={group.length} sx={styles.stickyBodyColumn(48)}>
                                {row.journalPrefix}
                              </TableCell>
                            </>
                          )}
                          <TableCell align="center">{row.payload.branch}</TableCell>
                          <TableCell align="center">{row.payload.currency}</TableCell>
                          <TableCell align="center">{row.payload.cgl}</TableCell>
                          <TableCell align="center">{formatCurrency(row.amount)}</TableCell>
                          <TableCell align="center">
                            <Chip label={row.transactionType} color={getTxnTypeColor(row.transactionType)} size="small" />
                          </TableCell>
                        </TableRow>
                      ));
                    })}
                  </TableBody>
                </Table>
              </TableContainer>
            ) : <Typography sx={{ p: 3 }}>No details found.</Typography>
          )}
        </DialogContent>
        <DialogActions sx={styles.dialogActions}>
          <Grid container spacing={2} alignItems="center">
            <Grid item xs={12} md={6}>
              <TextField
                margin="dense"
                label="Approver Remarks (Selected)"
                fullWidth
                size="small"
                value={partialExecutorRemarks}
                onChange={(e) => setPartialExecutorRemarks(e.target.value)}
                disabled={isPartialSubmitting || selectedJournalPrefixes.length === 0 || isViewingSelfCreated}
              />
            </Grid>
            <Grid item xs={12} md={6}>
              <Stack direction="row" spacing={1} justifyContent="flex-end">
                <Button variant="contained" color="success" onClick={() => handleSubmitPartialAction("ACCEPTED")} disabled={isPartialSubmitting || selectedJournalPrefixes.length === 0 || isViewingSelfCreated}>
                  Accept Selected
                </Button>
                <Button variant="contained" color="error" onClick={() => handleSubmitPartialAction("REJECTED")} disabled={isPartialSubmitting || selectedJournalPrefixes.length === 0 || isViewingSelfCreated}>
                  Reject Selected
                </Button>
              </Stack>
            </Grid>
            {isViewingSelfCreated && (
              <Grid item xs={12}>
                <Typography color="error" variant="body2" textAlign="center" sx={{ pt: 1 }}>
                  You cannot approve or reject entries from your own request.
                </Typography>
              </Grid>
            )}
          </Grid>
        </DialogActions>
      </Dialog>
    </Box>
  );
}
