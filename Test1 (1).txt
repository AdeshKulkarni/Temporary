import React, { useState, useEffect, useCallback } from "react";
import {
  Box,
  Typography,
  Chip,
  Dialog,
  DialogActions,
  DialogContent,
  DialogContentText,
  DialogTitle,
  Button,
  CircularProgress,
  Tooltip,
  IconButton,
  Stack,
} from "@mui/material";
import { styled } from "@mui/material/styles";
import { DataGrid } from "@mui/x-data-grid";
import {
  Ballot,
  Cancel as CancelIcon,
  HelpOutline as HelpIcon,
} from "@mui/icons-material";
import AccountCircleIcon from "@mui/icons-material/AccountCircle";
import InfoIcon from "@mui/icons-material/Info";

import useApi from "../../../hooks/useApi";
import useCustomSnackbar from "../../../utils/useCustomSnackbar";

// --- STYLED COMPONENTS (Unchanged) ---

const StyledRoot = styled(Box)(({ theme }) => ({
  padding: theme.spacing(3),
  width: "100%",
}));

const PageTitle = styled(Typography)(({ theme }) => ({
  fontWeight: "bold",
  color: theme.palette.primary.main,
  marginBottom: theme.spacing(3),
}));

const StyledDataGrid = styled(DataGrid)({
  height: 600,
  width: "100%",
  "& .MuiDataGrid-cell:focus": {
    outline: "none",
  },
});

const NoRowsContainer = styled(Stack)({
  height: "100%",
  alignItems: "center",
  justifyContent: "center",
});

const NoRowsIcon = styled(InfoIcon)(({ theme }) => ({
  fontSize: 48,
  color: theme.palette.text.secondary,
  marginBottom: theme.spacing(2),
}));

const ModalTitle = styled(DialogTitle)({
  display: "flex",
  alignItems: "center",
});

const WarningIcon = styled(Ballot)(({ theme }) => ({
  marginRight: theme.spacing(1),
  color: theme.palette.warning.main,
}));

const ModalContent = styled(DialogContent)({
  paddingTop: theme.spacing(3),
});

const ModalActions = styled(DialogActions)(({ theme }) => ({
  padding: theme.spacing(2),
}));

// --- END OF STYLED COMPONENTS ---

const getStatusChipColor = (status) => {
  switch (status) {
    case "PENDING":
      return "warning";
    case "ACCEPTED":
      return "success";
    case "REJECTED":
      return "error";
    default:
      return "default";
  }
};

function CustomNoRowsOverlay() {
  return (
    <NoRowsContainer>
      <NoRowsIcon />
      <Typography variant="h6">No Requests Found</Typography>
      <Typography>You have not created any MOC requests.</Typography>
    </NoRowsContainer>
  );
}

export default function JournalPostingStatus() {
  const { callApi } = useApi();
  const showSnackBar = useCustomSnackbar();

  const [loading, setLoading] = useState(true);
  const [rows, setRows] = useState([]);

  const [isModalOpen, setIsModalOpen] = useState(false);
  const [isSubmitting, setIsSubmitting] = useState(false);
  const [selectedRequest, setSelectedRequest] = useState(null);

  const fetchMyRequests = useCallback(async () => {
    setLoading(true);
    try {
      const myRequests = await callApi("MS/moc/my-requests", null, "GET");
      if (Array.isArray(myRequests)) {
        setRows(myRequests);
      } else {
        setRows([]);
      }
    } catch (err) {
      console.error("Failed to fetch user's requests:", err);
      const errorMsg =
        err.response?.data?.message || "Failed to load your requests.";
      showSnackBar(errorMsg, "error");
    } finally {
      setLoading(false);
    }
  }, [callApi, showSnackBar]);

  useEffect(() => {
    fetchMyRequests();
  }, [fetchMyRequests]);

  const handleOpenModal = (request) => {
    setSelectedRequest(request);
    setIsModalOpen(true);
  };

  const handleCloseModal = () => {
    if (isSubmitting) return;
    setIsModalOpen(false);
    setSelectedRequest(null);
  };

  const handleConfirmCancel = async () => {
    if (!selectedRequest) return;
    setIsSubmitting(true);
    try {
      await callApi(
        `MS/moc/my-request/${selectedRequest.id}`,
        null,
        "DELETE"
      );
      showSnackBar(
        `Request (Journal ID: ${selectedRequest.journalId}) was successfully canceled.`,
        "success"
      );
      handleCloseModal();
      fetchMyRequests(); 
    } catch (error) {
      console.error("Failed to cancel request:", error);
      const errorMsg =
        error.response?.data?.message || "Failed to cancel the request.";
      showSnackBar(errorMsg, "error");
    } finally {
      setIsSubmitting(false);
    }
  };

  // --- COLUMNS ARRAY MODIFIED ---
  const columns = [
    {
      field: "requestStatus",
      headerName: "Req Status", // Header text updated
      flex: 1,
      minWidth: 100,
      renderCell: (params) => (
        <Chip
          label={params.value}
          color={getStatusChipColor(params.value)}
          size="small"
        />
      ),
    },
    {
      field: "batchId",
      headerName: "Batch ID",
      flex: 1.5,
      minWidth: 150,
    },
    {
      field: "journalId",
      headerName: "Journal ID",
      flex: 1.5,
      minWidth: 150,
    },
    {
      field: "requestDate",
      headerName: "Req Date", // Header text updated
      flex: 1.5,
      minWidth: 180,
      renderCell: (params) => (
        <Box>
          {new Date(params.value).toLocaleString("en-IN", {
            dateStyle: "medium",
            timeStyle: "short",
          })}
        </Box>
      ),
    },
    // --- NEW COLUMN ADDED ---
    {
      field: "creatorId",
      headerName: "Creator ID",
      flex: 1,
      minWidth: 120,
      renderCell: (params) => (
        <Chip
          icon={<AccountCircleIcon />}
          label={params.value}
          size="small"
          variant="outlined"
        />
      ),
    },
    // --- END OF NEW COLUMN ---
    {
      field: "executorId",
      headerName: "Executor ID", // Header text updated
      flex: 1,
      minWidth: 120,
      renderCell: (params) =>
        params.value ? (
          <Chip
            icon={<AccountCircleIcon />}
            label={params.value}
            size="small"
          />
        ) : (
          <Chip label="N/A" size="small" variant="outlined" />
        ),
    },
    {
      field: "executorRemarks",
      headerName: "Executor Remarks", // Header text updated
      flex: 2,
      minWidth: 200,
    },
    {
      field: "actions",
      headerName: "Actions",
      flex: 0.8,
      minWidth: 100,
      align: "center",
      headerAlign: "center",
      sortable: false,
      renderCell: (params) => (
        <Tooltip title="Cancel Request">
          <Box>
            <IconButton
              onClick={() => handleOpenModal(params.row)}
              color="error"
              disabled={params.row.requestStatus !== "PENDING"}
            >
              <CancelIcon />
            </IconButton>
          </Box>
        </Tooltip>
      ),
    },
  ];
  // --- END OF COLUMN MODIFICATION ---

  return (
    <StyledRoot>
      <PageTitle variant="h4" gutterBottom>
        My MOC Requests
      </PageTitle>

      <StyledDataGrid
        rows={rows}
        columns={columns}
        loading={loading}
        getRowId={(row) => row.id} 
        disableRowSelectionOnClick
        pageSizeOptions={[10, 25, 50]}
        initialState={{
          pagination: {
            paginationModel: { page: 0, pageSize: 10 },
          },
          sorting: { sortModel: [{ field: "requestDate", sort: "desc" }] },
        }}
        slots={{
          noRowsOverlay: CustomNoRowsOverlay,
        }}
      />

      {/* --- Cancel Confirmation Modal (Unchanged) --- */}
      <Dialog
        open={isModalOpen}
        onClose={handleCloseModal}
        fullWidth
        maxWidth="sm"
      >
        <ModalTitle>
          <WarningIcon />
          Confirm Cancellation
        </ModalTitle>
        <ModalContent>
          <DialogContentText>
            Are you sure you want to cancel and delete this request?
            <br />
            <br />
            <strong>Batch ID:</strong> {selectedRequest?.batchId}
            <br />
            <strong>Journal ID:</strong> {selectedRequest?.journalId}
            <br />
            <br />
            This action cannot be undone.
          </DialogContentText>
        </ModalContent>
        <ModalActions>
          <Button onClick={handleCloseModal} disabled={isSubmitting}>
            Back
          </Button>
          <Button
            onClick={handleConfirmCancel}
            variant="contained"
            color="error"
            disabled={isSubmitting}
          >
            {isSubmitting ? (
              <CircularProgress size={24} color="inherit" />
            ) : (
              "Yes, Cancel Request"
            )}
          </Button>
        </ModalActions>
      </Dialog>
    </StyledRoot>
  );
}
