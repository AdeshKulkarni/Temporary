SELECT
    -- Grouping columns for context
    SC14_GRP1,
    SC14_GRP2,
    SC14_GRP3,
    SC14_GRP4,
    SC14_GRP5,
    SC14_TBCODE,

    -- [DEBUG] Current Year (cy) calculation breakdown
    sum(nvl(POSTMOC,0)) AS sum_postmoc,
    sum(nvl(a.wsa_adj,0)) AS sum_wsa_adj_cy,
    sum(nvl(POSTMOC,0) + nvl(a.wsa_adj,0)) AS total_cy_base,
    round(sum(nvl(POSTMOC,0) + nvl(a.wsa_adj,0)) * (-1) / 1000) AS transformed_cy_before_case,

    -- [DEBUG] Previous Year (py) calculation breakdown
    sum(nvl(b.wsa_tot,0)) AS sum_wsa_tot_py,
    round(nvl((sum(b.wsa_tot) * -1) / 1000, 0)) AS transformed_py_before_case,
    
    -- Original final calculations for comparison
    CASE 
        WHEN SC14_TBCODE IN ('Y020', 'Y040', 'Y060', 'Y080')
        THEN CASE WHEN round(sum(nvl(POSTMOC,0) + nvl(a.wsa_adj,0))*(-1)/1000) > 0 
                  THEN round(sum(nvl(POSTMOC,0) + nvl(a.wsa_adj,0))*(-1)/1000) ELSE 0 END 
        WHEN SC14_TBCODE IN ('Y030', 'Y050', 'Y070', 'Y090')
        THEN CASE WHEN round(sum(nvl(POSTMOC,0) + nvl(a.wsa_adj,0))*(-1)/1000) < 0 
                  THEN round(sum(nvl(POSTMOC,0) + nvl(a.wsa_adj,0))*(-1)/1000) ELSE 0 END 
        ELSE round((sum(nvl(POSTMOC,0) + nvl(a.wsa_adj,0))*(-1)/1000))
    END AS final_cy,
    CASE 
        WHEN SC14_TBCODE IN ('Y020', 'Y040', 'Y060', 'Y080')
        THEN CASE WHEN round(nvl((sum(b.wsa_tot)*-1)/1000,0)) > 0 
                  THEN round(nvl((sum(b.wsa_tot)*-1)/1000,0)) ELSE 0 END 
        WHEN SC14_TBCODE IN ('Y030', 'Y050', 'Y070', 'Y090')
        THEN CASE WHEN round(nvl((sum(b.wsa_tot)*-1)/1000,0)) < 0 
                  THEN round(nvl((sum(b.wsa_tot)*-1)/1000,0)) ELSE 0 END 
        ELSE round(nvl((sum(b.wsa_tot)*-1)/1000,0))
    END AS final_py

FROM 
    (SELECT SC14_sch, SC14_GRP1, SC14_GRP2, SC14_GRP3, SC14_GRP4, SC14_GRP5, SC14_TYPE, SC14_TBCODE,
            sum(PRE + MOC) postmoc 
     FROM fr_sc14_map, bs_postmoc
     WHERE comp_code = 'PNL' || fr_sc14_map.sc14_comp
     GROUP BY SC14_sch, SC14_GRP1, SC14_GRP2, SC14_GRP3, SC14_GRP4, SC14_GRP5, SC14_TBCODE, SC14_TYPE)
LEFT OUTER JOIN bs_wsa A ON (A.WSA_COMP = SC14_sch)
LEFT OUTER JOIN ${PREYR}.bs_wsa B ON SC14_sch = B.WSA_COMP
GROUP BY 
    SC14_GRP1, SC14_GRP2, SC14_GRP3, SC14_GRP4, SC14_GRP5, SC14_TBCODE
ORDER BY 
    SC14_TBCODE;
