import React, { useState, useMemo, useEffect } from "react";
import axios from "axios";
import {
  StyledBox,
  StyledDataGrid,
  StyledFab,
  StyledDialog,
  StyledDialogTitle,
  StyledDialogContent,
  StyledDialogActions,
  StyledTextField,
  StyledButton,
} from "./CurrencyMasterStyles";
import { GridActionsCellItem } from "@mui/x-data-grid";
import AddIcon from "@mui/icons-material/Add";
import EditIcon from "@mui/icons-material/Edit";
import useApi from "../../hooks/useApi";
import useCustomSnackbar from "../../utils/useCustomSnackbar";
import { timeStampFormatter } from "../../utils/CommonUtilities"; // Imported the formatter

const initialFormState = {
  currencyCode: "",
  currencyName: "",
  currencyRate: "",
};

// Use the existing StyledBox for the main container and remove internal sx props.
// Assuming StyledBox already provides necessary styling for layout.

export default function CurrencyMaster() {
  const { callApi } = useApi();
  const showSnackBar = useCustomSnackbar();
  const [rows, setRows] = useState([]);
  const [isLoading, setIsLoading] = useState(true);

  // State to hold Dynamically loaded JSON data
  const [content, setContent] = useState(null);

  const [isDialogOpen, setIsDialogOpen] = useState(false);
  const [formData, setFormData] = useState(initialFormState);
  const [editingRow, setEditingRow] = useState(null);
  const [errors, setErrors] = useState({});

  useEffect(() => {
    const loadContent = async () => {
      try {
        const res = await axios.get("./currencyMasterData.json");
        setContent(res.data);
      } catch (error) {
        console.log("Failed to load content JSON.", error);
        showSnackBar("Failed to load interface content.", "error");
      }
    };
    loadContent();
    // No need to call fetchData twice here.
  }, []); // Empty dependency array means this runs only once on mount

  useEffect(() => {
    fetchData();
  }, []); // Run on mount

  const fetchData = async () => {
    setIsLoading(true);
    try {
      const response = await callApi("CM/currency-master", null, "GET");

      if (response && response.data && Array.isArray(response.data)) {
        const formattedData = response.data.map((row) => {
          let formattedDate = "N/A";

          // --- CHANGE 1: Use timeStampFormatter for updatedAt ---
          if (row.updatedAt) {
            formattedDate = timeStampFormatter(row.updatedAt);
          }
          
          return {
            ...row,
            id: row.currencyCode,
            updatedAt: formattedDate,
            // Format currencyRate for display in the DataGrid later
            currencyRateDisplay: Number(row.currencyRate).toFixed(4),
          };
        });

        setRows(formattedData);
        if (formattedData.length === 0) {
          showSnackBar("No currency data found.", "info");
        }
      } else {
        showSnackBar("Received invalid data format.", "error");
        setRows([]);
      }
    } catch (error) {
      console.error("Error fetching currency data:", error);
      showSnackBar(
        "Something went wrong while fetching data. Please try again later.",
        "error"
      );
      setRows([]);
    } finally {
      setIsLoading(false);
    }
  };

  const handleOpenDialog = () => {
    setEditingRow(null);
    setFormData(initialFormState);
    setErrors({});
    setIsDialogOpen(true);
  };

  const handleCloseDialog = () => {
    setIsDialogOpen(false);
    setEditingRow(null);
    setFormData(initialFormState);
    setErrors({});
  };

  const handleEditClick = (id) => () => {
    const rowToEdit = rows.find((row) => row.id === id);
    if (rowToEdit) {
      setEditingRow(rowToEdit);
      setFormData({
        currencyCode: rowToEdit.currencyCode,
        // When editing, we only allow changing the name, so the rate and code are disabled.
        currencyName: rowToEdit.currencyName, 
        currencyRate: rowToEdit.currencyRate, 
      });
      setErrors({});
      // --- CHANGE 2: Ensure dialog fields are visible/not truncated (addressed in render logic by disabling code/rate) ---
      setIsDialogOpen(true);
    }
  };

  const validate = () => {
    let tempErrors = {};

    if (!formData.currencyName?.trim()) {
      tempErrors.currencyName = "Currency Name is required.";
    } else if (/\d/.test(formData.currencyName)) {
      tempErrors.currencyName = "Numbers are not allowed in Currency Name.";
    }

    // Only validate code and rate on ADD, not on UPDATE (as they are disabled/not changed)
    if (!editingRow) {
      if (!formData.currencyCode?.trim()) {
        tempErrors.currencyCode = "Currency Code is required.";
      } else if (formData.currencyCode.length !== 3) {
        tempErrors.currencyCode = "Currency Code must be exactly 3 characters.";
      }

      if (!String(formData.currencyRate).trim()) {
        tempErrors.currencyRate = "Currency Rate is required.";
      }
    }

    setErrors(tempErrors);
    return Object.keys(tempErrors).length === 0;
  };

  const handleInputChange = (e) => {
    const { name, value } = e.target;
    let processedValue = value;

    if (name === content?.formFields?.currencyCode?.name) {
      processedValue = value
        .toUpperCase()
        .slice(0, 3)
        .replace(/[^A-Z]/g, "");
    }
    if (name === content?.formFields?.currencyName?.name) {
      processedValue = value.slice(0, 30).replace(/[^a-zA-Z\s]/g, "");
    }
    if (name === content?.formFields?.currencyRate?.name) {
      // Allow up to 6 digits before and up to 6 after the decimal point
      if (value !== "" && !/^\d{0,6}(\.\d{0,6})?$/.test(value)) {
        return;
      }
      processedValue = value;
    }

    setFormData({ ...formData, [name]: processedValue });
    if (errors[name]) {
      setErrors({ ...errors, [name]: "" });
    }
  };

  const handleSubmit = async () => {
    if (!validate()) {
      return;
    }

    const isEditing = !!editingRow;
    let apiPayload;

    if (isEditing) {
      apiPayload = {
        requestType: "CURRENCY",
        changeType: "UPDATE",
        payload: {
          currencyCode: editingRow.currencyCode,
          // Only currencyName is truly editable
          currencyName: formData.currencyName, 
          // currencyRate is disabled in the form, so use the original value
          currencyRate: editingRow.currencyRate, 
          flag: 1,
        },
      };
    } else {
      apiPayload = {
        requestType: "CURRENCY",
        changeType: "ADD",
        payload: {
          currencyCode: formData.currencyCode,
          currencyName: formData.currencyName,
          currencyRate: parseFloat(formData.currencyRate) || 0,
          flag: 0,
        },
      };
    }

    try {
      await callApi("CR/create-request", apiPayload, "POST");

      showSnackBar(
        `Request to ${
          isEditing ? "update" : "add"
        } currency submitted successfully!`,
        "success"
      );
      handleCloseDialog();
      fetchData();
    } catch (error) {
      console.error("Error submitting request:", error);
      showSnackBar(
        error.response?.data?.message ||
          `Failed to submit ${isEditing ? "update" : "add"} request.`,
        "error"
      );
    }
  };

  const columns = useMemo(
    () => [
      {
        field: "currencyCode",
        headerName: content?.dataGridHeaders?.currencyCode || "Currency Code",
        flex: 1,
        sortable: true,
        align: "center",
        headerAlign: "center",
      },
      {
        field: "currencyName",
        headerName: content?.dataGridHeaders?.currencyName || "Currency Name",
        flex: 2,
        sortable: true,
        align: "center",
        headerAlign: "center",
      },
      {
        field: "currencyRate",
        headerName: content?.dataGridHeaders?.currencyRate || "Currency Rate",
        type: "number",
        flex: 1,
        sortable: true,
        align: "center",
        headerAlign: "center",
        // --- CHANGE 4: Format currencyRate to 4 decimals without commas ---
        valueGetter: (params) => params.row.currencyRateDisplay,
      },
      {
        field: "updatedAt",
        headerName: content?.dataGridHeaders?.updatedAt || "Last Updated",
        flex: 1,
        sortable: true,
        align: "center",
        headerAlign: "center",
      },
      {
        field: "actions",
        type: "actions",
        headerName: content?.dataGridHeaders?.actions || "Actions",
        width: 80,
        headerAlign: "center",
        align: "center",
        cellClassName: "actions",
        getActions: ({ id }) => [
          <GridActionsCellItem
            icon={<EditIcon />}
            label="Edit"
            onClick={handleEditClick(id)}
            color="primary"
          />,
        ],
      },
    ],
    [content] // Depend on content instead of rows, as rows changes often
  );

  return (
    // StyledBox replaces the use of 'sx' for the main container
    // --- CHANGE 5: Remove 'sx' usage and rely on Styled components/external styling ---
    <StyledBox> 
      <StyledDataGrid
        rows={rows}
        columns={columns}
        disableRowSelectionOnClick
        pageSizeOptions={[5, 10, 25, 50]}
        initialState={{
          pagination: { paginationModel: { pageSize: 10, page: 0 } },
          sorting: { sortModel: [{ field: "updatedAt", sort: "desc" }] },
        }}
        loading={isLoading}
        autoHeight
      />

      <StyledFab aria-label="add" onClick={handleOpenDialog}>
        <AddIcon />
      </StyledFab>

      <StyledDialog open={isDialogOpen} onClose={handleCloseDialog} fullWidth>
        <StyledDialogTitle>
          {editingRow ? content?.dialog?.editTitle : content?.dialog?.addTitle}
        </StyledDialogTitle>
        <StyledDialogContent>
          <StyledTextField
            autoFocus
            name={content?.formFields?.currencyCode?.name}
            label={content?.formFields?.currencyCode?.label}
            type="text"
            fullWidth
            variant="outlined"
            value={formData.currencyCode}
            onChange={handleInputChange}
            error={!!errors.currencyCode}
            helperText={
              errors.currencyCode ||
              content?.formFields?.currencyCode?.helperText
            }
            // Disabled in both ADD and EDIT to prevent changes
            disabled={true} // Making it permanently disabled (assuming code is immutable after creation)
            inputProps={{ maxLength: 3 }}
          />
          <StyledTextField
            name={content?.formFields?.currencyName?.name}
            label={content?.formFields?.currencyName?.label}
            type="text"
            fullWidth
            variant="outlined"
            value={formData.currencyName}
            onChange={handleInputChange}
            error={!!errors.currencyName}
            helperText={
              errors.currencyName ||
              content?.formFields?.currencyName?.helperText
            }
            inputProps={{ maxLength: 30 }}
          />
          <StyledTextField
            name={content?.formFields?.currencyRate?.name}
            label={content?.formFields?.currencyRate?.label}
            type="text"
            fullWidth
            variant="outlined"
            value={formData.currencyRate}
            onChange={handleInputChange}
            error={!!errors.currencyRate}
            helperText={
              errors.currencyRate ||
              content?.formFields?.currencyRate?.helperText
            }
            // Disabled in both ADD and EDIT to prevent changes
            disabled={true} // Making it permanently disabled (assuming rate is managed elsewhere/immutable)
          />
        </StyledDialogContent>
        <StyledDialogActions>
          <StyledButton onClick={handleCloseDialog}>
            {content?.buttons?.cancel}
          </StyledButton>
          <StyledButton onClick={handleSubmit} variant="contained">
            {/* --- CHANGE 3: Dynamic button text --- */}
            {editingRow ? content?.buttons?.update : content?.buttons?.submit}
          </StyledButton>
        </StyledDialogActions>
      </StyledDialog>
    </StyledBox>
  );
}
