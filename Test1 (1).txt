class Solution {
    public long Maxi(int x , int y , int[] nums) {

        int maxi = Integer.MIN_VALUE;
        for(int i=x; i<y; i++) {
            if(nums[i] > maxi) {
                maxi = nums[i];
            }
        }

        return maxi;
    }

    public long mini(int x , int y , int[] nums) {

        int mini = Integer.MAX_VALUE;
        for(int i=x; i<y; i++) {
            if(nums[i] < mini) {
                mini = nums[i];
            }
        }

        return mini;
    }

    public long countSubarrays(int[] nums, int minK, int maxK) {

        int count = 0;

        int n = nums.length;

        int low = 0;      

        for(int high = 0; high < n; high++) {
                   
             D.p(low , high);
             long minimum = mini(low , high , nums);
             long maximum = Maxi(low , high , nums);

             D.p(minimum , maximum);
             D.p("--------");

             if(minimum == minK && maximum == maxK) {
                count++;
                low++;
             }

          }

          return count;
        
    }
}
