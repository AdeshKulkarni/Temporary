# =====================================================================
# 3. CONFIGURATION
# =====================================================================
CURRENCY = "INR"
PRODUCT_CODE = "12345678"
REMARKS = "Bulk Script Entry"

# 50,000 pairs = 100,000 Total Rows (1 Lakh)
TOTAL_PAIRS_TO_GENERATE = 50000 
OUTPUT_FILENAME = "Voucher_Bulk_Upload_1Lakh.xlsx"

# =====================================================================
# 4. DATA GENERATION LOGIC
# =====================================================================
def generate_data():
    print("Parsing raw input lists...")
    # Clean the raw text blocks into lists (ignores blank lines automatically)
    branch_list = [b.strip() for b in RAW_BRANCH_LIST.splitlines() if b.strip()]
    opposite_cgl_list = [c.strip() for c in RAW_OPPOSITE_CGL_LIST.splitlines() if c.strip()]
    
    if not branch_list:
        print("ERROR: Please paste at least one Branch into the RAW_BRANCH_LIST.")
        return
    if not opposite_cgl_list:
        print("ERROR: Please paste at least one CGL into the RAW_OPPOSITE_CGL_LIST.")
        return

    print(f"Generating {TOTAL_PAIRS_TO_GENERATE * 2} rows of balanced entries...")
    rows = []

    for i in range(TOTAL_PAIRS_TO_GENERATE):
        # 1. Randomly pick required fields
        branch = random.choice(branch_list)
        amount = round(random.uniform(100.0, 99999.0), 2)  # Random amount between 100 and 99,999
        fixed_item = random.choice(FIXED_CGLS)
        opposite_cgl = random.choice(opposite_cgl_list)
        
        # 2. Determine Dr/Cr based on I/E rules
        if fixed_item["type"] == "I":
            fixed_leg_type = "Credit"
            opposite_leg_type = "Debit"
        else:
            fixed_leg_type = "Debit"
            opposite_leg_type = "Credit"
            
        # 3. Create Row 1 (Using the Fixed CGL)
        row1 = {
            "Branch*": branch,
            "Currency*": CURRENCY,
            "CGL*": fixed_item["cgl"],
            "Amount*": amount,
            "Type*": fixed_leg_type,
            "Remarks*": f"{REMARKS} {i+1}",
            "Product Code": PRODUCT_CODE
        }
        
        # 4. Create Row 2 (Using the Opposite pasted CGL to balance the entry)
        row2 = {
            "Branch*": branch,
            "Currency*": CURRENCY,
            "CGL*": opposite_cgl,
            "Amount*": amount,
            "Type*": opposite_leg_type,
            "Remarks*": f"{REMARKS} {i+1}",
            "Product Code": PRODUCT_CODE
        }
        
        rows.append(row1)
        rows.append(row2)
        
        # Print progress every 10,000 pairs
        if (i + 1) % 10000 == 0:
            print(f"Processed {i + 1} pairs ({(i + 1) * 2} rows)...")

    # =====================================================================
    # 5. WRITE TO EXCEL
    # =====================================================================
    print("Data generated! Writing to Excel (this might take a few moments for 1 Lakh rows)...")
    df = pd.DataFrame(rows)
    
    # Save using pandas and openpyxl
    df.to_excel(OUTPUT_FILENAME, index=False)
    print(f"\nSUCCESS! Created {OUTPUT_FILENAME} with {len(df)} rows.")

if __name__ == "__main__":
    generate_data()
