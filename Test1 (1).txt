-- Oracle SQL Query to create the REQUEST_MASTER table

CREATE TABLE REQUEST_MASTER (
    REQ_ID              NUMBER         NOT NULL,
    REQ_STATUS          VARCHAR2(10 BYTE) DEFAULT 'P' NOT NULL,
    CHANGE_TYPE         VARCHAR2(10 BYTE) NOT NULL,
    REQ_DATE            DATE             DEFAULT SYSDATE NOT NULL,
    CREATOR_ID          VARCHAR2(12 BYTE) NOT NULL,
    EXECUTOR_ID         VARCHAR2(12 BYTE),
    EXECUTION_DATE      DATE,
    EXECUTOR_REMARKS    VARCHAR2(500 BYTE),
    PAYLOAD             CLOB,
    BATCH_ID            VARCHAR2(50 BYTE),
    JOURNAL_ID          VARCHAR2(50 BYTE),
    COMMON_BATCH_REMARKS VARCHAR2(200 BYTE),
    CREATOR_ROLE        NUMBER(10,0)
);
