import React, { useEffect, useState } from "react";
import { Divider, Grid, Button, Autocomplete, TextField, Paper, Box, Typography, Dialog, DialogContent, CircularProgress, Snackbar, Alert } from "@mui/material";
import { Download } from "@mui/icons-material";
import { SnackbarProvider } from "notistack";
import axios from "axios";
import { encrypt } from "../../Security/AES-GCM256";
import downloadFile from "../../DownloadFile/DownloadFile";

function FrtUserEvidenceDownload() {
  const [quarter, setQuarter] = useState(null);
  const [quarterList, setQuarterList] = useState([]); // State to hold the dynamic quarter list
  const [loadOpen, setLoadOpen] = useState(false);
  const [snackbar, setSnackbar] = useState(null);
  const [circleCode, setCircleCode] = useState(null);
  const [circleOptions, setCircleOptions] = useState([]);

  const iv = crypto.getRandomValues(new Uint8Array(12));
  const ivBase64 = btoa(String.fromCharCode.apply(null, iv));
  const salt = crypto.getRandomValues(new Uint8Array(16));
  const saltBase64 = btoa(String.fromCharCode.apply(null, salt));

  const handleCloseSnackbar = () => setSnackbar(null);

  // --- NEW DYNAMIC DATE LOGIC ---
  const generateQuarterEndDates = (previousYearDate) => {
    // Expects date in "DD/MM/YYYY" format
    if (!previousYearDate) {
      return []; // Return empty if no date is provided
    }
    const parts = previousYearDate.split('/');
    const month = parseInt(parts[1], 10);
    const year = parseInt(parts[2], 10);
    let targetYear;

    // If last quarter was June (6), Sept (9), or Dec (12), the financial year ends in the next calendar year.
    if ([6, 9, 12].includes(month)) {
      targetYear = year + 1;
    } else {
      // Otherwise (e.g., March), it ends in the same calendar year.
      targetYear = year;
    }
    const newDate = `31-03-${targetYear}`;
    return [{ DATE: newDate }];
  };

  // useEffect to generate the quarter list when the component first loads
  useEffect(() => {
    // In a real app, you would get this date from your token or user session.
    // For this example, we'll simulate it.
    // const previousDateFromToken = localStorage.getItem("previousYearDate");
    const previousDateFromToken = "30/09/2025"; // EXAMPLE: Since today is Sep 19, 2025

    const generatedQuarters = generateQuarterEndDates(previousDateFromToken);
    setQuarterList(generatedQuarters);

    // Automatically select the first (and only) date in the list
    if (generatedQuarters.length > 0) {
      setQuarter(generatedQuarters[0]);
    }
  }, []); // Empty array ensures this runs only once on mount

  const loadingScreenOpen = () => setLoadOpen(true);
  const loadingScreenClose = () => setLoadOpen(false);

  const getCicleList = async () => {
    if (!quarter) return;
    try {
      loadingScreenOpen();
      const response = await axios.post(
        "Server/frtUserService/getDashCircleList",
        {},
        {
          headers: { Authorization: `Bearer ${localStorage.getItem("token")}` },
        }
      );
      if (response.data.result && response.data.result.length > 0) {
        setCircleOptions(response.data.result);
      } else {
        setSnackbar({
          children: "Failed to fetch list of circles.",
          severity: "error",
        });
      }
    } catch (e) {
      console.error(e);
      setSnackbar({
        children: "An error occurred while fetching circles.",
        severity: "error",
      });
    } finally {
      loadingScreenClose();
    }
  };

  useEffect(() => {
    setCircleCode(null);
    setCircleOptions([]);
    getCicleList();
  }, [quarter]);

  const handleDownload = async (e) => {
    e.preventDefault();
    if (!quarter || !circleCode) {
      setSnackbar({
        children: "Please select a Quarter and a Circle.",
        severity: "error",
      });
      return;
    }

    try {
      loadingScreenOpen();
      const dataToSend = {
        quarterDate: quarter.DATE,
        circleCode: [circleCode.CRS_SYS_PARAM_ID],
      };
      let jsonFormData = JSON.stringify(dataToSend);
      const encryptedData = await encrypt(iv, salt, jsonFormData);
      const payload = {
        iv: ivBase64,
        salt: saltBase64,
        data: encryptedData,
      };
      const response = await axios.post(
        "Server/frtUserService/download",
        payload,
        {
          headers: { Authorization: `Bearer ${localStorage.getItem("token")}` },
        }
      );
      if (response.data.result) {
        downloadFile(
          response.data.result.zipContent,
          "zip",
          response.data.result.zipName
        );
        setSnackbar({
          children: "Download successful!",
          severity: "success",
        });
      } else {
        setSnackbar({
          children: "Failed to prepare download.",
          severity: "warning",
        });
      }
    } catch (e) {
      console.error(e);
      setSnackbar({
        children: "An error occurred during download.",
        severity: "error",
      });
    } finally {
      loadingScreenClose();
    }
  };

  return (
    <>
      <Box sx={{ p: 2 }}>
        <Typography variant={"h5"} gutterBottom>
          Evidence Download
        </Typography>
        <Divider />

        <Paper sx={{ mt: 3, p: 4 }}>
          <Grid container spacing={4} justifyContent="center">
            <Grid item xs={12} md={5}>
              <Autocomplete
                options={quarterList}
                getOptionLabel={(option) => option.DATE || ""}
                value={quarter}
                onChange={(event, newValue) => {
                  setQuarter(newValue);
                }}
                renderInput={(params) => (
                  <TextField {...params} label="Select Quarter" />
                )}
                readOnly // Since the date is now auto-set, make it read-only
              />
            </Grid>
            <Grid item xs={12} md={5}>
              <Autocomplete
                options={circleOptions}
                getOptionLabel={(option) => option.CRS_SYS_PARAM_NAME || ""}
                value={circleCode}
                onChange={(event, newValue) => {
                  setCircleCode(newValue);
                }}
                disabled={!quarter}
                renderInput={(params) => (
                  <TextField {...params} label="Select Circle" />
                )}
              />
            </Grid>
          </Grid>

          <Box sx={{ display: "flex", justifyContent: "center", mt: 5 }}>
            <Button
              variant="contained"
              color="warning"
              startIcon={<Download />}
              onClick={handleDownload}
              size="large"
              sx={{ px: 4, py: 1.5 }}
            >
              Download Evidence
            </Button>
          </Box>
        </Paper>
      </Box>

      {!!snackbar && (
        <SnackbarProvider maxSnack={3}>
          <Snackbar
            open
            anchorOrigin={{ vertical: "bottom", horizontal: "right" }}
            onClose={handleCloseSnackbar}
            autoHideDuration={5000}
          >
            <Alert {...snackbar} onClose={handleCloseSnackbar} variant="filled" />
          </Snackbar>
        </SnackbarProvider>
      )}

      <Dialog
        open={loadOpen}
        PaperProps={{ style: { backgroundColor: "transparent", boxShadow: "none" } }}
      >
        <DialogContent>
          <CircularProgress />
        </DialogContent>
      </Dialog>
    </>
  );
}

export default FrtUserEvidenceDownload;
