import React, { useState, useEffect, useMemo } from "react";
import {
  Divider,
  Typography,
  Box,
  Stack,
  Button,
  Paper,
  MenuItem,
  TextField,
  Tabs,
  Tab,
} from "@mui/material";
import { DataGrid, GridToolbar, GridOverlay } from "@mui/x-data-grid";
import { encrypt } from "../../Security/AES-GCM256";
import axios from "axios";
import downloadFile from "../../DownloadFile/DownloadFile";

// Generate IV and salt for encryption
const iv = crypto.getRandomValues(new Uint8Array(12));
const ivBase64 = btoa(String.fromCharCode.apply(null, iv));
const salt = crypto.getRandomValues(new Uint8Array(16));
const saltBase64 = btoa(String.fromCharCode.apply(null, salt));

// DataGrid columns definition (UPDATED)
const columns = [
  { field: "FRN", headerName: "FRN", width: 170, flex: 1 },
  { field: "FIRM_NAME", headerName: "FIRM NAME", width: 200, flex: 1.5 },
  {
    field: "EMPANELMENT_SUB_TYPE",
    headerName: "EMPANELMENT SUB TYPE",
    width: 280,
    flex: 2,
  },
  // { field: "EMPANELED_BY", headerName: "EMPANELED BY", width: 140, flex: 1 },
  // {
  //   field: "EMPANELED_STATUS",
  //   headerName: "EMPANELED STATUS",
  //   width: 160,
  //   flex: 1,
  // },
  {
    field: "EMPANELMENT_TYPE",
    headerName: "EMPANELMENT TYPE",
    width: 160,
    flex: 1,
  },
  // {
  //   field: "FINANCIAL_YEAR",
  //   headerName: "FINANCIAL YEAR",
  //   width: 160,
  //   flex: 1,
  // },
  // {
  //   field: "REQUEST_STATUS",
  //   headerName: "REQUEST STATUS",
  //   width: 160,
  //   flex: 1,
  // },
  { field: "CONTACT_PERSON", headerName: "Contact Person", width: 180, flex: 1 },
  { field: "POC_EMAIL", headerName: "Contact Email", width: 220, flex: 1 },
];

const EmpanelmentData = () => {
  const [empanelmentData, setEmpanelmentData] = useState([]);
  const [userIp, setUserIp] = useState({
    datatype: "",
    type: "",
    subType: "",
    year: "",
  });
  const [selectionOptions, setSelectionOptions] = useState({});
  const [archiveYears, setArchiveYears] = useState([]);
  const [loadingYears, setLoadingYears] = useState(false);

  /**
   * Returns filtered empanelment data based on selected type and subType.
   */
  const filteredRows = useMemo(() => {
    let data = empanelmentData;
    if (userIp.type) {
      data = data.filter((row) => row.EMPANELMENT_TYPE === userIp.type);
    }
    if (userIp.subType) {
      data = data.filter(
        (row) => row.EMPANELMENT_SUB_TYPE === String(userIp.subType)
      );
    }
    return data;
  }, [empanelmentData, userIp.type, userIp.subType]);

  /**
   * Fetches dropdown options for Empanelment Type and SubType.
   */
  useEffect(() => {
    fetchOptions();
    // eslint-disable-next-line
  }, []);

  /**
   * Fetches main grid data on subType change.
   */
  useEffect(() => {
    if (userIp.type !== "") {
      fetchData();
    }
    // eslint-disable-next-line
  }, [userIp.subType]);

  /**
   * Fetches archive years when Archive tab is selected.
   */
  useEffect(() => {
    if (userIp.datatype === "Archive") {
      fetchArchiveYears();
    }
    // eslint-disable-next-line
  }, [userIp.datatype]);

  /**
   * Fetches available empanelment types/subtypes from the backend.
   */
  const fetchOptions = async () => {
    try {
      const response = await axios.post(
        "Server/IAMServices/getAssignmentType",
        {},
        {
          headers: {
            Authorization: `Bearer ${localStorage.getItem("token")}`,
          },
        }
      );
      const result = response.data.result || {};
      const options = {};
      let set1 = new Set(result.map((op) => op.ASSIGNMENT_TYPE_NAME));
      set1.forEach((entry) => {
        options[entry] = result.filter(
          (op) => op.ASSIGNMENT_TYPE_NAME === entry
        );
      });
      setSelectionOptions(options);
    } catch (e) {
      // Handle error if needed
    }
  };

  /**
   * Fetches archive years for Archive tab.
   */
  const fetchArchiveYears = async () => {
    try {
      setLoadingYears(true);
      const response = await axios.post(
        "Server/IAMServices/getArchiveYears",
        {},
        {
          headers: {
            Authorization: `Bearer ${localStorage.getItem("token")}`,
          },
        }
      );
      setArchiveYears(response.data.result || []);
    } catch (e) {
      // Handle error if needed
    } finally {
      setLoadingYears(false);
    }
  };

  /**
   * Fetches empanelment data based on user input.
   */
  const fetchData = async () => {
    const dataToSend = {
      Type: userIp.datatype,
      EMPANELMENT_TYPE: userIp.type,
      EMPANELMENT_SUB_TYPE: userIp.subType,
      financial_year: userIp.year,
    };

    try {
      const jsonFormData = JSON.stringify(dataToSend);
      const encryptedData = await encrypt(iv, salt, jsonFormData);
      const payload = {
        iv: ivBase64,
        salt: saltBase64,
        data: encryptedData,
      };
      const response = await axios.post(
        "Server/IAMServices/getActiveFRN",
        payload,
        {
          headers: {
            Authorization: `Bearer ${localStorage.getItem("token")}`,
          },
        }
      );
      setEmpanelmentData(response.data.result || []);
    } catch (e) {
      // Handle error if needed
    }
  };

  /**
   * Downloads the filtered empanelment data as an Excel file.
   */
  const handleDownload = async () => {
    let dataToSend = filteredRows;
    try {
      let jsonFormData = JSON.stringify(dataToSend);
      const encryptedData = await encrypt(iv, salt, jsonFormData);
      const payload = {
        iv: ivBase64,
        salt: saltBase64,
        data: encryptedData,
      };
      const response = await axios.post(
        "Server/IAMServices/download_empanelment_data",
        payload,
        {
          headers: {
            Authorization: `Bearer ${localStorage.getItem("token")}`,
          },
        }
      );
      const resData = response.data.result;
      if (resData) {
        downloadFile(resData, "excel", `${userIp.datatype}_Empanelment_list`);
      }
    } catch (e) {
      // Handle download error if needed
    }
  };

  /**
   * Custom overlay for DataGrid when no rows are available.
   * Shows a message based on selected year and subtype.
   */
  const CustomNoRowsOverlay = () => {
    let message = "No data available. Please Check back later!";
    if (userIp.year && userIp.subType) {
      message = `Data is Not Available for Financial Year :${userIp.year}, Empanelment type:${userIp.type} & Subtype:${userIp.subType} `;
    } else if (userIp.subType) {
      message = `For ${userIp.subType} Subtype, Data is Not Available.`;
    } else if (userIp.year) {
      message = `For year ${userIp.year}  Data is Not Available.`;
    }
    return (
      <GridOverlay>
        <div style={{ padding: 16, fontSize: 16, color: "#555" }}>
          {message}
        </div>
      </GridOverlay>
    );
  };

  /**
   * Handles tab change (Active/Archive).
   */
  const handleTabChange = (event, newValue) => {
    setUserIp({ datatype: newValue, type: "", subType: "", year: "" });
    setEmpanelmentData([]);
  };

  /**
   * Handles Empanelment Type selection.
   */
  const handleSelectType = (type) => {
    setUserIp((prev) => ({ ...prev, type, subType: "" }));
  };

  /**
   * Handles Empanelment SubType selection.
   */
  const handleSelectSubType = (subType) => {
    setUserIp((prev) => ({ ...prev, subType }));
  };

  /**
   * Handles Year selection for Archive tab.
   */
  const handleSelectYear = (year) => {
    setUserIp((prev) => ({ ...prev, year }));
  };

  return (
    <Box sx={{ p: 3, width: "100%" }}>
      <Typography variant="h5" sx={{ mb: 2 }}>
        Empanelment Data
      </Typography>
      <Divider sx={{ mb: 3 }} />

      <Paper sx={{ p: 3, width: "100%", overflow: "hidden" }}>
        <Box sx={{ mb: 3 }}>
          <Tabs
            value={userIp.datatype}
            onChange={handleTabChange}
            sx={{ borderBottom: 1, borderColor: "divider" }}
          >
            <Tab label="Active" value="Active" />
            <Tab label="Archive" value="Archive" />
          </Tabs>
        </Box>

        {userIp.datatype && (
          <Stack
            direction={{ xs: "column", sm: "row" }}
            spacing={2}
            sx={{ mb: 3 }}
            alignItems={{ xs: "stretch", sm: "center" }}
          >
            {userIp.datatype === "Archive" && (
              <TextField
                select
                label="Year"
                value={userIp.year}
                onChange={(e) => handleSelectYear(e.target.value)}
                fullWidth={{ xs: true, sm: false }}
                sx={{ minWidth: 200 }}
                size="small"
                disabled={loadingYears}
              >
                <MenuItem value="">Select Year</MenuItem>
                {archiveYears.map((year) => (
                  <MenuItem key={year} value={year}>
                    {year}
                  </MenuItem>
                ))}
              </TextField>
            )}

            <TextField
              select
              label="Type"
              value={userIp.type}
              onChange={(e) => handleSelectType(e.target.value)}
              fullWidth={{ xs: true, sm: false }}
              sx={{ minWidth: 200 }}
              size="small"
            >
              <MenuItem value="">Select Type</MenuItem>
              {Object.keys(selectionOptions).map((type) => (
                <MenuItem key={type} value={type}>
                  {type}
                </MenuItem>
              ))}
            </TextField>

            {userIp.type && (
              <TextField
                select
                label="Sub Type"
                value={userIp.subType}
                onChange={(e) => handleSelectSubType(e.target.value)}
                fullWidth={{ xs: true, sm: false }}
                sx={{ minWidth: 220 }}
                size="small"
              >
                {selectionOptions[userIp.type]?.map((option) => (
                  <MenuItem
                    key={option.ASSIGNMENT_SUBTYPE}
                    value={option.ASSIGNMENT_TYPE_ID}
                  >
                    {option.ASSIGNMENT_SUBTYPE}
                  </MenuItem>
                ))}
              </TextField>
            )}
          </Stack>
        )}

        {userIp.subType !== "" && (
          <Box sx={{ width: "100%" }}>
            <Box sx={{ display: "flex", justifyContent: "flex-end", mb: 2 }}>
              <Button
                onClick={handleDownload}
                variant="contained"
                size="medium"
                sx={{ ml: "auto" }}
              >
                Download Excel
              </Button>
            </Box>
            <Typography
              variant="body1"
              style={{
                fontSize: "0.875rem",
                fontStyle: "italic",
                backgroundColor: "#f9f9f9",
                padding: "4px 8px",
                borderLeft: "4px solid #ccc",
                color: "#555",
                mb: 2,
                textAlign: "right",
              }}
            >
              This Excel download contains all fields
            </Typography>

            <Box sx={{ height: 500, width: "100%" }}>
              <DataGrid
                rows={filteredRows}
                columns={columns}
                getRowId={(row) => row.FRN}
                slots={{
                  noRowsOverlay: CustomNoRowsOverlay,
                  toolbar: GridToolbar,
                }}
                pageSizeOptions={[10, 25, 50]}
                sx={{
                  "& .MuiDataGrid-columnHeaders": {
                    backgroundColor: "grey.100",
                  },
                }}
              />
            </Box>
          </Box>
        )}
      </Paper>
    </Box>
  );
};

export default EmpanelmentData;
