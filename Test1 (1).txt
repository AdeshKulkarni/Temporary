    @Transactional
    public String createBatchFromCacheAsync(String requestId, String commonRemarks, String creatorId, Integer creatorRole, String vid , LocalDateTime scheduledTime) {
        String lockKey = "LOCK_REQ_" + requestId;
        Boolean acquired;

        try {
            acquired = redisTemplate.opsForValue().setIfAbsent(lockKey, new byte[0], 60, TimeUnit.MINUTES);
        } catch (DataAccessException e) {
            log.error("REDIS_LOCK_FAIL | ReqID: {} | Status: Connection to lock service failed", requestId);
            throw new CustomExceptions.DatabaseOperationException("Distributed lock service unavailable", e);
        }

        if (Boolean.FALSE.equals(acquired)) {
            log.warn("CONCURRENCY_LOCKED | ReqID: {} | Status: Processing already in progress", requestId);
            throw new CustomExceptions.ConcurrencyLockException("Batch creation is already in progress for this Request ID.");
        }

        try {
            // FIX 1: Validate VID existence to prevent "null" crashes
            if (vid == null || vid.trim().isEmpty()) {
                throw new CustomExceptions.InvalidDataException("VID is missing from the payload. Cannot save batch.");
            }

            // --- ATOMIC LOCK FOR MANUAL POSTING ---
            String lockSql = "SELECT STATUS FROM REQUEST_VOUCHER WHERE VID = ? FOR UPDATE";
            String vidStatus;
            try {
                vidStatus = jdbcTemplate.queryForObject(lockSql, String.class, vid);
            } catch (EmptyResultDataAccessException e) {
                throw new CustomExceptions.InvalidDataException("VID (" +vid + ") does not exist.");
            }

            if (!"ACTIVE".equals(vidStatus)) {
                throw new CustomExceptions.InvalidDataException("VID (" + vid + ") is currently in use or locked.");
            }

            // Immediately mark as USED
            workflowEngine.updateVidStatus(vid, VoucherStatus.USED);

            // 1. GENERATE IDs
            String batchId = sequenceService.getNextBatchId();
            String journalPrefix = sequenceService.getNextJournalIdPrefix();

            // 2. FETCH ROW COUNT
            byte[] metaBytes = redisTemplate.opsForValue().get("DATA_META_" + requestId);
            long totalRows = 0;
            if (metaBytes != null) {
                try {
                    Map<String, Object> meta = objectMapper.readValue(metaBytes, new TypeReference<>() {
                    });
                    totalRows = ((Number) meta.get("totalRows")).longValue();
                } catch (Exception ignored) {
                }
            }

            // 3. WORKFLOW ROUTING
            String workflowCode;
            try {
                workflowCode = jdbcTemplate.queryForObject(
                        "SELECT WORKFLOW_CODE FROM WORKFLOW_TRANSITIONS WHERE ALLOWED_ROLE_ID = ? AND WORKFLOW_ORDER = 1 FETCH FIRST 1 ROWS ONLY",
                        String.class, creatorRole
                );
            } catch (EmptyResultDataAccessException e) {
                throw new CustomExceptions.InvalidDataException("Configuration Error: No workflow mapped for Role ID " + creatorRole);
            }

            Long firstApproverWorkflowId;
            try {
                firstApproverWorkflowId = jdbcTemplate.queryForObject(
                        "SELECT WORKFLOW_ID FROM WORKFLOW_TRANSITIONS WHERE WORKFLOW_CODE = ? AND WORKFLOW_ORDER = 2",
                        Long.class, workflowCode
                );
            } catch (EmptyResultDataAccessException e) {
                throw new CustomExceptions.InvalidDataException("Configuration Error: Missing Step 2 for " + workflowCode);
            }

            // 4. SYNCHRONOUS MASTER RECORD CREATION
            VoucherPosting posting = new VoucherPosting();
            posting.setBatchId(batchId);
            posting.setVidFk(vid);
            posting.setCommonBatchRemarks(commonRemarks);
            posting.setCreatorRole(creatorRole);
            posting.setCreatedBy(creatorId);
            posting.setWorkflowCode(workflowCode);
            posting.setPendingAt(firstApproverWorkflowId);
            posting.setBatchStatus("S");
            posting.setRecordsCount(totalRows);

            // Save the schedule directly from Step 3 UI
            if (scheduledTime != null) {
                workflowEngine.validateScheduledTime(scheduledTime, false);
                posting.setIS_EXECUTION_SCHEDULED("YES");
                posting.setScheduledTime(scheduledTime);
            } else {
                posting.setIS_EXECUTION_SCHEDULED("NO");
                posting.setScheduledTime(null);
            }
            posting.setIsScheduleOverride("NO");
            posting.setRecordsCount(totalRows);

            voucherPostingRepository.save(posting);
            String action = (scheduledTime != null) ? "Submitted (Scheduled)" : "Submitted";
            workflowEngine.logWorkflowAction(batchId, vid, action, creatorId, commonRemarks, scheduledTime);
            progressService.sendProgress(batchId, creatorId, 0, "PROCESSING", "Initializing Batch Creation...");

            // 6. ASYNC HANDOFF (Pass to background thread)
            asyncWorker.executeAsyncBatchCreation(batchId, journalPrefix, requestId, commonRemarks, creatorId, creatorRole, vid);

            return batchId;


        } catch (Exception e) {
            // ==============================================================================
            // THE FIX: IF ANY VALIDATION FAILS BEFORE REACHING THE ASYNC WORKER,
            // WE MUST MANUALLY DELETE THE ORPHANED LOCK HERE SO THE USER CAN TRY AGAIN!
            // ==============================================================================
            redisTemplate.delete(lockKey);

            if (e instanceof org.springframework.dao.DataIntegrityViolationException) {
                log.error("DB_CONSTRAINT_FAIL | ReqID: {} | Error: {}", requestId, ((org.springframework.dao.DataIntegrityViolationException) e).getRootCause() != null ? ((org.springframework.dao.DataIntegrityViolationException) e).getRootCause().getMessage() : e.getMessage());
                throw new CustomExceptions.InvalidDataException("Database constraint failed. Ensure the VID is not duplicated or missing.");
            } else if (e instanceof DataAccessException) {
                log.error("SEQUENCE_GEN_FAIL | ReqID: {} | Status: Failed to generate sequence", requestId);
                throw new CustomExceptions.DatabaseOperationException("Failed to generate Batch ID sequence", e);
            }

            // Rethrow the 30-minute time validation error or any other exception back to the UI
            throw e;
        }
        finally {
            // --- NEW: MICROSECOND RELEASE ---
            // Free the VID instantly whether the save succeeded or crashed.
            // If it succeeded, the DB is already marked as 'USED', so it's safe to drop the Redis lock.
            if (vid != null) {
                releaseVidLock(vid, creatorId);
            }
        }
    }

    @Transactional
    public void processBulkRequestsAsync(BulkProcessVoucherPostingDto dto, String executorId, Integer executorRole) {
        // Robust batch update lock statement safely scoping both standard and timer pending steps
        String lockSql = "UPDATE VOUCHER_POSTING SET BATCH_STATUS = 'A' WHERE BATCH_ID = ? AND BATCH_STATUS IN ('P', 'PENDING')";
        int updated = jdbcTemplate.update(lockSql, dto.getBatchId());

        if (updated == 0) {
            log.warn("APPROVAL_LOCK_FAIL | Batch: {} | Status: Already processing or invalid status", dto.getBatchId());
            throw new IllegalStateException("Batch is already being processed or is not in a valid state for execution.");
        }

        progressService.sendProgress(dto.getBatchId(), executorId, 0, "PROCESSING", "Approval Queued...");
        asyncWorker.executeAsyncBatchProcessing(dto, executorId);
    }

    @Transactional
    public List<JournalRequest> createBatchRequest(BatchPostingDto dto, String creatorId, Integer creatorRole) {

        if (dto.getVid() == null || dto.getVid().trim().isEmpty()) {
            throw new CustomExceptions.InvalidDataException("VID is missing from the payload. Cannot save batch.");
        }

        // NOTE: The old LOCK_MANUAL_VID_ redis logic was completely removed from here.
        // The UI lock now handles concurrency safely.

        try {
            String batchId = sequenceService.getNextBatchId();
            String journalPrefix = sequenceService.getNextJournalIdPrefix();

            // Checks if the VID is already used by an active/pending batch, ignoring rejected ones
            Integer vidExists = jdbcTemplate.queryForObject(
                    "SELECT COUNT(*) FROM VOUCHER_POSTING WHERE VID_FK = ? AND BATCH_STATUS NOT IN ('R', 'REJECTED', 'ERROR')",
                    Integer.class, dto.getVid()
            );

            if (vidExists != null && vidExists > 0) {
                throw new CustomExceptions.InvalidDataException("Voucher ID (" + dto.getVid() + ") is currently in use by an active batch. Please select a fresh VID.");
            }

            List<JournalRequest> entities = new ArrayList<>();
            Map<String, BigDecimal> balancingMap = new java.util.HashMap<>();
            int rowIndex = 1;

            for (BatchPostingDto.JournalRequestRow row : dto.getRows()) {
                JournalRequest entity = new JournalRequest();
                entity.setRequestStatus(RequestStatus.PENDING);
                entity.setRequestDate(LocalDateTime.now());
                entity.setCreatorId(creatorId);
                entity.setBatchId(batchId);

                String journalId = journalPrefix + "-" + rowIndex;
                entity.setJournalId(journalId);
                entity.setBranchCode(row.getBranch());
                entity.setCurrency(row.getCurrency());

                String cgl = row.getCgl() != null ? row.getCgl().trim() : "";
                entity.setCgl(cgl);

                BigDecimal amount = row.getAmount();

                if (amount == null || amount.compareTo(BigDecimal.ZERO) == 0) {
                    throw new CustomExceptions.InvalidDataException("Transaction amount cannot be zero.");
                }

                entity.setAmount(amount);
                entity.setNarration(row.getRemarks());
                entity.setProductCode(row.getProductType());

                // --- STRICT BACKEND BALANCING RULE ---
                String cglType = cgl.startsWith("5") ? "MEMO" : "NORMAL";
                String compositeKey = row.getBranch() + " - " + row.getCurrency() + " - " + cglType;
                balancingMap.put(compositeKey, balancingMap.getOrDefault(compositeKey, BigDecimal.ZERO).add(amount));

                entities.add(entity);
                rowIndex++;
            }

            // --- ENFORCE BALANCE ---
            for (Map.Entry<String, BigDecimal> entry : balancingMap.entrySet()) {
                if (entry.getValue().compareTo(BigDecimal.ZERO) != 0) {
                    throw new CustomExceptions.InvalidDataException(
                            "Balancing rule failed for [" + entry.getKey() + "]. Total Debits and Credits must exactly match."
                    );
                }
            }

            List<JournalRequest> savedEntities = journalRequestRepository.saveAll(entities);

            // ==========================================
            // 2. DYNAMIC WORKFLOW INITIALIZATION
            // ==========================================
            String workflowCode;
            try {
                workflowCode = jdbcTemplate.queryForObject(
                        "SELECT WORKFLOW_CODE FROM WORKFLOW_TRANSITIONS WHERE ALLOWED_ROLE_ID = ? AND WORKFLOW_ORDER = 1 FETCH FIRST 1 ROWS ONLY",
                        String.class, creatorRole
                );
            } catch (EmptyResultDataAccessException e) {
                throw new CustomExceptions.InvalidDataException("Configuration Error: No workflow mapped for Role ID " + creatorRole);
            }

            Map<String, Object> step2Info;
            try {
                step2Info = jdbcTemplate.queryForMap(
                        "SELECT WORKFLOW_ID, ALLOWED_ROLE_ID, WORKFLOW_STAGE  FROM WORKFLOW_TRANSITIONS WHERE WORKFLOW_CODE = ? AND WORKFLOW_ORDER = 2",
                        workflowCode
                );
            } catch (EmptyResultDataAccessException e) {
                throw new CustomExceptions.InvalidDataException("Configuration Error: Missing Step 2 for " + workflowCode);
            }

            Long firstApproverWorkflowId = ((Number) step2Info.get("WORKFLOW_ID")).longValue();

            VoucherPosting posting = new VoucherPosting();
            posting.setBatchId(batchId);
            posting.setVidFk(dto.getVid());
            posting.setCreatedBy(creatorId);
            posting.setPendingAt(firstApproverWorkflowId);
            posting.setBatchStatus("P"); // 'P' = Pending
            posting.setWorkflowCode(workflowCode);
            posting.setCommonBatchRemarks(dto.getCommonBatchRemarks());
            posting.setCreatorRole(creatorRole);

            if (dto.getRows() != null) {
                posting.setRecordsCount((long) dto.getRows().size());
            }

            // Handle Maker Scheduling
            if (dto.getScheduledTime() != null) {
                workflowEngine.validateScheduledTime(dto.getScheduledTime() , false);
                posting.setIS_EXECUTION_SCHEDULED("YES");
                posting.setScheduledTime(dto.getScheduledTime());
                posting.setIsScheduleOverride("NO");
            } else {
                posting.setIS_EXECUTION_SCHEDULED("NO");
                posting.setIsScheduleOverride("NO");
            }

            voucherPostingRepository.save(posting);
            logAudit(creatorId, "BATCH_MANUAL", "ADD", "Manual Batch " + batchId + " Created under " + workflowCode);

            if (dto.getVid() != null && !dto.getVid().isEmpty()) {
                workflowEngine.updateVidStatus(dto.getVid(), VoucherStatus.USED);
            }

            String action = (dto.getScheduledTime() != null) ? "Submitted (Scheduled)" : "Submitted";
            workflowEngine.logWorkflowAction(batchId, dto.getVid(), action, creatorId, dto.getCommonBatchRemarks(), dto.getScheduledTime());

            return savedEntities;

        } finally {
            // --- NEW: MICROSECOND RELEASE ---
            // Drops the UI-driven lock instantly so if validation failed, the VID is freed for others.
            releaseVidLock(dto.getVid(), creatorId);
        }
    }

    public void deleteBatchRequest(String batchId, String userId) {
        // 1. Fetch the batch
        VoucherPosting batch = voucherPostingRepository.findById(batchId)
                .orElseThrow(() -> new CustomExceptions.InvalidDataException("Batch not found."));

        // 2. SECURITY CHECK: Only the creator can delete it
        if (!batch.getCreatedBy().equalsIgnoreCase(userId)) {
            log.warn("SECURITY_VIOLATION | User {} attempted to delete batch {} owned by {}", userId, batchId, batch.getCreatedBy());
            throw new IllegalStateException("You do not have permission to delete this batch.");
        }

        // 3. STATE CHECK: Can only delete if it is still PENDING ('P')
        if (!"P".equalsIgnoreCase(batch.getBatchStatus())) {
            throw new IllegalStateException("Only Pending batches can be deleted. Current status: " + batch.getBatchStatus());
        }

        // 4. STRICT WORKFLOW ORDER CHECK: Ensure it is exactly at Order 2
        String orderSql = """
            SELECT wt.WORKFLOW_ORDER 
            FROM VOUCHER_POSTING vp 
            JOIN WORKFLOW_TRANSITIONS wt ON vp.PENDING_AT = wt.WORKFLOW_ID 
            WHERE vp.BATCH_ID = ?
        """;
        Integer currentOrder;
        try {
            currentOrder = jdbcTemplate.queryForObject(orderSql, Integer.class, batchId);
        } catch (EmptyResultDataAccessException e) {
            currentOrder = -1; // Failsafe
        }

        if (currentOrder == null || currentOrder != 2) {
            throw new IllegalStateException("Batch cannot be deleted as it has already been processed or moved from the initial stage.");
        }


        // 6. Hand off to the background thread to do the heavy database deletion
        asyncWorker.executeAsyncBatchCancellation(batchId, userId);
    }


    @Transactional
    public void cancelMyRequest(Long requestId, String userId) {
        JournalRequest request = journalRequestRepository.findById(requestId)
                .orElseThrow(() -> new RuntimeException("Request not found: " + requestId)); // Replace RuntimeException with your custom ResourceNotFoundException if you have one

        if (!request.getCreatorId().equalsIgnoreCase(userId)) {
            log.warn("UNAUTHORIZED_DELETE | User: {} | RequestOwner: {} | ReqID: {}", userId, request.getCreatorId(), requestId);
            throw new IllegalStateException("You can only cancel your own requests.");
        }

        if (request.getRequestStatus() != RequestStatus.PENDING) {
            throw new IllegalStateException("Only PENDING requests can be cancelled. Current: " + request.getRequestStatus());
        }

        journalRequestRepository.delete(request);
        log.info("REQUEST_CANCELLED | User: {} | ReqID: {}", userId, requestId);
    }

    @Transactional
    public void updateBatchSchedule(String batchId, String scheduledTime) {
        Integer exists = jdbcTemplate.queryForObject("SELECT COUNT(*) FROM VOUCHER_POSTING WHERE BATCH_ID = ?", Integer.class, batchId);
        if (exists == null || exists == 0) {
            throw new CustomExceptions.InvalidDataException("The batch is still being initialized. Please wait a few seconds.");
        }

        // --- ADD THIS LOCK ---
        VoucherPosting posting = voucherPostingRepository.findById(batchId)
                .orElseThrow(() -> new CustomExceptions.InvalidDataException("Batch not found."));

        if (posting.getVidFk() != null) {
            // This is where you flip it to USED at the final moment
            workflowEngine.updateVidStatus(posting.getVidFk(), VoucherStatus.USED);
        }


        if (scheduledTime != null && !scheduledTime.isEmpty()) {
            LocalDateTime parsedTime = LocalDateTime.parse(scheduledTime, java.time.format.DateTimeFormatter.ofPattern("yyyy-MM-dd HH:mm:ss"));
            workflowEngine.validateScheduledTime(parsedTime , false);

            jdbcTemplate.update("UPDATE VOUCHER_POSTING SET SCHEDULED_TIME = TO_TIMESTAMP(?, 'YYYY-MM-DD HH24:MI:SS'), IS_SCHEDULE_OVERRIDE = 'NO', IS_EXECUTION_SCHEDULED = 'YES' WHERE BATCH_ID = ?", scheduledTime, batchId);
            jdbcTemplate.update("UPDATE VOUCHER_WORKFLOW SET LAST_ACTION = 'Submitted (Scheduled)', SCHEDULED_TIME = TO_TIMESTAMP(?, 'YYYY-MM-DD HH24:MI:SS') WHERE BATCH_ID_FK = ? AND LAST_ACTION LIKE 'Submit%'", scheduledTime, batchId);
        }

        // SMART DOUBLE-LOCK LOGIC:
        // Check if the AsyncWorker is still writing data. (If actual rows match requested rows, it's done!)
        String checkDataLoadedSql = """
            SELECT CASE WHEN p.RECORDS_COUNT = (SELECT COUNT(*) FROM VOUCHER_POSTING_DATA d WHERE d.BATCH_ID = p.BATCH_ID) 
            THEN 1 ELSE 0 END FROM VOUCHER_POSTING p WHERE p.BATCH_ID = ?
        """;

        Integer isDataLoaded = jdbcTemplate.queryForObject(checkDataLoadedSql, Integer.class, batchId);

        if (isDataLoaded != null && isDataLoaded == 1) {
            // 1. Data is fully loaded. Safe to show to checkers!
            jdbcTemplate.update("UPDATE VOUCHER_POSTING SET BATCH_STATUS = 'P' WHERE BATCH_ID = ?", batchId);

            // 2. FETCH WORKFLOW DETAILS FOR NOTIFICATION
            // 2. DYNAMIC NOTIFICATION TRIGGER
            // This is the only place it happens now!
            try {
                // REPLACEMENT: Changed DESIGNATION to WORKFLOW_STAGE
                String wfSql = """
                    SELECT wt.ALLOWED_ROLE_ID, wt.WORKFLOW_STAGE 
                    FROM VOUCHER_POSTING vp 
                    JOIN WORKFLOW_TRANSITIONS wt ON vp.PENDING_AT = wt.WORKFLOW_ID 
                    WHERE vp.BATCH_ID = ? AND wt.WORKFLOW_ORDER = 2
                """;

                Map<String, Object> step2Info = jdbcTemplate.queryForMap(wfSql, batchId);


                String nextApproverRoleId = String.valueOf(step2Info.get("ALLOWED_ROLE_ID"));
                String stageName = (String) step2Info.get("WORKFLOW_STAGE"); // Update key here too

                // 3. SEND THE NOTIFICATION TO THE CHECKERS NOW
                notificationWriterService.createNextStepApprovalNotification(batchId, nextApproverRoleId, stageName);

            } catch (EmptyResultDataAccessException e) {
                log.error("Could not find Step 2 workflow configuration for Batch {} to send notification.", batchId);
            }
        } else {
            // Data is still loading, do not notify yet.
            // Note: Your AsyncWorker MUST also call this notification logic
            // once it finishes its final row insertion and performs the BATCH_STATUS = 'P' update.
            log.info("Schedule updated for Batch {}, data still loading. Notification deferred.", batchId);
        }

    }

    private void logAudit(String user, String action, String type, String val) {
        try {

            JournalLog l = new JournalLog();
            l.setUserId(user);
            l.setActionType(action);
            l.setChangeType(type);
            l.setNewValue(val.length() > 3900 ? val.substring(0, 3900) : val);
            l.setActionTime(LocalDateTime.now());
            voucherLogRepository.save(l);
        }
        catch (DataAccessException e) {
            log.error("AUDIT_LOG_FAIL | User: {} | Status: Persistence error", user);
        }
    }

    public void acquireVidLock(String vid, String userId) {
        String lockKey = "LOCK_VID_" + vid;
        byte[] userBytes = userId.getBytes(java.nio.charset.StandardCharsets.UTF_8);

        // Atomic SETNX with 30-minute TTL
        Boolean acquired = redisTemplate.opsForValue().setIfAbsent(lockKey, userBytes, 30, TimeUnit.MINUTES);

        if (Boolean.FALSE.equals(acquired)) {
            // Lock is already held by someone else
            throw new CustomExceptions.ConcurrencyLockException("This VID is currently locked by another user. Please select a different VID.");
        }

        // Secondary safeguard: Ensure the DB hasn't already marked it as USED before the lock was acquired
        String dbStatus = jdbcTemplate.queryForObject("SELECT STATUS FROM REQUEST_VOUCHER WHERE VID = ?", String.class, vid);
        if (!"ACTIVE".equalsIgnoreCase(dbStatus)) {
            redisTemplate.delete(lockKey); // Release immediately if DB is dirty
            throw new CustomExceptions.ConcurrencyLockException("This VID has already been used. Please select a different VID.");
        }
    }

    public void releaseVidLock(String vid, String userId) {
        String lockKey = "LOCK_VID_" + vid;
        byte[] existingUserBytes = redisTemplate.opsForValue().get(lockKey);

        // Only release the lock if it belongs to the user who requested the release
        if (existingUserBytes != null) {
            String lockedBy = new String(existingUserBytes, java.nio.charset.StandardCharsets.UTF_8);
            if (lockedBy.equals(userId)) {
                redisTemplate.delete(lockKey);
            }
        }
    }
