public class Rec {

    public static int solve(int[] arr, int i) {
        // Base Case 1: Reached past the end of the array
        if (i >= arr.length) {
            return 0;
        }

        // Base Case 2: On the very last element (arr[i + 1] does not exist)
        if (i == arr.length - 1) {
            return Math.max(0, arr[i]);
        }

        // Choice 1: Skip arr[i] -> move to i + 1
        int skip = solve(arr, i + 1);

        // Choice 2: Take arr[i] only -> MUST skip i + 1 -> jump to i + 2
        int takeOne = arr[i] + solve(arr, i + 2);

        // Choice 3: Take arr[i] and arr[i + 1] -> MUST skip i + 2 -> jump to i + 3
        int takeTwo = arr[i] + arr[i + 1] + solve(arr, i + 3);

        return Math.max(skip, Math.max(takeOne, takeTwo));
    }

    public static void main(String[] args) {
        int[] arr = {1, 2, 3, 4, 5, 6};
        System.out.println(solve(arr, 0)); // Output: 16
    }
}
