    private void handleAsyncApprovalError(String batchId, String executorId, RequestStatus actionStatus, String defaultErrorMsg) {

        try {
            // 1. Fetch WORKFLOW_CODE directly from the main VOUCHER_POSTING table 
            // (No blind lookups to REQUEST_VOUCHER)
            String workflowCode = jdbcTemplate.queryForObject(
                "SELECT WORKFLOW_CODE FROM VOUCHER_POSTING WHERE BATCH_ID = ?", 
                String.class, batchId
            );

            if (workflowCode != null) {
                // 2. Find the Final Step (Highest Order) for this specific workflow
                Map<String, Object> finalStepInfo = jdbcTemplate.queryForMap(
                    "SELECT WORKFLOW_ID FROM (" +
                    "   SELECT WORKFLOW_ID " +
                    "   FROM WORKFLOW_TRANSITIONS " +
                    "   WHERE WORKFLOW_CODE = ? " +
                    "   ORDER BY WORKFLOW_ORDER DESC" +
                    ") WHERE ROWNUM = 1", workflowCode
                );
                
                String finalPendingAt = String.valueOf(finalStepInfo.get("WORKFLOW_ID"));

                // 3. Revert to Pending ('P') AND restore the PENDING_AT column
                // (Removed CURRENT_ORDER entirely to perfectly match your DB schema)
                jdbcTemplate.update(
                    "UPDATE VOUCHER_POSTING SET BATCH_STATUS = 'P', IS_EXECUTION_SCHEDULED = 'NO', PENDING_AT = ? WHERE BATCH_ID = ?",
                    finalPendingAt, batchId
                );
                
                log.info("Workflow state successfully reverted to final approver stage for Batch: {}", batchId);
            } else {
                throw new RuntimeException("Workflow Code is null in VOUCHER_POSTING table.");
            }

        } catch (Exception e) {
            log.error("Failed to revert workflow state deeply for Batch: {}. Using fallback.", batchId, e);
            // Safe fallback
            jdbcTemplate.update(
                "UPDATE VOUCHER_POSTING SET BATCH_STATUS = 'P', IS_EXECUTION_SCHEDULED = 'NO' WHERE BATCH_ID = ?",
                batchId
            );
        }

        // 4. The exact professional message requested
        String finalMsg = "Execution failed due to a database or server timeout. Please retry the approval process.";
        progressService.sendProgress(batchId, executorId, 0, "FAILED", finalMsg);

        String vid = null;
        String creatorId = null;

        try {
            Map<String, Object> batchInfo = jdbcTemplate.queryForMap(
                    "SELECT VID_FK, CREATED_BY FROM VOUCHER_POSTING WHERE BATCH_ID = ?", batchId
            );
            vid = (String) batchInfo.get("VID_FK");
            creatorId = (String) batchInfo.get("CREATED_BY");
        } catch (org.springframework.dao.EmptyResultDataAccessException e) {
            log.warn("Batch info not found for Batch: {}", batchId);
        }

        // 5. Log the "Execution Failed" event in Workflow History so the Frontend knows to show the Retry button
        workflowEngine.logWorkflowAction(batchId, vid, "Execution Failed", executorId, finalMsg, null);

        // 6. Send the notification explicitly telling the user to retry
        try {
            String checkerMsg = String.format("Execution alert for Batch '%s': %s", batchId, finalMsg);
            notificationWriterService.createNotification(executorId, null, checkerMsg, "/voucher-view/" + batchId, batchId, "VoucherService");

            if (creatorId != null && !creatorId.equalsIgnoreCase(executorId)) {
                String makerMsg = String.format("Execution alert for your Batch '%s': %s", batchId, finalMsg);
                notificationWriterService.createNotification(creatorId, null, makerMsg, "/voucher-view/" + batchId, batchId, "VoucherService");
            }
        } catch (Exception e) {
            log.warn("NOTIFICATION_WARN | Batch: {} | Could not send failure notifications", batchId);
        }
    }
