import React, { useState, useEffect, useCallback } from "react";
import {
  Box,
  Typography,
  Chip,
  Dialog,
  DialogActions,
  DialogContent,
  DialogContentText,
  DialogTitle,
  Button,
  CircularProgress,
  Tooltip,
  IconButton,
  Stack,
} from "@mui/material";
import { DataGrid } from "@mui/x-data-grid";
import {
  Ballot,
  Cancel as CancelIcon,
  HelpOutline as HelpIcon,
} from "@mui/icons-material";
import AccountCircleIcon from "@mui/icons-material/AccountCircle";
import InfoIcon from "@mui/icons-material/Info";

import useApi from "../../../hooks/useApi";
import useCustomSnackbar from "../../../utils/useCustomSnackbar";

// Helper to get chip color based on status
const getStatusChipColor = (status) => {
  switch (status) {
    case "PENDING":
      return "warning";
    case "ACCEPTED":
      return "success";
    case "REJECTED":
      return "error";
    default:
      return "default";
  }
};

function CustomNoRowsOverlay() {
  return (
    <Stack height="100%" alignItems="center" justifyContent="center">
      <InfoIcon sx={{ fontSize: 48, color: "text.secondary", mb: 2 }} />
      <Typography variant="h6">No Requests Found</Typography>
      <Typography>You have not created any MOC requests.</Typography>
    </Stack>
  );
}

// Renamed component to follow React standards
export default function JournalPostingStatus() {
  const { callApi } = useApi();
  const showSnackBar = useCustomSnackbar();

  const [loading, setLoading] = useState(true);
  const [rows, setRows] = useState([]);

  const [isModalOpen, setIsModalOpen] = useState(false);
  const [isSubmitting, setIsSubmitting] = useState(false);
  const [selectedRequest, setSelectedRequest] = useState(null);

  const fetchMyRequests = useCallback(async () => {
    setLoading(true);
    try {
      const myRequests = await callApi("MS/moc/my-requests", null, "GET");
      console.log(myRequests);

      if (Array.isArray(myRequests)) {
        setRows(myRequests);
      } else {
        setRows([]);
      }
    } catch (err) {
      console.error("Failed to fetch user's requests:", err);
      const errorMsg =
        err.response?.data?.message || "Failed to load your requests.";
      showSnackBar(errorMsg, "error");
    } finally {
      setLoading(false);
    }
  }, [callApi, showSnackBar]);

  useEffect(() => {
    fetchMyRequests();
  }, [fetchMyRequests]);

  const handleOpenModal = (request) => {
    setSelectedRequest(request);
    setIsModalOpen(true);
  };

  const handleCloseModal = () => {
    if (isSubmitting) return;
    setIsModalOpen(false);
    setSelectedRequest(null);
  };

  const handleConfirmCancel = async () => {
    if (!selectedRequest) return;

    setIsSubmitting(true);
    try {
      await callApi(
        `MS/moc/my-request/${selectedRequest.id}`,
        null,
        "DELETE"
      );
      showSnackBar(
        `Request (Journal ID: ${selectedRequest.journalId}) was successfully canceled.`,
        "success"
      );
      handleCloseModal();
      fetchMyRequests(); 
    } catch (error) {
      console.error("Failed to cancel request:", error);
      const errorMsg =
        error.response?.data?.message || "Failed to cancel the request.";
      showSnackBar(errorMsg, "error");
    } finally {
      setIsSubmitting(false);
    }
  };

 
  const columns = [
    {
      field: "requestStatus",     
      headerName: "Req Status",
      flex: 1,
      minWidth: 100,
      renderCell: (params) => (
        <Chip
          label={params.value}
          color={getStatusChipColor(params.value)}
          size="small"
        />
      ),
    },
    {
      field: "batchId",          
      headerName: "Batch ID",
      flex: 1.5,
      minWidth: 150,
    },
    {
      field: "journalId",          
      headerName: "Journal ID",
      flex: 1.5,
      minWidth: 150,
    },
    {
      field: "requestDate",       
      headerName: "Req Date",
      flex: 1.5,
      minWidth: 180,
      renderCell: (params) => (
        <Box>
          {new Date(params.value).toLocaleString("en-IN", {
            dateStyle: "medium",
            timeStyle: "short",
          })}
        </Box>
      ),
    },
    {
      field: "creatorId",        
      headerName: "Creator ID",
      flex: 1,
      minWidth: 120,
      renderCell: (params) => (
        <Chip
          icon={<AccountCircleIcon />}
          label={params.value}
          size="small"
          variant="outlined"
        />
      ),
    },
    {
      field: "executorId",        
      headerName: "Executor ID",
      flex: 1,
      minWidth: 120,
      renderCell: (params) =>
        params.value ? (
          <Chip
            icon={<AccountCircleIcon />}
            label={params.value}
            size="small"
          />
        ) : (
          <Chip label="N/A" size="small" variant="outlined" />
        ),
    },
    {
      field: "executorRemarks",    
      headerName: "Executor Remarks",
      flex: 2,
      minWidth: 200,
    },
    {
      field: "actions",           
      headerName: "Actions",
      flex: 0.8,
      minWidth: 100,
      align: "center",
      headerAlign: "center",
      sortable: false,
      renderCell: (params) => (
        <Tooltip title="Cancel Request">
          {/* Button is disabled if the request is not PENDING */}
          <Box>
            <IconButton
              onClick={() => handleOpenModal(params.row)}
              color="error"
              disabled={params.row.requestStatus !== "PENDING"}
            >
              <CancelIcon />
            </IconButton>
          </Box>
        </Tooltip>
      ),
    },
  ];


  return (
    <Box sx={{ p: 3, width: "100%" }}>
      <Typography
        variant="h4"
        gutterBottom
        sx={{ fontWeight: "bold", color: "primary.main", mb: 3 }}
      >
        My MOC Requests
      </Typography>

      <DataGrid
        rows={rows}
        columns={columns}
        loading={loading}
        getRowId={(row) => row.id} 
        disableRowSelectionOnClick
        pageSizeOptions={[10, 25, 50]}
        initialState={{
          pagination: {
            paginationModel: { page: 0, pageSize: 10 },
          },
          sorting: { sortModel: [{ field: "requestDate", sort: "desc" }] },
        }}
        slots={{
          noRowsOverlay: CustomNoRowsOverlay,
        }}
        sx={{
          height: 600,
          width: "100%",
          "& .MuiDataGrid-cell:focus": { outline: "none" },
        }}
      />

      {/* --- Cancel Confirmation Modal --- */}
      <Dialog
        open={isModalOpen}
        onClose={handleCloseModal}
        fullWidth
        maxWidth="sm"
      >
        <DialogTitle sx={{ display: "flex", alignItems: "center" }}>
          <Ballot sx={{ mr: 1, color: "warning.main" }} />
          Confirm Cancellation
        </DialogTitle>
        <DialogContent sx={{ pt: 3 }}>
          <DialogContentText>
            Are you sure you want to cancel and delete this request?
            <br />
            <br />
            <strong>Batch ID:</strong> {selectedRequest?.batchId}
            <br />
            <strong>Journal ID:</strong> {selectedRequest?.journalId}
            <br />
            <br />
            This action cannot be undone.
          </DialogContentText>
        </DialogContent>
        <DialogActions sx={{ p: 2 }}>
          <Button onClick={handleCloseModal} disabled={isSubmitting}>
            Back
          </Button>
          <Button
            onClick={handleConfirmCancel}
            variant="contained"
            color="error"
            disabled={isSubmitting}
          >
            {isSubmitting ? (
              <CircularProgress size={24} color="inherit" />
            ) : (
              "Yes, Cancel Request"
            )}
          </Button>
        </DialogActions>
      </Dialog>
    </Box>
  );
}
