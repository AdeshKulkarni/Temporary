import { Box, Chip } from "@mui/material";
import { DataGrid } from "@mui/x-data-grid";
import React, { useEffect, useMemo, useState } from "react";

import useApi from "../../hooks/useApi";
import CheckCircleIcon from "@mui/icons-material/CheckCircle";
import BlockFlippedIcon from "@mui/icons-material/BlockFlipped";
import DateRangeIcon from "@mui/icons-material/DateRange";
import PendingActionsIcon from "@mui/icons-material/PendingActions";
import { currencyFormatterWith4Decimals } from "../../utils/NumberFormat";
import {
  StyledDateChip,
  StyledStatusChip,
  StyledStatusBox,
  StyledRequestDataGrid,
  StyledRequestContainerBox,
} from "./CurrencyRateChangeStyles";

export default function MyCurrencyRequestsTab() {
  const [rows, setRows] = useState([]);
  const { callApi } = useApi();

  const fetchData = async () => {
    const payload = { requestType: "CURRENCY_RATE_CHANGE" };
    const response = await callApi("/CR/my-requests", payload, "POST");
    if (response) {
      const result = response.data?.map((row) => {
        let parsedPayload = {};
        parsedPayload = row.payload ? row.payload : {};
        parsedPayload = {
          ...parsedPayload,
          id: row.id,
          reqStatus: row.reqStatus,
          reqDate: row.reqDate,
        };
        return { ...parsedPayload };
      });
      setRows(result);
    } else {
      setRows([]);
    }
  };

  useEffect(() => {
    fetchData();
  }, []);

  const columns = useMemo(
    () => [
      {
        field: "currencyCode",
        headerName: "Currency Code",
        flex: 1,
        sortable: true,
      },
      {
        field: "currencyRate",
        headerName: "Currency Rate",
        flex: 1,
        sortable: true,
        align: "right",
        headerAlign: "right",
        renderCell: (params) => currencyFormatterWith4Decimals("INR").format(params.value),
      },
      {
        field: "reqDate",
        headerName: "Date of Submission",
        flex: 1.5,
        sortable: true,
        align: "center",
        headerAlign: "center",
        renderCell: (params) => (
          <Box>
            <StyledDateChip
              icon={<DateRangeIcon />}
              label={params.value.split("T")[0]}
            />
          </Box>
        ),
      },
      {
        field: "reqStatus",
        headerName: "Status",
        flex: 0.7,
        sortable: true,
        align: "center",
        headerAlign: "center",
        justifyContent: "center",
        renderCell: (params) => {
          const status = params.value;
          let color = "";
          let label = "";
          let IconComponent = null;

          if (status === "REJECTED") {
            color = "error";
            label = "Rejected";
            IconComponent = BlockFlippedIcon;
          } else if (status === "ACCEPTED") {
            color = "success";
            label = "Approved";
            IconComponent = CheckCircleIcon;
          } else {
            color = "warning";
            label = "Pending";
            IconComponent = PendingActionsIcon;
          }

          return (
            <StyledStatusBox color={color}>
              <StyledStatusChip
                // Avoid passing 'color' if it causes conflicts with standard Chip color props
                // or ensure StyledStatusChip handles it safely
                icon={IconComponent ? <IconComponent /> : undefined}
                label={label}
                sx={{
                  color: `${color}.dark`,
                  borderColor: `${color}.main`,
                  "& .MuiChip-icon": { color: `${color}.dark` }
                }}
              />
            </StyledStatusBox>
          );
        },
      },
    ],
    []
  );

  return (
    <StyledRequestContainerBox>
      <StyledRequestDataGrid
        rows={rows}
        columns={columns}
        disableRowSelectionOnClick
        pageSizeOptions={[5, 10, 25, 50]}
        initialState={{
          pagination: { paginationModel: { pageSize: 10, page: 0 } },
          sorting: { sortModel: [{ field: "currencyCode", sort: "asc" }] },
        }}
        getRowId={(row) => row.id}
      />
    </StyledRequestContainerBox>
  );
}
