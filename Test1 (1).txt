import React, { useState, useRef } from "react";
import {
  Box,
  Typography,
  Paper,
  Grid,
  TextField,
  Button,
  Alert,
  CircularProgress,
  Chip,
  Divider,
} from "@mui/material";
import {
  CloudUpload as CloudUploadIcon,
  RestartAlt as RestartAltIcon,
  Description as DescriptionIcon,
  CheckCircle as CheckCircleIcon,
  Error as ErrorIcon,
} from "@mui/icons-material";

// Import styles
import {
  mainContainerStyle,
  uploadBoxStyle,
  controlPanelStyle,
} from "./JournalBulkStyle";

const JournalBulkUpload = () => {
  const [selectedDate, setSelectedDate] = useState("");
  const [file, setFile] = useState(null);
  const [loading, setLoading] = useState(false);
  const [error, setError] = useState("");
  const [successMsg, setSuccessMsg] = useState("");
  const [errorCount, setErrorCount] = useState(0);

  const fileInputRef = useRef(null);

  // --- Configuration ---
  // Adjust this URL based on your API Gateway or local setup.
  // Based on your previous code, it seems you use 'JS' as a prefix.
//   const API_BASE_URL = "http://localhost:8080/JS"; 

  const handleDateChange = (e) => {
    setSelectedDate(e.target.value);
    setSuccessMsg("");
    setError("");
    setErrorCount(0);
  };

  const handleFileChange = (e) => {
    const selectedFile = e.target.files[0];
    if (selectedFile) {
      const fileType = selectedFile.name.split(".").pop().toLowerCase();
      if (fileType !== "xlsx" && fileType !== "xls") {
        setError("Invalid file type. Please upload .xlsx or .xls");
        setFile(null);
        return;
      }
      setFile(selectedFile);
      setError("");
      setSuccessMsg("");
      setErrorCount(0);
    }
  };

  const handleUploadBoxClick = () => {
    fileInputRef.current.click();
  };

  const resetForm = () => {
    setFile(null);
    setSelectedDate("");
    setError("");
    setSuccessMsg("");
    setErrorCount(0);
    if (fileInputRef.current) fileInputRef.current.value = "";
  };

  const handleValidation = async () => {
    if (!selectedDate || !file) {
      setError("Please select a Date and upload a File.");
      return;
    }

    setLoading(true);
    setError("");
    setSuccessMsg("");
    setErrorCount(0);

    const formData = new FormData();
    formData.append("file", file);
    // If backend needs the date later, uncomment this:
    // formData.append("postingDate", selectedDate);

    try {
      // retrieve token from storage (Adjust key 'token' or 'persist:auth' as needed)
      // If you store token in session storage, change this line.
      const token = sessionStorage.getItem("token") || localStorage.getItem("token") || "";

      // We use native fetch here to handle Blob (File Download) responses easily
      const response = await fetch(`/journals/bulk-validate`, {
        method: "POST",
        headers: {
            "Authorization": token.startsWith("Bearer") ? token : `Bearer ${token}`
            // NOTE: Do NOT set Content-Type header for FormData, browser does it automatically
        },
        body: formData,
      });

      const contentType = response.headers.get("content-type");

      if (!response.ok) {
        // --- CASE 1: Validation Failed (Download Error Excel) ---
        if (contentType && contentType.includes("application/octet-stream")) {
          const count = response.headers.get("X-Error-Count");
          setErrorCount(count || "Multiple");
          setError("Validation Failed. Downloading error report...");

          // Convert response to Blob
          const blob = await response.blob();
          
          // Create a temporary link to trigger download
          const url = window.URL.createObjectURL(blob);
          const link = document.createElement("a");
          link.href = url;
          link.setAttribute("download", `Error_Report_${new Date().getTime()}.xlsx`);
          document.body.appendChild(link);
          link.click();
          
          // Cleanup
          link.parentNode.removeChild(link);
          window.URL.revokeObjectURL(url);
        } 
        // --- CASE 2: Standard JSON Error (e.g., 500 Server Error) ---
        else {
          const errData = await response.json().catch(() => ({})); // Prevent crash if body empty
          setError(errData.message || `Server Error: ${response.status}`);
        }
      } else {
        // --- CASE 3: Success ---
        const data = await response.json();
        setSuccessMsg(data.message || "File validated successfully!");
      }

    } catch (err) {
      console.error(err);
      setError("Unable to connect to the server. Please check your connection.");
    } finally {
      setLoading(false);
    }
  };

  return (
    <Box sx={mainContainerStyle}>
      <Paper elevation={3} sx={controlPanelStyle}>
        <Grid container spacing={3}>
          {/* 1. Date Picker */}
          <Grid item xs={12} md={4}>
            <Typography variant="subtitle2" gutterBottom>
              1. Select Posting Date
            </Typography>
            <TextField
              type="date"
              fullWidth
              size="small"
              value={selectedDate}
              onChange={handleDateChange}
              InputLabelProps={{ shrink: true }}
            />
          </Grid>

          {/* 2. File Upload */}
          <Grid item xs={12} md={4}>
            <Typography variant="subtitle2" gutterBottom>
              2. Upload Excel File
            </Typography>
            <Box sx={uploadBoxStyle} onClick={handleUploadBoxClick}>
              <input
                type="file"
                hidden
                accept=".xlsx, .xls"
                onChange={handleFileChange}
                ref={fileInputRef}
              />
              {file ? (
                <>
                  <DescriptionIcon color="success" fontSize="large" />
                  <Typography variant="body2" sx={{ mt: 1, fontWeight: "bold" }}>
                    {file.name}
                  </Typography>
                  <Typography variant="caption" color="textSecondary">
                    Click to change
                  </Typography>
                </>
              ) : (
                <>
                  <CloudUploadIcon color="action" fontSize="large" />
                  <Typography variant="body2" sx={{ mt: 1 }}>
                    Click to Upload Excel
                  </Typography>
                  <Typography variant="caption" color="textSecondary">
                    .xlsx or .xls
                  </Typography>
                </>
              )}
            </Box>
          </Grid>

          {/* 3. Actions */}
          <Grid
            item
            xs={12}
            md={4}
            sx={{
              display: "flex",
              flexDirection: "column",
              justifyContent: "center",
              gap: 2,
            }}
          >
            <Button
              variant="contained"
              color="primary"
              onClick={handleValidation}
              disabled={loading || !file || !selectedDate}
              startIcon={
                loading ? <CircularProgress size={20} color="inherit" /> : <CheckCircleIcon />
              }
              fullWidth
            >
              {loading ? "Validating..." : "Validate & Process"}
            </Button>

            <Button
              variant="outlined"
              color="error"
              onClick={resetForm}
              startIcon={<RestartAltIcon />}
              fullWidth
            >
              Reset
            </Button>
          </Grid>
        </Grid>

        <Divider sx={{ my: 3 }} />

        {/* Feedback Section */}
        <Box>
          {loading && (
            <Alert severity="info" icon={<CircularProgress size={20} />}>
              Uploading and validating file on server...
            </Alert>
          )}

          {error && (
            <Alert 
                severity="error" 
                sx={{ mt: 1 }}
                action={
                    errorCount > 0 && (
                        <Chip 
                            icon={<ErrorIcon sx={{color: 'white !important'}}/>} 
                            label={`${errorCount} Errors`} 
                            color="error" 
                            variant="filled"
                        />
                    )
                }
            >
              <Typography variant="subtitle2" fontWeight="bold">
                Validation Failed
              </Typography>
              <Typography variant="body2">
                {error}
              </Typography>
              {errorCount > 0 && (
                <Typography variant="caption" display="block" sx={{mt: 1}}>
                    <strong>Action Required:</strong> An Error Report file has been downloaded. Open it to see specific issues (highlighted in the last column), correct them, and upload again.
                </Typography>
              )}
            </Alert>
          )}

          {successMsg && (
            <Alert severity="success" sx={{ mt: 1 }}>
              <Typography variant="subtitle2" fontWeight="bold">Success!</Typography>
              {successMsg}
            </Alert>
          )}
        </Box>
      </Paper>
      
      {/* Rules Section */}
      <Box sx={{ mt: 2, p: 2, bgcolor: "#f5f5f5", borderRadius: 1 }}>
        <Typography variant="subtitle2" gutterBottom>
          Validation Rules
        </Typography>
        <ul style={{ margin: 0, paddingLeft: "20px", fontSize: "0.875rem", color: "#666" }}>
            <li><strong>Branch, Currency, CGL:</strong> Must exist in the System Master tables.</li>
            <li><strong>Amount:</strong> Cannot be zero.</li>
            <li><strong>Balancing:</strong> Total Debits must equal Total Credits for every Branch.</li>
            <li><strong>Remarks:</strong> Maximum 30 characters allowed.</li>
        </ul>
      </Box>
    </Box>
  );
};

export default JournalBulkUpload;
