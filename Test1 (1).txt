    @Override
    @Transactional
    public String createBulkBatchRequest(BatchRequestDto batchDto, String creatorId, Integer creatorRole) throws JsonProcessingException {
        long startTime = System.currentTimeMillis();

        // 1. Get Batch ID (One DB Call)
        String batchId = sequenceService.getNextBatchId();
        
        LocalDate postingDate = LocalDate.now();
        String commonRemarks = (batchDto.getCommonBatchRemarks() != null && !batchDto.getCommonBatchRemarks().isEmpty())
                ? batchDto.getCommonBatchRemarks()
                : "Bulk Upload Batch " + batchId;

        // 2. Group by Branch+Currency (Group = 1 Journal ID)
        Map<String, List<BatchRequestDto.JournalRequestRow>> groupedRows = batchDto.getRows().stream()
                .collect(Collectors.groupingBy(row -> row.getBranch() + ":" + row.getCurrency()));

        int totalJournals = groupedRows.size();
        log.info("Processing {} rows into {} journals for Batch {}", batchDto.getRows().size(), totalJournals, batchId);

        // 3. Bulk Fetch Journal Sequences (One DB Call)
        List<String> journalPrefixes = fetchJournalSequences(totalJournals);
        Iterator<String> prefixIterator = journalPrefixes.iterator();
        
        DateTimeFormatter dateFmt = DateTimeFormatter.ISO_DATE;
        
        String sql = "INSERT INTO JOURNAL_REQUEST " +
                     "(REQ_ID, REQ_STATUS, CHANGE_TYPE, REQ_DATE, CREATOR_ID, CREATOR_ROLE, BATCH_ID, JOURNAL_ID, COMMON_BATCH_REMARKS, PAYLOAD) " +
                     "VALUES (JOURNAL_REQUEST_SEQ.nextval, ?, ?, ?, ?, ?, ?, ?, ?, ?)";

        // 4. CHUNKING LOGIC: Flush to DB every 1000 rows to save Memory
        final int INSERT_BATCH_SIZE = 1000;
        List<Object[]> batchArgs = new ArrayList<>(INSERT_BATCH_SIZE);
        int processedCount = 0;

        for (Map.Entry<String, List<BatchRequestDto.JournalRequestRow>> entry : groupedRows.entrySet()) {
            String journalIdPrefix = prefixIterator.next();
            int journalSequence = 1;

            for (BatchRequestDto.JournalRequestRow rowDto : entry.getValue()) {
                String journalId = String.format("%s-%03d", journalIdPrefix, journalSequence);
                
                // Manual JSON builder (No Jackson overhead)
                String payload = buildJsonPayloadFast(rowDto, postingDate, batchId, journalId, commonRemarks, journalSequence, dateFmt);

                batchArgs.add(new Object[]{
                    "PENDING", rowDto.getChangeType().name(), LocalDateTime.now(), creatorId, creatorRole,
                    batchId, journalId, commonRemarks, payload
                });
                
                // FLUSH CONDITION
                if (batchArgs.size() >= INSERT_BATCH_SIZE) {
                    jdbcTemplate.batchUpdate(sql, batchArgs);
                    batchArgs.clear(); // Free up memory immediately
                }
                
                journalSequence++;
                processedCount++;
            }
        }

        // 5. Final Flush (for remaining rows)
        if (!batchArgs.isEmpty()) {
            jdbcTemplate.batchUpdate(sql, batchArgs);
        }

        log.info("Finished inserting {} rows in {} ms", processedCount, (System.currentTimeMillis() - startTime));
        createNotification(batchId, creatorId, processedCount);

        return batchId;
    }

    private List<String> fetchJournalSequences(int count) {
        if (count == 0) return Collections.emptyList();
        String sql = "SELECT JRNL_SEQ.nextval FROM dual CONNECT BY LEVEL <= ?";
        List<Long> seqs = jdbcTemplate.queryForList(sql, Long.class, count);
        return seqs.stream().map(id -> String.format("%07d", id)).collect(Collectors.toList());
    }

    private String buildJsonPayloadFast(BatchRequestDto.JournalRequestRow row, LocalDate pDate, String batchId, String jId, String rem, int count, DateTimeFormatter fmt) {
        StringBuilder sb = new StringBuilder(400); 
        sb.append("{");
        sb.append("\"changeType\":\"").append(row.getChangeType()).append("\",");
        sb.append("\"masterJournalId\":").append(row.getMasterJournalId() == null ? "null" : row.getMasterJournalId()).append(",");
        sb.append("\"pDate\":\"").append(pDate.format(fmt)).append("\",");
        sb.append("\"branch\":\"").append(row.getBranch()).append("\",");
        sb.append("\"currency\":\"").append(row.getCurrency()).append("\",");
        sb.append("\"cgl\":\"").append(row.getCgl()).append("\",");
        sb.append("\"amount\":").append(row.getAmount()).append(","); 
        sb.append("\"productType\":").append(row.getProductType() == null ? "null" : "\"" + row.getProductType() + "\"").append(",");
        sb.append("\"remarks\":").append(row.getRemarks() == null ? "null" : "\"" + escapeJson(row.getRemarks()) + "\"").append(",");
        sb.append("\"arFlag\":\"").append(row.getArFlag() == null ? "A" : row.getArFlag()).append("\",");
        sb.append("\"acClassification\":\"").append(row.getAcClassification() == null ? "A" : row.getAcClassification()).append("\",");
        sb.append("\"batchId\":\"").append(batchId).append("\",");
        sb.append("\"journalId\":\"").append(jId).append("\",");
        sb.append("\"commonBatchRemarks\":\"").append(escapeJson(rem)).append("\",");
        sb.append("\"transactionCount\":").append(count);
        sb.append("}");
        return sb.toString();
    }
    
    private String escapeJson(String s) {
        if(s == null) return "";
        return s.replace("\"", "\\\"").replace("\\", "\\\\");
    }

    @Override
    @Transactional
    public List<JournalRequest> processBulkRequests(BulkProcessJournalRequestDto dto, String executorId, Integer executorRole) throws JsonProcessingException {
        List<JournalRequest> requestsToProcess = StringUtils.hasText(dto.getBatchId()) 
             ? journalRequestRepository.findByBatchIdAndRequestStatus(dto.getBatchId(), RequestStatus.PENDING)
             : journalRequestRepository.findAll((root, query, cb) -> null); // Stub for brevity

        if (requestsToProcess.isEmpty()) return Collections.emptyList();

        List<GlTransaction> transactionsToSave = new ArrayList<>();
        List<JournalRequest> processedRequests = new ArrayList<>();

        for (JournalRequest request : requestsToProcess) {
            if (executorId.equals(request.getCreatorId()) &&
                    (executorRole != null && executorRole.equals(request.getCreatorRole()))) {
                continue;
            }

            request.setExecutorId(executorId);
            request.setExecutorRemarks(dto.getRemarks());
            request.setExecutionDate(LocalDateTime.now());
            request.setRequestStatus(dto.getStatus());

            if (RequestStatus.ACCEPTED.equals(dto.getStatus())) {
                CreateJournalRequestDto payloadDto = objectMapper.readValue(request.getPayload(), CreateJournalRequestDto.class);
                if (payloadDto.getChangeType() == ChangeType.ADD) {
                    GlTransaction newGlTxn = new GlTransaction();
                    newGlTxn.setBatchId(payloadDto.getBatchId());
                    newGlTxn.setJournalId(payloadDto.getJournalId());
                    newGlTxn.setPostDate(LocalDateTime.now());
                    newGlTxn.setTransactionDate(payloadDto.getPDate() != null ? payloadDto.getPDate() : LocalDate.now());
                    newGlTxn.setBranchCode(payloadDto.getBranch());
                    newGlTxn.setCurrency(payloadDto.getCurrency());
                    newGlTxn.setCgl(payloadDto.getCgl());
                    newGlTxn.setNarration(payloadDto.getRemarks());

                    BigDecimal amount = payloadDto.getAmount(); 
                    if (amount.signum() < 0) {
                        newGlTxn.setCreditAmount(amount.abs());
                        newGlTxn.setDebitAmount(BigDecimal.ZERO);
                    } else {
                        newGlTxn.setDebitAmount(amount);
                        newGlTxn.setCreditAmount(BigDecimal.ZERO);
                    }
                    
                    newGlTxn.setTransactionCount(payloadDto.getTransactionCount());
                    newGlTxn.setSourceFlag("J");
                    transactionsToSave.add(newGlTxn);
                }
            }
            processedRequests.add(request);
        }

        if (!transactionsToSave.isEmpty()) {
            glTransactionRepository.saveAll(transactionsToSave);
        }
        return journalRequestRepository.saveAll(processedRequests);
    }

        return (
          <Stack direction="row" spacing={1}>
            <IconButton onClick={() => handleOpenViewModal(p.row.batchId, p.row.creatorId)} color="primary">
              <VisibilityIcon />
            </IconButton>
            <IconButton onClick={() => handleOpenModal(p.row, "ACCEPTED")} color="success" disabled={isSelf}>
              <CheckIcon />
            </IconButton>
            <IconButton onClick={() => handleOpenModal(p.row, "REJECTED")} color="error" disabled={isSelf}>
              <ClearIcon />
            </IconButton>
          </Stack>
        );
      }
    }
  ];

  return (
    <Box sx={styles.mainContainer}>
      <DataGrid
        rows={rows}
        columns={columns}
        loading={loading}
        getRowId={(r) => r.batchId}
        sx={styles.dataGridContainer}
        disableRowSelectionOnClick
      />

      {/* --- Action Dialog --- */}
      <Dialog open={isModalOpen} onClose={handleCloseModal} fullWidth maxWidth="sm">
        <DialogTitle>{modalMode === "ACCEPTED" ? "Approve" : "Reject"} Batch</DialogTitle>
        <DialogContent>
          <DialogContentText sx={{ mb: 2 }}>
            Confirm {modalMode === "ACCEPTED" ? "Approval" : "Rejection"} for Batch <strong>{selectedBatch?.batchId}</strong>?
          </DialogContentText>
          <TextField
            autoFocus={modalMode === "REJECTED"}
            label="Approver Remarks"
            fullWidth
            variant="outlined"
            required={modalMode === "REJECTED"}
            value={executorRemarks}
            onChange={(e) => setExecutorRemarks(e.target.value)}
          />
        </DialogContent>
        <DialogActions sx={styles.dialogActions}>
          <Button onClick={handleCloseModal} disabled={isApproveLoading || isRejectLoading}>
            Cancel
          </Button>
          <Button
            onClick={handleSubmitAction}
            variant="contained"
            color={modalMode === "ACCEPTED" ? "success" : "error"}
            disabled={isApproveLoading || isRejectLoading}
            startIcon={(isApproveLoading || isRejectLoading) ? <CircularProgress size={20} color="inherit" /> : null}
          >
            Confirm
          </Button>
        </DialogActions>
      </Dialog>

      {/* --- View Details Dialog --- */}
      <Dialog open={isViewModalOpen} onClose={() => setIsViewModalOpen(false)} fullWidth maxWidth="lg">
        <DialogTitle>Batch Details: {viewingBatchId}</DialogTitle>
        <DialogContent dividers>
          {isViewLoading ? (
            <Box sx={{ display: 'flex', justifyContent: 'center', p: 5 }}>
              <CircularProgress />
            </Box>
          ) : (
            <TableContainer component={Paper} sx={styles.tableContainer}>
              <Table size="small" stickyHeader>
                <TableHead>
                  <TableRow>
                    <TableCell sx={{ ...styles.stickyHeaderColumn, zIndex: 20, left: 0 }} padding="checkbox">
                      <Checkbox disabled />
                    </TableCell>
                    <TableCell sx={{ ...styles.stickyHeaderColumn, zIndex: 20, left: 48 }}>
                      Journal ID
                    </TableCell>
                    <TableCell sx={styles.tableHeader}>Branch</TableCell>
                    <TableCell sx={styles.tableHeader}>Currency</TableCell>
                    <TableCell sx={styles.tableHeader}>CGL</TableCell>
                    <TableCell align="right" sx={styles.tableHeader}>Amount</TableCell>
                    <TableCell align="center" sx={styles.tableHeader}>Type</TableCell>
                  </TableRow>
                </TableHead>
                <TableBody>{renderDetailRows()}</TableBody>
              </Table>
            </TableContainer>
          )}
        </DialogContent>
        <DialogActions sx={styles.dialogActions}>
          <Button onClick={() => setIsViewModalOpen(false)}>Close</Button>
        </DialogActions>
      </Dialog>
    </Box>
  );
}
