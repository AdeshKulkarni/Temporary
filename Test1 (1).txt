class Solution {

    public long Helper(int[] nums, int index, int isEven, long[][] memo) {
        if (index == nums.length) {
            return 0;
        }

        if (memo[index][isEven] != -1) {
            return memo[index][isEven];
        }

        long val = (isEven == 1) ? nums[index] : -nums[index];

        long ans1 = val + Helper(nums, index + 1, 1 - isEven, memo); // take -> flip sign
        long ans2 = Helper(nums, index + 1, isEven, memo);           // skip -> keep sign

        return memo[index][isEven] = Math.max(ans1, ans2);
    }

    public long maxAlternatingSum(int[] nums) {
        long[][] memo = new long[nums.length][2];
        for (long[] row : memo) {
            Arrays.fill(row, -1);
        }

        return Helper(nums, 0, 1, memo); // first picked element is at even index 0 (+)
    }
}
