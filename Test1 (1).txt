const BASE_URL_CONFIG = '/RB/template-config';
const BASE_URL_REPORT = '/RB/report-builder'; // New Base URL for generation

const handleResponse = async (response) => {
  const result = await response.json().catch(() => ({ 
    message: "Failed to parse server response", 
    status: "ERROR" 
  }));

  if (!response.ok) {
    throw new Error(result.message || `Error ${response.status}: ${response.statusText}`);
  }
  return result;
};

export const templateApi = {
  // ... (Existing get/delete methods remain the same) ...

  getAllTemplates: async () => {
    try {
      const response = await fetch(`${BASE_URL_CONFIG}/get-templates`, {
        method: 'GET',
        headers: { 'Content-Type': 'application/json' },
      });
      return await handleResponse(response);
    } catch (error) {
      console.error("API Error [getAllTemplates]:", error.message);
      throw error;
    }
  },

  getTemplateById: async (templateId) => {
    try {
      const response = await fetch(`${BASE_URL_CONFIG}/${templateId}`, {
        method: 'GET',
        headers: { 'Content-Type': 'application/json' },
      });
      return await handleResponse(response);
    } catch (error) {
      console.error(`API Error [getTemplateById] ID: ${templateId}:`, error.message);
      throw error;
    }
  },

  updateTemplateStatus: async (templateId, status) => {
    try {
      const response = await fetch(`${BASE_URL_CONFIG}/update-status/${templateId}`, {
        method: 'PUT',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ status: status }),
      });
      return await handleResponse(response);
    } catch (error) {
      console.error(`API Error [updateStatus] ID: ${templateId}:`, error.message);
      throw error;
    }
  },

  deleteTemplate: async (templateId) => {
    try {
      const response = await fetch(`${BASE_URL_CONFIG}/${templateId}`, {
        method: 'DELETE',
        headers: { 'Content-Type': 'application/json' },
      });
      return await handleResponse(response);
    } catch (error) {
      console.error(`API Error [deleteTemplate] ID: ${templateId}:`, error.message);
      throw error;
    }
  },

  // --- NEW: Generate Report Function ---
  generateReport: async (payload, format) => {
    // 1. Determine Endpoint based on format
    let endpoint = '';
    switch (format) {
      case 'excel': endpoint = `${BASE_URL_REPORT}/view/excel`; break;
      case 'text':  endpoint = `${BASE_URL_REPORT}/view/text`; break;
      case 'pdf':   
      default:      endpoint = `${BASE_URL_REPORT}/view/pdf`; break;
    }

    try {
      const response = await fetch(endpoint, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify(payload),
      });

      if (!response.ok) {
        const errorText = await response.text();
        throw new Error(errorText || "Failed to generate report");
      }

      // 2. Return Blob for file download
      return await response.blob();
    } catch (error) {
      console.error(`API Error [generateReport] Format: ${format}:`, error.message);
      throw error;
    }
  }
};

import React, { useState, useEffect } from 'react';
import { 
  Dialog, DialogTitle, DialogContent, DialogActions, 
  Button, MenuItem, TextField, Grid, Divider, Alert 
} from '@mui/material';
import { templateApi } from '../api/templateApi';

const TemplateViewDialog = ({ open, handleClose, template }) => {
  const [selectedVariantCode, setSelectedVariantCode] = useState('');
  const [formData, setFormData] = useState({});
  const [exportFormat, setExportFormat] = useState('pdf');
  const [error, setError] = useState(null);
  const [loading, setLoading] = useState(false);

  // Get the full variant object based on the selected code
  const activeVariant = template?.variants?.find(v => v.variantCode === selectedVariantCode);

  // Reset state when dialog opens/closes
  useEffect(() => {
    if (open) {
      setSelectedVariantCode('');
      setFormData({});
      setError(null);
      setLoading(false);
    }
  }, [open]);

  const handleInputChange = (name, value) => {
    setFormData(prev => ({ ...prev, [name]: value }));
    // Clear error if user starts typing
    if (error) setError(null);
  };

  const handleGenerate = async () => {
    if (!activeVariant) return;

    // 1. Validation Logic
    const missingParams = [];
    const payloadParams = {};

    activeVariant.params.forEach(param => {
      const value = formData[param.paramName];
      
      // Check if required and missing (null, undefined, or empty string)
      if (param.required && (value === undefined || value === null || value === '')) {
        missingParams.push(param.label || param.paramName);
      }
      
      // Only add to payload if value exists
      if (value !== undefined && value !== '') {
        payloadParams[param.paramName] = value;
      }
    });

    if (missingParams.length > 0) {
      setError(`Please fill required fields: ${missingParams.join(', ')}`);
      return;
    }

    // 2. Construct Payload
    const payload = {
      templateId: template.templateId,
      variantCode: activeVariant.variantCode,
      params: payloadParams
    };

    setLoading(true);
    setError(null);

    try {
      // 3. Call API
      const blob = await templateApi.generateReport(payload, exportFormat);

      // 4. Handle File Download
      const url = window.URL.createObjectURL(blob);
      const link = document.createElement('a');
      link.href = url;
      
      // Set filename based on format
      const extension = exportFormat === 'excel' ? 'xlsx' : exportFormat === 'text' ? 'txt' : 'pdf';
      link.setAttribute('download', `${template.templateName}_${activeVariant.variantName}.${extension}`);
      
      document.body.appendChild(link);
      link.click();
      link.parentNode.removeChild(link);
      
      // Optional: Close dialog on success
      // handleClose(); 
    } catch (err) {
      setError("Failed to generate report. Please try again.");
    } finally {
      setLoading(false);
    }
  };

  return (
    <Dialog open={open} onClose={handleClose} maxWidth="sm" fullWidth>
      <DialogTitle>Generate Report: {template.templateName}</DialogTitle>
      
      <DialogContent dividers>
        {error && (
            <Alert severity="error" sx={{ mb: 2 }}>{error}</Alert>
        )}

        <Grid container spacing={3}>
          {/* Variant Selection */}
          <Grid item xs={12}>
            <TextField
              select
              fullWidth
              label="Select Report Variant"
              value={selectedVariantCode}
              onChange={(e) => {
                setSelectedVariantCode(e.target.value);
                setFormData({}); // Reset params when variant changes
              }}
            >
              {template.variants && template.variants.map((v) => (
                <MenuItem key={v.variantCode} value={v.variantCode}>
                  {v.variantName}
                </MenuItem>
              ))}
            </TextField>
          </Grid>

          {/* Dynamic Inputs based on Params */}
          {activeVariant?.params.map((param) => (
            <Grid item xs={12} key={param.paramName}>
              <TextField
                fullWidth
                label={param.label || param.paramName}
                // Handle DATE type vs STRING type
                type={param.paramType === 'DATE' ? 'date' : 'text'}
                required={param.required}
                value={formData[param.paramName] || ''}
                InputLabelProps={{ shrink: true }} // Ensures label doesn't overlap date placeholder
                onChange={(e) => handleInputChange(param.paramName, e.target.value)}
                helperText={param.required ? "Required" : "Optional"}
              />
            </Grid>
          ))}

          {/* Export Format */}
          {activeVariant && (
            <Grid item xs={12}>
              <Divider sx={{ my: 1 }}>Export Options</Divider>
              <TextField 
                select 
                fullWidth 
                label="Export Format" 
                value={exportFormat}
                onChange={(e) => setExportFormat(e.target.value)}
              >
                <MenuItem value="pdf">PDF Document</MenuItem>
                <MenuItem value="excel">Excel Spreadsheet</MenuItem>
                <MenuItem value="text">Plain Text</MenuItem>
              </TextField>
            </Grid>
          )}
        </Grid>
      </DialogContent>

      <DialogActions sx={{ p: 2 }}>
        <Button onClick={handleClose} disabled={loading}>
          Cancel
        </Button>
        <Button 
          variant="contained" 
          color="primary" 
          disabled={!selectedVariantCode || loading}
          onClick={handleGenerate}
        >
          {loading ? 'Generating...' : 'Generate Report'}
        </Button>
      </DialogActions>
    </Dialog>
  );
};

export default TemplateViewDialog;
