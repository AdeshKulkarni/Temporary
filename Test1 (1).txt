import React, { useState, useEffect } from 'react';

// --- Material-UI Imports ---
import {
    Box,
    Typography,
    Fab,
    Dialog,
    DialogActions,
    DialogContent,
    DialogTitle,
    TextField,
    Button,
    Container,
    Paper
} from '@mui/material';
import AddIcon from '@mui/icons-material/Add';
import EditIcon from '@mui/icons-material/Edit';
import DeleteIcon from '@mui/icons-material/Delete';
import { DataGrid, GridActionsCellItem } from '@mui/x-data-grid';


const fetchCurrencyData = () => {
    return new Promise((resolve) => {
        const dataFromBackend = [
            { id: 1, currencyCode: 'USD', currencyName: 'United States Dollar', currencyRate: 1.00, rateDate: '2025-09-29' },
            { id: 2, currencyCode: 'EUR', currencyName: 'Euro', currencyRate: 0.92, rateDate: '2025-09-29' },
            { id: 3, currencyCode: 'INR', currencyName: 'Indian Rupee', currencyRate: 83.50, rateDate: '2025-09-29' },
            { id: 4, currencyCode: 'GBP', currencyName: 'British Pound', currencyRate: 0.79, rateDate: '2025-09-28' },
            { id: 5, currencyCode: 'JPY', currencyName: 'Japanese Yen', currencyRate: 149.82, rateDate: '2025-09-29' },
        ];
        setTimeout(() => resolve(dataFromBackend), 1500);
    });
};

// --- CHANGED: Define an initial blank state for the form ---
const initialFormState = {
    currencyCode: '',
    currencyName: '',
    currencyRate: '',
    rateDate: '',
};

function CurrencyMaster() {
    // --- State Management ---
    const [rows, setRows] = useState([]);
    const [loading, setLoading] = useState(true);
    const [isDialogOpen, setIsDialogOpen] = useState(false);
    
    // --- CHANGED: We now use a single state for form data and another for the row being edited ---
    const [formData, setFormData] = useState(initialFormState);
    const [editingRow, setEditingRow] = useState(null); // This will hold the row object when editing
    const [errors, setErrors] = useState({});

    // --- Data Fetching ---
    useEffect(() => {
        setLoading(true);
        fetchCurrencyData().then(data => {
            setRows(data);
            setLoading(false);
        }).catch(error => {
            console.error("Failed to fetch currency data:", error);
            setLoading(false);
        });
    }, []);

    // --- CHANGED: Action Handlers are now fully functional ---
    const handleEditClick = (id) => () => {
        const rowToEdit = rows.find((row) => row.id === id);
        if (rowToEdit) {
            // The date from the grid is a Date object, format it to YYYY-MM-DD for the input field
            const formattedDate = rowToEdit.rateDate ? new Date(rowToEdit.rateDate).toISOString().split('T')[0] : '';
            const formDataToEdit = { ...rowToEdit, rateDate: formattedDate };

            setEditingRow(formDataToEdit); // Set the row to be edited
            setFormData(formDataToEdit);   // Pre-populate the form
            setIsDialogOpen(true);         // Open the dialog
        }
    };

    const handleDeleteClick = (id) => () => {
        // In a real app, you'd show a confirmation dialog here
        setRows(rows.filter((row) => row.id !== id));
    };

    const columns = [
        { field: 'id', headerName: 'ID', width: 70 },
        { field: 'currencyCode', headerName: 'Currency Code', width: 120 },
        { field: 'currencyName', headerName: 'Currency Name', flex: 1, minWidth: 200 },
        { field: 'currencyRate', headerName: 'Currency Rate', type: 'number', width: 130, align: 'right', headerAlign: 'right' },
        {
            field: 'rateDate',
            headerName: 'Rate Date',
            type: 'date',
            width: 130,
            valueGetter: (params) => params.value ? new Date(params.value) : null,
            renderCell: (params) => params.value ? new Date(params.value).toLocaleDateString() : ''
        },
        {
            field: 'actions',
            type: 'actions',
            headerName: 'Actions',
            width: 100,
            cellClassName: 'actions',
            getActions: ({ id }) => [
                <GridActionsCellItem icon={<EditIcon />} label="Edit" onClick={handleEditClick(id)} color="primary" />,
                <GridActionsCellItem icon={<DeleteIcon />} label="Delete" onClick={handleDeleteClick(id)} color="error" />,
            ],
        },
    ];

    // --- CHANGED: handleOpenDialog now just prepares for adding a *new* item ---
    const handleOpenDialog = () => {
        setEditingRow(null);       // Ensure we are in "Add" mode
        setFormData(initialFormState); // Clear the form
        setIsDialogOpen(true);
    };

    // --- CHANGED: handleCloseDialog now also resets the editing state ---
    const handleCloseDialog = () => {
        setIsDialogOpen(false);
        setEditingRow(null);
        setFormData(initialFormState);
        setErrors({});
    };

    const handleInputChange = (e) => {
        const { name, value } = e.target;
        setFormData(prev => ({ ...prev, [name]: value }));
    };

    // --- CHANGED: Validation now uses the `formData` state ---
    const validateForm = () => {
        let tempErrors = {};
        if (!formData.currencyCode) tempErrors.currencyCode = "Currency Code is required.";
        if (!formData.currencyName) tempErrors.currencyName = "Currency Name is required.";
        if (!formData.currencyRate) tempErrors.currencyRate = "Rate is required.";
        else if (isNaN(formData.currencyRate)) tempErrors.currencyRate = "Rate must be a number.";
        if (!formData.rateDate) tempErrors.rateDate = "Rate Date is required.";
        setErrors(tempErrors);
        return Object.keys(tempErrors).length === 0;
    };

    // --- CHANGED: handleSubmit now handles both Add and Edit logic ---
    const handleSubmit = () => {
        if (validateForm()) {
            if (editingRow) {
                // We are in Edit mode
                console.log("Updating row:", formData);
                setRows(rows.map((row) => (row.id === editingRow.id ? formData : row)));
            } else {
                // We are in Add mode
                console.log("Adding new row:", formData);
                const newId = rows.length > 0 ? Math.max(...rows.map(r => r.id)) + 1 : 1;
                setRows([...rows, { ...formData, id: newId }]);
            }
            handleCloseDialog();
        }
    };

    return (
        <Container maxWidth="xl" sx={{ mt: 4, mb: 4, display: 'flex', justifyContent: 'center', width: '100%' }}>
            <Paper elevation={3} sx={{ p: 3, borderRadius: 2, width: '80%', maxWidth: 1200 }}>
                <Typography variant="h4" component="h1" gutterBottom sx={{ fontWeight: '600', color: 'primary.main' }}>
                    Currency Master
                </Typography>
                <Box sx={{ height: '70vh', width: '100%' }}>
                    <DataGrid rows={rows} columns={columns} loading={loading} disableRowSelectionOnClick />
                </Box>
            </Paper>

            <Fab color="primary" aria-label="add" onClick={handleOpenDialog} sx={{ position: 'fixed', bottom: 30, right: 30 }}>
                <AddIcon />
            </Fab>

            <Dialog open={isDialogOpen} onClose={handleCloseDialog} fullWidth maxWidth="sm">
                {/* --- CHANGED: Dialog title is now dynamic --- */}
                <DialogTitle sx={{ fontWeight: 'bold' }}>
                    {editingRow ? 'Edit Currency' : 'Add New Currency'}
                </DialogTitle>
                <DialogContent>
                    {/* --- CHANGED: All TextFields now use the `formData` state --- */}
                    <TextField autoFocus margin="dense" name="currencyCode" label="Currency Code" type="text" fullWidth variant="outlined" value={formData.currencyCode} onChange={handleInputChange} error={!!errors.currencyCode} helperText={errors.currencyCode} />
                    <TextField margin="dense" name="currencyName" label="Currency Name" type="text" fullWidth variant="outlined" value={formData.currencyName} onChange={handleInputChange} error={!!errors.currencyName} helperText={errors.currencyName} />
                    <TextField margin="dense" name="currencyRate" label="Currency Rate" type="number" fullWidth variant="outlined" value={formData.currencyRate} onChange={handleInputChange} error={!!errors.currencyRate} helperText={errors.currencyRate} />
                    <TextField margin="dense" name="rateDate" label="Rate Date" type="date" fullWidth variant="outlined" value={formData.rateDate} onChange={handleInputChange} error={!!errors.rateDate} helperText={errors.rateDate} InputLabelProps={{ shrink: true }} />
                </DialogContent>
                <DialogActions sx={{ p: '16px 24px' }}>
                    <Button onClick={handleCloseDialog}>Cancel</Button>
                    <Button onClick={handleSubmit} variant="contained">Submit</Button>
                </DialogActions>
            </Dialog>
        </Container>
    );
}

export default CurrencyMaster;
