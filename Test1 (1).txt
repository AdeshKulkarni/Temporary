import { Box, Grid, Typography, keyframes } from '@mui/material';

// Define the keyframes for a smooth, continuous rotation
const smoothRotate = keyframes`
  from {
    transform: rotate(0deg);
  }
  to {
    transform: rotate(360deg);
  }
`;

function AnimatedBorderBox() {
  return (
    <Grid item xs={12} sm={5} md={4}>
      <Box
        sx={{
          position: 'relative',
          borderRadius: 1, // Matches the inner content's border-radius
          p: 1, // The padding creates space for the border to appear
          overflow: 'hidden',

          // The pseudo-element that creates the animated gradient border
          '&::before': {
            content: '""',
            position: 'absolute',
            inset: -20, // Make it larger than the box to ensure the gradient covers the edges
            zIndex: 0,
            
            // --- MODIFICATION START ---
            // A softer, more elegant gradient. It uses the theme's primary color
            // and fades to transparent, creating a "spotlight" effect.
            background: (theme) => `conic-gradient(
              from 180deg at 50% 50%,
              transparent 0%,
              ${theme.palette.primary.main} 30%, 
              transparent 60%
            )`,
            
            // A single, continuous, and slower animation for a smooth, looping effect.
            animation: `${smoothRotate} 4s linear infinite`,
            // --- MODIFICATION END ---
          },
          
          // The pseudo-element that creates the "cutout" for the border
          '&::after': {
            content: '""',
            position: 'absolute',
            inset: '1px', // This determines the border thickness
            background: (theme) => theme.palette.background.paper, // Must match the inner content background
            borderRadius: 'inherit', // Inherits the parent's border-radius
            zIndex: 1,
          },
        }}
      >
        {/* The actual content sits on top with a higher z-index */}
        <Box
          className="inner-content"
          sx={{
            position: 'relative',
            zIndex: 2, // Sits on top of the ::after element
            borderRadius: 'inherit',
            display: 'flex',
            alignItems: 'center',
            p: 1, // Match the outer padding for alignment
            backgroundColor: 'background.paper', // Ensures content background is solid
          }}
        >
          <Typography variant="body1" color="text.secondary">
            Please Note: All fields are required for successful user creation.
          </Typography>
        </Box>
      </Box>
    </Grid>
  );
}

export default AnimatedBorderBox;
