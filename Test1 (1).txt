class Solution {
    int maxDiff = 0; // Global Record Book

    public int maxAncestorDiff(TreeNode root) {
        // Kick off the helper, passing the root's value as both initial min and max
        helper(root, root.val, root.val);
        return maxDiff;
    }

    private void helper(TreeNode node, int currentMin, int currentMax) {
        if (node == null) {
            return;
        }

        // 1. Check the difference between current node and the ancestor extremes
        int diffFromMin = Math.abs(node.val - currentMin);
        int diffFromMax = Math.abs(node.val - currentMax);
        maxDiff = Math.max(maxDiff, Math.max(diffFromMin, diffFromMax));

        // 2. Update the min and max for the path ahead
        currentMin = Math.min(currentMin, node.val);
        currentMax = Math.max(currentMax, node.val);

        // 3. Pass the updated backpack down to children (Top-Down Pre-order)
        helper(node.left, currentMin, currentMax);
        helper(node.right, currentMin, currentMax);
    }
}
