export default function ReportTemplatesScreen() {
  // Hooks
  const { callApi } = useApi();
  const showSnackBar = useCustomSnackbar();
  const isMounted = useRef(true);
  
  // State
  const [loading, setLoading] = useState(true);
  const [rows, setRows] = useState([]);
  
  // Dialog State
  const [isViewDialogOpen, setIsViewDialogOpen] = useState(false);
  const [selectedTemplate, setSelectedTemplate] = useState(null);

  // Cleanup on unmount
  useEffect(() => {
    isMounted.current = true;
    return () => { isMounted.current = false; };
  }, []);

  // 1. Fetch Data
  const fetchTemplates = useCallback(async () => {
    setLoading(true);
    try {
      const response = await callApi(
        "/template-config/get-templates",
        null,
        "GET"
      );

      if (isMounted.current) {
        // Handling { message: "...", data: [...] } structure
        const dataList = response && Array.isArray(response.data) 
          ? response.data 
          : [];
        setRows(dataList);
      }
    } catch (err) {
      if (isMounted.current && err.name !== "CanceledError") {
        showSnackBar("Failed to load templates", "error");
      }
    } finally {
      if (isMounted.current) setLoading(false);
    }
  }, [callApi, showSnackBar]);

  // Initial Load
  useEffect(() => {
    fetchTemplates();
  }, [fetchTemplates]);

  // 2. Handle View (Open Dialog)
  const handleView = (row) => {
    setSelectedTemplate(row);
    setIsViewDialogOpen(true);
  };

  const handleCloseDialog = () => {
    setIsViewDialogOpen(false);
    setSelectedTemplate(null);
  };

  // 3. Handle Status Change (Dummy API Call)
  const handleStatusChange = async (row) => {
    // Determine new status for toggling
    const newStatus = row.status === 'ACTIVE' ? 'INACTIVE' : 'ACTIVE';
    
    // Optimistic Update (Optional: remove if you only want to update after success)
    const originalRows = [...rows];
    // setRows(prev => prev.map(r => r.templateId === row.templateId ? {...r, status: newStatus} : r));

    try {
      // DUMMY API Call

      // Simulating API delay for 1 second
      await new Promise(resolve => setTimeout(resolve, 800)); 

      if(isMounted.current) {
         showSnackBar(`Status updated to ${newStatus}`, "success");
         // Refresh list to get actual server state
         fetchTemplates();
      }

    } catch (error) {
       if(isMounted.current) {
          showSnackBar("Failed to update status", "error");
          // Revert if using optimistic update
          setRows(originalRows);
       }
    }
  };

  const columns = [

      minWidth: 180,
    },

      renderCell: (params) => (

                sx={{ minWidth: 80, fontWeight: 'bold' }}

This file handles the variant selection. I have ensured useState is imported correctly.
import React, { useState, useEffect } from 'react';
import { 
  Dialog, DialogTitle, DialogContent, DialogActions, 
  Button, MenuItem, TextField, Divider, Grid, Typography 
} from '@mui/material';

const TemplateViewDialog = ({ open, handleClose, template }) => {
  // State definitions
  const [selectedVariant, setSelectedVariant] = useState('');
  const [formData, setFormData] = useState({});

  // Reset state when the modal opens with a new template
  useEffect(() => {
    if (open && template) {
        setSelectedVariant('');
        setFormData({});
    }
  }, [open, template]);

  // Safety check: if no template data, return null
  if (!template) return null;

  // Find the variant object based on user selection
  const activeVariant = template.variants?.find(v => v.variantCode === selectedVariant);

  const handleInputChange = (name, value) => {
    setFormData(prev => ({ ...prev, [name]: value }));
  };

  const handleGenerate = () => {
    // Generate Logic here
    console.log("Generating Report for:", {
        template: template.templateName,
        variant: selectedVariant,
        data: formData
    });
    handleClose();
  };

  return (
    <Dialog open={open} onClose={handleClose} maxWidth="sm" fullWidth>
      <DialogTitle>View Report: {template.templateName}</DialogTitle>
      <DialogContent dividers>
        <Grid container spacing={3}>
          {/* Variant Selection */}
          <Grid item xs={12}>
            <TextField
              select
              fullWidth
              label="Select Report Variant"
              value={selectedVariant}
              onChange={(e) => setSelectedVariant(e.target.value)}
              helperText={template.variants?.length === 0 ? "No variants available" : ""}
            >
              {template.variants?.map((v) => (
                <MenuItem key={v.variantCode} value={v.variantCode}>
                  {v.variantName}
                </MenuItem>
              ))}
            </TextField>
            {activeVariant && (
                <Typography variant="caption" color="text.secondary" sx={{ mt: 1, display: 'block' }}>
                    {activeVariant.description}
                </Typography>
            )}
          </Grid>

          {/* Dynamic Inputs based on Params */}
          {activeVariant?.params?.map((param) => (
            <Grid item xs={12} key={param.paramName}>
              <TextField
                fullWidth
                label={param.label}
                // Determine type: if DATE use date picker, else text
                type={param.paramType === 'DATE' ? 'date' : 'text'}
                required={param.required}
                // Ensure label floats for date inputs
                InputLabelProps={{ shrink: true }} 
                onChange={(e) => handleInputChange(param.paramName, e.target.value)}
              />
            </Grid>
          ))}

          {/* Export Format */}
          <Grid item xs={12}>
            <Divider sx={{ my: 1 }}>Export Options</Divider>
            <TextField select fullWidth label="Export Format" defaultValue="pdf">
              <MenuItem value="pdf">PDF Document</MenuItem>
              <MenuItem value="excel">Excel Spreadsheet</MenuItem>
              <MenuItem value="text">Plain Text</MenuItem>
            </TextField>
          </Grid>
        </Grid>
      </DialogContent>
      <DialogActions sx={{ p: 2 }}>
        <Button onClick={handleClose}>Cancel</Button>
        <Button 
            variant="contained" 
            color="primary" 
            onClick={handleGenerate}
            disabled={!selectedVariant}
        >
          Generate Report
        </Button>
      </DialogActions>
    </Dialog>
  );
};

export default TemplateViewDialog;
