import React, { useState, useEffect, useCallback } from "react";
import {
  Box,
  Typography,
  Chip,
  Dialog,
  DialogActions,
  DialogContent,
  DialogContentText,
  DialogTitle,
  Button,
  CircularProgress,
  Tooltip,
  IconButton,
  Stack,
  Grid,
  Table,
  TableBody,
  TableCell,
  TableContainer,
  TableHead,
  TableRow,
  Paper,
  Checkbox,
} from "@mui/material";
import { DataGrid } from "@mui/x-data-grid";
import {
  DeleteSweep as DeleteSweepIcon,
  Visibility as VisibilityIcon,
  Close as CloseIcon,
} from "@mui/icons-material";
import AccountCircleIcon from "@mui/icons-material/AccountCircle";
import InfoIcon from "@mui/icons-material/Info";
import DeleteIcon from "@mui/icons-material/Delete";

import useApi from "../../hooks/useApi";
import useCustomSnackbar from "../../utils/useCustomSnackbar";

import {
  cellBorderStyle,
  tableHeaderCellStyle,
  stickyColumnCell,
  stickyBoxStyle,
  dialogTitleStyle,
  loadingContainerStyle,
  dialogActionsStyle,
  dataGridContainerStyle,
  mainContainerStyle,
  tableContainerStyle,
} from "./JournalBulkStyle";

const formatCurrency = (num) => {
  return Number(num).toFixed(4);
};

const getTxnTypeColor = (txnType) => {
  return txnType === "Credit" ? "error" : "success";
};

const getStatusChipColor = (status) => {
  switch (status) {
    case "ACCEPTED":
      return "success";
    case "REJECTED":
      return "error";
    case "PENDING":
    default:
      return "warning";
  }
};

function CustomNoRowsOverlay() {
  return (
    <Stack height="100%" alignItems="center" justifyContent="center">
      <InfoIcon sx={{ fontSize: 48, color: "text.secondary", mb: 2 }} />
      <Typography variant="h6">No Journal Requests Found</Typography>
      <Typography>You have not created any Journal requests.</Typography>
    </Stack>
  );
}

export default function JournalPostingStatus() {
  const { callApi } = useApi();
  const showSnackBar = useCustomSnackbar();

  const [loading, setLoading] = useState(true);
  const [rows, setRows] = useState([]);

  const [isModalOpen, setIsModalOpen] = useState(false);
  const [isSubmitting, setIsSubmitting] = useState(false);
  const [selectedBatch, setSelectedBatch] = useState(null);

  const [isViewModalOpen, setIsViewModalOpen] = useState(false);
  const [isViewLoading, setIsViewLoading] = useState(false);

  const [batchDetails, setBatchDetails] = useState([]);
  const [viewingBatchId, setViewingBatchId] = useState(null);
  const [viewingBatchStatus, setViewingBatchStatus] = useState(null);

  const [selectedJournalPrefixes, setSelectedJournalPrefixes] = useState([]);
  const [isPartialSubmitting, setIsPartialSubmitting] = useState(false);

  const fetchMyBatches = useCallback(async () => {
    setLoading(true);
    try {
      const myRequests = await callApi("/JS/journals/my-requests", null, "GET");

      if (!Array.isArray(myRequests) || myRequests.length === 0) {
        setRows([]);
        setLoading(false);
        return;
      }

      const groupedBatches = new Map();
      for (const request of myRequests) {
        const batchId = request.batchId;

        if (!groupedBatches.has(batchId)) {
          groupedBatches.set(batchId, {
            id: batchId,
            batchId: batchId,
            creatorId: request.creatorId,
            requestStatus: request.requestStatus,
            requestDate: request.requestDate,
            commonBatchRemarks: request.commonBatchRemarks,
            executorId: request.executorId,
            executionDate: request.executionDate,
            executorRemarks: request.executorRemarks,
            totalDebit: 0,
            totalCredit: 0,
            requestCount: 0,
          });
        }

        const batchDetails = groupedBatches.get(batchId);
        batchDetails.requestCount++;

        try {
          const payload = JSON.parse(request.payload);
          const amount = parseFloat(payload.amount) || 0;
          if (amount < 0) {
            batchDetails.totalCredit += Math.abs(amount);
          } else {
            batchDetails.totalDebit += amount;
          }
        } catch (e) {
          console.error(`Failed to parse payload for request ${request.id}`, e);
        }
      }
      setRows(Array.from(groupedBatches.values()));
    } catch (err) {
      console.error("Failed to fetch user's requests:", err);
      const errorMsg =
        err.response?.data?.message || "Failed to load your requests.";
      showSnackBar(errorMsg, "error");
    } finally {
      setLoading(false);
    }
  }, [callApi, showSnackBar]);

  useEffect(() => {
    fetchMyBatches();
  }, [fetchMyBatches]);

  const handleOpenModal = (batch) => {
    setSelectedBatch(batch);
    setIsModalOpen(true);
  };

  const handleCloseModal = () => {
    if (isSubmitting) return;
    setIsModalOpen(false);
    setSelectedBatch(null);
  };

  const handleOpenViewModal = async (batchId, status) => {
    setIsViewModalOpen(true);
    setIsViewLoading(true);
    setBatchDetails([]);
    setViewingBatchId(batchId);
    setViewingBatchStatus(status);

    try {
      const data = await callApi(
        `/JS/journals/by-batch/${batchId}`,
        null,
        "GET"
      );

      setTimeout(() => {
        if (Array.isArray(data)) {
          const journalGroups = new Map();

          const processedData = data
            .map((req) => {
              try {
                const payload = JSON.parse(req.payload);
                const prefix = req.journalId.split("-")[0];
                const amount = parseFloat(payload.amount) || 0;
                return {
                  ...req,
                  payload,
                  journalPrefix: prefix,
                  amount: Math.abs(amount),
                  transactionType: amount < 0 ? "Credit" : "Debit",
                };
              } catch (e) {
                console.error("Failed to parse payload for req:", req.id, e);
                return null;
              }
            })
            .filter(Boolean);

          processedData.sort((a, b) => a.journalId.localeCompare(b.journalId));

          for (const row of processedData) {
            const prefix = row.journalPrefix;
            if (!journalGroups.has(prefix)) {
              journalGroups.set(prefix, []);
            }
            journalGroups.get(prefix).push(row);
          }

          setBatchDetails(Array.from(journalGroups.values()));
        } else {
          setBatchDetails([]);
          showSnackBar("No details found for this batch.", "info");
        }
        setIsViewLoading(false);
      }, 100);
    } catch (error) {
      console.error("Failed to fetch batch details:", error);
      const errorMsg =
        error.response?.data?.message || "Failed to load batch details.";
      showSnackBar(errorMsg, "error");
      setBatchDetails([]);
      setIsViewLoading(false);
    }
  };

  const handleCloseViewModal = () => {
    setIsViewModalOpen(false);
    setBatchDetails([]);
    setViewingBatchId(null);
    setSelectedJournalPrefixes([]);
    setIsPartialSubmitting(false);
    setViewingBatchStatus(null);
  };

  const handleSubmitBatchCancel = async () => {
    setIsSubmitting(true);
    try {
      await callApi(
        `/JS/journals/my-requests/by-batch/${selectedBatch.batchId}`,
        null,
        "DELETE"
      );
      showSnackBar(
        `Batch ${selectedBatch.batchId} successfully canceled.`,
        "success"
      );
      handleCloseModal();
      fetchMyBatches();
    } catch (error) {
      console.error("Failed to cancel batch:", error);
      const errorMsg =
        error.response?.data?.message || "Failed to cancel batch.";
      showSnackBar(errorMsg, "error");
    } finally {
      setIsSubmitting(false);
    }
  };

  const handleCancelSelectedJournals = async () => {
    if (selectedJournalPrefixes.length === 0) {
      showSnackBar(
        "Please select at least one journal group to cancel.",
        "warning"
      );
      return;
    }

    setIsPartialSubmitting(true);

    try {
      await callApi(
        "/JS/journals/my-requests/by-journal-list",
        selectedJournalPrefixes,
        "DELETE"
      );
      showSnackBar(
        "Selected journal entries successfully canceled.",
        "success"
      );
      handleCloseViewModal();
      fetchMyBatches();
    } catch (error) {
      console.error("Failed to process partial batch:", error);
      const errorMsg =
        error.response?.data?.message || "Failed to cancel selected entries.";
      showSnackBar(errorMsg, "error");
    } finally {
      setIsPartialSubmitting(false);
    }
  };

  const handleSelectAllClick = (event) => {
    if (event.target.checked) {
      const newSelecteds = batchDetails.map((group) => group[0].journalPrefix);
      setSelectedJournalPrefixes(newSelecteds);
      return;
    }
    setSelectedJournalPrefixes([]);
  };

  const handleRowCheckboxClick = (event, prefix) => {
    if (event) {
      event.stopPropagation();
    }

    const selectedIndex = selectedJournalPrefixes.indexOf(prefix);
    let newSelected = [];

    if (selectedIndex === -1) {
      newSelected = newSelected.concat(selectedJournalPrefixes, prefix);
    } else if (selectedIndex === 0) {
      newSelected = newSelected.concat(selectedJournalPrefixes.slice(1));
    } else if (selectedIndex === selectedJournalPrefixes.length - 1) {
      newSelected = newSelected.concat(selectedJournalPrefixes.slice(0, -1));
    } else if (selectedIndex > 0) {
      newSelected = newSelected.concat(
        selectedJournalPrefixes.slice(0, selectedIndex),
        selectedJournalPrefixes.slice(selectedIndex + 1)
      );
    }
    setSelectedJournalPrefixes(newSelected);
  };

  const columns = [
    {
      field: "batchId",
      headerName: "Batch ID",
      width: 125,
      align: "center",
      headerAlign: "center",
    },
    {
      field: "requestStatus",
      headerName: "Status",
      width: 125,
      align: "center",
      headerAlign: "center",
      renderCell: (params) => (
        <Chip
          label={params.value}
          color={getStatusChipColor(params.value)}
          size="small"
        />
      ),
    },
    {
      field: "creatorId",
      headerName: "Creator",
      width: 150,
      align: "center",
      headerAlign: "center",
      renderCell: (params) => (
        <Chip icon={<AccountCircleIcon />} label={params.value} size="small" />
      ),
    },
    {
      field: "requestDate",
      headerName: "Submitted On",
      width: 200,
      align: "center",
      headerAlign: "center",
      renderCell: (params) => (
        <Box>
          {new Date(params.value).toLocaleString("en-IN", {
            dateStyle: "medium",
            timeStyle: "short",
          })}
        </Box>
      ),
    },
    {
      field: "commonBatchRemarks",
      headerName: "Creator Remarks",
      width: 200,
      align: "center",
      headerAlign: "center",
    },
    {
      field: "requestCount",
      headerName: "Entries",
      width: 100,
      align: "center",
      headerAlign: "center",
    },
    {
      field: "totalCredit",
      headerName: "Total Credit",
      width: 250,
      align: "right",
      headerAlign: "center",

      renderCell: (params) => (
        <Typography marginTop={"5%"}>{formatCurrency(params.value)}</Typography>
      ),
    },
    {
      field: "totalDebit",
      headerName: "Total Debit",
      width: 250,
      align: "right",
      headerAlign: "center",
      renderCell: (params) => (
        <Typography marginTop={"5%"}>{formatCurrency(params.value)}</Typography>
      ),
    },
    {
      field: "actions",
      headerName: "Actions",
      flex: 0.8,
      minWidth: 130,
      align: "center",
      headerAlign: "center",
      sortable: false,
      renderCell: (params) => {
        const isPending = params.row.requestStatus === "PENDING";
        return (
          <Stack direction="row" spacing={0.5} marginTop={"5%"}>
            <Tooltip title="View Details">
              <IconButton
                onClick={() =>
                  handleOpenViewModal(
                    params.row.batchId,
                    params.row.requestStatus
                  )
                }
                color="primary"
              >
                <VisibilityIcon />
              </IconButton>
            </Tooltip>
            <Tooltip
              title={
                isPending ? "Cancel Entire Batch" : "Batch already processed"
              }
            >
              <span>
                <IconButton
                  onClick={() => handleOpenModal(params.row)}
                  disabled={!isPending}
                >
                  <DeleteIcon />
                </IconButton>
              </span>
            </Tooltip>
          </Stack>
        );
      },
    },
  ];

  return (
    <Box sx={mainContainerStyle}>
      <DataGrid
        rows={rows}
        columns={columns}
        loading={loading}
        getRowId={(row) => row.batchId}
        disableRowSelectionOnClick
        pageSizeOptions={[5, 10, 25]}
        disableColumnResize={true}
        rowHeight={52}
        initialState={{
          pagination: {
            paginationModel: { page: 0, pageSize: 10 },
          },
          sorting: { sortModel: [{ field: "requestDate", sort: "desc" }] },
        }}
        slots={{
          noRowsOverlay: CustomNoRowsOverlay,
        }}
        sx={dataGridContainerStyle}
      />

      <Dialog
        open={isModalOpen}
        onClose={handleCloseModal}
        fullWidth
        maxWidth="sm"
      >
        <DialogTitle sx={{ display: "flex", alignItems: "center" }}>
          <DeleteSweepIcon sx={{ mr: 1, color: "error.main" }} />
          Confirm Batch Cancellation
        </DialogTitle>
        <DialogContent>
          <DialogContentText>
            Are you sure you want to cancel all your pending requests in batch{" "}
            <strong>{selectedBatch?.batchId}</strong>?
            <br />
            <br />
            This will permanently delete {selectedBatch?.requestCount}{" "}
            entry/entries created by you and cannot be undone.
          </DialogContentText>
        </DialogContent>
        <DialogActions sx={dialogActionsStyle}>
          <Stack direction="row" spacing={1} justifyContent="flex-end">
            <Button onClick={handleCloseModal} disabled={isSubmitting}>
              Back
            </Button>
            <Button
              onClick={handleSubmitBatchCancel}
              variant="contained"
              color="error"
              disabled={isSubmitting}
            >
              {isSubmitting ? (
                <CircularProgress size={24} color="inherit" />
              ) : (
                "Confirm Cancel"
              )}
            </Button>
          </Stack>
        </DialogActions>
      </Dialog>

      <Dialog
        open={isViewModalOpen}
        onClose={handleCloseViewModal}
        fullWidth
        maxWidth="lg"
      >
        <DialogTitle sx={dialogTitleStyle}>
          Batch Details: {viewingBatchId}
          <IconButton
            aria-label="close"
            onClick={handleCloseViewModal}
            sx={{
              color: (theme) => theme.palette.grey[500],
            }}
          >
            <CloseIcon />
          </IconButton>
        </DialogTitle>

        <DialogContent dividers sx={{ p: 0 }}>
          {isViewLoading ? (
            <Box sx={loadingContainerStyle}>
              <CircularProgress />
            </Box>
          ) : (
            <>
              {batchDetails.length > 0 ? (
                <TableContainer component={Paper} sx={tableContainerStyle}>
                  <Table sx={{ minWidth: 650 }} size="small" stickyHeader>
                    <TableHead>
                      <TableRow>
                        {viewingBatchStatus === "PENDING" && (
                          <TableCell
                            padding="checkbox"
                            sx={tableHeaderCellStyle}
                          >
                            <Checkbox
                              color="primary"
                              indeterminate={
                                selectedJournalPrefixes.length > 0 &&
                                selectedJournalPrefixes.length <
                                  batchDetails.length
                              }
                              checked={
                                batchDetails.length > 0 &&
                                selectedJournalPrefixes.length ===
                                  batchDetails.length
                              }
                              onChange={handleSelectAllClick}
                            />
                          </TableCell>
                        )}
                        <TableCell align="center" sx={tableHeaderCellStyle}>
                          Journal ID
                        </TableCell>
                        <TableCell align="center" sx={tableHeaderCellStyle}>
                          Status
                        </TableCell>
                        <TableCell align="center" sx={tableHeaderCellStyle}>
                          Branch
                        </TableCell>
                        <TableCell align="center" sx={tableHeaderCellStyle}>
                          Currency
                        </TableCell>
                        <TableCell align="center" sx={tableHeaderCellStyle}>
                          CGL
                        </TableCell>
                        <TableCell align="center" sx={tableHeaderCellStyle}>
                          Amount
                        </TableCell>
                        <TableCell align="center" sx={tableHeaderCellStyle}>
                          Txn Type
                        </TableCell>
                      </TableRow>
                    </TableHead>

                    <TableBody>
                      {batchDetails.map((group, groupIndex) => {
                        const isSelected = selectedJournalPrefixes.includes(
                          group[0].journalPrefix
                        );
                        const groupRowCount = group.length;

                        return group.map((row, rowIndex) => (
                          <TableRow
                            key={row.id}
                            hover={viewingBatchStatus === "PENDING"}
                            onClick={() =>
                              viewingBatchStatus === "PENDING" &&
                              handleRowCheckboxClick(null, row.journalPrefix)
                            }
                            role="checkbox"
                            aria-checked={isSelected}
                            selected={
                              viewingBatchStatus === "PENDING" && isSelected
                            }
                            sx={{
                              cursor:
                                viewingBatchStatus === "PENDING"
                                  ? "pointer"
                                  : "default",
                              ...(rowIndex === 0 &&
                                groupIndex > 0 && {
                                  borderTop: (theme) =>
                                    `2px solid ${theme.palette.divider}`,
                                }),
                            }}
                          >
                            {viewingBatchStatus === "PENDING" &&
                              rowIndex === 0 && (
                                <TableCell
                                  padding="checkbox"
                                  rowSpan={groupRowCount}
                                  sx={stickyColumnCell}
                                >
                                  <Box sx={stickyBoxStyle}>
                                    <Checkbox
                                      color="primary"
                                      checked={isSelected}
                                      onClick={(e) =>
                                        handleRowCheckboxClick(
                                          e,
                                          row.journalPrefix
                                        )
                                      }
                                    />
                                  </Box>
                                </TableCell>
                              )}
                            {rowIndex === 0 && (
                              <TableCell
                                align="center"
                                rowSpan={groupRowCount}
                                sx={stickyColumnCell}
                              >
                                <Box sx={stickyBoxStyle}>
                                  {row.journalPrefix}
                                </Box>
                              </TableCell>
                            )}
                            {rowIndex === 0 && (
                              <TableCell
                                align="center"
                                rowSpan={groupRowCount}
                                sx={stickyColumnCell}
                              >
                                <Box sx={stickyBoxStyle}>
                                  <Chip
                                    label={row.requestStatus}
                                    color={getStatusChipColor(
                                      row.requestStatus
                                    )}
                                    size="small"
                                  />
                                </Box>
                              </TableCell>
                            )}
                            <TableCell align="center" sx={cellBorderStyle}>
                              {row.payload.branch}
                            </TableCell>
                            <TableCell align="center" sx={cellBorderStyle}>
                              {row.payload.currency}
                            </TableCell>
                            <TableCell align="center" sx={cellBorderStyle}>
                              {row.payload.cgl}
                            </TableCell>
                            <TableCell align="center" sx={cellBorderStyle}>
                              {formatCurrency(row.amount)}
                            </TableCell>
                            <TableCell align="center" sx={cellBorderStyle}>
                              <Chip
                                label={row.transactionType}
                                color={getTxnTypeColor(row.transactionType)}
                                size="small"
                              />
                            </TableCell>
                          </TableRow>
                        ));
                      })}
                    </TableBody>
                  </Table>
                </TableContainer>
              ) : (
                <Typography sx={{ p: 3 }}>
                  No details found for this batch.
                </Typography>
              )}
            </>
          )}
        </DialogContent>

        <DialogActions sx={dialogActionsStyle}>
          <Grid
            container
            spacing={2}
            alignItems="center"
            justifyContent="flex-end"
          >
            <Grid item xs={12} md={6}></Grid>
            <Grid item xs={12} md={6}>
              <Stack direction="row" spacing={1} justifyContent="flex-end">
                {viewingBatchStatus === "PENDING" && (
                  <Button
                    variant="contained"
                    color="error"
                    onClick={handleCancelSelectedJournals}
                    disabled={
                      isPartialSubmitting ||
                      selectedJournalPrefixes.length === 0
                    }
                  >
                    {isPartialSubmitting ? (
                      <CircularProgress size={24} color="inherit" />
                    ) : (
                      "Cancel Selected"
                    )}
                  </Button>
                )}
              </Stack>
            </Grid>
          </Grid>
        </DialogActions>
      </Dialog>
    </Box>
  );
}
