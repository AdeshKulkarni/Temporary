import com.fincore.VoucherService.Dto.BranchDto;
import com.fincore.VoucherService.Service.BranchService;
import lombok.RequiredArgsConstructor;
import lombok.extern.slf4j.Slf4j;
import org.springframework.http.ResponseEntity;
import org.springframework.validation.annotation.Validated;
import jakarta.validation.constraints.Pattern;
import org.springframework.web.bind.annotation.GetMapping;
import org.springframework.web.bind.annotation.RequestMapping;
import org.springframework.web.bind.annotation.RequestParam;
import org.springframework.web.bind.annotation.RestController;

import java.util.Collections;
import java.util.List;
import java.util.Map;

@RestController
@RequestMapping("/api/branches-journal")
@RequiredArgsConstructor
@Slf4j
@Validated // FORTIFY FIX: Enable method-level validation to sanitize input
public class BranchController {

    private final BranchService branchService;

    @GetMapping("/map")
    public ResponseEntity<Map<String, String>> getBranchMap() {
        log.info("BRANCH_CONTROLLER_MAP_REQUEST | Action: Fetching all active branch mappings");
        Map<String, String> branchMap = branchService.getBranchCodeAndNameMap();
        return ResponseEntity.ok(branchMap);
    }

    @GetMapping("/search")
    public ResponseEntity<List<BranchDto>> searchBranches(
            // FORTIFY FIX: Restrict search strings to alphanumeric inputs to defeat DB access injection
            @RequestParam @Pattern(regexp = "^[a-zA-Z0-9]{1,20}$", message = "Invalid search query") String query) {
        
        log.info("BRANCH_CONTROLLER_PREFIX_SEARCH | Query: {}", query);

        if (query == null || query.trim().isEmpty()) {
            return ResponseEntity.ok(Collections.emptyList());
        }

        List<BranchDto> results = branchService.searchBranches(query.trim());
        return ResponseEntity.ok(results);
    }
}
