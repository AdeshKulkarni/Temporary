  // Helper to handle downloads (BLOB response)
  const downloadFileHelper = async (url, filename) => {
    try {
      console.log(`Starting download for: ${filename}`);

      // FIXED CALL API ARGUMENTS
      const response = await callApi(
        url,
        null,      // 2. payload
        "GET",     // 3. method
        "blob",    // 4. responseType (MUST BE STRING "blob")
        null,      // 5. contentType (null is fine for GET)
        {},        // 6. extraConfig (Empty object, don't pass responseType here)
        false      // 7. returnDataOnly (False to get full response object)
      );

      // Handle response structure based on returnDataOnly flag
      const blobData = response.data || response;

      if (!blobData) {
        throw new Error("Download failed: Empty response data");
      }

      // Create a Blob from the data
      // For Excel files, explicitly setting the type helps
      const blob = new Blob([blobData], { 
        type: "application/vnd.openxmlformats-officedocument.spreadsheetml.sheet" 
      });

      // Create download link
      const link = document.createElement("a");
      const objectUrl = window.URL.createObjectURL(blob);
      
      link.href = objectUrl;
      link.setAttribute("download", filename);
      document.body.appendChild(link);
      link.click();
      
      // Cleanup
      link.remove();
      window.URL.revokeObjectURL(objectUrl);
      
      return true;
    } catch (e) {
      console.error("Download Error:", e);
      showSnackBar("File download failed", "error");
      return false;
    }
  };
