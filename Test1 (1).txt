The NullPointerException is likely occurring in one of two places:
 * In the Controller: For CSV uploads, the successFile is null. Java's Map.of(...) method throws a NullPointerException if you try to put a null value into it.
 * In the Service (Error Report): If your CSV doesn't have a Product Code, that field remains null. When generating an Error Excel file, cell.setCellValue(null) can throw an exception in some versions of POI.
Here are the corrected files.

        response.put("message", "Validation Successful!");
        response.put("successFileBase64", base64File); // Safe to put null in HashMap
        response.put("previewData", result.get("previewData"));

        return ResponseEntity.ok(response);
    }

package com.fincore.JournalService.Service;

import com.fincore.JournalService.Models.BranchMaster;
import com.fincore.JournalService.Models.CglMaster;
import com.fincore.JournalService.Models.CurrencyMaster;
import com.fincore.JournalService.Repository.BranchMasterRepository;
import com.fincore.JournalService.Repository.CglMasterRepository;
import com.fincore.JournalService.Repository.CurrencyMasterRepository;
import lombok.RequiredArgsConstructor;
import lombok.extern.slf4j.Slf4j;

import org.apache.poi.ss.usermodel.*;
import org.apache.poi.xssf.usermodel.XSSFWorkbook;
import org.springframework.stereotype.Service;
import org.springframework.web.multipart.MultipartFile;

import java.io.BufferedReader;
import java.io.ByteArrayOutputStream;
import java.io.IOException;
import java.io.InputStreamReader;
import java.math.BigDecimal;
import java.math.RoundingMode;
import java.nio.charset.StandardCharsets;
import java.time.LocalDate;
import java.time.format.DateTimeFormatter;
import java.util.*;
import java.util.stream.Collectors;

@Service
@RequiredArgsConstructor
@Slf4j
public class JournalBulkValidationService {

    private final BranchMasterRepository branchRepo;
    private final CglMasterRepository cglRepo;
    private final CurrencyMasterRepository currencyRepo;

    private static class ExcelRowData {
        int rowIndex;
        String branch = "";
        String currency = "";
        String cgl = "";
        BigDecimal amount; 
        String txnType = "";
        String remarks = "";
        String productCode = ""; // Initialize to empty string to avoid NPE
        List<String> errors = new ArrayList<>();
    }

    public Map<String, Object> validateAndProcessFile(MultipartFile file, LocalDate postingDate) throws IOException {
        String filename = file.getOriginalFilename();
        List<ExcelRowData> parsedRows;

        if (filename != null && (filename.toLowerCase().endsWith(".csv") || filename.toLowerCase().endsWith(".txt"))) {
            parsedRows = parseCsvFile(file);
        } else {
            parsedRows = parseExcelFile(file);
        }

        boolean isInputCsv = filename != null && filename.toLowerCase().endsWith(".csv");
        return validateRows(parsedRows, postingDate, isInputCsv);
    }

    // --- A. PARSE EXCEL ---
    private List<ExcelRowData> parseExcelFile(MultipartFile file) throws IOException {
        List<ExcelRowData> rows = new ArrayList<>();
        Workbook workbook = new XSSFWorkbook(file.getInputStream());
        Sheet sheet = workbook.getSheetAt(0);

        for (Row row : sheet) {
            if (isRowEmpty(row)) continue;

            ExcelRowData data = new ExcelRowData();
            data.rowIndex = row.getRowNum();

            data.branch = parseCode(row.getCell(0));     
            data.currency = getCellValueAsString(row.getCell(1)).toUpperCase().trim(); 
            data.cgl = parseCode(row.getCell(2));        
            
            String amountRaw = getCellValueAsString(row.getCell(3)).trim(); 
            String typeRaw = getCellValueAsString(row.getCell(4)).trim().toUpperCase(); 
            
            data.remarks = getCellValueAsString(row.getCell(5));
            data.productCode = parseCode(row.getCell(6)); // Might be empty string
            data.txnType = typeRaw; 

            parseAmountAndType(data, amountRaw, typeRaw);
            rows.add(data);
        }
        workbook.close();
        return rows;
    }

    // --- B. PARSE CSV ---
    private List<ExcelRowData> parseCsvFile(MultipartFile file) throws IOException {
        List<ExcelRowData> rows = new ArrayList<>();
        BufferedReader reader = new BufferedReader(new InputStreamReader(file.getInputStream(), StandardCharsets.UTF_8));
        String line;
        int lineNum = 0;

        while ((line = reader.readLine()) != null) {
            if (line.trim().isEmpty()) continue;
            
            // Split with -1 to keep empty trailing fields
            String[] cols = line.split(",", -1); 
            
            // Need at least up to Amount/Type (index 8)
            if (cols.length < 9) continue; 

            ExcelRowData data = new ExcelRowData();
            data.rowIndex = lineNum++;

            // Based on CSV format: 4=Branch, 5=Currency, 6=CGL, 7=Amount, 8=Type, 9=Remarks, 10=Product
            data.branch = cols[4].trim();
            data.currency = cols[5].trim();
            data.cgl = cols[6].trim();
            
            String amountRaw = cols[7].trim();
            String typeRaw = cols[8].trim();
            
            // Bounds check for Remarks
            if (cols.length > 9) {
                data.remarks = cols[9].trim();
            }
            
            // Bounds check for Product Code
            if (cols.length > 10) {
                String prod = cols[10].trim();
                data.productCode = "A".equalsIgnoreCase(prod) ? "" : prod;
            }

            data.txnType = typeRaw;
            parseAmountAndType(data, amountRaw, typeRaw);
            rows.add(data);
        }
        return rows;
    }

    // --- C. SHARED VALIDATION ---
    private Map<String, Object> validateRows(List<ExcelRowData> parsedRows, LocalDate postingDate, boolean isInputCsv) throws IOException {
        Set<String> validBranches = branchRepo.findAll().stream().map(BranchMaster::getCode).map(String::trim).collect(Collectors.toSet());
        Set<String> validCgls = cglRepo.findAll().stream().map(CglMaster::getCglNumber).map(String::trim).collect(Collectors.toSet());
        Set<String> validCurrencies = currencyRepo.findAll().stream().map(CurrencyMaster::getCurrencyCode).map(String::trim).collect(Collectors.toSet());

        Map<String, BigDecimal> balanceMap = new HashMap<>();

        for (ExcelRowData data : parsedRows) {
            if (!validBranches.contains(data.branch)) data.errors.add("Invalid Branch Code (" + data.branch + ")");
            if (!validCurrencies.contains(data.currency)) data.errors.add("Invalid Currency (" + data.currency + ")");
            if (!validCgls.contains(data.cgl)) data.errors.add("Invalid CGL (" + data.cgl + ")");

            if (data.productCode != null && !data.productCode.isEmpty()) {
                if (!data.productCode.matches("\\d{8}")) data.errors.add("Product Code must be 8 digits");
            }

            if (data.amount == null || data.amount.compareTo(BigDecimal.ZERO) == 0) {
                data.errors.add("Amount cannot be Zero or Invalid");
            }

            if (data.remarks != null && data.remarks.length() > 30) data.errors.add("Remarks > 30 chars");

            if (validBranches.contains(data.branch) && validCurrencies.contains(data.currency) && data.amount != null) {
                String key = data.branch + "_" + data.currency;
                balanceMap.put(key, balanceMap.getOrDefault(key, BigDecimal.ZERO).add(data.amount));
            }
        }

        for (ExcelRowData data : parsedRows) {
            if (!validBranches.contains(data.branch) || !validCurrencies.contains(data.currency)) continue;
            String key = data.branch + "_" + data.currency;
            BigDecimal netTotal = balanceMap.getOrDefault(key, BigDecimal.ZERO);
            if (netTotal.setScale(4, RoundingMode.HALF_UP).compareTo(BigDecimal.ZERO) != 0) {
                data.errors.add("Branch Total Mismatch (Diff: " + netTotal.toPlainString() + ")");
            }
        }

        long errorCount = parsedRows.stream().filter(r -> !r.errors.isEmpty()).count();
        Map<String, Object> result = new HashMap<>();

        if (errorCount > 0) {
            byte[] errorFile = generateErrorExcel(parsedRows);
            result.put("status", "ERROR");
            result.put("errorCount", errorCount);
            result.put("file", errorFile);
            result.put("message", "Validation failed. Found " + errorCount + " errors.");
        } else {
            List<Map<String, Object>> previewList = parsedRows.stream().map(row -> {
                Map<String, Object> map = new HashMap<>();
                map.put("id", row.rowIndex);
                map.put("branch", row.branch);
                map.put("currency", row.currency);
                map.put("cgl", row.cgl);
                map.put("amount", row.amount.abs().toPlainString()); 
                map.put("txnType", row.amount.compareTo(BigDecimal.ZERO) < 0 ? "Credit" : "Debit");
                map.put("remarks", row.remarks);
                map.put("productCode", row.productCode);
                return map;
            }).collect(Collectors.toList());

            result.put("status", "SUCCESS");
            result.put("previewData", previewList);
            result.put("message", "Validation Successful!");
            
            // Only generate download file if input was NOT CSV
            if (!isInputCsv) {
                result.put("file", generateSuccessCsv(parsedRows, postingDate));
            }
        }
        return result;
    }

    private void parseAmountAndType(ExcelRowData data, String amountRaw, String typeRaw) {
        try {
            String cleanAmountStr = amountRaw.replaceAll("[^0-9.-]", "");
            if (cleanAmountStr.isEmpty()) throw new NumberFormatException("Empty");
            
            BigDecimal amt = new BigDecimal(cleanAmountStr);
            boolean isCredit = false;

            if (typeRaw.startsWith("C") || typeRaw.contains("CR")) isCredit = true;
            if (amountRaw.toUpperCase().contains("CR")) isCredit = true;
            if (amountRaw.toUpperCase().contains("DR")) isCredit = false;
            if (amountRaw.startsWith("-")) isCredit = true;

            data.amount = isCredit ? amt.abs().negate() : amt.abs();
        } catch (Exception e) {
            data.amount = BigDecimal.ZERO; 
        }
    }

    // --- Output Generators ---
    private byte[] generateSuccessCsv(List<ExcelRowData> rows, LocalDate postingDate) {
        StringBuilder csv = new StringBuilder();
        DateTimeFormatter ddMMyyyy = DateTimeFormatter.ofPattern("ddMMyyyy");
        String col2 = postingDate.format(ddMMyyyy);

        int month = postingDate.getMonthValue();
        int year = postingDate.getYear();
        String col3 = (month >= 4) ? String.valueOf(year) : String.valueOf(year - 1);
        String col4 = (month >= 4) ? String.format("%02d", month - 3) : String.format("%02d", month + 9);

        for (ExcelRowData row : rows) {
            String col1 = "01";
            String col5 = String.format("%5s", row.branch).replace(' ', '0');
            String col6 = (row.currency == null || row.currency.isEmpty()) ? "INR" : row.currency;
            String col7 = row.cgl;

            BigDecimal absAmt = row.amount.abs();
            String plainAmt = absAmt.setScale(4, RoundingMode.HALF_UP).toPlainString();
            String[] parts = plainAmt.split("\\.");
            String intPart = String.format("%16s", parts[0]).replace(' ', '0');
            String decPart = parts.length > 1 ? parts[1] : "0000";
            String col8 = intPart + "." + decPart;

            String col9 = (row.amount.compareTo(BigDecimal.ZERO) < 0) ? "Cr" : "Dr";
            String col10 = (row.remarks != null) ? row.remarks.replace(",", " ") : "";
            String col11 = (row.productCode != null && !row.productCode.isEmpty()) ? row.productCode : "A";

            csv.append(String.join(",", col1, col2, col3, col4, col5, col6, col7, col8, col9, col10, col11, "B", "C", "D")).append("\n");
        }
        return csv.toString().getBytes();
    }

    private byte[] generateErrorExcel(List<ExcelRowData> rows) throws IOException {
        Workbook workbook = new XSSFWorkbook();
        Sheet sheet = workbook.createSheet("Error Report");
        CellStyle headerStyle = workbook.createCellStyle();
        Font font = workbook.createFont();
        font.setBold(true);
        font.setColor(IndexedColors.RED.getIndex());
        headerStyle.setFont(font);

        Row header = sheet.createRow(0);
        String[] headers = {"Branch", "Currency", "CGL", "Amount", "TxnType", "Remarks", "Product Code", "ERRORS"};
        for (int i = 0; i < headers.length; i++) {
            Cell cell = header.createCell(i);
            cell.setCellValue(headers[i]);
            cell.setCellStyle(headerStyle);
        }

        int rowIdx = 1;
        for (ExcelRowData data : rows) {
            Row row = sheet.createRow(rowIdx++);
            row.createCell(0).setCellValue(data.branch);
            row.createCell(1).setCellValue(data.currency);
            row.createCell(2).setCellValue(data.cgl);
            row.createCell(3).setCellValue(data.amount != null ? data.amount.abs().toPlainString() : "0");
            row.createCell(4).setCellValue(data.txnType);
            row.createCell(5).setCellValue(data.remarks);
            // Fix: Check for null to avoid NPE in old POI versions
            row.createCell(6).setCellValue(data.productCode != null ? data.productCode : "");

            if (!data.errors.isEmpty()) {
                Cell errorCell = row.createCell(7);
                errorCell.setCellValue(String.join("; ", data.errors));
                errorCell.setCellStyle(headerStyle);
            }
        }
        for(int i=0; i<8; i++) sheet.autoSizeColumn(i);
        ByteArrayOutputStream out = new ByteArrayOutputStream();
        workbook.write(out);
        workbook.close();
        return out.toByteArray();
    }

    private String parseCode(Cell cell) {
        if (cell == null) return "";
        String val = getCellValueAsString(cell).trim();
        if (val.endsWith(".0")) return val.substring(0, val.length() - 2);
        return val;
    }

    private String getCellValueAsString(Cell cell) {
        if (cell == null) return "";
        if (cell.getCellType() == CellType.STRING) return cell.getStringCellValue().trim();
        if (cell.getCellType() == CellType.NUMERIC) {
            if (DateUtil.isCellDateFormatted(cell)) return cell.getDateCellValue().toString();
            return BigDecimal.valueOf(cell.getNumericCellValue()).toPlainString();
        }
        if (cell.getCellType() == CellType.BOOLEAN) return String.valueOf(cell.getBooleanCellValue());
        return "";
    }

    private boolean isRowEmpty(Row row) {
        if (row == null) return true;
        for (int c = row.getFirstCellNum(); c < row.getLastCellNum(); c++) {
            Cell cell = row.getCell(c);
            if (cell != null && cell.getCellType() != CellType.BLANK && !getCellValueAsString(cell).trim().isEmpty())
                return false;
        }
        return true;
    }
