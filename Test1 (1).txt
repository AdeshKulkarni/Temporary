package com.tcs.fincore.CommonRequestService.repository;

import com.tcs.fincore.CommonRequestService.model.CglMaster;
import org.springframework.data.jpa.repository.JpaRepository;
import org.springframework.stereotype.Repository;

import java.util.List;

/**
 * Spring Data JPA repository for the CglMaster entity.
 */
@Repository
public interface CglMasterRepository extends JpaRepository<CglMaster, String> {

    /**
     * Finds all CGLs matching a specific account classification code.
     * This is used for the dropdown filter functionality.
     *
     * @param acClassification The classification code ('A', 'L', 'I', 'E', 'M').
     * @return A list of matching CglMaster entities.
     */
    List<CglMaster> findByAcClassification(String acClassification);

    /**
     * Finds CGLs where the CGL number contains the given search query.
     * Note: This performs a 'LIKE %query%' search, which can be slow on very large tables.
     * For CGL numbers, a 'StartingWith' search is often faster if applicable.
     *
     * @param cglNumber The partial or full CGL number to search for.
     * @return A list of matching CglMaster entities.
     */
    List<CglMaster> findByCglNumberContainingIgnoreCase(String cglNumber);
}
