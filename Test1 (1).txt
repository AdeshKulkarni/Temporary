import React, { useState, useEffect, useCallback } from "react";
import {
  Box, Typography, Chip, Dialog, DialogActions, DialogContent, DialogContentText,
  DialogTitle, TextField, Button, CircularProgress, IconButton, Stack,
  Table, TableBody, TableCell, TableContainer, TableHead, TableRow, Paper, Checkbox
} from "@mui/material";
import { DataGrid } from "@mui/x-data-grid";
import {
  Check as CheckIcon,
  Clear as ClearIcon,
  Visibility as VisibilityIcon,
  AccountCircle
} from "@mui/icons-material";
import { alpha } from "@mui/material/styles";

import useApi from "../../hooks/useApi";
import useCustomSnackbar from "../../utils/useCustomSnackbar";

// --- INTERNAL STYLING ---
const styles = {
  mainContainer: {
    padding: 3,
    height: "100%",
    width: "100%",
    backgroundColor: "background.default",
  },
  dataGridContainer: {
    height: 600,
    width: "100%",
    backgroundColor: "background.paper",
    "& .MuiDataGrid-columnHeaders": {
      backgroundColor: (theme) => alpha(theme.palette.primary.main, 0.1),
      fontWeight: "bold",
    },
  },
  tableContainer: {
    marginTop: 2,
    borderRadius: "8px",
    border: "1px solid",
    borderColor: "divider",
    overflow: "auto",
    maxHeight: "600px",
    boxShadow: "0px 2px 8px rgba(0,0,0,0.05)",
  },
  tableHeader: {
    backgroundColor: (theme) => alpha(theme.palette.primary.main, 0.1),
    fontWeight: "bold",
    position: "sticky",
    top: 0,
    zIndex: 10,
    whiteSpace: "nowrap",
  },
  // Sticky Column Logic
  stickyColumn: {
    position: "sticky",
    left: 0,
    backgroundColor: "background.paper",
    zIndex: 5,
    borderRight: "2px solid rgba(224, 224, 224, 1)", 
  },
  stickyHeaderColumn: {
    position: "sticky",
    left: 0,
    top: 0,
    zIndex: 15,
    backgroundColor: "#f5f5f5",
    fontWeight: "bold",
    borderRight: "2px solid rgba(224, 224, 224, 1)",
    borderBottom: "1px solid rgba(224, 224, 224, 1)",
  },
  dialogActions: {
    padding: 2,
    display: "flex",
    gap: 1,
    justifyContent: "flex-end",
  },
};

const formatCurrency = (num) => Number(num).toFixed(4);

export default function JournalRequestApproval() {
  const { callApi } = useApi();
  const showSnackBar = useCustomSnackbar();

  // --- State ---
  const [currentUserId, setCurrentUserId] = useState(null);
  const [loading, setLoading] = useState(true);
  const [rows, setRows] = useState([]);

  // Action Modal
  const [isModalOpen, setIsModalOpen] = useState(false);
  const [isApproveLoading, setIsApproveLoading] = useState(false);
  const [isRejectLoading, setIsRejectLoading] = useState(false);
  const [selectedBatch, setSelectedBatch] = useState(null);
  const [modalMode, setModalMode] = useState("ACCEPTED");
  const [executorRemarks, setExecutorRemarks] = useState("");

  // View Details Modal
  const [isViewModalOpen, setIsViewModalOpen] = useState(false);
  const [isViewLoading, setIsViewLoading] = useState(false);
  const [batchDetails, setBatchDetails] = useState([]);
  const [viewingBatchId, setViewingBatchId] = useState(null);
  const [selectedJournalPrefixes, setSelectedJournalPrefixes] = useState([]);

  useEffect(() => {
    // Basic Auth Check from Storage
    try {
      const auth = sessionStorage.getItem("persist:auth") || localStorage.getItem("persist:auth");
      if (auth) {
        const parsed = JSON.parse(auth);
        const user = JSON.parse(parsed.user);
        if (user?.userId) setCurrentUserId(String(user.userId));
      }
    } catch (e) { }
  }, []);

  const fetchPendingBatches = useCallback(async () => {
    setLoading(true);
    try {
      const pendingRequests = await callApi("/JS/journals/pending-requests", null, "GET");

      if (!Array.isArray(pendingRequests) || pendingRequests.length === 0) {
        setRows([]);
        return;
      }

      // Group individual requests into Batches
      const groupedBatches = new Map();
      pendingRequests.forEach(req => {
        const batchId = req.batchId;
        if (!groupedBatches.has(batchId)) {
          groupedBatches.set(batchId, {
            id: batchId,
            batchId: batchId,
            creatorId: req.creatorId,
            requestStatus: req.requestStatus,
            requestDate: req.requestDate,
            commonBatchRemarks: req.commonBatchRemarks,
            totalDebit: 0,
            totalCredit: 0,
            requestCount: 0
          });
        }
        const detail = groupedBatches.get(batchId);
        detail.requestCount++;
        try {
          const pl = JSON.parse(req.payload);
          const amt = parseFloat(pl.amount) || 0;
          if (amt < 0) detail.totalCredit += Math.abs(amt);
          else detail.totalDebit += amt;
        } catch (e) { }
      });

      setRows(Array.from(groupedBatches.values()));
    } catch (err) {
      showSnackBar("Failed to load requests", "error");
    } finally {
      setLoading(false);
    }
  }, [callApi, showSnackBar]);

  useEffect(() => { fetchPendingBatches(); }, [fetchPendingBatches]);

  // --- Handlers ---
  const handleOpenModal = (batch, mode) => {
    setSelectedBatch(batch);
    setModalMode(mode);
    setExecutorRemarks("");
    setIsModalOpen(true);
  };

  const handleCloseModal = () => {
    if (isApproveLoading || isRejectLoading) return;
    setIsModalOpen(false);
    setSelectedBatch(null);
  };

  const handleSubmitAction = async () => {
    if (modalMode === "REJECTED" && !executorRemarks.trim()) {
      showSnackBar("Remarks are required for rejection.", "warning");
      return;
    }

    const setLoadingFn = modalMode === "ACCEPTED" ? setIsApproveLoading : setIsRejectLoading;
    setLoadingFn(true);

    try {
      const payload = {
        batchId: selectedBatch.batchId,
        status: modalMode,
        remarks: executorRemarks.trim() || (modalMode === "ACCEPTED" ? "Accepted" : "Rejected"),
      };

      await callApi("/JS/journals/process-bulk", payload, "POST");
      showSnackBar(`Batch ${modalMode === "ACCEPTED" ? "Approved" : "Rejected"} Successfully`, "success", 5000);
      handleCloseModal();
      fetchPendingBatches();
    } catch (e) {
      showSnackBar("Process Failed", "error");
    } finally {
      setLoadingFn(false);
    }
  };

  const handleOpenViewModal = async (batchId, creatorId) => {
    setIsViewModalOpen(true);
    setIsViewLoading(true);
    setViewingBatchId(batchId);
    setBatchDetails([]);
    setSelectedJournalPrefixes([]);

    try {
      const data = await callApi(`/JS/journals/by-batch/${batchId}`, null, "GET");

      // Group Requests by Journal Prefix (e.g., JRNL-001-01, JRNL-001-02 -> JRNL-001)
      const groups = new Map();
      data.filter(r => r.requestStatus === "PENDING").forEach(req => {
        const prefix = req.journalId.split("-")[0];
        if (!groups.has(prefix)) groups.set(prefix, []);
        try {
          const pl = JSON.parse(req.payload);
          groups.get(prefix).push({
            ...req,
            payload: pl,
            journalPrefix: prefix,
            amount: Math.abs(parseFloat(pl.amount) || 0)
          });
        } catch (e) { }
      });

      setBatchDetails(Array.from(groups.values()));
    } catch (e) {
      showSnackBar("Failed to load details", "error");
    } finally {
      setIsViewLoading(false);
    }
  };

  // --- Render Sticky Detail Rows ---
  const renderDetailRows = () => {
    return batchDetails.map((group, gIdx) => (
      group.map((row, rIdx) => {
        const isSelected = selectedJournalPrefixes.includes(row.journalPrefix);
        return (
          <TableRow key={row.id} selected={isSelected} hover>
            
            {/* STICKY COLUMN 1: Checkbox */}
            {rIdx === 0 && (
              <TableCell
                rowSpan={group.length}
                padding="checkbox"
                sx={{ ...styles.stickyColumn, top: 0, zIndex: 5 }} // Lower zIndex than header
              >
                <Checkbox
                  checked={isSelected}
                  onClick={() => {
                    const newSel = isSelected
                      ? selectedJournalPrefixes.filter(p => p !== row.journalPrefix)
                      : [...selectedJournalPrefixes, row.journalPrefix];
                    setSelectedJournalPrefixes(newSel);
                  }}
                />
              </TableCell>
            )}

            {/* STICKY COLUMN 2: Journal ID */}
            {rIdx === 0 && (
              <TableCell
                rowSpan={group.length}
                sx={{
                  ...styles.stickyColumn,
                  left: 48, // Offset for checkbox width
                  zIndex: 5,
                  fontWeight: 'bold'
                }}
              >
                {row.journalPrefix}
              </TableCell>
            )}

            {/* Standard Columns */}
            <TableCell>{row.payload.branch}</TableCell>
            <TableCell>{row.payload.currency}</TableCell>
            <TableCell>{row.payload.cgl}</TableCell>
            <TableCell align="right">{formatCurrency(row.amount)}</TableCell>
            <TableCell align="center">
               <Chip label={row.amount < 0 || row.payload.amount < 0 ? "Credit" : "Debit"} 
                     size="small" 
                     color={row.amount < 0 || row.payload.amount < 0 ? "error" : "success"} />
            </TableCell>
          </TableRow>
        );
      })
    ));
  };

  const columns = [
    { field: "batchId", headerName: "Batch ID", width: 130, disableColumnMenu: true, align: 'center', headerAlign: 'center' },
    {
      field: "requestStatus", headerName: "Status", width: 120, disableColumnMenu: true, align: 'center', headerAlign: 'center',
      renderCell: (p) => <Chip label={p.value} color="warning" size="small" />
    },
    {
      field: "creatorId", headerName: "Creator", width: 150, disableColumnMenu: true, align: 'center', headerAlign: 'center',
      renderCell: (p) => <Chip icon={<AccountCircle />} label={p.value} variant="outlined" size="small" />
    },
    {
      field: "requestDate", headerName: "Date", width: 180, disableColumnMenu: true, align: 'center', headerAlign: 'center',
      valueFormatter: (p) => new Date(p.value).toLocaleString()
    },
    { field: "commonBatchRemarks", headerName: "Remarks", flex: 1, disableColumnMenu: true, headerAlign: 'center' },
    {
      field: "actions", headerName: "Actions", width: 180, disableColumnMenu: true, sortable: false, align: 'center', headerAlign: 'center',
      renderCell: (p) => {
        const isSelf = String(currentUserId) === String(p.row.creatorId);
        return (
          <Stack direction="row" spacing={1}>
            <IconButton onClick={() => handleOpenViewModal(p.row.batchId, p.row.creatorId)} color="primary">
              <VisibilityIcon />
            </IconButton>
            <IconButton onClick={() => handleOpenModal(p.row, "ACCEPTED")} color="success" disabled={isSelf}>
              <CheckIcon />
            </IconButton>
            <IconButton onClick={() => handleOpenModal(p.row, "REJECTED")} color="error" disabled={isSelf}>
              <ClearIcon />
            </IconButton>
          </Stack>
        );
      }
    }
  ];

  return (
    <Box sx={styles.mainContainer}>
      <DataGrid
        rows={rows}
        columns={columns}
        loading={loading}
        getRowId={(r) => r.batchId}
        sx={styles.dataGridContainer}
        disableRowSelectionOnClick
      />

      {/* --- Action Dialog --- */}
      <Dialog open={isModalOpen} onClose={handleCloseModal} fullWidth maxWidth="sm">
        <DialogTitle>{modalMode === "ACCEPTED" ? "Approve" : "Reject"} Batch</DialogTitle>
        <DialogContent>
          <DialogContentText sx={{ mb: 2 }}>
            Confirm {modalMode === "ACCEPTED" ? "Approval" : "Rejection"} for Batch <strong>{selectedBatch?.batchId}</strong>?
          </DialogContentText>
          <TextField
            autoFocus={modalMode === "REJECTED"}
            label="Approver Remarks"
            fullWidth
            variant="outlined"
            required={modalMode === "REJECTED"}
            value={executorRemarks}
            onChange={(e) => setExecutorRemarks(e.target.value)}
          />
        </DialogContent>
        <DialogActions sx={styles.dialogActions}>
          <Button onClick={handleCloseModal} disabled={isApproveLoading || isRejectLoading}>
            Cancel
          </Button>
          <Button
            onClick={handleSubmitAction}
            variant="contained"
            color={modalMode === "ACCEPTED" ? "success" : "error"}
            disabled={isApproveLoading || isRejectLoading}
            startIcon={(isApproveLoading || isRejectLoading) ? <CircularProgress size={20} color="inherit" /> : null}
          >
            Confirm
          </Button>
        </DialogActions>
      </Dialog>

      {/* --- View Details Dialog --- */}
      <Dialog open={isViewModalOpen} onClose={() => setIsViewModalOpen(false)} fullWidth maxWidth="lg">
        <DialogTitle>Batch Details: {viewingBatchId}</DialogTitle>
        <DialogContent dividers>
          {isViewLoading ? (
            <Box sx={{ display: 'flex', justifyContent: 'center', p: 5 }}>
              <CircularProgress />
            </Box>
          ) : (
            <TableContainer component={Paper} sx={styles.tableContainer}>
              <Table size="small" stickyHeader>
                <TableHead>
                  <TableRow>
                    {/* Header for Sticky Column 1 */}
                    <TableCell sx={{ ...styles.stickyHeaderColumn, zIndex: 20, left: 0 }} padding="checkbox">
                      <Checkbox disabled />
                    </TableCell>
                    {/* Header for Sticky Column 2 */}
                    <TableCell sx={{ ...styles.stickyHeaderColumn, zIndex: 20, left: 48 }}>
                      Journal ID
                    </TableCell>
                    <TableCell sx={styles.tableHeader}>Branch</TableCell>
                    <TableCell sx={styles.tableHeader}>Currency</TableCell>
                    <TableCell sx={styles.tableHeader}>CGL</TableCell>
                    <TableCell align="right" sx={styles.tableHeader}>Amount</TableCell>
                    <TableCell align="center" sx={styles.tableHeader}>Type</TableCell>
                  </TableRow>
                </TableHead>
                <TableBody>{renderDetailRows()}</TableBody>
              </Table>
            </TableContainer>
          )}
        </DialogContent>
        <DialogActions sx={styles.dialogActions}>
          <Button onClick={() => setIsViewModalOpen(false)}>Close</Button>
        </DialogActions>
      </Dialog>
    </Box>
  );
}
