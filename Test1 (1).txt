@Service

    /**
     * Retrieves a mapping of branch codes to branch names.
     * Useful for dropdown population or lookup optimization.
     * * @return A Map containing branch codes as keys and names as values.
     * @throws org.springframework.dao.DataAccessException if the database retrieval fails.
     */
    public List<BranchDto> searchBranches(String query, java.time.LocalDate backDate) {
        if (query == null || query.trim().isEmpty()) return Collections.emptyList();

        try {
            String sql = "SELECT CODE, NAME FROM BRANCH_MASTER WHERE (LOWER(CODE) LIKE LOWER(?) OR LOWER(NAME) LIKE LOWER(?))";
            List<Object> params = new java.util.ArrayList<>();
            params.add("%" + query.trim() + "%");
            params.add("%" + query.trim() + "%");

            if (backDate != null) {
                sql += " AND TRUNC(OPEN_DATE) <= ? AND (CLOSE_DATE IS NULL OR TRUNC(CLOSE_DATE) >= ?)";
                params.add(java.sql.Date.valueOf(backDate));
                params.add(java.sql.Date.valueOf(backDate));
            } else {
                sql += " AND STATUS = 1";
            }
            sql += " FETCH FIRST 50 ROWS ONLY";

            return jdbcTemplate.query(sql, (rs, rowNum) -> new BranchDto(rs.getString("CODE"), rs.getString("NAME")), params.toArray());
        } catch (DataAccessException e) {
            return Collections.emptyList();
        }
    }


    /**
     * Performs a paginated search for branches by name or code.
     * Limits results to the first 50 matches to ensure high system responsiveness.[span_0](end_span)
     * * @param query The search string (branch name or code).
     * @return A list of matching BranchDto objects.
     */
    public List<BranchDto> searchBranches(String query) {
        log.info("BRANCH_SEARCH_INIT | QueryLength: {}", (query != null ? query.length() : 0)); //

        if (query == null || query.trim().isEmpty()) {
            return Collections.emptyList();
        }

        try {
            // Preserving the exact pagination and search logic
            Pageable limit = PageRequest.of(0, 50);


            List<BranchMaster> results = branchMasterRepository
                    .findByNameContainingIgnoreCaseOrCodeContainingIgnoreCase(query, query, limit);

            log.info("BRANCH_SEARCH_SUCCESS | ResultsFound: {}", results.size());

            return results.stream()
                    .map(branch -> new BranchDto(branch.getCode(), branch.getName()))
                    .collect(Collectors.toList());

        } catch (DataAccessException e) {
            log.error("BRANCH_SEARCH_FAIL | Query: [REDACTED] | Error: {}", e.getMessage());
            return Collections.emptyList();
        }
