    // ==================================================================================
    // 1. BUSINESS LOGIC & BALANCING TESTS (VoucherRuleValidator)
    // ==================================================================================
    @Nested
    @DisplayName("Voucher Rule Validator Tests")
    class VoucherRuleValidatorTest {

        @Mock
        private ValidationMasterService validationMasterService;

        @InjectMocks
        private VoucherRuleValidator ruleValidator;

        @Test
        @DisplayName("Format Check: Valid CR/DR should normalize to Credit/Debit")
        void testFormatCheck_ValidTransactionTypes() {
            ExcelRowData row1 = createMockRow("10301", "INR", "1102003001", new BigDecimal("500.00"), "CR");
            ExcelRowData row2 = createMockRow("10301", "INR", "1102003002", new BigDecimal("500.00"), "debit");

            ruleValidator.runFormatCheck(Arrays.asList(row1, row2));

            assertEquals("Credit", row1.txnType);
            assertEquals("Debit", row2.txnType);
            assertTrue(row1.formatErrors.isEmpty());
            assertTrue(row2.formatErrors.isEmpty());
        }

        @Test
        @DisplayName("Format Check: Invalid amounts should trigger format errors")
        void testFormatCheck_InvalidAmounts() {
            ExcelRowData row = createMockRow("10301", "INR", "1102003001", new BigDecimal("0.00"), "CR");

            ruleValidator.runFormatCheck(Collections.singletonList(row));

            assertFalse(row.formatErrors.isEmpty());
            assertTrue(row.formatErrors.contains("Amount must be greater than 0."));
        }

        @Test
        @DisplayName("Balance Check: Perfectly balanced Debits and Credits should pass")
        void testBalanceCheck_PerfectBalance() {
            ExcelRowData crRow = createMockRow("10301", "INR", "1102003001", new BigDecimal("1000.50"), "Credit");
            ExcelRowData drRow1 = createMockRow("10301", "INR", "1102003002", new BigDecimal("500.25"), "Debit");
            ExcelRowData drRow2 = createMockRow("10301", "INR", "1102003003", new BigDecimal("500.25"), "Debit");

            ruleValidator.runBalanceCheck(Arrays.asList(crRow, drRow1, drRow2));

            assertTrue(crRow.balErrors.isEmpty(), "Credit row should have no balance errors");
            assertTrue(drRow1.balErrors.isEmpty(), "Debit row should have no balance errors");
        }

        @Test
        @DisplayName("Balance Check: Unbalanced entries should trigger specific error")
        void testBalanceCheck_Unbalanced() {
            ExcelRowData crRow = createMockRow("10301", "INR", "1102003001", new BigDecimal("1000.50"), "Credit");
            ExcelRowData drRow = createMockRow("10301", "INR", "1102003002", new BigDecimal("900.00"), "Debit");

            ruleValidator.runBalanceCheck(Arrays.asList(crRow, drRow));

            assertTrue(crRow.balErrors.contains("Branch, Currency, and CGL-wise balance mismatch."));
            assertTrue(drRow.balErrors.contains("Branch, Currency, and CGL-wise balance mismatch."));
        }

        @Test
        @DisplayName("DB Check: Invalid branch triggers DB error")
        void testDbCheck_InvalidBranch() {
            when(validationMasterService.getAllActiveBranches()).thenReturn(Set.of("10001", "10002"));
            when(validationMasterService.getAllActiveCurrencies()).thenReturn(Set.of("INR"));
            when(validationMasterService.getAllActiveCgls()).thenReturn(Set.of("1102003001"));

            ExcelRowData row = createMockRow("99999", "INR", "1102003001", new BigDecimal("100.00"), "Debit");

            ruleValidator.runDbCheck(Collections.singletonList(row));

            assertTrue(row.dbErrors.contains("Branch is inactive or not found: 99999"));
        }
    }

    // ==================================================================================
    // 2. STREAMING & CACHE TESTS (ValidationStateManager)
    // ==================================================================================
    @Nested
    @DisplayName("Streaming & Redis State Manager Tests")
    class ValidationStateManagerTest {

        @Mock
        private ObjectMapper objectMapper;

        @Mock
        private RedisTemplate<String, byte[]> redisTemplate;

        @Mock
        private ValueOperations<String, byte[]> valueOperations;

        @InjectMocks
        private ValidationStateManager stateManager;

        @BeforeEach
        void setUp() {
            when(redisTemplate.opsForValue()).thenReturn(valueOperations);
        }

        @Test
        @DisplayName("getTotalChunks: Should successfully parse metadata from Redis")
        void testGetTotalChunks_Success() throws Exception {
            String reqId = "REQ-123";
            byte[] mockMetaBytes = "{\"totalChunks\": 5, \"totalRows\": 25000}".getBytes();

            when(valueOperations.get("DATA_META_" + reqId)).thenReturn(mockMetaBytes);
            when(objectMapper.readValue(eq(mockMetaBytes), any(TypeReference.class)))
                    .thenReturn(Map.of("totalChunks", 5, "totalRows", 25000));

            int chunks = stateManager.getTotalChunks(reqId);

            assertEquals(5, chunks);
        }

        @Test
        @DisplayName("getTotalChunks: Missing meta throws CacheExpiredException")
        void testGetTotalChunks_MissingMeta() {
            String reqId = "REQ-123";
            when(valueOperations.get("DATA_META_" + reqId)).thenReturn(null);

            assertThrows(CustomExceptions.CacheExpiredException.class, () -> {
                stateManager.getTotalChunks(reqId);
            });
        }
    }

    // ==================================================================================
    // 3. FILE SECURITY & UPLOAD TESTS (VoucherBulkValidationService)
    // ==================================================================================
    @Nested
    @DisplayName("File Security & Upload Initialization Tests")
    class VoucherBulkValidationServiceTest {

        @Mock
        private ValidationStateManager stateManager;

        @InjectMocks
        private VoucherBulkValidationService validationService;

        @Test
        @DisplayName("Upload Security: File too small is rejected")
        void testInitiateValidation_FileTooSmall() {
            byte[] smallContent = "Tiny".getBytes(); // Less than 1KB
            MultipartFile file = new MockMultipartFile("file", "test.xlsx", 
                    "application/vnd.openxmlformats-officedocument.spreadsheetml.sheet", smallContent);

            CustomExceptions.FileProcessingException ex = assertThrows(
                    CustomExceptions.FileProcessingException.class,
                    () -> validationService.initiateValidation(file, LocalDate.now())
            );

            assertTrue(ex.getMessage().contains("File is too small"));
        }

        @Test
        @DisplayName("Upload Security: Double extensions trigger rejection")
        void testInitiateValidation_DoubleExtension() {
            byte[] validContent = new byte[2048]; // 2KB valid size
            MultipartFile file = new MockMultipartFile("file", "test.exe.xlsx", 
                    "application/vnd.openxmlformats-officedocument.spreadsheetml.sheet", validContent);

            CustomExceptions.FileProcessingException ex = assertThrows(
                    CustomExceptions.FileProcessingException.class,
                    () -> validationService.initiateValidation(file, LocalDate.now())
            );

            assertTrue(ex.getMessage().contains("multiple extensions are strictly prohibited"));
        }

        @Test
        @DisplayName("Upload Security: Invalid MIME type triggers rejection")
        void testInitiateValidation_InvalidMimeType() {
            byte[] validContent = new byte[2048];
            MultipartFile file = new MockMultipartFile("file", "valid_name.xlsx", 
                    "image/png", validContent); // Spoofed MIME

            CustomExceptions.FileProcessingException ex = assertThrows(
                    CustomExceptions.FileProcessingException.class,
                    () -> validationService.initiateValidation(file, LocalDate.now())
            );

            assertTrue(ex.getMessage().contains("Detected MIME type does not match Excel"));
        }
    }

    // ==================================================================================
    // TEST UTILITIES
    // ==================================================================================
    private ExcelRowData createMockRow(String branch, String currency, String cgl, BigDecimal amount, String type) {
        ExcelRowData row = new ExcelRowData();
        row.branch = branch;
        row.currency = currency;
        row.cgl = cgl;
        row.amount = amount;
        row.txnType = type;
        row.remarks = "Test remarks";
        
        // Initialize the synchronized lists used by the validator
        row.formatErrors = Collections.synchronizedList(new ArrayList<>());
        row.dbErrors = Collections.synchronizedList(new ArrayList<>());
        row.balErrors = Collections.synchronizedList(new ArrayList<>());
        return row;
    }
