To process 100,000+ records and return an immediate response while ensuring Kubernetes (K8s) compatibility, we must move to an Asynchronous Architecture.
Here is the strategy to make it "Production Optimized":
 * Async Processing: The API will receive the file and immediately return a processId (in milliseconds). The actual heavy lifting happens in a background thread.
 * Database State Management: Since K8s has multiple pods, we cannot store the status in the server's RAM (memory). We will store the processing status (QUEUED, PROCESSING, COMPLETED) and the result in the Database. This ensures that even if the polling request hits a different pod, it finds the data.
 * Frontend Polling: The Frontend will get the processId and ping the server every 2 seconds: "Is job 123 done?" until it says "Yes".

package com.fincore.JournalService.config;

import org.springframework.context.annotation.Bean;
import org.springframework.context.annotation.Configuration;
import org.springframework.scheduling.annotation.EnableAsync;
import org.springframework.scheduling.concurrent.ThreadPoolTaskExecutor;

import java.util.concurrent.Executor;

@Configuration
@EnableAsync
public class AsyncConfig {

    @Bean(name = "bulkUploadExecutor")
    public Executor bulkUploadExecutor() {
        ThreadPoolTaskExecutor executor = new ThreadPoolTaskExecutor();
        executor.setCorePoolSize(10); // Start with 10 threads
        executor.setMaxPoolSize(50);  // Scale up to 50 under load
        executor.setQueueCapacity(500); // Queue up to 500 requests
        executor.setThreadNamePrefix("JournalBulk-");
        executor.initialize();
        return executor;
    }
}

    public String initiateValidation(MultipartFile file, LocalDate postingDate) throws IOException {
        String requestId = UUID.randomUUID().toString();

        BulkUploadRequest request = new BulkUploadRequest();
        request.setRequestId(requestId);
        request.setStatus("PROCESSING");
        request.setStartTime(LocalDateTime.now());

        byte[] fileBytes = file.getBytes();
        String filename = file.getOriginalFilename();

        // Trigger Async Process
        processAsync(requestId, fileBytes, filename, postingDate);

        return requestId;
    }

    @Async("bulkUploadExecutor")
    public void processAsync(String requestId, byte[] fileBytes, String filename, LocalDate postingDate) {
        log.info("Starting Async Processing for ID: {}", requestId);
        try {
            List<ExcelRowData> parsedRows;
            
            // Re-create stream from bytes
            if (filename != null && (filename.toLowerCase().endsWith(".csv") || filename.toLowerCase().endsWith(".txt"))) {
                parsedRows = parseCsvBytes(fileBytes);
            } else {
                parsedRows = parseExcelBytes(fileBytes);
            }

            // Heavy Validation
            processValidationLogic(requestId, parsedRows, postingDate, filename != null && filename.toLowerCase().endsWith(".csv"));

        } catch (Exception e) {
            log.error("Async Processing Failed for ID: {}", requestId, e);
            updateStatus(requestId, "FAILED", "System Error: " + e.getMessage());
        }
    }

    private void processValidationLogic(String requestId, List<ExcelRowData> parsedRows, LocalDate postingDate, boolean isInputCsv) {
        try {
            // ... (Keep your optimized fetching logic here: A. Extract codes, B. Fetch DB) ...
            Set<String> branchesInFile = parsedRows.parallelStream().map(r -> r.branch).filter(s -> !s.isEmpty()).collect(Collectors.toSet());
            Set<String> cglsInFile = parsedRows.parallelStream().map(r -> r.cgl).filter(s -> !s.isEmpty()).collect(Collectors.toSet());
            Set<String> currenciesInFile = parsedRows.parallelStream().map(r -> r.currency).filter(s -> !s.isEmpty()).collect(Collectors.toSet());

            Set<String> validBranches = branchRepo.findAllById(branchesInFile).stream().map(BranchMaster::getCode).map(String::trim).collect(Collectors.toSet());
            Set<String> validCgls = cglRepo.findAllById(cglsInFile).stream().map(CglMaster::getCglNumber).map(String::trim).collect(Collectors.toSet());
            Set<String> validCurrencies = currencyRepo.findAllById(currenciesInFile).stream().map(CurrencyMaster::getCurrencyCode).map(String::trim).collect(Collectors.toSet());

            Map<String, BigDecimal> balanceMap = new HashMap<>();

            // 1. Validation Loop
            for (ExcelRowData data : parsedRows) {
                if (!validBranches.contains(data.branch)) data.errors.add("Invalid Branch Code");
                if (!validCurrencies.contains(data.currency)) data.errors.add("Invalid Currency");
                if (!validCgls.contains(data.cgl)) data.errors.add("Invalid CGL");
                
                if (data.productCode != null && !data.productCode.isEmpty() && !data.productCode.matches("\\d{8}")) {
                    data.errors.add("Product Code must be 8 digits");
                }
                if (data.amount == null || data.amount.compareTo(BigDecimal.ZERO) == 0) {
                    data.errors.add("Amount cannot be Zero");
                }
                if (data.remarks != null && data.remarks.length() > 30) data.errors.add("Remarks > 30 chars");

                if (validBranches.contains(data.branch) && validCurrencies.contains(data.currency) && data.amount != null) {
                    String cglType = (data.cgl != null && data.cgl.startsWith("5")) ? "MEMO" : "NORMAL";
                    String key = data.branch + "_" + data.currency + "_" + cglType;
                    balanceMap.put(key, balanceMap.getOrDefault(key, BigDecimal.ZERO).add(data.amount));
                }
            }

            // 2. Balancing Check
            for (ExcelRowData data : parsedRows) {
                if (!validBranches.contains(data.branch) || !validCurrencies.contains(data.currency)) continue;
                String cglType = (data.cgl != null && data.cgl.startsWith("5")) ? "MEMO" : "NORMAL";
                String key = data.branch + "_" + data.currency + "_" + cglType;
                
                BigDecimal netTotal = balanceMap.getOrDefault(key, BigDecimal.ZERO);
                if (netTotal.setScale(4, RoundingMode.HALF_UP).compareTo(BigDecimal.ZERO) != 0) {
                    data.errors.add(cglType + " Total Mismatch (" + netTotal.toPlainString() + ")");
                }
            }

            // 3. Prepare Result for DB
            long errorCount = parsedRows.stream().filter(r -> !r.errors.isEmpty()).count();
            BulkUploadRequest request = bulkUploadRepo.findById(requestId).orElse(new BulkUploadRequest());
            
            request.setTotalRows(parsedRows.size());
            request.setErrorCount(errorCount);
            request.setEndTime(LocalDateTime.now());

            // Prepare Preview (Top 100)
            List<Map<String, Object>> previewList = parsedRows.stream().limit(100).map(row -> {
                Map<String, Object> map = new HashMap<>();
                map.put("id", row.rowIndex);
                map.put("branch", row.branch);
                map.put("currency", row.currency);
                map.put("cgl", row.cgl);
                map.put("amount", row.amount.abs().toPlainString());
                map.put("txnType", row.amount.compareTo(BigDecimal.ZERO) < 0 ? "Credit" : "Debit");
                map.put("remarks", row.remarks);
                map.put("productCode", row.productCode);
                return map;
            }).collect(Collectors.toList());
            request.setPreviewDataJson(objectMapper.writeValueAsString(previewList));

            if (errorCount > 0) {
                request.setStatus("ERROR");
                request.setMessage("Validation failed with " + errorCount + " errors.");
                request.setErrorFile(generateErrorExcel(parsedRows));
            } else {
                request.setStatus("SUCCESS");
                request.setMessage("Validation Successful!");
                if (!isInputCsv) {
                    request.setSuccessFile(generateSuccessCsv(parsedRows, postingDate));
                }
            }

            bulkUploadRepo.save(request);
            log.info("Async Processing Completed for ID: {}", requestId);

        } catch (Exception e) {
            log.error("Logic Error", e);
            updateStatus(requestId, "FAILED", e.getMessage());
        }
    }

    private List<ExcelRowData> parseWorkbookLogic(Workbook workbook) {
        List<ExcelRowData> rows = new ArrayList<>();
        Sheet sheet = workbook.getSheetAt(0);
        DataFormatter formatter = new DataFormatter();
        for (Row row : sheet) {
            if (isRowEmpty(row)) continue;
            ExcelRowData data = new ExcelRowData();
            data.rowIndex = row.getRowNum();
            data.branch = parseCode(row.getCell(0), formatter);     
            data.currency = getCellValueAsString(row.getCell(1), formatter).toUpperCase().trim(); 
            data.cgl = parseCode(row.getCell(2), formatter);        
            String amountRaw = getCellValueAsString(row.getCell(3), formatter).trim(); 
            String typeRaw = getCellValueAsString(row.getCell(4), formatter).trim().toUpperCase(); 
            data.remarks = getCellValueAsString(row.getCell(5), formatter);
            data.productCode = parseCode(row.getCell(6), formatter); 
            data.txnType = typeRaw; 
            parseAmountAndType(data, amountRaw, typeRaw);
            rows.add(data);
        }
        return rows;
    }

    private List<ExcelRowData> parseCsvReaderLogic(BufferedReader reader) throws IOException {
        List<ExcelRowData> rows = new ArrayList<>();
        String line;
        int lineNum = 0;
        while ((line = reader.readLine()) != null) {
            if (line.trim().isEmpty()) continue;
            String[] cols = line.split(",", -1); 
            if (cols.length < 9) continue; 
            ExcelRowData data = new ExcelRowData();
            data.rowIndex = lineNum++;
            data.branch = cols[4].trim();
            data.currency = cols[5].trim();
            data.cgl = cols[6].trim();
            String amountRaw = cols[7].trim();
            String typeRaw = cols[8].trim();
            data.remarks = (cols.length > 9) ? cols[9].trim() : "";
            if (cols.length > 10) { String prod = cols[10].trim(); data.productCode = "A".equalsIgnoreCase(prod) ? "" : prod; }
            data.txnType = typeRaw;
            parseAmountAndType(data, amountRaw, typeRaw);
            rows.add(data);
        }
        return rows;
    }

    private void parseAmountAndType(ExcelRowData data, String amountRaw, String typeRaw) {
        try {
            String cleanAmountStr = CLEAN_AMOUNT_REGEX.matcher(amountRaw).replaceAll("");
            if (cleanAmountStr.isEmpty()) throw new NumberFormatException("Empty");
            BigDecimal amt = new BigDecimal(cleanAmountStr);
            boolean isCredit = false;
            if (typeRaw.startsWith("C") || typeRaw.contains("CR")) isCredit = true;
            if (amountRaw.toUpperCase().contains("CR")) isCredit = true;
            if (amountRaw.toUpperCase().contains("DR")) isCredit = false;
            if (amountRaw.startsWith("-")) isCredit = true;
            data.amount = isCredit ? amt.abs().negate() : amt.abs();
        } catch (Exception e) { data.amount = BigDecimal.ZERO; }
    }

        return ResponseEntity.ok(Map.of("status", "QUEUED", "requestId", requestId));
    }

    // 2. CHECK STATUS (Polling)
    @GetMapping("/bulk-status/{requestId}")
    public ResponseEntity<Map<String, Object>> checkBulkStatus(@PathVariable String requestId) {
        BulkUploadRequest request = bulkUploadRepo.findById(requestId)
                .orElseThrow(() -> new ResourceNotFoundException("Request ID not found"));

        Map<String, Object> response = new HashMap<>();
        response.put("status", request.getStatus()); // PROCESSING, SUCCESS, ERROR
        response.put("message", request.getMessage());
        
        if ("SUCCESS".equals(request.getStatus()) || "ERROR".equals(request.getStatus())) {
            response.put("totalRows", request.getTotalRows());
            response.put("errorCount", request.getErrorCount());
            
            // Return files as Base64 if they exist
            if (request.getErrorFile() != null) {
                response.put("errorReportBase64", Base64.getEncoder().encodeToString(request.getErrorFile()));
            }
            if (request.getSuccessFile() != null) {
                response.put("successFileBase64", Base64.getEncoder().encodeToString(request.getSuccessFile()));
            }
            
            // Preview Data
            try {
                if (request.getPreviewDataJson() != null) {
                    ObjectMapper mapper = new ObjectMapper();
                    response.put("previewData", mapper.readTree(request.getPreviewDataJson()));
                }
            } catch (Exception e) {
                log.error("JSON Parse Error", e);
            }
        }
        
        return ResponseEntity.ok(response);
    }
