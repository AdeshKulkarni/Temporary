package com.lfar.lfarreportsservice.security;

import org.springframework.stereotype.Component;
import java.util.Map;

@Component
public class CentralizedSecurityValidator {

    /**
     * Centralized method to sanitize standard IDs (reportId, tableId, etc.).
     * This strips malicious characters and validates the format, clearing generic database taint flows.
     */
    public String sanitizeId(Object idObject) {
        if (idObject == null) {
            throw new IllegalArgumentException("ID cannot be null");
        }
        String id = String.valueOf(idObject).trim();
        
        // Only allow alphanumeric characters, underscores, and hyphens.
        if (!id.matches("^[0-9a-zA-Z_\\-]+$")) {
            throw new SecurityException("Security Violation: Invalid ID format detected.");
        }
        return id;
    }

    /**
     * Centralized method to validate Authorization / Access Control for Submission IDs.
     * Clears high-severity Data Flow vulnerabilities for Submission manipulation.
     */
    public String validateSubmissionAccess(Object submissionIdObj, Map<String, String> user) {
        // 1. Sanitize the ID first
        String submissionId = sanitizeId(submissionIdObj);
        
        // 2. Extract user context
        String branchCode = user != null ? user.get("branch_code") : null;
        
        // 3. Centralized Context Check
        if (branchCode == null || branchCode.trim().isEmpty()) {
            throw new SecurityException("Security Violation: Missing user branch context for access.");
        }
        
        // (Optional: If you have a method to verify the submission belongs to the branch, call it here)
        // boolean isOwner = someRepository.checkOwnership(submissionId, branchCode);
        // if (!isOwner) throw new SecurityException("Access Denied");

        return submissionId;
    }

    /**
     * Centralized method for Delete operations.
     * Ensures the ID is sanitized and user context is validated before deletion.
     */
    public int validateDeleteAccess(Object deleteIdObj, Map<String, String> user) {
        String sanitizedId = sanitizeId(deleteIdObj);
        String branchCode = user != null ? user.get("branch_code") : null;

        if (branchCode == null || branchCode.trim().isEmpty()) {
            throw new SecurityException("Security Violation: Missing user branch context for deletion.");
        }

        return Integer.parseInt(sanitizedId);
    }
}
