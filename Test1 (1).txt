package com.tcs.fincore.CommonRequestService.service.strategy;

import com.fasterxml.jackson.core.JsonProcessingException;
import com.fasterxml.jackson.databind.ObjectMapper;
import com.tcs.fincore.CommonRequestService.dto.payload.MocPayloadDto;
import com.tcs.fincore.CommonRequestService.model.CommonReq;
import com.tcs.fincore.CommonRequestService.model.master.FnMoc;
import com.tcs.fincore.CommonRequestService.repository.FnMocRepository;
import lombok.RequiredArgsConstructor;
import lombok.extern.slf4j.Slf4j;
import org.springframework.stereotype.Component;

@Component("MOC") // Name MUST match the RequestType enum value
@Slf4j
@RequiredArgsConstructor
public class MocApprovalStrategy implements RequestApprovalStrategy { // Assuming your approval interface is named this

    private final FnMocRepository fnMocRepository;
    private final ObjectMapper objectMapper;

    @Override
    public void processApproval(CommonReq request) throws JsonProcessingException {
        log.info("Executing MOC approval strategy for request ID: {}", request.getId());

        // 1. Deserialize the payload from the request into our MOC DTO
        MocPayloadDto payload = objectMapper.readValue(request.getPayload(), MocPayloadDto.class);

        // 2. Create the master entity from the payload
        FnMoc masterRecord = new FnMoc();
        masterRecord.setTmmocPdate(payload.getTmmocPdate());
        masterRecord.setTmmocBranch(payload.getTmmocBranch());
        masterRecord.setCurrency(payload.getCurrency());
        masterRecord.setTmmocCgl(payload.getTmmocCgl());
        masterRecord.setTmmocAmt(payload.getTmmocAmt());
        masterRecord.setTxnType(payload.getTxnType());
        masterRecord.setProductType(payload.getProductType());
        masterRecord.setRemarks(payload.getRemarks());
        masterRecord.setArFlag("A"); // 'A' for Accepted

        // 3. Save to the final FN_MOC master table
        fnMocRepository.save(masterRecord);
        log.info("Successfully saved MOC data to FN_MOC master table for targetId: {}", request.getTargetId());
    }
}
