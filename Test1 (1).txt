package com.tcs.fincore.MOCService.Models;
import com.tcs.fincore.MOCService.Models.enums.ChangeType;
import com.tcs.fincore.MOCService.Models.enums.RequestStatus;
import jakarta.persistence.*;
import lombok.Data;
import java.time.LocalDateTime;

@Entity
@Table(name = "MOC_REQUEST")
@Data
public class MocRequest {
     @Id
     @GeneratedValue(strategy = GenerationType.SEQUENCE, generator = "MOC_REQUEST_SEQ")
     @SequenceGenerator(name = "MOC_REQUEST_SEQ", sequenceName = "MOC_REQUEST_SEQ", allocationSize = 1)
     @Column(name = "REQ_ID")
     private Long id;

     @Enumerated(EnumType.STRING)
     @Column(name = "REQ_STATUS", length = 10)
     private RequestStatus requestStatus;

     @Enumerated(EnumType.STRING)
     @Column(name = "CHANGE_TYPE", length = 10)
     private ChangeType changeType;

     @Column(name = "REQ_DATE", updatable = false)
     private LocalDateTime requestDate = LocalDateTime.now();

     @Column(name = "CREATOR_ID", length = 12, updatable = false)
     private String creatorId;

     @Column(name = "EXECUTOR_ID", length = 12)
     private String executorId;

     @Column(name = "EXECUTION_DATE")
     private LocalDateTime executionDate;

     @Column(name = "EXECUTOR_REMARKS", length = 500)
     private String executorRemarks;

     @Lob
     @Column(name = "PAYLOAD")
     private String payload;

     // --- targetId field REMOVED from here ---

     @Column(name = "BATCH_ID", length = 50)
     private String batchId;

    
     @Column(name = "JOURNAL_ID", length = 50) 
     private String journalId;
     
     @Column(name = "COMMON_BATCH_REMARKS" , length = 200)
     private String commonBatchRemarks;
     
}
