package com.fincore.VoucherService.Controllers;


import com.fincore.VoucherService.Dto.BulkUploadStateDto;
import com.fincore.VoucherService.Exception.CustomExceptions;
import com.fincore.VoucherService.Service.validation.ValidationReportGenerator;
import com.fincore.VoucherService.Service.validation.VoucherBulkValidationService;
import jakarta.servlet.http.HttpServletRequest;
import lombok.RequiredArgsConstructor;
import lombok.extern.slf4j.Slf4j;
import org.springframework.core.io.ByteArrayResource;
import org.springframework.core.io.Resource;
import org.springframework.format.annotation.DateTimeFormat;
import org.springframework.http.HttpHeaders;
import org.springframework.http.MediaType;
import org.springframework.http.ResponseEntity;
import org.springframework.web.bind.annotation.*;
import org.springframework.web.multipart.MultipartFile;

import java.io.IOException;
import java.time.LocalDate;
import java.util.Map;

@RestController
@RequestMapping("/api/voucher")
@RequiredArgsConstructor
@Slf4j
public class VoucherBulkUploadController {

    private final VoucherBulkValidationService voucherBulkValidationService;
    private final ValidationReportGenerator validationReportGenerator;
    /**
     * Entry point for Journal File uploads. Validates file existence and starts parsing.
     */

    @PostMapping(value = "/bulk-validate-init", consumes = MediaType.MULTIPART_FORM_DATA_VALUE)
    public ResponseEntity<Map<String, Object>> initiateValidation(
            @RequestParam("file") MultipartFile file,
            @RequestParam("postingDate") @DateTimeFormat(pattern = "dd/MM/yyyy") LocalDate postingDate,
            HttpServletRequest request) {

        if (file == null || file.isEmpty()) {
            return ResponseEntity.status(org.springframework.http.HttpStatus.BAD_REQUEST)
                    .body(Map.of("message", "File is missing or empty."));
        }

        try {
            String reqId = voucherBulkValidationService.initiateValidation(file, postingDate);

            HttpHeaders headers = new HttpHeaders();
            headers.setCacheControl(org.springframework.http.CacheControl.noCache().mustRevalidate().sMaxAge(java.time.Duration.ZERO));
            headers.setPragma("no-cache");
            headers.setExpires(0L);
            headers.set("X-Content-Type-Options", "nosniff");
            headers.set("Content-Security-Policy", "default-src 'self'");

            return ResponseEntity.ok()
                    .headers(headers)
                    .body(Map.of("status", "QUEUED", "requestId", reqId));

        } catch (CustomExceptions.FileProcessingException e) {
            // FIX: Explicitly catch our custom validation errors and send the EXACT message back to React
            return ResponseEntity.status(org.springframework.http.HttpStatus.BAD_REQUEST)
                    .body(Map.of("message", e.getMessage()));

        } catch (Exception e) {
            // Fallback for unexpected system errors
            log.error("VALIDATION_INIT_FAIL | File: {} | Error: {}", file.getOriginalFilename(), e.getMessage());
            return ResponseEntity.status(org.springframework.http.HttpStatus.INTERNAL_SERVER_ERROR)
                    .body(Map.of("message", "An unexpected error occurred during upload."));
        }
    }


    @GetMapping("/bulk-status/{requestId}")
    public ResponseEntity<BulkUploadStateDto> checkStatus(@PathVariable String requestId) {
        BulkUploadStateDto state = voucherBulkValidationService.getState(requestId);
        return state != null ? ResponseEntity.ok(state) : ResponseEntity.notFound().build();
    }


    /**
     * Downloads the error report excel file.
     */
    @GetMapping("/download-bulk-file/{requestId}")
    public ResponseEntity<Resource> downloadFile(@PathVariable String requestId, @RequestParam String type) {
        byte[] data = voucherBulkValidationService.getFileBytes(requestId, type);
        if (data == null) {
            log.warn("DOWNLOAD_FAIL | File not found | ReqID: {} | Type: {}", requestId, type);
            return ResponseEntity.notFound().build();
        }

        // Retrieves the strict date format (e.g., "2026-06-18") without timestamps
        String filename = "Error_Report" + ".xlsx";


        // Exclusively sets the MIME type to Excel (.xlsx) since CSV is no longer needed
        MediaType mediaType = MediaType.parseMediaType("application/vnd.openxmlformats-officedocument.spreadsheetml.sheet");

        return ResponseEntity.ok()
                .header(HttpHeaders.CONTENT_DISPOSITION, "attachment; filename=\"" + filename + "\"")
                .contentType(mediaType)
                .body(new ByteArrayResource(data));
    }

    @GetMapping("/download-template")
    public ResponseEntity<org.springframework.core.io.Resource> downloadTemplate() throws IOException {
        return ResponseEntity.ok().header(HttpHeaders.CONTENT_DISPOSITION, "attachment; filename=\"Voucher_Template.xlsx\"").body(new ByteArrayResource(validationReportGenerator.generateExcelTemplateWithGuidelines()));
    }

}

@RestController
@RequestMapping("/api/voucher")
@RequiredArgsConstructor
@Slf4j
public class VoucherQueryController {



    private final VoucherQueryService queryService;
    private final JsEodMasterService jsEodMasterService;
    private final RequestVoucherService requestVoucherService;
    private final JwtUtil jwtUtil;



    /**
     * Safeguard: Forces pagination if batch rows exceed 2000 to avoid Server OOM.
     */
    @GetMapping("/by-batch/{batchId}")
    public ResponseEntity<?> getRequestsByBatchId(@PathVariable String batchId) {
        long count = queryService.getRequestCountByBatchId(batchId);
        if (count > 2000) return ResponseEntity.status(HttpStatus.PAYLOAD_TOO_LARGE).body(Map.of("error", "Batch too large.", "suggestion", "/api/voucher/by-batch-paginated/" + batchId));
        return ResponseEntity.ok(queryService.getRequestsByBatchId(batchId));
    }



    @GetMapping("/by-batch-paginated/{batchId}")
    public ResponseEntity<?> getRequestsByBatchIdPaginated(@PathVariable String batchId, @RequestParam(defaultValue = "0") int page, @RequestParam(defaultValue = "10") int size) {
        return ResponseEntity.ok(queryService.getRequestsByBatchIdPaginated(batchId, PageRequest.of(page, size)));
    }

    @GetMapping("/download-batch/{batchId}")
    public ResponseEntity<?> downloadBatchExcel(@PathVariable String batchId, @RequestHeader("Authorization") String token) {
        try {
            if (!queryService.isUserAuthorizedForBatch(batchId, jwtUtil.getUserIdFromToken(token.substring(7)), jwtUtil.getUserRoleFromToken(token.substring(7)))) {
                return ResponseEntity.status(HttpStatus.FORBIDDEN).body(Map.of("message", "You are not authorized to view this batch."));
            }

            // Simple synchronous call returning byte[]
            byte[] fileData = queryService.generateBatchExcel(batchId);

            org.springframework.http.HttpHeaders headers = new org.springframework.http.HttpHeaders();
            headers.setCacheControl(org.springframework.http.CacheControl.noCache().mustRevalidate().sMaxAge(java.time.Duration.ZERO));
            headers.set(org.springframework.http.HttpHeaders.CONTENT_DISPOSITION, "attachment; filename=\"Batch_" + batchId + ".xlsx\"");
            headers.set(org.springframework.http.HttpHeaders.CONTENT_TYPE, "application/vnd.openxmlformats-officedocument.spreadsheetml.sheet");

            return ResponseEntity.ok().headers(headers).body(fileData);

        } catch (CustomExceptions.InvalidDataException e) {
            return ResponseEntity.status(HttpStatus.BAD_REQUEST).body(Map.of("message", e.getMessage()));
        } catch (Exception e) {
            log.error("DOWNLOAD_ERROR | Batch: {} | Reason: {}", batchId, e.getMessage());
            return ResponseEntity.status(HttpStatus.INTERNAL_SERVER_ERROR).body(Map.of("message", "Failed to generate Excel file."));
        }
    }

    @GetMapping("/pending-requests")
    public List<JournalRequest> getPendingRequests(@RequestHeader("Authorization") String token) {
        return queryService.getPendingRequests(jwtUtil.getUserIdFromToken(token), jwtUtil.getUserRoleFromToken(token));
    }

    @GetMapping("/checker/pending-approvals")
    public ResponseEntity<List<Map<String, Object>>> getPendingApprovals(@RequestHeader("Authorization") String token) {
        Integer userRole = jwtUtil.getUserRoleFromToken(token);
        String userId = jwtUtil.getUserIdFromToken(token);
        return ResponseEntity.ok(queryService.getPendingBatchesForChecker(userRole, userId));
    }

    // ==================================================================================
    // 2. DASHBOARDS & SUMMARIES
    // ==================================================================================

    @GetMapping("/all-requests-summary")
    public ResponseEntity<Map<String, Object>> getAllRequestsSummary(@RequestHeader("Authorization") String token , @RequestParam(defaultValue = "0") int page,
                                                                           @RequestParam(defaultValue = "10") int size) {
        String userId = jwtUtil.getUserIdFromToken(token);
        Integer userRole = jwtUtil.getUserRoleFromToken(token);
        Map<String, Object> response = queryService.getAllRequestsSummary(userId, userRole, page, size);
        return ResponseEntity.ok(response);
    }


    @GetMapping("/workflow-history/{batchId}")
    public ResponseEntity<List<Map<String, Object>>> getWorkflowHistory(@PathVariable String batchId) {
        return ResponseEntity.ok(queryService.getWorkflowHistory(batchId));
    }

    @GetMapping("/batch-summary/{batchId}")
    public ResponseEntity<Map<String, Object>> getBatchSummaryDetails(@PathVariable String batchId) {
        return ResponseEntity.ok(queryService.getBatchSummaryDetails(batchId));
    }

    // ==================================================================================

    @GetMapping("/current-posting-date")

    @GetMapping("/eod_status")
    public ResponseEntity<List<JsEodMasterDTO>> getEodStatus() { return new ResponseEntity<>(jsEodMasterService.getAllData(), HttpStatus.OK); }

    @GetMapping("/filter")
    public ResponseEntity<List<RequestVoucher>> getFilteredVouchers(@RequestParam(required = false) String category, @RequestParam(required = false) String role, @RequestParam(required = false) @DateTimeFormat(pattern = "yyyy-MM-dd") LocalDate date) {
        return ResponseEntity.ok(requestVoucherService.getFilteredVouchers(category, role, date));
    }

    @GetMapping("/workflow/steps/{workflowCode}")
    public ResponseEntity<List<Map<String, Object>>> getWorkflowSteps(@PathVariable String workflowCode) {
        return ResponseEntity.ok(queryService.getWorkflowSteps(workflowCode)); }

    @GetMapping("/active-roles")
    public ResponseEntity<Map<String, String>> getActiveRoles() { return ResponseEntity.ok(queryService.getActiveRoles()); }

    public ResponseEntity<Map<String, Object>> getVidDetails(@PathVariable String vid) { return ResponseEntity.ok(queryService.getVidDetails(vid)); }

        Map<String, Object> response = new HashMap<>(); response.put("data", queryService.getAvailableVouchersVid(roleId)); response.put("message", "Success"); response.put("status", "OK");
        return ResponseEntity.ok(response);
    }
