void solve(int[] arr, int i, List<Integer> temp) {

    if (i == arr.length) {
        System.out.println(temp);
        return;
    }

    // INCLUDE
    temp.add(arr[i]);
    solve(arr, i + 1, temp);

    // BACKTRACK
    temp.remove(temp.size() - 1);

    // EXCLUDE
    solve(arr, i + 1, temp);
}
