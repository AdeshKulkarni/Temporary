class Solution {

    public List<Integer> SizeList(int n ,  List<List<Integer>> adj , int node , List<Boolean> visit , int count) {

        List<Integer> ans = new ArrayList<>();

        visit.set(node, true); 
        visit.get(node); 
        // D.p(node);

        for(int i=0; i<adj.get(node).size(); i++) {

            int nabar = adj.get(node).get(i);
            D.p(count);

            if(visit.get(nabar) == false) {
                count++;
                SizeList(n , adj , nabar , visit , count);
            }          
            ans.add(count);
                      
        }


        return ans;

    }
    public long countPairs(int n, int[][] edges) {

        List<List<Integer>> adj = new ArrayList<>(n);

        for (int i = 0; i < n; i++) {
            adj.add(new ArrayList<>());
        }

        for(int i=0; i<edges.length; i++) {

            int[] temp = edges[i];

            int src = temp[0];
            int dis = temp[1];

            adj.get(src).add(dis);
            adj.get(dis).add(src);
                

        }

        List<Boolean> visit = new ArrayList<>(Collections.nCopies(n, false));

        List<Integer> anss = SizeList(n , adj , 0 ,visit , 0);

        D.p(anss);





        return 2323L;
        
    }
}
