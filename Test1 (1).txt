import * as React from "react";
import PropTypes from "prop-types";
import { Tabs, Tab, Box } from "@mui/material";
import CurrencyRupeeIcon from "@mui/icons-material/CurrencyRupee";
import PendingActionsIcon from "@mui/icons-material/PendingActions";
import CurrencyRateChangeTab from "./CurrencyRateChangeTab";
import MyCurrencyRateRequestsTab from "./MyCurrencyRateRequestsTab";
import { CurrencyRateMasterStyles } from "./CurrencyRateChangeStyles";

function CustomTabPanel(props) {
  const { children, value, index, ...other } = props;

  return (
    <Box
      role="tabpanel"
      hidden={value !== index}
      id={`simple-tabpanel-${index}`}
      aria-labelledby={`simple-tab-${index}`}
      {...other}
    >
      {value === index && (
        <Box sx={CurrencyRateMasterStyles.customTabPanelBox}>{children}</Box>
      )}
    </Box>
  );
}

CustomTabPanel.propTypes = {
  children: PropTypes.node,
  index: PropTypes.number.isRequired,
  value: PropTypes.number.isRequired,
};

function a11yProps(index) {
  return {
    id: `simple-tab-${index}`,
    "aria-controls": `simple-tabpanel-${index}`,
  };
}

export default function CurrencyRateChange() {
  const [value, setValue] = React.useState(0);

  const handleChange = (event, newValue) => {
    setValue(newValue);
  };

  return (
    <Box>
      <Box sx={CurrencyRateMasterStyles.tabBox}>
        <Tabs
          sx={CurrencyRateMasterStyles.tabs}
          value={value}
          onChange={handleChange}
          aria-label="basic tabs example"
          variant="fullWidth"
        >
          <Tab
            label="Manage Currency Rate"
            icon={<CurrencyRupeeIcon />}
            iconPosition="start"
            {...a11yProps(0)}
          />
          <Tab
            icon={<PendingActionsIcon />}
            iconPosition="start"
            label="My Requests"
            {...a11yProps(1)}
          />
        </Tabs>
      </Box>
      <CustomTabPanel value={value} index={0}>
        <CurrencyRateChangeTab />
      </CustomTabPanel>
      <CustomTabPanel value={value} index={1}>
        <MyCurrencyRateRequestsTab />
      </CustomTabPanel>
    </Box>
  );
}
