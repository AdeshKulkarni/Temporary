1. src/hooks/useApi.js (Frontend)

// useApi.js
import { useEffect, useRef, useState, useCallback, useMemo } from "react";
import axios from "axios";
import { useLocation, useNavigate } from "react-router-dom";
import useCustomSnackbar from "../utils/useCustomSnackbar";
import { useSelector } from "react-redux";
import { resolveConfig } from "../config/EnvironmentConfig";
import { convertToKebabCase } from "../utils/CommonUtilities";

const useApi = () => {
  const [data, setData] = useState(null);
  const [error, setError] = useState(null);
  const [loading, setLoading] = useState(false);

  const navigate = useNavigate();
  const location = useLocation();
  const snackbar = useCustomSnackbar();

  const token = useSelector((state) => state.auth.token);
  const menus = useSelector((s) => s.menus.menus);
  const selectedMenuItem = useSelector((s) => s.menus.selectedMenuItem);

  const activeRequestsRef = useRef(new Set());

  const refreshToken = async () => {
    try {
      await axios.post(
        `${resolveConfig("/LS") ?? ""}/refresh-token`,
        {},
        { withCredentials: true }
      );
    } catch (err) {
      console.error("Token refresh failed", err);
    }
  };

  const cancelAllRequests = () => {
    activeRequestsRef.current.forEach((controller) => {
      if (controller && !controller.signal.aborted) {
        controller.abort();
      }
    });
    activeRequestsRef.current.clear();
    setLoading(false);
  };

  useEffect(() => {
    return () => cancelAllRequests();
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, []);

  const findMenuItem = useCallback((items, route) => {
    if (!items || !items.length) return null;

    for (const item of items) {
      if (item.route === route) return item;
      if (item.children && item.children.length > 0) {
        const isSubmenuSelectionPage = route.includes("/choose-option");
        if (isSubmenuSelectionPage) {
          const prefix = "/choose-option/";
          const startIndex = route.indexOf(prefix) + prefix.length;
          const parameter = route.substring(startIndex);
          if (parameter === convertToKebabCase(item.title)) {
            return item;
          }
        }
        const childMatch = findMenuItem(item.children, route);
        if (childMatch) return childMatch;
      }
    }
    return null;
  }, []);

  const currentMenuFromRoute = useMemo(() => {
    if (!menus || !menus.length) return null;
    return findMenuItem(menus, location.pathname);
  }, [menus, location.pathname, findMenuItem]);

  const xRequestType = useMemo(() => {
    return (
      currentMenuFromRoute?.requestType || selectedMenuItem?.requestType || "*"
    );
  }, [currentMenuFromRoute, selectedMenuItem]);

  const callApi = useCallback(
    async (
      url,
      payload = null,
      method = "GET",
      responseType = "json",
      contentType = "application/json",
      extraConfig = {},
      returnDataOnly = true
    ) => {
      const parts = url.split("/");
      const service = parts[1];
      const resolved = resolveConfig(`/${service}`);
      parts[1] = resolved;
      const Final_Url = parts.slice(1).join("/");

      const controller = new AbortController();
      activeRequestsRef.current.add(controller);

      setLoading(true);
      setError(null);

      try {
        // --- FIX STARTS HERE ---
        // 1. Detect if payload is FormData
        const isFormData = payload instanceof FormData;

        // 2. Build Headers
        const headers = {
          Authorization: `Bearer ${token}`,
          "X-Request-Type": xRequestType,
        };

        // 3. Only attach Content-Type if it is NOT FormData.
        // If it IS FormData, we leave it undefined so Axios/Browser sets the boundary.
        if (!isFormData) {
          headers["Content-Type"] = contentType;
        }
        // --- FIX ENDS HERE ---

        const config = {
          method: method.toUpperCase(),
          url: Final_Url,
          headers,
          responseType,
          withCredentials: true,
          signal: controller.signal,
          ...extraConfig,
        };

        if (payload !== null) {
          if (method.toUpperCase() === "GET") {
            config.params = payload;
          } else {
            config.data = payload;
          }
        }

        const response = await axios(config);

        if (!controller.signal.aborted) {
          if (response.data.status === 440) {
            console.error("You have logged in another session");
            navigate("/");
          }

          if (response?.data?.status === 401) {
            console.error("You have to login again as the session is expired");
            navigate("/");
          }

          setData(response.data);
          return returnDataOnly ? response.data : response;
        }
      } catch (err) {
        const status = err.response?.status;

        if (status === 440) {
          console.error("You have logged in another session");
          navigate("/");
        }

        if (!axios.isCancel(err) && !controller.signal.aborted) {
          console.error("API call error:", err);

          const apiMessage =
            err.response?.data?.result?.message ||
            err.response?.data?.message ||
            err.response?.statusText ||
            "Something went wrong";

          snackbar(apiMessage, "error");
          setError(err);
        }

        if (axios.isCancel(err)) {
          throw err;
        }
      } finally {
        activeRequestsRef.current.delete(controller);
        setLoading(false);
      }
    },
    [navigate, snackbar, token, xRequestType]
  );

  return { data, error, loading, callApi, cancelAllRequests, refreshToken };
};

export default useApi;

                      <TableCell align="center">{row.cgl}</TableCell>
                      <TableCell align="right">{row.amount}</TableCell>
                      <TableCell align="center"><Chip label={row.txnType} color={row.txnType.includes("Credit") ? "error" : "success"} size="small" variant="filled" /></TableCell>
                      <TableCell align="center">{row.remarks}</TableCell>
                      <TableCell align="center">{row.productCode || "-"}</TableCell>
                    </TableRow>
                  ))}
                </TableBody>
              </Table>
            </TableContainer>
            <TablePagination rowsPerPageOptions={[50, 100, 200]} component="div" count={previewRows.length} rowsPerPage={rowsPerPage} page={page} onPageChange={handleChangePage} onRowsPerPageChange={handleChangeRowsPerPage} labelDisplayedRows={({ from, to }) => `${from}-${to} of Preview`} />
            <Stack direction="row" sx={actionButtonsContainer}>
              <Button variant="outlined" onClick={handleBack} disabled={isCreatingBatch} startIcon={<ArrowBackIcon />}>Back</Button>
              <Button variant="text" onClick={resetAll} disabled={isCreatingBatch} startIcon={<RestartAltIcon />}>Cancel</Button>
              <Button variant="contained" color="success" size="large" onClick={handleCreateBatch} disabled={isCreatingBatch} startIcon={isCreatingBatch ? <CircularProgress size={20} color="inherit" /> : <SendIcon />}>{isCreatingBatch ? "Creating Batch..." : "Submit Batch"}</Button>
            </Stack>
          </Box>
        )}
      </Paper>
    </Box>
  );
};
export default JournalBulkUpload;
