import React, { useState, useMemo, useEffect } from "react";
import { Box } from "@mui/material";
import { DataGrid } from "@mui/x-data-grid";
import useApi from "../../../hooks/useApi"; 
import useCustomSnackbar from "../../../utils/useCustomSnackbar"; 

export default function CurrencyMaster() {
  const { callApi } = useApi();
  const showSnackBar = useCustomSnackbar();
  const [rows, setRows] = useState([]);
  const [isLoading, setIsLoading] = useState(true);

  const fetchData = async () => {
    setIsLoading(true);
    try {
      const response = await callApi("/currency-master", {}, "GET");
      
      if (response && response.data && response.data.length > 0) {
        setRows(response.data);
      } else {
        showSnackBar("No currency data found.", "info");
        setRows([]);
      }
    } catch (error) {
      console.error("Error fetching currency data:", error);
      showSnackBar(
        "Something went wrong while fetching data. Please try again later.",
        "error"
      );
      setRows([]);
    } finally {
      setIsLoading(false);
    }
  };

  useEffect(() => {
    fetchData();
  }, []); // The empty dependency array ensures this runs only once on mount

  const columns = useMemo(
    () => [
      {
        field: "currencyCode",
        headerName: "Currency Code",
        flex: 1,
        sortable: true,
      },
      {
        field: "currencyName",
        headerName: "Currency Name",
        flex: 2,
        sortable: true,
      },
      {
        field: "currencyRate",
        headerName: "Currency Rate",
        flex: 1,
        sortable: true,
        align: "right",
        headerAlign: "right",
      },
      {
        field: "createdAt",
        headerName: "Created At",
        flex: 1,
        sortable: true,
        align: "center",
        headerAlign: "center",
        valueGetter: (params) => params.value ? new Date(params.value).toLocaleString() : 'N/A',
      },
    ],
    []
  );

  return (
    <Box>
      <DataGrid
        rows={rows}
        columns={columns}
        disableRowSelectionOnClick
        pageSizeOptions={[5, 10, 25, 50]}
        initialState={{
          pagination: { paginationModel: { pageSize: 10, page: 0 } },
          sorting: { sortModel: [{ field: "createdAt", sort: "desc" }] },
        }}
        getRowId={(row) => row.currencyCode} 
        loading={isLoading}
        sx={{
          height: 600,
          width: "100%",
          "& .MuiDataGrid-cell:focus": {
            outline: "none",
          },
        }}
      />
    </Box>
  );
}
