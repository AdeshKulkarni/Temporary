public class Task {
    protected String status;

    public Task() {
        this.status = "OPEN";
    }

    public void completeTask() {
        this.status = "COMPLETED";
    }

    public String getStatus() {
        return this.status;
    }
}

public class DependentTask extends Task {
    private boolean dependenciesResolved;

    public DependentTask(boolean dependenciesResolved) {
        super();
        this.dependenciesResolved = dependenciesResolved;
    }

    @Override
    public void completeTask() {
        if (dependenciesResolved) {
            super.completeTask();
        } else {
            // Silently ignore the request to complete the task
            System.out.println("Cannot complete. Dependencies are still pending.");
        }
    }
}

public class TaskManager {
    // This method receives a mix of normal Tasks and DependentTasks
    public void closeAllTasks(List<Task> allTasks) {
        for (Task task : allTasks) {
            task.completeTask();
        }
        
        // The manager assumes that after calling completeTask() on the list, 
        // every single task is now "COMPLETED".
        System.out.println("Success: All tasks are now completed!");
    }
}
