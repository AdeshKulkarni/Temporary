    }

    private List<ticket> ticketListAgent;
    private boolean currentlyWorking;

    public void addTicketToList(ticket ticket) {
        ticketListAgent.add(ticket);
    }

    private void startWorking(ticket ticket , String agentName) {

        List<ticket> highPriority = ticketListAgent.stream()
                .filter(t -> t.getPrioritylevels() == priorityLevels.HIGH)
                .collect(Collectors.toList());

        currentlyWorking = true;

        for (ticket  t : highPriority) {
            // here we will first check who is try to work on this ticket is
            // that ticket assigend to that agent

            // logic using the agent name



            // getting the first ticket form this high property list
            int count = 0;
            if(t.isAssigned() == true)  {
                count++;
                if(count == 1) {
                    t.setTicketStatus(ticketStatus.IN_PROGRESS);
                    break;
                }

            }

        }

    }

    public void taskresolved(ticket ticket) {
        ticket.setTicketStatus(ticketStatus.RESOLVED);
        System.out.println("ticket resolved");
    }

        if(ticket.getTicketStatus() == ticketStatus.RESOLVED) {
            ticket.setTicketStatus(ticketStatus.CLOSED);
            System.out.println("ticket closed");
