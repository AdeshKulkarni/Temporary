package com.tcs.fincore.MOCService.Repository;

import java.util.List;

import org.springframework.data.jpa.repository.JpaRepository;
import org.springframework.stereotype.Repository;

import com.tcs.fincore.MOCService.Models.MocRequest;
import com.tcs.fincore.MOCService.Models.enums.RequestStatus;

@Repository
public interface MocRequestRepository extends JpaRepository<MocRequest, Long> {
    
    List<MocRequest> findByJournalIdAndRequestStatus(String journalId, RequestStatus status);
    
    List<MocRequest> findByCreatorId(String creatorId);
    List<MocRequest> findByRequestStatus(RequestStatus status);
    List<MocRequest> findByBatchIdAndRequestStatus(String batchId, RequestStatus status);

    List<MocRequest> findByJournalIdInAndRequestStatus(List<String> journalIds, RequestStatus status);
    
    List<MocRequest> findByJournalIdStartingWithAndRequestStatus(String journalIdPrefix, RequestStatus status);
    
    List<MocRequest> findByBatchId(String batchId);

    // --- NEW METHOD ---
    // This finds all requests that share the same journal ID prefix (e.g., "0000092")
    List<MocRequest> findByJournalIdStartingWith(String journalIdPrefix);
    // --- END OF NEW METHOD ---
}
