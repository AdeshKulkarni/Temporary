
import java.util.ArrayList;
import java.util.List;

public class allSongItrator implements SongItrator<song> {

    int index = 0;

    List<song> playList = new ArrayList<>();

    public allSongItrator(List<song> playList) {
        this.playList = playList;
    }

    @Override
    public boolean hasNext() {
       return index < playList.size();
    }
    ///  returning the every song one by one
    @Override
    public song next() {
       if(!hasNext()) {
           System.out.println("song is not there");
       }

       return playList.get(index++);
    }
}

import java.util.ArrayList;

public class noPopSongItrator implements SongItrator<song> {

    int index = 0;

    List<song> NonPopplayList = new ArrayList<>();

    public  noPopSongItrator(List<song> nonPopplayList) {
        nonPopplayList.removeIf(song -> song.getGenre() == GENRE.POP);
    }


    @Override
    public boolean hasNext() {
       return index < NonPopplayList.size();
    }


    ///  here we need to return the song by filtering the pop song like we dont want pop song in this

    @Override
    public song next()  {
        if(!hasNext()) {
             /// we will throw Exceptions
            System.out.println("song is not there");
        }

        return NonPopplayList.get(index++);
    }
}



import java.util.ArrayList;
import java.util.List;

public class PlayList {

    private List<song> playList = new ArrayList<>();

    public void addSong(song song) {
        // consider we have the Exception here we are not adding for now
        playList.add(song);
    }

    public void RemoveSong(song song) {
        // consider we have the Exception here we are not adding for now

        playList.remove(song);
    }

    public SongItrator<song> getAllItrator() {
        return new allSongItrator(playList);
    }

    public SongItrator<song> getPopFilterItrator() {
        return new noPopSongItrator(playList);
    }
}


public class song {

    private String title;
    private String artist;
    private GENRE genre;

    public song(String title, String artist, GENRE genre) {
        this.title = title;
        this.artist = artist;
        this.genre = genre;
    }

    public String getTitle() {
        return title;
    }



    public GENRE getGenre() {
        return genre;
    }


}



public interface SongItrator<T> {

    boolean hasNext();
    T next();

}
