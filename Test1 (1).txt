package com.tcs.fincore.CommonRequestService.repository;

import com.tcs.fincore.CommonRequestService.model.CglMaster;
import com.tcs.fincore.CommonRequestService.model.enums.AcClassification;
import org.springframework.data.jpa.repository.JpaRepository;
import org.springframework.stereotype.Repository;

import java.util.List;

@Repository
public interface CglMasterRepository extends JpaRepository<CglMaster, String> {

    // This method is used by the /filter endpoint
    List<CglMaster> findByAcClassification(AcClassification classification);

    // This is the original search
    List<CglMaster> findByCglNumberContaining(String query);

    // --- NEW METHOD ---
    // This is the new method that allows combined searching.
    List<CglMaster> findByCglNumberContainingAndAcClassification(String query, AcClassification classification);
}
