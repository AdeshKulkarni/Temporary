import React, { useState, useEffect } from 'react';
import { Box, Typography, Container, Paper } from '@mui/material';
import TemplateTable from './TemplateTable';
import { templateApi } from '../api/templateApi';

const TemplateConfiguration = () => {
  const [templates, setTemplates] = useState([]);

  useEffect(() => {

  }, []);

  return (

      <Paper sx={{width: "100%", p: 3 }}>
        <Typography variant="h5" gutterBottom sx={{ fontWeight: 'bold' }}>

export default TemplateConfiguration;



import React, { useState } from 'react';
import { DataGrid } from '@mui/x-data-grid';
import Box from '@mui/material/Box';
import Paper from '@mui/material/Paper';
import IconButton from '@mui/material/IconButton';
import Tooltip from '@mui/material/Tooltip';
import Dialog from '@mui/material/Dialog';
import DialogTitle from '@mui/material/DialogTitle';
import DialogContent from '@mui/material/DialogContent';
import DialogContentText from '@mui/material/DialogContentText';
import DialogActions from '@mui/material/DialogActions';
import Button from '@mui/material/Button';

// Icons (Edit removed)
import Visibility from '@mui/icons-material/Visibility';
import Delete from '@mui/icons-material/Delete';
import ReportProblemOutlined from '@mui/icons-material/ReportProblemOutlined';

// Internal Dialogs
import TemplateViewDialog from './TemplateViewDialog';
import { templateApi } from '../api/templateApi';

const TemplateTable = ({ data, onRefresh }) => {
  const [selectedTemplate, setSelectedTemplate] = useState(null);
  const [openView, setOpenView] = useState(false);
  const [openDelete, setOpenDelete] = useState(false);

  // --- Handlers ---
  const handleAction = (template, actionType) => {
    setSelectedTemplate(template);
    if (actionType === 'view') setOpenView(true);
    if (actionType === 'delete') setOpenDelete(true);
  };

  const closeAllDialogs = () => {
    setOpenView(false);
    setOpenDelete(false);
  };

  const confirmDelete = async () => {
    try {
      // Deletes based on templateId as seen in controller
      await templateApi.deleteTemplate(selectedTemplate.templateId);
      onRefresh(); 
      closeAllDialogs();
    } catch (error) {
      console.error("Delete failed:", error.message);
    }
  };

  // --- Data Grid Column Definitions ---
  const columns = [
    { 
      field: 'templateId', 
      headerName: 'ID', 
      flex: 1 
    },
    { 
      field: 'templateName', 
      headerName: 'Report Name', 
      flex: 2,
      renderCell: (params) => (
        <Box sx={{ fontWeight: 500 }}>{params.value}</Box>
      )
    },
    { 
      field: 'description', 
      headerName: 'Description', 
      flex: 3 
    },
    {
      field: 'actions',
      headerName: 'Actions',
      flex: 1, // Reduced flex since we have fewer buttons
      sortable: false,
      filterable: false,
    //   align: 'right',
    //   headerAlign: 'right',
      renderCell: (params) => (
        <Box sx={{ display: 'flex', gap: 1, width: '100%' }}>
          <Tooltip title="View & Run">
            <IconButton 
              size="small" 
              color="primary" 
              onClick={() => handleAction(params.row, 'view')}
            >
              <Visibility fontSize="small" />
            </IconButton>
          </Tooltip>

          <Tooltip title="Delete Template">
            <IconButton 
              size="small" 
              color="error" 
              onClick={() => handleAction(params.row, 'delete')}
            >
              <Delete fontSize="small" />
            </IconButton>
          </Tooltip>
        </Box>
      )
    }
  ];

  return (
    <Paper sx={{width: "100%"}} variant="outlined">
      <DataGrid
        rows={data}
        columns={columns}
        getRowId={(row) => row.templateId} // Maps unique ID from API
        pageSizeOptions={[5, 10, 25]}
        initialState={{
          pagination: { paginationModel: { pageSize: 10 } },
        }}
        disableRowSelectionOnClick
        sx={{
          border: 'none',
          '& .MuiDataGrid-columnHeader': {
            backgroundColor: (theme) => theme.palette.grey[50],
            fontWeight: 'bold'
          },
        }}
      />

      {/* View Dialog (Handles Variants and Export Formats) */}
      {selectedTemplate && (
        <TemplateViewDialog
          open={openView}
          handleClose={closeAllDialogs}
          template={selectedTemplate}
        />
      )}

      {/* Delete Confirmation */}
      <Dialog open={openDelete} onClose={closeAllDialogs}>
        <DialogTitle sx={{ display: 'flex', alignItems: 'center', gap: 1 }}>
          <ReportProblemOutlined color="error" /> Confirm Delete
        </DialogTitle>
        <DialogContent>
          <DialogContentText>
            Are you sure you want to delete <strong>{selectedTemplate?.templateName}</strong>?
          </DialogContentText>
        </DialogContent>
        <DialogActions sx={{ p: 2 }}>
          <Button onClick={closeAllDialogs}>Cancel</Button>
          <Button onClick={confirmDelete} color="error" variant="contained">
            Delete
          </Button>
        </DialogActions>
      </Dialog>
    </Paper>
  );
};

export default TemplateTable;



import React, { useState } from 'react';
import { 
  Dialog, DialogTitle, DialogContent, DialogActions, 
  Button, MenuItem, TextField, Typography, Divider 
} from '@mui/material';
import {Grid} from '@mui/material';

const TemplateViewDialog = ({ open, handleClose, template }) => {
  const [selectedVariant, setSelectedVariant] = useState('');
  const [formData, setFormData] = useState({});

  const activeVariant = template.variants.find(v => v.variantCode === selectedVariant);

  const handleInputChange = (name, value) => {
    setFormData(prev => ({ ...prev, [name]: value }));
  };

  return (
    <Dialog open={open} onClose={handleClose} maxWidth="sm" fullWidth>
      <DialogTitle>View Report: {template.templateName}</DialogTitle>
      <DialogContent dividers>
        <Grid container spacing={3}>
          {/* Variant Selection */}
          <Grid size={12}>
            <TextField
              select
              fullWidth
              label="Select Report Variant"
              value={selectedVariant}
              onChange={(e) => setSelectedVariant(e.target.value)}
            >
              {template.variants.map((v) => (
                <MenuItem key={v.variantCode} value={v.variantCode}>
                  {v.variantName}
                </MenuItem>
              ))}
            </TextField>
          </Grid>

          {/* Dynamic Inputs based on Params */}
          {activeVariant?.params.map((param) => (
            <Grid size={12} key={param.paramName}>
              <TextField
                fullWidth
                label={param.label}
                type={param.paramType === 'DATE' ? 'date' : 'text'}
                required={param.required}
                slotProps={{ inputLabel: { shrink: true } }}
                onChange={(e) => handleInputChange(param.paramName, e.target.value)}
              />
            </Grid>
          ))}

          {/* Export Format */}
          <Grid size={12}>
            <Divider sx={{ my: 1 }}>Export Options</Divider>
            <TextField select fullWidth label="Export Format" defaultValue="pdf">
              <MenuItem value="pdf">PDF Document</MenuItem>
              <MenuItem value="excel">Excel Spreadsheet</MenuItem>
              <MenuItem value="text">Plain Text</MenuItem>
            </TextField>
          </Grid>
        </Grid>
      </DialogContent>
      <DialogActions sx={{ p: 2 }}>
        <Button onClick={handleClose}>Cancel</Button>
        <Button variant="contained" color="primary" disabled={!selectedVariant}>
          Generate Report
        </Button>
      </DialogActions>
    </Dialog>
  );
};

export default TemplateViewDialog;



const BASE_URL = '/RB/template-config';

/**
 * Utility to handle fetch responses and throw meaningful errors
 * matching the backend's ApiResponse structure.
 */
const handleResponse = async (response) => {
  const result = await response.json().catch(() => ({ 
    message: "Failed to parse server response", 
    status: "ERROR" 
  }));

  if (!response.ok) {
    // Throws the message from your ResponseMessages (e.g., TEMPLATE_NOT_FOUND)
    throw new Error(result.message || `Error ${response.status}: ${response.statusText}`);
  }
  return result;
};

export const templateApi = {
  /**
   * Matches @GetMapping in your Controller
   * Fetches List<TemplateResponse>
   */
  getAllTemplates: async () => {
    try {
      const response = await fetch(`${BASE_URL}/get-templates`, {
        method: 'GET',
        headers: { 'Content-Type': 'application/json' },
      });
      return await handleResponse(response);
    } catch (error) {
      console.error("API Error [getAllTemplates]:", error.message);
      throw error;
    }
  },

  /**
   * Matches @GetMapping("/{templateId}") in your Controller
   * Fetches a single template with its variants and params
   */
  getTemplateById: async (templateId) => {
    try {
      const response = await fetch(`${BASE_URL}/${templateId}`, {
        method: 'GET',
        headers: { 'Content-Type': 'application/json' },
      });
      return await handleResponse(response);
    } catch (error) {
      console.error(`API Error [getTemplateById] ID: ${templateId}:`, error.message);
      throw error;
    }
  },

  /**
   * Matches @DeleteMapping("/{templateId}") in your Controller
   */
  deleteTemplate: async (templateId) => {
    try {
      const response = await fetch(`${BASE_URL}/${templateId}`, {
        method: 'DELETE',
        headers: { 'Content-Type': 'application/json' },
      });
      return await handleResponse(response);
    } catch (error) {
      console.error(`API Error [deleteTemplate] ID: ${templateId}:`, error.message);
      throw error;
    }
  },

  /**
   * Placeholder for Update functionality 
   * (Standardizing for your PUT mapping when implemented)
   */
  updateTemplate: async (templateId, data) => {
    try {
      const response = await fetch(`${BASE_URL}/save`, {
        method: 'PUT',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify(data),
      });
      return await handleResponse(response);
    } catch (error) {
      console.error(`API Error [updateTemplate] ID: ${templateId}:`, error.message);
      throw error;
    }
  }
};
