import React, { useState, useEffect, useCallback } from "react";
import {
  Box,
  Typography,
  Grid,
  TextField,
  Button,
  Paper,
  IconButton,
  MenuItem,
  CircularProgress,
  Autocomplete,
  Tooltip,
  Checkbox,
  FormControlLabel,
} from "@mui/material";
import AddIcon from "@mui/icons-material/Add";
import DeleteIcon from "@mui/icons-material/Delete";
import { debounce } from "lodash";

import useApi from "../../hooks/useApi";
import useCustomSnackbar from "../../utils/useCustomSnackbar";

const initialRowState = {
  id: Date.now(),
  branch: null,
  currency: "",
  cgl: null,
  amount: "",
  txnType: "",
  remarks: "",
  productCode: "",
};

const formatBigIntForDisplay = (val) => {
  if (typeof val !== "bigint") {
    val = BigInt(val || 0);
  }
  const str = val.toString().padStart(5, "0");
  const integer = str.slice(0, -4);
  const fractional = str.slice(-4);
  return `${integer || "0"}.${fractional}`;
};

export default function JournalPosting() {
  const { callApi } = useApi();
  const showSnackBar = useCustomSnackbar();

  const [rows, setRows] = useState([initialRowState]);
  const [currencyOptions, setCurrencyOptions] = useState([]);

  const [isSubmitting, setIsSubmitting] = useState(false);

  const [commonBatchRemarks, setCommonBatchRemarks] = useState("");
  const [postingDate, setPostingDate] = useState("");

  const [checked, setChecked] = useState(false);

  const [branchSearch, setBranchSearch] = useState({ loading: false, options: [] });
  const [cglSearch, setCglSearch] = useState({ loading: false, options: [] });

  const [activeBranchRowId, setActiveBranchRowId] = useState(null);
  const [activeCglRowId, setActiveCglRowId] = useState(null);

  useEffect(() => {
    const fetchInitialData = async () => {
      try {
        const dateResponse = await callApi("/JS/journals/current-posting-date", null, "GET");
        if (dateResponse) {
          try {
            const [year, month, day] = dateResponse.split("-");
            setPostingDate(`${day}-${month}-${year}`);
          } catch (e) {
            setPostingDate(dateResponse);
          }
        }

        const currencyResponse = await callApi("/CM/currency", null, "GET");
        if (currencyResponse?.success && Array.isArray(currencyResponse.data)) {
          setCurrencyOptions([...new Set(currencyResponse.data)]);
        }
      } catch (error) {
        showSnackBar("An error occurred while loading page data.", "error");
      }
    };
    fetchInitialData();
  }, [callApi, showSnackBar]);

  const handleAddRow = () => {
    const lastRow = rows[rows.length - 1];
    if (!lastRow.branch || !lastRow.currency || !lastRow.cgl || !lastRow.amount || !lastRow.txnType) {
      showSnackBar("Please fill all required fields (*) in the current row before adding a new one.", "warning");
      return;
    }
    setRows([...rows, { ...initialRowState, id: Date.now(), remarks: checked ? commonBatchRemarks : "" }]);
  };

  const handleChange = (event) => {
    const isChecked = event.target.checked;
    setChecked(isChecked);
    setRows(rows.map((row) => ({ ...row, remarks: isChecked ? commonBatchRemarks : "" })));
  };

  const handleDeleteRow = (id) => {
    if (rows.length > 1) {
      setRows(rows.filter((row) => row.id !== id));
    } else {
      showSnackBar("Cannot delete the last row.", "warning");
    }
  };

  const handleRowChange = (id, field, value) => {
    setRows(rows.map((row) => (row.id === id ? { ...row, [field]: value } : row)));
  };

  const searchBranches = useCallback(
    debounce(async (query, rowId) => {
      if (!query) {
        setBranchSearch({ loading: false, options: [] });
        setActiveBranchRowId(null);
        return;
      }
      setActiveBranchRowId(rowId);
      setBranchSearch((prev) => ({ ...prev, loading: true }));
      try {
        const response = await callApi(`/JS/branches-journal/search?query=${encodeURIComponent(query)}`, null, "GET");
        setBranchSearch({ loading: false, options: response || [] });
      } catch (error) {
        setBranchSearch({ loading: false, options: [] });
      }
    }, 300), [callApi]
  );

  const searchCgls = useCallback(
    debounce(async (query, rowId) => {
      if (!query) {
        setCglSearch({ loading: false, options: [] });
        setActiveCglRowId(null);
        return;
      }
      setActiveCglRowId(rowId);
      setCglSearch((prev) => ({ ...prev, loading: true }));
      try {
        const response = await callApi(`/JS/cgl-journal/search?query=${encodeURIComponent(query)}`, null, "GET");
        setCglSearch({ loading: false, options: response || [] });
      } catch (error) {
        setCglSearch({ loading: false, options: [] });
      }
    }, 300), [callApi]
  );

  const handleSubmit = async () => {
    setIsSubmitting(true);

    if (!commonBatchRemarks.trim()) {
      showSnackBar("Please enter Common Batch Remarks before submitting.", "error");
      setIsSubmitting(false);
      return;
    }

    const cglCheck = new Map();
    // Two separate maps for balancing checks
    const normalTotals = new Map();
    const memoTotals = new Map();
    const rowPayloads = [];

    for (const [index, row] of rows.entries()) {
      if (!row.branch || !row.currency || !row.cgl || !row.amount || !row.txnType) {
        showSnackBar(`Row ${index + 1} is incomplete. Please fill all required fields.`, "error");
        setIsSubmitting(false);
        return;
      }

      // 1. Duplicate CGL Check
      const key = `${row.branch.code}-${row.currency}`;
      const cglNumber = row.cgl.cglNumber;
      
      if (!cglCheck.has(key)) cglCheck.set(key, new Set());
      if (cglCheck.get(key).has(cglNumber)) {
        showSnackBar(`Duplicate CGL entry: "${cglNumber}" used multiple times for same branch/currency.`, "error");
        setIsSubmitting(false);
        return;
      }
      cglCheck.get(key).add(cglNumber);

      // 2. Parse Amount
      const [intPart = "0", fracPart = "0"] = (row.amount || "0").split(".");
      const fracPadded = fracPart.substring(0, 4).padEnd(4, "0");
      const amountBigInt = BigInt(intPart + fracPadded);

      // 3. Determine CGL Type (Memo starts with '5')
      const isMemo = cglNumber.startsWith("5");
      const targetMap = isMemo ? memoTotals : normalTotals;

      // 4. Initialize Totals if needed
      if (!targetMap.has(key)) {
        targetMap.set(key, { 
            totalDebit: 0n, 
            totalCredit: 0n, 
            branchName: row.branch.name, 
            branchCode: row.branch.code, 
            currency: row.currency 
        });
      }

      // 5. Accumulate
      const currentTotal = targetMap.get(key);
      if (row.txnType === "Debit") currentTotal.totalDebit += amountBigInt;
      else currentTotal.totalCredit += amountBigInt;

      // 6. Payload Construction
      rowPayloads.push({
        changeType: "ADD",
        branch: row.branch.code,
        currency: row.currency,
        cgl: cglNumber,
        amount: row.txnType === "Credit" ? "-" + row.amount : row.amount,
        productType: row.productCode,
        remarks: row.remarks,
        arFlag: "A",
      });
    }

    // 7. Validate Normal CGLs
    for (const [key, totals] of normalTotals.entries()) {
      if (totals.totalDebit !== totals.totalCredit) {
        showSnackBar(
          `Normal CGL Mismatch (${totals.branchCode} - ${totals.currency}): Debit ${formatBigIntForDisplay(totals.totalDebit)} != Credit ${formatBigIntForDisplay(totals.totalCredit)}`,
          "error"
        );
        setIsSubmitting(false);
        return;
      }
    }

    // 8. Validate Memo CGLs
    for (const [key, totals] of memoTotals.entries()) {
      if (totals.totalDebit !== totals.totalCredit) {
        showSnackBar(
          `Memo CGL Mismatch (${totals.branchCode} - ${totals.currency}): Debit ${formatBigIntForDisplay(totals.totalDebit)} != Credit ${formatBigIntForDisplay(totals.totalCredit)}`,
          "error"
        );
        setIsSubmitting(false);
        return;
      }
    }

    try {
      const response = await callApi("/JS/journals/create-batch", { commonBatchRemarks: commonBatchRemarks.trim(), rows: rowPayloads }, "POST");
      const newBatchId = response?.[0]?.batchId || "N/A";
      showSnackBar(`Batch ${newBatchId} created successfully!`, "success");
      setRows([{ ...initialRowState, id: Date.now() }]);
      setCommonBatchRemarks("");
      setChecked(false);
    } catch (error) {
      showSnackBar("Submission failed", "error");
    } finally {
      setIsSubmitting(false);
    }
  };

  return (
    <Box sx={{ p: 3, maxWidth: "98%" }}>
      <Paper elevation={3} sx={{ p: 3 }}>
        {/* ... (Header and Remarks Section - No Changes) ... */}
        <Box>
          <Box sx={{ display: "flex", alignItems: "center", gap: 0.5 }}>
            <Typography variant="body1" fontWeight="bold" component="span">Posting Date:</Typography>
            <Typography variant="body1" fontWeight="bold" component="span">{postingDate}</Typography>
          </Box>
          <Typography variant="body2" sx={{ color: "#ec0101ff", mt: 0.5 }}>Note: All Journals will be Posted on this Date</Typography>
        </Box>

        <Grid container spacing={2} sx={{ my: 2 }}>
            <Grid item xs={12}>
                <TextField
                    label="Common Batch Remarks"
                    value={commonBatchRemarks}
                    onChange={(e) => {
                        const val = e.target.value;
                        if (val.length <= 50) {
                            setCommonBatchRemarks(val);
                            if (checked) setRows(rows.map(r => ({ ...r, remarks: val.trim() ? val : "" })));
                        }
                    }}
                    fullWidth
                    required
                    multiline
                    inputProps={{ maxLength: 50 }}
                />
                <FormControlLabel
                    control={<Checkbox checked={checked} onChange={handleChange} disabled={!commonBatchRemarks.trim()} />}
                    label="Apply This Remark To All Below Transaction"
                />
            </Grid>
        </Grid>

        <Box>
          {rows.map((row, index) => (
            <React.Fragment key={row.id}>
              <Grid container spacing={2} alignItems="center">
                {/* Branch */}
                <Grid item xs={12} md={3}>
                  <Autocomplete
                    popupIcon={null}
                    options={branchSearch.options}
                    getOptionLabel={(o) => `${o.code} - ${o.name}`}
                    value={row.branch}
                    onChange={(e, v) => handleRowChange(row.id, "branch", v)}
                    onInputChange={(e, v) => searchBranches(v, row.id)}
                    loading={branchSearch.loading && activeBranchRowId === row.id}
                    onClose={() => setActiveBranchRowId(null)}
                    renderInput={(params) => (
                        <TextField {...params} label="Branch *" placeholder="Search Name/Code" 
                            InputProps={{ ...params.InputProps, endAdornment: (
                                <>{branchSearch.loading && activeBranchRowId === row.id ? <CircularProgress size={20} /> : null}{params.InputProps.endAdornment}</>
                            )}} 
                        />
                    )}
                  />
                </Grid>

                {/* Currency */}
                <Grid item xs={12} md={1.5}>
                  <TextField select label="Currency *" value={row.currency} onChange={(e) => handleRowChange(row.id, "currency", e.target.value)} fullWidth>
                    {currencyOptions.map((c) => <MenuItem key={c} value={c}>{c}</MenuItem>)}
                  </TextField>
                </Grid>

                {/* CGL */}
                <Grid item xs={12} md={2}>
                  <Autocomplete
                    popupIcon={null}
                    options={cglSearch.options}
                    getOptionLabel={(o) => `${o.cglNumber} - ${o.description}`}
                    value={row.cgl}
                    onChange={(e, v) => handleRowChange(row.id, "cgl", v)}
                    onInputChange={(e, v) => searchCgls(v, row.id)}
                    loading={cglSearch.loading && activeCglRowId === row.id}
                    onClose={() => setActiveCglRowId(null)}
                    renderInput={(params) => (
                        <TextField {...params} label="CGL *" placeholder="Search CGL" 
                            InputProps={{ ...params.InputProps, endAdornment: (
                                <>{cglSearch.loading && activeCglRowId === row.id ? <CircularProgress size={20} /> : null}{params.InputProps.endAdornment}</>
                            )}} 
                        />
                    )}
                  />
                </Grid>

                {/* Amount */}
                <Grid item xs={12} md={1.5}>
                  <TextField label="Amount *" value={row.amount} onChange={(e) => {
                      if (/^\d{0,21}(\.\d{0,4})?$/.test(e.target.value)) handleRowChange(row.id, "amount", e.target.value);
                  }} fullWidth />
                </Grid>

                {/* Txn Type */}
                <Grid item xs={12} md={1}>
                  <TextField select label="Txn Type *" value={row.txnType} onChange={(e) => handleRowChange(row.id, "txnType", e.target.value)} fullWidth>
                    <MenuItem value="Debit">Debit</MenuItem>
                    <MenuItem value="Credit">Credit</MenuItem>
                  </TextField>
                </Grid>

                {/* Product Code */}
                <Grid item xs={12} md={1.5}>
                  <TextField label="Product Code" value={row.productCode} onChange={(e) => {
                      if (/^\d{0,8}$/.test(e.target.value)) handleRowChange(row.id, "productCode", e.target.value);
                  }} fullWidth inputProps={{ maxLength: 8 }} />
                </Grid>

                {/* Remarks */}
                <Grid item xs={12} md={1.5}>
                  <TextField label="Remarks" value={row.remarks} onChange={(e) => {
                      if (e.target.value.length <= 50) handleRowChange(row.id, "remarks", e.target.value);
                  }} fullWidth inputProps={{ maxLength: 50, readOnly: checked }} />
                </Grid>

                {/* Delete */}
                <Grid item xs={12} md={0.5} textAlign="center">
                  <IconButton onClick={() => handleDeleteRow(row.id)}><DeleteIcon /></IconButton>
                </Grid>
              </Grid>
              {index < rows.length - 1 && <Box sx={{ borderBottom: 1, borderColor: "divider", my: 2 }} />}
            </React.Fragment>
          ))}

          <Box sx={{ mt: 3, display: "flex", gap: 2, justifyContent: "right" }}>
            <Button variant="contained" startIcon={<AddIcon />} onClick={handleAddRow}>Add Row</Button>
            <Button variant="contained" color="success" onClick={handleSubmit} disabled={isSubmitting}>
              {isSubmitting ? <CircularProgress size={24} /> : "Submit"}
            </Button>
          </Box>
        </Box>
      </Paper>
    </Box>
  );
}
