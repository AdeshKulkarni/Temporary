1. src/api/templateApi.js
Contains all endpoints including the new report generation and status update logic.
const BASE_URL_CONFIG = '/RB/template-config';
const BASE_URL_REPORT = '/RB/report-builder';

const handleResponse = async (response) => {
  const result = await response.json().catch(() => ({ 
    message: "Failed to parse server response", 
    status: "ERROR" 
  }));

  if (!response.ok) {
    throw new Error(result.message || `Error ${response.status}: ${response.statusText}`);
  }
  return result;
};

export const templateApi = {
  // Fetch all templates
  getAllTemplates: async () => {
    try {
      const response = await fetch(`${BASE_URL_CONFIG}/get-templates`, {
        method: 'GET',
        headers: { 'Content-Type': 'application/json' },
      });
      return await handleResponse(response);
    } catch (error) {
      console.error("API Error [getAllTemplates]:", error.message);
      throw error;
    }
  },

  // Fetch single template details
  getTemplateById: async (templateId) => {
    try {
      const response = await fetch(`${BASE_URL_CONFIG}/${templateId}`, {
        method: 'GET',
        headers: { 'Content-Type': 'application/json' },
      });
      return await handleResponse(response);
    } catch (error) {
      console.error(`API Error [getTemplateById] ID: ${templateId}:`, error.message);
      throw error;
    }
  },

  // Update Status Only
  updateTemplateStatus: async (templateId, status) => {
    try {
      const response = await fetch(`${BASE_URL_CONFIG}/update-status/${templateId}`, {
        method: 'PUT',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ status: status }),
      });
      return await handleResponse(response);
    } catch (error) {
      console.error(`API Error [updateStatus] ID: ${templateId}:`, error.message);
      throw error;
    }
  },

  // Delete Template
  deleteTemplate: async (templateId) => {
    try {
      const response = await fetch(`${BASE_URL_CONFIG}/${templateId}`, {
        method: 'DELETE',
        headers: { 'Content-Type': 'application/json' },
      });
      return await handleResponse(response);
    } catch (error) {
      console.error(`API Error [deleteTemplate] ID: ${templateId}:`, error.message);
      throw error;
    }
  },

  // Generate Report (PDF/Excel/Text)
  generateReport: async (payload, format) => {
    let endpoint = '';
    switch (format) {
      case 'excel': endpoint = `${BASE_URL_REPORT}/view/excel`; break;
      case 'text':  endpoint = `${BASE_URL_REPORT}/view/text`; break;
      case 'pdf':   
      default:      endpoint = `${BASE_URL_REPORT}/view/pdf`; break;
    }

    try {
      const response = await fetch(endpoint, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify(payload),
      });

      if (!response.ok) {
        const errorText = await response.text();
        throw new Error(errorText || "Failed to generate report");
      }

      // Return Blob for file download
      return await response.blob();
    } catch (error) {
      console.error(`API Error [generateReport] Format: ${format}:`, error.message);
      throw error;
    }
  }
};

2. src/components/TemplateConfiguration.js (Admin Screen)
This screen allows full access (Edit, Delete, Status Change).
import React, { useState, useEffect, useCallback } from 'react';
import { Typography, Container, Paper } from '@mui/material';
import TemplateTable from './TemplateTable';
import { templateApi } from '../api/templateApi';

const TemplateConfiguration = () => {
  const [templates, setTemplates] = useState([]);

  const fetchTemplates = useCallback(() => {
    templateApi.getAllTemplates()
      .then(res => setTemplates(res.data))
      .catch(err => console.error("Failed to load templates", err));
  }, []);

  useEffect(() => {
    fetchTemplates();
  }, [fetchTemplates]);

  return (
    <Container maxWidth={false} disableGutters sx={{ mt: 4 }}>
      <Paper sx={{ width: "100%", p: 3 }}>
        <Typography variant="h5" gutterBottom sx={{ fontWeight: 'bold' }}>
          Report Management
        </Typography>
        {/* readOnly defaults to false, enabling admin features */}
        <TemplateTable data={templates} onRefresh={fetchTemplates} />
      </Paper>
    </Container>
  );
};

export default TemplateConfiguration;

4. src/components/TemplateTable.js
Handles the display logic based on the readOnly prop.
import React, { useState } from 'react';
import { DataGrid } from '@mui/x-data-grid';
import Box from '@mui/material/Box';
import Paper from '@mui/material/Paper';
import IconButton from '@mui/material/IconButton';
import Tooltip from '@mui/material/Tooltip';
import Dialog from '@mui/material/Dialog';
import DialogTitle from '@mui/material/DialogTitle';
import DialogContent from '@mui/material/DialogContent';
import DialogContentText from '@mui/material/DialogContentText';
import DialogActions from '@mui/material/DialogActions';
import Button from '@mui/material/Button';
import Chip from '@mui/material/Chip';
import MenuItem from '@mui/material/MenuItem';
import TextField from '@mui/material/TextField';

// Icons
import Visibility from '@mui/icons-material/Visibility';
import Delete from '@mui/icons-material/Delete';
import Edit from '@mui/icons-material/Edit';
import ReportProblemOutlined from '@mui/icons-material/ReportProblemOutlined';

// Internal Components
import TemplateViewDialog from './TemplateViewDialog';
import { templateApi } from '../api/templateApi';

const TemplateTable = ({ data, onRefresh, readOnly = false }) => {
  const [selectedTemplate, setSelectedTemplate] = useState(null);
  
  // Dialog States
  const [openView, setOpenView] = useState(false);
  const [openDelete, setOpenDelete] = useState(false);
  const [openStatusEdit, setOpenStatusEdit] = useState(false);
  const [openGeneralEdit, setOpenGeneralEdit] = useState(false);

  const [newStatus, setNewStatus] = useState('');

  // --- Handlers ---
  const handleAction = (template, actionType) => {
    setSelectedTemplate(template);
    
    // View is always allowed
    if (actionType === 'view') {
        setOpenView(true);
        return;
    }

    // Security Check: Block other actions if readOnly
    if (readOnly) return; 

    if (actionType === 'delete') setOpenDelete(true);
    
    if (actionType === 'status') {
      setNewStatus(template.status || 'INACTIVE');
      setOpenStatusEdit(true);
    }
    
    if (actionType === 'edit') {
        setOpenGeneralEdit(true); 
    }
  };

  const closeAllDialogs = () => {
    setOpenView(false);
    setOpenDelete(false);
    setOpenStatusEdit(false);
    setOpenGeneralEdit(false);
    setSelectedTemplate(null);
  };

  const confirmDelete = async () => {
    try {
      await templateApi.deleteTemplate(selectedTemplate.templateId);
      onRefresh(); 
      closeAllDialogs();
    } catch (error) {
      console.error("Delete failed:", error.message);
    }
  };

  const confirmStatusUpdate = async () => {
    if (!selectedTemplate) return;
    try {
      await templateApi.updateTemplateStatus(selectedTemplate.templateId, newStatus);
      onRefresh(); 
      closeAllDialogs();
    } catch (error) {
      console.error("Status update failed:", error.message);
    }
  };

  // --- Data Grid Column Definitions ---
  const columns = [
    { field: 'templateId', headerName: 'ID', flex: 0.5 },
    { 
      field: 'templateName', 
      headerName: 'Report Name', 
      flex: 1.5,
      renderCell: (params) => <Box sx={{ fontWeight: 500 }}>{params.value}</Box>
    },
    { field: 'description', headerName: 'Description', flex: 2 },
    
    // --- Status Column ---
    {
      field: 'status',
      headerName: 'Status',
      flex: 1.2,
      renderCell: (params) => {
        const status = params.value || 'INACTIVE';
        const isSuccess = status === 'ACTIVE';
        
        return (
            <Box sx={{ display: 'flex', alignItems: 'center', gap: 1 }}>
                <Chip 
                    label={status} 
                    color={isSuccess ? "success" : "default"} 
                    size="small" 
                    variant="outlined"
                />
                
                {/* Only show Edit Icon if NOT readOnly */}
                {!readOnly && (
                    <Tooltip title="Change Status">
                        <IconButton 
                            size="small" 
                            onClick={() => handleAction(params.row, 'status')}
                        >
                            <Edit fontSize="inherit" sx={{ fontSize: '1rem', color: 'text.secondary' }} />
                        </IconButton>
                    </Tooltip>
                )}
            </Box>
        );
      }
    },
    
    // --- Actions Column ---
    {
      field: 'actions',
      headerName: 'Actions',
      flex: 1.2,
      sortable: false,
      filterable: false,
      renderCell: (params) => (
        <Box sx={{ display: 'flex', gap: 0.5, width: '100%' }}>
          <Tooltip title="View & Run">
            <IconButton 
              size="small" 
              color="primary" 
              onClick={() => handleAction(params.row, 'view')}
            >
              <Visibility fontSize="small" />
            </IconButton>
          </Tooltip>

          {/* Only show Edit/Delete if NOT readOnly */}
          {!readOnly && (
            <>
                <Tooltip title="Edit Template Details">
                    <IconButton 
                    size="small" 
                    color="info" 
                    onClick={() => handleAction(params.row, 'edit')}
                    >
                    <Edit fontSize="small" />
                    </IconButton>
                </Tooltip>

                <Tooltip title="Delete Template">
                    <IconButton 
                    size="small" 
                    color="error" 
                    onClick={() => handleAction(params.row, 'delete')}
                    >
                    <Delete fontSize="small" />
                    </IconButton>
                </Tooltip>
            </>
          )}
        </Box>
      )
    }
  ];

  return (
    <Paper sx={{ width: "100%" }} variant="outlined">
      <DataGrid
        rows={data}
        columns={columns}
        getRowId={(row) => row.templateId}
        pageSizeOptions={[5, 10, 25]}
        initialState={{
          pagination: { paginationModel: { pageSize: 10 } },
        }}
        disableRowSelectionOnClick
        sx={{
          border: 'none',
          '& .MuiDataGrid-columnHeader': {
            backgroundColor: (theme) => theme.palette.grey[50],
            fontWeight: 'bold'
          },
        }}
      />

      {/* View Dialog (Always Available) */}
      {selectedTemplate && (
        <TemplateViewDialog
          open={openView}
          handleClose={closeAllDialogs}
          template={selectedTemplate}
        />
      )}

      {/* Admin Dialogs (Conditional Render) */}
      {!readOnly && (
        <>
            {/* Status Edit Dialog */}
            <Dialog open={openStatusEdit} onClose={closeAllDialogs} maxWidth="xs" fullWidth>
                <DialogTitle>Update Status</DialogTitle>
                <DialogContent>
                    <Box sx={{ mt: 1 }}>
                        <TextField
                            select
                            fullWidth
                            label="Select New Status"
                            value={newStatus}
                            onChange={(e) => setNewStatus(e.target.value)}
                            size="small"
                            sx={{ mt: 2 }}
                        >
                            <MenuItem value="ACTIVE">ACTIVE</MenuItem>
                            <MenuItem value="INACTIVE">INACTIVE</MenuItem>
                        </TextField>
                    </Box>

                    <Button onClick={closeAllDialogs}>Cancel</Button>
                    <Button onClick={confirmStatusUpdate} variant="contained" size="small">Update</Button>

                </DialogContentText>

                </DialogActions>
            </Dialog>
        </>
      )}
    </Paper>
  );
};

5. src/components/TemplateViewDialog.js
Handles parameter inputs, validation, and file downloading.
import React, { useState, useEffect } from 'react';
import { 
  Dialog, DialogTitle, DialogContent, DialogActions, 
  Button, MenuItem, TextField, Grid, Divider, Alert 
} from '@mui/material';
import { templateApi } from '../api/templateApi';

const TemplateViewDialog = ({ open, handleClose, template }) => {
  const [selectedVariantCode, setSelectedVariantCode] = useState('');
  const [formData, setFormData] = useState({});
  const [exportFormat, setExportFormat] = useState('pdf');
  const [error, setError] = useState(null);
  const [loading, setLoading] = useState(false);

  // Find active variant object
  const activeVariant = template?.variants?.find(v => v.variantCode === selectedVariantCode);

  useEffect(() => {
    if (open) {
      setSelectedVariantCode('');
      setFormData({});
      setError(null);
      setLoading(false);
    }
  }, [open]);

  const handleInputChange = (name, value) => {
    setFormData(prev => ({ ...prev, [name]: value }));
    if (error) setError(null);
  };

  const handleGenerate = async () => {
    if (!activeVariant) return;

    // 1. Validate Required Params
    const missingParams = [];
    const payloadParams = {};

    activeVariant.params.forEach(param => {
      const value = formData[param.paramName];
      
      // Validation check
      if (param.required && (value === undefined || value === null || value === '')) {
        missingParams.push(param.label || param.paramName);
      }
      
      // Build payload
      if (value !== undefined && value !== '') {
        payloadParams[param.paramName] = value;
      }
    });

    if (missingParams.length > 0) {
      setError(`Please fill required fields: ${missingParams.join(', ')}`);
      return;
    }

    // 2. Prepare Payload
    const payload = {
      templateId: template.templateId,
      variantCode: activeVariant.variantCode,
      params: payloadParams
    };

    setLoading(true);

    try {
      // 3. API Call
      const blob = await templateApi.generateReport(payload, exportFormat);

      // 4. Handle Download
      const url = window.URL.createObjectURL(blob);
      const link = document.createElement('a');
      link.href = url;
      
      const extension = exportFormat === 'excel' ? 'xlsx' : exportFormat === 'text' ? 'txt' : 'pdf';
      link.setAttribute('download', `${template.templateName}_${activeVariant.variantName}.${extension}`);
      
      document.body.appendChild(link);
      link.click();
      link.parentNode.removeChild(link);
      
    } catch (err) {
      setError("Failed to generate report. Please try again.");
    } finally {
      setLoading(false);
    }
  };

  return (
    <Dialog open={open} onClose={handleClose} maxWidth="sm" fullWidth>
      <DialogTitle>Generate Report: {template.templateName}</DialogTitle>
      
      <DialogContent dividers>
        {error && (
            <Alert severity="error" sx={{ mb: 2 }}>{error}</Alert>
        )}

        <Grid container spacing={3}>
          {/* Variant Selection */}
          <Grid item xs={12}>
            <TextField
              select
              fullWidth
              label="Select Report Variant"
              value={selectedVariantCode}
              onChange={(e) => {
                setSelectedVariantCode(e.target.value);
                setFormData({}); // Clear form when variant changes
              }}
            >
              {template.variants && template.variants.map((v) => (
                <MenuItem key={v.variantCode} value={v.variantCode}>
                  {v.variantName}
                </MenuItem>
              ))}
            </TextField>
          </Grid>

          {/* Dynamic Input Fields */}
          {activeVariant?.params.map((param) => (
            <Grid item xs={12} key={param.paramName}>
              <TextField
                fullWidth
                label={param.label || param.paramName}
                type={param.paramType === 'DATE' ? 'date' : 'text'}
                required={param.required}
                value={formData[param.paramName] || ''}
                InputLabelProps={{ shrink: true }}
                onChange={(e) => handleInputChange(param.paramName, e.target.value)}
                helperText={param.required ? "Required" : "Optional"}
              />
            </Grid>
          ))}

          {/* Export Format */}
          {activeVariant && (
            <Grid item xs={12}>
              <Divider sx={{ my: 1 }}>Export Options</Divider>
              <TextField 
                select 
                fullWidth 
                label="Export Format" 
                value={exportFormat}
                onChange={(e) => setExportFormat(e.target.value)}
              >
                <MenuItem value="pdf">PDF Document</MenuItem>
                <MenuItem value="excel">Excel Spreadsheet</MenuItem>
                <MenuItem value="text">Plain Text</MenuItem>
              </TextField>
            </Grid>
          )}
        </Grid>
      </DialogContent>

      <DialogActions sx={{ p: 2 }}>
        <Button onClick={handleClose} disabled={loading}>
          Cancel
        </Button>
        <Button 
          variant="contained" 
          color="primary" 
          disabled={!selectedVariantCode || loading}
          onClick={handleGenerate}
        >
          {loading ? 'Generating...' : 'Generate Report'}
        </Button>
      </DialogActions>
    </Dialog>
  );
};

export default TemplateViewDialog;
