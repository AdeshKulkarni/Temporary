import csv
import random

# --- CONFIGURATION ---
FILENAME = "balanced_real_data.csv"
NUM_PAIRS = 750  # 750 Pairs = 1500 Rows Total
SELECTED_DATE = "17122025"  # ddMMyyyy

# --- CONSTANTS ---
BATCH_ID = "01"
YEAR = "2025"
MONTH = "09"
VALID_CURRENCY = "USD"

# --- DB DATA ---
VALID_CGLS = [
    "1208505003", "8812505008", "1122505001", "5051030601", "2051030601",
    "5051502501", "2051502501", "5051500501", "2051500501", "2051270801",
    "2051140870", "2051080670", "2051500270", "2051501170", "2051500170",
    "5051080601", "2782080630", "2011500105", "8463505001", "1257505001"
]
VALID_BRANCHES = [
    "32121", "32122", "32130", "32131", "32132", "31774", "31776", "31777",
    "31778", "31779", "31781", "31783", "31784", "31785", "31787", "31788",
    "17386", "17383", "18381", "18378", "17336", "14747", "32108", "32109"
]

# --- GENERATION LOGIC ---
def generate_balanced_pair(index):
    # 1. Generate Random Amount for this Pair
    amount = f"{random.randint(10, 10000)}.{random.randint(10, 99)}"
    
    # 2. Select Random Branch/CGL
    branch = random.choice(VALID_BRANCHES)
    cgl = random.choice(VALID_CGLS)
    
    # 3. Create Debit Row
    row_dr = [
        BATCH_ID, SELECTED_DATE, YEAR, MONTH,
        branch, VALID_CURRENCY, cgl, amount,
        "Dr", f"Txn_{index}_Dr", "12345678", "B", "C", "D"
    ]
    
    # 4. Create Matching Credit Row (Same Amount)
    row_cr = [
        BATCH_ID, SELECTED_DATE, YEAR, MONTH,
        branch, VALID_CURRENCY, cgl, amount,
        "Cr", f"Txn_{index}_Cr", "12345678", "B", "C", "D"
    ]
    
    return row_dr, row_cr

# --- WRITE FILE ---
with open(FILENAME, mode='w', newline='') as file:
    writer = csv.writer(file)
    
    print(f"Generating {NUM_PAIRS * 2} rows (Balanced)...")
    for i in range(1, NUM_PAIRS + 1):
        dr, cr = generate_balanced_pair(i)
        writer.writerow(dr)
        writer.writerow(cr)

print(f"Successfully created '{FILENAME}'. Total Debit matches Total Credit.")
