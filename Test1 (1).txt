class Solution {
    public int MaxElement(int[] nums , int low , int high) {
        int maxi = 0;

        for(int i=low; i<=high; i++) {
           
            if(nums[i] > maxi) {
                maxi = nums[i];
            }
        }

        return maxi;
    }


    public int maxFrequency(int[] nums, int k) {

        int low = 0;
        int n = nums.length;      
           Arrays.sort(nums);       

        int currentSum = 0;
        int ans = 1;

        for(int high=0; high<n; high++) {  
                 
            
            currentSum += nums[high];
            int maxiEle = MaxElement(nums , low , high);  
           
            int len = Math.abs(high - low) + 1;
            
            while(Math.abs(maxiEle - currentSum) > k) {
                if(len > ans) {
                    ans = len;
                }
            }

            if(Math.abs(maxiEle - currentSum) > k) {
                low++;
            }
                
        }

        return ans;
        
    }
}

class Solution {
    public int MaxElement(int[] nums, int low, int high) {
        int maxi = 0;
        for(int i = low; i <= high; i++) {
            if(nums[i] > maxi) {
                maxi = nums[i];
            }
        }
        return maxi;
    }

    public int maxFrequency(int[] nums, int k) {
        Arrays.sort(nums);  // ← needed for the window logic to work

        int low = 0;
        int n = nums.length;
        int currentSum = 0;
        int ans = 1;  // ← at least 1 element always works

        for(int high = 0; high < n; high++) {
            currentSum += nums[high];
            int maxiEle = MaxElement(nums, low, high);
            int len = high - low + 1;

            // ← while, not if; and subtract leaving element
            while(maxiEle * len - currentSum > k) {
                currentSum -= nums[low];
                low++;
                len = high - low + 1;
                maxiEle = MaxElement(nums, low, high);
            }

            if(maxiEle * len - currentSum <= k) {  // ← <= not <
                if(len > ans) {
                    ans = len;
                }
            }
        }

        return ans;
    }
}
