class Solution {

    public boolean valid(int i, int j, int n, int m) {
        if (i < 0 || i >= n || j < 0 || j >= m) {
            return false;
        }
        if (i == 0 || i == n - 1 || j == 0 || j == m - 1) {
            return false;
        }
        return true;
    }

    public void dfs(char[][] board, int i, int j, int m, int n, boolean[][] visited) {

        visited[i][j] = true;
        board[i][j] = 'X';

        int[] x = { -1, 1, 0, 0 };
        int[] y = { 0, 0, -1, 1 };

        for (int k = 0; k < 4; k++) {

            int row = i + x[k];
            int col = j + y[k];

            if (valid(row, col, n, m) && (visited[row][col] == false && board[row][col] == 'O')) {
                D.p(row, col);
                board[row][col] = 'X';
                dfs(board, row, col, m, n, visited);
            }

        }
    }

    public void solve(char[][] board) {

        int n = board.length;
        int m = board[0].length;

        boolean[][] visited = new boolean[n][m];

        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {

                if (visited[i][j] == false && board[i][j] == 'O') {
                    // D.p(i , j);
                    dfs(board, i, j, m, n, visited);
                }
            }
        }

    }
}

class D {
    static void p(Object... o) {
        StringBuilder sb = new StringBuilder();
        for (int i = 0; i < o.length; i += 2) {
            String val = (i + 1 < o.length) ? Arrays.deepToString(new Object[] { o[i + 1] }) : "[]";
            sb.append(o[i]).append(": ").append(val.substring(1, val.length() - 1)).append(" | ");
        }
        System.out.println(sb);
    }
