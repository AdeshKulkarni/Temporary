import React, { useState, useEffect } from 'react';
import axios from 'axios';
import {
    Box,
    Typography,
    Fab,
    Dialog,
    DialogActions,
    DialogContent,
    DialogTitle,
    TextField,
    Button,
    Container,
    Paper,
    Snackbar,
    Alert
} from '@mui/material';
import AddIcon from '@mui/icons-material/Add';
import EditIcon from '@mui/icons-material/Edit';
import { DataGrid, GridActionsCellItem } from '@mui/x-data-grid';

// --- API Configuration ---
const API_BASE_URL_GET = "http://10.0.19.101:2000";
const API_BASE_URL_POST = "http://10.0.19.143:9000";

const apiClientGet = axios.create({ baseURL: API_BASE_URL_GET });
const apiClientPost = axios.create({ baseURL: API_BASE_URL_POST });


const initialFormState = {
    currencyCode: '',
    currencyName: '',
    currencyRate: '',
};

function CurrencyMaster() {

    const [rows, setRows] = useState([]);
    const [loading, setLoading] = useState(true);
    const [isDialogOpen, setIsDialogOpen] = useState(false);
    const [formData, setFormData] = useState(initialFormState);
    const [editingRow, setEditingRow] = useState(null);
    const [errors, setErrors] = useState({});

    // --- State for Snackbar ---
    const [snackbar, setSnackbar] = useState({
        open: false,
        message: '',
        severity: 'info' // can be 'success', 'error', 'warning', 'info'
    });

    const showSnackBar = (message, severity = 'info') => {
        setSnackbar({ open: true, message, severity });
    };

    const handleSnackbarClose = (event, reason) => {
        if (reason === 'clickaway') {
            return;
        }
        setSnackbar({ ...snackbar, open: false });
    };


    // --- Data Fetching Effect ---
    useEffect(() => {
        fetchCurrencyData();
    }, []);

    const fetchCurrencyData = async () => {
        setLoading(true);
        try {
            console.log("DEBUG: API HIT [GET]: Fetching all currency data from", `${API_BASE_URL_GET}/currency-master`);
            const response = await apiClientGet.get('/currency-master');
            
            // --- CONSOLE LOG FOR DEBUGGING ---
            console.log("DEBUG: API RESPONSE [GET]:", response);

            if (response.data && response.data.success) {
                const processedData = response.data.data.map(item => ({
                    ...item,
                    id: item.currencyCode
                }));
                setRows(processedData);
                // Not showing a snackbar on initial load to keep the UI clean.
                // You can uncomment the line below if you want a notification.
                // showSnackBar("Currency data loaded successfully!", "success");
            } else {
                const errorMessage = response.data.message || "Failed to fetch data due to an unknown error.";
                showSnackBar(errorMessage, "error");
            }
        } catch (error) {
            // --- CONSOLE LOG FOR DEBUGGING ---
            console.error("DEBUG: API ERROR [GET]:", error);
            const errorMessage = error.response?.data?.message || "An error occurred while fetching data.";
            showSnackBar(errorMessage, "error");
        } finally {
            setLoading(false);
        }
    };


    const validate = (fieldValues = formData) => {
        let tempErrors = { ...errors };
        if ('currencyName' in fieldValues) {
            tempErrors.currencyName = '';
            if (!fieldValues.currencyName) tempErrors.currencyName = "Currency Name is required.";
            else if (/[0-9]/.test(fieldValues.currencyName)) tempErrors.currencyName = "Numbers are not allowed.";
        }
        if (!editingRow) {
            if ('currencyCode' in fieldValues) {
                tempErrors.currencyCode = '';
                if (!fieldValues.currencyCode) tempErrors.currencyCode = "Currency Code is required.";
                else if (fieldValues.currencyCode.length !== 3) tempErrors.currencyCode = "Must be exactly 3 alphabets.";
            }
            if ('currencyRate' in fieldValues) {
                tempErrors.currencyRate = '';
                if (!fieldValues.currencyRate) tempErrors.currencyRate = "Rate is required.";
            }
        }
        setErrors(tempErrors);
        return Object.values(tempErrors).every(x => x === "");
    };

    const handleInputChange = (e) => {
        const { name, value } = e.target;
        let processedValue = value;

        if (name === 'currencyCode') {
            if (!/^[a-zA-Z]*$/.test(value)) return;
            processedValue = value.toUpperCase().slice(0, 3);
        }
        if (name === 'currencyName') {
            if (/[0-9]/.test(value)) return;
            processedValue = value.slice(0, 30);
        }
        if (name === 'currencyRate') {
            if (value !== "" && !/^\d*\.?\d*$/.test(value)) return;
             const [integerPart, decimalPart] = value.split('.');
             if (integerPart.length > 6 || (decimalPart && decimalPart.length > 6)) return;
        }

        const newFormData = { ...formData, [name]: processedValue };
        setFormData(newFormData);
        validate({ [name]: processedValue });
    };

    const handleEditClick = (id) => () => {
        const rowToEdit = rows.find((row) => row.id === id);
        if (rowToEdit) {
            setEditingRow(rowToEdit);
            setFormData(rowToEdit);
            setIsDialogOpen(true);
        }
    };

    const columns = [
        { field: 'id', headerName: 'ID', width: 90, headerAlign: 'center', align: 'center', valueGetter: (params) => params.row.currencyCode },
        { field: 'currencyCode', headerName: 'Currency Code', width: 130, headerAlign: 'center', align: 'center' },
        { field: 'currencyName', headerName: 'Currency Name', flex: 1, minWidth: 200, headerAlign: 'center', align: 'left' },
        { field: 'currencyRate', headerName: 'Currency Rate', type: 'number', width: 130, headerAlign: 'center', align: 'right' },
        {
            field: 'createdAt', headerName: 'Created At', type: 'dateTime', width: 200,
            valueGetter: (params) => params.value ? new Date(params.value) : null,
            renderCell: (params) => params.value ? new Date(params.value).toLocaleString() : '',
            headerAlign: 'center', align: 'right'
        },
        {
            field: 'actions', type: 'actions', headerName: 'Actions', width: 80,
            cellClassName: 'actions', headerAlign: 'center', align: 'center',
            getActions: ({ id }) => [<GridActionsCellItem icon={<EditIcon />} label="Edit" onClick={handleEditClick(id)} color="primary" />],
        },
    ];

    const handleOpenDialog = () => {
        setEditingRow(null);
        setFormData(initialFormState);
        setErrors({});
        setIsDialogOpen(true);
    };

    const handleCloseDialog = () => {
        setIsDialogOpen(false);
        setEditingRow(null);
        setFormData(initialFormState);
        setErrors({});
    };

    const handleSubmit = async () => {
        if (!validate()) {
            showSnackBar("Please fix the validation errors.", "warning");
            return;
        }

        const isEditing = !!editingRow;
        const changeType = isEditing ? "UPDATE" : "ADD";

        const payload = {
            requestType: "CURRENCY_MASTER",
            changeType: changeType,
            payload: {
                currencyCode: isEditing ? editingRow.currencyCode : formData.currencyCode,
                currencyName: formData.currencyName,
                currencyRate: isEditing ? editingRow.currencyRate : Number(formData.currencyRate),
                flag: 0
            }
        };

        try {
            console.log(`DEBUG: API HIT [POST] (${changeType}): Creating request at`, `${API_BASE_URL_POST}/create-request`);
            // --- CONSOLE LOG FOR DEBUGGING ---
            console.log("DEBUG: Request Payload:", JSON.stringify(payload, null, 2));

            const response = await apiClientPost.post('/create-request', payload);
            // --- CONSOLE LOG FOR DEBUGGING ---
            console.log("DEBUG: API RESPONSE [POST]:", response);

            if (response.data && response.data.success) {
                showSnackBar(response.data.message || "Request submitted successfully!", "success");
                handleCloseDialog();
            } else {
                const errorMessage = response.data.message || "Failed to submit request.";
                showSnackBar(errorMessage, "error");
            }
        } catch (error) {
            // --- CONSOLE LOG FOR DEBUGGING ---
            console.error(`DEBUG: API ERROR [POST] (${changeType}):`, error);
            const errorMessage = error.response?.data?.message || "An error occurred while submitting the request.";
            showSnackBar(errorMessage, "error");
        }
    };

    return (
        <Container maxWidth="xl" sx={{ mt: 4, mb: 4, display: 'flex', justifyContent: 'center', width: '100%' }}>
            <Paper elevation={3} sx={{ p: 3, borderRadius: 2, width: '90%', maxWidth: 1400 }}>
                <Typography variant="h4" component="h1" sx={{ fontWeight: '600', color: 'primary.main', textAlign: 'center', mb: 3 }}>
                    Currency Master
                </Typography>
                <Box sx={{ height: '70vh', width: '100%' }}>
                    <DataGrid
                        rows={rows}
                        columns={columns}
                        loading={loading}
                        getRowId={(row) => row.id}
                        disableRowSelectionOnClick
                        sx={{
                            '& .MuiDataGrid-cell:focus-within, & .MuiDataGrid-cell:focus': { outline: 'none !important' },
                            '& .MuiDataGrid-columnHeaderTitle': { fontWeight: 'bold' },
                            '& .MuiDataGrid-row:nth-of-type(odd)': { backgroundColor: (theme) => theme.palette.action.hover },
                        }}
                    />
                </Box>
            </Paper>

            <Fab color="primary" aria-label="add" onClick={handleOpenDialog} sx={{ position: 'fixed', bottom: 30, right: 30 }}>
                <AddIcon />
            </Fab>

            <Dialog open={isDialogOpen} onClose={handleCloseDialog} fullWidth maxWidth="sm">
                <DialogTitle sx={{ fontWeight: 'bold' }}>
                    {editingRow ? 'Edit Currency' : 'Add New Currency'}
                </DialogTitle>
                <DialogContent>
                    <TextField autoFocus margin="dense" name="currencyCode" label="Currency Code" type="text" fullWidth variant="outlined" value={formData.currencyCode} onChange={handleInputChange} error={!!errors.currencyCode} helperText={errors.currencyCode} disabled={!!editingRow} />
                    <TextField margin="dense" name="currencyName" label="Currency Name" type="text" fullWidth variant="outlined" value={formData.currencyName} onChange={handleInputChange} error={!!errors.currencyName} helperText={errors.currencyName} />
                    <TextField margin="dense" name="currencyRate" label="Currency Rate" type="text" fullWidth variant="outlined" value={formData.currencyRate} onChange={handleInputChange} error={!!errors.currencyRate} helperText={errors.currencyRate} disabled={!!editingRow} />
                </DialogContent>
                <DialogActions sx={{ p: '16px 24px' }}>
                    <Button onClick={handleCloseDialog} color="secondary">Cancel</Button>
                    <Button onClick={handleSubmit} variant="contained">Submit Request</Button>
                </DialogActions>
            </Dialog>

            {/* --- Snackbar Component for Notifications --- */}
            <Snackbar
                open={snackbar.open}
                autoHideDuration={6000}
                onClose={handleSnackbarClose}
                anchorOrigin={{ vertical: 'top', horizontal: 'right' }}
            >
                <Alert onClose={handleSnackbarClose} severity={snackbar.severity} sx={{ width: '100%' }}>
                    {snackbar.message}
                </Alert>
            </Snackbar>
        </Container>
    );
}

export default CurrencyMaster;
