Import React, { useState, useEffect, useCallback } from "react";
import {
  Box,
  Typography,
  Chip,
  Dialog,
  DialogActions,
  DialogContent,
  DialogContentText,
  DialogTitle,
  TextField,
  Button,
  CircularProgress,
  Tooltip,
  IconButton,
  Stack,
} from "@mui/material";
import { DataGrid } from "@mui/x-data-grid";
import {
  Check as CheckIcon,
  Clear as ClearIcon,
  Visibility as VisibilityIcon,
} from "@mui/icons-material";
import AccountCircleIcon from "@mui/icons-material/AccountCircle";
import InfoIcon from "@mui/icons-material/Info";

import useApi from "../../../hooks/useApi";
import useCustomSnackbar from "../../../utils/useCustomSnackbar";

const formatCurrency = (num) => {
  if (typeof num !== "number") {
    num = parseFloat(num) || 0;
  }
  return new Intl.NumberFormat("en-IN", {
    style: "decimal",
    minimumFractionDigits: 2,
    maximumFractionDigits: 2,
  }).format(num);
};

function CustomNoRowsOverlay() {
  return (
    <Stack height="100%" alignItems="center" justifyContent="center">
      <InfoIcon sx={{ fontSize: 48, color: "text.secondary", mb: 2 }} />
      <Typography variant="h6">No Pending Requests Found</Typography>
      <Typography>All MOC requests have been processed.</Typography>
    </Stack>
  );
}

export default function MocRequestApproval() {
  const { callApi } = useApi();
  const showSnackBar = useCustomSnackbar();

  const [loading, setLoading] = useState(true);
  const [rows, setRows] = useState([]);
  const [allPendingRequests, setAllPendingRequests] = useState([]);

  const [isModalOpen, setIsModalOpen] = useState(false);
  const [isViewModalOpen, setIsViewModalOpen] = useState(false);

  const [isSubmitting, setIsSubmitting] = useState(false);
  const [selectedBatch, setSelectedBatch] = useState(null);
  const [modalMode, setModalMode] = useState("ACCEPTED");
  const [executorRemarks, setExecutorRemarks] = useState("");
  const [partialApprovalRemarks, setPartialApprovalRemarks] = useState("");

  const [viewModalRows, setViewModalRows] = useState([]);
  const [viewModalSelection, setViewModalSelection] = useState([]);

  const fetchPendingBatches = useCallback(async () => {
    setLoading(true);
    try {
      const pendingRequests = await callApi(
        "MS/moc/pending-requests",
        null,
        "GET"
      );

      if (!Array.isArray(pendingRequests) || pendingRequests.length === 0) {
        setRows([]);
        setAllPendingRequests([]);
        setLoading(false);
        showSnackBar("No pending requests found.", "info");
        return;
      }

      setAllPendingRequests(pendingRequests);

      const groupedBatches = new Map();
      for (const request of pendingRequests) {
        const batchId = request.batchId;

        if (!groupedBatches.has(batchId)) {
          groupedBatches.set(batchId, {
            id: batchId,
            batchId: batchId,
            creatorId: request.creatorId,
            requestStatus: request.requestStatus,
            requestDate: request.requestDate,
            commonBatchRemarks: request.commonBatchRemarks,
            totalDebit: 0,
            totalCredit: 0,
            requestCount: 0,
          });
        }

        const batchDetails = groupedBatches.get(batchId);
        batchDetails.requestCount++;

        try {
          const payload = JSON.parse(request.payload);
          const amount = parseFloat(payload.amount) || 0;
          if (payload.transactionType === "Debit") {
            batchDetails.totalDebit += amount;
          } else if (payload.transactionType === "Credit") {
            batchDetails.totalCredit += amount;
          }
        } catch (e) {
          console.error(`Failed to parse payload for request ${request.id}`, e);
        }
      }
      setRows(Array.from(groupedBatches.values()));
    } catch (err) {
      console.error("Failed to fetch pending requests:", err);
      showSnackBar("Failed to load pending requests.", "error");
    } finally {
      setLoading(false);
    }
  }, [callApi, showSnackBar]);

  useEffect(() => {
    fetchPendingBatches();
  }, [fetchPendingBatches]);

  const handleOpenModal = (batch, mode) => {
    setSelectedBatch(batch);
    setModalMode(mode);
    setExecutorRemarks("");
    setIsModalOpen(true);
  };

  const handleCloseModal = () => {
    if (isSubmitting) return;
    setIsModalOpen(false);
    setSelectedBatch(null);
  };

  const handleOpenViewModal = (batch) => {
    setSelectedBatch(batch);
    const filteredRequests = allPendingRequests.filter(
      (req) => req.batchId === batch.batchId && req.requestStatus === "PENDING"
    );

    const detailRows = filteredRequests
      .map((req) => {
        try {
          const payloadData = JSON.parse(req.payload);
          return {
            id: req.journalId,
            journalId: req.journalId,
            ...payloadData,
            amount: parseFloat(payloadData.amount) || 0,
          };
        } catch (e) {
          console.error("Failed to parse payload for view modal:", e);
          return null;
        }
      })
      .filter(Boolean);

    setViewModalRows(detailRows);
    setPartialApprovalRemarks("");
    setViewModalSelection([]);
    setIsViewModalOpen(true);
  };

  const handleCloseViewModal = () => {
    if (isSubmitting) return;
    setIsViewModalOpen(false);
    setSelectedBatch(null);
    setViewModalRows([]);
    setViewModalSelection([]);
  };

  const handleSubmitAction = async () => {
    if (modalMode === "REJECTED" && !executorRemarks.trim()) {
      showSnackBar("Approver remarks are required for rejection.", "warning");
      return;
    }

    setIsSubmitting(true);
    const payload = {
      batchId: selectedBatch.batchId,
      status: modalMode,
      remarks:
        executorRemarks.trim() ||
        (modalMode === "ACCEPTED" ? "Accepted" : "Rejected"),
    };

    try {
      await callApi("MS/moc/process-bulk", payload, "POST");
      showSnackBar(
        `Batch ${
          selectedBatch.batchId
        } successfully ${modalMode.toLowerCase()}ed.`,
        "success"
      );
      handleCloseModal();
      fetchPendingBatches();
    } catch (error) {
      console.error("Failed to process batch:", error);
      const errorMsg =
        error.response?.data?.message ||
        `Failed to ${modalMode.toLowerCase()} batch.`;
      showSnackBar(errorMsg, "error");
    } finally {
      setIsSubmitting(false);
    }
  };

  const handlePartialApprovalSubmit = async () => {
    if (viewModalSelection.length === 0) {
      showSnackBar("Please select at least one entry to approve.", "warning");
      return;
    }
    if (!partialApprovalRemarks.trim()) {
      showSnackBar("Remarks are required for partial approval.", "warning");
      return;
    }

    setIsSubmitting(true);
    const payload = {
      journalIds: viewModalSelection,
      status: "ACCEPTED",
      remarks: partialApprovalRemarks.trim(),
    };

    try {
      await callApi("MS/moc/process-bulk", payload, "POST");
      showSnackBar("Selected entries approved successfully.", "success");
      handleCloseViewModal();
      fetchPendingBatches();
    } catch (error) {
      console.error("Failed to process partial batch:", error);
      const errorMsg =
        error.response?.data?.message || "Failed to approve selected entries.";
      showSnackBar(errorMsg, "error");
    } finally {
      setIsSubmitting(false);
    }
  };

  const mainColumns = [
    {
      field: "batchId",
      headerName: "Batch ID",
      flex: 1.2,
      minWidth: 150,
    },
    {
      field: "requestStatus",
      headerName: "Status",
      flex: 0.5,
      align: "center",
      headerAlign: "center",
      renderCell: (params) => (
        <Chip label={params.value} color="warning" size="small" />
      ),
    },
    {
      field: "creatorId",
      headerName: "Creator",
      flex: 0.6,
      align: "center",
      headerAlign: "center",
      renderCell: (params) => (
        <Chip icon={<AccountCircleIcon />} label={params.value} size="small" />
      ),
    },
    {
      field: "requestDate",
      headerName: "Submitted On",
      flex: 1,
      minWidth: 180,
      renderCell: (params) => (
        <Box>
          {new Date(params.value).toLocaleString("en-IN", {
            dateStyle: "medium",
            timeStyle: "short",
          })}
        </Box>
      ),
    },
    {
      field: "commonBatchRemarks",
      headerName: "Creator Remarks",
      flex: 1.5,
      minWidth: 200,
    },
    {
      field: "requestCount",
      headerName: "Entries",
      flex: 0.4,
      minWidth: 50,
      align: "center",
      headerAlign: "center",
    },
    {
      field: "totalCredit",
      headerName: "Total Credit",
      flex: 0.7,
      minWidth: 120,
      align: "right",
      headerAlign: "right",
      renderCell: (params) => (
        <Typography sx={{ color: "success.dark", fontWeight: "bold" }}>
          {formatCurrency(params.value)}
        </Typography>
      ),
    },
    {
      field: "totalDebit",
      headerName: "Total Debit",
      flex: 0.7,
      minWidth: 120,
      align: "right",
      headerAlign: "right",
      renderCell: (params) => (
        <Typography sx={{ color: "error.dark", fontWeight: "bold" }}>
          {formatCurrency(params.value)}
        </Typography>
      ),
    },
    {
      field: "actions",
      headerName: "Actions",
      flex: 0.8,
      minWidth: 140,
      align: "center",
      headerAlign: "center",
      sortable: false,
      renderCell: (params) => (
        <Stack direction="row" spacing={0.5}>
          <Tooltip title="View Details">
            <IconButton
              onClick={() => handleOpenViewModal(params.row)}
              color="primary"
            >
              <VisibilityIcon />
            </IconButton>
          </Tooltip>
          <Tooltip title="Accept Batch">
            <IconButton
              onClick={() => handleOpenModal(params.row, "ACCEPTED")}
              color="success"
            >
              <CheckIcon />
            </IconButton>
          </Tooltip>
          <Tooltip title="Reject Batch">
            <IconButton
              onClick={() => handleOpenModal(params.row, "REJECTED")}
              color="error"
            >
              <ClearIcon />
            </IconButton>
          </Tooltip>
        </Stack>
      ),
    },
  ];

  const viewModalColumns = [
    {
      field: "journalId",
      headerName: "Journal ID",
      flex: 1.5,
      minWidth: 180,
    },
    {
      field: "branch",
      headerName: "Branch",
      flex: 0.7,
      minWidth: 80,
    },
    {
      field: "cgl",
      headerName: "CGL",
      flex: 1,
      minWidth: 120,
    },
    {
      field: "amount",
      headerName: "Amount",
      flex: 1,
      minWidth: 120,
      align: "right",
      headerAlign: "right",
      renderCell: (params) => formatCurrency(params.value),
    },
    {
      field: "transactionType",
      headerName: "Type",
      flex: 0.7,
      minWidth: 90,
      align: "center",
      headerAlign: "center",
      renderCell: (params) => (
        <Chip
          label={params.value}
          color={params.value === "Debit" ? "error" : "success"}
          size="small"
          variant="outlined"
        />
      ),
    },
    {
      field: "productType",
      headerName: "Product",
      flex: 1,
      minWidth: 100,
    },
    {
      field: "remarks",
      headerName: "Remarks",
      flex: 1.5,
      minWidth: 150,
    },
  ];

  return (
    <Box sx={{ p: 3, width: "100%" }}>
      <Typography
        variant="h4"
        gutterBottom
        sx={{ fontWeight: "bold", color: "primary.main", mb: 3 }}
      >
        MOC Request Approval
      </Typography>

      <DataGrid
        rows={rows}
        columns={mainColumns}
        loading={loading}
        getRowId={(row) => row.batchId}
        disableRowSelectionOnClick
        pageSizeOptions={[5, 10, 25]}
        initialState={{
          pagination: { paginationModel: { pageSize: 10, page: 0 } },
          sorting: { sortModel: [{ field: "requestDate", sort: "desc" }] },
        }}
        slots={{
          noRowsOverlay: CustomNoRowsOverlay,
        }}
        sx={{
          height: 600,
          width: "100%",
          "& .MuiDataGrid-cell:focus": { outline: "none" },
        }}
      />

      <Dialog
        open={isModalOpen}
        onClose={handleCloseModal}
        fullWidth
        maxWidth="sm"
      >
        <DialogTitle
          sx={{
            backgroundColor:
              modalMode === "ACCEPTED" ? "success.main" : "error.main",
            color: "white",
            fontWeight: "bold",
          }}
        >
          {modalMode === "ACCEPTED" ? "Approve" : "Reject"} Batch
        </DialogTitle>
        <DialogContent sx={{ pt: 3, mt: 2 }}>
          <DialogContentText sx={{ mb: 2 }}>
            You are about to{" "}
            <strong
              style={{
                color: modalMode === "ACCEPTED" ? "green" : "red",
                textTransform: "uppercase",
              }}
            >
              {modalMode}
            </strong>{" "}
            the entire batch <strong>{selectedBatch?.batchId}</strong>.
            {modalMode === "REJECTED" &&
              " Please provide mandatory remarks below."}
          </DialogContentText>
          <TextField
            autoFocus={modalMode === "REJECTED"}
            required={modalMode === "REJECTED"}
            margin="dense"
            id="executorRemarks"
            label="Approver Remarks"
            type="text"
            fullWidth
            multiline
            rows={3}
            value={executorRemarks}
            onChange={(e) => setExecutorRemarks(e.target.value)}
            disabled={isSubmitting}
            InputProps={{
              inputProps: {
                maxLength: 200,
              },
            }}
          />
        </DialogContent>
        <DialogActions sx={{ p: 2 }}>
          <Button onClick={handleCloseModal} disabled={isSubmitting}>
            Cancel
          </Button>
          <Button
            onClick={handleSubmitAction}
            variant="contained"
            color={modalMode === "ACCEPTED" ? "success" : "error"}
            disabled={isSubmitting}
          >
            {isSubmitting ? (
              <CircularProgress size={24} color="inherit" />
            ) : (
              `Confirm ${modalMode === "ACCEPTED" ? "Approval" : "Rejection"}`
            )}
          </Button>
        </DialogActions>
      </Dialog>

      <Dialog
        open={isViewModalOpen}
        onClose={handleCloseViewModal}
        fullWidth
        maxWidth="xl"
      >
        <DialogTitle
          sx={{
            backgroundColor: "primary.main",
            color: "white",
            fontWeight: "bold",
          }}
        >
          Batch Details - {selectedBatch?.batchId}
        </DialogTitle>
        <DialogContent sx={{ pt: 3, mt: 2 }}>
          <DataGrid
            rows={viewModalRows}
            columns={viewModalColumns}
            getRowId={(row) => row.journalId}
            checkboxSelection
            disableRowSelectionOnClick
            onRowSelectionModelChange={(newSelectionModel) => {
              setViewModalSelection(newSelectionModel);
            }}
            rowSelectionModel={viewModalSelection}
            loading={isSubmitting}
            pageSizeOptions={[5, 10, 25]}
            initialState={{
              pagination: { paginationModel: { pageSize: 5, page: 0 } },
            }}
            sx={{
              height: 400,
              width: "100%",
              mb: 2,
              "& .MuiDataGrid-cell:focus": { outline: "none" },
            }}
          />
          <TextField
            required
            margin="dense"
            id="partialApprovalRemarks"
            label="Approver Remarks for Selected"
            type="text"
            fullWidth
            multiline
            rows={3}
            value={partialApprovalRemarks}
            onChange={(e) => setPartialApprovalRemarks(e.target.value)} // <-- THIS WAS THE FIX
            disabled={isSubmitting}
            InputProps={{
              inputProps: {
                maxLength: 200,
              },
            }}
          />
        </DialogContent>
        <DialogActions sx={{ p: 2 }}>
          <Button onClick={handleCloseViewModal} disabled={isSubmitting}>
            Cancel
          </Button>
          <Button
            onClick={handlePartialApprovalSubmit}
            variant="contained"
            color="success"
            disabled={isSubmitting || viewModalSelection.length ===0}
          >
            {isSubmitting ? (
              <CircularProgress size={24} color="inherit" />
            ) : (
              `Approve Selected (${viewModalSelection.length})`
            )}
          </Button>
        </DialogActions>
      </Dialog>
    </Box>
  );
}
