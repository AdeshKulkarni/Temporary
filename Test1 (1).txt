const handleEditorDrop = (e) => {
  e.preventDefault();
  const text = e.dataTransfer.getData("text");
  if (!text) return;

  const quill = bodyEditorRef.current?.getEditor();
  if (!quill) return;
  
  // Get the location where the user is dropping the tag
  const range = quill.getSelection(true);
  if (!range) return;

  // --- VALIDATION LOGIC ---

  // 1. Check the content immediately BEFORE the drop position
  if (range.index > 0) {
    const precedingContent = quill.getContents(range.index - 1, 1);
    const precedingOp = precedingContent.ops[0];
    if (
      precedingOp &&
      typeof precedingOp.insert === "object" &&
      precedingOp.insert.variable === text
    ) {
      setSbar({
        isOpen: true,
        Msg: "❌ Cannot add the same variable consecutively.",
      });
      return; // Block the drop
    }
  }

  // 2. Check the content immediately AFTER the drop position
  const succeedingContent = quill.getContents(range.index, 1);
  if (succeedingContent.ops.length > 0) {
    const succeedingOp = succeedingContent.ops[0];
    if (
      succeedingOp &&
      typeof succeedingOp.insert === "object" &&
      succeedingOp.insert.variable === text
    ) {
      setSbar({
        isOpen: true,
        Msg: "❌ Cannot add the same variable consecutively.",
      });
      return; // Block the drop
    }
  }

  // --- END VALIDATION ---

  // If validation passes, insert the variable
  quill.insertEmbed(range.index, "variable", text, Quill.sources.USER);
  quill.setSelection(range.index + 1, Quill.sources.USER);
};
