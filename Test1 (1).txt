class Solution {
    public int widthOfBinaryTree(TreeNode root) {

        if (root == null) {
            return 0;
        }

        Queue<TreeNode> q = new LinkedList<>();

        int MaxCount = 0;

        int DoubleCounter = 1;

        q.offer(root);

        while (!q.isEmpty()) {

            int Qsize = q.size();

            for (int i = 0; i < Qsize; i++) {

                if (Qsize > 1) {

                    DoubleCounter = DoubleCounter * 2;

                    int diff = DoubleCounter - Qsize;

                    MaxCount = diff + DoubleCounter;

                }

                if (root.left != null) {
                    q.offer(root.left);
                }

                if (root.right != null) {
                    q.offer(root.right);
                }

            }

        }

        return MaxCount;

    }
}
