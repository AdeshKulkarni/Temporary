import java.util.*;
import java.util.logging.Logger;



public class Main {

   
public static Boolean  rec(int[] arr , int index , int target) {
    
    if(index == arr.length) {
        return false;
    }
    
    int x =  Math.abs(arr[index] - target);
    
    D.p(x);
    
    // if(arr[index])
    
    Boolean ans = findRest(arr , x , index);
    
    Boolean finalAns = rec(arr , index+1 , target);
    
    return ans || finalAns;
    
}


public static Boolean findRest(int[] arr , int x , int index) {
    
    if(index == arr.length) {
        return false;
    }
    
    if(x == arr[index]) {
        return true;
    }
    
    return findRest(arr , x , index + 1);
}
            

    
    
    
    public static void main(String[] args) {
        
    
      
      int arr[]=  {1,2,3,4,5};
      
      System.out.print(rec(arr , 0 , 3));
      
    }
}



class D {
    static void p(Object... o) {
        StringBuilder sb = new StringBuilder();
        for (int i = 0; i < o.length; i += 2) {
            String val = (i + 1 < o.length) ? Arrays.deepToString(new Object[]{o[i + 1]}) : "[]";
            sb.append(o[i]).append(": ").append(val.substring(1, val.length() - 1)).append("  |  ");
        }
        System.out.println(sb);
    }
}
