import React, { useState, useEffect } from 'react';
import {
    Box,
    Typography,
    Fab,
    Dialog,
    DialogActions,
    DialogContent,
    DialogTitle,
    TextField,
    Button
    // --- REMOVED --- Container and Paper are no longer needed for the main layout
} from '@mui/material';
import AddIcon from '@mui/icons-material/Add';
import EditIcon from '@mui/icons-material/Edit';
import { DataGrid, GridActionsCellItem } from '@mui/x-data-grid';


const fetchCurrencyData = () => {
    return new Promise((resolve) => {
        const dataFromBackend = [
            { id: 1, currencyCode: 'USD', currencyName: 'United States Dollar', currencyRate: 1.00, createdAt: '2025-09-29T10:00:00Z' },
            { id: 2, currencyCode: 'EUR', currencyName: 'Euro', currencyRate: 0.92, createdAt: '2025-09-29T11:30:00Z' },
            { id: 3, currencyCode: 'INR', currencyName: 'Indian Rupee', currencyRate: 83.50, createdAt: '2025-09-29T12:00:00Z' },
            { id: 4, currencyCode: 'GBP', currencyName: 'British Pound', currencyRate: 0.79, createdAt: '2025-09-28T09:00:00Z' },
            { id: 5, currencyCode: 'JPY', currencyName: 'Japanese Yen', currencyRate: 149.82, createdAt: '2025-09-29T14:45:00Z' },
        ];
        setTimeout(() => resolve(dataFromBackend), 1500);
    });
};

const initialFormState = {
    currencyCode: '',
    currencyName: '',
    currencyRate: '',
};

function CurrencyMaster() {

    const [rows, setRows] = useState([]);
    const [loading, setLoading] = useState(true);
    const [isDialogOpen, setIsDialogOpen] = useState(false);
    const [formData, setFormData] = useState(initialFormState);
    const [editingRow, setEditingRow] = useState(null);
    const [errors, setErrors] = useState({});

    useEffect(() => {
        setLoading(true);
        fetchCurrencyData().then(data => {
            setRows(data);
            setLoading(false);
        }).catch(error => {
            console.error("Failed to fetch currency data:", error);
            setLoading(false);
        });
    }, []);

    const validate = (fieldValues = formData) => {
        let tempErrors = { ...errors };

        if ('currencyName' in fieldValues) {
            tempErrors.currencyName = '';
            if (!fieldValues.currencyName) tempErrors.currencyName = "Currency Name is required.";
            else if (/[0-9]/.test(fieldValues.currencyName)) tempErrors.currencyName = "Numbers are not allowed.";
        }

        if (!editingRow) {
            if ('currencyCode' in fieldValues) {
                tempErrors.currencyCode = '';
                if (!fieldValues.currencyCode) tempErrors.currencyCode = "Currency Code is required.";
                else if (fieldValues.currencyCode.length < 3) tempErrors.currencyCode = "Must be exactly 3 alphabets.";
            }
            if ('currencyRate' in fieldValues) {
                tempErrors.currencyRate = '';
                if (!fieldValues.currencyRate) tempErrors.currencyRate = "Rate is required.";
            }
        }
        setErrors(tempErrors);
        return Object.values(tempErrors).every(x => x === "");
    };

    const handleInputChange = (e) => {
        const { name, value } = e.target;

        switch (name) {
            case 'currencyCode':
                if (!/^[a-zA-Z]*$/.test(value)) {
                    setErrors({ ...errors, currencyCode: "Only letters are allowed." });
                    return;
                }
                break;
            case 'currencyName':
                if (/[0-9]/.test(value)) {
                    setErrors({ ...errors, currencyName: "Numbers are not allowed." });
                    return;
                }
                break;
            case 'currencyRate':
                if (value !== "" && !/^\d*\.?\d*$/.test(value)) {
                    setErrors({ ...errors, currencyRate: "Only numbers and '.' are allowed." });
                    return;
                }
                const [integerPart, decimalPart] = value.split('.');
                if (integerPart.length > 6 || (decimalPart && decimalPart.length > 6)) {
                    setErrors({ ...errors, currencyRate: "Input is too long." });
                    return;
                }
                break;
            default:
                break;
        }

        let processedValue = value;
        if (name === 'currencyCode') processedValue = value.toUpperCase().slice(0, 3);
        if (name === 'currencyName') processedValue = value.slice(0, 30);

        const newFormData = { ...formData, [name]: processedValue };
        setFormData(newFormData);

        validate({ [name]: processedValue });
    };


    const handleEditClick = (id) => () => {
        const rowToEdit = rows.find((row) => row.id === id);
        if (rowToEdit) {
            setEditingRow(rowToEdit);
            setFormData(rowToEdit);
            setIsDialogOpen(true);
        }
    };

    const columns = [
        { field: 'id', headerName: 'ID', width: 70, headerAlign: 'center', align: 'center' },
        { field: 'currencyCode', headerName: 'Currency Code', width: 150, headerAlign: 'center', align: 'center' },
        { field: 'currencyName', headerName: 'Currency Name', flex: 1, minWidth: 200, headerAlign: 'center' },
        { field: 'currencyRate', headerName: 'Currency Rate', type: 'number', width: 150, headerAlign: 'center', align: 'center' },
        {
            field: 'createdAt', headerName: 'Created At', type: 'dateTime', width: 220,
            valueGetter: (params) => params.value ? new Date(params.value) : null,
            renderCell: (params) => params.value ? new Date(params.value).toLocaleString() : '',
            headerAlign: 'center', align: 'center'
        },
        {
            field: 'actions', type: 'actions', headerName: 'Actions', width: 100,
            cellClassName: 'actions', headerAlign: 'center', align: 'center',
            getActions: ({ id }) => [<GridActionsCellItem icon={<EditIcon />} label="Edit" onClick={handleEditClick(id)} color="primary" />],
        },
    ];

    const handleOpenDialog = () => {
        setEditingRow(null);
        setFormData(initialFormState);
        setIsDialogOpen(true);
    };

    const handleCloseDialog = () => {
        setIsDialogOpen(false);
        setEditingRow(null);
        setFormData(initialFormState);
        setErrors({});
    };

    const handleSubmit = () => {
        if (validate()) {
            const now = new Date().toISOString();
            if (editingRow) {
                setRows(rows.map((row) => (row.id === editingRow.id ? formData : row)));
            } else {
                const newId = rows.length > 0 ? Math.max(...rows.map(r => r.id)) + 1 : 1;
                setRows([...rows, { ...formData, id: newId, createdAt: now }]);
            }
            handleCloseDialog();
        }
    };

    return (
        // --- CHANGED ---
        // Replaced Container and Paper with a single Box to allow full-width layout
        <Box sx={{ width: '100%', py: 2 }}>
            <Typography variant="h4" component="h1" sx={{ fontWeight: '600', color: 'primary.main', textAlign: 'center', mb: 3 }}>
                Currency Master
            </Typography>

            {/* This Box contains the DataGrid and controls its height */}
            <Box sx={{ height: 650, width: '100%' }}>
                <DataGrid
                    rows={rows}
                    columns={columns}
                    loading={loading}
                    disableRowSelectionOnClick
                    sx={{
                        '& .MuiDataGrid-row:nth-of-type(odd)': {
                            backgroundColor: (theme) => theme.palette.action.hover,
                        },
                        // Ensure headers are bold and centered
                        '& .MuiDataGrid-columnHeaderTitle': {
                            fontWeight: 'bold',
                        }
                    }}
                />
            </Box>

            <Fab color="primary" aria-label="add" onClick={handleOpenDialog} sx={{ position: 'fixed', bottom: 30, right: 30 }}>
                <AddIcon />
            </Fab>

            <Dialog open={isDialogOpen} onClose={handleCloseDialog} fullWidth maxWidth="sm">
                <DialogTitle sx={{ fontWeight: 'bold' }}>
                    {editingRow ? 'Edit Currency' : 'Add New Currency'}
                </DialogTitle>
                <DialogContent>
                    <TextField autoFocus margin="dense" name="currencyCode" label="Currency Code" type="text" fullWidth variant="outlined" value={formData.currencyCode} onChange={handleInputChange} error={!!errors.currencyCode} helperText={errors.currencyCode} disabled={!!editingRow} />
                    <TextField margin="dense" name="currencyName" label="Currency Name" type="text" fullWidth variant="outlined" value={formData.currencyName} onChange={handleInputChange} error={!!errors.currencyName} helperText={errors.currencyName} />
                    <TextField margin="dense" name="currencyRate" label="Currency Rate" type="text" fullWidth variant="outlined" value={formData.currencyRate} onChange={handleInputChange} error={!!errors.currencyRate} helperText={errors.currencyRate} disabled={!!editingRow} />
                </DialogContent>
                <DialogActions sx={{ p: '16px 24px' }}>
                    <Button onClick={handleCloseDialog}>Cancel</Button>
                    <Button onClick={handleSubmit} variant="contained">Submit</Button>
                </DialogActions>
            </Dialog>
        </Box>
    );
}

export default CurrencyMaster;
