package com.fincore.VoucherService.Exception;

import com.fincore.VoucherService.Dto.ApiErrorResponseDto;
import jakarta.servlet.http.HttpServletRequest;
import lombok.extern.slf4j.Slf4j;
import org.springframework.dao.DataAccessException;
import org.springframework.dao.DataIntegrityViolationException;
import org.springframework.http.HttpStatus;
import org.springframework.http.ResponseEntity;
import org.springframework.web.bind.MethodArgumentNotValidException;
import org.springframework.web.bind.annotation.ExceptionHandler;
import org.springframework.web.bind.annotation.RestControllerAdvice;
import org.springframework.web.multipart.MaxUploadSizeExceededException;

import java.io.IOException;
import java.time.LocalDateTime;
import java.util.List;
import java.util.stream.Collectors;

@RestControllerAdvice
@Slf4j
public class GlobalExceptionHandler {

    @ExceptionHandler(MethodArgumentNotValidException.class)
    public ResponseEntity<ApiErrorResponseDto> handleValidation(MethodArgumentNotValidException ex, HttpServletRequest request) {
        List<ApiErrorResponseDto.ValidationError> errors = ex.getBindingResult().getFieldErrors().stream()
                .map(err -> ApiErrorResponseDto.ValidationError.builder()
                        .field(err.getField())
                        .message(err.getDefaultMessage())
                        .build())
                .collect(Collectors.toList());

        String joinedMessage = ex.getBindingResult().getFieldErrors().stream()
                .map(org.springframework.validation.FieldError::getDefaultMessage)
                .collect(Collectors.joining(" | "));

        log.warn("VALIDATION_FAILED | Path: {}", sanitizeLog(request.getRequestURI()));
        return buildResponse(HttpStatus.BAD_REQUEST, "Validation Failed", joinedMessage, request.getRequestURI(), errors);
    }

    @ExceptionHandler(MaxUploadSizeExceededException.class)
    public ResponseEntity<ApiErrorResponseDto> handleMaxSize(MaxUploadSizeExceededException ex, HttpServletRequest request) {
        log.error("FILE_TOO_LARGE | Path: {}", sanitizeLog(request.getRequestURI()));
        return buildResponse(HttpStatus.PAYLOAD_TOO_LARGE, "File Too Large", "Maximum upload size exceeded", request.getRequestURI(), null);
    }

    @ExceptionHandler(CustomExceptions.FileProcessingException.class)
    public ResponseEntity<ApiErrorResponseDto> handleFileProcessing(CustomExceptions.FileProcessingException ex, HttpServletRequest request) {
        log.error("FILE_PROCESS_ERROR | Path: {}", sanitizeLog(request.getRequestURI()));
        // FORTIFY FIX: System Info Leak (Do not expose ex.getMessage() to frontend)
        return buildResponse(HttpStatus.BAD_REQUEST, "File Error", "Failed to process the uploaded file due to invalid format or corrupted data.", request.getRequestURI(), null);
    }

    @ExceptionHandler(CustomExceptions.ConcurrencyLockException.class)
    public ResponseEntity<ApiErrorResponseDto> handleConcurrency(CustomExceptions.ConcurrencyLockException ex, HttpServletRequest request) {
        log.warn("CONCURRENCY_CONFLICT | Path: {}", sanitizeLog(request.getRequestURI()));
        return buildResponse(HttpStatus.CONFLICT, "Operation Conflict", "The requested resource is currently locked by another process.", request.getRequestURI(), null);
    }

    @ExceptionHandler(ResourceNotFoundException.class)
    public ResponseEntity<ApiErrorResponseDto> handleNotFound(ResourceNotFoundException ex, HttpServletRequest request) {
        log.info("RESOURCE_NOT_FOUND | Path: {}", sanitizeLog(request.getRequestURI()));
        return buildResponse(HttpStatus.NOT_FOUND, "Not Found", "The requested resource could not be found.", request.getRequestURI(), null);
    }

    @ExceptionHandler(CustomExceptions.CacheExpiredException.class)
    public ResponseEntity<ApiErrorResponseDto> handleCacheExpired(CustomExceptions.CacheExpiredException ex, HttpServletRequest request) {
        log.warn("CACHE_EXPIRED | Path: {}", sanitizeLog(request.getRequestURI()));
        return buildResponse(HttpStatus.GONE, "Session Expired", "Data cache expired. Please re-initiate the process.", request.getRequestURI(), null);
    }

    @ExceptionHandler(DataIntegrityViolationException.class)
    public ResponseEntity<ApiErrorResponseDto> handleSqlIntegrity(DataIntegrityViolationException ex, HttpServletRequest request) {
        log.error("DB_INTEGRITY_VIOLATION | Path: {}", sanitizeLog(request.getRequestURI()));
        return buildResponse(HttpStatus.CONFLICT, "Data Integrity Error", "Constraint violation or duplicate entry encountered.", request.getRequestURI(), null);
    }

    @ExceptionHandler(IOException.class)
    public ResponseEntity<ApiErrorResponseDto> handleIoException(IOException ex, HttpServletRequest request) {
        log.error("IO_STREAM_FAILURE | Path: {}", sanitizeLog(request.getRequestURI()));
        return buildResponse(HttpStatus.INTERNAL_SERVER_ERROR, "IO Error", "Internal storage access failure.", request.getRequestURI(), null);
    }

    @ExceptionHandler(IllegalStateException.class)
    public ResponseEntity<ApiErrorResponseDto> handleStateErrors(IllegalStateException ex, HttpServletRequest request) {
        log.warn("INVALID_STATE | Path: {}", sanitizeLog(request.getRequestURI()));
        return buildResponse(HttpStatus.CONFLICT, "Invalid State", "The action cannot be performed in the current workflow state.", request.getRequestURI(), null);
    }

    @ExceptionHandler({NullPointerException.class, ArrayIndexOutOfBoundsException.class})
    public ResponseEntity<ApiErrorResponseDto> handleInternalLogicErrors(Exception ex, HttpServletRequest request) {
        log.error("INTERNAL_LOGIC_ERROR | Path: {}", sanitizeLog(request.getRequestURI()));
        return buildResponse(HttpStatus.INTERNAL_SERVER_ERROR, "Processing Error", "An internal processing error occurred.", request.getRequestURI(), null);
    }

    @ExceptionHandler(org.springframework.dao.DataAccessException.class)
    public ResponseEntity<ApiErrorResponseDto> handleDatabaseOperation(org.springframework.dao.DataAccessException ex, HttpServletRequest request) {
        log.error("DATABASE_FAULT | Path: {}", sanitizeLog(request.getRequestURI()));
        return buildResponse(HttpStatus.INTERNAL_SERVER_ERROR, "Database Error", "A technical error occurred in the data layer. Please try again later.", request.getRequestURI(), null);
    }

    @ExceptionHandler(CustomExceptions.InvalidDataException.class)
    public ResponseEntity<ApiErrorResponseDto> handleInvalidData(CustomExceptions.InvalidDataException ex, HttpServletRequest request) {
        log.warn("INVALID_DATA | Path: {}", sanitizeLog(request.getRequestURI()));
        return buildResponse(HttpStatus.BAD_REQUEST, "Invalid Data", "The provided payload contains invalid or business-restricted data.", request.getRequestURI(), null);
    }

    @ExceptionHandler(CustomExceptions.ScheduleLapsedException.class)
    public ResponseEntity<ApiErrorResponseDto> handleScheduleLapsed(CustomExceptions.ScheduleLapsedException ex, HttpServletRequest request) {
        log.warn("SCHEDULE_LAPSED | Path: {}", sanitizeLog(request.getRequestURI()));
        return buildResponse(HttpStatus.CONFLICT, "SCHEDULE_LAPSED", "The scheduled time for this action has already passed.", request.getRequestURI(), null);
    }

    @ExceptionHandler(org.springframework.http.converter.HttpMessageNotReadableException.class)
    public ResponseEntity<ApiErrorResponseDto> handleJsonParseError(org.springframework.http.converter.HttpMessageNotReadableException ex, HttpServletRequest request) {
        log.error("JSON_PARSE_ERROR | Path: {}", sanitizeLog(request.getRequestURI()));
        return buildResponse(HttpStatus.BAD_REQUEST, "Malformed JSON Payload", "The server could not read the malformed request payload.", request.getRequestURI(), null);
    }

    private ResponseEntity<ApiErrorResponseDto> buildResponse(HttpStatus status, String error, String message, String path, List<ApiErrorResponseDto.ValidationError> validationErrors) {
        ApiErrorResponseDto response = ApiErrorResponseDto.builder()
                .timestamp(LocalDateTime.now())
                .status(status.value())
                .error(error)
                .message(message)
                .path(sanitizeLog(path))
                .validationErrors(validationErrors)
                .build();
        return new ResponseEntity<>(response, status);
    }

    // FORTIFY FIX: Log Forging (Removes CRLF characters from logged strings)
    private String sanitizeLog(String input) {
        if (input == null) return "null";
        return input.replaceAll("[\r\n]", "_");
    }
}
package com.fincore.VoucherService.Service.validation;

import com.fasterxml.jackson.core.JsonProcessingException;
import com.fasterxml.jackson.core.type.TypeReference;
import com.fasterxml.jackson.databind.ObjectMapper;
import com.fincore.VoucherService.Dto.BulkUploadStateDto;
import com.fincore.VoucherService.Dto.ExcelRowData;
import com.fincore.VoucherService.Exception.CustomExceptions;
import lombok.extern.slf4j.Slf4j;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.beans.factory.annotation.Qualifier;
import org.springframework.dao.DataAccessException;
import org.springframework.data.redis.core.RedisTemplate;
import org.springframework.stereotype.Service;

import java.io.ByteArrayInputStream;
import java.io.ByteArrayOutputStream;
import java.io.IOException;
import java.util.HashMap;
import java.util.List;
import java.util.Map;
import java.util.concurrent.TimeUnit;
import java.util.concurrent.locks.LockSupport;
import java.util.function.Consumer;
import java.util.zip.GZIPInputStream;
import java.util.zip.GZIPOutputStream;

@Service
@Slf4j
public class ValidationStateManager {

    private final ObjectMapper objectMapper;

    @Autowired
    @Qualifier("byteArrayRedisTemplate")
    private RedisTemplate<String, byte[]> redisTemplate;

    public ValidationStateManager(ObjectMapper objectMapper) {
        this.objectMapper = objectMapper;
    }

    public void saveStatusToRedis(String reqId, BulkUploadStateDto state) {
        try {
            byte[] json = objectMapper.writeValueAsBytes(state);
            redisTemplate.opsForValue().set("BULK_STATUS_" + reqId, json, 2, TimeUnit.HOURS);
        } catch (JsonProcessingException e) {
            log.error("SERIALIZATION_FAIL | ReqID: {} | Status: Object mapping error", sanitizeLog(reqId));
        } catch (DataAccessException e) {
            log.error("REDIS_WRITE_FAIL | ReqID: {} | Status: Connection/Persistence failure", sanitizeLog(reqId));
        }
    }

    public BulkUploadStateDto getStatusFromRedis(String reqId) {
        try {
            byte[] json = redisTemplate.opsForValue().get("BULK_STATUS_" + reqId);
            if (json == null) return null;
            return objectMapper.readValue(json, BulkUploadStateDto.class);
        } catch (IOException | DataAccessException e) {
            log.error("REDIS_READ_FAIL | ReqID: {} | Type: Error reading from cache", sanitizeLog(reqId));
            return null;
        }
    }

    public void updateState(String requestId, Consumer<BulkUploadStateDto> updater) {
        String lockKey = "LOCK_STATE_" + requestId;
        int maxRetries = 50; 
        long delayMs = 50;

        for (int i = 0; i < maxRetries; i++) {
            Boolean acquired = redisTemplate.opsForValue().setIfAbsent(lockKey, new byte[0], 5, TimeUnit.SECONDS);

            if (Boolean.TRUE.equals(acquired)) {
                try {
                    BulkUploadStateDto state = getStatusFromRedis(requestId);
                    if (state == null) {
                        state = new BulkUploadStateDto();
                        state.setRequestId(requestId);
                    }
                    updater.accept(state);
                    saveStatusToRedis(requestId, state);
                    return; 
                } finally {
                    redisTemplate.delete(lockKey);
                }
            }

            // FORTIFY FIX: J2EE Bad Practices: Threads (Replaced forbidden Thread.sleep)
            LockSupport.parkNanos(TimeUnit.MILLISECONDS.toNanos(delayMs));
            
            if (Thread.currentThread().isInterrupted()) {
                log.error("STATE_UPDATE_INTERRUPTED | ReqID: {}", sanitizeLog(requestId));
                return;
            }
        }

        log.error("STATE_UPDATE_TIMEOUT | ReqID: {} | Status: Failed to acquire distributed lock", sanitizeLog(requestId));
    }

    public int getTotalChunks(String requestId) {
        byte[] metaBytes = redisTemplate.opsForValue().get("DATA_META_" + requestId);
        if (metaBytes == null) {
            throw new CustomExceptions.CacheExpiredException("Validation data expired (TTL 1 Hour). Please re-upload.");
        }

        try {
            Map<String, Object> meta = objectMapper.readValue(metaBytes, new TypeReference<>() {});
            return (int) meta.get("totalChunks");
        } catch (IOException | DataAccessException e) {
            throw new CustomExceptions.FileProcessingException("Failed to read chunk metadata", e);
        }
    }

    public List<ExcelRowData> getChunkFromCache(String requestId, int chunkIndex) {
        byte[] compressedChunk = redisTemplate.opsForValue().get("DATA_CHUNK_" + requestId + "_" + chunkIndex);

        if (compressedChunk == null) {
            throw new CustomExceptions.CacheExpiredException("Missing data chunk " + chunkIndex + ". Please re-upload.");
        }

        try (GZIPInputStream gis = new GZIPInputStream(new ByteArrayInputStream(compressedChunk))) {
            return objectMapper.readValue(gis, new TypeReference<List<ExcelRowData>>() {});
        } catch (IOException | DataAccessException e) {
            throw new CustomExceptions.FileProcessingException("Failed to reconstruct cached data chunk " + chunkIndex, e);
        }
    }

    public void saveRowsToRedis(String requestId, List<ExcelRowData> rows) {
        if (rows == null || rows.isEmpty()) return;
        try {
            int chunkSize = 5000;
            int totalRows = rows.size();
            int totalChunks = (int) Math.ceil((double) totalRows / chunkSize);

            for (int i = 0; i < totalChunks; i++) {
                int start = i * chunkSize;
                int end = Math.min(start + chunkSize, totalRows);
                List<ExcelRowData> chunk = rows.subList(start, end);

                try (ByteArrayOutputStream baos = new ByteArrayOutputStream();
                     GZIPOutputStream gos = new GZIPOutputStream(baos)) {
                    objectMapper.writeValue(gos, chunk);
                    gos.finish();
                    String chunkKey = "DATA_CHUNK_" + requestId + "_" + i;
                    redisTemplate.opsForValue().set(chunkKey, baos.toByteArray(), 1, TimeUnit.HOURS);
                }
            }

            Map<String, Object> meta = new HashMap<>();
            meta.put("totalChunks", totalChunks);
            meta.put("totalRows", totalRows);
            byte[] metaBytes = objectMapper.writeValueAsBytes(meta);
            redisTemplate.opsForValue().set("DATA_META_" + requestId, metaBytes, 1, TimeUnit.HOURS);

        } catch (IOException | DataAccessException e) {
            throw new CustomExceptions.RedisOperationException("Failed to cache large dataset", e);
        }
    }

    public byte[] getFileBytes(String reqId, String type) {
        try {
            return redisTemplate.opsForValue().get("FILE_" + reqId + "_" + type);
        } catch (DataAccessException e) {
            return null;
        }
    }

    public void saveErrorFileToRedis(String reqId, byte[] fileData) {
        try {
            redisTemplate.opsForValue().set("FILE_" + reqId + "_ERROR", fileData, 30, TimeUnit.MINUTES);
        } catch (DataAccessException e) {
            log.error("REDIS_WRITE_FAIL | ReqID: {} | Could not save error file to cache.", sanitizeLog(reqId));
        }
    }

    private String sanitizeLog(String input) {
        if (input == null) return "null";
        return input.replaceAll("[\r\n]", "_");
    }
}

package com.fincore.VoucherService.Controllers;

import com.fincore.VoucherService.Dto.BulkUploadStateDto;
import com.fincore.VoucherService.Exception.CustomExceptions;
import com.fincore.VoucherService.Service.validation.ValidationReportGenerator;
import com.fincore.VoucherService.Service.validation.VoucherBulkValidationService;
import jakarta.servlet.http.HttpServletRequest;
import lombok.RequiredArgsConstructor;
import lombok.extern.slf4j.Slf4j;
import org.springframework.core.io.ByteArrayResource;
import org.springframework.core.io.Resource;

import org.springframework.http.ResponseEntity;
import org.springframework.validation.annotation.Validated;
import jakarta.validation.constraints.Pattern;
import org.springframework.web.bind.annotation.*;
import org.springframework.web.multipart.MultipartFile;

import java.io.File;
import java.io.IOException;
import java.time.LocalDate;
import java.util.Map;

@RestController
@RequestMapping("/api/voucher")
@RequiredArgsConstructor
@Slf4j
@Validated
public class VoucherBulkUploadController {

    private final VoucherBulkValidationService voucherBulkValidationService;
    private final ValidationReportGenerator validationReportGenerator;

    @PostMapping(value = "/bulk-validate-init", consumes = MediaType.MULTIPART_FORM_DATA_VALUE)
    public ResponseEntity<Map<String, Object>> initiateValidation(
            @RequestParam("file") MultipartFile file,
            @RequestParam("postingDate") @DateTimeFormat(pattern = "dd/MM/yyyy") LocalDate postingDate,
            HttpServletRequest request) {

        if (file == null || file.isEmpty()) {
            return ResponseEntity.status(org.springframework.http.HttpStatus.BAD_REQUEST)
                    .body(Map.of("message", "File is missing or empty."));
        }

        // FORTIFY FIX: Often Misused: File Upload (Strips path traversal sequences)
        String originalFilename = file.getOriginalFilename() != null ? file.getOriginalFilename() : "unknown.xlsx";
        String safeFilename = new File(originalFilename).getName();

        try {
            String reqId = voucherBulkValidationService.initiateValidation(file, postingDate);

            HttpHeaders headers = new HttpHeaders();
            headers.setCacheControl(org.springframework.http.CacheControl.noCache().mustRevalidate().sMaxAge(java.time.Duration.ZERO));
            headers.setPragma("no-cache");
            headers.setExpires(0L);
            headers.set("X-Content-Type-Options", "nosniff");
            headers.set("Content-Security-Policy", "default-src 'self'");

            return ResponseEntity.ok()
                    .headers(headers)
                    .body(Map.of("status", "QUEUED", "requestId", reqId));

        } catch (CustomExceptions.FileProcessingException e) {
            return ResponseEntity.status(org.springframework.http.HttpStatus.BAD_REQUEST)
                    .body(Map.of("message", "File processing failed. Please ensure the file matches the expected format."));

        } catch (Exception e) {
            log.error("VALIDATION_INIT_FAIL | File: {} | Error: Unexpected Error", sanitizeLog(safeFilename));
            return ResponseEntity.status(org.springframework.http.HttpStatus.INTERNAL_SERVER_ERROR)
                    .body(Map.of("message", "An unexpected error occurred during upload."));
        }
    }

    @GetMapping("/bulk-status/{requestId}")
    public ResponseEntity<BulkUploadStateDto> checkStatus(
            @PathVariable @Pattern(regexp = "^[a-zA-Z0-9-]{1,50}$", message = "Invalid ID") String requestId) {
        BulkUploadStateDto state = voucherBulkValidationService.getState(requestId);
        return state != null ? ResponseEntity.ok(state) : ResponseEntity.notFound().build();
    }

    @GetMapping("/download-bulk-file/{requestId}")
    public ResponseEntity<Resource> downloadFile(
            @PathVariable @Pattern(regexp = "^[a-zA-Z0-9-]{1,50}$", message = "Invalid ID") String requestId, 
            @RequestParam @Pattern(regexp = "^[a-zA-Z0-9_]{1,20}$", message = "Invalid type") String type) {
        
        byte[] data = voucherBulkValidationService.getFileBytes(requestId, type);
        if (data == null) {
            log.warn("DOWNLOAD_FAIL | File not found | ReqID: {} | Type: {}", sanitizeLog(requestId), sanitizeLog(type));
            return ResponseEntity.notFound().build();
        }

        // FORTIFY FIX: Header Manipulation (Ensures no CRLF in header values)
        String safeRequestId = requestId.replaceAll("[^a-zA-Z0-9-]", "");
        String filename = "Error_Report_" + safeRequestId + ".xlsx";

        MediaType mediaType = MediaType.parseMediaType("application/vnd.openxmlformats-officedocument.spreadsheetml.sheet");

        return ResponseEntity.ok()
                .header(HttpHeaders.CONTENT_DISPOSITION, "attachment; filename=\"" + filename + "\"")
                .contentType(mediaType)
                .body(new ByteArrayResource(data));
    }

    @GetMapping("/download-template")
    public ResponseEntity<org.springframework.core.io.Resource> downloadTemplate() throws IOException {
        return ResponseEntity.ok().header(HttpHeaders.CONTENT_DISPOSITION, "attachment; filename=\"Voucher_Template.xlsx\"").body(new ByteArrayResource(validationReportGenerator.generateExcelTemplateWithGuidelines()));
    }

    private String sanitizeLog(String input) {
        if (input == null) return "null";
        return input.replaceAll("[\r\n]", "_");
    }
}

@RestController

    private final VoucherQueryService queryService;

    private final RequestVoucherService requestVoucherService;
    private final JwtUtil jwtUtil;

    @GetMapping("/by-batch/{batchId}")
    public ResponseEntity<?> getRequestsByBatchId(
            @PathVariable @Pattern(regexp = "^[0-9]{1,10}$", message = "Invalid Batch ID") String batchId) {
        long count = queryService.getRequestCountByBatchId(batchId);
        if (count > 2000) return ResponseEntity.status(HttpStatus.PAYLOAD_TOO_LARGE).body(Map.of("error", "Batch too large.", "suggestion", "/api/voucher/by-batch-paginated/" + sanitizeLog(batchId)));
        return ResponseEntity.ok(queryService.getRequestsByBatchId(batchId));
    }

    @GetMapping("/by-batch-paginated/{batchId}")
    public ResponseEntity<?> getRequestsByBatchIdPaginated(
            @PathVariable @Pattern(regexp = "^[0-9]{1,10}$", message = "Invalid Batch ID") String batchId, 
            @RequestParam(defaultValue = "0") int page, 
            @RequestParam(defaultValue = "10") int size) {
        return ResponseEntity.ok(queryService.getRequestsByBatchIdPaginated(batchId, PageRequest.of(page, size)));
    }

    @GetMapping("/download-batch/{batchId}")
    public ResponseEntity<?> downloadBatchExcel(
            @PathVariable @Pattern(regexp = "^[0-9]{1,10}$", message = "Invalid Batch ID") String batchId, 
            @RequestHeader("Authorization") String token) {
        try {
            if (!queryService.isUserAuthorizedForBatch(batchId, jwtUtil.getUserIdFromToken(token.substring(7)), jwtUtil.getUserRoleFromToken(token.substring(7)))) {
                return ResponseEntity.status(HttpStatus.FORBIDDEN).body(Map.of("message", "You are not authorized to view this batch."));
            }

            byte[] fileData = queryService.generateBatchExcel(batchId);

            // FORTIFY FIX: Header Manipulation (Clean strings entering headers)
            String safeBatchId = batchId.replaceAll("[^0-9]", "");

            org.springframework.http.HttpHeaders headers = new org.springframework.http.HttpHeaders();
            headers.setCacheControl(org.springframework.http.CacheControl.noCache().mustRevalidate().sMaxAge(java.time.Duration.ZERO));
            headers.set(org.springframework.http.HttpHeaders.CONTENT_DISPOSITION, "attachment; filename=\"Batch_" + safeBatchId + ".xlsx\"");
            headers.set(org.springframework.http.HttpHeaders.CONTENT_TYPE, "application/vnd.openxmlformats-officedocument.spreadsheetml.sheet");

            return ResponseEntity.ok().headers(headers).body(fileData);

        } catch (CustomExceptions.InvalidDataException e) {
            return ResponseEntity.status(HttpStatus.BAD_REQUEST).body(Map.of("message", "Invalid Request Data"));
        } catch (Exception e) {
            log.error("DOWNLOAD_ERROR | Batch: {} | Reason: Internal Error", sanitizeLog(batchId));
            return ResponseEntity.status(HttpStatus.INTERNAL_SERVER_ERROR).body(Map.of("message", "Failed to generate Excel file."));
        }
    }

            @RequestParam(required = false, defaultValue = "") @Pattern(regexp = "^[a-zA-Z0-9-]{0,30}$", message = "Invalid format") String vid) {
