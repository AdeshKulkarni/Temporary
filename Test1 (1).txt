    @Override
    @Transactional
    public List<JournalRequest> processBulkRequests(BulkProcessJournalRequestDto dto, String executorId, Integer executorRole) {
        long startTime = System.currentTimeMillis();
        
        // 1. Validate Input
        if (!StringUtils.hasText(dto.getBatchId()) && (dto.getJournalIdPrefixes() == null || dto.getJournalIdPrefixes().isEmpty())) {
            throw new IllegalArgumentException("Either Batch ID or Journal ID list is required.");
        }

        // 2. Identify Target Requests (Standard JPA is fine for selecting IDs, or use JDBC for speed)
        // We need to ensure we don't approve our own requests.
        // Instead of fetching entities, we execute updates directly.

        String targetClause;
        List<Object> args = new ArrayList<>();
        
        // Build WHERE clause based on input
        if (StringUtils.hasText(dto.getBatchId())) {
            targetClause = "BATCH_ID = ?";
            args.add(dto.getBatchId());
        } else {
            // "JOURNAL_ID LIKE 'PREFIX%'" logic
            List<String> conditions = new ArrayList<>();
            for (String prefix : dto.getJournalIdPrefixes()) {
                conditions.add("JOURNAL_ID LIKE ?");
                args.add(prefix + "%");
            }
            targetClause = "(" + String.join(" OR ", conditions) + ")";
        }

        // 3. IF APPROVING: Perform Bulk Insert into GL_TRANSACTIONS directly from JOURNAL_REQUEST
        if (RequestStatus.ACCEPTED.equals(dto.getStatus())) {
            log.info("Starting Bulk Approval Insert for executor: {}", executorId);
            
            // SQL Explanation:
            // Selects JSON fields from JOURNAL_REQUEST and Inserts directly into GL_TRANSACTIONS
            // Filters out Self-Approval (CREATOR_ID != executorId)
            
            String insertSql = """
                INSERT INTO GL_TRANSACTIONS (
                    TRANSACTION_ID, BATCH_ID, JOURNAL_ID, TRANSACTION_DATE, POST_DATE,
                    BRANCH_CODE, CURRENCY, CGL, NARRATION,
                    DEBIT_AMOUNT, CREDIT_AMOUNT, TRANSACTION_COUNT, SOURCE_FLAG
                )
                SELECT
                    GL_TRANSACTIONS_SEQ.nextval,
                    jr.BATCH_ID,
                    jr.JOURNAL_ID,
                    TO_DATE(JSON_VALUE(jr.PAYLOAD, '$.pDate'), 'YYYY-MM-DD'), -- Extract Date
                    SYSTIMESTAMP,
                    JSON_VALUE(jr.PAYLOAD, '$.branch'),
                    JSON_VALUE(jr.PAYLOAD, '$.currency'),
                    JSON_VALUE(jr.PAYLOAD, '$.cgl'),
                    JSON_VALUE(jr.PAYLOAD, '$.remarks'),
                    -- Debit Amount Logic: If Amount > 0, use it, else 0
                    CASE WHEN TO_NUMBER(JSON_VALUE(jr.PAYLOAD, '$.amount')) > 0 
                         THEN TO_NUMBER(JSON_VALUE(jr.PAYLOAD, '$.amount')) 
                         ELSE 0 END,
                    -- Credit Amount Logic: If Amount < 0, use ABS(Amount), else 0
                    CASE WHEN TO_NUMBER(JSON_VALUE(jr.PAYLOAD, '$.amount')) < 0 
                         THEN ABS(TO_NUMBER(JSON_VALUE(jr.PAYLOAD, '$.amount'))) 
                         ELSE 0 END,
                    TO_NUMBER(JSON_VALUE(jr.PAYLOAD, '$.transactionCount')),
                    'J'
                FROM JOURNAL_REQUEST jr
                WHERE jr.REQ_STATUS = 'PENDING'
                  AND jr.CREATOR_ID != ? -- SECURITY: Prevent Self-Approval
                  AND 
            """ + targetClause;

            // Add executorId to the beginning of args list for the query
            List<Object> insertArgs = new ArrayList<>();
            insertArgs.add(executorId);
            insertArgs.addAll(args);

            int rowsInserted = jdbcTemplate.update(insertSql, insertArgs.toArray());
            log.info("Bulk Approval: Inserted {} rows into GL_TRANSACTIONS", rowsInserted);
        }

        // 4. Update Status in JOURNAL_REQUEST (Bulk Update)
        String updateSql = "UPDATE JOURNAL_REQUEST SET REQ_STATUS = ?, EXECUTOR_ID = ?, EXECUTOR_REMARKS = ?, EXECUTION_DATE = SYSDATE " +
                           "WHERE REQ_STATUS = 'PENDING' AND CREATOR_ID != ? AND " + targetClause;

        List<Object> updateArgs = new ArrayList<>();
        updateArgs.add(dto.getStatus().getCode()); // 'A' or 'R'
        updateArgs.add(executorId);
        updateArgs.add(dto.getRemarks());
        updateArgs.add(executorId); // For CREATOR_ID != ? check
        updateArgs.addAll(args); // Add the Batch/Journal filters

        int rowsUpdated = jdbcTemplate.update(updateSql, updateArgs.toArray());
        log.info("Bulk Status Update: Updated {} rows to {}", rowsUpdated, dto.getStatus());

        log.info("Total Processing Time: {} ms", (System.currentTimeMillis() - startTime));
        
        // Return empty list to avoid fetching back 200,000 rows to Java
        return Collections.emptyList(); 
    }

    // --- SINGLE APPROVAL (Also Optimized) ---
    @Override
    @Transactional
    public Optional<JournalRequest> updateRequestStatus(ProcessJournalRequestDto dto, String executorId, Integer executorRole) {
        // Since we want speed, even single approval can use the bulk logic logic under hood
        // But for single items, JPA is acceptable. For consistency, let's use the bulk logic 
        // targeted at one specific ID to reuse the SQL optimization.
        
        BulkProcessJournalRequestDto bulkDto = new BulkProcessJournalRequestDto();
        bulkDto.setStatus(dto.getStatus());
        bulkDto.setRemarks(dto.getRemarks());
        // We need to fetch the specific journal ID first to use the Bulk Logic filter
        // OR just keep JPA for single item as 1 row is fast.
        
        // Keeping JPA for Single Item for simplicity unless you want it changed:
        JournalRequest request = journalRequestRepository.findById(dto.getRequestId())
                .orElseThrow(() -> new ResourceNotFoundException("Journal Request not found."));

        if (request.getRequestStatus() != RequestStatus.PENDING) throw new IllegalStateException("Already processed.");
        if (executorId.equals(request.getCreatorId())) throw new IllegalStateException("Self-approval not allowed.");

        // If Approved, Insert into GL
        if (RequestStatus.ACCEPTED.equals(dto.getStatus())) {
             // Use the same native SQL approach for single item to ensure consistency
             String insertSql = """
                INSERT INTO GL_TRANSACTIONS (
                    TRANSACTION_ID, BATCH_ID, JOURNAL_ID, TRANSACTION_DATE, POST_DATE,
                    BRANCH_CODE, CURRENCY, CGL, NARRATION,
                    DEBIT_AMOUNT, CREDIT_AMOUNT, TRANSACTION_COUNT, SOURCE_FLAG
                )
                SELECT
                    GL_TRANSACTIONS_SEQ.nextval,
                    BATCH_ID, JOURNAL_ID, TO_DATE(JSON_VALUE(PAYLOAD, '$.pDate'), 'YYYY-MM-DD'), SYSTIMESTAMP,
                    JSON_VALUE(PAYLOAD, '$.branch'), JSON_VALUE(PAYLOAD, '$.currency'), JSON_VALUE(PAYLOAD, '$.cgl'), JSON_VALUE(PAYLOAD, '$.remarks'),
                    CASE WHEN TO_NUMBER(JSON_VALUE(PAYLOAD, '$.amount')) > 0 THEN TO_NUMBER(JSON_VALUE(PAYLOAD, '$.amount')) ELSE 0 END,
                    CASE WHEN TO_NUMBER(JSON_VALUE(PAYLOAD, '$.amount')) < 0 THEN ABS(TO_NUMBER(JSON_VALUE(PAYLOAD, '$.amount'))) ELSE 0 END,
                    TO_NUMBER(JSON_VALUE(PAYLOAD, '$.transactionCount')), 'J'
                FROM JOURNAL_REQUEST WHERE REQ_ID = ?
            """;
            jdbcTemplate.update(insertSql, request.getId());
        }

        request.setExecutorId(executorId);
        request.setExecutorRemarks(dto.getRemarks());
        request.setExecutionDate(LocalDateTime.now());
        request.setRequestStatus(dto.getStatus());
        return Optional.of(journalRequestRepository.save(request));
    }

    // --- CANCELLATION METHODS ---
    @Override
    @Transactional
    public void cancelMyRequest(Long requestId, String userId) {
        JournalRequest request = journalRequestRepository.findById(requestId)
                .orElseThrow(() -> new ResourceNotFoundException("Request not found."));
        if (!userId.equals(request.getCreatorId())) throw new IllegalStateException("Unauthorized.");
        if (request.getRequestStatus() != RequestStatus.PENDING) throw new IllegalStateException("Already processed.");
        journalRequestRepository.delete(request);
    }

    @Override
    @Transactional
    public void cancelMyRequestsByBatchId(String batchId, String userId) {
        List<JournalRequest> list = journalRequestRepository.findByBatchIdAndCreatorIdAndRequestStatus(batchId, userId, RequestStatus.PENDING);
        if (list.isEmpty()) throw new ResourceNotFoundException("No pending requests found.");
        journalRequestRepository.deleteAll(list);
    }

    @Override
    @Transactional
    public void cancelMyRequestsByJournalPrefix(String prefix, String userId) {
        List<JournalRequest> list = journalRequestRepository.findByJournalIdStartingWithAndCreatorIdAndRequestStatus(prefix, userId, RequestStatus.PENDING);
        if (list.isEmpty()) throw new ResourceNotFoundException("No pending requests found.");
        journalRequestRepository.deleteAll(list);
    }

    @Override
    @Transactional
    public void cancelMyRequestsByJournalPrefixes(List<String> prefixes, String userId) {
         List<JournalRequest> requestsToCancel = journalRequestRepository.findAll((root, query, cb) -> {
            List<Predicate> predicates = new ArrayList<>();
            for (String prefix : prefixes) {
                predicates.add(cb.like(root.get("journalId"), prefix + "%"));
            }
            Predicate journalIdOr = cb.or(predicates.toArray(new Predicate[0]));
            Predicate status = cb.equal(root.get("requestStatus"), RequestStatus.PENDING);
            Predicate creator = cb.equal(root.get("creatorId"), userId);
            return cb.and(journalIdOr, status, creator);
        });
        if (requestsToCancel.isEmpty()) throw new ResourceNotFoundException("No pending requests found.");
        journalRequestRepository.deleteAll(requestsToCancel);
    }
