import React, { useState, useEffect, useCallback } from "react";
import {
  Box,
  Typography,
  Paper,
  Grid,
  Button,
  CircularProgress,
  Chip,
  Dialog,
  DialogActions,
  DialogContent,
  DialogContentText,
  DialogTitle,
  TextField,
  IconButton,
  Divider,
} from "@mui/material";
import CheckCircleIcon from "@mui/icons-material/CheckCircle";
import CancelIcon from "@mui/icons-material/Cancel";
import InfoIcon from "@mui/icons-material/Info";

// Import your custom hooks
import useApi from "../../../hooks/useApi"; // Adjust path as needed
import useCustomSnackbar from "../../../utils/useCustomSnackbar"; // Adjust path as needed

// Helper function to format currency
const formatCurrency = (num) => {
  return new Intl.NumberFormat("en-IN", {
    style: "decimal",
    minimumFractionDigits: 2,
    maximumFractionDigits: 2,
  }).format(num);
};

export default function MocRequestApproval() {
  const { callApi } = useApi();
  const showSnackBar = useCustomSnackbar();

  const [isLoading, setIsLoading] = useState(true);
  const [batches, setBatches] = useState([]);
  const [error, setError] = useState(null);

  const [isModalOpen, setIsModalOpen] = useState(false);
  const [isSubmitting, setIsSubmitting] = useState(false);
  const [selectedBatch, setSelectedBatch] = useState(null);
  const [modalMode, setModalMode] = useState("ACCEPT"); // 'ACCEPT' or 'REJECT'
  const [executorRemarks, setExecutorRemarks] = useState("");

  // Function to fetch and process data
  const fetchPendingBatches = useCallback(async () => {
    setIsLoading(true);
    setError(null);
    try {
      const pendingRequests = await callApi(
        "MS/moc/pending-requests",
        null,
        "GET"
      );

      if (!Array.isArray(pendingRequests) || pendingRequests.length === 0) {
        setBatches([]);
        setIsLoading(false);
        return;
      }

      // Group requests by batchId
      const groupedBatches = new Map();

      for (const request of pendingRequests) {
        const batchId = request.batchId;

        if (!groupedBatches.has(batchId)) {
          // Initialize batch details from the first request
          groupedBatches.set(batchId, {
            batchId: batchId,
            creatorId: request.creatorId,
            requestStatus: request.requestStatus,
            requestDate: request.requestDate,
            commonBatchRemarks: request.commonBatchRemarks,
            totalDebit: 0,
            totalCredit: 0,
            requestCount: 0,
          });
        }

        const batchDetails = groupedBatches.get(batchId);
        batchDetails.requestCount++;

        // Parse payload to calculate totals
        try {
          const payload = JSON.parse(request.payload);
          const amount = parseFloat(payload.amount) || 0;

          if (payload.transactionType === "Debit") {
            batchDetails.totalDebit += amount;
          } else if (payload.transactionType === "Credit") {
            batchDetails.totalCredit += amount;
          }
        } catch (e) {
          console.error(`Failed to parse payload for request ${request.id}`, e);
          // Handle parse error if necessary
        }
      }

      setBatches(Array.from(groupedBatches.values()));
    } catch (err) {
      console.error("Failed to fetch pending requests:", err);
      setError("Failed to load data. Please try again.");
      showSnackBar("Failed to load pending requests.", "error");
    } finally {
      setIsLoading(false);
    }
  }, [callApi, showSnackBar]);

  // Fetch data on component mount
  useEffect(() => {
    fetchPendingBatches();
  }, [fetchPendingBatches]);

  // --- Modal Handlers ---

  const handleOpenModal = (batch, mode) => {
    setSelectedBatch(batch);
    setModalMode(mode);
    setExecutorRemarks("");
    setIsModalOpen(true);
  };

  const handleCloseModal = () => {
    if (isSubmitting) return; // Prevent closing while submitting
    setIsModalOpen(false);
    setSelectedBatch(null);
  };

  const handleSubmitAction = async () => {
    if (!executorRemarks.trim()) {
      showSnackBar("Approver remarks are required.", "warning");
      return;
    }

    setIsSubmitting(true);
    const payload = {
      batchId: selectedBatch.batchId,
      status: modalMode,
      remarks: executorRemarks.trim(),
    };

    try {
      await callApi("MS/moc/process-bulk", payload, "POST");
      showSnackBar(
        `Batch ${selectedBatch.batchId} successfully ${modalMode.toLowerCase()}ed.`,
        "success"
      );
      handleCloseModal();
      // Refresh the list after successful action
      fetchPendingBatches();
    } catch (error) {
      console.error("Failed to process batch:", error);
      const errorMsg =
        error.response?.data?.message || `Failed to ${modalMode.toLowerCase()} batch.`;
      showSnackBar(errorMsg, "error");
    } finally {
      setIsSubmitting(false);
    }
  };

  // --- Render Functions ---

  if (isLoading) {
    return (
      <Box
        sx={{
          display: "flex",
          justifyContent: "center",
          alignItems: "center",
          height: "80vh",
        }}
      >
        <CircularProgress />
        <Typography sx={{ ml: 2 }}>Loading Pending Requests...</Typography>
      </Box>
    );
  }

  if (error) {
    return (
      <Box sx={{ p: 3 }}>
        <Paper elevation={3} sx={{ p: 3, color: "error.main" }}>
          <Typography variant="h6">Error</Typography>
          <Typography>{error}</Typography>
          <Button
            variant="contained"
            onClick={fetchPendingBatches}
            sx={{ mt: 2 }}
          >
            Retry
          </Button>
        </Paper>
      </Box>
    );
  }

  return (
    <Box sx={{ p: 3, maxWidth: "95%", margin: "auto" }}>
      <Typography
        variant="h4"
        gutterBottom
        sx={{ fontWeight: "bold", color: "primary.main", mb: 3 }}
      >
        MOC Pending Requests
      </Typography>

      {batches.length === 0 ? (
        <Paper elevation={3} sx={{ p: 3, textAlign: "center" }}>
          <InfoIcon sx={{ fontSize: 48, color: "text.secondary" }} />
          <Typography variant="h6" sx={{ mt: 2 }}>
            No Pending Requests Found
          </Typography>
          <Typography>
            All MOC requests have been processed.
          </Typography>
        </Paper>
      ) : (
        <Grid container spacing={3}>
          {batches.map((batch) => (
            <Grid item xs={12} key={batch.batchId}>
              <Paper elevation={3} sx={{ p: 3 }}>
                <Grid container spacing={2} alignItems="center">
                  {/* Batch Info Column */}
                  <Grid item xs={12} md={5}>
                    <Typography
                      variant="h6"
                      sx={{ fontWeight: "bold", color: "primary.dark" }}
                    >
                      Batch ID: {batch.batchId}
                    </Typography>
                    <Typography variant="body2" color="text.secondary">
                      Creator ID: {batch.creatorId}
                    </Typography>
                    <Typography variant="body2" color="text.secondary">
                      Submitted:{" "}
                      {new Date(batch.requestDate).toLocaleString()}
                    </Typography>
                    <Chip
                      label={batch.requestStatus}
                      color="warning"
                      size="small"
                      sx={{ mt: 1, fontWeight: "bold" }}
                    />
                  </Grid>

                  {/* Totals Column */}
                  <Grid item xs={12} md={4}>
                    <Typography
                      variant="body1"
                      sx={{ color: "success.dark", fontWeight: "bold" }}
                    >
                      Total Credit: {formatCurrency(batch.totalCredit)}
                    </Typography>
                    <Typography
                      variant="body1"
                      sx={{ color: "error.dark", fontWeight: "bold" }}
                    >
                      Total Debit: {formatCurrency(batch.totalDebit)}
                    </Typography>
                     <Typography
                      variant="body1"
                      color="text.secondary"
                    >
                      Entries: {batch.requestCount}
                    </Typography>
                  </Grid>

                  {/* Actions Column */}
                  <Grid item xs={12} md={3} sx={{ textAlign: "right" }}>
                    <Button
                      variant="contained"
                      color="success"
                      startIcon={<CheckCircleIcon />}
                      onClick={() => handleOpenModal(batch, "ACCEPTED")}
                      sx={{ mb: { xs: 1, md: 0 }, mr: { xs: 0, md: 1 } }}
                    >
                      Accept
                    </Button>
                    <Button
                      variant="contained"
                      color="error"
                      startIcon={<CancelIcon />}
                      onClick={() => handleOpenModal(batch, "REJECTED")}
                    >
                      Reject
                    </Button>
                  </Grid>

                  {/* Remarks Row */}
                  <Grid item xs={12} sx={{ pt: 2 }}>
                     <Divider sx={{ mb: 2 }} />
                    <Typography variant="body2" sx={{ fontStyle: "italic" }}>
                      <span style={{ fontWeight: "bold" }}>Creator Remarks:</span>{" "}
                      {batch.commonBatchRemarks || "N/A"}
                    </Typography>
                  </Grid>
                </Grid>
              </Paper>
            </Grid>
          ))}
        </Grid>
      )}

      {/* --- Action Confirmation Modal --- */}
      <Dialog open={isModalOpen} onClose={handleCloseModal} fullWidth maxWidth="sm">
        <DialogTitle
          sx={{
            backgroundColor:
              modalMode === "ACCEPTED" ? "success.main" : "error.main",
            color: "white",
            fontWeight: "bold",
          }}
        >
          {modalMode === "ACCEPTED" ? "Approve" : "Reject"} Batch
        </DialogTitle>
        <DialogContent sx={{ pt: 3 }}>
          <DialogContentText sx={{ mb: 2 }}>
            You are about to{" "}
            <strong style={{ color: modalMode === "ACCEPTED" ? "green" : "red" }}>
              {modalMode.toLowerCase()}
            </strong>{" "}
            the entire batch{" "}
            <strong>{selectedBatch?.batchId}</strong>.
          </DialogContentText>
          <TextField
            autoFocus
            required
            margin="dense"
            id="executorRemarks"
            label="Approver Remarks"
            type="text"
            fullWidth
            multiline
            rows={3}
            value={executorRemarks}
            onChange={(e) => setExecutorRemarks(e.target.value)}
            disabled={isSubmitting}
            InputProps={{
              inputProps: {
                maxLength: 200, // Or your desired max length
              },
            }}
          />
        </DialogContent>
        <DialogActions sx={{ p: 2 }}>
          <Button onClick={handleCloseModal} disabled={isSubmitting}>
            Cancel
          </Button>
          <Button
            onClick={handleSubmitAction}
            variant="contained"
            color={modalMode === "ACCEPTED" ? "success" : "error"}
            disabled={isSubmitting}
          >
            {isSubmitting ? (
              <CircularProgress size={24} color="inherit" />
            ) : (
              `Confirm ${modalMode === "ACCEPTED" ? "Approval" : "Rejection"}`
            )}
          </Button>
        </DialogActions>
      </Dialog>
    </Box>
  );
}
