// EmailTemplateSlate.jsx
import React, { useCallback, useMemo, useRef, useState } from "react";
import {
  Typography, Box, Grid, Card, Snackbar, Button, Divider,
  CardHeader, CardContent, Chip, Paper, IconButton, List, ListItem,
  ListItemIcon, ListItemText
} from "@mui/material";
import CheckCircleOutlineIcon from "@mui/icons-material/CheckCircleOutline";
import EditIcon from "@mui/icons-material/Edit";
import PreviewIcon from "@mui/icons-material/Preview";
import RestartAltIcon from "@mui/icons-material/RestartAlt";
import SaveIcon from "@mui/icons-material/Save";
import CancelIcon from "@mui/icons-material/Cancel";
import { createEditor, Transforms, Editor, Range, Element as SlateElement, Node } from "slate";
import { Slate, Editable, withReact, useSlateStatic, useSlate } from "slate-react";
import { css } from "@emotion/css";

/////////////////
// Variable & Toolbar Config
const draftVar = [
  "{{DATE}}",
  "{{REF_NO}}",
  "{{FIRM_NAME}}",
  "{{FRN_NO}}",
  "{{FIRM_ADDR}}",
  "{{ASSIGNMENT_TYPE}}",
  "{{GSTN}}",
];

const WORD_LIMIT = 2000;

/**
 * A custom Slate element for template variables (non-editable inline chips).
 */
const VariableElement = ({ attributes, children, element }) => (
  <span
    {...attributes}
    contentEditable={false}
    className={css`
      background: #a9a9a9;
      color: black;
      padding: 3px 14px 3px 6px;
      border-radius: 3px;
      user-select: none;
      display: inline-flex;
      align-items: center;
      position: relative;
      margin-right: 4px;
    `}
  >
    <span>{element.value}</span>
    <span
      contentEditable={false}
      style={{
        marginLeft: 4,
        cursor: "pointer",
        fontWeight: "bold",
        color: "#533",
        fontSize: 14,
      }}
      // Variable X button: remove variable
      onMouseDown={e => {
        e.preventDefault();
        const editor = window._editingEditor;
        if (editor) {
          const path = ReactEditor.findPath(editor, element);
          Transforms.removeNodes(editor, { at: path });
        }
      }}
      className="variable-delete-icon"
    >
      ×
    </span>
    {children}
  </span>
);

// Standard paragraph element for Slate
const DefaultElement = props => <p {...props.attributes}>{props.children}</p>;

// Insert variable helper
const insertVariable = (editor, value) => {
  const variable = {
    type: "variable",
    value,
    children: [{ text: "" }],
  };
  // Insert variable at current selection
  Transforms.insertNodes(editor, variable);
  // Place a space after
  Transforms.insertText(editor, " ");
};

//////////////////
// Slate plugins:
const withVariables = editor => {
  // Use isInline to make "variable" element inline
  const { isInline } = editor;
  editor.isInline = elem => elem.type === "variable" ? true : isInline(elem);
  return editor;
};

//////////////////
// Toolbar for bold/italic/underline
const ToolbarButton = ({ format, icon, ...props }) => {
  // For demo, only bold/italic/underline
  const editor = useSlate();
  const isActive = isMarkActive(editor, format);
  return (
    <Button
      size="small"
      variant={isActive ? "contained" : "outlined"}
      sx={{ minWidth: 32, fontWeight: isActive ? "bold" : "normal", mx: 0.5 }}
      onMouseDown={e => {
        e.preventDefault();
        toggleMark(editor, format);
      }}
      {...props}
    >
      {icon}
    </Button>
  );
};

function isMarkActive(editor, format) {
  const marks = Editor.marks(editor);
  return marks ? marks[format] === true : false;
}

function toggleMark(editor, format) {
  const isActive = isMarkActive(editor, format);
  if (isActive) {
    Editor.removeMark(editor, format);
  } else {
    Editor.addMark(editor, format, true);
  }
}

// Used for validating variable inserts (disallow consecutive variables)
function lastNodeIsVariable(editor) {
  const { selection } = editor;
  if (selection && Range.isCollapsed(selection)) {
    const [node] = Editor.node(editor, selection.focus.path);
    if (node && SlateElement.isElement(node) && node.type === "variable") {
      return true;
    }
    // Also look at previous sibling; more robust checking can be added
  }
  return false;
}

//////////////////
// Email Slate Template Component
export default function EmailTemplateSlate() {
  // Header is non-editable, here it's kept as a string (but you could use Slate if you want to allow formatting)
  const headerContent = useMemo(
    () => `<h4>NON-EDITABLE HEADER<br/>INTIMATION</h4><br/>Dear Sir/Madam,`,
    []
  );
  // Body (initial, editable)
  const initialBodyValue = useMemo(
    () => [
      { type: "paragraph", children: [{ text: "Editable body will appear here. Drag variables to insert into email." }] }
    ],
    []
  );
  const [bodyValue, setBodyValue] = useState(initialBodyValue);
  const [isEditing, setIsEditing] = useState(false);
  const [isPreviewing, setIsPreviewing] = useState(false);
  const [templateContent, setTemplateContent] = useState(null);
  const [sbar, setSbar] = useState({ isOpen: false, Msg: "" });

  const editor = useMemo(() => withVariables(withReact(createEditor())), []);
  window._editingEditor = isEditing ? editor : null; // For use in variable X-button (it's hacky, you can refactor to better state mgmt)

  // For word count
  const wordCount = useMemo(() => {
    return Node.string({ children: bodyValue }).split(/\s+/).filter(Boolean).length;
  }, [bodyValue]);

  // -- PREVIEW substitution: replace all variable elements with dummy values
  const previewDummyData = {
    DATE: "xx/xx/xxxx",
    REF_NO: "9999xxx",
    FIRM_NAME: "xxxxxxxxxx",
    FRN_NO: "999999999",
    GSTN: "99xx99xx",
    FIRM_ADDR: "xxxxxxxx,xxxxxxxx",
    ASSIGNMENT_TYPE: "xxxxxxx"
  };
  const renderPreviewNodes = useCallback(nodes => {
    return nodes.map((node, i) => {
      if (SlateElement.isElement(node) && node.type === "variable") {
        const key = node.value.replace(/[{}]/g, "").trim();
        return (
          <span key={i} style={{ background: "#ffe", color: "black", fontWeight: "bold", borderRadius: 3, padding: "2px 6px" }}>
            {previewDummyData[key] || node.value}
          </span>
        );
      } else if (SlateElement.isElement(node)) {
        return React.createElement(
          node.type === "paragraph" ? "p" : "span",
          { key: i },
          renderPreviewNodes(node.children)
        );
      } else {
        return node.text;
      }
    });
  }, []);

  // -- Rendering Slate elements
  const renderElement = useCallback(props => {
    switch (props.element.type) {
      case "variable":
        return <VariableElement {...props} />;
      default:
        return <DefaultElement {...props} />;
    }
  }, []);

  // -- Rendering Slate leafs (bold, italic, underline)
  const renderLeaf = useCallback(props => {
    let { children } = props;
    if (props.leaf.bold) children = <strong>{children}</strong>;
    if (props.leaf.italic) children = <em>{children}</em>;
    if (props.leaf.underline) children = <u>{children}</u>;
    return <span {...props.attributes}>{children}</span>;
  }, []);

  // -- Drag and drop for variables
  const onDropVariable = useCallback((e, editor) => {
    const value = e.dataTransfer.getData("variable");
    if (value) {
      e.preventDefault();
      // Prevent consecutive duplicate variables
      if (lastNodeIsVariable(editor)) {
        setSbar({ isOpen: true, Msg: "❌ Cannot insert duplicate variable here." });
        return;
      }
      insertVariable(editor, value.replaceAll("{{", "").replaceAll("}}", ""));
    }
  }, [setSbar]);

  // -- On body change, validate word count and block input if over
  const onBodyChange = useCallback(
    newValue => {
      const text = Node.string({ children: newValue });
      if (text.split(/\s+/).filter(Boolean).length > WORD_LIMIT) {
        setSbar({ isOpen: true, Msg: "Word limit exceeded. Please reduce content." });
        return;
      }
      setBodyValue(newValue);
    },
    [setSbar, setBodyValue]
  );

  // -- PREVIEW: substitute all variables and display as HTML
  const handlePreview = () => {
    setIsPreviewing(true);
    setTemplateContent(
      <div>
        <div
          className={css`background: #f5f5f5; padding: 10px 14px; border: 1px solid #eee; margin-bottom: 2rem;`}
          dangerouslySetInnerHTML={{ __html: headerContent }}
        />
        <div style={{ fontSize: 16 }}>
          {renderPreviewNodes(bodyValue)}
        </div>
      </div>
    );
  };

  // -- SAVE: Would normally send to backend
  const handleSave = () => {
    setIsEditing(false);
    setIsPreviewing(false);
    setSbar({ isOpen: true, Msg: "✅ Template saved successfully!" });
    // TODO: serialize bodyValue and headerContent and send to backend (with encryption if needed)
  };

  // -- RESET: Set back to initial body value
  const handleReset = () => {
    setBodyValue(initialBodyValue);
    setSbar({ isOpen: true, Msg: "✅ Template reset." });
  };

  // -- Copy handler for header: block copying.
  const handleHeaderCopy = e => {
    e.preventDefault();
    setSbar({ isOpen: true, Msg: "❌ Copying header is not allowed!" });
  };

  return (
    <>
      <Typography gutterBottom variant="h5" sx={{ mb: 1 }}>
        Empanelment Intimation Letter Template
      </Typography>
      <Divider sx={{ mb: 1 }} />
      <Box sx={{ p: 1, minHeight: "80%" }}>
        <Grid container spacing={3}>
          {/* VARIABLES + INSTRUCTIONS */}
          <Grid item xs={12} md={4}>
            {/* Variable chips area */}
            <Card sx={{ mb: 3 }}>
              <CardHeader title="Variables" titleTypographyProps={{ variant: "h6" }} />
              <CardContent sx={{ display: "flex", flexWrap: "wrap", gap: 1 }}>
                {draftVar.map(label => (
                  <Chip
                    key={label}
                    label={label}
                    color="primary"
                    draggable={isEditing}
                    onDragStart={e => {
                      e.dataTransfer.setData("variable", label.replaceAll("{", "").replaceAll("}", ""));
                    }}
                    sx={{
                      borderRadius: "0",
                      cursor: isEditing ? "move" : "not-allowed",
                      fontWeight: "medium",
                    }}
                  />
                ))}
              </CardContent>
            </Card>
            {/* Instructions */}
            <Card>
              <CardHeader title="Instructions" titleTypographyProps={{ variant: "h6" }} />
              <CardContent>
                <List dense>
                  {[
                    "Special characters are not allowed.",
                    "Only use variables from the list above.",
                    "Header section is non-editable.",
                    "Click 'Edit' to modify the template body.",
                    `Maximum word limit is ${WORD_LIMIT} words.`,
                  ].map(text => (
                    <ListItem key={text} disablePadding>
                      <ListItemIcon sx={{ minWidth: 32 }}>
                        <CheckCircleOutlineIcon color="primary" fontSize="small" />
                      </ListItemIcon>
                      <ListItemText primary={text} />
                    </ListItem>
                  ))}
                </List>
              </CardContent>
            </Card>
          </Grid>
          {/* MAIN EDITOR */}
          <Grid item xs={12} md={8}>
            <Card>
              <CardHeader title="Empanelment Intimation Letter Draft" titleTypographyProps={{ variant: "h6" }} />
              <CardContent>
                {/* Header (non-configurable, non-editable) */}
                <Typography variant="subtitle1" sx={{ fontWeight: "medium", color: "text.secondary" }}>Non Configurable</Typography>
                <Paper
                  variant="outlined"
                  sx={{
                    bgcolor: "#f5f5f5",
                    p: 1,
                    mb: 3,
                    border: "1px solid #e0e0e0",
                    borderRadius: 1,
                  }}
                  onCopy={handleHeaderCopy}
                >
                  {/* Render as HTML or with Slate in readOnly mode */}
                  <div dangerouslySetInnerHTML={{ __html: headerContent }} style={{ minHeight: 50, fontSize: 16 }} />
                </Paper>
                {/* Body (editable) */}
                <Typography variant="subtitle1" sx={{ fontWeight: "medium", color: "text.secondary" }}>Configurable</Typography>
                <Paper variant="outlined">
                  {/* Formatting toolbar */}
                  <Box sx={{ mb: 1, display: "flex", gap: 1 }}>
                    <ToolbarButton format="bold" icon={<b>B</b>} disabled={!isEditing} />
                    <ToolbarButton format="italic" icon={<i>I</i>} disabled={!isEditing} />
                    <ToolbarButton format="underline" icon={<u>U</u>} disabled={!isEditing} />
                  </Box>
                  {/* The Slate editor */}
                  <Slate
                    editor={editor}
                    value={bodyValue}
                    onChange={onBodyChange}
                  >
                    <Editable
                      renderElement={renderElement}
                      renderLeaf={renderLeaf}
                      readOnly={!isEditing}
                      spellCheck
                      className={css`
                        font-size: 1rem;
                        min-height: 23vh;
                        margin-bottom: 10px;
                        padding: 8px;
                      `}
                      onDrop={e => {
                        if (!isEditing) return;
                        onDropVariable(e, editor);
                      }}
                    />
                  </Slate>
                  {/* Bottom bar: word count and buttons */}
                  <Box sx={{
                    display: "flex",
                    justifyContent: "space-between",
                    alignItems: "center",
                    mt: 2,
                    mb: 1,
                  }}>
                    <Typography variant="body2" color="textSecondary">
                      Word Count: {wordCount} / {WORD_LIMIT}
                    </Typography>
                    <Box sx={{ display: "flex", gap: 1 }}>
                      {!isEditing ? (
                        <Button
                          variant="contained"
                          startIcon={<EditIcon />}
                          onClick={() => setIsEditing(true)}
                        >
                          Edit
                        </Button>
                      ) : (
                        <Button
                          variant="outlined"
                          color="error"
                          startIcon={<CancelIcon />}
                          onClick={() => setIsEditing(false)}
                        >
                          Cancel
                        </Button>
                      )}
                      <Button
                        variant="outlined"
                        color="info"
                        startIcon={<PreviewIcon />}
                        onClick={handlePreview}
                      >
                        Preview
                      </Button>
                      <Button
                        variant="outlined"
                        color="secondary"
                        startIcon={<RestartAltIcon />}
                        onClick={handleReset}
                      >
                        Reset
                      </Button>
                    </Box>
                  </Box>
                </Paper>
              </CardContent>
            </Card>
          </Grid>
        </Grid>
        {/* Preview Dialog */}
        {isPreviewing && (
          <PreviewDialog
            open={isPreviewing}
            close={() => setIsPreviewing(false)}
            save={handleSave}
            content={templateContent}
          />
        )}
        {/* Snackbar */}
        <Snackbar
          open={sbar.isOpen}
          autoHideDuration={6000}
          onClose={() => setSbar({ isOpen: false, Msg: "" })}
          message={sbar.Msg}
          anchorOrigin={{ vertical: "bottom", horizontal: "right" }}
        />
      </Box>
    </>
  );
}

/** Preview dialog using MUI **/
function PreviewDialog({ open, close, save, content }) {
  return (
    <Dialog open={open} onClose={close} maxWidth="md" fullWidth>
      <DialogTitle sx={{ m: 0, p: 2, fontWeight: "bold" }}>
        Intimation Letter Draft Preview
      </DialogTitle>
      <IconButton
        aria-label="close"
        onClick={close}
        sx={{
          position: "absolute",
          right: 8,
          top: 8,
          color: theme => theme.palette.grey[500],
        }}
      >
        <CloseIcon />
      </IconButton>
      <DialogContent>
        <Paper
          elevation={3}
          sx={{
            width: "210mm",
            minHeight: "297mm",
            margin: "2rem auto",
            padding: "20mm",
            backgroundColor: "#fff",
            overflow: "auto",
            overflowWrap: "break-word",
          }}
        >
          {content}
        </Paper>
      </DialogContent>
      <DialogActions>
        <Button onClick={close} color="secondary">Cancel</Button>
        <Button
          autoFocus
          onClick={save}
          variant="contained"
          startIcon={<SaveIcon />}
        >
          Save Draft
        </Button>
      </DialogActions>
    </Dialog>
  );
}
