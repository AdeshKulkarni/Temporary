import React, { useState, useEffect, useMemo } from "react";
import {
  Divider,
  Typography,
  Box,
  Stack,
  Button,
  Paper,
  MenuItem,
  TextField,
  Tabs,
  Tab,
} from "@mui/material";
import { DataGrid, GridToolbar } from "@mui/x-data-grid";
import { encrypt } from "../../Security/AES-GCM256";
import axios from "axios";
import downloadFile from "../../DownloadFile/DownloadFile";
import { Assignment } from "@mui/icons-material";

const iv = crypto.getRandomValues(new Uint8Array(12));
const ivBase64 = btoa(String.fromCharCode.apply(null, iv));
const salt = crypto.getRandomValues(new Uint8Array(16));
const saltBase64 = btoa(String.fromCharCode.apply(null, salt));

const columns = [
  { field: "FRN", headerName: "FRN", width: 170 },
  { field: "FIRM_NAME", headerName: "FIRM NAME", width: 200 },
  {
    field: "EMPANELMENT_SUB_TYPE",
    headerName: "EMPANELMENT SUB TYPE",
    width: 280,
  },
  { field: "EMPANELED_BY", headerName: "EMPANELED BY", width: 140 },
  { field: "EMPANELED_STATUS", headerName: "EMPANELED STATUS", width: 160 },
  { field: "EMPANELMENT_TYPE", headerName: "EMPANELMENT TYPE", width: 160 },
  { field: "FINANCIAL_YEAR", headerName: "FINANCIAL YEAR", width: 160 },
  { field: "REQUEST_STATUS", headerName: "REQUEST STATUS", width: 160 },
];

const EmpanelmentData = () => {
  const [empanelmentData, setEmpanelmentData] = useState([]);
  const [userIp, setUserIp] = useState({ datatype: "", type: "", subType: "" });
  const [selectionOptions, setSelectionOptions] = useState({
    IA: [],
    SA: [],
  });

  // Filtered rows for DataGrid
  const filteredRows = useMemo(() => {
    let data = empanelmentData;
    if (userIp.type) {
      data = data.filter((row) => row.EMPANELMENT_TYPE === userIp.type);
    }
    if (userIp.subType) {
      data = data.filter((row) => row.EMPANELMENT_SUB_TYPE === userIp.subType);
    }
    return data;
  }, [empanelmentData, userIp.type, userIp.subType]);

  useEffect(() => {
    fetchOptions();
  }, []);

  useEffect(() => {
    // Fetch data only after type is selected
    if (userIp.type !== "") {
      fetchData();
    }
    // eslint-disable-next-line
  }, [userIp.subType]);

  const fetchOptions = async () => {
    try {
      const response = await axios.post(
        "Server/IAMServices/getAssignmentType",
        {},
        {
          headers: {
            Authorization: `Bearer ${localStorage.getItem("token")}`,
          },
        }
      );
      
      // Process the API response to get IA and SA options
      const result = response.data.result[0] || {};
      const options = {
        IA: result.IA ? result.IA.split(",").map(item => item.trim()) : [],
        SA: result.SA ? result.SA.split(",").map(item => item.trim()) : [],
      };
      
      setSelectionOptions(options);
    } catch (e) {
      console.error("Error fetching options:", e);
    }
  };

  const fetchData = async () => {
    const dataToSend = {
      EMPANELMENT_TYPE: userIp.type,
      EMPANELMENT_SUB_TYPE: userIp.subType,
    };
    try {
      const jsonFormData = JSON.stringify(dataToSend);
      const encryptedData = await encrypt(iv, salt, jsonFormData);
      const payload = {
        iv: ivBase64,
        salt: saltBase64,
        data: encryptedData,
      };
      const response = await axios.post(
        "Server/IAMServices/getActiveFRN",
        payload,
        {
          headers: {
            Authorization: `Bearer ${localStorage.getItem("token")}`,
          },
        }
      );
      setEmpanelmentData(response.data.result || []);
    } catch (e) {
      console.error("Fetch Error:", e);
    }
  };

  const handleDownload = async () => {
    let dataToSend = filteredRows;
    try {
      let jsonFormData = JSON.stringify(dataToSend);
      const encryptedData = await encrypt(iv, salt, jsonFormData);
      const payload = {
        iv: ivBase64,
        salt: saltBase64,
        data: encryptedData,
      };
      const response = await axios.post(
        "Server/IAMServices/download_empanelment_data",
        payload,
        {
          headers: {
            Authorization: `Bearer ${localStorage.getItem("token")}`,
          },
        }
      );
      const resData = response.data.result;
      if (resData) {
        downloadFile(resData, "excel", `${userIp.datatype}_Empanelment_list`);
      }
    } catch (e) {
      console.error("Download Error:", e);
    }
  };

  const handleTabChange = (event, newValue) => {
    setUserIp({ datatype: newValue, type: "", subType: "" });
    setEmpanelmentData([]);
  };

  const handleSelectYear = (year) => {
    setUserIp((prev) => ({ ...prev, year, type: "", subType: "" }));
  };

  const handleSelectType = (type) => {
    setUserIp((prev) => ({ ...prev, type, subType: "" }));
  };

  const handleSelectSubType = (subType) => {
    setUserIp((prev) => ({ ...prev, subType }));
  };

  return (
    <>
      <Typography variant="h5">Empanelment Active/Archive Data</Typography>
      <Divider sx={{ mb: 3 }} />
      <Paper
        sx={{
          p: 3,
          maxWidth: "90%",
          height: "80%",
          mx: "auto",
          my: 4,
          boxShadow: 4,
        }}
      >
        <Box sx={{ mb: 3 }}>
          <Tabs
            value={userIp.datatype}
            onChange={handleTabChange}
            textColor="primary"
            indicatorColor="primary"
            sx={{
              "& .MuiTab-root": {
                fontWeight: 500,
                fontSize: 16,
                minWidth: 120,
              },
            }}
          >
            <Tab label="Active" value="Active" />
            <Tab label="Archive" value="Archive" />
          </Tabs>
        </Box>
        {userIp.datatype && (
          <Stack direction="row" spacing={2} alignItems="center" sx={{ mb: 3 }}>
            {userIp.datatype == "Archive" && (
              <TextField
                select
                label="Year"
                value={userIp.year}
                onChange={(e) => handleSelectYear(e.target.value)}
                sx={{ minWidth: 200 }}
                size="small"
                InputLabelProps={{ shrink: true }}
              >
                <MenuItem disabled value="">
                  -- Select Year --
                </MenuItem>
                <MenuItem value="2025-26">2025-26</MenuItem>
              </TextField>
            )}
            <TextField
              select
              label="Type"
              value={userIp.type}
              onChange={(e) => handleSelectType(e.target.value)}
              sx={{ minWidth: 200 }}
              size="small"
              InputLabelProps={{ shrink: true }}
            >
              <MenuItem disabled value="">
                -- Select Type --
              </MenuItem>
              {Object.keys(selectionOptions).map((type) => (
                <MenuItem key={type} value={type}>
                  {type}
                </MenuItem>
              ))}
            </TextField>

            {/* Show Sub Type dropdown if Type is selected */}
            {userIp.type && (
              <TextField
                select
                label="Empanelment Sub Type"
                value={userIp.subType}
                onChange={(e) => handleSelectSubType(e.target.value)}
                sx={{ minWidth: 220 }}
                size="small"
                InputLabelProps={{ shrink: true }}
              >
                <MenuItem value="">-- All Sub Types --</MenuItem>
                {selectionOptions[userIp.type]?.map((option) => (
                  <MenuItem key={option} value={option}>
                    {option}
                  </MenuItem>
                ))}
              </TextField>
            )}
          </Stack>
        )}

        {userIp.subType !== "" && (
          <Box sx={{ width: "100%", mt: 2, justifyItems: "right" }}>
            <Button
              onClick={handleDownload}
              variant="contained"
              sx={{
                mb: 2,
                ml: "auto",
                display: "block",
                fontWeight: 600,
                px: 3,
                borderRadius: 2,
              }}
            >
              Download in Excel
            </Button>
            <Typography
              variant="caption"
              color="textSecondary"
              sx={{ mb: 1, display: "block" }}
            >
              Note: This Excel file contains all fields.
            </Typography>
            <Paper
              elevation={2}
              sx={{
                borderRadius: 2,
                overflow: "hidden",
                minHeight: 420,
                background: "#f9f9f9",
              }}
            >
              <DataGrid
                rows={filteredRows}
                columns={columns}
                getRowId={(row) => row.FRN}
                pageSizeOptions={[5, 10, 20, 50]}
                initialState={{
                  pagination: { paginationModel: { pageSize: 10 } },
                }}
                disableRowSelectionOnClick
                slots={{
                  toolbar: () => (
                    <Box
                      sx={{
                        display: "flex",
                        justifyContent: "flex-end",
                        width: "100%",
                      }}
                    >
                      <GridToolbar />
                    </Box>
                  ),
                }}
                sx={{
                  border: "none",
                  "& .MuiDataGrid-columnHeaders": {
                    background: "#e3e7ef",
                    fontWeight: 600,
                    fontSize: 15,
                  },
                  "& .MuiDataGrid-row": {
                    background: "#fff",
                  },
                  "& .MuiDataGrid-footerContainer": {
                    background: "#fafafa",
                  },
                }}
                autoHeight
              />
            </Paper>
          </Box>
        )}
      </Paper>
    </>
  );
};

export default EmpanelmentData;
