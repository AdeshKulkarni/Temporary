package com.fincore.commonutilities.jwt;

import com.fasterxml.jackson.databind.JsonNode;
import com.fasterxml.jackson.databind.ObjectMapper;

import java.io.IOException;
import java.util.Base64;

public class JwtUtil {
    private static final ObjectMapper objectMapper = new ObjectMapper();

    /**
     * Extracts the User ID (subject) from a JWT token.
     * NOTE: This method DOES NOT verify the token's signature.
     * ...
     */
    public static String getUserIdFromToken(String token) {
        if (token == null || !token.startsWith("Bearer ")) {
            throw new IllegalArgumentException("Invalid Authorization header");
        }
        String jwt = token.substring(7);
        String[] chunks = jwt.split("\\.");
        if (chunks.length < 2) {
            throw new IllegalArgumentException("Invalid JWT token");
        }
        Base64.Decoder decoder = Base64.getUrlDecoder();
        String payload = new String(decoder.decode(chunks[1]));
        try {
            JsonNode payloadNode = objectMapper.readTree(payload);
            if (payloadNode.has("sub")) {
                return payloadNode.get("sub").asText();
            }
            throw new IllegalArgumentException("JWT token does not contain 'sub' (subject) claim");
        } catch (IOException e) {
            throw new RuntimeException("Failed to parse JWT payload", e);
        }
    }
    
    // --- THIS IS THE NEW METHOD YOU NEED TO ADD ---

    /**
     * Parses the JWT payload and returns it as a JsonNode.
     * NOTE: This method DOES NOT verify the token's signature.
     *
     * @param token The full "Bearer <token>" string.
     * @return The entire payload as a JsonNode.
     */
    public static JsonNode getClaimsFromToken(String token) {
        if (token == null || !token.startsWith("Bearer ")) {
            throw new IllegalArgumentException("Invalid Authorization header");
        }
        String jwt = token.substring(7);
        String[] chunks = jwt.split("\\.");
        if (chunks.length < 2) {
            throw new IllegalArgumentException("Invalid JWT token");
        }
        Base64.Decoder decoder = Base64.getUrlDecoder();
        String payload = new String(decoder.decode(chunks[1]));
        try {
            return objectMapper.readTree(payload);
        } catch (IOException e) {
            throw new RuntimeException("Failed to parse JWT payload", e);
        }
    }
}
