Here is the specific optimized SQL query that handles the date-specific logic.
This query groups by Date and matches on it. If a record exists for that specific date, it updates the balance. If not, it inserts a new entry for that date.
MERGE INTO GL_BALANCE B
USING (
    SELECT 
        jt.branch AS BRANCH_CODE, 
        jt.currency AS CURRENCY, 
        jt.cgl AS CGL,
        TO_DATE(jt.pDate, 'YYYY-MM-DD') AS TXN_DATE, 
        SUM(
            CASE 
                WHEN jt.currency = 'INR' THEN jt.amount
                ELSE jt.amount * NVL(cm.CURRENCY_RATE, 1) 
            END
        ) AS NET_AMOUNT
    FROM JOURNAL_REQUEST jr
    CROSS JOIN JSON_TABLE(jr.PAYLOAD, '$' 
        COLUMNS (
            branch   VARCHAR2(50) PATH '$.branch',
            currency VARCHAR2(3)  PATH '$.currency',
            cgl      VARCHAR2(50) PATH '$.cgl',
            amount   NUMBER       PATH '$.amount',
            pDate    VARCHAR2(10) PATH '$.pDate'
        )
    ) jt
    LEFT JOIN CURRENCY_MASTER cm ON jt.currency = cm.CURRENCY_CODE
    WHERE jr.REQ_STATUS = 'P' 
      AND jr.CREATOR_ID != ? 
      AND jr.BATCH_ID = ?  -- Or use the dynamic target clause here
    GROUP BY jt.branch, jt.currency, jt.cgl, jt.pDate
) S
ON (
    B.BRANCH_CODE = S.BRANCH_CODE AND 
    B.CURRENCY = S.CURRENCY AND 
    B.CGL = S.CGL AND 
    TRUNC(B.BALANCE_DATE) = S.TXN_DATE 
)
WHEN MATCHED THEN
    UPDATE SET B.BALANCE = B.BALANCE + S.NET_AMOUNT
WHEN NOT MATCHED THEN
    INSERT (ID, BALANCE_DATE, BRANCH_CODE, CURRENCY, CGL, BALANCE)
    VALUES (GL_BALANCE_SEQ.nextval, S.TXN_DATE, S.BRANCH_CODE, S.CURRENCY, S.CGL, S.NET_AMOUNT)
