import java.util.ArrayList;
import java.util.List;

public class CompositeSurge implements SurgeStrategy {
    
    // 1. It secretly holds a list of other surges
    private List<SurgeStrategy> activeSurges = new ArrayList<>();

    // 2. A method to load it up
    public void addSurge(SurgeStrategy surge) {
        this.activeSurges.add(surge);
    }

    @Override
    public double CalculateSurge(rideDetails ride) {
        double totalSurge = 1.0; // Base multiplier is 1.0 (no surge)

        // 3. Loop through all active surges and multiply them!
        for (SurgeStrategy surge : activeSurges) {
            double currentSurge = surge.CalculateSurge(ride);
            
            // Only multiply if it actually returned a surge (> 0)
            // (Based on your hardcoded logic where 0 means "doesn't apply")
            if (currentSurge > 0) {
                totalSurge = totalSurge * currentSurge;
            }
        }

        // Rule: Surge cannot exceed 3.0x
        return Math.min(totalSurge, 3.0);
    }
}


public class PricingEngine {
    // The 3 Slots!
    private BaseFareStrategy baseFareStrategy;
    private SurgeStrategy surgeStrategy;
    private DiscountStrategy discountStrategy;

    // Setters to plug in the cartridges
    public void setBaseFareStrategy(BaseFareStrategy baseFareStrategy) {
        this.baseFareStrategy = baseFareStrategy;
    }

    public void setSurgeStrategy(SurgeStrategy surgeStrategy) {
        this.surgeStrategy = surgeStrategy;
    }

    public void setDiscountStrategy(DiscountStrategy discountStrategy) {
        this.discountStrategy = discountStrategy;
    }

    // The Master Calculation!
    public double calculateFinalPrice(rideDetails ride) {
        // 1. Get Base Fare
        double base = baseFareStrategy.baseFare(ride);

        // 2. Get Surge
        double surge = surgeStrategy.CalculateSurge(ride);
        if (surge == 0) surge = 1.0; // Failsafe for your hardcoded 0s

        // 3. Get Discount (Notice we pass the base fare in!)
        double discount = discountStrategy.ApplyDiscount(ride, base);

        // 4. The Final Formula: (Base * Surge) - Discount + ServiceFee
        double finalPrice = (base * surge) - discount + 10.0; // 10 is service fee

        // Rule: Price cannot be negative
        return Math.max(finalPrice, 0.0);
    }
}
