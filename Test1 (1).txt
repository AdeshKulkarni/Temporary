import org.springframework.data.jpa.repository.JpaRepository;

import java.util.List;

public interface BranchRepository extends JpaRepository<Branch, Long> { // Assuming <Branch, Long>

    /**
     * This is the new method you need to add.
     *
     * It will find a branch if:
     * 1. The branchName contains the query (ignoring case)
     * OR
     * 2. The branchCode contains the query (ignoring case)
     *
     * We pass the same 'query' string to both parameters from the service.
     */
    List<Branch> findByBranchNameContainingIgnoreCaseOrBranchCodeContainingIgnoreCase(String branchName, String branchCode);

    /* * You can now remove your old, separate search methods if you want,
     * like "findByBranchNameContainingIgnoreCase" or "findByBranchCode"
     * to avoid confusion.
     */
}
