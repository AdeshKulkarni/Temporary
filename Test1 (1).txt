1. TemplateViewDialog.js
This is your detailed view modal. I have maintained the logic from your snippet (selecting variants, dynamic parameters) but added a safety check if (!template) return null; to prevent crashes while the modal is opening.
import React, { useState, useEffect } from 'react';
import { 
  Dialog, DialogTitle, DialogContent, DialogActions, 
  Button, MenuItem, TextField, Divider, Grid, Typography 
} from '@mui/material';

const TemplateViewDialog = ({ open, handleClose, template }) => {
  const [selectedVariant, setSelectedVariant] = useState('');
  const [formData, setFormData] = useState({});

  // Reset state when the modal opens with a new template
  useEffect(() => {
    if (open && template) {
        setSelectedVariant('');
        setFormData({});
    }
  }, [open, template]);

  if (!template) return null;

  const activeVariant = template.variants?.find(v => v.variantCode === selectedVariant);

  const handleInputChange = (name, value) => {
    setFormData(prev => ({ ...prev, [name]: value }));
  };

  const handleGenerate = () => {
    console.log("Generating Report for:", {
        template: template.templateName,
        variant: selectedVariant,
        data: formData
    });
    // Add logic to call generation API here
    handleClose();
  };

  return (
    <Dialog open={open} onClose={handleClose} maxWidth="sm" fullWidth>
      <DialogTitle>View Report: {template.templateName}</DialogTitle>
      <DialogContent dividers>
        <Grid container spacing={3}>
          {/* Variant Selection */}
          <Grid item xs={12}>
            <TextField
              select
              fullWidth
              label="Select Report Variant"
              value={selectedVariant}
              onChange={(e) => setSelectedVariant(e.target.value)}
              helperText={template.variants?.length === 0 ? "No variants available" : ""}
            >
              {template.variants?.map((v) => (
                <MenuItem key={v.variantCode} value={v.variantCode}>
                  {v.variantName}
                </MenuItem>
              ))}
            </TextField>
            {activeVariant && (
                <Typography variant="caption" color="text.secondary" sx={{ mt: 1, display: 'block' }}>
                    {activeVariant.description}
                </Typography>
            )}
          </Grid>

          {/* Dynamic Inputs based on Params */}
          {activeVariant?.params?.map((param) => (
            <Grid item xs={12} key={param.paramName}>
              <TextField
                fullWidth
                label={param.label}
                // Simple mapping for types; expand as needed
                type={param.paramType === 'DATE' ? 'date' : 'text'}
                required={param.required}
                InputLabelProps={{ shrink: true }} // Ensures label doesn't overlap for date inputs
                onChange={(e) => handleInputChange(param.paramName, e.target.value)}
              />
            </Grid>
          ))}

          {/* Export Format */}
          <Grid item xs={12}>
            <Divider sx={{ my: 1 }}>Export Options</Divider>
            <TextField select fullWidth label="Export Format" defaultValue="pdf">
              <MenuItem value="pdf">PDF Document</MenuItem>
              <MenuItem value="excel">Excel Spreadsheet</MenuItem>
              <MenuItem value="text">Plain Text</MenuItem>
            </TextField>
          </Grid>
        </Grid>
      </DialogContent>
      <DialogActions sx={{ p: 2 }}>
        <Button onClick={handleClose}>Cancel</Button>
        <Button 
            variant="contained" 
            color="primary" 
            onClick={handleGenerate}
            disabled={!selectedVariant}
        >
          Generate Report
        </Button>
      </DialogActions>
    </Dialog>
  );
};

export default TemplateViewDialog;

import React, { useState, useEffect, useCallback, useRef } from "react";
import {

// Custom Hooks
import useApi from "../../hooks/useApi";

// Import the Dialog Component

const styles = {

function CustomNoRowsOverlay() {
  return (
    <Stack height="100%" alignItems="center" justifyContent="center">
      <InfoIcon sx={{ fontSize: 48, color: "text.secondary", mb: 2 }} />
      <Typography variant="h6">No Templates Found</Typography>
    </Stack>
  );
}

export default function ReportTemplatesScreen() {
  const { callApi } = useApi();
  const showSnackBar = useCustomSnackbar();
  const isMounted = useRef(true);
  
  const [loading, setLoading] = useState(true);
  const [rows, setRows] = useState([]);
  
  // Dialog State
  const [isViewDialogOpen, setIsViewDialogOpen] = useState(false);
  const [selectedTemplate, setSelectedTemplate] = useState(null);

  useEffect(() => {
    isMounted.current = true;
    return () => { isMounted.current = false; };
  }, []);

  // 1. Fetch Data
  const fetchTemplates = useCallback(async () => {
    setLoading(true);
    try {
      const response = await callApi(
        "/template-config/get-templates",
        null,
        "GET"
      );

      if (isMounted.current) {
        // Handling { message: "...", data: [...] } structure
        const dataList = response && Array.isArray(response.data) 
          ? response.data 
          : [];
        setRows(dataList);
      }
    } catch (err) {
      if (isMounted.current && err.name !== "CanceledError") {
        showSnackBar("Failed to load templates", "error");
      }
    } finally {
      if (isMounted.current) setLoading(false);
    }
  }, [callApi, showSnackBar]);

  useEffect(() => {
    fetchTemplates();
  }, [fetchTemplates]);

  // 2. Handle View (Open Dialog)
  const handleView = (row) => {
    setSelectedTemplate(row);
    setIsViewDialogOpen(true);
  };

  const handleCloseDialog = () => {
    setIsViewDialogOpen(false);
    setSelectedTemplate(null);
  };

  // 3. Handle Status Change (Dummy API Call)
  const handleStatusChange = async (row) => {
    // Determine new status for toggling
    const newStatus = row.status === 'ACTIVE' ? 'INACTIVE' : 'ACTIVE';
    
    // Optional: Optimistic UI update (update row immediately before API)
    // const originalRows = [...rows];
    // setRows(rows.map(r => r.templateId === row.templateId ? {...r, status: newStatus} : r));

    try {
      // DUMMY API Call
      /* Uncomment and adjust URL when backend is ready:
         await callApi('/template-config/update-status', { id: row.templateId, status: newStatus }, 'POST');
      */
      
      // Simulating API delay
      await new Promise(resolve => setTimeout(resolve, 500)); 

      showSnackBar(`Status updated to ${newStatus}`, "success");
      
      // Refresh list to get actual server state
      fetchTemplates();

    } catch (error) {
       showSnackBar("Failed to update status", "error");
       // Revert optimistic update if used
       // setRows(originalRows);
    }
  };

  const columns = [
    {
      field: "templateName",
      headerName: "Template Name",
      flex: 1,
      minWidth: 180,
    },
    {
      field: "description",
      headerName: "Description",
      flex: 1.5,
      minWidth: 220,
    },
    {
      field: "status",
      headerName: "Status",
      width: 180, 
      align: "center",
      headerAlign: "center",
      renderCell: (params) => (
        <Stack direction="row" alignItems="center" spacing={1}>
            <Chip 
                label={params.value} 
                color={params.value === "ACTIVE" ? "success" : "default"} 
                size="small" 
                variant="outlined"
                sx={{ minWidth: 80 }}
            />
            <Tooltip title="Change Status">
                <IconButton 
                    size="small" 
                    onClick={(e) => {
                        e.stopPropagation(); // Prevent row click
                        handleStatusChange(params.row);
                    }}
                >
                    <EditIcon fontSize="small" />
                </IconButton>
            </Tooltip>
        </Stack>
      ),
    },
    {
      field: "versionNo",
      headerName: "Ver.",
      width: 80,
      align: "center",
      headerAlign: "center",
    },
    {
      field: "actions",
      headerName: "Action",
      width: 100,
      sortable: false,
      align: "center",
      headerAlign: "center",
      renderCell: (params) => (
        <Tooltip title="View Report">
          <IconButton 
            color="primary" 
            onClick={() => handleView(params.row)}
          >
            <VisibilityIcon />
          </IconButton>
        </Tooltip>
      ),
    },
  ];

  return (
    <Box sx={styles.mainContainer}>
      <Typography variant="h5" sx={{ mb: 2, fontWeight: 'bold' }}>
        Report Templates
      </Typography>

      <DataGrid
        rows={rows}
        columns={columns}
        loading={loading}
        getRowId={(row) => row.templateId}
        disableRowSelectionOnClick
        pageSizeOptions={[5, 10, 25]}
        rowHeight={60} // Slightly taller for the status stack
        initialState={{
          pagination: { paginationModel: { page: 0, pageSize: 10 } },
        }}
        slots={{ noRowsOverlay: CustomNoRowsOverlay }}
        sx={styles.dataGridContainer}
      />

      {/* View Dialog Component */}
      <TemplateViewDialog 
        open={isViewDialogOpen} 
        handleClose={handleCloseDialog} 
        template={selectedTemplate} 
      />
    </Box>
  );
}
