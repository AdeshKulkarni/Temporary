class Solution {

    static boolean valid(int i, int j, int n , int m) {
        if(i<0 || i>=n || j<0 || j>=m) {
            return false;
        }

        return true;
    }

    public int bfs(int[][] grid , int starti , int startj , int n , int m , boolean[][] visited , Queue<int[]> queue) {

        int[] x = { -1, 1, 0 , 0};
        int[] y = { 0, 0 , -1 , 1};      


        visited[starti][startj] = true;
        int maxDis = 0;
        queue.offer(new int[] {starti , startj , 0});

        while(!queue.isEmpty()) {

            int[] node = queue.poll();
            int row = node[0];
            int col = node[1];
            int dis = node[2];


            // checking what to do with dis
            if(dis > maxDis) {
                maxDis = dis;
            }



            for(int k=0; k<4; k++) {

                int newrow =  row  + x[k];
                int newcol =  col  + y[k];

                if(valid(newrow , newcol , n , m) && visited[newrow][newcol]== false && grid[newrow][newcol] == 0) {
                    visited[newrow][newcol] = true;
                    queue.offer(new int[] {newrow , newcol , dis + 1});
                }
            }
        }
        return maxDis;

    }
    public int maxDistance(int[][] grid) {

    int n = grid.length;
    int m = grid[0].length;

    boolean[][] visited = new boolean[n][m];
    Queue<int[]> queue = new LinkedList<>();

    int maxi = 0;

    for (int i = 0; i < n; i++) {
        for (int j = 0; j < m; j++) {

            if (grid[i][j] == 1) {
                visited[i][j] = true;
                queue.offer(new int[] {i, j, 0});
            }
        }
    }

    return bfs(grid, n, m, visited, queue);
}
