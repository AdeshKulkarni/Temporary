B. In BranchService.java, add JdbcTemplate and replace searchBranches:
    @org.springframework.beans.factory.annotation.Autowired
    @org.springframework.beans.factory.annotation.Qualifier("oracleJdbcTemplate")
    private org.springframework.jdbc.core.JdbcTemplate jdbcTemplate;

    public List<BranchDto> searchBranches(String query, java.time.LocalDate backDate) {
        log.info("BRANCH_SEARCH_INIT | QueryLength: {} | BackDate: {}", (query != null ? query.length() : 0), backDate);

        if (query == null || query.trim().isEmpty()) {
            return Collections.emptyList();
        }

        try {
            String sql = "SELECT CODE, NAME FROM BRANCH_MASTER WHERE (LOWER(CODE) LIKE LOWER(?) OR LOWER(NAME) LIKE LOWER(?))";
            List<Object> params = new java.util.ArrayList<>();
            params.add("%" + query + "%");
            params.add("%" + query + "%");

            // IF BACKDATED: Must be open before the date, and NOT closed before the date
            if (backDate != null) {
                sql += " AND TRUNC(OPEN_DATE) <= ? AND (CLOSE_DATE IS NULL OR TRUNC(CLOSE_DATE) >= ?)";
                params.add(java.sql.Date.valueOf(backDate));
                params.add(java.sql.Date.valueOf(backDate));
            } else {
                sql += " AND STATUS = 1"; // Normal live fetch
            }
            sql += " FETCH FIRST 50 ROWS ONLY";

            return jdbcTemplate.query(sql, (rs, rowNum) -> new BranchDto(rs.getString("CODE"), rs.getString("NAME")), params.toArray());

        } catch (DataAccessException e) {
            log.error("BRANCH_SEARCH_FAIL | Error: {}", e.getMessage());
            return Collections.emptyList();
        }
    }

    public Set<String> getAllActiveCurrencies(java.time.LocalDate backDate) {
        long start = System.currentTimeMillis();
        String sql;
        List<Object> params = new java.util.ArrayList<>();

        // Ensure we only show currencies created AT OR BEFORE the backdate
        if (backDate != null) {
            sql = "SELECT CURRENCY_CODE FROM CURRENCY_MASTER WHERE TRUNC(CREATED_AT) <= ?";
            params.add(java.sql.Date.valueOf(backDate));
        } else {
            sql = "SELECT CURRENCY_CODE FROM CURRENCY_MASTER WHERE FLAG = 1";
        }

        try {
            List<String> list = jdbcTemplate.query(con -> {
                var ps = con.prepareStatement(sql);
                ps.setFetchSize(1000);
                for (int i = 0; i < params.size(); i++) {
                    ps.setObject(i + 1, params.get(i));
                }
                return ps;
            }, (rs, rowNum) -> rs.getString(1));

            log.info("MASTER_DATA_LOADED | Type: CURRENCIES | Count: {} | Duration: {}ms", list.size(), System.currentTimeMillis() - start);
            return new HashSet<>(list);

        } catch (DataAccessException e) {
            log.error("MASTER_DATA_FETCH_FAIL | Type: CURRENCIES | Reason: {}", e.getMessage());
            return new HashSet<>();
        }
    }
