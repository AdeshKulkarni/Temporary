import React, { useState, useRef } from "react";
import {
  Box,
  Typography,
  Paper,
  TextField,
  Button,
  CircularProgress,
  Chip,
  Alert,
  Stack,
} from "@mui/material";
import {
  CloudUpload as CloudUploadIcon,
  RestartAlt as RestartAltIcon,
  Description as DescriptionIcon,
  CheckCircle as CheckCircleIcon,
  Error as ErrorIcon,
  Download as DownloadIcon,
  Send as SendIcon,
} from "@mui/icons-material";

import { DataGrid } from "@mui/x-data-grid";

import {
  mainContainerStyle,
  uploadBoxStyle,
  controlPanelStyle,
  iconStyle,
  headerTextStyle,
  subTextStyle,
  dateinputfield, // Imported your specific style
} from "./JournalBulkStyle";

import useApi from "../../hooks/useApi";
import useCustomSnackbar from "../../utils/useCustomSnackbar";

const JournalBulkUpload = () => {
  const { callApi, loading } = useApi();
  const showSnackBar = useCustomSnackbar();

  const [selectedDate, setSelectedDate] = useState("");
  const [file, setFile] = useState(null);
  const [isCsvInput, setIsCsvInput] = useState(false);

  const [uiError, setUiError] = useState("");
  const [errorCount, setErrorCount] = useState(0);

  const [successFile, setSuccessFile] = useState(null);
  const [previewRows, setPreviewRows] = useState([]);
  const [totalRowsProcessed, setTotalRowsProcessed] = useState(0);

  const [isSubmitting, setIsSubmitting] = useState(false);

  const fileInputRef = useRef(null);

  const downloadBase64File = (base64String, fileName, mimeType) => {
    try {
      const byteCharacters = atob(base64String);
      const byteNumbers = new Array(byteCharacters.length);
      for (let i = 0; i < byteCharacters.length; i++) {
        byteNumbers[i] = byteCharacters.charCodeAt(i);
      }
      const byteArray = new Uint8Array(byteNumbers);
      const blob = new Blob([byteArray], { type: mimeType });
      const url = window.URL.createObjectURL(blob);
      const link = document.createElement("a");
      link.href = url;
      link.setAttribute("download", fileName);
      document.body.appendChild(link);
      link.click();
      link.parentNode.removeChild(link);
      window.URL.revokeObjectURL(url);
    } catch (e) {
      console.error("Failed to download", e);
      showSnackBar("Failed to download file.", "error");
    }
  };

  const handleDateChange = (e) => {
    setSelectedDate(e.target.value);
    setUiError("");
    setSuccessFile(null);
    setPreviewRows([]);
    setTotalRowsProcessed(0);
  };

  const handleFileChange = (e) => {
    const selectedFile = e.target.files[0];
    if (selectedFile) {
      const name = selectedFile.name.toLowerCase();
      if (
        !name.endsWith(".xlsx") &&
        !name.endsWith(".xls") &&
        !name.endsWith(".csv")
      ) {
        const msg = "Invalid file type. Supported: .xlsx, .xls, .csv";
        setUiError(msg);
        showSnackBar(msg, "error");
        setFile(null);
        return;
      }
      setFile(selectedFile);
      setIsCsvInput(name.endsWith(".csv"));

      setUiError("");
      setErrorCount(0);
      setSuccessFile(null);
      setPreviewRows([]);
      setTotalRowsProcessed(0);
    }
  };

  const resetForm = () => {
    setFile(null);
    setSelectedDate("");
    setUiError("");
    setErrorCount(0);
    setSuccessFile(null);
    setPreviewRows([]);
    setTotalRowsProcessed(0);
    setIsSubmitting(false);
    setIsCsvInput(false);
    if (fileInputRef.current) fileInputRef.current.value = "";
  };

  const handleValidation = async () => {
    if (!selectedDate || !file) {
      const msg = "Please select a Date and upload a File.";
      setUiError(msg);
      showSnackBar(msg, "warning");
      return;
    }

    setUiError("");
    setErrorCount(0);
    setSuccessFile(null);
    setPreviewRows([]);
    setTotalRowsProcessed(0);

    const formData = new FormData();
    formData.append("file", file);
    formData.append("postingDate", selectedDate);

    try {
      const response = await callApi(
        "/JS/journals/bulk-validate",
        formData,
        "POST",
        "json",
        "multipart/form-data"
      );

      if (response && response.status === "ERROR") {
        const count = response.errorCount || 0;
        setErrorCount(count);
        const msg = `Validation Failed with ${count} errors. Downloading report...`;
        setUiError(msg);
        showSnackBar(msg, "error");

        if (response.errorReportBase64) {
          downloadBase64File(
            response.errorReportBase64,
            `Error_Report.xlsx`,
            "application/vnd.openxmlformats-officedocument.spreadsheetml.sheet"
          );
        }
      } else if (response && response.status === "SUCCESS") {
        showSnackBar(response.message, "success");
        if (response.successFileBase64)
          setSuccessFile(response.successFileBase64);
        if (response.previewData) setPreviewRows(response.previewData);
        
        // Logic for total count
        if (response.totalRows) setTotalRowsProcessed(response.totalRows);
        else if (response.previewData) setTotalRowsProcessed(response.previewData.length);
      }
    } catch (err) {
      console.error("Validation Error:", err);
    }
  };

  const handleSubmitBatch = async () => {
    if (previewRows.length === 0) {
      showSnackBar("No validated data to submit.", "warning");
      return;
    }

    setIsSubmitting(true);

    try {
      const payloadRows = previewRows.map((row) => {
        // Handle Amount: Remove commas from string, then parse
        let amt = parseFloat(row.amount.replace(/,/g, ""));
        if (row.txnType === "Credit") {
          amt = -Math.abs(amt);
        } else {
          amt = Math.abs(amt);
        }

        return {
          changeType: "ADD",
          branch: row.branch,
          currency: row.currency,
          cgl: row.cgl,
          amount: amt,
          productType: row.productCode || "",
          remarks: row.remarks || "",
          arFlag: "A",
          acClassification: "A",
        };
      });

      const finalPayload = {
        commonBatchRemarks: `Bulk Upload - ${selectedDate}`,
        rows: payloadRows,
      };

      const response = await callApi(
        "/JS/journals/create-bulk-batch",
        finalPayload,
        "POST"
      );

      if (response && response.status === "SUCCESS") {
        showSnackBar(
          `Batch Created Successfully! Batch ID: ${response.batchId}`,
          "success"
        );
        resetForm();
      }
    } catch (error) {
      console.error("Batch Creation Failed:", error);
      showSnackBar("Failed to create batch.", "error");
    } finally {
      setIsSubmitting(false);
    }
  };

  const columns = [
    {
      field: "branch",
      headerName: "Branch",
      flex: 1,
      align: "center",
      headerAlign: "center",
    },
    {
      field: "currency",
      headerName: "Currency",
      width: 100,
      align: "center",
      headerAlign: "center",
    },
    {
      field: "cgl",
      headerName: "CGL",
      flex: 1.5,
      align: "center",
      headerAlign: "center",
    },
    {
      field: "productCode",
      headerName: "Product",
      width: 120,
      align: "center",
      headerAlign: "center",
      renderCell: (p) => p.value || "-",
    },
    {
      field: "amount",
      headerName: "Amount",
      flex: 1.5,
      minWidth: 150,
      align: "right",
      headerAlign: "center",
      valueFormatter: (params) => {
        const val =
          params && typeof params === "object" && "value" in params
            ? params.value
            : params;
        if (!val) return "0.00";
        const parts = String(val).split(".");
        parts[0] = parts[0].replace(/\B(?=(\d{3})+(?!\d))/g, ",");
        return parts.join(".");
      },
    },
    {
      field: "txnType",
      headerName: "Type",
      width: 100,
      align: "center",
      headerAlign: "center",
      renderCell: (params) => (
        <Chip
          label={params.value}
          color={params.value === "Credit" ? "error" : "success"}
          size="small"
          variant="filled"
          sx={{ fontWeight: "bold", width: "80px" }}
        />
      ),
    },
    {
      field: "remarks",
      headerName: "Remarks",
      flex: 2,
      align: "center",
      headerAlign: "center",
    },
  ];

  return (
    <Box sx={mainContainerStyle}>
      <Paper elevation={1} sx={controlPanelStyle}>
        
        {/* Restored the styling box you liked */}
        <Box sx={dateinputfield}>
            <TextField
                type="date"
                label="Posting Date" // Added Label
                sx={{ alignItems: "center" }}
                size="small"
                value={selectedDate}
                onChange={handleDateChange}
                InputLabelProps={{ shrink: true }} // Added to fix styling
            />
        </Box>

        <br />

        <Box sx={uploadBoxStyle} onClick={() => fileInputRef.current.click()}>
          <input
            type="file"
            hidden
            accept=".xlsx, .xls, .csv"
            onChange={handleFileChange}
            ref={fileInputRef}
          />

          {file ? (
            <>
              <DescriptionIcon color="success" sx={{ fontSize: 60, mb: 1 }} />
              <Typography variant="h6" sx={headerTextStyle}>
                {file.name}
              </Typography>
              <Chip
                label="Ready"
                color="success"
                size="small"
                variant="outlined"
              />
              <Typography
                variant="caption"
                sx={{ mt: 1, color: "text.disabled" }}
              >
                Click to change
              </Typography>
            </>
          ) : (
            <>
              <CloudUploadIcon sx={iconStyle} />
              <Typography variant="h6" sx={headerTextStyle}>
                Click to Upload
              </Typography>
              <Typography sx={subTextStyle}>
                Supported: .xlsx, .xls, .csv
              </Typography>
            </>
          )}
        </Box>

        <br />
        <Stack
          direction="row"
          sx={{ display: "flex", justifyContent: "center" }}
          spacing={2}
        >
          <Button
            variant="contained"
            sx={{ cursor: "pointer" }}
            onClick={handleValidation}
            disabled={loading || !file || !selectedDate || isSubmitting}
            startIcon={
              loading ? (
                <CircularProgress size={20} color="inherit" />
              ) : (
                <CheckCircleIcon />
              )
            }
            size="large"
          >
            {loading ? "Validating..." : "Validate & Process"}
          </Button>

          <Button
            variant="outlined"
            color="error"
            onClick={resetForm}
            startIcon={<RestartAltIcon />}
            disabled={isSubmitting}
            size="large"
          >
            Reset
          </Button>
        </Stack>

        <Box sx={{ mt: 3 }}>
          {/* ERROR ALERT */}
          {uiError && (
            <Alert
              severity="error"
              sx={{ mb: 2, border: "1px solid #ee0404ff" }}
            >
              <Typography variant="subtitle2" fontWeight="bold">
                Validation Failed
              </Typography>
              <Typography variant="body2">{uiError}</Typography>
            </Alert>
          )}

          {/* SUCCESS PANEL */}
          {!uiError && previewRows.length > 0 && (
            <Paper
              variant="outlined"
              sx={{ p: 2, borderColor: "success.main" }}
            >
              <Stack
                direction={{ xs: "column", sm: "row" }}
                justifyContent="space-between"
                alignItems="center"
                spacing={2}
              >
                <Box>
                  <Stack direction="row" alignItems="center" spacing={1}>
                    <CheckCircleIcon color="success" />
                    <Typography variant="subtitle1" fontWeight="bold">
                      Validation Successful!
                    </Typography>
                  </Stack>
                  <Typography variant="body2">
                    {/* Show Total Count Logic */}
                    {totalRowsProcessed} rows valid. 
                    {totalRowsProcessed > previewRows.length && ` (Previewing first ${previewRows.length})`}
                  </Typography>
                </Box>
                <Stack direction="row" spacing={2}>
                  {/* HIDE DOWNLOAD IF CSV */}
                  {!isCsvInput && successFile && (
                    <Button
                      variant="outlined"
                      color="success"
                      startIcon={<DownloadIcon />}
                      disabled={isSubmitting}
                      onClick={() =>
                        downloadBase64File(
                          successFile,
                          `Processed_Journal_${selectedDate}.csv`,
                          "text/csv"
                        )
                      }
                      sx={{ bgcolor: "background.paper" }}
                    >
                      Download CSV
                    </Button>
                  )}

                  <Button
                    variant="contained"
                    color="success"
                    startIcon={
                      isSubmitting ? (
                        <CircularProgress size={20} color="inherit" />
                      ) : (
                        <SendIcon />
                      )
                    }
                    onClick={handleSubmitBatch}
                    disabled={isSubmitting}
                  >
                    {isSubmitting ? "Creating Batch..." : "Create Batch"}
                  </Button>
                </Stack>
              </Stack>
            </Paper>
          )}
        </Box>
      </Paper>

      {/* DATA GRID PREVIEW */}
      {previewRows.length > 0 && !uiError && (
        <Paper
          elevation={1}
          sx={{
            p: 0,
            mt: 3,
            height: 600,
            width: "100%",
            borderRadius: 2,
            overflow: "hidden",
          }}
        >
          <Box sx={{ p: 2, borderBottom: 1, borderColor: "divider" }}>
            <Typography variant="h6" sx={{ fontWeight: 600 }}>
              Data Preview 
              {totalRowsProcessed > previewRows.length ? ` (First ${previewRows.length} rows)` : ""}
            </Typography>
          </Box>
          <DataGrid
            rows={previewRows}
            columns={columns}
            pageSize={10}
            rowsPerPageOptions={[10, 25, 50, 100]}
            disableSelectionOnClick
            density="comfortable"
            sx={{ border: "none" }}
          />
        </Paper>
      )}
    </Box>
  );
};

export default JournalBulkUpload;
