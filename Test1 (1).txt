class Solution {

    public int Helper(int[] nums, int index, int end) {
        if (index > end) {
            return 0;
        }

        int take = nums[index] + Helper(nums, index + 2, end);
        int skip = Helper(nums, index + 1, end);

        return Math.max(take, skip);
    }

    public int rob(int[] nums) {
        if (nums.length == 1) {
            return nums[0];
        }

        int case1 = Helper(nums, 0, nums.length - 2); // Can rob first house, exclude last
        int case2 = Helper(nums, 1, nums.length - 1); // Exclude first house, can rob last

        return Math.max(case1, case2);
    }
}
