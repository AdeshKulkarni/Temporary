import { useCallback, useState } from "react";
import { useDropzone } from "react-dropzone";
import {
  Typography,
  Paper,
  Box,
  ListItem,
  ListItemIcon,
  ListItemText,
  Grid,
  Divider,
  TextField,
  Button,
} from "@mui/material";
import DescriptionIcon from "@mui/icons-material/Description";
import * as XLSX from "xlsx";
import SearchIcon from "@mui/icons-material/Search";
import FileDownloadOutlinedIcon from "@mui/icons-material/FileDownloadOutlined";
import { GetForFRNCompetency } from "./PreEmpanelmentService";
import useCustomSnackbar from "../../../Hooks/useCustomSnackbar";

const FileUploadDropBox = ({ onSendData }) => {
  const [selectedFile, setSelectedFile] = useState(null);
  const [frnValue, setFrnValue] = useState([]);
  const showSnackbar = useCustomSnackbar();

  const columnKeys = {
    FRN: "FRN",
    FIRM_NAME: "FRN Name",
    ELIGIBLE: "Eligible",
    ALREADY_EMPANELED: "Already Empaneled",
    EMPANELMENT_FOR: "Empaneled For",
    EMPANELED_BY: "Empaneled by",
  };

  const columnHeaders = Object.entries(columnKeys).map(([key, value]) => ({
    field: key,
    headerName: value,
    flex: 1,
    headerAlign: "center",
    align:
      key === "EMPANELED_BY" ||
      key === "ALREADY_EMPANELED" ||
      key === "ELIGIBLE"
        ? "center"
        : "left",
  }));

  const onDrop = useCallback((acceptedFiles) => {
    const excelFile = acceptedFiles.find((file) => file.name.endsWith(".xlsx"));
    setSelectedFile(excelFile || null);
    handleFileUpload(excelFile || null);
  }, [handleFileUpload]); // Added handleFileUpload to dependency array

  const { getRootProps, getInputProps, isDragActive } = useDropzone({
    onDrop,
    accept: {
      "application/vnd.openxmlformats-officedocument.spreadsheetml.sheet": [
        ".xlsx",
      ],
    },
    multiple: false,
  });

  // START: MODIFIED handleFileUpload FUNCTION
  const handleFileUpload = useCallback(async (file) => {
    if (!file) {
      onSendData({ columns: [], rows: [] }); // Clear DataGrid if no file
      return;
    }

    const reader = new FileReader();

    reader.onload = async (evt) => {
      try {
        const arrayBuffer = evt.target.result;
        const data = new Uint8Array(arrayBuffer);
        const workbook = XLSX.read(data, { type: "array" });

        if (!workbook.SheetNames?.length) {
          showSnackbar("No sheets found in the Excel file.", "error");
          onSendData({ columns: [], rows: [] }); // Send empty data
          return;
        }

        const worksheet = workbook.Sheets[workbook.SheetNames[0]];
        // Use { header: 1 } to get an array of arrays, useful for robust empty sheet check
        const parsedData = XLSX.utils.sheet_to_json(worksheet, { header: 1 });

        // NEW/MODIFIED VALIDATION FOR EMPTY OR HEADER-ONLY SHEETS
        if (!Array.isArray(parsedData) || parsedData.length === 0 || (parsedData.length === 1 && parsedData[0].every(cell => !cell))) {
          showSnackbar("The uploaded Excel file is empty or contains no data.", "warning");
          onSendData({ columns: [], rows: [] }); // Send empty data to clear DataGrid and trigger parent's empty file alert
          return; // Stop further processing
        }

        // Assuming the first row is headers if the sheet is not empty
        const headers = parsedData[0];
        const rows = parsedData.slice(1);

        // Check if headers are empty or invalid
        if (!headers || headers.length === 0 || headers.every(h => !h || String(h).trim() === '')) { // Added String(h).trim() for robustness
            showSnackbar("Excel file has no valid headers.", "error");
            onSendData({ columns: [], rows: [] }); // Send empty data
            return;
        }

        const validationErrors = [];
        const validFRNs = [];

        rows.forEach((row, index) => {
          // Map array-based row to object-based row using headers for validation
          const rowObject = {};
          headers.forEach((header, i) => {
            rowObject[header] = row[i];
          });

          const error = validateRow(rowObject, index); // Pass rowObject to validator
          if (error) {
            validationErrors.push(error);
          } else if (rowObject.FRN) { // Use rowObject.FRN
            validFRNs.push(String(rowObject.FRN).trim()); // trim to normalize, ensure string
          }
        });

        if (validationErrors.length > 0) {
          console.error("Validation Errors:", validationErrors);
          showSnackbar("Some FRNs in the Excel file are invalid. Please check console for details.", "error");
          onSendData({ columns: [], rows: [] }); // Clear DataGrid if validation errors
          return;
        }

        let rowData = [];
        const uniqueFRNs = [...new Set(validFRNs)];

        if (uniqueFRNs.length > 0) {
          const competencyData = await GetForFRNCompetency(uniqueFRNs);
          const existingFRNs = new Set();

          rowData = competencyData.map((row, index) => {
            const frn = row.FRN?.trim();
            existingFRNs.add(frn);
            return { id: index, ...row };
          });

          uniqueFRNs.forEach((frn) => {
            if (!existingFRNs.has(frn)) {
              rowData.push({
                id: rowData.length,
                FRN: frn,
                FIRM_NAME: "NA",
                ELIGIBLE: "YES",
                ALREADY_EMPANELED: "NO",
                EMPANELMENT_FOR: "NA",
                EMPANELLED_BY: "NA", // Corrected typo here, assuming it was intentional
              });
            }
          });
        }

        onSendData({
          columns: columnHeaders,
          rows: rowData,
        });
      } catch (error) {
        console.error("Error processing file:", error);
        showSnackbar(`Error processing Excel file: ${error.message}`, "error");
        onSendData({ columns: [], rows: [] }); // Send empty data in case of any processing error
      }
    };

    reader.readAsArrayBuffer(file);
  }, [onSendData, showSnackbar]); // Added dependencies for useCallback
  // END: MODIFIED handleFileUpload FUNCTION

  /**
   * Function to validate rows in excel
   */
  const validateRow = (row, index) => {
    const rowErrors = {};

    if (!row.FRN || String(row.FRN).trim() === "") {
      rowErrors.FRN = "Not a valid FRN"; // Changed from 'Name' to 'FRN'
    }

    return Object.keys(rowErrors).length > 0
      ? { rowIndex: index + 2, errors: rowErrors }
      : null; // +2 for Excel row (1-based + header)
  };

  const search = async () => {
    // START: MODIFIED search FUNCTION
    if (!validateAndSetString(frnValue)) {
      onSendData({ columns: [], rows: [] }); // Clear if validation fails
      return;
    }

    try {
      const payload = frnValue.split(",").map((s) => s.trim()).filter(s => s !== ''); // Filter out empty strings

      if (payload.length === 0) { // Check after trimming and filtering
        showSnackbar("No FRN Values Entered", "warning");
        onSendData({ columns: [], rows: [] }); // Clear DataGrid
        return;
      }

      const uniqueFRNs = [...new Set(payload)];

      const competencyData = await GetForFRNCompetency(uniqueFRNs);
      let rowData = competencyData.map((row, index) => ({
        id: index,
        ...row,
      }));

      const existingFRNs = new Set();

      rowData = competencyData.map((row, index) => {
        const frn = row.FRN?.trim();
        existingFRNs.add(frn);
        return { id: index, ...row };
      });

      uniqueFRNs.forEach((frn) => {
        if (!existingFRNs.has(frn)) {
          rowData.push({
            id: rowData.length,
            FRN: frn,
            FIRM_NAME: "NA",
            ELIGIBLE: "YES",
            ALREADY_EMPANELED: "NO",
            EMPANELMENT_FOR: "NA",
            EMPANELLED_BY: "NA",
          });
        }
      });

      onSendData({
        columns: columnHeaders,
        rows: rowData,
      });
    } catch (e) {
      console.error("Error while searching.", e);
      showSnackbar(`Error during search: ${e.message}`, "error");
      onSendData({ columns: [], rows: [] }); // Clear DataGrid on search error
    }
  };
  // END: MODIFIED search FUNCTION

  // START: MODIFIED validateAndSetString FUNCTION
  function validateAndSetString(inputString) {
    if (!inputString || typeof inputString !== "string" || inputString.trim() === '') {
      showSnackbar("No FRN Value Entered", "warning");
      return false;
    }

    const values = inputString.split(",").map((s) => s.trim()).filter(s => s !== '');

    if (values.length === 0) {
        showSnackbar("No FRN Value Entered", "warning");
        return false;
    }

    for (const value of values) {
      if (!/^[a-zA-Z0-9]+$/.test(value)) {
        showSnackbar(`Invalid FRN found: "${value}"`, "error");
        return false;
      }
    }
    return true;
  }
  // END: MODIFIED validateAndSetString FUNCTION

  const handleDownload = () => {
    const link = document.createElement("a");
    link.href = `${process.env.PUBLIC_URL}/Assets/Docs/FRN_TEMPL.xlsx`;
    link.download = "FRN_TEMPL.xlsx";
    document.body.appendChild(link);
    link.click();
    document.body.removeChild(link);
  };

  return (
    <Box sx={{ width: "100%", p: 2 }}>
      <Grid
        container
        spacing={2}
        sx={{ display: "flex", justifyContent: "center", alignItems: "top" }}
      >
        <Grid item xs={12} md={6}>
          <Paper
            elevation={3}
            sx={{
              display: "flex",
              justifyContent: "center",
              alignItems: "center",
              flexDirection: "column",
            }}
          >
            <Box
              {...getRootProps()}
              sx={{
                p: 4,
                textAlign: "center",
                width: "100%",
                border: "1px dashed #1976d2",
                backgroundColor: isDragActive ? "#f0f8ff" : "#fafafa",
                cursor: "pointer",
                transition: "background-color 0.3s ease",
                display: "flex",
                flexDirection: "column",
                alignItems: "center",
                justifyContent: "center",
                minHeight: 150,
              }}
            >
              <input {...getInputProps()} />
              {!selectedFile ? (
                <Box>
                  <Typography variant="h6">
                    {isDragActive
                      ? "Drop the .xlsx file here..."
                      : "Drag & drop or upload a file"}
                  </Typography>
                  <Typography variant="body2" color="textSecondary">
                    {`Excel (max. 5MB) use the templete provided on right side ->`}
                  </Typography>
                </Box>
              ) : (
                <ListItem disableGutters>
                  <Typography variant="subtitle1">Selected File:</Typography>
                  &nbsp;&nbsp;&nbsp;
                  <ListItemIcon>
                    <DescriptionIcon sx={{ color: "green", fontSize: 40 }} />
                  </ListItemIcon>
                  <ListItemText
                    primary={selectedFile.name}
                    secondary={`${(selectedFile.size / 1024).toFixed(2)} KB`}
                  />
                </ListItem>
              )}
            </Box>
            <Divider />
            <Typography variant="body">OR</Typography>
            <Box
              sx={{
                P: 2,
                display: "flex",
                justifyContent: "center",
                alignItems: "center",
                width: "100%",
                //flexDirection: "column",
                height: 50,
              }}
            >
              <TextField
                label="FRN"
                variant="standard"
                sx={{ m: 2, width: "50%" }}
                placeholder="AS978997432LKP, AS978911232LKP"
                inputProps={{ "aria-label": "search FRN Numbers" }}
                value={frnValue}
                onChange={(e) => setFrnValue(e.target.value)}
              />
              <Button
                startIcon={<SearchIcon />}
                onClick={search}
                variant="contained"
                size="small"
              >
                Search
              </Button>
            </Box>
          </Paper>
        </Grid>
        <Grid item xs={12} md={4}>
          <Box
            sx={{
              display: "flex",
              justifyContent: "center",
              alignItems: "top",
            }}
          >
            <FileDownloadOutlinedIcon sx={{ fontSize: 30 }} />
            &nbsp;
            <Typography
              variant="h6"
              color={"blue"}
              sx={{ cursor: "pointer" }}
              onClick={handleDownload}
            >
              Click Here to download template
            </Typography>
          </Box>
        </Grid>
      </Grid>
    </Box>
  );
};

export default FileUploadDropBox;
