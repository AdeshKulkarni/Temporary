flowchart TD
    
    %% ==========================================
    %% WORKFLOW ORDER 1: CREATION
    %% ==========================================
    subgraph Order1 ["1. CREATION STAGE (Maker - Role: 9001 TCSGLIF)"]
        Start(["Maker Selects VID<br/>*(VID Status: ACTIVE)*"]) --> Lock["VID Locked in Redis for 30m<br/>*(Batch Status: DRAFT)*"]
        
        Lock --> EntryChoice{"Choose Entry Method"}
        EntryChoice --> |Manual Grid| ValCheck{"System Validation"}
        EntryChoice --> |Excel Upload| ValCheck
        
        ValCheck --> |Errors Found| Fix["Show Errors to Maker"] --> EntryChoice
        ValCheck --> |Clean Data| Sched["Set Execution Time"]
        
        Sched --> Submit(["Submit Batch"])
    end
    
    %% ==========================================
    %% WORKFLOW ORDER 2: VERIFICATION
    %% ==========================================
    subgraph Order2 ["2. VERIFICATION STAGE (Checker 1 - Role: 9003)"]
        Submit --> |Routes to| Inbox2(["Inbox: Role 9003<br/>*(Batch: PENDING | VID: IN-USE)*"])
        Inbox2 --> Dec2{"Is Data Correct?"}
        
        Dec2 --> |YES| Acc2["Accept + Remarks"]
        Dec2 --> |NO| Rej2["Reject + Remarks"]
    end
    
    %% ==========================================
    %% WORKFLOW ORDER 3: REVIEW
    %% ==========================================
    subgraph Order3 ["3. REVIEW STAGE (Checker 2 - Role: 9002 SBIGLIF)"]
        Acc2 --> |Routes to| Inbox3(["Inbox: Role 9002<br/>*(Batch: PENDING | VID: IN-USE)*"])
        Inbox3 --> Dec3{"Is Data Correct?"}
        
        Dec3 --> |YES| Acc3["Accept + Remarks"]
        Dec3 --> |NO| Rej3["Reject + Remarks"]
    end
    
    %% ==========================================
    %% WORKFLOW ORDER 4: APPROVAL (FINAL)
    %% ==========================================
    subgraph Order4 ["4. APPROVAL STAGE (Final Approver - Role: 9002 SBIGLIF)"]
        Acc3 --> |Routes to| Inbox4(["Inbox: Role 9002<br/>*(Batch: PENDING | VID: IN-USE)*"])
        Inbox4 --> Dec4{"Is Data Correct?"}
        
        Dec4 --> |YES| Acc4["Accept + Remarks"]
        Dec4 --> |NO| Rej4["Reject + Remarks"]
        
        Acc4 --> SchedCheck{"Check Execution Schedule"}
        SchedCheck --> |Keep Maker's Schedule| Commit["Commit Final Approval"]
        SchedCheck --> |Override with New Time| Commit
    end
    
    %% ==========================================
    %% WORKFLOW ORDER 5: EXECUTION (System / PSO)
    %% ==========================================
    subgraph Order5 ["5. EXECUTION STAGE (System / PSO Role: 9001)"]
        Commit --> ExecCheck{"When to Execute?"}
        
        ExecCheck --> |Immediate| PostNow(["Post to Core Ledger Instantly"])
        ExecCheck --> |Future Time| Wait(["Wait for Scheduled Time<br/>*(Batch: SCHEDULED)*"]) --> PostNow
        
        PostNow --> |Success| Done(["Transaction Successful<br/>*(Batch: POSTED | VID: COMPLETED)*"])
        
        PostNow --> |Timeout / Database Error| FailQueue(["Execution Failed<br/>*(Batch: EXECUTION FAILED)*"])
        
        FailQueue --> |Routed to| RetryRole(["PSO Inbox: Role 9001 TCSGLIF"])
        RetryRole --> |Clicks Retry| PostNow
    end
    
    %% ==========================================
    %% GLOBAL REJECTION HANDLING
    %% ==========================================
    Rej2 --> Terminate
    Rej3 --> Terminate
    Rej4 --> Terminate
    
    Terminate["Workflow Terminated<br/>*(Batch Status: REJECTED)*"] --> UnlockVID(["VID UNLOCKED & AVAILABLE AGAIN<br/>*(VID Status reset to ACTIVE)*"])
    
    UnlockVID -.-> |Maker is Notified. Can reuse VID for a new batch.| Start

    %% ==========================================
    style Start fill:#e1f5fe,stroke:#0288d1,stroke-width:2px
    style Submit fill:#e8f5e9,stroke:#2e7d32,stroke-width:2px
    
    style Acc2 fill:#e8f5e9,stroke:#2e7d32
    style Acc3 fill:#e8f5e9,stroke:#2e7d32
    style Acc4 fill:#e8f5e9,stroke:#2e7d32
    
    style Rej2 fill:#ffebee,stroke:#c62828
    style Rej3 fill:#ffebee,stroke:#c62828
    style Rej4 fill:#ffebee,stroke:#c62828
    
    style Terminate fill:#b71c1c,stroke:#7f0000,stroke-width:2px,color:#fff
    style UnlockVID fill:#fff9c4,stroke:#fbc02d,stroke-width:2px,stroke-dasharray: 5 5
    
    style Commit fill:#a5d6a7,stroke:#1b5e20,stroke-width:2px
    style Done fill:#4caf50,stroke:#1b5e20,stroke-width:3px,color:#fff
    style FailQueue fill:#ff9800,stroke:#e65100,stroke-width:2px,color:#fff
