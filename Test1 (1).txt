// Get Icon component dynamically from icon name
const getIconComponent = (iconName) => {
  const Icon = MuiIcons[iconName];
  return Icon ? <Icon /> : <MuiIcons.HelpOutline />; // fallback icon
};

const SideBar = () => {
  const user = JSON.parse(sessionStorage.getItem("user"));
  const navigate = useNavigate();
  const location = useLocation();

  const [menuItems, setMenuItems] = useState([]);
  const [selectedMenuItem, setSelectedMenuItem] = useState(0);

  // Load menu items from sessionStorage or fallback to default
  useEffect(() => {
    let finalMenu = [];
    const storedMenu = sessionStorage.getItem("menuItems");

    if (storedMenu) {
      try {
        const parsedMenu = JSON.parse(storedMenu);
        if (Array.isArray(parsedMenu) && parsedMenu.length > 0) {
          finalMenu = parsedMenu.map((item, idx) => ({
            id: item.id || idx + 1, // Ensure unique id
            ...item,
          }));
        }
      } catch (err) {
        console.error("Invalid menu data in sessionStorage:", err);
      }
    }

    // Use default if session menu empty
    if (finalMenu.length === 0 && DEFAULT_MENU.length > 0) {
      finalMenu = DEFAULT_MENU;
    }
    console.log(finalMenu);
    setMenuItems(finalMenu);
  }, []);

  // Highlight menu based on current route
  useEffect(() => {
    if (menuItems.length > 0) {
      const activeIndex = menuItems.findIndex((item) =>
        location.pathname.includes(convertToKebabCase(item.text))
      );
      if (activeIndex >= 0) setSelectedMenuItem(activeIndex);
    }
  }, [location.pathname, menuItems]);

  const navigateToPage = (item, index) => {
    setSelectedMenuItem(index);
    if (user?.role) {
      navigate(`/${roleConstants[user.role]}/${convertToKebabCase(item.text)}`);
    }
  };

  const drawer = (
    <Box sx={{ height: "100%", display: "flex", flexDirection: "column" }}>
      {/* Logo Section */}
      <Box
        sx={{
          p: 2,
          borderBottom: "1px solid #E2E8F0",
          display: "flex",
          alignItems: "center",
          gap: 2,
        }}
      >
        <img src={f1Logo} alt="Logo" style={{ width: 50, height: 50 }} />
        <Box>
          <Typography variant="h6" sx={{ fontWeight: 700 }}>
            FinCore
          </Typography>
          <Typography variant="caption" color="text.secondary">
            Empowering Data
          </Typography>
        </Box>
      </Box>

      {/* Menu Items */}
      <List sx={{ flexGrow: 1, py: 1 }}>
        {menuItems.length > 0 ? (
          menuItems.map((item, index) => (
            <ListItem key={item.id} disablePadding>
              <ListItemButton
                selected={selectedMenuItem === index}
                onClick={() => navigateToPage(item, index)}
              >
                <ListItemIcon sx={{ minWidth: 40 }}>
                  {getIconComponent(item.icon)}
                </ListItemIcon>
                <ListItemText
                  primary={item.text}
                  sx={{
                    "& .MuiTypography-root": {
                      fontWeight: selectedMenuItem === index ? 600 : 400,
                      color:
                        selectedMenuItem === index
                          ? "primary.main"
                          : "text.primary",
                    },
                  }}
                />
                {item.badge && (
                  <Chip
                    size="small"
                    label={item.badge}
                    sx={{
                      height: 20,
                      fontSize: "0.7rem",
                      backgroundColor: "secondary.main",
                      color: "white",
                    }}
                  />
                )}
              </ListItemButton>
            </ListItem>
          ))
        ) : (
          <Typography
            variant="body2"
            color="text.secondary"
            sx={{ textAlign: "center", mt: 3 }}
          >
            No menu available
          </Typography>
        )}
      </List>

      {/* Support Section */}
      <Box sx={{ p: 2, borderTop: "1px solid #E2E8F0" }}>
        <Card>
          <CardContent sx={{ p: 2 }}>
            <Typography variant="subtitle2" sx={{ fontWeight: 600, mb: 1 }}>
              Need Help?
            </Typography>
            <Typography variant="body2" color="text.secondary" sx={{ mb: 1 }}>
              Contact our support team
            </Typography>
            <Button size="small" variant="contained" fullWidth>
              Connect to SCORE
            </Button>
          </CardContent>
        </Card>
      </Box>
    </Box>
  );

  return (
    <Box
      component="nav"
      sx={{ width: { md: drawerWidth }, flexShrink: { md: 0 } }}
    >
      <Drawer
        variant="permanent"
        sx={{
          display: { xs: "none", md: "block" },
          "& .MuiDrawer-paper": { width: drawerWidth, boxSizing: "border-box" },
        }}
        open
      >
        {drawer}
      </Drawer>
    </Box>
  );
};

export default SideBar;
