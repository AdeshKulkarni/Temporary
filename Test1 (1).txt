package com.tcs.fincore.MOCService.Models;

import com.tcs.fincore.MOCService.Models.enums.AcClassification;
import jakarta.persistence.*;
import lombok.Data;
import java.time.LocalDate;

@Entity
@Table(name = "CGL_MASTER") // Assuming this is the table name
@Data
public class CglMaster {

    @Id
    @Column(name = "CGL_NUMBER", length = 10, nullable = false)
    private String cglNumber;

    @Column(name = "COMP_1", length = 4, nullable = false)
    private String comp1;

    @Column(name = "SEGMENT_CODE", length = 4, nullable = false)
    private String segmentCode;

    @Column(name = "COMP_2", length = 2, nullable = false)
    private String comp2;

    @Column(name = "DESCRIPTION", length = 100, nullable = false)
    private String description;

    // --- MAJOR FIX ---
    // The field type is changed from 'AcClassification' (enum) to 'String'.
    // We remove the @Enumerated annotation.
    // Now, JPA will just store the raw string/char code (e.g., "E") directly,
    // which is exactly what we want and what the database expects.
    @Column(name = "AC_CLASSIFICATION", length = 1, nullable = false)
    private String acClassification;
    // --- END FIX ---

    @Column(name = "BAL_FWD", nullable = false)
    private Boolean balFwd;

    @Column(name = "DEF_BAL_TYPE", length = 1, nullable = false)
    private String defBalType;

    @Column(name = "STATUS", nullable = false)
    private Boolean status;

    @Column(name = "OPEN_DATE", nullable = false)
    private LocalDate openDate;

    @Column(name = "CLOSE_DATE")
    private LocalDate closeDate;

    @Column(name = "BAL_COMPARE", nullable = false)
    private Boolean balCompare;

    @Column(name = "MANUAL_POSTING", nullable = false)
    private Boolean manualPosting;
}
