import React, { useState, useEffect, useCallback, useRef } from "react";
import {
  Box,
  Chip,
  IconButton,
  Tooltip,
  Stack,
  Typography,
  alpha
} from "@mui/material";
import { DataGrid } from "@mui/x-data-grid";
import { 
    Visibility as VisibilityIcon, 
    Info as InfoIcon,
    Edit as EditIcon 
} from "@mui/icons-material";

// Adjust these paths based on your folder structure
import useApi from "../../hooks/useApi"; 
import useCustomSnackbar from "../../utils/useCustomSnackbar";
import TemplateViewDialog from "../viewReport/pages/TemplateViewDialog";

const styles = {
  mainContainer: {
    padding: 3,
    height: "100%",
    width: "100%",
    backgroundColor: "background.default",
  },
  dataGridContainer: {
    height: 600,
    width: "100%",
    backgroundColor: "background.paper",
    boxShadow: 2,
    borderRadius: 2,
    border: 1,
    borderColor: "divider",
    "& .MuiDataGrid-columnHeaders": {
      backgroundColor: (theme) => alpha(theme.palette.primary.main, 0.08),
      fontWeight: "bold",
      fontSize: "0.95rem",
    },
    "& .MuiDataGrid-row:hover": {
      backgroundColor: (theme) => alpha(theme.palette.primary.main, 0.04),
    },
  },
};

function CustomNoRowsOverlay() {
  return (
    <Stack height="100%" alignItems="center" justifyContent="center">
      <InfoIcon sx={{ fontSize: 48, color: "text.secondary", mb: 2 }} />
      <Typography variant="h6">No Templates Found</Typography>
    </Stack>
  );
}

export default function ReportViewScreen() {
  const { callApi } = useApi();
  const showSnackBar = useCustomSnackbar();
  const isMounted = useRef(true);
  
  const [loading, setLoading] = useState(true);
  const [rows, setRows] = useState([]);
  
  // Dialog State
  const [isViewDialogOpen, setIsViewDialogOpen] = useState(false);
  const [selectedTemplate, setSelectedTemplate] = useState(null);

  useEffect(() => {
    isMounted.current = true;
    return () => { isMounted.current = false; };
  }, []);

  const fetchTemplates = useCallback(async () => {
    setLoading(true);
    try {
      const response = await callApi("/RS/template-config/get-templates", null, "GET");

      if (isMounted.current) {
        const dataList = response && Array.isArray(response.data) 
          ? response.data 
          : [];
        setRows(dataList);
      }
    } catch (err) {
      if (isMounted.current && err.name !== "CanceledError") {
        showSnackBar("Failed to load templates", "error");
      }
    } finally {
      if (isMounted.current) setLoading(false);
    }
  }, [callApi, showSnackBar]);

  useEffect(() => {
    fetchTemplates();
  }, [fetchTemplates]);

  // --- HANDLERS ---

  const handleView = (row) => {
    setSelectedTemplate(row);
    setIsViewDialogOpen(true);
  };

  const handleEditRow = (row) => {
    console.log("Edit entire row:", row);
    // Logic to navigate to edit screen
  };

  const handleEditStatus = (row) => {
    console.log("Edit Status for:", row.templateName);
    // Logic to open status change menu/dialog
  };

  const columns = [
    {
      field: "templateName",
      headerName: "Template Name",
      flex: 1,
      minWidth: 180,
    },
    {
      field: "description",
      headerName: "Description",
      flex: 1.5,
      minWidth: 220,
    },
    {
      field: "status",
      headerName: "Status",
      width: 170, // Width for Chip + Edit Button
      align: "center",
      headerAlign: "center",
      renderCell: (params) => (
        <Stack direction="row" alignItems="center" spacing={1} justifyContent="center" sx={{ width: '100%' }}>
            <Chip 
                label={params.value} 
                color={params.value === "ACTIVE" ? "success" : "default"} 
                size="small" 
                variant="outlined"
                sx={{ fontWeight: 'bold', minWidth: 70 }}
            />
            {/* Status Edit Button (Outside the View Dialog) */}
            <Tooltip title="Change Status">
                <IconButton 
                    size="small" 
                    onClick={(e) => {
                        e.stopPropagation(); 
                        handleEditStatus(params.row);
                    }}
                    sx={{ 
                        color: 'text.secondary',
                        '&:hover': { color: 'primary.main' }
                    }}
                >
                    <EditIcon fontSize="small" />
                </IconButton>
            </Tooltip>
        </Stack>
      ),
    },
    {
      field: "versionNo",
      headerName: "Ver.",
      width: 80,
      align: "center",
      headerAlign: "center",
    },
    {
      field: "actions",
      headerName: "Action",
      width: 120,
      sortable: false,
      align: "center",
      headerAlign: "center",
      renderCell: (params) => (
        <Stack direction="row" spacing={0} justifyContent="center">
            {/* The Original "Eye" Button */}
            <Tooltip title="View Report">
                <IconButton 
                    color="primary" 
                    onClick={() => handleView(params.row)}
                    size="small"
                >
                    <VisibilityIcon />
                </IconButton>
            </Tooltip>
            
            {/* Row Edit Button */}
            <Tooltip title="Edit Details">
                <IconButton 
                    color="secondary" 
                    onClick={() => handleEditRow(params.row)}
                    size="small"
                >
                    <EditIcon />
                </IconButton>
            </Tooltip>
        </Stack>
      ),
    },
  ];

  return (
    <Box sx={styles.mainContainer}>
      <DataGrid
        rows={rows}
        columns={columns}
        loading={loading}
        getRowId={(row) => row.templateId}
        disableRowSelectionOnClick
        pageSizeOptions={[5, 10, 25]}
        rowHeight={60} 
        initialState={{
          pagination: { paginationModel: { page: 0, pageSize: 10 } },
        }}
        slots={{ noRowsOverlay: CustomNoRowsOverlay }}
        sx={styles.dataGridContainer}
      />

      {/* The Dialog is called here, kept original */}
      <TemplateViewDialog 
        open={isViewDialogOpen} 
        handleClose={() => setIsViewDialogOpen(false)} 
        template={selectedTemplate} 
      />
    </Box>
  );
}
