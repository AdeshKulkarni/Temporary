import React, { useState, useEffect } from 'react';
import { 
  Dialog, DialogTitle, DialogContent, DialogActions, 
  Button, TextField, Divider, Grid, Typography, Box, IconButton, Chip, Tooltip, Stack 
} from '@mui/material';
import { DataGrid } from "@mui/x-data-grid";
import { 
    Close as CloseIcon, 
    Description as DescriptionIcon,
    Edit as EditIcon,
    PlayArrow as PlayArrowIcon,
    Info as InfoIcon
} from '@mui/icons-material';
import { alpha } from "@mui/material/styles";

const styles = {
    dialogTitle: {
        display: "flex",
        justifyContent: "space-between",
        alignItems: "center",
        backgroundColor: (theme) => alpha(theme.palette.primary.main, 0.08),
        pb: 2,
    },
    sectionTitle: {
        mt: 2,
        mb: 1,
        display: 'flex',
        alignItems: 'center',
        gap: 1
    }
};

const TemplateViewDialog = ({ open, handleClose, template }) => {
  const [selectedVariantCode, setSelectedVariantCode] = useState('');
  const [formData, setFormData] = useState({});

  // Reset state when dialog opens or template changes
  useEffect(() => {
    if (open && template) {
        setSelectedVariantCode('');
        setFormData({});
    }
  }, [open, template]);

  if (!template) return null;

  const activeVariant = template.variants?.find(v => v.variantCode === selectedVariantCode);

  const handleInputChange = (name, value) => {
    setFormData(prev => ({ ...prev, [name]: value }));
  };

  const handleGenerate = () => {
    console.log("Generating Report for:", {
        template: template.templateName,
        variant: selectedVariantCode,
        data: formData
    });
    handleClose();
  };

  // --- Handlers for Table Actions ---
  
  const handleSelectVariant = (variantCode) => {
      setSelectedVariantCode(variantCode);
      setFormData({}); // Reset form when switching variants
  };

  const handleEditStatus = (row) => {
      console.log("Edit Status for variant:", row.variantName);
      // Open status change dialog or logic here
  };

  const handleEditVariant = (row) => {
      console.log("Edit Variant details:", row);
      // Open variant edit dialog here
  };

  // --- DataGrid Configuration ---

  const columns = [
    { 
        field: 'variantName', 
        headerName: 'Variant Name', 
        flex: 1, 
        minWidth: 150 
    },
    { 
        field: 'description', 
        headerName: 'Description', 
        flex: 1.5, 
        minWidth: 200 
    },
    {
        field: 'status',
        headerName: 'Status',
        width: 160,
        align: 'center',
        headerAlign: 'center',
        renderCell: (params) => (
            <Stack direction="row" alignItems="center" spacing={1} justifyContent="center" sx={{ width: '100%' }}>
                <Chip 
                    label={params.value || "ACTIVE"} // Defaulting to ACTIVE if undefined for demo
                    color={params.value === "INACTIVE" ? "default" : "success"} 
                    size="small" 
                    variant="outlined"
                    sx={{ fontWeight: 'bold', minWidth: 70 }}
                />
                <Tooltip title="Change Status">
                    <IconButton 
                        size="small" 
                        onClick={(e) => {
                            e.stopPropagation();
                            handleEditStatus(params.row);
                        }}
                        sx={{ 
                            color: 'text.secondary',
                            bgcolor: (theme) => alpha(theme.palette.action.hover, 0.5),
                            '&:hover': { bgcolor: 'action.hover' }
                        }}
                    >
                        <EditIcon fontSize="small" />
                    </IconButton>
                </Tooltip>
            </Stack>
        )
    },
    {
        field: 'actions',
        headerName: 'Actions',
        width: 120,
        sortable: false,
        align: 'center',
        headerAlign: 'center',
        renderCell: (params) => (
            <Stack direction="row" spacing={0.5}>
                 <Tooltip title="Edit Details">
                    <IconButton 
                        color="primary" 
                        size="small"
                        onClick={(e) => {
                            e.stopPropagation();
                            handleEditVariant(params.row);
                        }}
                    >
                        <EditIcon fontSize="small" />
                    </IconButton>
                </Tooltip>
                <Tooltip title="Select & Generate">
                    <IconButton 
                        color={selectedVariantCode === params.row.variantCode ? "success" : "default"}
                        size="small"
                        onClick={(e) => {
                            e.stopPropagation();
                            handleSelectVariant(params.row.variantCode);
                        }}
                    >
                        <PlayArrowIcon fontSize="small" />
                    </IconButton>
                </Tooltip>
            </Stack>
        )
    }
  ];

  return (
    <Dialog 
        open={open} 
        onClose={handleClose} 
        maxWidth="md" // Increased width for the table
        fullWidth
        PaperProps={{ sx: { borderRadius: 2, height: '80vh' } }} // Fixed height for consistency
    >
      {/* Styled Header */}
      <DialogTitle sx={styles.dialogTitle}>
        <Box display="flex" alignItems="center" gap={1}>
            <DescriptionIcon color="primary" />
            <Box>
                <Typography variant="h6" component="div" fontWeight="bold">
                    {template.templateName}
                </Typography>
                <Typography variant="caption" color="text.secondary">
                    Manage Variants & Generate Report
                </Typography>
            </Box>
        </Box>
        <IconButton onClick={handleClose} size="small">
            <CloseIcon />
        </IconButton>
      </DialogTitle>

      <DialogContent dividers sx={{ p: 0, display: 'flex', flexDirection: 'column' }}>
        
        {/* Top Section: Variants Management Table */}
        <Box sx={{ flex: 1, p: 2, pb: 0 }}>
            <DataGrid
                rows={template.variants || []}
                columns={columns}
                getRowId={(row) => row.variantCode} // Assuming variantCode is unique
                density="compact"
                hideFooter
                disableRowSelectionOnClick
                sx={{
                    border: 1,
                    borderColor: 'divider',
                    '& .MuiDataGrid-columnHeaders': {
                        backgroundColor: (theme) => alpha(theme.palette.primary.main, 0.05),
                    }
                }}
            />
        </Box>

        {/* Bottom Section: Parameters Form (Visible only when variant selected) */}
        <Box sx={{ p: 3, backgroundColor: 'background.default', borderTop: 1, borderColor: 'divider', mt: 2 }}>
            {!activeVariant ? (
                <Box display="flex" alignItems="center" justifyContent="center" py={2} color="text.secondary" gap={1}>
                    <InfoIcon fontSize="small" />
                    <Typography variant="body2">Select a variant from the table above to generate a report.</Typography>
                </Box>
            ) : (
                <>
                    <Typography variant="subtitle2" color="primary" gutterBottom fontWeight="bold">
                        Configuration for: {activeVariant.variantName}
                    </Typography>
                    
                    <Grid container spacing={2} sx={{ mt: 0.5 }}>
                        {activeVariant.params?.map((param) => (
                            <Grid item xs={12} sm={6} key={param.paramName}>
                                <TextField
                                    fullWidth
                                    size="small"
                                    label={param.label}
                                    type={param.paramType === 'DATE' ? 'date' : 'text'}
                                    required={param.required}
                                    InputLabelProps={{ shrink: true }} 
                                    onChange={(e) => handleInputChange(param.paramName, e.target.value)}
                                    placeholder={param.uiHint || ''}
                                />
                            </Grid>
                        ))}
                         <Grid item xs={12} sm={6}>
                            <TextField 
                                select 
                                fullWidth 
                                size="small" 
                                label="Export Format" 
                                defaultValue="pdf"
                            >
                                <Box component="option" value="pdf">PDF Document</Box>
                                <Box component="option" value="excel">Excel Spreadsheet</Box>
                                <Box component="option" value="text">Plain Text</Box>
                            </TextField>
                        </Grid>
                    </Grid>
                </>
            )}
        </Box>

      </DialogContent>
      
      <DialogActions sx={{ p: 2 }}>
        <Button onClick={handleClose} color="inherit">Close</Button>
        <Button 
            variant="contained" 
            color="primary" 
            onClick={handleGenerate}
            disabled={!selectedVariantCode}
            startIcon={<DescriptionIcon />}
        >
          Generate Report
        </Button>
      </DialogActions>
    </Dialog>
  );
};

export default TemplateViewDialog;
