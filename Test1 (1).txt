import React, { useState, useEffect } from "react";
import {
  Box,
  Dialog,
  DialogTitle,
  DialogContent,
  DialogActions,
  TextField,
  Typography,
  Button,
  Stack,
  Tooltip,
  IconButton,
  Chip,
} from "@mui/material";
import { DataGrid } from "@mui/x-data-grid";
import { Check as CheckIcon, Clear as ClearIcon } from "@mui/icons-material";
import AddBoxIcon from "@mui/icons-material/AddBox";
import AutoDeleteIcon from "@mui/icons-material/AutoDelete";
import EditDocumentIcon from "@mui/icons-material/EditDocument";
import AccountCircleIcon from "@mui/icons-material/AccountCircle";
import { useNavigate } from "react-router-dom";
import useApi from "../../hooks/useApi";
import useCustomSnackbar from "../../utils/useCustomSnackbar";
import { COMMON_REQ_SCREEN_CONFIGS } from "../common/CommonConstants";
import { useSelector } from "react-redux";

export default function CommonRequestScreen() {
  const selectedMenuItem = useSelector((state) => state.menus.selectedMenuItem);
  const navigate = useNavigate();
  const { callApi } = useApi();
  const showSnackBar = useCustomSnackbar();
  const screenConfig = COMMON_REQ_SCREEN_CONFIGS[selectedMenuItem?.id];
  const { requestType, columns, transformData } = screenConfig;
  const [rows, setRows] = useState([]);
  const [loading, setLoading] = useState(true);
  const [remarks, setRemarks] = useState("");
  const [confirmDialog, setConfirmDialog] = useState({
    open: false,
    type: null,
    rowId: null,
  });

  const fetchData = async () => {
    setLoading(true);
    try {
      const response = await callApi(
        "/CR/pending-requests",
        { requestType },
        "POST"
      );
      const result = response?.data || [];

      const data = result.map((row) => ({
        ...row,
        ...(typeof row.payload === "object" ? row.payload : {}),
      }));

      setRows(transformData ? transformData(data) : data);

      if (!data.length) showSnackBar("No pending requests found", "warning");
    } catch (err) {
      console.error(err);
      showSnackBar("Error fetching data", "error");
    } finally {
      setLoading(false);
    }
  };

  useEffect(() => {
    if (!screenConfig || !requestType) {
      navigate("/404");
      return;
    }
    setRows([]);
    setRemarks("");
    setConfirmDialog({ open: false, type: null, rowId: null });

    fetchData();
  }, [selectedMenuItem?.id]);

  const handleConfirm = async () => {
    if (confirmDialog.type === "reject" && !remarks.trim()) {
      showSnackBar("Please provide remarks before rejecting", "warning");
      return;
    }

    try {
      const payload = {
        requestId: confirmDialog.rowId,
        status: confirmDialog.type === "reject" ? "REJECTED" : "ACCEPTED",
        remarks:
          confirmDialog.type === "reject" ? remarks.trim() : "No Remarks",
      };

      const res = await callApi("/CR/update-request", payload, "PATCH");

      if (res?.success || res) {
        showSnackBar(
          confirmDialog.type === "accept"
            ? "Request Approved"
            : "Request Rejected",
          "success"
        );
        fetchData();
      } else {
        showSnackBar("Failed to process request", "error");
      }
    } catch (err) {
      console.error(err);
      showSnackBar("Error while processing request", "error");
    }
    closeConfirmDialog();
  };

  const openConfirmDialog = (type, rowId) =>
    setConfirmDialog({ open: true, type, rowId });

  const closeConfirmDialog = () =>
    setConfirmDialog({ open: false, type: null, rowId: null });

  const allColumns = [
    ...columns,
    {
      field: "changeType",
      headerName: "Request Type",
      flex: 0.5,
      align: "center",
      headerAlign: "center",
      renderCell: (params) => {
        const val = params.value;
        return (
          <Box>
            {val === "ADD" ? (
              <Chip icon={<AddBoxIcon />} label="Added" size="small" />
            ) : val === "DELETE" ? (
              <Chip icon={<AutoDeleteIcon />} label="Deleted" size="small" />
            ) : (
              <Chip icon={<EditDocumentIcon />} label="Modified" size="small" />
            )}
          </Box>
        );
      },
    },
    {
      field: "creatorId",
      headerName: "Creator ID",
      flex: 0.8,
      align: "center",
      headerAlign: "center",
      renderCell: (params) => (
        <Chip icon={<AccountCircleIcon />} label={params.value} size="small" />
      ),
    },
    {
      field: "reqDate",
      headerName: "Requested Date",
      flex: 1,
      align: "center",
      headerAlign: "center",
      renderCell: (params) => {
        const isoString = params.value;
        if (!isoString) return null;
        try {
          const date = new Date(isoString);
          return (
            <Box>
              {date.toLocaleString("en-US", {
                year: "numeric",
                month: "long",
                day: "numeric",
                hour: "2-digit",
                minute: "2-digit",
                second: "2-digit",
                hour12: true,
                timeZone: "Asia/Calcutta",
              })}
            </Box>
          );
        } catch {
          return <Box>{isoString}</Box>;
        }
      },
    },
    {
      field: "actions",
      headerName: "Actions",
      flex: 0.6,
      align: "center",
      headerAlign: "center",
      renderCell: (params) => (
        <Stack direction="row" spacing={1} justifyContent="center">
          <Tooltip title="Approve">
            <IconButton
              onClick={() => openConfirmDialog("accept", params.row.id)}
              color="success"
            >
              <CheckIcon />
            </IconButton>
          </Tooltip>
          <Tooltip title="Reject">
            <IconButton
              onClick={() => openConfirmDialog("reject", params.row.id)}
              color="error"
            >
              <ClearIcon />
            </IconButton>
          </Tooltip>
        </Stack>
      ),
    },
  ];

  return (
    <Box>
      <DataGrid
        rows={rows}
        columns={allColumns}
        disableRowSelectionOnClick
        pageSizeOptions={[5, 10, 25, 50]}
        initialState={{
          pagination: { paginationModel: { pageSize: 10, page: 0 } },
          sorting: { sortModel: [{ field: "reqDate", sort: "desc" }] },
        }}
        getRowId={(row) => row.id}
        loading={loading}
        sx={{
          height: 600,
          width: "100%",
          "& .MuiDataGrid-cell:focus": { outline: "none" },
        }}
      />

      <Dialog
        open={confirmDialog.open}
        onClose={closeConfirmDialog}
        maxWidth="sm"
        fullWidth
      >
        <DialogTitle>
          {confirmDialog.type === "accept"
            ? "Confirm Approval"
            : "Confirm Rejection"}
        </DialogTitle>
        <DialogContent>
          <Typography sx={{ p: 2 }}>
            {confirmDialog.type === "accept"
              ? "Are you sure you want to approve this request?"
              : "Are you sure you want to reject this request? Please enter remarks."}
          </Typography>
          {confirmDialog.type === "reject" && (
            <TextField
              fullWidth
              label="Rejection Remarks"
              value={remarks}
              onChange={(e) => setRemarks(e.target.value.slice(0, 100))}
              multiline
              rows={2}
              margin="dense"
              inputProps={{ maxLength: 100 }}
            />
          )}
        </DialogContent>
        <DialogActions>
          <Button
            onClick={closeConfirmDialog}
            color="inherit"
            variant="contained"
          >
            Cancel
          </Button>
          <Button
            onClick={handleConfirm}
            color={confirmDialog.type === "accept" ? "success" : "error"}
            variant="contained"
          >
            {confirmDialog.type === "accept" ? "Approve" : "Reject"}
          </Button>
        </DialogActions>
      </Dialog>
    </Box>
  );
}
