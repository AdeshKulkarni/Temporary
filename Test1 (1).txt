class Solution {
    public int widthOfBinaryTree(TreeNode root) {
        if (root == null) {
            return 0;
        }

        int MaxCount = 0;
        
        // CHANGE 1: Use Pair instead of TreeNode
        Queue<Pair> q = new LinkedList<>();
        
        // Give the root node seat number 0
        q.offer(new Pair(root, 0));

        while (!q.isEmpty()) {
            int Qsize = q.size();
            
            // Variables to track the first and last seat numbers on this specific level
            int first = 0; 
            int last = 0;

            for (int i = 0; i < Qsize; i++) {
                
                // Pull out the Pair to get BOTH the node and its seat number
                Pair current = q.poll();
                TreeNode node = current.node;
                int index = current.index;

                // CHANGE 2: Record the first and last seat numbers
                if (i == 0) {
                    first = index; // Left-most node of this level
                }
                if (i == Qsize - 1) {
                    last = index; // Right-most node of this level
                }

                // CHANGE 3: The Index Math!
                // Left child is always: (2 * parentIndex) + 1
                if (node.left != null) {
                    q.offer(new Pair(node.left, (2 * index) + 1));
                }
                
                // Right child is always: (2 * parentIndex) + 2
                if (node.right != null) {
                    q.offer(new Pair(node.right, (2 * index) + 2));
                }
            }

            // Once the level finishes, calculate the width and update MaxCount!
            int currentWidth = (last - first) + 1;
            MaxCount = Math.max(MaxCount, currentWidth);
        }

        return MaxCount;
    }
}
