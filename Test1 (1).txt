import React, { useState, useEffect, useCallback } from "react";
import {
  Box,
  Typography,
  Chip,
  Dialog,
  DialogActions,
  DialogContent,
  DialogContentText,
  DialogTitle,
  TextField,
  Button,
  CircularProgress,
  Tooltip,
  IconButton,
  Stack,
  Grid,
  Table,
  TableBody,
  TableCell,
  TableContainer,
  TableHead,
  TableRow,
  Paper,
  Checkbox,
} from "@mui/material";
import { DataGrid } from "@mui/x-data-grid";
import {
  Check as CheckIcon,
  Clear as ClearIcon,
  Visibility as VisibilityIcon,
  Close as CloseIcon,
} from "@mui/icons-material";
import AccountCircleIcon from "@mui/icons-material/AccountCircle";
import InfoIcon from "@mui/icons-material/Info";
import { alpha } from "@mui/material/styles";

import useApi from "../../hooks/useApi";
import useCustomSnackbar from "../../utils/useCustomSnackbar";

// --- DEFINED STYLES (Replaces external import to fix CSS issues) ---
const styles = {
  mainContainer: {
    padding: 3,
    height: "100%",
    width: "100%",
    backgroundColor: "background.default",
  },
  dataGridContainer: {
    height: 600,
    width: "100%",
    backgroundColor: "background.paper",
    boxShadow: 2,
    borderRadius: 2,
    border: "1px solid",
    borderColor: "divider",
    "& .MuiDataGrid-columnHeaders": {
      backgroundColor: (theme) => alpha(theme.palette.primary.main, 0.08),
      fontWeight: "bold",
      fontSize: "0.95rem",
    },
    "& .MuiDataGrid-row:hover": {
      backgroundColor: (theme) => alpha(theme.palette.primary.main, 0.04),
    },
  },
  tableContainer: {
    maxHeight: 500,
    border: "1px solid",
    borderColor: "divider",
    mt: 1,
  },
  // Base style for all header cells
  tableHeaderCell: {
    fontWeight: "bold",
    backgroundColor: (theme) => alpha(theme.palette.primary.main, 0.08),
    whiteSpace: "nowrap",
    zIndex: 10, // Base header z-index
  },
  // Specific style for Sticky Header Columns (Top Left corners)
  stickyHeaderColumn: (leftOffset) => ({
    position: "sticky",
    left: leftOffset,
    top: 0,
    zIndex: 20, // Must be higher than standard header (10)
    backgroundColor: "#f5f5f5", // Solid color to hide scroll
    fontWeight: "bold",
    borderRight: "1px solid rgba(224, 224, 224, 1)",
    borderBottom: "1px solid rgba(224, 224, 224, 1)",
  }),
  // Specific style for Sticky Body Columns (Left side)
  stickyBodyColumn: (leftOffset) => ({
    position: "sticky",
    left: leftOffset,
    zIndex: 5, // Higher than body (1) but lower than header (10)
    backgroundColor: "background.paper", // Crucial: Solid background
    borderRight: "1px solid rgba(224, 224, 224, 1)",
  }),
  dialogTitle: {
    display: "flex",
    justifyContent: "space-between",
    alignItems: "center",
    pb: 1,
  },
  dialogActions: {
    p: 2,
    borderTop: "1px solid",
    borderColor: "divider",
  },
  loadingContainer: {
    display: "flex",
    justifyContent: "center",
    alignItems: "center",
    minHeight: 200,
  },
};

// --- HELPER FUNCTIONS ---
const formatCurrency = (num) => {
  return Number(num).toFixed(4);
};

const getTxnTypeColor = (txnType) => {
  return txnType === "Credit" ? "error" : "success";
};

function CustomNoRowsOverlay() {
  return (
    <Stack height="100%" alignItems="center" justifyContent="center">
      <InfoIcon sx={{ fontSize: 48, color: "text.secondary", mb: 2 }} />
      <Typography variant="h6">No Pending Requests Found</Typography>
      <Typography color="text.secondary">All Journal requests have been processed.</Typography>
    </Stack>
  );
}

export default function JournalRequestApproval() {
  const { callApi } = useApi();
  const showSnackBar = useCustomSnackbar();

  // --- STATE MANAGEMENT ---
  const [currentUserId, setCurrentUserId] = useState(null);
  const [loading, setLoading] = useState(true);
  const [rows, setRows] = useState([]);

  // Modal State (Approve/Reject Batch)
  const [isModalOpen, setIsModalOpen] = useState(false);
  const [isSubmitting, setIsSubmitting] = useState(false);
  const [selectedBatch, setSelectedBatch] = useState(null);
  const [modalMode, setModalMode] = useState("ACCEPTED");
  const [executorRemarks, setExecutorRemarks] = useState("");

  // View Modal State
  const [isViewModalOpen, setIsViewModalOpen] = useState(false);
  const [isViewLoading, setIsViewLoading] = useState(false);
  const [batchDetails, setBatchDetails] = useState([]);
  const [viewingBatchId, setViewingBatchId] = useState(null);
  const [viewingBatchCreatorId, setViewingBatchCreatorId] = useState(null);

  // Partial Selection State
  const [selectedJournalPrefixes, setSelectedJournalPrefixes] = useState([]);
  const [partialExecutorRemarks, setPartialExecutorRemarks] = useState("");
  const [isPartialSubmitting, setIsPartialSubmitting] = useState(false);

  // --- AUTH CHECK ---
  useEffect(() => {
    let userIdFound = false;
    try {
      const authDataString =
        sessionStorage.getItem("persist:auth") ||
        localStorage.getItem("persist:auth");

      if (authDataString) {
        const authData = JSON.parse(authDataString);
        const userDataString = authData.user;

        if (userDataString) {
          const userData = JSON.parse(userDataString);
          if (userData && userData.userId) {
            setCurrentUserId(String(userData.userId));
            userIdFound = true;
          }
        }
      }
    } catch (error) {
      console.error("Failed to parse auth data from storage:", error);
    }

    if (!userIdFound) {
      console.warn("Could not verify user ID from storage.");
    }
  }, []);

  // --- API CALLS ---
  const fetchPendingBatches = useCallback(async () => {
    setLoading(true);
    try {
      const pendingRequests = await callApi(
        "/JS/journals/pending-requests",
        null,
        "GET"
      );

      if (!Array.isArray(pendingRequests) || pendingRequests.length === 0) {
        setRows([]);
        setLoading(false);
        return;
      }

      const groupedBatches = new Map();
      for (const request of pendingRequests) {
        const batchId = request.batchId;

        if (!groupedBatches.has(batchId)) {
          groupedBatches.set(batchId, {
            id: batchId,
            batchId: batchId,
            creatorId: request.creatorId,
            requestStatus: request.requestStatus,
            requestDate: request.requestDate,
            commonBatchRemarks: request.commonBatchRemarks,
            totalDebit: 0,
            totalCredit: 0,
            requestCount: 0,
          });
        }

        const batchDetails = groupedBatches.get(batchId);
        batchDetails.requestCount++;

        try {
          const payload = JSON.parse(request.payload);
          const amount = parseFloat(payload.amount) || 0;

          if (amount < 0) {
            batchDetails.totalCredit += Math.abs(amount);
          } else {
            batchDetails.totalDebit += amount;
          }
        } catch (e) {
          console.error(`Failed to parse payload for request ${request.id}`, e);
        }
      }
      setRows(Array.from(groupedBatches.values()));
    } catch (err) {
      showSnackBar("Failed to load pending requests", "error");
    } finally {
      setLoading(false);
    }
  }, [callApi, showSnackBar]);

  useEffect(() => {
    fetchPendingBatches();
  }, [fetchPendingBatches]);

  // --- HANDLERS: MAIN ACTION MODAL ---
  const handleOpenModal = (batch, mode) => {
    setSelectedBatch(batch);
    setModalMode(mode);
    setExecutorRemarks("");
    setIsModalOpen(true);
  };

  const handleCloseModal = () => {
    if (isSubmitting) return;
    setIsModalOpen(false);
    setSelectedBatch(null);
  };

  const handleSubmitAction = async () => {
    if (modalMode === "REJECTED" && !executorRemarks.trim()) {
      showSnackBar("Approver remarks are required for rejection.", "warning");
      return;
    }

    setIsSubmitting(true);
    const payload = {
      batchId: selectedBatch.batchId,
      status: modalMode,
      remarks:
        executorRemarks.trim() ||
        (modalMode === "ACCEPTED" ? "Accepted" : "Rejected"),
    };

    try {
      await callApi("/JS/journals/process-bulk", payload, "POST");
      showSnackBar(`Batch ${modalMode === "ACCEPTED" ? "Approved" : "Rejected"} Successfully`, "success");
      handleCloseModal();
      fetchPendingBatches();
    } catch (error) {
      console.error("Failed to process batch:", error);
      showSnackBar("Failed to process batch", "error");
    } finally {
      setIsSubmitting(false);
    }
  };

  // --- HANDLERS: VIEW DETAILS & PARTIAL ACTIONS ---
  const handleOpenViewModal = async (batchId, creatorId) => {
    setIsViewModalOpen(true);
    setIsViewLoading(true);
    setBatchDetails([]);
    setViewingBatchId(batchId);
    setViewingBatchCreatorId(String(creatorId));

    try {
      const data = await callApi(
        `/JS/journals/by-batch/${batchId}`,
        null,
        "GET"
      );

      // Timeout to allow UI to settle before processing large data
      setTimeout(() => {
        if (Array.isArray(data)) {
          const journalGroups = new Map();

          const processedData = data
            .map((req) => {
              if (req.requestStatus !== "PENDING") {
                return null;
              }
              try {
                const payload = JSON.parse(req.payload);
                const prefix = req.journalId.split("-")[0];
                const amount = parseFloat(payload.amount) || 0;

                return {
                  ...req,
                  payload,
                  journalPrefix: prefix,
                  amount: Math.abs(amount),
                  transactionType: amount < 0 ? "Credit" : "Debit",
                };
              } catch (e) {
                console.error("Failed to parse payload for req:", req.id, e);
                return null;
              }
            })
            .filter(Boolean);

          // Sort by Journal ID
          processedData.sort((a, b) => a.journalId.localeCompare(b.journalId));

          // Group by Prefix
          for (const row of processedData) {
            const prefix = row.journalPrefix;
            if (!journalGroups.has(prefix)) {
              journalGroups.set(prefix, []);
            }
            journalGroups.get(prefix).push(row);
          }

          setBatchDetails(Array.from(journalGroups.values()));
        } else {
          setBatchDetails([]);
          showSnackBar("No details found for this batch.", "info");
        }
        setIsViewLoading(false);
      }, 100);
    } catch (error) {
      showSnackBar("Failed to load batch details", "error");
      setBatchDetails([]);
      setIsViewLoading(false);
    }
  };

  const handleCloseViewModal = () => {
    setIsViewModalOpen(false);
    setBatchDetails([]);
    setViewingBatchId(null);
    setViewingBatchCreatorId(null);
    setSelectedJournalPrefixes([]);
    setPartialExecutorRemarks("");
    setIsPartialSubmitting(false);
  };

  const handleSubmitPartialAction = async (mode) => {
    if (mode === "REJECTED" && !partialExecutorRemarks.trim()) {
      showSnackBar("Approver remarks are required for rejection.", "warning");
      return;
    }

    if (selectedJournalPrefixes.length === 0) {
      showSnackBar("Please select at least one entry to process.", "warning");
      return;
    }

    setIsPartialSubmitting(true);

    const payload = {
      journalIdPrefixes: selectedJournalPrefixes,
      status: mode,
      remarks:
        partialExecutorRemarks.trim() ||
        (mode === "ACCEPTED" ? "Accepted" : "Rejected"),
    };

    try {
      await callApi("/JS/journals/process-bulk", payload, "POST");
      showSnackBar(`Selection ${mode === "ACCEPTED" ? "Approved" : "Rejected"} Successfully`, "success");
      handleCloseViewModal();
      fetchPendingBatches();
    } catch (error) {
      console.error("Failed to process partial batch:", error);
      showSnackBar("Failed to process partial selection", "error");
    } finally {
      setIsPartialSubmitting(false);
    }
  };

  // --- SELECTION LOGIC ---
  const handleSelectAllClick = (event) => {
    if (event.target.checked) {
      const newSelecteds = batchDetails.map((group) => group[0].journalPrefix);
      setSelectedJournalPrefixes(newSelecteds);
      return;
    }
    setSelectedJournalPrefixes([]);
  };

  const handleRowCheckboxClick = (event, prefix) => {
    if (event) {
      event.stopPropagation();
    }

    const selectedIndex = selectedJournalPrefixes.indexOf(prefix);
    let newSelected = [];

    if (selectedIndex === -1) {
      newSelected = newSelected.concat(selectedJournalPrefixes, prefix);
    } else if (selectedIndex === 0) {
      newSelected = newSelected.concat(selectedJournalPrefixes.slice(1));
    } else if (selectedIndex === selectedJournalPrefixes.length - 1) {
      newSelected = newSelected.concat(selectedJournalPrefixes.slice(0, -1));
    } else if (selectedIndex > 0) {
      newSelected = newSelected.concat(
        selectedJournalPrefixes.slice(0, selectedIndex),
        selectedJournalPrefixes.slice(selectedIndex + 1)
      );
    }
    setSelectedJournalPrefixes(newSelected);
  };

  const isViewingSelfCreated =
    currentUserId &&
    viewingBatchCreatorId &&
    String(currentUserId) === String(viewingBatchCreatorId);

  // --- DATA GRID COLUMNS ---
  const columns = [
    {
      field: "batchId",
      headerName: "Batch ID",
      width: 125,
      align: "center",
      headerAlign: "center",
    },
    {
      field: "requestStatus",
      headerName: "Status",
      width: 125,
      align: "center",
      headerAlign: "center",
      renderCell: (params) => (
        <Chip label={params.value} color="warning" size="small" variant="outlined" />
      ),
    },
    {
      field: "creatorId",
      headerName: "Creator",
      width: 150,
      align: "center",
      headerAlign: "center",
      renderCell: (params) => (
        <Chip icon={<AccountCircleIcon />} label={params.value} size="small" />
      ),
    },
    {
      field: "requestDate",
      headerName: "Submitted On",
      width: 200,
      align: "center",
      headerAlign: "center",
      renderCell: (params) => (
        <Box>
          {new Date(params.value).toLocaleString("en-IN", {
            dateStyle: "medium",
            timeStyle: "short",
          })}
        </Box>
      ),
    },
    {
      field: "commonBatchRemarks",
      headerName: "Creator Remarks",
      width: 200,
      align: "center",
      headerAlign: "center",
    },
    {
      field: "requestCount",
      headerName: "Entries",
      width: 100,
      align: "center",
      headerAlign: "center",
    },
    {
      field: "totalCredit",
      headerName: "Total Credit",
      width: 150,
      align: "right",
      headerAlign: "right",
      renderCell: (params) => (
        <Typography variant="body2" sx={{ mt: 1.5, fontWeight: 500 }}>
          {formatCurrency(params.value)}
        </Typography>
      ),
    },
    {
      field: "totalDebit",
      headerName: "Total Debit",
      width: 150,
      align: "right",
      headerAlign: "right",
      renderCell: (params) => (
        <Typography variant="body2" sx={{ mt: 1.5, fontWeight: 500 }}>
          {formatCurrency(params.value)}
        </Typography>
      ),
    },
    {
      field: "actions",
      headerName: "Actions",
      flex: 0.8,
      minWidth: 130,
      align: "center",
      headerAlign: "center",
      sortable: false,
      renderCell: (params) => {
        const isSelfCreated =
          currentUserId &&
          params.row.creatorId &&
          String(currentUserId) === String(params.row.creatorId);

        const selfCreateTooltip = "You cannot approve or reject your own request";

        return (
          <Stack direction="row" spacing={0.5} justifyContent="center">
            <Tooltip title="View Details">
              <IconButton
                onClick={() =>
                  handleOpenViewModal(params.row.batchId, params.row.creatorId)
                }
                color="primary"
              >
                <VisibilityIcon />
              </IconButton>
            </Tooltip>
            <Tooltip title={isSelfCreated ? selfCreateTooltip : "Accept Batch"}>
              <span>
                <IconButton
                  onClick={() => handleOpenModal(params.row, "ACCEPTED")}
                  color="success"
                  disabled={isSelfCreated}
                >
                  <CheckIcon />
                </IconButton>
              </span>
            </Tooltip>
            <Tooltip title={isSelfCreated ? selfCreateTooltip : "Reject Batch"}>
              <span>
                <IconButton
                  onClick={() => handleOpenModal(params.row, "REJECTED")}
                  color="error"
                  disabled={isSelfCreated}
                >
                  <ClearIcon />
                </IconButton>
              </span>
            </Tooltip>
          </Stack>
        );
      },
    },
  ];

  return (
    <Box sx={styles.mainContainer}>
      <DataGrid
        rows={rows}
        columns={columns}
        loading={loading}
        getRowId={(row) => row.batchId}
        disableRowSelectionOnClick
        pageSizeOptions={[5, 10, 25]}
        disableColumnResize={true}
        rowHeight={52}
        initialState={{
          pagination: { paginationModel: { page: 0, pageSize: 10 } },
          sorting: { sortModel: [{ field: "requestDate", sort: "desc" }] },
        }}
        slots={{ noRowsOverlay: CustomNoRowsOverlay }}
        sx={styles.dataGridContainer}
      />

      {/* --- BATCH ACTION MODAL --- */}
      <Dialog
        open={isModalOpen}
        onClose={handleCloseModal}
        fullWidth
        maxWidth="sm"
      >
        <DialogTitle>
          {modalMode === "ACCEPTED" ? "Approve" : "Reject"} Batch
        </DialogTitle>
        <DialogContent>
          <DialogContentText>
            Are you sure you want to {modalMode.toLowerCase()} the entire batch{" "}
            <strong>{selectedBatch?.batchId}</strong>?
            {modalMode === "REJECTED" && " Remarks are required for rejection."}
          </DialogContentText>

          <TextField
            autoFocus={modalMode === "REJECTED"}
            required={modalMode === "REJECTED"}
            margin="dense"
            id="executorRemarks"
            label="Approver Remarks"
            type="text"
            fullWidth
            variant="standard"
            value={executorRemarks}
            onChange={(e) => setExecutorRemarks(e.target.value)}
            disabled={isSubmitting}
            InputProps={{ inputProps: { maxLength: 200 } }}
          />
        </DialogContent>
        <DialogActions sx={styles.dialogActions}>
          <Stack direction="row" spacing={1} justifyContent="flex-end">
            <Button onClick={handleCloseModal} disabled={isSubmitting}>
              Cancel
            </Button>
            <Button
              onClick={handleSubmitAction}
              variant="contained"
              color={modalMode === "ACCEPTED" ? "success" : "error"}
              disabled={isSubmitting}
            >
              {isSubmitting ? (
                <CircularProgress size={24} color="inherit" />
              ) : (
                `Confirm ${modalMode === "ACCEPTED" ? "Approval" : "Rejection"}`
              )}
            </Button>
          </Stack>
        </DialogActions>
      </Dialog>

      {/* --- VIEW DETAILS MODAL --- */}
      <Dialog
        open={isViewModalOpen}
        onClose={handleCloseViewModal}
        fullWidth
        maxWidth="lg"
      >
        <DialogTitle sx={styles.dialogTitle}>
          <Typography variant="h6" component="div">
            Batch Details: {viewingBatchId}
          </Typography>
          <IconButton
            aria-label="close"
            onClick={handleCloseViewModal}
            sx={{ color: (theme) => theme.palette.grey[500] }}
          >
            <CloseIcon />
          </IconButton>
        </DialogTitle>

        <DialogContent dividers sx={{ p: 0 }}>
          {isViewLoading ? (
            <Box sx={styles.loadingContainer}>
              <CircularProgress />
            </Box>
          ) : (
            <>
              {batchDetails.length > 0 ? (
                <TableContainer component={Paper} sx={styles.tableContainer}>
                  <Table sx={{ minWidth: 650 }} size="small" stickyHeader>
                    <TableHead>
                      <TableRow>
                        {/* Sticky Header Column 1: Checkbox */}
                        <TableCell padding="checkbox" sx={styles.stickyHeaderColumn(0)}>
                          <Checkbox
                            color="primary"
                            indeterminate={
                              selectedJournalPrefixes.length > 0 &&
                              selectedJournalPrefixes.length < batchDetails.length
                            }
                            checked={
                              batchDetails.length > 0 &&
                              selectedJournalPrefixes.length === batchDetails.length
                            }
                            onChange={handleSelectAllClick}
                            disabled={isViewingSelfCreated}
                          />
                        </TableCell>
                        {/* Sticky Header Column 2: Journal ID */}
                        <TableCell align="center" sx={styles.stickyHeaderColumn(48)}>
                          Journal ID
                        </TableCell>
                        {/* Normal Columns */}
                        <TableCell align="center" sx={styles.tableHeaderCell}>Branch</TableCell>
                        <TableCell align="center" sx={styles.tableHeaderCell}>Currency</TableCell>
                        <TableCell align="center" sx={styles.tableHeaderCell}>CGL</TableCell>
                        <TableCell align="center" sx={styles.tableHeaderCell}>Amount</TableCell>
                        <TableCell align="center" sx={styles.tableHeaderCell}>Txn Type</TableCell>
                      </TableRow>
                    </TableHead>

                    <TableBody>
                      {batchDetails.map((group, groupIndex) => {
                        const isSelected = selectedJournalPrefixes.includes(
                          group[0].journalPrefix
                        );
                        const groupRowCount = group.length;

                        return group.map((row, rowIndex) => (
                          <TableRow
                            key={row.id}
                            hover={!isViewingSelfCreated}
                            onClick={() =>
                              !isViewingSelfCreated &&
                              handleRowCheckboxClick(null, row.journalPrefix)
                            }
                            role="checkbox"
                            aria-checked={isSelected}
                            selected={isSelected}
                            sx={{
                              cursor: isViewingSelfCreated ? "not-allowed" : "pointer",
                              // Border logic for grouping
                              ...(rowIndex === 0 && groupIndex > 0 && {
                                borderTop: (theme) => `2px solid ${theme.palette.divider}`,
                              }),
                            }}
                          >
                            {/* Sticky Body Column 1: Checkbox */}
                            {rowIndex === 0 && (
                              <TableCell
                                padding="checkbox"
                                rowSpan={groupRowCount}
                                sx={styles.stickyBodyColumn(0)}
                              >
                                <Checkbox
                                  color="primary"
                                  checked={isSelected}
                                  disabled={isViewingSelfCreated}
                                  onClick={(e) =>
                                    !isViewingSelfCreated &&
                                    handleRowCheckboxClick(e, row.journalPrefix)
                                  }
                                />
                              </TableCell>
                            )}
                            {/* Sticky Body Column 2: Journal ID */}
                            {rowIndex === 0 && (
                              <TableCell
                                align="center"
                                rowSpan={groupRowCount}
                                sx={styles.stickyBodyColumn(48)}
                              >
                                {row.journalPrefix}
                              </TableCell>
                            )}
                            {/* Normal Data Cells */}
                            <TableCell align="center">{row.payload.branch}</TableCell>
                            <TableCell align="center">{row.payload.currency}</TableCell>
                            <TableCell align="center">{row.payload.cgl}</TableCell>
                            <TableCell align="center">{formatCurrency(row.amount)}</TableCell>
                            <TableCell align="center">
                              <Chip
                                label={row.transactionType}
                                color={getTxnTypeColor(row.transactionType)}
                                size="small"
                              />
                            </TableCell>
                          </TableRow>
                        ));
                      })}
                    </TableBody>
                  </Table>
                </TableContainer>
              ) : (
                <Typography sx={{ p: 3 }}>
                  No pending details found for this batch.
                </Typography>
              )}
            </>
          )}
        </DialogContent>

        <DialogActions sx={styles.dialogActions}>
          <Grid container spacing={2} alignItems="center">
            <Grid item xs={12} md={6}>
              <TextField
                margin="dense"
                id="partialExecutorRemarks"
                label="Approver Remarks (for selected)"
                type="text"
                fullWidth
                size="small"
                value={partialExecutorRemarks}
                onChange={(e) => setPartialExecutorRemarks(e.target.value)}
                disabled={
                  isPartialSubmitting ||
                  selectedJournalPrefixes.length === 0 ||
                  isViewingSelfCreated
                }
                InputProps={{ inputProps: { maxLength: 200 } }}
              />
            </Grid>
            <Grid item xs={12} md={6}>
              <Stack direction="row" spacing={1} justifyContent="flex-end">
                <Button
                  variant="contained"
                  color="success"
                  onClick={() => handleSubmitPartialAction("ACCEPTED")}
                  disabled={
                    isPartialSubmitting ||
                    selectedJournalPrefixes.length === 0 ||
                    isViewingSelfCreated
                  }
                >
                  {isPartialSubmitting ? (
                    <CircularProgress size={24} color="inherit" />
                  ) : (
                    "Accept Selected"
                  )}
                </Button>
                <Button
                  variant="contained"
                  color="error"
                  onClick={() => handleSubmitPartialAction("REJECTED")}
                  disabled={
                    isPartialSubmitting ||
                    selectedJournalPrefixes.length === 0 ||
                    isViewingSelfCreated
                  }
                >
                  {isPartialSubmitting ? (
                    <CircularProgress size={24} color="inherit" />
                  ) : (
                    "Reject Selected"
                  )}
                </Button>
              </Stack>
            </Grid>
            {isViewingSelfCreated && (
              <Grid item xs={12}>
                <Typography
                  color="error"
                  variant="body2"
                  textAlign="center"
                  sx={{ pt: 1, width: "100%" }}
                >
                  You cannot approve or reject entries from your own request.
                </Typography>
              </Grid>
            )}
          </Grid>
        </DialogActions>
      </Dialog>
    </Box>
  );
}
