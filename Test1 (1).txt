package com.tcs.fincore.CommonRequestService.controller;

import com.tcs.fincore.CommonRequestService.dto.BranchDto;
import com.tcs.fincore.CommonRequestService.service.BranchService;
import lombok.RequiredArgsConstructor;
import org.springframework.web.bind.annotation.GetMapping;
import org.springframework.web.bind.annotation.RequestMapping;
import org.springframework.web.bind.annotation.RequestParam;
import org.springframework.web.bind.annotation.RestController;

import java.util.List;
import java.util.Map;

/**
 * REST controller for branch-related queries.
 */
@RestController
@RequestMapping("/api/branches")
@RequiredArgsConstructor
public class BranchController {

    private final BranchService branchService;

    /**
     * GET /api/branches/map
     * Retrieves all branches as a key-value map of { "branchCode": "branchName" }.
     *
     * @return A Map<String, String> containing all branches.
     */
    @GetMapping("/map")
    public Map<String, String> getBranchMap() {
        return branchService.getBranchCodeAndNameMap();
    }

    /**
     * GET /api/branches/search?query=...
     * Searches for branches by code or name.
     *
     * @param query The search term.
     * @return A list of branches matching the query.
     */
    @GetMapping("/search")
    public List<BranchDto> searchBranches(@RequestParam String query) {
        return branchService.searchBranches(query);
    }
}
