function CustomTabPanel(props) {
  const { children, value, index, ...other } = props;

  return (
    <Box
      role="tabpanel"
      hidden={value !== index}
      id={`simple-tabpanel-${index}`}
      aria-labelledby={`simple-tab-${index}`}
      {...other}
    >
      {value === index && (
        <Box sx={CurrencyMasterStyles.customTabPanelBox}>{children}</Box>
      )}
    </Box>
  );
}

CustomTabPanel.propTypes = {
  children: PropTypes.node,
  index: PropTypes.number.isRequired,
  value: PropTypes.number.isRequired,
};

function a11yProps(index) {
  return {
    id: `simple-tab-${index}`,
    "aria-controls": `simple-tabpanel-${index}`,
  };
}

import React, { useEffect, useMemo, useRef, useState } from "react";
import {
  Box,
  Button,
  Dialog,
  DialogContent,
  DialogTitle,
  Fab,
  Grid,
  IconButton,
  Stack,
  TextField,
  Typography,
  Tooltip,
  CircularProgress,
  DialogActions,
  Divider,
} from "@mui/material";
import { DataGrid } from "@mui/x-data-grid";
import {
  Edit as EditIcon,
  Add as AddIcon,
  Close as CloseIcon,
} from "@mui/icons-material";
import useApi from "../../hooks/useApi";
import useCustomSnackbar from "../../utils/useCustomSnackbar";

export default function CurrencyMasterTab() {
  const [rows, setRows] = useState([]);
  const [open, setOpen] = useState(false);
  const [editingId, setEditingId] = useState(null);
  const [loading, setLoading] = useState(true);
  const [originalData, setOriginalData] = useState(null); // New state for original data
  const [isDataSame, setIsDataSame] = useState(true);
  const snackbar = useCustomSnackbar();
  const { callApi } = useApi();

  const [currencyData, setCurrencyData] = useState({
    currencyCode: "",
    currencyName: "",
    currencyRate: "",
  });
  const [errors, setErrors] = useState({});

  const checkDataEquality = (current, original) => {
    if (!original) return true; // If no original data, assume they are "same" initially for disable state
    return (
      current.currencyName === original.currencyName &&
      current.currencyRate === original.currencyRate
      // currencyCode is excluded as it is the ID and likely read-only
    );
  };

  // --- debounce/snackbar control ---
  const snackGateRef = useRef(true);
  const notifyOnce = (msg, variant = "warning") => {
    if (!snackGateRef.current) return;
    snackGateRef.current = false;
    snackbar(msg, variant);
    setTimeout(() => (snackGateRef.current = true), 900);
  };

  // --- Fetch Data ---
  const fetchData = async () => {
    const currenciesResponse = await callApi(
      "/CM/currency-master",
      null,
      "GET"
    );
    setRows(currenciesResponse?.data ?? []);
    try {
      setLoading(true);
    } catch (err) {
      snackbar("Failed to fetch data", "error");
    } finally {
      setLoading(false);
    }
  };

  useEffect(() => {
    fetchData();
    if (editingId && originalData) {
      setIsDataSame(checkDataEquality(currencyData, originalData));
    }
  }, [currencyData, originalData, editingId]);

  // eslint-disable-next-line react-hooks/exhaustive-deps
  // }, []);

  // --- Validation Rules ---
  const reDigits = /^\d*\.?\d{0,4}$/;
  const reDesc = /^[A-Za-z\s]*$/;

  const validateField = (name, value) => {
    switch (name) {
      case "currencyCode":
        if (!value.trim()) return "Currency Code is required";
        if (!reDesc.test(value)) return "Only letters allowed";
        if (value.length !== 3) return "Must be 3 letters";
        break;

      case "currencyName":
        if (!value.trim()) return "Currency Name is required";
        if (!reDesc.test(value)) return "Invalid characters";
        break;

      case "currencyRate":
        if (!value.trim()) return "Currency Rate is required";
        if (!reDigits.test(value)) return "Only digits allowed";
        break;

      default:
        return "";
    }
    return "";
  };

  const isFormValid = () => {
    return (
      !validateField("currencyCode", currencyData.currencyCode) &&
        !validateField("CurrencyName", currencyData.currencyName) &&
        !validateField("CurrencyRate", currencyData.currencyRate),
      [currencyData]
    );
  };

  const handleChange = (e) => {
    const { name, value } = e.target;
    if (name === "currencyCode") {
      if (!reDesc.test(value)) {
        notifyOnce("Currency Code accepts letters only");
        return;
      }
      if (value.length > 3) {
        notifyOnce("Currency Code must be 3 letters");
        return;
      }
    } else if (name === "currencyName" && !reDesc.test(value)) {
      notifyOnce("Invalid character in Currency Name");
      return;
    } else if (name === "currencyRate" && !reDigits.test(value)) {
      notifyOnce("Currency Rate accepts only digits");
      return;
    }

    setCurrencyData((prev) => ({ ...prev, [name]: value }));
    setErrors((prev) => ({ ...prev, [name]: validateField(name, value) }));
  };

  const handleSubmit = async (e) => {
    e.preventDefault();

    if (editingId && isDataSame) {
      snackbar("Unable to update", "warning");
      return; // Stop submission
    }

    const isEditing = editingId;
    if (!isEditing) {
      const currencyExists = rows.some(
        (row) => row.currencyCode === currencyData.currencyCode
      );
      if (currencyExists) {
        snackbar("Currency code already exists", "warning");
        return;
      }
    }

    const allErrors = {
      currencyCode: validateField("CurrencyCode", currencyData.currencyCode),
      CurrencyName: validateField("CurrencyName", currencyData.currencyName),
      CurrencyRate: validateField("CurrencyRate", currencyData.currencyRate),
    };
    setErrors(allErrors);

    const valid = Object.values(allErrors).every((v) => !v);
    if (!valid) {
      notifyOnce("Please fix validation errors", "error");
      return;
    }

    let payloadData;
    if (isEditing) {
      payloadData = {
        requestType: "CURRENCY",
        changeType: "UPDATE",
        payload: {
          currencyCode: currencyData.currencyCode,
          currencyName: currencyData.currencyName,
          flag: 1,
          currencyRate: currencyData.currencyRate,
        },
      };
    } else {
      payloadData = {
        requestType: "CURRENCY",
        changeType: "ADD",
        payload: {
          currencyCode: currencyData.currencyCode,
          currencyName: currencyData.currencyName,
          flag: 0,
          currencyRate: currencyData.currencyRate,
        },
      };
    }

    try {
      const response = await callApi("/CR/create-request", payloadData, "POST");
      if (response) {
        snackbar(
          editingId ? "Currency updated" : "Currency created",
          "success"
        );
        handleClose();
        fetchData();
      }
    } catch (err) {
      snackbar(err?.message || "Request failed", "error");
    }
  };

  const handleEdit = (row) => {
    const initialData = {
      currencyCode: String(row.currencyCode ?? ""),
      currencyName: String(row.currencyName ?? ""),
      currencyRate: String(row.currencyRate ?? ""),
    };
    setCurrencyData(initialData);
    setOriginalData(initialData); // Store the original data here
    setEditingId(row.currencyCode);
    setOpen(true);
    setIsDataSame(true);
  };

  const handleClose = () => {
    setOpen(false);
    setCurrencyData({
      currencyCode: "",
      currencyName: "",
      currencyRate: "",
    });
    setOriginalData(null); // Clear original data on close
    setErrors({});
    setEditingId(null);
    setIsDataSame(true);
  };

  // --- Columns ---
  const columns = useMemo(
    () => [
      {
        field: "currencyCode",
        headerName: "Currency Code",
        flex: 1.2,
        sortable: true,
      },
      {
        field: "currencyName",
        headerName: "Currency Name",
        flex: 2,
        sortable: true,
      },
      {
        field: "currencyRate",
        headerName: "Currency Rate",
        flex: 2,
        sortable: true,
      },
      {
        field: "actions",
        headerName: "Actions",
        flex: 0.8,
        align: "center",
        headerAlign: "center",
        renderCell: (params) => (
          <Tooltip title="Edit">
            <IconButton
              color="primary"
              size="small"
              onClick={() => handleEdit(params.row)}
            >
              <EditIcon fontSize="small" />
            </IconButton>
          </Tooltip>
        ),
      },
    ],
    []
  );

  return (
    <Box sx={{ p: 2 }}>
      {loading ? (
        <Stack
          alignItems="center"
          justifyContent="center"
          sx={{ height: "60vh" }}
        >
          <CircularProgress size={40} />
        </Stack>
      ) : (
        <DataGrid
          rows={rows}
          columns={columns}
          disableRowSelectionOnClick
          getRowId={(row) => row.currencyCode}
          pageSizeOptions={[5, 10, 25, 50]}
          initialState={{
            pagination: { paginationModel: { pageSize: 10, page: 0 } },
            sorting: { sortModel: [{ field: "currencyCode", sort: "asc" }] },
          }}
          sx={{
            "& .MuiDataGrid-columnHeaders": { backgroundColor: "#f9f9f9" },
            border: "none",
            height: 500,
          }}
          localeText={{
            noRowsLabel: "No Currency Records Found",
          }}
        />
      )}

      {/* Floating Create Button */}
      <Fab
        onClick={() => setOpen(true)}
        color="primary"
        i
        variant="extended"
        sx={{
          position: "fixed",
          borderRadius: 1,
          fontSize: "15px",
          bottom: (theme) => theme.spacing(8),
          right: (theme) => theme.spacing(4),
          zIndex: 1200,
        }}
      >
        <AddIcon sx={{ mr: 1 }} />
        Create
      </Fab>

      {/* Dialog Form */}
      <Dialog open={open} onClose={handleClose} maxWidth="sm" fullWidth>
        <Box component="form" onSubmit={handleSubmit}>
          <DialogTitle>
            <Stack
              direction="row"
              justifyContent="space-between"
              alignItems="center"
            >
              <Typography variant="h6" color="primary">
                {editingId ? "Edit Currency" : "Add Currency"}
              </Typography>
              <IconButton onClick={handleClose}>
                <CloseIcon />
              </IconButton>
            </Stack>
          </DialogTitle>

          <DialogContent>
            <Grid container spacing={2} sx={{ mt: 1 }}>
              <Grid item xs={12}>
                <TextField
                  fullWidth
                  label="Currency Code"
                  name="currencyCode"
                  value={currencyData.currencyCode}
                  onChange={handleChange}
                  disabled={!!editingId}
                  error={!!errors.currencyCode}
                  helperText={errors.currencyCode || "Exactly 3 letters"}
                  inputProps={{ maxLength: 3, inputMode: "numeric" }}
                />
              </Grid>

              <Grid item xs={12}>
                <TextField
                  fullWidth
                  label="Currency Name"
                  name="currencyName"
                  value={currencyData.currencyName}
                  onChange={handleChange}
                  error={!!errors.currencyName}
                  inputProps={{ maxLength: 50 }}
                  helperText={errors.currencyName || "Allowed: letters only"}
                />
              </Grid>
              <Grid>
                <TextField
                  name="currencyRate"
                  label="Currency Rate"
                  fullWidth
                  value={currencyData.currencyRate}
                  onChange={handleChange}
                  error={!!errors.currencyRate}
                  helperText={errors.currencyRate}
                  inputProps={{ maxLength: 25 }}
                />
              </Grid>

              <Grid
                item
                xs={12}
                display="flex"
                justifyContent="flex-end"
              ></Grid>
            </Grid>
          </DialogContent>
          <Divider></Divider>
          <DialogActions>
            <Button
              type="submit"
              variant="contained"
              startIcon={editingId ? <EditIcon /> : <AddIcon />}
              disabled={!isFormValid()} // || (editingId && isDataSame)
            >
              {editingId ? "Update" : "Add"}
            </Button>
          </DialogActions>
        </Box>
      </Dialog>
    </Box>
  );
}

import { Box, Chip } from "@mui/material";
import { DataGrid } from "@mui/x-data-grid";
import React, { useEffect, useMemo, useState } from "react";

import useApi from "../../hooks/useApi";
import CheckCircleIcon from "@mui/icons-material/CheckCircle";
import BlockFlippedIcon from "@mui/icons-material/BlockFlipped";
import DateRangeIcon from "@mui/icons-material/DateRange";
import PendingActionsIcon from "@mui/icons-material/PendingActions";

export default function MyCurrencyRequestsTab() {
  const [rows, setRows] = useState([]);

  const { callApi } = useApi();

  const fetchData = async () => {
    const payload = { requestType: "CURRENCY" };
    const response = await callApi("/CR/my-requests", payload, "POST");
    if (response) {
      const result = response.data?.map((row) => {
        let parsedPayload = {};
        parsedPayload = row.payload ? row.payload : {};
        parsedPayload = {
          ...parsedPayload,
          id: row.id,
          reqStatus: row.reqStatus,
          reqDate: row.reqDate,
        };

        return {
          ...parsedPayload,
        };
      });
      setRows(result);
    } else {
      setRows([]);
    }
  };

  useEffect(() => {
    fetchData();
  }, []);

  const columns = useMemo(
    () => [
      {
        field: "currencyCode",
        headerName: "Currency Code",
        flex: 2,
        sortable: true,
        // valueGetter: (params) => params?.row?.payload?.code,
      },
      {
        field: "currencyName",
        headerName: "Currency Name",
        flex: 2,
        sortable: true,
        // valueGetter: (params) => params?.row?.payload?.name,
      },
      {
        field: "reqDate",
        headerName: "Date of Submission",
        flex: 1.5,
        sortable: true,
        align: "center",
        headerAlign: "center",
        renderCell: (params) => (
          <Box
            sx={
              {
                // fontWeight: "bold",
              }
            }
          >
            <Chip
              sx={{ color: "text.secondary" }}
              icon={<DateRangeIcon />}
              label={params.value.split("T")[0]}
            />
          </Box>
        ),
      },
      {
        field: "reqStatus",
        headerName: "Status",
        flex: 1,
        sortable: true,
        align: "center",
        headerAlign: "center",
        justifyContent: "center",
        renderCell: (params) => {
          const status = params.value;
          let color = "";
          let label = "";
          let IconComponent = null;

          if (status === "REJECTED") {
            color = "error";
            label = "Rejected";
            IconComponent = BlockFlippedIcon;
          } else if (status === "ACCEPTED") {
            color = "success";
            label = "Approved";
            IconComponent = CheckCircleIcon; // You'll need to import CheckCircleIcon
          } else {
            // Assuming "PENDING" or any other status
            color = "warning";
            label = "Pending";
            IconComponent = PendingActionsIcon;
          }

          return (
            <Box
              sx={{
                color: `${color}.main`,
              }}
            >
              <Chip
                icon={IconComponent ? <IconComponent /> : undefined}
                label={label}
                sx={{
                  p: 0.5,
                  m: 0,
                  width: "90%",
                  color: `${color}.dark`,
                  // backgroundColor: `${color}.light`, // Added light background for better visibility

                  "& .MuiChip-icon": {
                    color: `${color}.dark`,
                  },
                }}
              />
            </Box>
          );
        },
      },
    ],

    []
  );

  return (
    <Box>
      <DataGrid
        rows={rows}
        columns={columns}
        disableRowSelectionOnClick
        pageSizeOptions={[5, 10, 25, 50]}
        initialState={{
          pagination: { paginationModel: { pageSize: 10, page: 0 } },
          sorting: { sortModel: [{ field: "currencyCode", sort: "asc" }] },
        }}
        getRowId={(row) => row.currencyCode}
        sx={{ height: 400, width: "100%" }}
      />
    </Box>
  );
}
