package com.fincore.JournalService.Aspect;

import com.fasterxml.jackson.databind.ObjectMapper;
import com.fincore.JournalService.Dto.BatchRequestDto;
import com.fincore.JournalService.Dto.BulkProcessJournalRequestDto;
import com.fincore.JournalService.Dto.ProcessJournalRequestDto;

import com.fincore.JournalService.Models.JournalRequest;
import com.fincore.JournalService.Repository.JournalLogRepository;
import jakarta.servlet.http.HttpServletRequest;
import lombok.RequiredArgsConstructor;
import lombok.extern.slf4j.Slf4j;
import org.aspectj.lang.JoinPoint;
import org.aspectj.lang.annotation.AfterReturning;
import org.aspectj.lang.annotation.Aspect;
import org.springframework.stereotype.Component;
import org.springframework.web.context.request.RequestContextHolder;
import org.springframework.web.context.request.ServletRequestAttributes;

import java.time.LocalDateTime;

@Aspect
@Component
@Slf4j
@RequiredArgsConstructor
public class JournalActivityLogger {

    private final JournalLogRepository journalLogRepository;
    private final ObjectMapper objectMapper;

    // --- 1. Log Batch Creation (Standard) ---
    @AfterReturning(
            pointcut = "execution(* com.fincore.JournalService.Service.JournalRequestService.createBatchRequest(..)) && args(batchDto, creatorId, creatorRole)",
            returning = "result"
    )
    public void logCreateBatch(JoinPoint joinPoint, BatchRequestDto batchDto, String creatorId, Integer creatorRole, Object result) {
        String summary = "Created Batch with " + (batchDto.getRows() != null ? batchDto.getRows().size() : 0) + " rows.";
        // Passing null for requestId is correct here, but saveLog must handle it safely
        saveLog(creatorId, "CREATE", "BATCH_CREATION", summary, null, null);
    }

    // --- 2. Log Bulk Batch Creation (Optimized Upload) ---
    @AfterReturning(
            pointcut = "execution(* com.fincore.JournalService.Service.JournalRequestService.createBulkBatchRequest(..)) && args(batchDto, creatorId, creatorRole)",
            returning = "batchId"
    )
    public void logBulkCreate(JoinPoint joinPoint, BatchRequestDto batchDto, String creatorId, Integer creatorRole, Object batchId) {
        String summary = "Bulk Upload Created. Batch ID: " + batchId + ". Rows: " + (batchDto.getRows() != null ? batchDto.getRows().size() : 0);
        saveLog(creatorId, "CREATE", "BULK_UPLOAD", summary, null, null);
    }

    // --- 3. Log Single Status Update (Approve/Reject) ---
    @AfterReturning(
            pointcut = "execution(* com.fincore.JournalService.Service.JournalRequestService.updateRequestStatus(..)) && args(dto, executorId, executorRole)",
            returning = "result"
    )
    public void logStatusUpdate(JoinPoint joinPoint, ProcessJournalRequestDto dto, String executorId, Integer executorRole, Object result) {
        if (result instanceof JournalRequest) {
            JournalRequest req = (JournalRequest) result; // The updated entity
            String newValue = "Status: " + dto.getStatus() + ", Remarks: " + dto.getRemarks();
            // We can treat 'changeType' as the Status (ACCEPTED/REJECTED)
            saveLog(executorId, dto.getStatus().toString(), "STATUS_UPDATE", newValue, null, req.getId());
        }
    }

    // --- 4. Log Bulk Status Update (Bulk Approve/Reject) ---
    @AfterReturning(
            pointcut = "execution(* com.fincore.JournalService.Service.JournalRequestService.processBulkRequests(..)) && args(dto, executorId, executorRole)",
            returning = "result"
    )
    public void logBulkStatusUpdate(JoinPoint joinPoint, BulkProcessJournalRequestDto dto, String executorId, Integer executorRole, Object result) {
        String target = dto.getBatchId() != null ? "Batch ID: " + dto.getBatchId() : "Journals: " + dto.getJournalIdPrefixes();
        String newValue = "Bulk " + dto.getStatus() + " for " + target + ". Remarks: " + dto.getRemarks();
        
        saveLog(executorId, "BULK_" + dto.getStatus(), "BULK_PROCESS", newValue, null, null);
    }

    // --- 5. Log Single Cancellation ---
    @AfterReturning(
            pointcut = "execution(* com.fincore.JournalService.Service.JournalRequestService.cancelMyRequest(..)) && args(requestId, userId)"
    )
    public void logCancelRequest(JoinPoint joinPoint, Long requestId, String userId) {
        saveLog(userId, "DELETE", "CANCEL_REQUEST", "User canceled request ID: " + requestId, null, requestId);
    }

    // --- 6. Log Bulk Cancellation (By Batch) ---
    @AfterReturning(
            pointcut = "execution(* com.fincore.JournalService.Service.JournalRequestService.cancelMyRequestsByBatchId(..)) && args(batchId, userId)"
    )
    public void logCancelBatch(JoinPoint joinPoint, String batchId, String userId) {
        saveLog(userId, "DELETE", "CANCEL_BATCH", "User canceled all requests in Batch: " + batchId, null, null);
    }

    // --- Helper to Save to DB ---
    private void saveLog(String userId, String actionType, String changeType, String newValue, String oldValue, Long requestId) {
        try {
            JournalLog logEntry = new JournalLog();
            
            logEntry.setUserId(userId);
            
            logEntry.setActionType(actionType);
            logEntry.setChangeType(changeType);
            logEntry.setActionTime(LocalDateTime.now());
            logEntry.setRequestId(requestId);
            
            // Capture IP Address
            logEntry.setIpAddress(getClientIp());

            // Convert Objects to JSON String if they aren't already strings
            if (newValue != null && !newValue.getClass().equals(String.class)) {
                newValue = objectMapper.writeValueAsString(newValue);
            }

            logEntry.setNewValue(truncate(newValue, 250));
            logEntry.setOldValue(truncate(oldValue, 250));



            journalLogRepository.save(logEntry);
            log.info("✅ Audit Log Saved: {} performed {} on {}", userId, actionType, (requestId != null ? requestId : "Batch"));

        } catch (Exception e) {
            log.error("Failed to save audit log: {}", e.getMessage());
        }
    }

    private String getClientIp() {
        try {
            HttpServletRequest request = ((ServletRequestAttributes) RequestContextHolder.currentRequestAttributes()).getRequest();
            String xForwardedFor = request.getHeader("X-Forwarded-For");
            if (xForwardedFor != null && !xForwardedFor.isEmpty()) {
                return xForwardedFor.split(",")[0].trim();
            }
            return request.getRemoteAddr();
        } catch (Exception e) {
            return "UNKNOWN";
        }
    }

    private String truncate(String value, int length) {
        if (value != null && value.length() > length) {
            return value.substring(0, length - 3) + "...";
        }
        return value;
    }
}

package com.fincore.JournalService.config;

import jakarta.persistence.AttributeConverter;
import jakarta.persistence.Converter;
import java.util.stream.Stream;

import com.fincore.JournalService.Models.enums.RequestStatus;

@Converter(autoApply = true)
public class RequestStatusConverter implements AttributeConverter<RequestStatus, String> {

    @Override
    public String convertToDatabaseColumn(RequestStatus status) {
        if (status == null) {
            return null;
        }
        return status.getCode(); // Converts PENDING to "P"
    }

    @Override
    public RequestStatus convertToEntityAttribute(String code) {
        if (code == null) {
            return null;
        }
        // Finds the Enum constant that matches the code from the DB
        return Stream.of(RequestStatus.values())
                .filter(c -> c.getCode().equals(code))
                .findFirst()
                .orElseThrow(IllegalArgumentException::new);
    }
}

import org.springframework.security.config.annotation.web.builders.HttpSecurity;
import org.springframework.security.config.annotation.web.configuration.EnableWebSecurity;
import org.springframework.security.config.http.SessionCreationPolicy;
import org.springframework.security.web.SecurityFilterChain;
import org.springframework.security.web.authentication.UsernamePasswordAuthenticationFilter;

/**
 * Global security configuration for the application.
 * This class uses Spring Security's Java configuration to define security rules,
 * such as disabling CSRF, making the session stateless, and defining authorization policies.
 *
 * The `@Configuration` annotation marks this class as a source of bean definitions.
 * The `@EnableWebSecurity` annotation enables Spring Security's web security support and provides
 * the integration with the Spring MVC.
 */
@Configuration
@EnableWebSecurity
@Import({RedisConfig.class, JwtUtil.class}) // Import Redis Config from Library
public class SecurityConfig {

    @Autowired
    private StringRedisTemplate redisTemplate;

    @Autowired
    private JwtUtil jwtUtil;

    @Bean
    public SecurityFilterChain securityFilterChain(HttpSecurity http) throws Exception {
        http
                .csrf(csrf -> csrf.disable())
                .cors(cors -> {}) // Enable CORS (uses your existing WebConfig)
                .sessionManagement(s -> s.sessionCreationPolicy(SessionCreationPolicy.STATELESS))

                // LOCK DOWN ENDPOINTS
                .authorizeHttpRequests(authz -> authz
                        .requestMatchers("/**").permitAll() // Allows access to all your API endpoints
                        .requestMatchers("/actuator/**", "/auth/**", "/error").permitAll()
                        // All other requests MUST pass the ContextRbacFilter
                        .anyRequest().authenticated()
                )

                // ======== INSERTING THE GUARD HERE =======
                // This filter does two things:
                // 1. Validates JWT & Sets User Context
                // 2. Checks Redis for Permissions (e.g. "POST:/create-request|SEGMENT_CODE")
                .addFilterBefore(new ContextRbacFilter(redisTemplate, jwtUtil), UsernamePasswordAuthenticationFilter.class);

        return http.build();
    }
}

	private final JwtUtil jwtUtil;

	@GetMapping("/current-posting-date")
	public String getCurrentPostingDate() {
		return LocalDate.now().format(DateTimeFormatter.ISO_LOCAL_DATE);
	}

	@PostMapping(value = "/bulk-validate", consumes = MediaType.MULTIPART_FORM_DATA_VALUE)
	public ResponseEntity<?> validateBulkUpload(@RequestParam("file") MultipartFile file,
			@RequestParam("postingDate") String postingDateStr) throws IOException {

		LocalDate postingDate = LocalDate.parse(postingDateStr);

		Map<String, Object> result = journalBulkValidationService.validateAndProcessFile(file, postingDate);

		byte[] fileContent = (byte[]) result.get("file");
		String base64File = (fileContent != null) ? Base64.getEncoder().encodeToString(fileContent) : null;

		if ("ERROR".equals(result.get("status"))) {
			long count = (long) result.get("errorCount");

			// Use HashMap to allow potential nulls (though unlikely here)
			Map<String, Object> response = new HashMap<>();
			response.put("status", "ERROR");
			response.put("message", "Validation failed. Found " + count + " errors.");
			response.put("errorCount", count);
			response.put("errorReportBase64", base64File);

			return ResponseEntity.ok(response);
		}

		Map<String, Object> response = new HashMap<>();
		response.put("status", "SUCCESS");
		response.put("message", "Validation Successful!");
		response.put("successFileBase64", base64File); 
		response.put("previewData", result.get("previewData"));

		return ResponseEntity.ok(response);
	}

@Getter
public enum ChangeType {
	    ADD("A"),
	    UPDATE("U"),
	    DELETE("D");

@Service
@RequiredArgsConstructor
@Slf4j
public class JournalBulkValidationService {

    private final BranchMasterRepository branchRepo;
    private final CglMasterRepository cglRepo;
    private final CurrencyMasterRepository currencyRepo;

    // Compiled Pattern for cleanup (Performance)
    private static final Pattern CLEAN_AMOUNT_REGEX = Pattern.compile("[^0-9.-]");

    private static class ExcelRowData {
        int rowIndex;
        String branch = "";
        String currency = "";
        String cgl = "";
        BigDecimal amount; 
        String txnType = "";
        String remarks = "";
        String productCode = "";
        List<String> errors = new ArrayList<>();
    }

    public Map<String, Object> validateAndProcessFile(MultipartFile file, LocalDate postingDate) throws IOException {
        long startTime = System.currentTimeMillis();
        String filename = file.getOriginalFilename();
        List<ExcelRowData> parsedRows;

        // 1. PARSE FILE TO MEMORY
        if (filename != null && (filename.toLowerCase().endsWith(".csv") || filename.toLowerCase().endsWith(".txt"))) {
            parsedRows = parseCsvFile(file);
        } else {
            parsedRows = parseExcelFile(file);
        }

        // 2. VALIDATE LOGIC
        Map<String, Object> result = validateRows(parsedRows, postingDate, filename != null && filename.toLowerCase().endsWith(".csv"));
        
        log.info("File processed in {} ms", (System.currentTimeMillis() - startTime));
        return result;
    }

    // --- SMART VALIDATION (FETCH ONLY WHAT IS NEEDED) ---
    private Map<String, Object> validateRows(List<ExcelRowData> parsedRows, LocalDate postingDate, boolean isInputCsv) throws IOException {
        
        // A. Extract Unique Codes from the uploaded file
        Set<String> branchesInFile = parsedRows.stream()
                .map(r -> r.branch).filter(s -> !s.isEmpty()).collect(Collectors.toSet());
        Set<String> cglsInFile = parsedRows.stream()
                .map(r -> r.cgl).filter(s -> !s.isEmpty()).collect(Collectors.toSet());
        Set<String> currenciesInFile = parsedRows.stream()
                .map(r -> r.currency).filter(s -> !s.isEmpty()).collect(Collectors.toSet());

        // B. Fetch ONLY relevant data from DB (Super Fast)
        // Note: Assuming Entity IDs are Strings matching the codes. 
        // If IDs are different, use findByCodeIn(Set codes) repository method.
        Set<String> validBranches = branchRepo.findAllById(branchesInFile).stream()
                .map(BranchMaster::getCode).map(String::trim).collect(Collectors.toSet());
        
        Set<String> validCgls = cglRepo.findAllById(cglsInFile).stream()
                .map(CglMaster::getCglNumber).map(String::trim).collect(Collectors.toSet());
        
        Set<String> validCurrencies = currencyRepo.findAllById(currenciesInFile).stream()
                .map(CurrencyMaster::getCurrencyCode).map(String::trim).collect(Collectors.toSet());

        // Key Format: "BRANCH_CURRENCY_TYPE" (Type = MEMO or NORMAL)
        Map<String, BigDecimal> balanceMap = new HashMap<>();

        for (ExcelRowData data : parsedRows) {
            
            // 1. Validate Masters (InMemory Check)
            if (!validBranches.contains(data.branch)) data.errors.add("Invalid Branch Code");
            if (!validCurrencies.contains(data.currency)) data.errors.add("Invalid Currency");
            if (!validCgls.contains(data.cgl)) data.errors.add("Invalid CGL");

            if (data.productCode != null && !data.productCode.isEmpty()) {
                if (!data.productCode.matches("\\d{8}")) data.errors.add("Product Code must be 8 digits");
            }

            if (data.amount == null || data.amount.compareTo(BigDecimal.ZERO) == 0) {
                data.errors.add("Amount cannot be Zero");
            }

            if (data.remarks != null && data.remarks.length() > 30) data.errors.add("Remarks > 30 chars");

            // 2. Accumulate for Balancing (Separating MEMO vs NORMAL)
            if (validBranches.contains(data.branch) && validCurrencies.contains(data.currency) && data.amount != null) {
                
                String cglType = "NORMAL";
                if (data.cgl != null && data.cgl.startsWith("5")) {
                    cglType = "MEMO";
                }

                // Bucket: Branch + Currency + Type
                String key = data.branch + "_" + data.currency + "_" + cglType;
                
                balanceMap.put(key, balanceMap.getOrDefault(key, BigDecimal.ZERO).add(data.amount));
            }
        }

        // 3. Check Balancing
        for (ExcelRowData data : parsedRows) {
            if (!validBranches.contains(data.branch) || !validCurrencies.contains(data.currency)) continue;

            String cglType = (data.cgl != null && data.cgl.startsWith("5")) ? "MEMO" : "NORMAL";
            String key = data.branch + "_" + data.currency + "_" + cglType;
            
            BigDecimal netTotal = balanceMap.getOrDefault(key, BigDecimal.ZERO);

            if (netTotal.setScale(4, RoundingMode.HALF_UP).compareTo(BigDecimal.ZERO) != 0) {
                data.errors.add(cglType + " CGL Total Mismatch (Diff: " + netTotal.toPlainString() + ")");
            }
        }

        long errorCount = parsedRows.stream().filter(r -> !r.errors.isEmpty()).count();
        Map<String, Object> result = new HashMap<>();

        if (errorCount > 0) {
            byte[] errorFile = generateErrorExcel(parsedRows);
            result.put("status", "ERROR");
            result.put("errorCount", errorCount);
            result.put("file", errorFile);
            result.put("message", "Validation failed. Found " + errorCount + " errors.");
        } else {
            // OPTIMIZATION: Limit preview to 100 rows to prevent Frontend lag on massive files
            List<Map<String, Object>> previewList = parsedRows.stream().limit(100).map(row -> {
                Map<String, Object> map = new HashMap<>();
                map.put("id", row.rowIndex);
                map.put("branch", row.branch);
                map.put("currency", row.currency);
                map.put("cgl", row.cgl);
                map.put("amount", row.amount.abs().toPlainString()); 
                map.put("txnType", row.amount.compareTo(BigDecimal.ZERO) < 0 ? "Credit" : "Debit");
                map.put("remarks", row.remarks);
                map.put("productCode", row.productCode);
                return map;
            }).collect(Collectors.toList());

            result.put("status", "SUCCESS");
            result.put("previewData", previewList);
            result.put("totalRows", parsedRows.size());
            result.put("message", "Validation Successful!");
            
            if (!isInputCsv) {
                result.put("file", generateSuccessCsv(parsedRows, postingDate));
            }
        }
        return result;
    }

    private List<ExcelRowData> parseExcelFile(MultipartFile file) throws IOException {
        List<ExcelRowData> rows = new ArrayList<>();
        // Use try-with-resources
        try (Workbook workbook = new XSSFWorkbook(file.getInputStream())) {
            Sheet sheet = workbook.getSheetAt(0);
            DataFormatter formatter = new DataFormatter();

            for (Row row : sheet) {
                if (isRowEmpty(row)) continue;

                ExcelRowData data = new ExcelRowData();
                data.rowIndex = row.getRowNum();

                // Reading values
                data.branch = parseCode(row.getCell(0), formatter);     
                data.currency = getCellValueAsString(row.getCell(1), formatter).toUpperCase().trim(); 
                data.cgl = parseCode(row.getCell(2), formatter);        
                
                String amountRaw = getCellValueAsString(row.getCell(3), formatter).trim(); 
                String typeRaw = getCellValueAsString(row.getCell(4), formatter).trim().toUpperCase(); 
                data.remarks = getCellValueAsString(row.getCell(5), formatter);
                data.productCode = parseCode(row.getCell(6), formatter); 
                data.txnType = typeRaw; 

                parseAmountAndType(data, amountRaw, typeRaw);
                rows.add(data);
            }
        }
        return rows;
    }

    private List<ExcelRowData> parseCsvFile(MultipartFile file) throws IOException {
        List<ExcelRowData> rows = new ArrayList<>();
        try (BufferedReader reader = new BufferedReader(new InputStreamReader(file.getInputStream(), StandardCharsets.UTF_8))) {
            String line;
            int lineNum = 0;
            while ((line = reader.readLine()) != null) {
                if (line.trim().isEmpty()) continue;
                String[] cols = line.split(",", -1); 
                if (cols.length < 9) continue; 

                ExcelRowData data = new ExcelRowData();
                data.rowIndex = lineNum++;

                data.branch = cols[4].trim();
                data.currency = cols[5].trim();
                data.cgl = cols[6].trim();
                String amountRaw = cols[7].trim();
                String typeRaw = cols[8].trim();
                data.remarks = (cols.length > 9) ? cols[9].trim() : "";
                if (cols.length > 10) {
                    String prod = cols[10].trim();
                    data.productCode = "A".equalsIgnoreCase(prod) ? "" : prod;
                }
                data.txnType = typeRaw;
                parseAmountAndType(data, amountRaw, typeRaw);
                rows.add(data);
            }
        }
        return rows;
    }

    private void parseAmountAndType(ExcelRowData data, String amountRaw, String typeRaw) {
        try {
            // Fast regex replacement
            String cleanAmountStr = CLEAN_AMOUNT_REGEX.matcher(amountRaw).replaceAll("");
            if (cleanAmountStr.isEmpty()) throw new NumberFormatException("Empty");
            
            BigDecimal amt = new BigDecimal(cleanAmountStr);
            boolean isCredit = false;

            if (typeRaw.startsWith("C") || typeRaw.contains("CR")) isCredit = true;
            if (amountRaw.toUpperCase().contains("CR")) isCredit = true;
            if (amountRaw.toUpperCase().contains("DR")) isCredit = false;
            if (amountRaw.startsWith("-")) isCredit = true;

            data.amount = isCredit ? amt.abs().negate() : amt.abs();
        } catch (Exception e) {
            data.amount = BigDecimal.ZERO; 
        }
    }

    private byte[] generateSuccessCsv(List<ExcelRowData> rows, LocalDate postingDate) {
        // Estimate size: rows * 100 bytes avg
        StringBuilder csv = new StringBuilder(rows.size() * 100);
        DateTimeFormatter ddMMyyyy = DateTimeFormatter.ofPattern("ddMMyyyy");
        String col2 = postingDate.format(ddMMyyyy);

        int month = postingDate.getMonthValue();
        int year = postingDate.getYear();
        String col3 = (month >= 4) ? String.valueOf(year) : String.valueOf(year - 1);
        String col4 = (month >= 4) ? String.format("%02d", month - 3) : String.format("%02d", month + 9);

        // Pre-calculate fixed columns
        String col1 = "01";
        String col12 = "B";
        String col13 = "C";
        String col14 = "D";

        for (ExcelRowData row : rows) {
            String col5 = String.format("%5s", row.branch).replace(' ', '0');
            String col6 = (row.currency == null || row.currency.isEmpty()) ? "INR" : row.currency;
            
            BigDecimal absAmt = row.amount.abs();
            String plainAmt = absAmt.setScale(4, RoundingMode.HALF_UP).toPlainString();
            int dotIndex = plainAmt.indexOf('.');
            String intPart = (dotIndex == -1) ? plainAmt : plainAmt.substring(0, dotIndex);
            String decPart = (dotIndex == -1) ? "0000" : plainAmt.substring(dotIndex + 1);
            String col8 = String.format("%16s", intPart).replace(' ', '0') + "." + decPart;

            String col9 = (row.amount.compareTo(BigDecimal.ZERO) < 0) ? "Cr" : "Dr";
            String col10 = (row.remarks != null) ? row.remarks.replace(",", " ") : "";
            String col11 = (row.productCode != null && !row.productCode.isEmpty()) ? row.productCode : "A";

            csv.append(col1).append(",")
               .append(col2).append(",")
               .append(col3).append(",")
               .append(col4).append(",")
               .append(col5).append(",")
               .append(col6).append(",")
               .append(row.cgl).append(",")
               .append(col8).append(",")
               .append(col9).append(",")
               .append(col10).append(",")
               .append(col11).append(",")
               .append(col12).append(",")
               .append(col13).append(",")
               .append(col14).append("\n");
        }
        return csv.toString().getBytes();
    }

    private byte[] generateErrorExcel(List<ExcelRowData> rows) throws IOException {
        // Keep existing Error Excel generation logic...
        // (Just returning the standard byte array as before)
        Workbook workbook = new XSSFWorkbook();
        Sheet sheet = workbook.createSheet("Error Report");
        CellStyle headerStyle = workbook.createCellStyle();
        Font font = workbook.createFont();
        font.setBold(true);
        font.setColor(IndexedColors.RED.getIndex());
        headerStyle.setFont(font);

        Row header = sheet.createRow(0);
        String[] headers = {"Branch", "Currency", "CGL", "Amount", "TxnType", "Remarks", "Product Code", "ERRORS"};
        for (int i = 0; i < headers.length; i++) {
            Cell cell = header.createCell(i);
            cell.setCellValue(headers[i]);
            cell.setCellStyle(headerStyle);
        }

        int rowIdx = 1;
        for (ExcelRowData data : rows) {
            Row row = sheet.createRow(rowIdx++);
            row.createCell(0).setCellValue(data.branch);
            row.createCell(1).setCellValue(data.currency);
            row.createCell(2).setCellValue(data.cgl);
            row.createCell(3).setCellValue(data.amount != null ? data.amount.abs().toPlainString() : "0");
            row.createCell(4).setCellValue(data.txnType);
            row.createCell(5).setCellValue(data.remarks);
            row.createCell(6).setCellValue(data.productCode != null ? data.productCode : "");

            if (!data.errors.isEmpty()) {
                Cell errorCell = row.createCell(7);
                errorCell.setCellValue(String.join("; ", data.errors));
                errorCell.setCellStyle(headerStyle);
            }
        }
        for(int i=0; i<8; i++) sheet.autoSizeColumn(i);
        ByteArrayOutputStream out = new ByteArrayOutputStream();
        workbook.write(out);
        workbook.close();
        return out.toByteArray();
    }

    private String parseCode(Cell cell, DataFormatter formatter) {
        if (cell == null) return "";
        String val = getCellValueAsString(cell, formatter).trim();
        if (val.endsWith(".0")) return val.substring(0, val.length() - 2);
        return val;
    }

    private String getCellValueAsString(Cell cell, DataFormatter formatter) {
        if (cell == null) return "";
        if (cell.getCellType() == CellType.STRING) return cell.getStringCellValue().trim();
        if (cell.getCellType() == CellType.NUMERIC) {
            if (DateUtil.isCellDateFormatted(cell)) return cell.getDateCellValue().toString();
            return BigDecimal.valueOf(cell.getNumericCellValue()).toPlainString();
        }
        if (cell.getCellType() == CellType.BOOLEAN) return String.valueOf(cell.getBooleanCellValue());
        return "";
    }

    private boolean isRowEmpty(Row row) {
        if (row == null) return true;
        for (int c = row.getFirstCellNum(); c < row.getLastCellNum(); c++) {
            Cell cell = row.getCell(c);
            if (cell != null && cell.getCellType() != CellType.BLANK && !getCellValueAsString(cell, new DataFormatter()).trim().isEmpty())
                return false;
        }
        return true;
    }
}

    @Override
    @Transactional
    public List<JournalRequest> createBatchRequest(BatchRequestDto batchDto, String creatorId, Integer creatorRole) throws JsonProcessingException {
        
        String batchId = sequenceService.getNextBatchId();
        log.info("Creating new Journal batch with Batch ID: {}", batchId);
       
        LocalDate postingDate = LocalDate.now();
        
        List<JournalRequest> requestsToSave = new ArrayList<>();
        String commonRemarks = batchDto.getCommonBatchRemarks();

        Map<String, List<BatchRequestDto.JournalRequestRow>> groupedRows = batchDto.getRows().stream()
                .collect(Collectors.groupingBy(row -> row.getBranch() + ":" + row.getCurrency()));
        
        log.info("Batch {} contains {} distinct Branch-Currency groups.", batchId, groupedRows.size());

        for (Map.Entry<String, List<BatchRequestDto.JournalRequestRow>> entry : groupedRows.entrySet()) {
            
            String journalIdPrefix = sequenceService.getNextJournalIdPrefix();
            int journalSequence = 1;
            List<BatchRequestDto.JournalRequestRow> rowsInGroup = entry.getValue();
            
            for (BatchRequestDto.JournalRequestRow rowDto : rowsInGroup) {
                
                String journalId = String.format("%s-%03d", journalIdPrefix, journalSequence);
                
                CreateJournalRequestDto payloadDto = new CreateJournalRequestDto();
                
                payloadDto.setPDate(postingDate); 
                payloadDto.setChangeType(rowDto.getChangeType());
                payloadDto.setMasterJournalId(rowDto.getMasterJournalId());
                payloadDto.setBranch(rowDto.getBranch());
                payloadDto.setCurrency(rowDto.getCurrency());
                payloadDto.setCgl(rowDto.getCgl());
                payloadDto.setAmount(rowDto.getAmount()); 
                payloadDto.setProductType(rowDto.getProductType());
                payloadDto.setRemarks(rowDto.getRemarks());
                payloadDto.setArFlag(rowDto.getArFlag());
                payloadDto.setAcClassification(rowDto.getAcClassification());
                payloadDto.setBatchId(batchId); 
                payloadDto.setJournalId(journalId); 
                payloadDto.setCommonBatchRemarks(commonRemarks);
                payloadDto.setTransactionCount(journalSequence); 
                
                JournalRequest request = new JournalRequest();
                request.setCreatorId(creatorId);
                request.setCreatorRole(creatorRole);
                request.setChangeType(rowDto.getChangeType());
                request.setRequestStatus(RequestStatus.PENDING);
                request.setBatchId(batchId);
                request.setJournalId(journalId);
                request.setCommonBatchRemarks(commonRemarks);
                request.setPayload(objectMapper.writeValueAsString(payloadDto));

                requestsToSave.add(request);

                journalSequence++; 
            }
        }
        log.info("Saving {} JournalRequest entries for Batch ID: {}", requestsToSave.size(), batchId);
        List<JournalRequest> savedRequests = journalRequestRepository.saveAll(requestsToSave);
        
        try {
            NotificationConfigDto config = permissionConfigService.getConfig(JOURNAL_AUTH_TYPE);
            String message = String.format("New Journal request (Batch: %s) by user %s is pending approval.", batchId, creatorId);

            notificationWriterService.createNotification(
                null,                   
                config.getTargetRoles(),
                message,
                config.getTargetUrl(),  
                batchId,                
                EVENT_SOURCE
            );
            log.info("Successfully created 1-to-many notification for Batch ID: {}", batchId);

        } catch (Exception e) {
            log.error("CRITICAL: Failed to create 'pending' notification for Batch ID: {}. Rolling back transaction.", batchId, e);
            throw new RuntimeException("Failed to create notification event, rolling back batch creation.", e);
        }
   
        return savedRequests;
    }

    @Override
    @Transactional
    public Optional<JournalRequest> updateRequestStatus(ProcessJournalRequestDto dto, String executorId, Integer executorRole) throws JsonProcessingException {
        log.debug("Updating single request: {}", dto.getRequestId());
        JournalRequest request = journalRequestRepository.findById(dto.getRequestId())
                .orElseThrow(() -> new ResourceNotFoundException("Journal Request with ID " + dto.getRequestId() + " not found."));

        boolean isCreator = executorId.equals(request.getCreatorId());
        boolean isSameRole = executorRole != null && executorRole.equals(request.getCreatorRole());

        if (isCreator && isSameRole) {
            log.warn("SECURITY VIOLATION: User {} (Role: {}) attempted to approve/reject their own request (ID: {}).", 
                     executorId, executorRole, request.getId());
            throw new IllegalStateException("User cannot approve or reject their own request with the same role.");
        }

        if (request.getRequestStatus() != RequestStatus.PENDING) {
            throw new IllegalStateException("Journal Request has already been processed.");
        }

        request.setExecutorId(executorId);
        request.setExecutorRemarks(dto.getRemarks());
        request.setExecutionDate(LocalDateTime.now());
        request.setRequestStatus(dto.getStatus());

        if (RequestStatus.ACCEPTED.equals(dto.getStatus())) {
            log.info("Journal Request {} ACCEPTED. Processing for Batch ID: {}", dto.getRequestId(), request.getBatchId());
            processApproval(request); 
            sendCreatorNotification(request, true);
        } else {
            log.info("Journal Request {} REJECTED.", dto.getRequestId());
            sendCreatorNotification(request, false);
        }

        return Optional.of(journalRequestRepository.save(request));
    }
    
    private void processApproval(JournalRequest request) throws JsonProcessingException {
        CreateJournalRequestDto payloadDto = objectMapper.readValue(request.getPayload(), CreateJournalRequestDto.class);
        
        if (payloadDto.getChangeType() == ChangeType.ADD) {
       
            GlTransaction newGlTxn = new GlTransaction();
            newGlTxn.setBatchId(payloadDto.getBatchId());
            newGlTxn.setJournalId(payloadDto.getJournalId());
            
            
            if (payloadDto.getPDate() != null) {
                newGlTxn.setPostDate(payloadDto.getPDate().atStartOfDay());
                newGlTxn.setTransactionDate(payloadDto.getPDate()); // Set Date explicitly
            } else {
                newGlTxn.setPostDate(LocalDateTime.now());
                newGlTxn.setTransactionDate(LocalDate.now());
            }
            
            newGlTxn.setBranchCode(payloadDto.getBranch());
            newGlTxn.setCurrency(payloadDto.getCurrency());
            newGlTxn.setCgl(payloadDto.getCgl());
            newGlTxn.setNarration(payloadDto.getRemarks());
            
            BigDecimal amount = payloadDto.getAmount();
            if (amount.signum() < 0) { 
                newGlTxn.setCreditAmount(amount);
                newGlTxn.setDebitAmount(BigDecimal.ZERO);
            } else { 
                newGlTxn.setDebitAmount(amount);
                newGlTxn.setCreditAmount(BigDecimal.ZERO);
            }
            
            newGlTxn.setTransactionCount(payloadDto.getTransactionCount());
            newGlTxn.setSourceFlag("J"); 
            
            glTransactionRepository.save(newGlTxn);
        } 
    }

    @Override
    public List<JournalRequest> getPendingRequests(String userId, Integer userRole) {
        List<JournalRequest> allPending = journalRequestRepository.findByRequestStatus(RequestStatus.PENDING);

    @Override
    @Transactional
    public List<JournalRequest> processBulkRequests(BulkProcessJournalRequestDto dto, String executorId, Integer executorRole) throws JsonProcessingException {
        log.info("Starting processBulkRequests for user: {}", executorId);

        boolean hasBatchId = StringUtils.hasText(dto.getBatchId());
        boolean hasJournalPrefixes = dto.getJournalIdPrefixes() != null && !dto.getJournalIdPrefixes().isEmpty();

        if (!hasBatchId && !hasJournalPrefixes) {
            throw new IllegalArgumentException("Either batchId or a list of journalIdPrefixes must be provided.");
        }

        List<JournalRequest> requestsToProcess;
        if (hasBatchId) {
            requestsToProcess = journalRequestRepository.findByBatchIdAndRequestStatus(dto.getBatchId(), RequestStatus.PENDING);
        } else {
            Specification<JournalRequest> spec = (root, query, criteriaBuilder) -> {
                List<Predicate> predicates = new ArrayList<>();
                for (String prefix : dto.getJournalIdPrefixes()) {
                    predicates.add(criteriaBuilder.like(root.get("journalId"), prefix + "%"));
                }
                Predicate journalIdOrPredicates = criteriaBuilder.or(predicates.toArray(new Predicate[0]));
                Predicate statusPredicate = criteriaBuilder.equal(root.get("requestStatus"), RequestStatus.PENDING);
                return criteriaBuilder.and(journalIdOrPredicates, statusPredicate);
            };
            requestsToProcess = journalRequestRepository.findAll(spec);
        }

        if (requestsToProcess.isEmpty()) {
            return Collections.emptyList();
        }
        
        List<JournalRequest> processedRequests = new ArrayList<>();
        List<GlTransaction> transactionsToSave = new ArrayList<>();
        
        for (JournalRequest request : requestsToProcess) {
            boolean isCreator = executorId.equals(request.getCreatorId());
            boolean isSameRole = executorRole != null && executorRole.equals(request.getCreatorRole());

            if (isCreator && isSameRole) continue;
            if (request.getRequestStatus() != RequestStatus.PENDING) continue;

            request.setExecutorId(executorId);
            request.setExecutorRemarks(dto.getRemarks());
            request.setExecutionDate(LocalDateTime.now());
            request.setRequestStatus(dto.getStatus());
        
            if (RequestStatus.ACCEPTED.equals(dto.getStatus())) {
                CreateJournalRequestDto payloadDto = objectMapper.readValue(request.getPayload(), CreateJournalRequestDto.class);
                
                if (payloadDto.getChangeType() == ChangeType.ADD) {
                    GlTransaction newGlTxn = new GlTransaction();
                    newGlTxn.setBatchId(payloadDto.getBatchId());
                    newGlTxn.setJournalId(payloadDto.getJournalId());
                    
                    // UPDATED: Explicitly setting Transaction Date
                    if (payloadDto.getPDate() != null) {
                        newGlTxn.setPostDate(payloadDto.getPDate().atStartOfDay());
                        newGlTxn.setTransactionDate(payloadDto.getPDate());
                    } else {
                        newGlTxn.setPostDate(LocalDateTime.now());
                        newGlTxn.setTransactionDate(LocalDate.now());
                    }

                    newGlTxn.setBranchCode(payloadDto.getBranch());
                    newGlTxn.setCurrency(payloadDto.getCurrency());
                    newGlTxn.setCgl(payloadDto.getCgl());
                    newGlTxn.setNarration(payloadDto.getRemarks());
                    
                    BigDecimal amount = payloadDto.getAmount();
                    if (amount.signum() < 0) { 
                        newGlTxn.setCreditAmount(amount);
                        newGlTxn.setDebitAmount(BigDecimal.ZERO);
                    } else { 
                        newGlTxn.setDebitAmount(amount);
                        newGlTxn.setCreditAmount(BigDecimal.ZERO);
                    }
                    newGlTxn.setTransactionCount(payloadDto.getTransactionCount());
                    newGlTxn.setSourceFlag("J"); 
                    
                    transactionsToSave.add(newGlTxn);
                }
            }
            processedRequests.add(request);
        }
     
        if (!transactionsToSave.isEmpty()) {
            glTransactionRepository.saveAll(transactionsToSave);
        }
        
        List<JournalRequest> savedRequests = journalRequestRepository.saveAll(processedRequests);
        
        try {
            boolean isApproved = RequestStatus.ACCEPTED.equals(dto.getStatus());
            String status = isApproved ? "approved" : "rejected";
            String message;
            String aggregateId;

            if (hasBatchId) {
                aggregateId = dto.getBatchId();
                message = String.format("Your Journal batch (ID: %s) with %d entries has been %s by user %s.",
                                        aggregateId, savedRequests.size(), status, executorId);
            } else {

    @Override
    @Transactional
    public void cancelMyRequestsByJournalPrefixes(List<String> journalIdPrefixes, String userId) {
        Specification<JournalRequest> spec = (root, query, criteriaBuilder) -> {
            List<Predicate> predicates = new ArrayList<>();
            for (String prefix : journalIdPrefixes) {
                predicates.add(criteriaBuilder.like(root.get("journalId"), prefix + "%"));
            }
            Predicate journalIdOrPredicates = criteriaBuilder.or(predicates.toArray(new Predicate[0]));
            Predicate statusPredicate = criteriaBuilder.equal(root.get("requestStatus"), RequestStatus.PENDING);
            Predicate creatorPredicate = criteriaBuilder.equal(root.get("creatorId"), userId);
            return criteriaBuilder.and(journalIdOrPredicates, statusPredicate, creatorPredicate);
        };
        
        List<JournalRequest> requestsToCancel = journalRequestRepository.findAll(spec);
        if (requestsToCancel.isEmpty()) {
            throw new ResourceNotFoundException("No pending requests found for you with the specified journal prefixes.");
        }
        journalRequestRepository.deleteAll(requestsToCancel);
    }
    
    @Override
    @Transactional
    public String createBulkBatchRequest(BatchRequestDto batchDto, String creatorId, Integer creatorRole) throws JsonProcessingException {
        
        String batchId = sequenceService.getNextBatchId();
        log.info("Processing BULK Batch ID: {}", batchId);
       
        LocalDate postingDate = LocalDate.now();
        List<JournalRequest> requestsToSave = new ArrayList<>();
        
      
        String commonRemarks = (batchDto.getCommonBatchRemarks() != null && !batchDto.getCommonBatchRemarks().isEmpty()) 
                                ? batchDto.getCommonBatchRemarks() 
                                : "Bulk Upload Batch " + batchId;

        // Grouping logic (Same as original, ensures Journal IDs are sequenced per Branch/Currency)
        Map<String, List<BatchRequestDto.JournalRequestRow>> groupedRows = batchDto.getRows().stream()
                .collect(Collectors.groupingBy(row -> row.getBranch() + ":" + row.getCurrency()));

        for (Map.Entry<String, List<BatchRequestDto.JournalRequestRow>> entry : groupedRows.entrySet()) {
            
            String journalIdPrefix = sequenceService.getNextJournalIdPrefix();
            int journalSequence = 1;
            List<BatchRequestDto.JournalRequestRow> rowsInGroup = entry.getValue();
            
            for (BatchRequestDto.JournalRequestRow rowDto : rowsInGroup) {
                
                String journalId = String.format("%s-%03d", journalIdPrefix, journalSequence);
                
                CreateJournalRequestDto payloadDto = new CreateJournalRequestDto();
                payloadDto.setPDate(postingDate); 
                payloadDto.setChangeType(rowDto.getChangeType());
                payloadDto.setMasterJournalId(rowDto.getMasterJournalId());
                payloadDto.setBranch(rowDto.getBranch());
                payloadDto.setCurrency(rowDto.getCurrency());
                payloadDto.setCgl(rowDto.getCgl());
                payloadDto.setAmount(rowDto.getAmount()); 
                payloadDto.setProductType(rowDto.getProductType());
                payloadDto.setRemarks(rowDto.getRemarks());
     
                payloadDto.setArFlag(rowDto.getArFlag() != null ? rowDto.getArFlag() : "A");
                payloadDto.setAcClassification(rowDto.getAcClassification() != null ? rowDto.getAcClassification() : "A");
                
                payloadDto.setBatchId(batchId); 
                payloadDto.setJournalId(journalId); 
                payloadDto.setCommonBatchRemarks(commonRemarks);
                payloadDto.setTransactionCount(journalSequence); 
                
                JournalRequest request = new JournalRequest();
                request.setCreatorId(creatorId);
                request.setCreatorRole(creatorRole);
                request.setChangeType(rowDto.getChangeType()); // Default to ADD
                request.setRequestStatus(RequestStatus.PENDING);
                request.setBatchId(batchId);
                request.setJournalId(journalId);
                request.setCommonBatchRemarks(commonRemarks);
                request.setPayload(objectMapper.writeValueAsString(payloadDto));

                requestsToSave.add(request);
                journalSequence++; 
            }
        }

      
        journalRequestRepository.saveAll(requestsToSave);
        
     
        try {
            NotificationConfigDto config = permissionConfigService.getConfig("JOURNAL_AUTH"); 
            String message = String.format("Bulk Journal Batch %s (%d rows) by %s pending approval.", 
                                           batchId, requestsToSave.size(), creatorId);

            notificationWriterService.createNotification(
                null,                   
                config.getTargetRoles(),
                message,
                config.getTargetUrl(),  
                batchId,                
                "JournalService"
            );
        } catch (Exception e) {
            log.error("Failed to create notification for Bulk Batch: {}", batchId, e);
            
        }
   
        return batchId; // Return ID only
    }
}

import lombok.RequiredArgsConstructor;
import lombok.extern.slf4j.Slf4j;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Propagation;
import org.springframework.transaction.annotation.Transactional;

import com.fincore.JournalService.Models.Notifications;
import com.fincore.JournalService.Repository.NotificationRepository;

@Service
@RequiredArgsConstructor
@Slf4j
public class NotificationWriterService {

    private final NotificationRepository notificationRepository;

    @Transactional(propagation = Propagation.MANDATORY) 
    public void createNotification(String userId, String targetRole, String message, String linkUrl, String aggregateId, String eventSource) {

        if (userId == null && targetRole == null) {
            log.warn("Skipping notification creation: Both userId and targetRole are null. AggregateID: {}", aggregateId);
            return;
        }

        if (message == null || message.isBlank()) {
            log.warn("Skipping notification creation: Message is null or blank. AggregateID: {}", aggregateId);
            throw new IllegalArgumentException("Notification message cannot be null or blank.");
        }

        Notifications notification = Notifications.builder()
                .userId(userId)
                .targetRole(targetRole)
                .message(message)
                .linkUrl(linkUrl)
                .aggregateId(aggregateId)
                .eventSource(eventSource)
                .build();

        notificationRepository.save(notification);

        if (userId != null) {
            log.info("Saved 1-to-1 notification event for user: {} (AggregateID: {})", userId, aggregateId);
        } else {
            log.info("Saved 1-to-many notification event for role: {} (AggregateID: {})", targetRole, aggregateId);
        }
    }
}
package com.fincore.JournalService.Service;

import lombok.RequiredArgsConstructor;
import lombok.extern.slf4j.Slf4j;
import org.springframework.cache.annotation.Cacheable;
import org.springframework.stereotype.Service;

import com.fincore.JournalService.Dto.NotificationConfigDto;
import com.fincore.JournalService.Repository.PermissionRepository;

import java.util.List;
import java.util.stream.Collectors;

@Service
@RequiredArgsConstructor
@Slf4j
public class PermissionConfigService {

    private final PermissionRepository permissionRepository;

    @Cacheable(value = "notification_configs", key = "#requestType")
    public NotificationConfigDto getConfig(String requestType) {

        log.info("Cache Miss: Fetching DB permissions for MOC notification type: {}", requestType);

        List<Object[]> results = permissionRepository.findUrlAndRolesByRequestType(requestType);

        if (results.isEmpty()) {
            log.warn("No notification config found for request type {}. Using defaults.", requestType);
            return new NotificationConfigDto("/dashboard", null); // Default to no roles
        }

        String url = (String) results.get(0)[0];
        if (url == null) url = "/dashboard";

        String roles = results.stream()
                .map(row -> String.valueOf(row[1])) // Role ID
                .distinct()
                .collect(Collectors.joining(","));

        if (roles.isEmpty()) {
             log.warn("No roles found for request type {}. Notification will not be sent to a group.", requestType);
             return new NotificationConfigDto(url, null);
        }
        
        log.info("Fetched Notification Config for {}: URL=[{}], Roles=[{}]", requestType, url, roles);
        return new NotificationConfigDto(url, roles);
    }
}

import jakarta.persistence.EntityManager;
import jakarta.persistence.PersistenceContext;
import lombok.RequiredArgsConstructor;
import org.springframework.stereotype.Service;
import jakarta.transaction.Transactional;

@Service
@RequiredArgsConstructor
public class SequenceService {

    @PersistenceContext
    private EntityManager entityManager;

    /**
     * Pads the given number with leading zeros to a total length of 7.
     */
    private String padToSevenDigits(Long id) {
        if (id == null) {
            throw new IllegalStateException("Sequence query returned null");
        }
        return String.format("%07d", id);
    }

    /**
     * Gets the next BATCH_ID from the BATCH_SEQ sequence. 
     */
    @Transactional
    public String getNextBatchId() {
        Number nextId = (Number) entityManager
                .createNativeQuery("select BATCH_SEQ.nextval from dual")
                .getSingleResult();
        
        return padToSevenDigits(nextId.longValue());
    }

    /**
     * Gets the next JOURNAL_ID prefix from the JRNL_SEQ sequence.
     */
    @Transactional
    public String getNextJournalIdPrefix() {
        
      
        Number nextId = (Number) entityManager
                .createNativeQuery("select JRNL_SEQ.nextval from dual")
                .getSingleResult();
      

        return padToSevenDigits(nextId.longValue());
    }
}

          <Stack direction="row" spacing={0.5} justifyContent="center">
            <Tooltip title="View Details">
              <IconButton
                onClick={() =>
                  handleOpenViewModal(params.row.batchId, params.row.creatorId)
                }
                color="primary"
              >
                <VisibilityIcon />
              </IconButton>
            </Tooltip>
            <Tooltip title={isSelfCreated ? selfCreateTooltip : "Accept Batch"}>
              <span>
                <IconButton
                  onClick={() => handleOpenModal(params.row, "ACCEPTED")}
                  color="success"
                  disabled={isSelfCreated}
                >
                  <CheckIcon />
                </IconButton>
              </span>
            </Tooltip>
            <Tooltip title={isSelfCreated ? selfCreateTooltip : "Reject Batch"}>
              <span>
                <IconButton
                  onClick={() => handleOpenModal(params.row, "REJECTED")}
                  color="error"
                  disabled={isSelfCreated}
                >
                  <ClearIcon />
                </IconButton>
              </span>
            </Tooltip>
          </Stack>
        );
      },
    },
  ];

  const isViewingSelfCreated =
    currentUserId &&
    viewingBatchCreatorId &&
    String(currentUserId) === String(viewingBatchCreatorId);

  return (
    <Box sx={mainContainerStyle}>
      <DataGrid
        rows={rows}
        columns={columns}
        loading={loading}
        getRowId={(row) => row.batchId}
        disableRowSelectionOnClick
        pageSizeOptions={[5, 10, 25]}
        disableColumnResize={true}
        rowHeight={52}
        initialState={{
          pagination: {
            paginationModel: { page: 0, pageSize: 10 },
          },
          sorting: { sortModel: [{ field: "requestDate", sort: "desc" }] },
        }}
        slots={{
          noRowsOverlay: CustomNoRowsOverlay,
        }}
        sx={dataGridContainerStyle}
      />

      {/* --- Batch Action Modal --- */}
      <Dialog
        open={isModalOpen}
        onClose={handleCloseModal}
        fullWidth
        maxWidth="sm"
      >
        <DialogTitle>
          {modalMode === "ACCEPTED" ? "Approve" : "Reject"} Batch
        </DialogTitle>
        <DialogContent>
          <DialogContentText>
            Are you sure you want to {modalMode.toLowerCase()} the entire batch{" "}
            <strong>{selectedBatch?.batchId}</strong>?
            {modalMode === "REJECTED" && " Remarks are required for rejection."}
          </DialogContentText>

          <TextField
            autoFocus={modalMode === "REJECTED"}
            required={modalMode === "REJECTED"}
            margin="dense"
            id="executorRemarks"
            label="Approver Remarks"
            type="text"
            fullWidth
            variant="standard"
            value={executorRemarks}
            onChange={(e) => setExecutorRemarks(e.target.value)}
            disabled={isSubmitting}
            InputProps={{
              inputProps: {
                maxLength: 200,
              },
            }}
          />
        </DialogContent>
        <DialogActions sx={dialogActionsStyle}>
          <Stack direction="row" spacing={1} justifyContent="flex-end">
            <Button onClick={handleCloseModal} disabled={isSubmitting}>
              Cancel
            </Button>
            <Button
              onClick={handleSubmitAction}
              variant="contained"
              color={modalMode === "ACCEPTED" ? "success" : "error"}
              disabled={isSubmitting}
            >
              {isSubmitting ? (
                <CircularProgress size={24} color="inherit" />
              ) : (
                `Confirm ${modalMode === "ACCEPTED" ? "Approval" : "Rejection"}`
              )}
            </Button>
          </Stack>
        </DialogActions>
      </Dialog>

      {/* --- View Details Modal --- */}
      <Dialog
        open={isViewModalOpen}
        onClose={handleCloseViewModal}
        fullWidth
        maxWidth="lg"
      >
        <DialogTitle sx={dialogTitleStyle}>
          Batch Details: {viewingBatchId}
          <IconButton
            aria-label="close"
            onClick={handleCloseViewModal}
            sx={{
              color: (theme) => theme.palette.grey[500],
            }}
          >
            <CloseIcon />
          </IconButton>
        </DialogTitle>

        <DialogContent dividers sx={{ p: 0 }}>
          {isViewLoading ? (
            <Box sx={loadingContainerStyle}>
              <CircularProgress />
            </Box>
          ) : (
            <>
              {batchDetails.length > 0 ? (
                <TableContainer component={Paper} sx={tableContainerStyle}>
                  <Table sx={{ minWidth: 650 }} size="small" stickyHeader>
                    <TableHead>
                      <TableRow>
                        <TableCell padding="checkbox" sx={tableHeaderCellStyle}>
                          <Checkbox
                            color="primary"
                            indeterminate={
                              selectedJournalPrefixes.length > 0 &&
                              selectedJournalPrefixes.length <
                                batchDetails.length
                            }
                            checked={
                              batchDetails.length > 0 &&
                              selectedJournalPrefixes.length ===
                                batchDetails.length
                            }
                            onChange={handleSelectAllClick}
                            disabled={isViewingSelfCreated}
                          />
                        </TableCell>
                        <TableCell align="center" sx={tableHeaderCellStyle}>
                          Journal ID
                        </TableCell>
                        <TableCell align="center" sx={tableHeaderCellStyle}>
                          Branch
                        </TableCell>
                        <TableCell align="center" sx={tableHeaderCellStyle}>
                          Currency
                        </TableCell>
                        <TableCell align="center" sx={tableHeaderCellStyle}>
                          CGL
                        </TableCell>
                        <TableCell align="center" sx={tableHeaderCellStyle}>
                          Amount
                        </TableCell>
                        <TableCell align="center" sx={tableHeaderCellStyle}>
                          Txn Type
                        </TableCell>
                      </TableRow>
                    </TableHead>

                    <TableBody>
                      {batchDetails.map((group, groupIndex) => {
                        const isSelected = selectedJournalPrefixes.includes(
                          group[0].journalPrefix
                        );
                        const groupRowCount = group.length;

                        return group.map((row, rowIndex) => (
                          <TableRow
                            key={row.id}
                            hover={!isViewingSelfCreated}
                            onClick={() =>
                              !isViewingSelfCreated &&
                              handleRowCheckboxClick(null, row.journalPrefix)
                            }
                            role="checkbox"
                            aria-checked={isSelected}
                            selected={isSelected}
                            sx={{
                              cursor: isViewingSelfCreated
                                ? "not-allowed"
                                : "pointer",
                              ...(rowIndex === 0 &&
                                groupIndex > 0 && {
                                  borderTop: (theme) =>
                                    `2px solid ${theme.palette.divider}`,
                                }),
                            }}
                          >
                            {rowIndex === 0 && (
                              <TableCell
                                padding="checkbox"
                                rowSpan={groupRowCount}
                                sx={stickyColumnCell}
                              >
                                <Box sx={stickyBoxStyle}>
                                  <Checkbox
                                    color="primary"
                                    checked={isSelected}
                                    disabled={isViewingSelfCreated}
                                    onClick={(e) =>
                                      !isViewingSelfCreated &&
                                      handleRowCheckboxClick(
                                        e,
                                        row.journalPrefix
                                      )
                                    }
                                  />
                                </Box>
                              </TableCell>
                            )}
                            {rowIndex === 0 && (
                              <TableCell
                                align="center"
                                rowSpan={groupRowCount}
                                sx={stickyColumnCell}
                              >
                                <Box sx={stickyBoxStyle}>
                                  {row.journalPrefix}
                                </Box>
                              </TableCell>
                            )}
                            <TableCell align="center" sx={cellBorderStyle}>
                              {row.payload.branch}
                            </TableCell>
                            <TableCell align="center" sx={cellBorderStyle}>
                              {row.payload.currency}
                            </TableCell>
                            <TableCell align="center" sx={cellBorderStyle}>
                              {row.payload.cgl}
                            </TableCell>
                            <TableCell align="center" sx={cellBorderStyle}>
                              {formatCurrency(row.amount)}
                            </TableCell>
                            <TableCell align="center" sx={cellBorderStyle}>
                              <Chip
                                label={row.transactionType}
                                color={getTxnTypeColor(row.transactionType)}
                                size="small"
                              />
                            </TableCell>
                          </TableRow>
                        ));
                      })}
                    </TableBody>
                  </Table>
                </TableContainer>
              ) : (
                <Typography sx={{ p: 3 }}>
                  No pending details found for this batch.
                </Typography>
              )}
            </>
          )}
        </DialogContent>

        <DialogActions sx={dialogActionsStyle}>
          <Grid container spacing={2} alignItems="center">
            <Grid item xs={12} md={6}>
              <TextField
                margin="dense"
                id="partialExecutorRemarks"
                label="Approver Remarks (for selected)"
                type="text"
                fullWidth
                size="small"
                value={partialExecutorRemarks}
                onChange={(e) => setPartialExecutorRemarks(e.target.value)}
                disabled={
                  isPartialSubmitting ||
                  selectedJournalPrefixes.length === 0 ||
                  isViewingSelfCreated
                }
                InputProps={{
                  inputProps: {
                    maxLength: 200,
                  },
                }}
              />
            </Grid>
            <Grid item xs={12} md={6}>
              <Stack direction="row" spacing={1} justifyContent="flex-end">
                <Button
                  variant="contained"
                  color="success"
                  onClick={() => handleSubmitPartialAction("ACCEPTED")}
                  disabled={
                    isPartialSubmitting ||
                    selectedJournalPrefixes.length === 0 ||
                    isViewingSelfCreated
                  }
                >
                  {isPartialSubmitting ? (
                    <CircularProgress size={24} color="inherit" />
                  ) : (
                    "Accept Selected"
                  )}
                </Button>
                <Button
                  variant="contained"
                  color="error"
                  onClick={() => handleSubmitPartialAction("REJECTED")}
                  disabled={
                    isPartialSubmitting ||
                    selectedJournalPrefixes.length === 0 ||
                    isViewingSelfCreated
                  }
                >
                  {isPartialSubmitting ? (
                    <CircularProgress size={24} color="inherit" />
                  ) : (
                    "Reject Selected"
                  )}
                </Button>
              </Stack>
            </Grid>
            {isViewingSelfCreated && (
              <Grid item xs={12}>
                <Typography
                  color="error"
                  variant="body2"
                  textAlign="center"
                  sx={{ pt: 1, width: "100%" }}
                >
                  You cannot approve or reject entries from your own request.
                </Typography>
              </Grid>
            )}
          </Grid>
        </DialogActions>
      </Dialog>
    </Box>
  );
}

import React, { useState, useEffect, useCallback } from "react";
import {
  Box,
  Typography,
  Grid,
  TextField,
  Button,
  Paper,
  IconButton,
  MenuItem,
  CircularProgress,
  Autocomplete,
  Tooltip,
  Checkbox,
  FormControlLabel,
} from "@mui/material";
import AddIcon from "@mui/icons-material/Add";
import DeleteIcon from "@mui/icons-material/Delete";
import { debounce } from "lodash";

import useApi from "../../hooks/useApi";
import useCustomSnackbar from "../../utils/useCustomSnackbar";

const initialRowState = {
  id: Date.now(),
  branch: null,
  currency: "",
  cgl: null,
  amount: "",
  txnType: "",
  remarks: "",
  productCode: "",
};

const formatBigIntForDisplay = (val) => {
  if (typeof val !== "bigint") {
    val = BigInt(val || 0);
  }
  const str = val.toString().padStart(5, "0");
  const integer = str.slice(0, -4);
  const fractional = str.slice(-4);
  return `${integer || "0"}.${fractional}`;
};

export default function JournalPosting() {
  const { callApi } = useApi();
  const showSnackBar = useCustomSnackbar();

  const [rows, setRows] = useState([initialRowState]);
  const [currencyOptions, setCurrencyOptions] = useState([]);

  const [isSubmitting, setIsSubmitting] = useState(false);

  const [commonBatchRemarks, setCommonBatchRemarks] = useState("");
  const [postingDate, setPostingDate] = useState("");

  // Checkbox
  const [checked, setChecked] = useState(false);

  const [branchSearch, setBranchSearch] = useState({
    loading: false,
    options: [],
  });
  const [cglSearch, setCglSearch] = useState({ loading: false, options: [] });

  // --- NEW: Track active row for independent loaders ---
  const [activeBranchRowId, setActiveBranchRowId] = useState(null);
  const [activeCglRowId, setActiveCglRowId] = useState(null);

  useEffect(() => {
    const fetchInitialData = async () => {
      try {
        const dateResponse = await callApi(
          "/JS/journals/current-posting-date",
          null,
          "GET"
        );

        if (dateResponse) {
          try {
            const [year, month, day] = dateResponse.split("-");
            const formattedDate = `${day}-${month}-${year}`;
            setPostingDate(formattedDate);
          } catch (e) {
            console.error("Could not parse date:", dateResponse, e);
            setPostingDate(dateResponse);
          }
        } else {
          showSnackBar("Failed to load official posting date.", "error");
          setPostingDate("Error");
        }

        const currencyResponse = await callApi("/CM/currency", null, "GET");
        if (currencyResponse?.success && Array.isArray(currencyResponse.data)) {
          const uniqueCurrencies = [...new Set(currencyResponse.data)];
          setCurrencyOptions(uniqueCurrencies);
        } else {
          showSnackBar("Failed to load currencies.", "error");
        }
      } catch (error) {
        console.error("Error fetching initial data:", error);
        showSnackBar("An error occurred while loading page data.", "error");
      }
    };

    fetchInitialData();
  }, [callApi, showSnackBar]);

  const handleAddRow = () => {
    const lastRow = rows[rows.length - 1];

    if (
      !lastRow.branch ||
      !lastRow.currency ||
      !lastRow.cgl ||
      !lastRow.amount ||
      !lastRow.txnType
    ) {
      showSnackBar(
        "Please fill all required fields (*) in the current row before adding a new one.",
        "warning"
      );
      return;
    }

    const newRow = {
      ...initialRowState,
      id: Date.now(),
      remarks: checked ? commonBatchRemarks : "",
    };
    setRows([...rows, newRow]);
  };

  const handleChange = (event) => {
    const isChecked = event.target.checked;
    setChecked(isChecked);

    if (isChecked) {
      const newRows = rows.map((row) => ({
        ...row,
        remarks: commonBatchRemarks,
      }));
      setRows(newRows);
    } else {
      const newRows = rows.map((row) => ({
        ...row,
        remarks: "",
      }));
      setRows(newRows);
    }
  };

  const handleDeleteRow = (id) => {
    if (rows.length > 1) {
      setRows(rows.filter((row) => row.id !== id));
    } else {
      showSnackBar("Cannot delete the last row.", "warning");
    }
  };

  const handleRowChange = (id, field, value) => {
    const newRows = rows.map((row) => {
      if (row.id === id) {
        return { ...row, [field]: value };
      }
      return row;
    });
    setRows(newRows);
  };

  // --- UPDATED SEARCH: Accepts rowId ---
  const searchBranches = useCallback(
    debounce(async (query, rowId) => {
      if (!query) {
        setBranchSearch({ loading: false, options: [] });
        setActiveBranchRowId(null);
        return;
      }
      
      setActiveBranchRowId(rowId); // Mark this row as active
      setBranchSearch((prev) => ({ ...prev, loading: true }));
      
      try {
        const response = await callApi(
          `/JS/branches-journal/search?query=${encodeURIComponent(query)}`,
          null,
          "GET"
        );

        setBranchSearch({ loading: false, options: response || [] });
      } catch (error) {
        showSnackBar("Branch search failed.", "error");
        setBranchSearch({ loading: false, options: [] });
      }
    }, 300),
    [callApi, showSnackBar]
  );

  const searchCgls = useCallback(
    debounce(async (query, rowId) => {
      if (!query) {
        setCglSearch({ loading: false, options: [] });
        setActiveCglRowId(null);
        return;
      }

      setActiveCglRowId(rowId); // Mark this row as active
      setCglSearch((prev) => ({ ...prev, loading: true }));
      
      try {
        const response = await callApi(
          `/JS/cgl-journal/search?query=${encodeURIComponent(query)}`,
          null,
          "GET"
        );
        setCglSearch({ loading: false, options: response || [] });
      } catch (error) {
        showSnackBar("CGL search failed.", "error");
        setCglSearch({ loading: false, options: [] });
      }
    }, 300),
    [callApi, showSnackBar]
  );

  const handleSubmit = async () => {
    setIsSubmitting(true);

    if (!commonBatchRemarks.trim()) {
      showSnackBar(
        "Please enter Common Batch Remarks before submitting.",
        "error"
      );
      setIsSubmitting(false);
      return;
    }

    const cglCheck = new Map();
    for (const row of rows) {
      if (!row.branch || !row.currency || !row.cgl) {
        continue;
      }
      const key = `${row.branch.code}-${row.currency}`;
      const cgl = row.cgl.cglNumber;

      if (!cglCheck.has(key)) {
        cglCheck.set(key, new Set());
      }

      const cglSet = cglCheck.get(key);
      if (cglSet.has(cgl)) {
        showSnackBar(
          `Duplicate CGL entry: "${cgl}" is used more than once for branch "${row.branch.name}" and currency "${row.currency}".`,
          "error"
        );
        setIsSubmitting(false);
        return;
      }
      cglSet.add(cgl);
    }

    const branchCurrencyTotals = new Map();
    const branchEntryCheck = new Map();
    const rowPayloads = [];

    for (const [index, row] of rows.entries()) {
      if (
        !row.branch ||
        !row.currency ||
        !row.cgl ||
        !row.amount ||
        !row.txnType
      ) {
        showSnackBar(
          `Row ${index + 1} is incomplete. Please fill all required fields.`,
          "error"
        );
        setIsSubmitting(false);
        return;
      }

      const branchCode = row.branch.code;
      const branchName = row.branch.name;
      const currency = row.currency;

      const [integerPart = "0", fractionalPart = "0"] = (
        row.amount || "0"
      ).split(".");
      const fractionalPadded = fractionalPart.substring(0, 4).padEnd(4, "0");
      const amountAsBigInt = BigInt(integerPart + fractionalPadded);

      const compositeKey = `${branchCode}-${currency}`;

      if (!branchCurrencyTotals.has(compositeKey)) {
        branchCurrencyTotals.set(compositeKey, {
          totalDebit: 0n,
          totalCredit: 0n,
          branchName: branchName,
          branchCode: branchCode,
          currency: currency,
        });
      }
      const currencyTotals = branchCurrencyTotals.get(compositeKey);

      if (row.txnType === "Debit") {
        currencyTotals.totalDebit += amountAsBigInt;
      } else if (row.txnType === "Credit") {
        currencyTotals.totalCredit += amountAsBigInt;
      }

      if (!branchEntryCheck.has(branchCode)) {
        branchEntryCheck.set(branchCode, {
          hasDebit: false,
          hasCredit: false,
          branchName: branchName,
        });
      }
      const branchCheck = branchEntryCheck.get(branchCode);

      if (row.txnType === "Debit" && amountAsBigInt > 0n) {
        branchCheck.hasDebit = true;
      } else if (row.txnType === "Credit" && amountAsBigInt > 0n) {
        branchCheck.hasCredit = true;
      }

      const finalAmount =
        row.txnType === "Credit" ? "-" + row.amount : row.amount;

      rowPayloads.push({
        changeType: "ADD",
        branch: row.branch.code,
        currency: row.currency,
        cgl: row.cgl.cglNumber,
        amount: finalAmount,
        productType: row.productCode,
        remarks: row.remarks,
        arFlag: "A",
      });
    }

    for (const [branchCode, totals] of branchEntryCheck.entries()) {
      const { hasDebit, hasCredit, branchName } = totals;
      if (!hasDebit || !hasCredit) {
        showSnackBar(
          `Submission failed. Branch "${branchName}" (${branchCode}) must have at least one Debit and one Credit entry.`,
          "error"
        );
        setIsSubmitting(false);
        return;
      }
    }

    for (const [compositeKey, totals] of branchCurrencyTotals.entries()) {
      const { totalDebit, totalCredit, branchName, branchCode, currency } =
        totals;

      if (totalDebit !== totalCredit) {
        showSnackBar(
          `Submission failed. Totals for branch "${branchName}" (${branchCode}) and currency "${currency}" do not match. Debit: ${formatBigIntForDisplay(
            totalDebit
          )}, Credit: ${formatBigIntForDisplay(totalCredit)}`,
          "error"
        );
        setIsSubmitting(false);
        return;
      }
    }

    const finalPayload = {
      commonBatchRemarks: commonBatchRemarks.trim(),
      rows: rowPayloads,
    };

    try {
      const response = await callApi(
        "/JS/journals/create-batch",
        finalPayload,
        "POST"
      );

      const newBatchId = response?.[0]?.batchId || "N/A";

      showSnackBar(
        `All ${response.length} Journal entries submitted successfully! Batch ID: ${newBatchId}`,
        "success"
      );
      setRows([{ ...initialRowState, id: Date.now() }]);
      setCommonBatchRemarks("");
      setChecked(false);
    } catch (error) {
      showSnackBar("Journal Batch submission failed", "error");
    } finally {
      setIsSubmitting(false);
    }
  };

  return (
    <Box sx={{ p: 3, maxWidth: "98%" }}>
      <Paper elevation={3} sx={{ p: 3 }}>
        <Box>
          <Box sx={{ display: "flex", alignItems: "center", gap: 0.5 }}>
            <Typography variant="body1" fontWeight="bold" component="span">
              Posting Date:
            </Typography>
            <Typography variant="body1" fontWeight="bold" component="span">
              {postingDate}
            </Typography>
          </Box>
          <Box sx={{ marginTop: 0.5 }}>
            <Typography
              variant="body2"
              component="span"
              sx={{ color: "#ec0101ff" }}
            >
              Note: All Journals will be Posted on this Date
            </Typography>
          </Box>
        </Box>

        <Grid item xs={12} md={12} sx={{ margin: "1% 0%", width: "95%" }}>
          <TextField
            label="Common Batch Remarks"
            value={commonBatchRemarks}
            onChange={(e) => {
              const newValue = e.target.value;
              if (newValue.length <= 50) {
                setCommonBatchRemarks(newValue);

                if (checked && newValue.trim()) {
                  setRows(
                    rows.map((row) => ({
                      ...row,
                      remarks: newValue,
                    }))
                  );
                }

                if (!newValue.trim()) {
                  setChecked(false);
                  setRows(
                    rows.map((row) => ({
                      ...row,
                      remarks: "",
                    }))
                  );
                }
              }
            }}
            fullWidth
            required
            multiline
            rows={1}
            InputProps={{
              inputProps: {
                maxLength: 50,
              },
            }}
          />
          <Box sx={{ display: "flex", alignItems: "center" }}>
            <FormControlLabel
              control={
                <Checkbox
                  checked={checked}
                  onChange={handleChange}
                  color="primary"
                  disabled={!commonBatchRemarks.trim()}
                />
              }
              label="Apply This Remark To All Below Transaction"
            />
          </Box>
        </Grid>

        <Box>
          {rows.map((row, index) => (
            <React.Fragment key={row.id}>
              <Grid container spacing={2} alignItems="center">
                {/* Branch - RESTORED PROPS: md={3} size={1.8} */}
                <Grid item xs={12} md={3} size={1.8}>
                  <Autocomplete
                    popupIcon={null}
                    autoHighlight
                    options={branchSearch.options}
                    getOptionLabel={(option) =>
                      `${option.code} - ${option.name}`
                    }
                    value={row.branch}
                    onChange={(event, newValue) =>
                      handleRowChange(row.id, "branch", newValue)
                    }
                    onInputChange={(event, newInputValue) =>
                      searchBranches(newInputValue, row.id)
                    }
                    // FIX: Loader active only for this row
                    loading={branchSearch.loading && activeBranchRowId === row.id}
                    filterOptions={(x) => x}
                    isOptionEqualToValue={(option, value) =>
                      option.code === value.code
                    }
                    noOptionsText=""
                    onClose={() => setActiveBranchRowId(null)}
                    renderInput={(params) => (
                      <Tooltip
                        title="Search by Branch name or Branch code"
                        placement="top-start"
                      >
                        <TextField
                          {...params}
                          label="Branch *"
                          placeholder="Search by name or code"
                          InputLabelProps={{ shrink: true }}
                          InputProps={{
                            ...params.InputProps,
                            endAdornment: (
                              <Box>
                                {branchSearch.loading && activeBranchRowId === row.id ? (
                                  <CircularProgress color="inherit" size={20} />
                                ) : null}
                                {params.InputProps.endAdornment}
                              </Box>
                            ),
                          }}
                        />
                      </Tooltip>
                    )}
                  />
                </Grid>

                {/* Currency - RESTORED PROPS: md={1.5} size={1.5} */}
                <Grid item xs={12} md={1.5} size={1.5}>
                  <TextField
                    popupIcon={null}
                    select
                    label="Currency *"
                    value={row.currency}
                    onChange={(e) =>
                      handleRowChange(row.id, "currency", e.target.value)
                    }
                    fullWidth
                  >
                    <MenuItem value="" disabled>
                      <em>Select Currency</em>
                    </MenuItem>
                    {currencyOptions.map((c) => (
                      <MenuItem key={c} value={c}>
                        {c}
                      </MenuItem>
                    ))}
                  </TextField>
                </Grid>

                {/* CGL - RESTORED PROPS: md={2} size={1.8} */}
                <Grid item xs={12} md={2} size={1.8}>
                  <Autocomplete
                    popupIcon={null}
                    autoHighlight
                    options={cglSearch.options}
                    getOptionLabel={(option) =>
                      `${option.cglNumber} - ${option.description}`
                    }
                    value={row.cgl}
                    onChange={(event, newValue) =>
                      handleRowChange(row.id, "cgl", newValue)
                    }
                    onInputChange={(event, newInputValue) =>
                      searchCgls(newInputValue, row.id)
                    }
                    // FIX: Loader active only for this row
                    loading={cglSearch.loading && activeCglRowId === row.id}
                    filterOptions={(x) => x}
                    isOptionEqualToValue={(option, value) =>
                      option.cglNumber === value.cglNumber
                    }
                    noOptionsText=""
                    onClose={() => setActiveCglRowId(null)}
                    renderInput={(params) => (
                      <TextField
                        {...params}
                        label="CGL *"
                        placeholder="Search CGL by number or name"
                        InputLabelProps={{ shrink: true }}
                        InputProps={{
                          ...params.InputProps,
                          endAdornment: (
                            <Box sx={{ marginLeft: "2%" }}>
                              {cglSearch.loading && activeCglRowId === row.id ? (
                                <CircularProgress color="inherit" size={20} />
                              ) : null}
                              {params.InputProps.endAdornment}
                            </Box>
                          ),
                        }}
                      />
                    )}
                  />
                </Grid>

                {/* Amount - RESTORED PROPS: md={2} size={1.5} */}
                <Grid item xs={12} md={2} size={1.5}>
                  <TextField
                    label="Amount *"
                    type="text"
                    value={row.amount}
                    onChange={(e) => {
                      const val = e.target.value;
                      if (/^\d{0,21}(\.\d{0,4})?$/.test(val)) {
                        handleRowChange(row.id, "amount", val);
                      }
                    }}
                    fullWidth
                    placeholder="0.0000"
                  />
                </Grid>

                {/* Txn Type - RESTORED PROPS: md={1} size={1.3} */}
                <Grid item xs={12} md={1} size={1.3}>
                  <TextField
                    select
                    label="Txn Type *"
                    value={row.txnType}
                    onChange={(e) =>
                      handleRowChange(row.id, "txnType", e.target.value)
                    }
                    fullWidth
                  >
                    <MenuItem value="" disabled>
                      <em>Select Type</em>
                    </MenuItem>
                    <MenuItem value="Debit">Debit</MenuItem>
                    <MenuItem value="Credit">Credit</MenuItem>
                  </TextField>
                </Grid>

                {/* Product Code - RESTORED PROPS: md={2} size={1.5} */}
                <Grid item xs={12} md={2} size={1.5}>
                  <TextField
                    label="Product Code"
                    value={row.productCode}
                    onChange={(e) => {
                      const val = e.target.value;
                      if (/^\d{0,8}$/.test(val)) {
                        handleRowChange(row.id, "productCode", val);
                      }
                    }}
                    fullWidth
                    InputProps={{
                      inputProps: {
                        maxLength: 8,
                      },
                    }}
                  />
                </Grid>

                {/* Remarks - RESTORED PROPS: md={2} size={2} */}
                <Grid item xs={12} md={2} size={2}>
                  <TextField
                    label="Remarks"
                    value={row.remarks}
                    onChange={(e) => {
                      const val = e.target.value;

                      if (val.length <= 50) {
                        handleRowChange(row.id, "remarks", val);
                      }
                    }}
                    fullWidth
                    InputProps={{
                      inputProps: {
                        maxLength: 50,
                      },
                      readOnly: checked,
                    }}
                  />
                </Grid>

                {/* Actions - RESTORED PROPS: md={0.5} */}
                <Grid item xs={12} md={0.5} sx={{ textAlign: "center" }}>
                  <IconButton
                    onClick={() => handleDeleteRow(row.id)}
                    aria-label="delete row"
                  >
                    <DeleteIcon />
                  </IconButton>
                </Grid>
              </Grid>

              {index < rows.length - 1 && (
                <Box
                  sx={{
                    borderBottom: 1,
                    borderColor: "divider",
                    my: 2,
                  }}
                />
              )}
            </React.Fragment>
          ))}

          <Box sx={{ mt: 3, display: "flex", gap: 2, justifyContent: "right" }}>
            <Button
              variant="contained"
              startIcon={<AddIcon />}
              onClick={handleAddRow}
            >
              Add Row
            </Button>

            <Button
              variant="contained"
              color="success"
              onClick={handleSubmit}
              disabled={isSubmitting}
            >
              {isSubmitting ? <CircularProgress size={24} /> : "Submit"}
            </Button>
          </Box>
        </Box>
      </Paper>
    </Box>
  );
}
