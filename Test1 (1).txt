class Solution {
    public int findMin(int[] nums) {

        int n = nums.length;

        int ans = nums[0];

        int low = 0;
        int high = n - 1;

        while (low <= high) {

            int mid = low + (high - low) / 2;

            if (nums[mid] <= ans) {
                 System.out.println("ans" + ans);
                ans = nums[mid];
               
            }

            // first half

            if (nums[mid] >= nums[low]) {

                // ans is there in the first half
                if (ans >= nums[mid] && ans < nums[mid]) {

                    high = mid - 1;

                } else {

                    low = mid + 1;
                }

            }

            // second half 

            else {

                if(ans >= nums[mid] &&  ans <= nums[high]) {
                    low = mid+1;
                }
                else {

                    high = mid-1;

                }

                



            }

        }

        return ans;

    }
}
