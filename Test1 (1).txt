import React, { useState, useEffect } from 'react';

// --- Material-UI Imports ---
import {
    Box,
    Typography,
    Fab,
    Dialog,
    DialogActions,
    DialogContent,
    DialogTitle,
    TextField,
    Button,
    Container,
    Paper
} from '@mui/material';
import AddIcon from '@mui/icons-material/Add';
import { DataGrid } from '@mui/x-data-grid';

/**
 * --- MOCK API CALL ---
 */
const fetchCurrencyData = () => {
    return new Promise((resolve) => {
        const dataFromBackend = [
            { id: 1, currencyCode: 'USD', currencyName: 'United States Dollar', currencyRate: 1.00, rateDate: '2025-09-29' },
            { id: 2, currencyCode: 'EUR', currencyName: 'Euro', currencyRate: 0.92, rateDate: '2025-09-29' },
            { id: 3, currencyCode: 'INR', currencyName: 'Indian Rupee', currencyRate: 83.50, rateDate: '2025-09-29' },
            { id: 4, currencyCode: 'GBP', currencyName: 'British Pound', currencyRate: 0.79, rateDate: '2025-09-28' },
            { id: 5, currencyCode: 'JPY', currencyName: 'Japanese Yen', currencyRate: 149.82, rateDate: '2025-09-29' },
        ];
        setTimeout(() => {
            resolve(dataFromBackend);
        }, 1500);
    });
};


function CurrencyMaster() {
    // --- State Management ---
    const [rows, setRows] = useState([]);
    const [loading, setLoading] = useState(true);
    const [isDialogOpen, setIsDialogOpen] = useState(false);
    const [newCurrency, setNewCurrency] = useState({
        currencyCode: '',
        currencyName: '',
        currencyRate: '',
        rateDate: '',
    });
    const [errors, setErrors] = useState({});

    // --- Data Fetching ---
    useEffect(() => {
        setLoading(true);
        fetchCurrencyData().then(data => {
            setRows(data);
            setLoading(false);
        }).catch(error => {
            console.error("Failed to fetch currency data:", error);
            setLoading(false);
        });
    }, []);

    // --- Grid Column Definitions (FURTHER REFINED FOR BETTER SPACING) ---
    const columns = [
        { field: 'id', headerName: 'ID', width: 70 }, // Slightly smaller ID
        {
            field: 'currencyCode',
            headerName: 'Currency Code',
            width: 120, // Adjusted width
            // This is important: if you set a width, don't also set flex: 1 unless you want complex behavior
        },
        {
            field: 'currencyName',
            headerName: 'Currency Name',
            flex: 1, // Let this one take up remaining space
            minWidth: 200, // Ensure it doesn't get too narrow
        },
        {
            field: 'currencyRate',
            headerName: 'Currency Rate',
            type: 'number',
            width: 130, // Adjusted width
            align: 'right', // Align numbers to the right for readability
            headerAlign: 'right',
        },
        {
            field: 'rateDate',
            headerName: 'Rate Date',
            type: 'date',
            width: 130, // Adjusted width
            valueGetter: (params) => params.value ? new Date(params.value) : null, // Ensure date object for proper sorting/filtering
            renderCell: (params) => {
                // Format date for display
                return params.value ? new Date(params.value).toLocaleDateString() : '';
            }
        },
    ];

    // --- Event Handlers ---
    const handleOpenDialog = () => setIsDialogOpen(true);

    const handleCloseDialog = () => {
        setIsDialogOpen(false);
        setNewCurrency({ currencyCode: '', currencyName: '', currencyRate: '', rateDate: '' });
        setErrors({});
    };

    const handleInputChange = (e) => {
        const { name, value } = e.target;
        setNewCurrency(prev => ({ ...prev, [name]: value }));
    };

    const validateForm = () => {
        let tempErrors = {};
        if (!newCurrency.currencyCode) tempErrors.currencyCode = "Currency Code is required.";
        if (!newCurrency.currencyName) tempErrors.currencyName = "Currency Name is required.";
        if (!newCurrency.currencyRate) tempErrors.currencyRate = "Rate is required.";
        else if (isNaN(newCurrency.currencyRate)) tempErrors.currencyRate = "Rate must be a number.";
        if (!newCurrency.rateDate) tempErrors.rateDate = "Rate Date is required.";
        setErrors(tempErrors);
        return Object.keys(tempErrors).length === 0;
    };

    const handleSubmit = () => {
        if (validateForm()) {
            console.log("Submitting to backend:", JSON.stringify(newCurrency, null, 2));
            handleCloseDialog();
        }
    };

    // --- JSX Rendering ---
    return (
        <Container
            maxWidth="xl" // Allows the container to go wider
            sx={{
                mt: 4,
                mb: 4,
                display: 'flex',
                justifyContent: 'center', // Center the content horizontally
                width: '100%', // Take full width of parent
            }}
        >
            <Paper
                elevation={3}
                sx={{
                    p: 3,
                    borderRadius: 2,
                    width: '80%', // Explicitly set the Paper to 80% width
                    // Ensure the paper itself does not exceed a reasonable max width on very large screens
                    maxWidth: 1200, // Example max width for the grid area
                }}
            >
                <Typography variant="h4" component="h1" gutterBottom sx={{ fontWeight: '600', color: 'primary.main' }}>
                    Currency Master
                </Typography>
                <Box sx={{ height: '70vh', width: '100%' }}>
                    <DataGrid
                        rows={rows}
                        columns={columns}
                        loading={loading}
                        initialState={{
                            pagination: { paginationModel: { pageSize: 10 } },
                        }}
                        pageSizeOptions={[10, 25, 50]}
                        disableRowSelectionOnClick
                    />
                </Box>
            </Paper>

            <Fab color="primary" aria-label="add" onClick={handleOpenDialog} sx={{ position: 'fixed', bottom: 30, right: 30 }}>
                <AddIcon />
            </Fab>

            <Dialog open={isDialogOpen} onClose={handleCloseDialog} fullWidth maxWidth="sm">
                <DialogTitle sx={{ fontWeight: 'bold' }}>Add New Currency</DialogTitle>
                <DialogContent>
                    <TextField autoFocus margin="dense" name="currencyCode" label="Currency Code" type="text" fullWidth variant="outlined" value={newCurrency.currencyCode} onChange={handleInputChange} error={!!errors.currencyCode} helperText={errors.currencyCode} />
                    <TextField margin="dense" name="currencyName" label="Currency Name" type="text" fullWidth variant="outlined" value={newCurrency.currencyName} onChange={handleInputChange} error={!!errors.currencyName} helperText={errors.currencyName} />
                    <TextField margin="dense" name="currencyRate" label="Currency Rate" type="number" fullWidth variant="outlined" value={newCurrency.currencyRate} onChange={handleInputChange} error={!!errors.currencyRate} helperText={errors.currencyRate} />
                    <TextField margin="dense" name="rateDate" label="Rate Date" type="date" fullWidth variant="outlined" value={newCurrency.rateDate} onChange={handleInputChange} error={!!errors.rateDate} helperText={errors.rateDate} InputLabelProps={{ shrink: true }} />
                </DialogContent>
                <DialogActions sx={{ p: '16px 24px' }}>
                    <Button onClick={handleCloseDialog}>Cancel</Button>
                    <Button onClick={handleSubmit} variant="contained">Submit</Button>
                </DialogActions>
            </Dialog>
        </Container>
    );
}
