@RestController
@RequestMapping("moc") 
@RequiredArgsConstructor
@Slf4j
public class MocRequestController {

    private final MocRequestService mocRequestService;

    @PostMapping("/create-request")
    @ResponseStatus(HttpStatus.CREATED)
    public MocRequest createRequest(  @Valid
             @RequestBody CreateMocRequestDto createRequestDto,
            @RequestHeader("Authorization") String token) throws JsonProcessingException {
        String userId = JwtUtil.getUserIdFromToken(token);
        log.info("Received MOC request creation call from user: {}", userId);
        return mocRequestService.createRequest(createRequestDto, userId);
    }

    @PatchMapping("/update-request")
    public MocRequest updateRequestStatus(
            @RequestHeader("Authorization") String token,
            @RequestBody ProcessMocRequestDto processRequestDto) throws JsonProcessingException {
        String executorId = JwtUtil.getUserIdFromToken(token);
        return mocRequestService.updateRequestStatus(processRequestDto, executorId).get();
    }

    @GetMapping("/my-requests")
    public List<MocRequest> getMyRequests(@RequestHeader("Authorization") String token) {
        String userId = JwtUtil.getUserIdFromToken(token);
        return mocRequestService.getMyRequests(userId);
    }

    @GetMapping("/pending-requests")
    public List<MocRequest> getPendingRequests(@RequestHeader("Authorization") String token) {
        return mocRequestService.getPendingRequests();
    }
    
    
    @PostMapping("/process-bulk")
    public List<MocRequest> processBulkRequests(
            @RequestHeader("Authorization") String token,
            @Valid @RequestBody BulkProcessMocRequestDto processRequestDto) throws JsonProcessingException {
        
        String executorId = JwtUtil.getUserIdFromToken(token);
        log.info("Received bulk MOC process call from user: {}", executorId);
        return mocRequestService.processBulkRequests(processRequestDto, executorId);
    }
    @GetMapping("/by-batch/{batchId}")
    public ResponseEntity<List<MocRequest>> getRequestsByBatchId(@PathVariable String batchId) {
        log.info("Received request for all MOC data by Batch ID: {}", batchId);
        List<MocRequest> requests = mocRequestService.getRequestsByBatchId(batchId);
        return ResponseEntity.ok(requests);
    }
}


package com.tcs.fincore.MOCService.Controllers;

import com.tcs.fincore.MOCService.Models.enums.RequestStatus;

import jakarta.persistence.AttributeConverter;
import jakarta.persistence.Converter;
import java.util.stream.Stream;

@Converter(autoApply = true) // autoApply makes it work for all RequestStatus fields
public class RequestStatusConverter implements AttributeConverter<RequestStatus, String> {

    @Override
    public String convertToDatabaseColumn(RequestStatus status) {
        if (status == null) {
            return null;
        }
        return status.getCode(); // Converts PENDING to "P"
    }

    @Override
    public RequestStatus convertToEntityAttribute(String code) {
        if (code == null) {
            return null;
        }
        // Finds the Enum constant that matches the code from the DB
        return Stream.of(RequestStatus.values())
                .filter(c -> c.getCode().equals(code))
                .findFirst()
                .orElseThrow(IllegalArgumentException::new);
    }
}

package com.tcs.fincore.MOCService.Service;


import lombok.RequiredArgsConstructor;
import lombok.extern.slf4j.Slf4j;
import org.springframework.stereotype.Service;

import com.tcs.fincore.MOCService.Dto.BranchDto;
import com.tcs.fincore.MOCService.Models.BranchMaster;
import com.tcs.fincore.MOCService.Repository.BranchMasterRepository;

import java.util.Collections;
import java.util.List;
import java.util.Map;
import java.util.Optional;
import java.util.stream.Collectors;


@Service
@RequiredArgsConstructor
@Slf4j
public class BranchService {

    private final BranchMasterRepository branchMasterRepository;

  
    public Map<String, String> getBranchCodeAndNameMap() {
        log.info("Fetching all branches to create a code-name map.");
        List<BranchMaster> branches = branchMasterRepository.findAll();

        // Using Java Streams for an efficient and clean transformation
        Map<String, String> branchMap = branches.stream()
                .collect(Collectors.toMap(BranchMaster::getCode, BranchMaster::getName));
        
        log.info("Successfully created map with {} branch entries.", branchMap.size());
        return branchMap;
    }

   
    public List<BranchDto> searchBranches(String query) {
        log.info("Searching for branches with query: '{}'", query);

      
        Optional<BranchMaster> byId = branchMasterRepository.findById(query);
        if (byId.isPresent()) {
            log.info("Found exact match by code: {}", query);
            BranchMaster branch = byId.get();
            return Collections.singletonList(new BranchDto(branch.getCode(), branch.getName()));
        }

   
        log.info("No exact code match found. Searching by name instead.");
        List<BranchMaster> byName = branchMasterRepository.findByNameStartingWithIgnoreCase(query);

      
        return byName.stream()
                .map(branch -> new BranchDto(branch.getCode(), branch.getName()))
                .collect(Collectors.toList());
    }
}

package com.tcs.fincore.MOCService.Service;


import lombok.RequiredArgsConstructor;
import lombok.extern.slf4j.Slf4j;
import org.springframework.stereotype.Service;

import org.springframework.util.StringUtils;
import com.tcs.fincore.MOCService.Dto.CglDto;
import com.tcs.fincore.MOCService.Models.CglMaster;
import com.tcs.fincore.MOCService.Models.enums.AcClassification;
import com.tcs.fincore.MOCService.Repository.CglMasterRepository;

import java.util.Arrays;
import java.util.List;
import java.util.Map;
import java.util.stream.Collectors;

@Service
@RequiredArgsConstructor
@Slf4j
public class CglService {

	  private final CglMasterRepository cglMasterRepository;

      private CglDto convertToDto(CglMaster cgl) {
          return new CglDto(cgl.getCglNumber(), cgl.getDescription());
      }

  
      public List<CglDto> getCglsByClassification(String classification) {
          return cglMasterRepository.findByAcClassification(classification)
                  .stream()
                  .map(this::convertToDto)
                  .collect(Collectors.toList());
      }

      public List<CglDto> searchCglsByNumber(String query) {
          return cglMasterRepository.findByCglNumberContaining(query)
                  .stream()
                  .map(this::convertToDto)
                  .collect(Collectors.toList());
      }

     
      public List<CglDto> searchCgls(String query, String classification) {
        
          boolean hasQuery = StringUtils.hasText(query); 
          boolean hasClassification = StringUtils.hasText(classification);
          List<CglMaster> results;

          if (hasQuery && hasClassification) {
              log.info("Searching CGLs by query '{}' and classification '{}'", query, classification);
              results = cglMasterRepository.findByCglNumberContainingAndAcClassification(query, classification);
          } else if (!hasQuery && hasClassification) {
              log.info("Searching CGLs by classification '{}' only", classification);
              results = cglMasterRepository.findByAcClassification(classification);
          } else if (hasQuery && !hasClassification) {
              log.info("Searching CGLs by query '{}' only", query);
              results = cglMasterRepository.findByCglNumberContaining(query);
          } else {
              log.info("No CGL query or classification provided.");
              return List.of(); 
          }
          return results.stream()
                  .map(this::convertToDto)
                  .collect(Collectors.toList());
         
      }
    
      public Map<String, String> getClassificationTypes() {
          return Arrays.stream(AcClassification.values())
                  .collect(Collectors.toMap(
                        
                          ac -> String.valueOf(ac.getCode()),
                          AcClassification::getDescription
                  ));
      }
}

import com.fasterxml.jackson.databind.ObjectMapper;
import org.springframework.util.StringUtils;
import java.util.ArrayList;
import com.tcs.fincore.MOCService.Dto.BulkProcessMocRequestDto;
import com.tcs.fincore.MOCService.Dto.CreateMocRequestDto;
import com.tcs.fincore.MOCService.Dto.ProcessMocRequestDto;
import com.tcs.fincore.MOCService.Exception.ResourceNotFoundException;
import com.tcs.fincore.MOCService.Models.MasterMoc;
import com.tcs.fincore.MOCService.Models.MocRequest;
import com.tcs.fincore.MOCService.Models.enums.RequestStatus;
import com.tcs.fincore.MOCService.Repository.MasterMocRepository;
import com.tcs.fincore.MOCService.Repository.MocRequestRepository;

import lombok.RequiredArgsConstructor;
import lombok.extern.slf4j.Slf4j;
import org.springframework.dao.DataIntegrityViolationException;
import org.springframework.stereotype.Service;
import jakarta.transaction.Transactional;
import java.time.LocalDateTime;
import java.util.Collections;
import java.util.List;
import java.util.Optional;

@Service
@RequiredArgsConstructor
@Slf4j
public class MocRequestServiceImpl implements MocRequestService {

    private final MocRequestRepository mocRequestRepository;
    private final MasterMocRepository masterMocRepository;
    private final ObjectMapper objectMapper;

    @Override
    @Transactional
    public MocRequest createRequest(CreateMocRequestDto dto, String creatorId) throws JsonProcessingException {
        
        String targetId = dto.getBranch() + "-" + dto.getCgl() + "-" + dto.getPDate();
        
        List<MocRequest> existing = mocRequestRepository.findByTargetIdAndRequestStatus(targetId, RequestStatus.PENDING);
        if (!existing.isEmpty()) {
            throw new DataIntegrityViolationException("A pending request for this specific MOC record (Branch-CGL-Date) already exists.");
        }

        MocRequest request = new MocRequest();
        request.setCreatorId(creatorId);
        request.setChangeType(dto.getChangeType());
        request.setRequestStatus(RequestStatus.PENDING);
        request.setTargetId(targetId);
        request.setBatchId(dto.getBatchId());
   
       
        request.setJournalId(dto.getJournalId()); 
        request.setCommonBatchRemarks(dto.getCommonBatchRemarks());
      
   
        request.setPayload(objectMapper.writeValueAsString(dto)); 

        log.info("Saving new MOC request (Batch ID: {}, Journal ID: {}) for targetId {}", dto.getBatchId(), dto.getJournalId(), targetId);
        return mocRequestRepository.save(request);
    }

  

    @Override
    @Transactional
    public Optional<MocRequest> updateRequestStatus(ProcessMocRequestDto dto, String executorId) throws JsonProcessingException {
        MocRequest request = mocRequestRepository.findById(dto.getRequestId())
                .orElseThrow(() -> new ResourceNotFoundException("MOC Request with ID " + dto.getRequestId() + " not found."));

        if (executorId.equals(request.getCreatorId())) {
            throw new IllegalStateException("User cannot approve or reject their own MOC request.");
        }

        if (request.getRequestStatus() != RequestStatus.PENDING) {
            throw new IllegalStateException("MOC Request has already been processed.");
        }

        request.setExecutorId(executorId);
        request.setExecutorRemarks(dto.getRemarks());
        request.setExecutionDate(LocalDateTime.now());
        request.setRequestStatus(dto.getStatus());

        if (RequestStatus.ACCEPTED.equals(dto.getStatus())) {
            log.info("MOC Request {} ACCEPTED. Processing for Batch ID: {}", dto.getRequestId(), request.getBatchId());
            processApproval(request); 
        } else {
            log.info("MOC Request {} REJECTED.", dto.getRequestId());
        }

        return Optional.of(mocRequestRepository.save(request));
    }
    
    private void processApproval(MocRequest request) throws JsonProcessingException {
        CreateMocRequestDto payloadDto = objectMapper.readValue(request.getPayload(), CreateMocRequestDto.class);

        switch (request.getChangeType()) {
            case ADD:
                MasterMoc newMoc = new MasterMoc();
                mapDtoToEntity(payloadDto, newMoc);
                masterMocRepository.save(newMoc);
                log.info("Successfully ADDED new MOC record to master table (Batch ID: {}, Journal ID: {}).", payloadDto.getBatchId(), payloadDto.getJournalId());
                break;

            case UPDATE:
                MasterMoc existingMoc = masterMocRepository.findById(payloadDto.getMasterMocId())
                        .orElseThrow(() -> new ResourceNotFoundException("MasterMOC with ID " + payloadDto.getMasterMocId() + " not found for update."));
                mapDtoToEntity(payloadDto, existingMoc); 
                masterMocRepository.save(existingMoc);
                log.info("Successfully UPDATED MOC record in master table (Batch ID: {}, Journal ID: {}).", payloadDto.getBatchId(), payloadDto.getJournalId());
                break;

            case DELETE:
                masterMocRepository.deleteById(payloadDto.getMasterMocId());
                log.info("Successfully DELETED MOC record from master table (Was Batch ID: {}, Journal ID: {}).", payloadDto.getBatchId(), payloadDto.getJournalId());
                break;
        }
    }

    private void mapDtoToEntity(CreateMocRequestDto dto, MasterMoc entity) {
        entity.setPDate(dto.getPDate());
        entity.setBranch(dto.getBranch());
        entity.setCurrency(dto.getCurrency());
        entity.setCgl(dto.getCgl());
        entity.setAmount(dto.getAmount());
        entity.setTransactionType(dto.getTransactionType());
        entity.setProductType(dto.getProductType());
        entity.setRemarks(dto.getRemarks());
        entity.setArFlag(dto.getArFlag());
        entity.setAcClassification(dto.getAcClassification());
        entity.setBatchId(dto.getBatchId());
        entity.setJournalId(dto.getJournalId());
        entity.setCommonBatchRemarks(dto.getCommonBatchRemarks());
    }

    @Override
    public List<MocRequest> getMyRequests(String userId) {
        return mocRequestRepository.findByCreatorId(userId);
    }

    @Override
    public List<MocRequest> getPendingRequests() {
        return mocRequestRepository.findByRequestStatus(RequestStatus.PENDING);
    }
    
    @Override
    @Transactional
    public List<MocRequest> processBulkRequests(BulkProcessMocRequestDto dto, String executorId) throws JsonProcessingException {

        boolean hasBatchId = StringUtils.hasText(dto.getBatchId());
        boolean hasJournalIds = dto.getJournalIds() != null && !dto.getJournalIds().isEmpty();

        
        if (!hasBatchId && !hasJournalIds) {
            throw new IllegalArgumentException("Either batchId or a list of journalIds must be provided.");
        }
        if (hasBatchId && hasJournalIds) {
            throw new IllegalArgumentException("Provide either batchId or journalIds, not both.");
        }

        // 2. Find all PENDING requests based on the criteria
        List<MocRequest> requestsToProcess;
        if (hasBatchId) {
            log.info("Processing bulk request by Batch ID: {} for user: {}", dto.getBatchId(), executorId);
            requestsToProcess = mocRequestRepository.findByBatchIdAndRequestStatus(dto.getBatchId(), RequestStatus.PENDING);
        } else {
            log.info("Processing bulk request by Journal IDs: {} for user: {}", dto.getJournalIds().size(), executorId);
            requestsToProcess = mocRequestRepository.findByJournalIdInAndRequestStatus(dto.getJournalIds(), RequestStatus.PENDING);
        }

        if (requestsToProcess.isEmpty()) {
            log.warn("No pending requests found for the given criteria.");
            return Collections.emptyList();
        }

        List<MocRequest> processedRequests = new ArrayList<>();

        
        for (MocRequest request : requestsToProcess) {
            
            // Self-approval check
            if (executorId.equals(request.getCreatorId())) {
                log.warn("Skipping request {}: User {} cannot approve/reject their own request.", request.getId(), executorId);
                continue; 
            }

          
            if (request.getRequestStatus() != RequestStatus.PENDING) {
                log.warn("Skipping request {}: It is not in PENDING status.", request.getId());
                continue;
            }

            // 4. Update the request entity
            request.setExecutorId(executorId);
            request.setExecutorRemarks(dto.getRemarks());
            request.setExecutionDate(LocalDateTime.now());
            request.setRequestStatus(dto.getStatus());

        
            if (RequestStatus.ACCEPTED.equals(dto.getStatus())) {
                log.info("MOC Request {} ACCEPTED. Processing...", request.getId());
                processApproval(request); 
            } else {
                log.info("MOC Request {} REJECTED.", request.getId());
            }

            processedRequests.add(request);
        }

     
        return mocRequestRepository.saveAll(processedRequests);
    }
    
    @Override
    public List<MocRequest> getRequestsByBatchId(String batchId) {
        log.info("Fetching all MOC requests for Batch ID: {}", batchId);
        List<MocRequest> requests = mocRequestRepository.findByBatchId(batchId);
        
        if (requests.isEmpty()) {
            log.warn("No requests found for Batch ID: {}", batchId);
          
            throw new ResourceNotFoundException("No MOC requests found for Batch ID: " + batchId);
        }
        
      
        return requests;
    }
}

frontend
import React, { useState, useEffect, useCallback } from "react";
import {
  Box,
  Typography,
  Grid,
  TextField,
  Button,
  Paper,
  IconButton,
  MenuItem,
  CircularProgress,
  Autocomplete,
} from "@mui/material";
import AddIcon from "@mui/icons-material/Add";
import DeleteIcon from "@mui/icons-material/Delete";
import { debounce } from "lodash";

import useApi from "../../../hooks/useApi";
import useCustomSnackbar from "../../../utils/useCustomSnackbar";

const initialRowState = {
  id: Date.now(),
  branch: null,
  postDate: new Date().toISOString().split("T")[0],
  currency: "",
  head: "",
  cgl: null,
  amount: "",
  txnType: "Debit",
  remarks: "", 
  productCode: "",
};

export default function MOCPosting() {
  const { callApi } = useApi();
  const showSnackBar = useCustomSnackbar();

  const [rows, setRows] = useState([initialRowState]);
  const [currencyOptions, setCurrencyOptions] = useState([]);
  const [headOptions, setHeadOptions] = useState([]);
  const [isSubmitting, setIsSubmitting] = useState(false);

  
  const [commonBatchRemarks, setCommonBatchRemarks] = useState("");

  const [branchSearch, setBranchSearch] = useState({
    loading: false,
    options: [],
  });
  const [cglSearch, setCglSearch] = useState({ loading: false, options: [] });

  useEffect(() => {
    const fetchInitialData = async () => {
      try {
        const currencyResponse = await callApi("CM/currency", null, "GET");
        if (currencyResponse?.success && Array.isArray(currencyResponse.data)) {
          const uniqueCurrencies = [...new Set(currencyResponse.data)];
          setCurrencyOptions(uniqueCurrencies);
        } else {
          showSnackBar("Failed to load currencies.", "error");
        }

        const headResponse = await callApi(
          "MS/api/cgls/classifications",
          null,
          "GET"
        );

        if (headResponse) {
          const formattedHeads = Object.entries(headResponse).map(
            ([key, value]) => ({ key, value })
          );
          setHeadOptions(formattedHeads);
        } else {
          showSnackBar("Failed to load head classifications.", "error");
        }
      } catch (error) {
        console.error("Error fetching initial data:", error);
        showSnackBar("An error occurred while loading page data.", "error");
      }
    };

    fetchInitialData();
  }, [callApi, showSnackBar]);

  const handleAddRow = () => {
    setRows([...rows, { ...initialRowState, id: Date.now() }]);
  };

  const handleDeleteRow = (id) => {
    if (rows.length > 1) {
      setRows(rows.filter((row) => row.id !== id));
    } else {
      showSnackBar("Cannot delete the last row.", "warning");
    }
  };

  const handleRowChange = (id, field, value) => {
    const newRows = rows.map((row) => {
      if (row.id === id) {
        if (field === "head" && row.head !== value) {
          setCglSearch({ loading: false, options: [] });
          searchCgls("", value);
          return { ...row, [field]: value, cgl: null };
        }
        return { ...row, [field]: value };
      }
      return row;
    });
    setRows(newRows);
  };

  const searchBranches = useCallback(
    debounce(async (query) => {
      if (!query) {
        setBranchSearch({ loading: false, options: [] });
        return;
      }
      setBranchSearch((prev) => ({ ...prev, loading: true }));
      try {
        const response = await callApi(
          `MS/branches/search?query=${encodeURIComponent(query)}`,
          null,
          "GET"
        );
        setBranchSearch({ loading: false, options: response || [] });
      } catch (error) {
        console.error("Branch search failed:", error);
        showSnackBar("Branch search failed. See console.", "error");
        setBranchSearch({ loading: false, options: [] });
      }
    }, 300),
    [callApi, showSnackBar]
  );

  const searchCgls = useCallback(
    debounce(async (query, headKey) => {
      if (!headKey) {
        setCglSearch({ loading: false, options: [] });
        return;
      }
      setCglSearch((prev) => ({ ...prev, loading: true }));
      try {
        const response = await callApi(
          `MS/api/cgls/search?query=${encodeURIComponent(
            query
          )}&classification=${encodeURIComponent(headKey)}`,
          null,
          "GET"
        );
        setCglSearch({ loading: false, options: response || [] });
      } catch (error) {
        console.error("CGL search failed:", error);
        showSnackBar("CGL search failed. See console.", "error");
        setCglSearch({ loading: false, options: [] });
      }
    }, 300),
    [callApi, showSnackBar]
  );

  
  const handleSubmit = async () => {
    setIsSubmitting(true);

    
    if (!commonBatchRemarks.trim()) {
      showSnackBar(
        "Please enter Common Batch Remarks before submitting.",
        "error"
      );
      setIsSubmitting(false);
      return;
    }

    const batchId = String(Date.now());
    console.log("Generated Batch ID:", batchId);

    const branchTotals = new Map();
    let journalCounter = 0;

    
    for (const [index, row] of rows.entries()) {
      if (
        !row.branch ||
        !row.currency ||
        !row.head ||
        !row.cgl ||
        !row.amount ||
        !row.txnType ||
        !row.productCode
      ) {
        showSnackBar(
          `Row ${index + 1} is incomplete. Please fill all fields.`,
          "error"
        );
        setIsSubmitting(false);
        return;
      }

      const branchCode = row.branch.code;
      const branchName = row.branch.name;

      // Totals calculation remains per-branch
      if (!branchTotals.has(branchCode)) {
        branchTotals.set(branchCode, {
          totalDebit: 0,
          totalCredit: 0,
          hasDebit: false,
          hasCredit: false,
          branchName: branchName,
        });
      }

      const totals = branchTotals.get(branchCode);
      const amount = parseFloat(row.amount) || 0;

      if (row.txnType === "Debit") {
        totals.totalDebit += amount;
        totals.hasDebit = true;
      } else if (row.txnType === "Credit") {
        totals.totalCredit += amount;
        totals.hasCredit = true;
      }
    }

    // Validate balancing per branch
    for (const [branchCode, totals] of branchTotals.entries()) {
      const { totalDebit, totalCredit, hasDebit, hasCredit, branchName } =
        totals;
      if (!hasDebit || !hasCredit) {
        showSnackBar(
          `Submission failed. Branch "${branchName}" (${branchCode}) must have at least one Debit and one Credit entry.`,
          "error"
        );
        setIsSubmitting(false);
        return;
      }
      const EPSILON = 0.000001;
      if (Math.abs(totalDebit - totalCredit) > EPSILON) {
        showSnackBar(
          `Submission failed. Totals for branch "${branchName}" (${branchCode}) do not match. Debit: ${totalDebit.toFixed(
            2
          )}, Credit: ${totalCredit.toFixed(2)}`,
          "error"
        );
        setIsSubmitting(false);
        return;
      }
    }

    let successfulSubmissions = 0;
    let failedSubmissions = 0;

    for (const [index, row] of rows.entries()) {
      journalCounter++;
      const sequenceNumber = String(journalCounter).padStart(3, "0");
      const journalId = `${batchId}-${sequenceNumber}`;
      console.log(`Row ${index + 1}: Generated Journal ID: ${journalId}`);

      const payload = {
        changeType: "ADD",
        pDate: row.postDate,
        branch: row.branch.code,
        currency: row.currency,
        cgl: row.cgl.cglNumber,
        amount: parseFloat(row.amount),
        transactionType: row.txnType,
        productType: row.productCode,
        remarks: row.remarks, 
        arFlag: "A",
        acClassification: row.head,
        batchId: batchId,
        journalId: journalId,

        
        commonBatchRemarks: commonBatchRemarks.trim(),
      };

      try {
        await callApi("MS/moc/create-request", payload, "POST");
        successfulSubmissions++;
      } catch (error) {
        console.error(
          `Failed to submit row ${index + 1} (Journal ID: ${journalId}):`,
          error
        );
        const errorMsg =
          error.response?.data?.message || error.message || "Unknown error";
        showSnackBar(
          `Failed to submit row ${
            index + 1
          } (Journal ID: ${journalId}): ${errorMsg}`,
          "error"
        );
        failedSubmissions++;
      }
    }

    if (successfulSubmissions > 0 && failedSubmissions === 0) {
      showSnackBar(
        `All ${successfulSubmissions} MOC entries submitted successfully! Batch ID: ${batchId}`,
        "success"
      );
      setRows([{ ...initialRowState, id: Date.now() }]);
  
      setCommonBatchRemarks("");
    } else if (successfulSubmissions > 0 && failedSubmissions > 0) {
      showSnackBar(
        `Partially successful for Batch ID ${batchId}: ${successfulSubmissions} submitted, ${failedSubmissions} failed.`,
        "warning"
      );
    } else if (successfulSubmissions === 0 && failedSubmissions > 0) {
      showSnackBar(
        `Submission failed for Batch ID ${batchId}. Please check errors and try again.`,
        "error"
      );
    }

    setIsSubmitting(false);
  };

  return (
    <Box sx={{ p: 3, maxWidth: "98%", margin: "auto" }}>
      <Paper elevation={3} sx={{ p: 3 }}>
        <Typography
          variant="h4"
          gutterBottom
          sx={{ fontWeight: "bold", color: "primary.main", mb: 3 }}
        >
          MOC Posting
        </Typography>

      
        <TextField
          label="Common Batch Remarks"
          value={commonBatchRemarks}
          onChange={(e) => {
          
            if (e.target.value.length <= 200) {
              setCommonBatchRemarks(e.target.value);
            }
          }}
          fullWidth
          required
          multiline
          rows={2}
          placeholder="Enter remarks for the entire batch (max 200 chars)..."
          InputProps={{
            inputProps: {
              maxLength: 200,
            },
          }}
          sx={{ mb: 3 }} 
        />

        {rows.map((row, index) => (
          <React.Fragment key={row.id}>
            <Grid container spacing={2} alignItems="center">
              {/* Branch */}
              <Grid item xs={12} md={3}>
                <Autocomplete
                  options={branchSearch.options}
                  getOptionLabel={(option) => `${option.code} - ${option.name}`}
                  value={row.branch}
                  onChange={(event, newValue) =>
                    handleRowChange(row.id, "branch", newValue)
                  }
                  onInputChange={(event, newInputValue) =>
                    searchBranches(newInputValue)
                  }
                  loading={branchSearch.loading}
                  filterOptions={(x) => x}
                  isOptionEqualToValue={(option, value) =>
                    option.code === value.code
                  }
                  renderInput={(params) => (
                    <TextField
                      {...params}
                      label="Branch"
                      placeholder="Search by name or code"
                      InputProps={{
                        ...params.InputProps,
                        endAdornment: (
                          <Box
                            sx={{
                              minWidth: 24,
                              minHeight: 24,
                              display: "flex",
                              alignItems: "center",
                              justifyContent: "center",
                            }}
                          >
                            {branchSearch.loading ? (
                              <CircularProgress color="inherit" size={20} />
                            ) : null}
                            {params.InputProps.endAdornment}
                          </Box>
                        ),
                      }}
                    />
                  )}
                />
              </Grid>

              {/* Post Date */}
              <Grid item xs={12} md={1.5}>
                <TextField
                  label="Post Date"
                  value={row.postDate}
                  fullWidth
                  InputProps={{
                    readOnly: true,
                  }}
                />
              </Grid>

              {/* Currency */}
              <Grid item xs={12} md={1.5}>
                <TextField
                  select
                  label="Currency"
                  value={row.currency}
                  onChange={(e) =>
                    handleRowChange(row.id, "currency", e.target.value)
                  }
                  fullWidth
                  sx={{ minWidth: 140 }}
                >
                  {currencyOptions.map((c) => (
                    <MenuItem key={c} value={c}>
                      {c}
                    </MenuItem>
                  ))}
                </TextField>
              </Grid>

              {/* Head */}
              <Grid item xs={12} md={2}>
                <TextField
                  select
                  label="Head"
                  value={row.head}
                  onChange={(e) =>
                    handleRowChange(row.id, "head", e.target.value)
                  }
                  fullWidth
                  sx={{ minWidth: 160 }}
                >
                  {headOptions.map((h) => (
                    <MenuItem key={h.key} value={h.key}>
                      {h.value}
                    </MenuItem>
                  ))}
                </TextField>
              </Grid>

              {/* CGL */}
              <Grid item xs={12} md={3}>
                <Autocomplete
                  options={cglSearch.options}
                  getOptionLabel={(option) =>
                    `${option.cglNumber} - ${option.description}`
                  }
                  value={row.cgl}
                  onChange={(event, newValue) =>
                    handleRowChange(row.id, "cgl", newValue)
                  }
                  onInputChange={(event, newInputValue) =>
                    searchCgls(newInputValue, row.head)
                  }
                  loading={cglSearch.loading}
                  disabled={!row.head}
                  filterOptions={(x) => x}
                  isOptionEqualToValue={(option, value) =>
                    option.cglNumber === value.cglNumber
                  }
                  renderInput={(params) => (
                    <TextField
                      {...params}
                      label="CGL"
                      placeholder={
                        row.head ? "Search CGL" : "Select Head first"
                      }
                      InputProps={{
                        ...params.InputProps,
                        endAdornment: (
                          <Box
                            sx={{
                              minWidth: 24,
                              minHeight: 24,
                              display: "flex",
                              alignItems: "center",
                              justifyContent: "center",
                            }}
                          >
                            {cglSearch.loading ? (
                              <CircularProgress color="inherit" size={20} />
                            ) : null}
                            {params.InputProps.endAdornment}
                          </Box>
                        ),
                      }}
                    />
                  )}
                />
              </Grid>

              {/* Amount */}
              <Grid item xs={12} md={1.5}>
                <TextField
                  label="Amount"
                  type="text"
                  value={row.amount}
                  onChange={(e) => {
                    const val = e.target.value;
                    if (/^\d{0,16}(\.\d{0,6})?$/.test(val)) {
                      handleRowChange(row.id, "amount", val);
                    }
                  }}
                  fullWidth
                  placeholder="0.00"
                />
              </Grid>

              {/* Txn Type */}
              <Grid item xs={12} md={1}>
                <TextField
                  select
                  label="Txn Type"
                  value={row.txnType}
                  onChange={(e) =>
                    handleRowChange(row.id, "txnType", e.target.value)
                  }
                  fullWidth
                  sx={{ minWidth: 90 }}
                >
                  <MenuItem value="Debit">Debit</MenuItem>
                  <MenuItem value="Credit">Credit</MenuItem>
                </TextField>
              </Grid>

              {/* Product Code */}
              <Grid item xs={12} md={1.5}>
                <TextField
                  label="Product Code"
                  value={row.productCode}
                  onChange={(e) => {
                    const val = e.target.value;
                    if (/^\d{0,8}$/.test(val)) {
                      handleRowChange(row.id, "productCode", val);
                    }
                  }}
                  fullWidth
                  InputProps={{
                    inputProps: {
                      maxLength: 8,
                    },
                  }}
                />
              </Grid>

              {/* Remarks (Individual) */}
              <Grid item xs={12} md={1.5}>
                <TextField
                  label="Remarks"
                  value={row.remarks}
                  onChange={(e) => {
                    const val = e.target.value;
                    if (val.length <= 40) {
                      handleRowChange(row.id, "remarks", val);
                    }
                  }}
                  fullWidth
                  InputProps={{
                    inputProps: {
                      maxLength: 40,
                    },
                  }}
                />
              </Grid>

              {/* Actions */}
              <Grid
                item
                xs={12}
                md={0.5}
                sx={{ textAlign: "center", minWidth: 50 }}
              >
                <IconButton
                  onClick={() => handleDeleteRow(row.id)}
                  color="error"
                  aria-label="delete row"
                >
                  <DeleteIcon />
                </IconButton>
              </Grid>
            </Grid>

            {index < rows.length - 1 && (
              <Box
                sx={{
                  borderBottom: 1,
                  borderColor: "divider",
                  my: 2,
                }}
              />
            )}
          </React.Fragment>
        ))}

        <Box sx={{ mt: 3, display: "flex", gap: 2 }}>
          <Button
            variant="contained"
            startIcon={<AddIcon />}
            onClick={handleAddRow}
          >
            Add Row
          </Button>
          <Button
            variant="contained"
            color="success"
            onClick={handleSubmit}
            disabled={isSubmitting}
          >
            {isSubmitting ? <CircularProgress size={24} /> : "Submit"}
          </Button>
        </Box>
      </Paper>
    </Box>
  );
}
