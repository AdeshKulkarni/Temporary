import random
import csv

# =====================================================================
# 1. PASTE YOUR RAW DATA HERE (NO QUOTES OR COMMAS NEEDED!)
# =====================================================================

RAW_BRANCH_LIST = """
10400
10001
10002
"""

RAW_OPPOSITE_CGL_LIST = """
5051030601
5051240801
1102003001
2115505002
"""

# =====================================================================
# 2. FIXED CURRENCY LIST
# =====================================================================
FIXED_CURRENCIES = [
    "USD", "AUD", "CAD", "CHF", "SGD", "HKD", 
    "NZD", "EUR", "GBP", "JPY", "DKK", "SEK", "INR"
]

# =====================================================================
# 3. FIXED CGL LIST (I = Income/Cr | E = Expense/Dr)
# =====================================================================
FIXED_CGLS = [
    {"cgl": "7016500301", "type": "I"}, {"cgl": "7016500401", "type": "I"},
    {"cgl": "7016500501", "type": "I"}, {"cgl": "7016501101", "type": "I"},
    {"cgl": "7016500101", "type": "I"}, {"cgl": "7049500101", "type": "I"},
    {"cgl": "7049500301", "type": "I"}, {"cgl": "7049500501", "type": "I"},
    {"cgl": "7049501101", "type": "I"}, {"cgl": "7049501501", "type": "I"},
    {"cgl": "7913505001", "type": "I"}, {"cgl": "7914505001", "type": "I"},
    
    {"cgl": "8041080670", "type": "E"}, {"cgl": "8041080680", "type": "E"},
    {"cgl": "8051080670", "type": "E"}, {"cgl": "8051080690", "type": "E"},
    {"cgl": "8051500170", "type": "E"}, {"cgl": "8051501170", "type": "E"},
    {"cgl": "8001500160", "type": "E"}, {"cgl": "8001500560", "type": "E"},
    {"cgl": "8041070660", "type": "E"}, {"cgl": "8057070601", "type": "E"},
    {"cgl": "8972505001", "type": "E"}, {"cgl": "8972030601", "type": "E"},
    {"cgl": "8972070601", "type": "E"}
]

# =====================================================================
# 4. CONFIGURATION
# =====================================================================
PRODUCT_CODE = "12345678"
REMARKS = "Bulk Script Entry"

# =====================================================================
# 5. DATA GENERATION LOGIC
# =====================================================================
def generate_data():
    print("\n--- Voucher Bulk Data Generator ---")
    
    try:
        user_input = input("Enter the total number of rows you want to generate (e.g., 100): ")
        total_rows = int(user_input.strip())
    except ValueError:
        print("Invalid number. Defaulting to 100 rows.")
        total_rows = 100
        
    total_pairs = max(1, total_rows // 2)

    print("\nParsing and segregating raw input lists...")
    branch_list = [b.strip() for b in RAW_BRANCH_LIST.splitlines() if b.strip()]
    opp_list = [c.strip() for c in RAW_OPPOSITE_CGL_LIST.splitlines() if c.strip()]
    
    if not branch_list or not opp_list:
        print("ERROR: Please paste at least one Branch and one CGL into the RAW lists.")
        return

    # NEW RULE: Rigorous Segregation to prevent mixing!
    fixed_real = [c for c in FIXED_CGLS if not c["cgl"].startswith("5")]
    fixed_memo = [c for c in FIXED_CGLS if c["cgl"].startswith("5")]
    
    opp_real = [c for c in opp_list if not c.startswith("5")]
    opp_memo = [c for c in opp_list if c.startswith("5")]

    print(f"Generating {total_pairs * 2} rows of perfectly balanced entries...")
    rows = []

    for i in range(total_pairs):
        branch = random.choice(branch_list)
        amount = random.randint(10, 999) * 100  
        
        # Check what kind of pairs we are actually capable of making
        can_do_real = bool(opp_real) and (bool(fixed_real) or len(opp_real) > 1)
        can_do_memo = bool(opp_memo) and (bool(fixed_memo) or len(opp_memo) > 1)
        
        choices = []
        if can_do_real: choices.append("REAL")
        if can_do_memo: choices.append("MEMO")
        
        if not choices:
            print("ERROR: Not enough CGLs to make valid matching pairs! Check your lists.")
            return
            
        pair_type = random.choice(choices)
        
        # Build the perfectly segregated pair
        if pair_type == "REAL":
            if fixed_real:
                cgl1_obj = random.choice(fixed_real)
                cgl2_str = random.choice(opp_real)
            else:
                cgl1_obj = {"cgl": random.choice(opp_real), "type": random.choice(["I", "E"])}
                cgl2_str = random.choice(opp_real)
        else:
            if fixed_memo:
                cgl1_obj = random.choice(fixed_memo)
                cgl2_str = random.choice(opp_memo)
            else:
                cgl1_obj = {"cgl": random.choice(opp_memo), "type": random.choice(["I", "E"])}
                cgl2_str = random.choice(opp_memo)
                
        cgl1 = cgl1_obj["cgl"]
        cgl2 = cgl2_str

        # NEW RULE: Force INR if either CGL is '2115505002'
        if cgl1 == "2115505002" or cgl2 == "2115505002":
            currency = "INR"
        else:
            currency = random.choice(FIXED_CURRENCIES)
        
        # Set Types
        if cgl1_obj["type"] == "I":
            leg1_type, leg2_type = "Cr", "Dr"
        else:
            leg1_type, leg2_type = "Dr", "Cr"
            
        row1 = {
            "Branch*": branch, "Currency*": currency, "CGL*": cgl1,
            "Amount*": amount, "Type*": leg1_type, "Remarks*": f"{REMARKS} {i+1}",
            "Product Code": PRODUCT_CODE
        }
        
        row2 = {
            "Branch*": branch, "Currency*": currency, "CGL*": cgl2,
            "Amount*": amount, "Type*": leg2_type, "Remarks*": f"{REMARKS} {i+1}",
            "Product Code": PRODUCT_CODE
        }
        
        rows.append(row1)
        rows.append(row2)

    # =====================================================================
    # 6. WRITE TO CSV (Download and Save As -> .xlsx before uploading)
    # =====================================================================
    output_filename = f"Voucher_Upload_{total_pairs * 2}_Rows.csv"
    print(f"Writing to CSV ({output_filename})...")
    
    headers = ["Branch*", "Currency*", "CGL*", "Amount*", "Type*", "Remarks*", "Product Code"]
    
    with open(output_filename, mode='w', newline='', encoding='utf-8') as file:
        writer = csv.DictWriter(file, fieldnames=headers)
        writer.writeheader()
        writer.writerows(rows)
        
    print(f"\nSUCCESS! Created {output_filename} with {len(rows)} rows.")
    print("NOTE: Download this CSV, open it in Excel, and 'Save As' an Excel Workbook (.xlsx) before uploading!")

if __name__ == "__main__":
    generate_data()
