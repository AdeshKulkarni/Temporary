package com.tcs.userservice.aspect;

import com.fasterxml.jackson.databind.ObjectMapper;
import com.tcs.userservice.model.UserLogs;
import com.tcs.userservice.repository.UserLogsRepository;
import lombok.RequiredArgsConstructor;
import lombok.extern.slf4j.Slf4j;
import org.aspectj.lang.JoinPoint;
import org.aspectj.lang.annotation.AfterReturning;
import org.aspectj.lang.annotation.Aspect;
import org.springframework.stereotype.Component;

import java.sql.Timestamp;
import java.time.Instant;
import java.util.Map;
import org.springframework.http.ResponseEntity;
import com.tcs.userservice.ResponseVO;


@Aspect
@Component
@Slf4j
@RequiredArgsConstructor
public class UserActivityLogger {

    private final UserLogsRepository userLogsRepository;
    private final ObjectMapper objectMapper;

    // 1. Log User Creation/Modification Requests
    @AfterReturning(
            pointcut = "execution(* com.tcs.userservice.service.UserRequestService.createNewRequest(..)) && args(request, userId)",
            returning = "result"
    )
    public void logUserRequest(JoinPoint joinPoint, Map<String, Object> request, String userId, Object result) {
        if (isSuccess(result)) {
            saveLog(userId, "USER_REQUEST", "CREATE/MODIFY Request Raised", request.toString());
        } else {
            log.debug("Skipping audit log for failed/conflicted user request.");
        }
    }

    // 2. Log User Approval/Rejection
    @AfterReturning(pointcut = "execution(* com.tcs.userservice.service.UserRequestService.acceptOrRejectUserRequest(..)) && args(dto, ip)", returning = "result")
    public void logUserApproval(JoinPoint joinPoint, Object dto, String ip, Object result) {
        // We can extract Approver ID from DTO if needed, or assume ContextHolder has it
        // For simplicity, just logging the event content
        saveLog("SYSTEM", "USER_APPROVAL", "User Request Processed", dto.toString());
    }

    // 2. Log Role Requests
    @AfterReturning(
            pointcut = "execution(* com.tcs.userservice.service.RoleRequestService.createNewRoleRequest(..)) && args(request, userId)",
            returning = "result"
    )
    public void logRoleRequest(JoinPoint joinPoint, Map<String, Object> request, String userId, Object result) {
        if (isSuccess(result)) {
            saveLog(userId, "ROLE_REQUEST", "Role Request Raised", request.toString());
        }
    }

    // 4. Log Role Approval
    @AfterReturning(pointcut = "execution(* com.tcs.userservice.service.RoleRequestService.acceptOrRejectRoleRequest(..)) && args(request, userId)", returning = "result")
    public void logRoleApproval(JoinPoint joinPoint, Map<String, Object> request, String userId, Object result) {
        saveLog(userId, "ROLE_APPROVAL", "Role Request Processed", request.toString());
    }

    // --- Helper to Save to DB ---
    private void saveLog(String userId, String actionType, String changeType, String newValue) {
        try {
            UserLogs logEntry = new UserLogs();
            logEntry.setUserId(userId);
            logEntry.setActionType(actionType);
            logEntry.setChangeType(changeType);
            logEntry.setActionTime(Timestamp.from(Instant.now()));
            // Truncate if too long to prevent DB errors
            if (newValue != null && newValue.length() > 3900) {
                newValue = newValue.substring(0, 3900) + "...";
            }
            logEntry.setNewValue(newValue);
            userLogsRepository.save(logEntry);
            log.info("✅ Auto-Logged User Activity: {} -> {}", userId, actionType);
        } catch (Exception e) {
            log.error("Failed to save audit log: {}", e.getMessage());
        }
    }

    // --- Helper Method to Determine Success ---
    private boolean isSuccess(Object result) {
        if (result instanceof ResponseEntity) {
            ResponseEntity<?> response = (ResponseEntity<?>) result;
            // Check HTTP Status (200 OK or 201 CREATED)
            if (!response.getStatusCode().is2xxSuccessful()) {
                return false;
            }
            // Optional: Dig deeper into your ResponseVO "status" boolean if needed
            if (response.getBody() instanceof ResponseVO) {
                ResponseVO<?> vo = (ResponseVO<?>) response.getBody();
                // if ResponseVO holds a Map with a "status" key,
                // but checking HTTP code is usually sufficient for these flow.
                return true;
            }
        }
        return false;
    }

}
