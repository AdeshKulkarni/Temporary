package com.fincore.JournalService.Aspect;

import com.fasterxml.jackson.databind.ObjectMapper;
import com.fincore.JournalService.Dto.BatchRequestDto;
import com.fincore.JournalService.Dto.BulkProcessJournalRequestDto;
import com.fincore.JournalService.Dto.ProcessJournalRequestDto;

import com.fincore.JournalService.Models.JournalRequest;
import com.fincore.JournalService.Repository.JournalLogRepository;
import jakarta.servlet.http.HttpServletRequest;
import lombok.RequiredArgsConstructor;
import lombok.extern.slf4j.Slf4j;
import org.aspectj.lang.JoinPoint;
import org.aspectj.lang.annotation.AfterReturning;
import org.aspectj.lang.annotation.Aspect;
import org.springframework.stereotype.Component;
import org.springframework.web.context.request.RequestContextHolder;
import org.springframework.web.context.request.ServletRequestAttributes;

import java.time.LocalDateTime;

@Aspect
@Component
@Slf4j
@RequiredArgsConstructor
public class JournalActivityLogger {

    private final JournalLogRepository journalLogRepository;
    private final ObjectMapper objectMapper;

    // --- 1. Log Batch Creation (Standard) ---
    @AfterReturning(
            pointcut = "execution(* com.fincore.JournalService.Service.JournalRequestService.createBatchRequest(..)) && args(batchDto, creatorId, creatorRole)",
            returning = "result"
    )
    public void logCreateBatch(JoinPoint joinPoint, BatchRequestDto batchDto, String creatorId, Integer creatorRole, Object result) {
        String summary = "Created Batch with " + (batchDto.getRows() != null ? batchDto.getRows().size() : 0) + " rows.";
        // Passing null for requestId is correct here, but saveLog must handle it safely
        saveLog(creatorId, "CREATE", "BATCH_CREATION", summary, null, null);
    }

    // --- 2. Log Bulk Batch Creation (Optimized Upload) ---
    @AfterReturning(
            pointcut = "execution(* com.fincore.JournalService.Service.JournalRequestService.createBulkBatchRequest(..)) && args(batchDto, creatorId, creatorRole)",
            returning = "batchId"
    )
    public void logBulkCreate(JoinPoint joinPoint, BatchRequestDto batchDto, String creatorId, Integer creatorRole, Object batchId) {
        String summary = "Bulk Upload Created. Batch ID: " + batchId + ". Rows: " + (batchDto.getRows() != null ? batchDto.getRows().size() : 0);
        saveLog(creatorId, "CREATE", "BULK_UPLOAD", summary, null, null);
    }

    // --- 3. Log Single Status Update (Approve/Reject) ---
    @AfterReturning(
            pointcut = "execution(* com.fincore.JournalService.Service.JournalRequestService.updateRequestStatus(..)) && args(dto, executorId, executorRole)",
            returning = "result"
    )
    public void logStatusUpdate(JoinPoint joinPoint, ProcessJournalRequestDto dto, String executorId, Integer executorRole, Object result) {
        if (result instanceof JournalRequest) {
            JournalRequest req = (JournalRequest) result; // The updated entity
            String newValue = "Status: " + dto.getStatus() + ", Remarks: " + dto.getRemarks();
            // We can treat 'changeType' as the Status (ACCEPTED/REJECTED)
            saveLog(executorId, dto.getStatus().toString(), "STATUS_UPDATE", newValue, null, req.getId());
        }
    }

    // --- 4. Log Bulk Status Update (Bulk Approve/Reject) ---
    @AfterReturning(
            pointcut = "execution(* com.fincore.JournalService.Service.JournalRequestService.processBulkRequests(..)) && args(dto, executorId, executorRole)",
            returning = "result"
    )
    public void logBulkStatusUpdate(JoinPoint joinPoint, BulkProcessJournalRequestDto dto, String executorId, Integer executorRole, Object result) {
        String target = dto.getBatchId() != null ? "Batch ID: " + dto.getBatchId() : "Journals: " + dto.getJournalIdPrefixes();
        String newValue = "Bulk " + dto.getStatus() + " for " + target + ". Remarks: " + dto.getRemarks();
        
        saveLog(executorId, "BULK_" + dto.getStatus(), "BULK_PROCESS", newValue, null, null);
    }

    // --- 5. Log Single Cancellation ---
    @AfterReturning(
            pointcut = "execution(* com.fincore.JournalService.Service.JournalRequestService.cancelMyRequest(..)) && args(requestId, userId)"
    )
    public void logCancelRequest(JoinPoint joinPoint, Long requestId, String userId) {
        saveLog(userId, "DELETE", "CANCEL_REQUEST", "User canceled request ID: " + requestId, null, requestId);
    }

    // --- 6. Log Bulk Cancellation (By Batch) ---
    @AfterReturning(
            pointcut = "execution(* com.fincore.JournalService.Service.JournalRequestService.cancelMyRequestsByBatchId(..)) && args(batchId, userId)"
    )
    public void logCancelBatch(JoinPoint joinPoint, String batchId, String userId) {
        saveLog(userId, "DELETE", "CANCEL_BATCH", "User canceled all requests in Batch: " + batchId, null, null);
    }

    // --- Helper to Save to DB ---
    private void saveLog(String userId, String actionType, String changeType, String newValue, String oldValue, Long requestId) {
        try {
            JournalLog logEntry = new JournalLog();
            
            logEntry.setUserId(userId);
            
            logEntry.setActionType(actionType);
            logEntry.setChangeType(changeType);
            logEntry.setActionTime(LocalDateTime.now());
            logEntry.setRequestId(requestId);
            
            // Capture IP Address
            logEntry.setIpAddress(getClientIp());

            // Convert Objects to JSON String if they aren't already strings
            if (newValue != null && !newValue.getClass().equals(String.class)) {
                newValue = objectMapper.writeValueAsString(newValue);
            }

            logEntry.setNewValue(truncate(newValue, 250));
            logEntry.setOldValue(truncate(oldValue, 250));



            journalLogRepository.save(logEntry);
            log.info("? Audit Log Saved: {} performed {} on {}", userId, actionType, (requestId != null ? requestId : "Batch"));

        } catch (Exception e) {
            log.error("Failed to save audit log: {}", e.getMessage());
        }
    }

    private String getClientIp() {
        try {
            HttpServletRequest request = ((ServletRequestAttributes) RequestContextHolder.currentRequestAttributes()).getRequest();
            String xForwardedFor = request.getHeader("X-Forwarded-For");
            if (xForwardedFor != null && !xForwardedFor.isEmpty()) {
                return xForwardedFor.split(",")[0].trim();
            }
            return request.getRemoteAddr();
        } catch (Exception e) {
            return "UNKNOWN";
        }
    }

    private String truncate(String value, int length) {
        if (value != null && value.length() > length) {
            return value.substring(0, length - 3) + "...";
        }
        return value;
    }
}

@RestController
@RequestMapping("journals")
@RequiredArgsConstructor
@Slf4j
public class JournalRequestController {

    private final JournalRequestService journalRequestService;
    private final JournalBulkValidationService journalBulkValidationService;
    private final JwtUtil jwtUtil;

    // --- BULK VALIDATION ENDPOINTS ---

    @PostMapping(value = "/bulk-validate-init", consumes = MediaType.MULTIPART_FORM_DATA_VALUE)
    public ResponseEntity<?> initiateValidation(@RequestParam("file") MultipartFile file,
                                                @RequestParam("postingDate") String date) {
        try {
            if(file.isEmpty()) return ResponseEntity.badRequest().body(Map.of("error", "File is empty"));
            LocalDate pDate = LocalDate.parse(date);
            String reqId = journalBulkValidationService.initiateValidation(file.getBytes(), file.getOriginalFilename(), pDate);
            return ResponseEntity.ok(Map.of("status", "QUEUED", "requestId", reqId));
        } catch (Exception e) {
            log.error("Validation init failed", e);
            return ResponseEntity.badRequest().body(Map.of("error", "Invalid Request: " + e.getMessage()));
        }
    }

    @GetMapping("/bulk-status/{requestId}")
    public ResponseEntity<BulkUploadStateDto> checkStatus(@PathVariable String requestId) {
        BulkUploadStateDto state = journalBulkValidationService.getState(requestId);
        if (state == null) return ResponseEntity.notFound().build();
        return ResponseEntity.ok(state);
    }

    @GetMapping("/download-bulk-file/{requestId}")
    public ResponseEntity<Resource> downloadFile(@PathVariable String requestId, @RequestParam String type) {
        byte[] data = journalBulkValidationService.getFileBytes(requestId, type);
        if (data == null) return ResponseEntity.notFound().build();

        String name = type.equals("ERROR") ? "Error_Report.xlsx" : "Success.csv";
        String mime = type.equals("ERROR") ? "application/vnd.openxmlformats-officedocument.spreadsheetml.sheet" : "text/csv";

        return ResponseEntity.ok()
                .header(HttpHeaders.CONTENT_DISPOSITION, "attachment; filename=\"" + name + "\"")
                .contentType(MediaType.parseMediaType(mime))
                .body(new ByteArrayResource(data));
    }

    @GetMapping("/download-template")
    public ResponseEntity<Resource> downloadTemplate() throws IOException {
        byte[] data = journalBulkValidationService.generateTemplateBytes();
        return ResponseEntity.ok()
                .header(HttpHeaders.CONTENT_DISPOSITION, "attachment; filename=\"Journal_Upload_Template.xlsx\"")
                .contentType(MediaType.parseMediaType("application/vnd.openxmlformats-officedocument.spreadsheetml.sheet"))
                .body(new ByteArrayResource(data));
    }

    // --- BATCH CREATION ENDPOINTS ---

    @PostMapping("/create-bulk-batch-optimized")
    public ResponseEntity<Map<String, Object>> createBulkBatchOptimized(@RequestBody Map<String, String> payload,
                                                                        @RequestHeader("Authorization") String token) {
        try {
            String userId = jwtUtil.getUserIdFromToken(token);
            Integer userRole = jwtUtil.getUserRoleFromToken(token);
            String requestId = payload.get("requestId");
            String remarks = payload.getOrDefault("commonBatchRemarks", "Bulk Upload");

            String batchId = journalRequestService.createBatchFromCache(requestId, remarks, userId, userRole);
            return ResponseEntity.ok(Map.of("status", "SUCCESS", "message", "Batch created successfully", "batchId", batchId));
        } catch (Exception e) {
            return ResponseEntity.status(HttpStatus.INTERNAL_SERVER_ERROR).body(Map.of("status", "ERROR", "message", e.getMessage()));
        }
    }

    @PostMapping("/create-bulk-batch")
    public ResponseEntity<Map<String, Object>> createBulkBatchRequest(@RequestBody BatchRequestDto batchDto,
                                                                      @RequestHeader("Authorization") String token) throws JsonProcessingException {
        try {
            String userId = jwtUtil.getUserIdFromToken(token);
            Integer userRole = jwtUtil.getUserRoleFromToken(token);
            String batchId = journalRequestService.createBulkBatchRequest(batchDto, userId, userRole);
            return ResponseEntity.ok(Map.of("status", "SUCCESS", "message", "Batch created successfully", "batchId", batchId));
        } catch (Exception e) {
            return ResponseEntity.status(HttpStatus.INTERNAL_SERVER_ERROR).body(Map.of("status", "ERROR", "message", e.getMessage()));
        }
    }

    @PostMapping("/create-batch")
    public List<JournalRequest> createBatchRequest(@Valid @RequestBody BatchRequestDto batchDto,

        String userId = jwtUtil.getUserIdFromToken(token);

    // --- APPROVAL / PROCESS ENDPOINTS ---

    @PostMapping("/process-bulk")

    }

@Service

@Slf4j
public class JournalBulkValidationService {

    private final BranchMasterRepository branchRepo;
    private final CglMasterRepository cglRepo;
    private final CurrencyMasterRepository currencyRepo;
    private final StringRedisTemplate redisTemplate;
    private final ObjectMapper objectMapper;

    @Autowired
    @Lazy
    private JournalBulkValidationService self;

    private static final Pattern CLEAN_AMOUNT_REGEX = Pattern.compile("[^0-9.]");
    private static final Pattern PRODUCT_CODE_REGEX = Pattern.compile("^\\d{8}$");
    private static final Pattern CGL_FORMAT_REGEX = Pattern.compile("^\\d{10}$");
    private static final DateTimeFormatter SYSTEM_DATE_FMT = DateTimeFormatter.ofPattern("ddMMyyyy");

    private static final String KEY_META = "journal:upload:meta:";
    private static final String KEY_FILE_ERR = "journal:upload:file:err:";
    private static final String KEY_FILE_SUC = "journal:upload:file:suc:";
    private static final String KEY_DATA_CACHE = "journal:upload:data:";
    private static final long TTL_MINUTES = 60;

    public static class ExcelRowData {
        public int rowIndex;
        public String site="", dateRaw="", year="", period=""; // New fields for System file reconstruction
        public String branch = "", currency = "", cgl = "", txnType = "", remarks = "", productCode = "";
        public BigDecimal amount;
        public boolean isSystemFormat = false;
        public List<String> errors = new ArrayList<>();
        public boolean hasErrors() { return !errors.isEmpty(); }
        public void addError(String err) { if(!errors.contains(err)) errors.add(err); }
    }

    public String initiateValidation(byte[] fileBytes, String filename, LocalDate postingDate) throws IOException {
        String requestId = UUID.randomUUID().toString();
        BulkUploadStateDto state = new BulkUploadStateDto();
        state.setRequestId(requestId);
        state.setStatus("PROCESSING");
        state.setCurrentStage(1);
        state.setMessage("Initializing Upload...");
        state.setTotalRows(0);
        saveStateToRedis(state);

        self.processAsync(requestId, fileBytes, filename, postingDate);
        return requestId;
    }

    @Async("bulkExecutor")
    public void processAsync(String requestId, byte[] fileBytes, String filename, LocalDate postingDate) {
        try {
            updateState(requestId, s -> { s.setMessage("Parsing File..."); s.setCurrentStage(1); });

            List<ExcelRowData> rows;
            boolean isCsv = filename != null && (filename.toLowerCase().endsWith(".csv") || filename.toLowerCase().endsWith(".txt"));

            if (isCsv) rows = parseCsvBytes(fileBytes);
            else rows = parseExcelBytes(fileBytes);

            if (rows.isEmpty()) throw new IllegalArgumentException("File contains no valid data rows.");

            updateState(requestId, s -> { s.setTotalRows(rows.size()); s.setMessage("Validating Database..."); s.setCurrentStage(2); });

            validateAll(rows, postingDate, requestId);

            boolean hasErrors = rows.stream().anyMatch(ExcelRowData::hasErrors);

            if (hasErrors) {
                // Generates error file containing ALL rows (with error column filled where applicable)
                failRequest(requestId, rows, "Validation Failed. Please correct the errors in the downloaded report and re-upload.", 1);
            } else {
                completeRequest(requestId, rows, postingDate, isCsv);
            }

        } catch (Exception e) {
            log.error("Async Validation Error for {}", requestId, e);
            updateState(requestId, s -> {
                s.setStatus("ERROR");
                s.setMessage("System Error: " + e.getMessage());
                s.setHasErrorFile(false);
            });
        }
    }

    private void validateAll(List<ExcelRowData> rows, LocalDate uiDate, String requestId) {
        // Fetch Reference Data
        Set<String> branches = rows.stream().map(r -> r.branch).collect(Collectors.toSet());
        Set<String> cgls = rows.stream().map(r -> r.cgl).collect(Collectors.toSet());
        Set<String> currs = rows.stream().map(r -> r.currency).collect(Collectors.toSet());

        Set<String> validBranches = branchRepo.findAllById(branches).stream().map(BranchMaster::getCode).collect(Collectors.toSet());
        Set<String> validCgls = cglRepo.findAllById(cgls).stream().map(CglMaster::getCglNumber).collect(Collectors.toSet());
        Set<String> validCurrs = currencyRepo.findAllById(currs).stream().map(CurrencyMaster::getCurrencyCode).collect(Collectors.toSet());

        ConcurrentHashMap<String, BigDecimal> balanceMap = new ConcurrentHashMap<>();
        String uiDateStr = uiDate.format(SYSTEM_DATE_FMT);

        // --- ROW LEVEL VALIDATION (Format + DB + Date) ---
        // Using parallelStream for speed, but ensure list integrity
        rows.parallelStream().forEach(d -> {

            // 1. Format Checks
            if (d.branch == null || d.branch.isEmpty()) d.addError("Branch is Empty");
            if (d.currency == null || d.currency.length() != 3) d.addError("Invalid Currency");

            if (d.cgl == null || d.cgl.isEmpty()) d.addError("CGL is Empty");
            else if (!CGL_FORMAT_REGEX.matcher(d.cgl).matches()) d.addError("Invalid CGL Format (10 digits required)");

            if (d.amount == null || d.amount.compareTo(BigDecimal.ZERO) == 0) d.addError("Amount cannot be Zero");

            if (d.amount != null && "INR".equalsIgnoreCase(d.currency)) {
                if (d.amount.stripTrailingZeros().scale() > 2) d.addError("INR Amount allows max 2 decimal places");
            }

            if (d.productCode != null && !d.productCode.isEmpty() && !d.productCode.equals("A")) {
                if (!PRODUCT_CODE_REGEX.matcher(d.productCode).matches()) d.addError("Product Code must be 'A' or 8 digits");
            }

            // 2. Date Check (Independent of other checks)
            if (d.isSystemFormat) {
                if (d.dateRaw == null || !d.dateRaw.equals(uiDateStr)) {
                    d.addError("Date Mismatch: File date (" + (d.dateRaw==null?"null":d.dateRaw) + ") != Selected (" + uiDateStr + ")");
                }
                if (!"01".equals(d.site)) d.addError("Site ID must be '01'");
            }

            // 3. Database Checks (Independent of format checks to catch multiple errors)
            if (!d.branch.isEmpty() && !validBranches.contains(d.branch)) d.addError("Branch Not Found");
            if (!d.currency.isEmpty() && !validCurrs.contains(d.currency)) d.addError("Currency Not Found");

            if (!d.cgl.isEmpty() && CGL_FORMAT_REGEX.matcher(d.cgl).matches() && !validCgls.contains(d.cgl)) {
                String type = d.cgl.startsWith("5") ? "Memo CGL" : "CGL";
                d.addError(type + " Not Found/Mismatch");
            }

            // 4. Accumulate Balances (Only if amount valid)
            if (d.amount != null && d.errors.isEmpty()) {
                // Only sum if row is valid so far, otherwise balance check is noisy
                String type = d.cgl.startsWith("5") ? "MEMO" : "NORMAL";
                String key = d.branch + "_" + d.currency + "_" + type;
                balanceMap.merge(key, d.amount, BigDecimal::add);
            }
        });

        // Stage 3: Balance Checks (Global)
        updateState(requestId, s -> { s.setMessage("Checking Balances..."); s.setCurrentStage(3); });

        rows.forEach(d -> {
            // Only add balance error to valid rows to avoid clutter
            if (!d.hasErrors()) {
                String type = d.cgl.startsWith("5") ? "MEMO" : "NORMAL";
                String key = d.branch + "_" + d.currency + "_" + type;
                BigDecimal diff = balanceMap.getOrDefault(key, BigDecimal.ZERO);
                if (diff.compareTo(BigDecimal.ZERO) != 0) {
                    String cglType = d.cgl.startsWith("5") ? "Memo CGL" : "CGL";
                    d.addError(cglType + " Balance Mismatch (Diff: " + diff.toPlainString() + ")");
                }
            }
        });
    }

    // --- PARSERS ---
    private List<ExcelRowData> parseCsvBytes(byte[] bytes) throws IOException {
        List<ExcelRowData> list = new ArrayList<>();
        try(BufferedReader br = new BufferedReader(new InputStreamReader(new ByteArrayInputStream(bytes), StandardCharsets.UTF_8))) {
            String line; int i=0;
            while((line=br.readLine())!=null) {
                if(line.trim().isEmpty()) continue;
                String[] c = line.split(",", -1);
                if (i == 0 || (c.length > 0 && isHeader(c[0]))) { i++; continue; } // Skip header

                ExcelRowData d = new ExcelRowData();
                d.rowIndex = i++;

                if (c.length >= 14) {
                    d.isSystemFormat = true;
                    // Store raw system headers for error file reconstruction
                    d.site = c[0].trim(); d.dateRaw = c[1].trim(); d.year = c[2].trim(); d.period = c[3].trim();

                    d.branch = c[4].trim(); d.currency = c[5].trim(); d.cgl = c[6].trim();
                    parseAmount(d, c[7].trim(), c[8].trim());
                    d.remarks = c[9].trim();
                    d.productCode = c[10].trim().isEmpty() ? "A" : c[10].trim();
                } else if (c.length >= 7) {
                    d.branch = c[0].trim(); d.currency = c[1].trim(); d.cgl = c[2].trim();
                    parseAmount(d, c[3].trim(), c[4].trim()); d.remarks = c[5].trim();
                    d.productCode = c.length > 6 && !c[6].trim().isEmpty() ? c[6].trim() : "A";
                } else { d.addError("Invalid Column Count"); }
                list.add(d);
            }
        }
        return list;
    }

    private List<ExcelRowData> parseExcelBytes(byte[] bytes) throws IOException {
        try(Workbook wb = new XSSFWorkbook(new ByteArrayInputStream(bytes))) {
            Sheet sheet = wb.getSheetAt(0);
            List<ExcelRowData> list = new ArrayList<>();
            DataFormatter fmt = new DataFormatter();
            for(Row r : sheet) {
                if(r == null) continue;
                String c0 = getCellVal(r.getCell(0), fmt);
                if(c0.isEmpty() || isHeader(c0)) continue;
                ExcelRowData d = new ExcelRowData();
                d.rowIndex = r.getRowNum();
                if(c0.equals("01")) {
                    d.isSystemFormat = true;
                    d.site = c0; d.dateRaw = getCellVal(r.getCell(1), fmt);
                    d.year = getCellVal(r.getCell(2), fmt); d.period = getCellVal(r.getCell(3), fmt);

                    d.branch = getCellVal(r.getCell(4), fmt); d.currency = getCellVal(r.getCell(5), fmt); d.cgl = getCellVal(r.getCell(6), fmt);
                    parseAmount(d, getCellVal(r.getCell(7), fmt), getCellVal(r.getCell(8), fmt));
                    d.remarks = getCellVal(r.getCell(9), fmt);
                    d.productCode = getCellVal(r.getCell(10), fmt).isEmpty() ? "A" : getCellVal(r.getCell(10), fmt);
                } else {
                    d.branch = c0; d.currency = getCellVal(r.getCell(1), fmt); d.cgl = getCellVal(r.getCell(2), fmt);
                    parseAmount(d, getCellVal(r.getCell(3), fmt), getCellVal(r.getCell(4), fmt));
                    d.remarks = getCellVal(r.getCell(5), fmt);
                    d.productCode = getCellVal(r.getCell(6), fmt).isEmpty() ? "A" : getCellVal(r.getCell(6), fmt);
                }
                list.add(d);
            }
            return list;
        }
    }

    private boolean isHeader(String cell) { String h = cell.toLowerCase(); return h.contains("branch") || h.contains("site") || h.contains("date"); }
    private String getCellVal(Cell c, DataFormatter fmt) { if(c==null) return ""; return fmt.formatCellValue(c).trim(); }

    private void parseAmount(ExcelRowData d, String amountRaw, String typeRaw) {
        try {
            if (amountRaw == null) amountRaw = "";
            String clean = CLEAN_AMOUNT_REGEX.matcher(amountRaw).replaceAll("");
            if (clean.isEmpty()) { d.amount = BigDecimal.ZERO; return; }
            BigDecimal v = new BigDecimal(clean);
            boolean isCredit = typeRaw != null && (typeRaw.toUpperCase().startsWith("C") || typeRaw.toUpperCase().contains("CR"));
            d.amount = isCredit ? v.abs().negate() : v.abs();
            d.txnType = isCredit ? "Credit" : "Debit";
        } catch(Exception e) { d.amount = BigDecimal.ZERO; d.addError("Invalid Amount Format"); }
    }

    private void failRequest(String reqId, List<ExcelRowData> rows, String msg, int stage) throws IOException {
        byte[] excel = generateErrorExcel(rows);
        saveFileToRedis(reqId, "ERROR", excel);
        updateState(reqId, s -> {
            s.setStatus("ERROR"); s.setMessage(msg); s.setCurrentStage(stage);
            s.setErrorCount(rows.stream().filter(ExcelRowData::hasErrors).count()); s.setHasErrorFile(true);
        });
    }

    // --- CSV Generation for Success (Supports downloading CSV even if Excel uploaded) ---
    private void completeRequest(String reqId, List<ExcelRowData> rows, LocalDate pDate, boolean isCsv) throws IOException {
        String fullDataJson = objectMapper.writeValueAsString(rows);
        redisTemplate.opsForValue().set(KEY_DATA_CACHE + reqId, fullDataJson, TTL_MINUTES, TimeUnit.MINUTES);

        // Always generate a Success CSV bytes in case they want to download it
        byte[] successCsv = generateSuccessCsv(rows, pDate);
        saveFileToRedis(reqId, "SUCCESS", successCsv);

        List<Map<String, Object>> preview = rows.stream().limit(2000).map(this::mapToPreview).collect(Collectors.toList());
        String previewJson = objectMapper.writeValueAsString(preview);

        updateState(reqId, s -> {
            s.setCurrentStage(4); s.setStatus("SUCCESS"); s.setMessage("Validation Successful");
            s.setTotalRows(rows.size()); s.setPreviewDataJson(previewJson);
            s.setHasSuccessFile(true); // Enable download button
        });
    }

    private byte[] generateErrorExcel(List<ExcelRowData> rows) throws IOException {
        try (SXSSFWorkbook wb = new SXSSFWorkbook(100)) {
            Sheet sheet = wb.createSheet("Error Report");
            Row h = sheet.createRow(0);
            String[] hd = {"Site", "Date", "Year", "Period", "Branch", "Currency", "CGL", "Amount", "Dr/Cr", "Remarks", "Product", "ERRORS"};
            for(int i=0; i<hd.length; i++) h.createCell(i).setCellValue(hd[i]);

            CellStyle errStyle = wb.createCellStyle();
            Font f = wb.createFont(); f.setColor(IndexedColors.RED.getIndex()); errStyle.setFont(f);

            int idx = 1;
            for(ExcelRowData d : rows) {
                // WRITE ALL ROWS (Valid & Invalid) so user can fix in place
                Row r = sheet.createRow(idx++);
                r.createCell(0).setCellValue(d.isSystemFormat ? d.site : "01");
                r.createCell(1).setCellValue(d.isSystemFormat ? d.dateRaw : "");
                r.createCell(2).setCellValue(d.isSystemFormat ? d.year : "");
                r.createCell(3).setCellValue(d.isSystemFormat ? d.period : "");

                r.createCell(4).setCellValue(d.branch);
                r.createCell(5).setCellValue(d.currency);
                r.createCell(6).setCellValue(d.cgl);
                r.createCell(7).setCellValue(d.amount != null ? d.amount.abs().toString() : "");
                r.createCell(8).setCellValue(d.txnType);
                r.createCell(9).setCellValue(d.remarks);
                r.createCell(10).setCellValue(d.productCode);

                if(d.hasErrors()) {
                    Cell c = r.createCell(11);
                    c.setCellValue(String.join("; ", d.errors));
                    c.setCellStyle(errStyle);
                }
            }
            ByteArrayOutputStream out = new ByteArrayOutputStream();
            wb.write(out); return out.toByteArray();
        }
    }

    private byte[] generateSuccessCsv(List<ExcelRowData> rows, LocalDate pDate) {
        StringBuilder sb = new StringBuilder();
        // Basic CSV construction matching system format or template
        for(ExcelRowData d : rows) {
            String amt = d.amount != null ? d.amount.abs().toString() : "0.00";
            String type = d.txnType != null && d.txnType.startsWith("Cr") ? "Cr" : "Dr";
            // Reconstruct valid system CSV line for download
            sb.append("01,").append(d.dateRaw != null ? d.dateRaw : pDate.format(SYSTEM_DATE_FMT)).append(",")
                    .append(d.year).append(",").append(d.period).append(",")
                    .append(d.branch).append(",").append(d.currency).append(",").append(d.cgl).append(",")
                    .append(amt).append(",").append(type).append(",").append(d.remarks).append(",").append(d.productCode).append(",B,C,D\n");
        }
        return sb.toString().getBytes(StandardCharsets.UTF_8);
    }

    public byte[] generateTemplateBytes() throws IOException {
        try (SXSSFWorkbook wb = new SXSSFWorkbook()) {
            Sheet sheet = wb.createSheet("Journal Template");
            Row row = sheet.createRow(0);
            String[] headers = {"Branch", "Currency", "CGL", "Amount", "Dr/Cr", "Remarks", "Product"};
            for (int i = 0; i < headers.length; i++) sheet.createRow(0).createCell(i).setCellValue(headers[i]);
            ByteArrayOutputStream out = new ByteArrayOutputStream();
            wb.write(out);
            return out.toByteArray();
        }
    }

    private Map<String, Object> mapToPreview(ExcelRowData r) {
        Map<String, Object> m = new HashMap<>();
        m.put("id", r.rowIndex); m.put("branch", r.branch); m.put("currency", r.currency);
        m.put("cgl", r.cgl); m.put("amount", r.amount!=null?r.amount.abs().toString():"");
        m.put("txnType", r.txnType); m.put("remarks", r.remarks); m.put("productCode", r.productCode);
        return m;
    }

    // Redis Helpers
    private void saveStateToRedis(BulkUploadStateDto state) { try { redisTemplate.opsForValue().set(KEY_META + state.getRequestId(), objectMapper.writeValueAsString(state), TTL_MINUTES, TimeUnit.MINUTES); } catch (Exception e) {} }
    private void updateState(String requestId, java.util.function.Consumer<BulkUploadStateDto> updater) { try { String json = redisTemplate.opsForValue().get(KEY_META + requestId); if (json != null) { BulkUploadStateDto state = objectMapper.readValue(json, BulkUploadStateDto.class); updater.accept(state); saveStateToRedis(state); } } catch (Exception e) {} }
    private void saveFileToRedis(String requestId, String type, byte[] data) { String key = (type.equals("ERROR") ? KEY_FILE_ERR : KEY_FILE_SUC) + requestId; String base64 = Base64.getEncoder().encodeToString(data); redisTemplate.opsForValue().set(key, base64, TTL_MINUTES, TimeUnit.MINUTES); }
    public BulkUploadStateDto getState(String reqId) { try { String json = redisTemplate.opsForValue().get(KEY_META + reqId); return json != null ? objectMapper.readValue(json, BulkUploadStateDto.class) : null; } catch (Exception e) { return null; } }
    public byte[] getFileBytes(String reqId, String type) { String key = (type.equals("ERROR") ? KEY_FILE_ERR : KEY_FILE_SUC) + reqId; String base64 = redisTemplate.opsForValue().get(key); return base64 != null ? Base64.getDecoder().decode(base64) : null; }
}

    public String createBatchFromCache(String requestId, String commonRemarks, String creatorId, Integer creatorRole) throws IOException {

        if (cachedRows == null || cachedRows.isEmpty()) {
            throw new IllegalArgumentException("Session expired or invalid Request ID. Please upload again.");
        }

        String batchId = sequenceService.getNextBatchId();
        LocalDate postingDate = LocalDate.now();
        List<JournalRequest> requestsToSave = new ArrayList<>();

        Map<String, List<JournalBulkValidationService.ExcelRowData>> groupedRows = cachedRows.stream()
                .collect(Collectors.groupingBy(row -> row.branch + ":" + row.currency));

        for (Map.Entry<String, List<JournalBulkValidationService.ExcelRowData>> entry : groupedRows.entrySet()) {
            String journalIdPrefix = sequenceService.getNextJournalIdPrefix();
            int seq = 1;
            for (JournalBulkValidationService.ExcelRowData row : entry.getValue()) {
                String journalId = String.format("%s-%03d", journalIdPrefix, seq++);

                CreateJournalRequestDto payload = new CreateJournalRequestDto();
                payload.setPDate(postingDate);
                payload.setChangeType(ChangeType.ADD);
                payload.setBranch(row.branch);
                payload.setCurrency(row.currency);
                payload.setCgl(row.cgl);
                payload.setAmount(row.amount);
                payload.setProductType(row.productCode);
                payload.setRemarks(row.remarks);
                payload.setBatchId(batchId);
                payload.setJournalId(journalId);
                payload.setCommonBatchRemarks(commonRemarks);
                payload.setTransactionCount(seq-1);
                payload.setArFlag("A");
                payload.setAcClassification("A");

                JournalRequest req = new JournalRequest();
                req.setCreatorId(creatorId);
                req.setCreatorRole(creatorRole);
                req.setRequestStatus(RequestStatus.PENDING);
                req.setBatchId(batchId);
                req.setJournalId(journalId);
                req.setCommonBatchRemarks(commonRemarks);
                req.setPayload(objectMapper.writeValueAsString(payload));
                req.setChangeType(ChangeType.ADD);

                requestsToSave.add(req);
            }
        }

        journalRequestRepository.saveAll(requestsToSave);
        createNotification(batchId, creatorId, requestsToSave.size());

        return batchId;
    }

    // --- 3. APPROVAL / REJECTION ---
    @Override
    @Transactional
    public Optional<JournalRequest> updateRequestStatus(ProcessJournalRequestDto dto, String executorId, Integer executorRole) throws JsonProcessingException {
        JournalRequest request = journalRequestRepository.findById(dto.getRequestId())
                .orElseThrow(() -> new ResourceNotFoundException("Journal Request not found."));

        if (request.getRequestStatus() != RequestStatus.PENDING) throw new IllegalStateException("Already processed.");

        request.setExecutorId(executorId);
        request.setExecutorRemarks(dto.getRemarks());
        request.setExecutionDate(LocalDateTime.now());
        request.setRequestStatus(dto.getStatus());

        if (RequestStatus.ACCEPTED.equals(dto.getStatus())) {
            processApproval(request);
        }
        return Optional.of(journalRequestRepository.save(request));
    }

    @Override
    @Transactional
    public List<JournalRequest> processBulkRequests(BulkProcessJournalRequestDto dto, String executorId, Integer executorRole) throws JsonProcessingException {
        boolean hasBatchId = StringUtils.hasText(dto.getBatchId());
        List<JournalRequest> requestsToProcess;

        if (hasBatchId) {
            requestsToProcess = journalRequestRepository.findByBatchIdAndRequestStatus(dto.getBatchId(), RequestStatus.PENDING);
        } else {
            Specification<JournalRequest> spec = (root, query, criteriaBuilder) -> {
                List<Predicate> predicates = new ArrayList<>();
                if (dto.getJournalIdPrefixes() != null) {
                    for (String prefix : dto.getJournalIdPrefixes()) {
                        predicates.add(criteriaBuilder.like(root.get("journalId"), prefix + "%"));
                    }
                }
                Predicate journalIdOrPredicates = criteriaBuilder.or(predicates.toArray(new Predicate[0]));
                Predicate statusPredicate = criteriaBuilder.equal(root.get("requestStatus"), RequestStatus.PENDING);
                return criteriaBuilder.and(journalIdOrPredicates, statusPredicate);
            };
            requestsToProcess = journalRequestRepository.findAll(spec);
        }

        if (requestsToProcess.isEmpty()) return Collections.emptyList();

        List<GlTransaction> transactionsToSave = new ArrayList<>();
        List<JournalRequest> processedRequests = new ArrayList<>();

        for (JournalRequest request : requestsToProcess) {
            // Self-approval check
            if (executorId.equals(request.getCreatorId())) continue;

            request.setExecutorId(executorId);
            request.setExecutorRemarks(dto.getRemarks());
            request.setExecutionDate(LocalDateTime.now());
            request.setRequestStatus(dto.getStatus());

            if (RequestStatus.ACCEPTED.equals(dto.getStatus())) {
                CreateJournalRequestDto payloadDto = objectMapper.readValue(request.getPayload(), CreateJournalRequestDto.class);
                if (payloadDto.getChangeType() == ChangeType.ADD) {
                    GlTransaction newGlTxn = new GlTransaction();
                    newGlTxn.setBatchId(payloadDto.getBatchId());
                    newGlTxn.setJournalId(payloadDto.getJournalId());
                    newGlTxn.setPostDate(payloadDto.getPDate() != null ? payloadDto.getPDate().atStartOfDay() : LocalDateTime.now());
                    newGlTxn.setTransactionDate(payloadDto.getPDate() != null ? payloadDto.getPDate() : LocalDate.now());
                    newGlTxn.setBranchCode(payloadDto.getBranch());
                    newGlTxn.setCurrency(payloadDto.getCurrency());
                    newGlTxn.setCgl(payloadDto.getCgl());
                    newGlTxn.setNarration(payloadDto.getRemarks());

                    BigDecimal amount = payloadDto.getAmount();
                    if (amount.signum() < 0) {
                        newGlTxn.setCreditAmount(amount); newGlTxn.setDebitAmount(BigDecimal.ZERO);
                    } else {
                        newGlTxn.setDebitAmount(amount); newGlTxn.setCreditAmount(BigDecimal.ZERO);
                    }
                    newGlTxn.setTransactionCount(payloadDto.getTransactionCount());
                    newGlTxn.setSourceFlag("J");
                    transactionsToSave.add(newGlTxn);
                }
            }
            processedRequests.add(request);
        }

        if (!transactionsToSave.isEmpty()) glTransactionRepository.saveAll(transactionsToSave);
        return journalRequestRepository.saveAll(processedRequests);
    }

    private void processApproval(JournalRequest request) throws JsonProcessingException {
        // Reuse logic from bulk process or separate method
        CreateJournalRequestDto payloadDto = objectMapper.readValue(request.getPayload(), CreateJournalRequestDto.class);
        if (payloadDto.getChangeType() == ChangeType.ADD) {
            GlTransaction newGlTxn = new GlTransaction();
            newGlTxn.setBatchId(payloadDto.getBatchId());
            newGlTxn.setJournalId(payloadDto.getJournalId());
            newGlTxn.setPostDate(payloadDto.getPDate() != null ? payloadDto.getPDate().atStartOfDay() : LocalDateTime.now());
            newGlTxn.setTransactionDate(payloadDto.getPDate() != null ? payloadDto.getPDate() : LocalDate.now());

    @Override
    @Transactional
    public String createBatchFromCache(String requestId, String commonRemarks, String creatorId, Integer creatorRole) throws IOException {
        List<JournalBulkValidationService.ExcelRowData> cachedRows = validationService.getValidRowsFromCache(requestId);

        if (cachedRows == null || cachedRows.isEmpty()) {
            throw new IllegalArgumentException("Session expired or invalid Request ID. Please upload again.");
        }

        String batchId = sequenceService.getNextBatchId();
        LocalDate postingDate = LocalDate.now();
        List<JournalRequest> requestsToSave = new ArrayList<>();

        Map<String, List<JournalBulkValidationService.ExcelRowData>> groupedRows = cachedRows.stream()
                .collect(Collectors.groupingBy(row -> row.branch + ":" + row.currency));

        for (Map.Entry<String, List<JournalBulkValidationService.ExcelRowData>> entry : groupedRows.entrySet()) {
            String journalIdPrefix = sequenceService.getNextJournalIdPrefix();
            int seq = 1;
            for (JournalBulkValidationService.ExcelRowData row : entry.getValue()) {
                String journalId = String.format("%s-%03d", journalIdPrefix, seq++);

                CreateJournalRequestDto payload = new CreateJournalRequestDto();
                payload.setPDate(postingDate);
                payload.setChangeType(ChangeType.ADD);
                payload.setBranch(row.branch);
                payload.setCurrency(row.currency);
                payload.setCgl(row.cgl);
                payload.setAmount(row.amount);
                payload.setProductType(row.productCode);
                payload.setRemarks(row.remarks);
                payload.setBatchId(batchId);
                payload.setJournalId(journalId);
                payload.setCommonBatchRemarks(commonRemarks);
                payload.setTransactionCount(seq-1);
                payload.setArFlag("A");
                payload.setAcClassification("A");

                JournalRequest req = new JournalRequest();
                req.setCreatorId(creatorId);
                req.setCreatorRole(creatorRole);
                req.setRequestStatus(RequestStatus.PENDING);
                req.setBatchId(batchId);
                req.setJournalId(journalId);
                req.setCommonBatchRemarks(commonRemarks);
                req.setPayload(objectMapper.writeValueAsString(payload));
                req.setChangeType(ChangeType.ADD);

                requestsToSave.add(req);
            }
        }

        journalRequestRepository.saveAll(requestsToSave);
        createNotification(batchId, creatorId, requestsToSave.size());

        return batchId;
    }

    @Override
    @Transactional
    public Optional<JournalRequest> updateRequestStatus(ProcessJournalRequestDto dto, String executorId, Integer executorRole) throws JsonProcessingException {
        JournalRequest request = journalRequestRepository.findById(dto.getRequestId())
                .orElseThrow(() -> new ResourceNotFoundException("Journal Request not found."));

        if (request.getRequestStatus() != RequestStatus.PENDING) throw new IllegalStateException("Already processed.");

        request.setExecutorId(executorId);
        request.setExecutorRemarks(dto.getRemarks());
        request.setExecutionDate(LocalDateTime.now());
        request.setRequestStatus(dto.getStatus());

        if (RequestStatus.ACCEPTED.equals(dto.getStatus())) {
            processApproval(request);
        }
        return Optional.of(journalRequestRepository.save(request));
    }

    @Override
    @Transactional
    public List<JournalRequest> processBulkRequests(BulkProcessJournalRequestDto dto, String executorId, Integer executorRole) throws JsonProcessingException {
        boolean hasBatchId = StringUtils.hasText(dto.getBatchId());
        List<JournalRequest> requestsToProcess;

        if (hasBatchId) {
            requestsToProcess = journalRequestRepository.findByBatchIdAndRequestStatus(dto.getBatchId(), RequestStatus.PENDING);
        } else {
            Specification<JournalRequest> spec = (root, query, criteriaBuilder) -> {
                List<Predicate> predicates = new ArrayList<>();
                if (dto.getJournalIdPrefixes() != null) {
                    for (String prefix : dto.getJournalIdPrefixes()) {
                        predicates.add(criteriaBuilder.like(root.get("journalId"), prefix + "%"));
                    }
                }
                Predicate journalIdOrPredicates = criteriaBuilder.or(predicates.toArray(new Predicate[0]));
                Predicate statusPredicate = criteriaBuilder.equal(root.get("requestStatus"), RequestStatus.PENDING);
                return criteriaBuilder.and(journalIdOrPredicates, statusPredicate);
            };
            requestsToProcess = journalRequestRepository.findAll(spec);
        }

        if (requestsToProcess.isEmpty()) return Collections.emptyList();

        List<GlTransaction> transactionsToSave = new ArrayList<>();
        List<JournalRequest> processedRequests = new ArrayList<>();

        for (JournalRequest request : requestsToProcess) {
            // Self-approval check
            if (executorId.equals(request.getCreatorId())) continue;

            request.setExecutorId(executorId);
            request.setExecutorRemarks(dto.getRemarks());
            request.setExecutionDate(LocalDateTime.now());
            request.setRequestStatus(dto.getStatus());

            if (RequestStatus.ACCEPTED.equals(dto.getStatus())) {
                CreateJournalRequestDto payloadDto = objectMapper.readValue(request.getPayload(), CreateJournalRequestDto.class);
                if (payloadDto.getChangeType() == ChangeType.ADD) {
                    GlTransaction newGlTxn = new GlTransaction();
                    newGlTxn.setBatchId(payloadDto.getBatchId());
                    newGlTxn.setJournalId(payloadDto.getJournalId());
                    newGlTxn.setPostDate(payloadDto.getPDate() != null ? payloadDto.getPDate().atStartOfDay() : LocalDateTime.now());
                    newGlTxn.setTransactionDate(payloadDto.getPDate() != null ? payloadDto.getPDate() : LocalDate.now());
                    newGlTxn.setBranchCode(payloadDto.getBranch());
                    newGlTxn.setCurrency(payloadDto.getCurrency());
                    newGlTxn.setCgl(payloadDto.getCgl());
                    newGlTxn.setNarration(payloadDto.getRemarks());

                    BigDecimal amount = payloadDto.getAmount();
                    if (amount.signum() < 0) {
                        newGlTxn.setCreditAmount(amount); newGlTxn.setDebitAmount(BigDecimal.ZERO);
                    } else {
                        newGlTxn.setDebitAmount(amount); newGlTxn.setCreditAmount(BigDecimal.ZERO);
                    }
                    newGlTxn.setTransactionCount(payloadDto.getTransactionCount());
                    newGlTxn.setSourceFlag("J");
                    transactionsToSave.add(newGlTxn);
                }
            }
            processedRequests.add(request);
        }

        if (!transactionsToSave.isEmpty()) glTransactionRepository.saveAll(transactionsToSave);
        return journalRequestRepository.saveAll(processedRequests);
    }

    private void processApproval(JournalRequest request) throws JsonProcessingException {
        // Reuse logic from bulk process or separate method
        CreateJournalRequestDto payloadDto = objectMapper.readValue(request.getPayload(), CreateJournalRequestDto.class);

      const sAuth = sessionStorage.getItem("persist:auth") || localStorage.getItem("persist:auth");
      if (sAuth) { try { token = JSON.parse(sAuth).token.replace(/"/g, ""); } catch (e) { } }

      const res = await fetch("/JS/journals/bulk-validate-init", {
        method: "POST", headers: { "Authorization": `Bearer ${token}` }, body: formData
      });
      if (!res.ok) throw new Error(await res.text());
      const data = await res.json();
      setRequestId(data.requestId);
    } catch (err) {
      setIsProcessing(false); setResultStatus("ERROR"); setErrorMsg("Init Failed: " + err.message);
    }
  };

  // Polling
  useEffect(() => {
    let interval;
    if (isProcessing && requestId) {
      interval = setInterval(async () => {
        try {
          const res = await callApi(`/JS/journals/bulk-status/${requestId}`, null, "GET");
          if (res.currentStage) setValidationStage(res.currentStage);

          if (res.status === "ERROR") {
            setIsProcessing(false); setResultStatus("ERROR"); setErrorMsg(res.message);
            clearInterval(interval);
            showSnackBar(res.message, "error", 5000);
            if (res.hasErrorFile && !autoDlTriggered) {
              setAutoDlTriggered(true);
              handleDownloadFile("ERROR", res.requestId);
            }
          }
          else if (res.status === "SUCCESS") {
            setIsProcessing(false); setResultStatus("SUCCESS"); setValidationStage(4);
            setTotalRows(res.totalRows);
            setHasSuccessFile(res.hasSuccessFile);
            if (res.previewDataJson) {
              const rows = JSON.parse(res.previewDataJson);
              setPreviewRows(rows);
              setHasProductColumn(rows.some(r => r.productCode && r.productCode !== "A"));
            }
            // Auto-Forward
            setTimeout(() => { setActiveStep(1); }, 800);
            clearInterval(interval);
            showSnackBar("Validation Complete. Moving to Preview.", "success");
          }
        } catch (e) { console.error(e); }
      }, 800);
    }
    return () => clearInterval(interval);
  }, [isProcessing, requestId, callApi, showSnackBar, autoDlTriggered]);

  const handleDownloadFile = async (type, reqId = requestId) => {
    try {
      if(type === "ERROR") showSnackBar("Downloading Error Report...", "info");
      
      let token = "";
      const sAuth = sessionStorage.getItem("persist:auth") || localStorage.getItem("persist:auth");
      if (sAuth) { try { token = JSON.parse(sAuth).token.replace(/"/g, ""); } catch (e) { } }
      const response = await fetch(`/JS/journals/download-bulk-file/${reqId}?type=${type}`, {
        headers: { Authorization: `Bearer ${token}` }
      });
      if (!response.ok) throw new Error("Download failed");
      const blob = await response.blob();
      const url = window.URL.createObjectURL(blob);
      const link = document.createElement("a");
      link.href = url;
      link.setAttribute("download", type === "ERROR" ? "Error_Report.xlsx" : "Success_Batch.csv");
      document.body.appendChild(link);
      link.click();
      link.remove();
      
      showSnackBar("Downloaded successfully.", "success"); // Update text after download
    } catch (e) { showSnackBar("Download failed", "error"); }
  };

  const handleCreateBatch = async () => {
    setIsCreatingBatch(true);
    try {
      const payload = { requestId: requestId, commonBatchRemarks: `Bulk Upload - ${selectedDate}` };
      const res = await callApi("/JS/journals/create-bulk-batch-optimized", payload, "POST");
      if (res.status === "SUCCESS") {
        showSnackBar(`Batch ${res.batchId} Created Successfully!`, "success", 5000);
        resetAll();
      } else { throw new Error(res.message); }
    } catch (e) { showSnackBar("Failed to create batch: " + e.message, "error"); } 
    finally { setIsCreatingBatch(false); }
  };

  // UI Helper: Horizontal Validation Items
  const renderValidationItem = (stageNum, label) => {
    let icon = <RadioButtonUnchecked color="disabled" />;
    let textSx = { color: 'text.secondary' };

    if (validationStage > stageNum || (validationStage === stageNum && resultStatus === 'SUCCESS')) {
        icon = <CheckCircle color="success" />;
        textSx = { color: 'text.primary', fontWeight: 'bold' };
    } else if (validationStage === stageNum && resultStatus === 'ERROR') {
        icon = <ErrorIcon color="error" />;
        textSx = { color: 'error.main', fontWeight: 'bold' };
    } else if (validationStage === stageNum && isProcessing) {
        icon = <CircularProgress size={20} color="primary" />;
        textSx = { color: 'primary.main', fontWeight: 'bold' };
    }

    return (
        <Stack direction="row" spacing={1} alignItems="center">
            {icon}
            <Typography variant="body2" sx={textSx}>{label}</Typography>
        </Stack>
    );
  };

  return (
    <Box sx={styles.mainContainer}>
      <Paper sx={styles.controlPanel}>
        
        <Box sx={{ display: 'flex', justifyContent: 'flex-end', mb: 2 }}>
          <Button startIcon={<FileDownload />} onClick={handleDownloadTemplate}>Download Template</Button>
        </Box>

        <Stepper activeStep={activeStep} sx={{ mb: 4 }} alternativeLabel>
          {steps.map((label) => (
            <Step key={label}><StepLabel>{label}</StepLabel></Step>
          ))}
        </Stepper>

        {/* --- STEP 1: UPLOAD & VALIDATE --- */}
        {activeStep === 0 && (
          <Fade in={true}>
            <Box>
              <Stack direction="row" spacing={2} justifyContent="center" alignItems="center">
                 <TextField
                    type="date" label="Posting Date" value={selectedDate} onChange={handleDateChange}
                    InputLabelProps={{ shrink: true }} size="small" disabled={isProcessing} sx={{ width: 200 }}
                 />
              </Stack>

              <Box
                sx={{
                    ...styles.uploadBox, 
                    borderColor: resultStatus === 'ERROR' ? 'error.main' : 'primary.main',
                    opacity: isProcessing ? 0.6 : 1
                }}
                onClick={() => { if (!isProcessing) fileInputRef.current.click(); }}
              >
                <input type="file" hidden ref={fileInputRef} onChange={handleFileSelect} accept=".xlsx,.csv" />
                {file ? (
                  <>
                    <CheckCircle sx={{ fontSize: 60, color: "success.main" }} />
                    <Typography sx={styles.uploadText}>{file.name}</Typography>
                    <Typography variant="caption">Click to change</Typography>
                  </>
                ) : (
                  <>
                    <CloudUpload sx={{ fontSize: 60, color: "primary.main", mb: 1 }} />
                    <Typography sx={styles.uploadText}>Click to Upload File</Typography>
                    <Typography sx={styles.uploadSubText}>.xlsx, .csv (Max 50MB)</Typography>
                  </>
                )}
              </Box>

              {/* HORIZONTAL VALIDATION LIST */}
              {(isProcessing || resultStatus) && (
                  <Box sx={styles.validationContainer}>
                      <Typography variant="subtitle2" sx={{ mb: 2, color: 'text.secondary' }}>Validation Status:</Typography>
                      <Grid container spacing={2}>
                          <Grid item xs={12} md={4}>{renderValidationItem(1, "Format Checks (Date, Headers)")}</Grid>
                          <Grid item xs={12} md={4}>{renderValidationItem(2, "Database Checks (Branch, CGL)")}</Grid>
                          <Grid item xs={12} md={4}>{renderValidationItem(3, "Balance Checks (Dr/Cr)")}</Grid>
                      </Grid>
                      
                      {resultStatus === 'ERROR' && (
                          <Alert severity="error" sx={{ mt: 2 }}>
                              Validation Failed. Please correct the errors in the downloaded report and re-upload.
                          </Alert>
                      )}
                  </Box>
              )}

              <Box sx={styles.actionButtons}>
                <Button
                  variant="contained" size="large" onClick={handleStartValidation}
                  disabled={!file || !selectedDate || isProcessing}
                  startIcon={isProcessing ? <CircularProgress size={20} color="inherit"/> : <Send/>}
                >
                  {isProcessing ? "Validating..." : "Start Validation"}
                </Button>
              </Box>
            </Box>
          </Fade>
        )}

        {/* --- STEP 2: PREVIEW & SUBMIT --- */}
        {activeStep === 1 && (
          <Fade in={true}>
            <Box>
              <Stack direction="row" justifyContent="space-between" alignItems="center" sx={{mb: 2}}>
                  <Alert severity="success" sx={{ flexGrow: 1, mr: 2 }}>
                     Validation Successful! Total Rows: {totalRows.toLocaleString()}
                  </Alert>
                  {/* Download CSV Button (Available for Excel uploads too) */}
                  <Button variant="outlined" startIcon={<FileDownload />} onClick={() => handleDownloadFile("SUCCESS")}>
                      Download CSV
                  </Button>
              </Stack>

              <TableContainer component={Paper} sx={styles.tableContainer}>
                <Table stickyHeader size="small">
                  <TableHead>
                    <TableRow>
                      <TableCell sx={styles.tableHeader}>Row #</TableCell>
                      <TableCell sx={styles.tableHeader}>Branch</TableCell>
                      <TableCell sx={styles.tableHeader}>Currency</TableCell>
                      <TableCell sx={styles.tableHeader}>CGL</TableCell>
                      <TableCell align="right" sx={styles.tableHeader}>Amount</TableCell>
                      <TableCell align="center" sx={styles.tableHeader}>Type</TableCell>
                      <TableCell sx={styles.tableHeader}>Remarks</TableCell>
                      {hasProductColumn && <TableCell sx={styles.tableHeader}>Product</TableCell>}
                    </TableRow>
                  </TableHead>
                  <TableBody>
                    {previewRows
                      .slice(page * rowsPerPage, page * rowsPerPage + rowsPerPage)
                      .map((row, idx) => (
                      <TableRow key={idx} hover>
                        <TableCell>{row.id}</TableCell>
                        <TableCell>{row.branch}</TableCell>
                        <TableCell>{row.currency}</TableCell>
                        <TableCell>{row.cgl}</TableCell>
                        <TableCell align="right">{row.amount}</TableCell>
                        <TableCell align="center">
                          <Chip
                            label={row.txnType} size="small"
                            color={row.txnType.includes("Credit") ? "error" : "success"}
                          />
                        </TableCell>
                        <TableCell>{row.remarks}</TableCell>
                        {hasProductColumn && <TableCell>{row.productCode}</TableCell>}
                      </TableRow>
                    ))}
                  </TableBody>
                </Table>
              </TableContainer>
              
              <TablePagination
                rowsPerPageOptions={[50, 100, 200]}
                component="div"
                count={previewRows.length}
                rowsPerPage={rowsPerPage}
                page={page}
                onPageChange={(e, p) => setPage(p)}
                onRowsPerPageChange={(e) => { setRowsPerPage(parseInt(e.target.value, 10)); setPage(0); }}
              />

              <Box sx={styles.actionButtons}>
                <Button variant="outlined" onClick={() => setActiveStep(0)} disabled={isCreatingBatch} startIcon={<RestartAlt />}>
                  Back to Upload
                </Button>
                <Button
                  variant="contained" color="success" size="large" onClick={handleCreateBatch}
                  disabled={isCreatingBatch}
                  startIcon={isCreatingBatch ? <CircularProgress size={20} color="inherit" /> : <Send />}
                >
                  {isCreatingBatch ? "Creating Batch..." : "Create Batch"}
                </Button>
              </Box>
            </Box>
          </Fade>
        )}

      </Paper>
    </Box>
  );
};
export default JournalBulkUpload;
