  const user = localStorage.getItem('user');
  return user ? JSON.parse(user) : {};
};

// styled tab

  textTransform: 'none', // Prevent uppercase text
  minWidth: 0,
  [theme.breakpoints.up('sm')]: {
    minWidth: 0,
  },
  fontWeight: theme.typography.fontWeightRegular,
  marginRight: theme.spacing(1),
  color: theme.palette.text.secondary,
  '&:hover': {
    color: theme.palette.primary.main, // Hover effect for interactivity
    opacity: 1,
  },
  '&.Mui-selected': {
    color: theme.palette.primary.main,
    fontWeight: theme.typography.fontWeightMedium,
  },
  '&.Mui-focusVisible': {
    backgroundColor: 'rgba(100, 95, 228, 0.32)',
  },
}));

// Custom Tabs Indicator for a smoother transition
const StyledTabs = styled(Tabs)({
  borderBottom: '1px solid #e8e8e8',
  '& .MuiTabs-indicator': {
    backgroundColor: '#1890ff',
    height: 6, // Adjust indicator thickness
  },
});

// TabPanel component to display content for each tab
function TabPanel(props) {
  const { children, value, index, ...other } = props;

  return (
    <div
      role="tabpanel"
      hidden={value !== index}
      id={`simple-tabpanel-${index}`}
      aria-labelledby={`simple-tab-${index}`}
      {...other}
    >
      {value === index && <Box sx={{ pt: 2 }}>{children}</Box>}
    </div>
  );
}

function a11yProps(index) {
  return {
    id: `simple-tab-${index}`,
    'aria-controls': `simple-tabpanel-${index}`,
  };
}

const UserManagement = () => {
  const [tabIndex, setTabIndex] = useState(0);
  const [loginUser] = useState(getLoginUser());
  const [roleList, setRoleList] = useState([]);
  const [circleList, setCircleList] = useState([]);
  const [isLoading, setIsLoading] = useState(false);
  const { callApi } = useApi();
  const snackbar = useCustomSnackbar();

  // States for User Creation
  const [newUser, setNewUser] = useState({
    userId: '',
    userName: '',
    role: '',
    circleCode: '',
  });
  const [isCircleDisabled, setIsCircleDisabled] = useState(false);

  // States for User Modification
  const [userIdToModify, setUserIdToModify] = useState('');
  const [fetchedUserDetails, setFetchedUserDetails] = useState(null);
  const [isModifyDialogOpen, setIsModifyDialogOpen] = useState(false);
  const [confirmDeleteDialogOpen, setConfirmDeleteDialogOpen] = useState(false);

  // Initial data fetch for dropdowns
  useEffect(() => {
    const fetchData = async () => {
      setIsLoading(true);
      const roles = await getRoleList(loginUser.capacity, callApi);
      const circles = await getCircleList(callApi);
      setRoleList(roles);
      setCircleList(circles);
      setIsLoading(false);
    };
    fetchData();
  }, [loginUser.capacity, callApi]);

  useEffect(() => {
    if (loginUser.capacity === '52') {
      const prefilledCircle = circleList.find((c) => c.circleCode === loginUser.circleCode);
      if (prefilledCircle) {
        setNewUser((prev) => ({ ...prev, circleCode: prefilledCircle }));
      }
      setIsCircleDisabled(true);
    }
  }, [loginUser.capacity, loginUser.circleCode, circleList]);

  const handleTabChange = (event, newValue) => {
    setTabIndex(newValue);
    setFetchedUserDetails(null);
    setUserIdToModify('');
  };

  const handleResetSearch = () => {
    setFetchedUserDetails(null);
    setUserIdToModify('');
  };

  const handleResetForm = () => {
    setNewUser({
      userId: '',
      userName: '',
      role: null,
      circleCode: loginUser.capacity === '52' ? newUser.circleCode : null,
    });
  };

  // This function handles Select component changes
  const handleNewUserChange = (field, selectedValue) => {
    const list = field === 'role' ? roleList : circleList;
    const key = field === 'role' ? 'capacity' : 'circleCode';
    const selectedObject = list.find((item) => item[key] === selectedValue);

    const updatedUser = { ...newUser, [field]: selectedObject };
    setNewUser(updatedUser);

    // Logic to disable/enable circle based on role selection
    if (field === 'role') {
      if (selectedObject && parseInt(selectedObject.capacity, 10) > 60) {
        setIsCircleDisabled(true);
        const frtCircle = { circleCode: '100', circleName: 'FRT/GITC Users' };
        updatedUser.circleCode = frtCircle;
      } else if (loginUser.capacity !== '52') {
        setIsCircleDisabled(false);
        // If the role is changed back, clear the circle selection
        if (newUser.circleCode?.circleCode === '100') {
          updatedUser.circleCode = null;
        }
      }
      setNewUser(updatedUser);
    }
  };

  const handleCreateUser = async () => {
    if (!newUser.userId || newUser.userId.trim() === '') {
      return snackbar('User ID field cannot be left blank. Please provide User ID to Create a new User.', 'info');
    } else if (newUser.userId && newUser.userId.length < 7) {
      return snackbar(
        'User ID must be a 7-digit number. Please ensure User ID meets this requirement to create a new user.',
        'info'
      );
    } else if (!newUser.userName || newUser.userName.trim() === '') {
      return snackbar('User Name field cannot be left blank. Please provide User Name to Create a new User.', 'info');
    } else if (!newUser.role || newUser.role.capacity === '') {
      return snackbar(
        'You need to select a User Role before you can proceed. Please choose one from the dropdown.',
        'info'
      );
    } else if (!newUser.circleCode || newUser.circleCode.circleCode === '') {
      return snackbar('You need to choose a circle before proceeding. Please select one from the dropdown.', 'info');
    }

    setIsLoading(true);
    try {
      const payload = {
        userId: newUser.userId,
        userName: newUser.userName,
        role: newUser.role.capacity,
        circleCode: newUser.circleCode.circleCode,
      };
      const result = await createUser(payload, loginUser, callApi);
      if (result == 1) {
        snackbar('User Creation request has submitted successfully.', 'success');
        setNewUser({
          userId: '',
          userName: '',
          role: null,
          circleCode: loginUser.capacity === '52' ? newUser.circleCode : null,
        });
      } else if (result == 0) {
        snackbar('User already exists and acceptance pending with FRT Checker.', 'warning');
      } else if (result == 2) {
        snackbar('An account with this User ID already exists.', 'warning');
      } else {
        snackbar('User creation failed.', 'error');
      }
    } catch (e) {
      snackbar('An error occurred during user creation.', 'error');
    }
    setIsLoading(false);
  };

  // USER MODIFICATION LOGIC
  const handleCheckDetails = async () => {
    if (!userIdToModify) {
      snackbar('Please enter a User ID to Search.', 'warning');
      return;
    } else if (userIdToModify.length < 7) {
      snackbar('The User ID you entered is too short. Please enter a 7-digit numeric User ID', 'info');
      return;
    }
    setIsLoading(true);
    setFetchedUserDetails(null);
    try {
      const data = await checkUserDetails(userIdToModify, loginUser, callApi);
      if (data.isUserExist === '2') {
        snackbar("User doesn't exists", 'warning');
      } else if (data.isUserExist === '-1') {
        snackbar("User doesn't exists / User does not belong to your circle", 'warning');
      } else {
        data.userStatus = data.status === 'A' ? 'Active' : data.status === 'P' ? 'Approval Pending' : 'Unknown';
        setFetchedUserDetails(data);
      }
    } catch (e) {
      snackbar('Failed to fetch user details.', 'error');
    }
    setIsLoading(false);
  };

  const handleKeyDown = (event) => {
    if (event.key === 'Enter') {
      event.preventDefault();
      handleCheckDetails();
    }
  };

  const handleUpdateUser = async (updatedData) => {
    setIsLoading(true);
    try {
      const result = await updateUser(updatedData, loginUser, callApi);
      if (result == 1) {
        snackbar('User Updated Successfully.', 'success');
        handleCheckDetails(); // Refresh details
      } else {
        snackbar('User has not been Updated.', 'error');
      }
    } catch (e) {
      snackbar('An error occurred while updating the user.', 'error');
    }
    setIsLoading(false);
  };

  const handleDeleteUser = async () => {
    setConfirmDeleteDialogOpen(false);
    if (!fetchedUserDetails) return;
    setIsLoading(true);
    try {
      const result = await deleteUser(fetchedUserDetails, loginUser, callApi);
      if (result == 1) {
        snackbar('User Deleted Successfully.', 'warning');
        setFetchedUserDetails(null);
        setUserIdToModify('');
      } else {
        snackbar('User has not been deleted.', 'error');
      }
    } catch (e) {
      snackbar('An error occurred while deleting the user.', 'error');
    }
    setIsLoading(false);
  };

  return (
    <Box sx={{ width: '100%' }}>
      <Box sx={{ borderBottom: 1, borderColor: 'divider' }}>
        <StyledTabs value={tabIndex} onChange={handleTabChange} aria-label="user management tabs" variant="fullWidth">
          <StyledTab icon={<PersonAddIcon />} iconPosition="start" label="Create User" {...a11yProps(0)} />
          <StyledTab icon={<ManageAccountsIcon />} iconPosition="start" label="Modify User" {...a11yProps(1)} />
        </StyledTabs>
      </Box>

      <Box sx={{ width: '100%' }}>
        {/* --- CREATE USER PANEL --- */}
        <TabPanel value={tabIndex} index={0} sx={{ p: 0 }}>
          <Grid sx={{ p: 0 }} container spacing={3} alignItems="center">
            <Grid item xs={12} md={5} fullWidth sx={{ width: '100%' }}>
              <Card
                sx={(theme) => ({
                  minHeight: '100%',
                  display: { xs: 'none', md: 'flex' },
                  flexDirection: 'column',
                  alignItems: 'center',
                  justifyContent: 'center',
                  backgroundColor: theme.palette.mode === 'dark' ? 'neutral.800' : 'background.paper',
                  boxShadow: theme.shadows[1],
                })}
              >
                <CardContent sx={{ textAlign: 'center' }}>
                  <GroupAddIcon sx={(theme) => ({ fontSize: 80, color: theme.palette.primary.main })} />
                  <Typography variant="h5" component="div" gutterBottom sx={{ mt: 2, color: 'text.primary' }}>
                    New User Registration
                  </Typography>
                  <Typography color="text.secondary">
                    Fill in the details on the below to create a new user request. Ensure all information is accurate
                    before submission.
                  </Typography>
                </CardContent>
              </Card>
            </Grid>
            {isLoading && (
              <Box sx={{ width: '100%' }}>
                <LinearProgress />
              </Box>
            )}
            {/* Form Panel */}
            <Grid item xs={12} md={7} sx={{ width: '100%' }}>
              <Card>
                <CardHeader title="User Creation" sx={{ color: 'text.primary', mb: 1 }} />
                <Divider sx={{ mb: 2 }} />
                <CardContent>
                  <Grid container spacing={3}>
                    <Grid item xs={12} sm={6}>
                      <FormControl sx={{ minWidth: 300 }}>
                        <Typography sx={{ mb: 1 }}>User ID</Typography>
                        <FormInput
                          value={newUser.userId}
                          onChange={(e) => setNewUser({ ...newUser, userId: e.target.value })}
                          inputType={'numInput'}
                          maxLength={7}
                          required
                          alignment={'left'}
                          debounceDuration={1}
                        />
                      </FormControl>
                    </Grid>
                    <Grid item xs={12} sm={6}>
                      <FormControl sx={{ minWidth: 300 }}>
                        <Typography sx={{ mb: 1 }}>User Name</Typography>
                        <FormInput
                          value={newUser.userName}
                          onChange={(e) => setNewUser({ ...newUser, userName: e.target.value })}
                          inputType={'alphaNumWithSpaceTrim'}
                          maxLength={50}
                          required
                          debounceDuration={1}
                        />
                      </FormControl>
                    </Grid>
                    <Grid item xs={12} sm={6}>
                      <FormControl sx={{ minWidth: 300 }} required>
                        <Typography sx={{ mb: 1 }}>User Role</Typography>
                        <Select
                          variant="outlined"
                          value={newUser.role ? newUser.role.capacity : ''}
                          onChange={(e) => handleNewUserChange('role', e.target.value)}
                        >
                          {roleList.map((r) => (
                            <MenuItem key={r.capacity} value={r.capacity}>
                              {r.role}
                            </MenuItem>
                          ))}
                        </Select>
                      </FormControl>
                    </Grid>
                    <Grid item xs={12} sm={6}>
                      {(!isCircleDisabled || loginUser.capacity === '52') && (
                        <FormControl sx={{ minWidth: 350 }} required>
                          <Typography sx={{ mb: 1 }}>User Circle</Typography>
                          <Select
                            variant="outlined"
                            value={newUser.circleCode ? newUser.circleCode.circleCode : ''}
                            onChange={(e) => handleNewUserChange('circleCode', e.target.value)}
                            disabled={isCircleDisabled}
                          >
                            {circleList.map((c) => (
                              <MenuItem
                                key={c.circleCode}
                                value={c.circleCode}
                              >{`${c.circleCode} - ${c.circleName}`}</MenuItem>
                            ))}
                          </Select>
                        </FormControl>
                      )}
                    </Grid>
                  </Grid>
                </CardContent>

                <Divider sx={{ my: 2 }} />

                <CardActions sx={{ justifyContent: 'space-between', py: 2 }}>
                  <Grid item xs={12} sm={5} md={4}>
                    <Box
                      sx={{
                        position: 'relative',
                        borderRadius: 1,
                        p: 1,
                        display: 'flex',
                        alignItems: 'center',
                        overflow: 'hidden',

                        '&::before': {
                          content: '""',
                          position: 'absolute',
                          inset: 0,
                          padding: '1px',
                          borderRadius: 'inherit',
                          background:
                            'conic-gradient(from 0deg at 50% 50%, rgba(0, 170, 255, 1) 0%, rgba(255, 100, 100, 1) 5%, rgba(255, 200, 50, 1) 15%, rgba(0, 170, 255, 1) 25%)',
                          // --- MODIFICATION START ---
                          // The animation now completes the sweep in 1 second and finalizes in 0.5 seconds.
                          // This is much faster than the original 3-second duration and removes the delay.
                          animation: `${sweepBorder} 1s linear forwards, ${finalizeBorder} 0.5s linear forwards 1s`,
                          // --- MODIFICATION END ---
                          WebkitMask: 'linear-gradient(#fff 0 0) content-box, linear-gradient(#fff 0 0)',
                          mask: 'linear-gradient(#fff 0 0) content-box, linear-gradient(#fff 0 0)',
                          maskComposite: 'exclude',
                          WebkitMaskComposite: 'xor',
                          //   willChange: 'background',
                        },

                        '& .inner-content': {
                          position: 'relative',
                          zIndex: 1,
                          borderRadius: 'inherit',
                          display: 'flex',
                          alignItems: 'center',
                          height: '100%',
                          width: '100%',
                          padding: 'inherit',
                          backgroundColor: 'background.paper', // Ensures proper masking
                        },
                      }}
                    >
                      <div className="inner-content">
                        <Typography variant="body1" color="text.secondary">
                          Please Note: All fields are required for successful user creation.
                        </Typography>
                      </div>
                    </Box>
                  </Grid>

                  {/* Action buttons at flex-end */}
                  <Grid item xs={12} sm={7} md={8} sx={{ display: 'flex', gap: 2, justifyContent: 'flex-end' }}>
                    <Button
                      variant="outlined"
                      onClick={handleResetForm}
                      startIcon={<ResetIcon />}
                      sx={{
                        height: 37,
                        alignItems: 'center', //
                        transition: 'background-color 0.3s ease',
                        borderRadius: 1,
                      }}
                    >
                      Clear Form
                    </Button>
                    <CustomButton
                      buttonType="custom"
                      label="Create User"
                      onClickHandler={handleCreateUser}
                      customButtonProps={{
                        bgColor: 'success.deep',
                        hoverColor: 'success.dark',
                      }}
                      disabled={isLoading}
                      startIcon={isLoading ? <CircularProgress size={20} color="inherit" /> : <PersonAddIcon />}
                    />
                  </Grid>
                </CardActions>
              </Card>
            </Grid>
          </Grid>
        </TabPanel>

        {/* --- MODIFY USER PANEL --- */}
        <TabPanel value={tabIndex} index={1} sx={{ p: 0 }}>
          <Typography variant="h6" gutterBottom>
            Find and Modify an Existing User
          </Typography>
          <Grid component={Card} container spacing={2} alignItems="flex-end" sx={{ boxShadow: 1, mt: 1 }}>
            <Grid item xs={12} sm={5} md={4}>
              <FormControl fullWidth sx={{ minWidth: 300 }}>
                <Typography sx={{ mb: 1 }}>Enter User ID</Typography>
                <FormInput
                  value={userIdToModify}
                  onChange={(e) => setUserIdToModify(e.target.value)}
                  inputType={'numInput'}
                  maxLength={7}
                  alignment={'left'}
                  debounceDuration={1}
                  placeholder={'Enter User ID'}
                  onKeyDown={handleKeyDown}
                />
              </FormControl>
            </Grid>
            <Grid item xs={12} sm={7} md={8} sx={{ display: 'flex', gap: 2 }}>
              <Button
                variant="contained"
                color="secondary"
                onClick={handleCheckDetails}
                disabled={isLoading}
                startIcon={<SearchIcon />}
              >
                {isLoading ? <CircularProgress size={24} color="inherit" /> : 'Search Details'}
              </Button>
              {fetchedUserDetails && (
                <Button variant="outlined" onClick={handleResetSearch} startIcon={<ResetIcon />}>
                  Clear Search Results
                </Button>
              )}
            </Grid>
          </Grid>
          <Divider sx={{ mt: 2 }} />
          {isLoading && (
            <Box sx={{ width: '100%' }}>
              <LinearProgress />
            </Box>
          )}

          {!isLoading && !fetchedUserDetails && (
            <Box
              sx={{
                display: 'flex',
                justifyContent: 'center',
                alignItems: 'center',
                minHeight: 150,
                textAlign: 'center',
                p: 3,
              }}
            >
              <Typography
                variant="h6" // Or a variant that suits your design
                color="text.secondary" // or "text.disabled", which is a lighter gray
                sx={{ opacity: 0.4 }} // Adjust opacity for desired fade effect
              >
                To find and modify an existing user, use the search bar above.
              </Typography>
            </Box>
          )}
          {fetchedUserDetails && (
            <UserDetailsCard
              fetchedUserDetails={fetchedUserDetails}
              roleList={roleList}
              circleList={circleList}
              isLoading={isLoading}
              setIsModifyDialogOpen={setIsModifyDialogOpen}
              setConfirmDeleteDialogOpen={setConfirmDeleteDialogOpen}
            />
          )}
        </TabPanel>
      </Box>

      {/* Reusable Dialog for Update */}
      {fetchedUserDetails && (
        <UserModifyDialog
          open={isModifyDialogOpen}
          onClose={() => setIsModifyDialogOpen(false)}
          userDetails={fetchedUserDetails}
          roleList={roleList}
          circleList={circleList}
          onUpdate={handleUpdateUser}
        />
      )}

      {/* Confirmation Dialog for Delete */}
      <Dialog open={confirmDeleteDialogOpen} onClose={() => setConfirmDeleteDialogOpen(false)}>
        <DialogTitle>Confirm Deletion</DialogTitle>
        <DialogContent>
          <DialogContentText>
            Are you sure you want to delete this user ({fetchedUserDetails?.userId})?
          </DialogContentText>
        </DialogContent>
        <DialogActions>
          <Button onClick={() => setConfirmDeleteDialogOpen(false)}>Cancel</Button>

          <CustomButton buttonType="delete" label="Delete" onClickHandler={handleDeleteUser} autoFocus />
        </DialogActions>
      </Dialog>
    </Box>
  );
};

export default UserManagement;
