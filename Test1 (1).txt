class Solution {

    public int Helper(TreeNode root , int count , int max) {
         if(root == null) {
            return 0;
        }      
       
        if(root.val >= max) {

            count++;
            max = Math.max(max , root.val);          

        }


        count += Helper(root.left , 0 , max);
        count += Helper(root.right , 0 , max);

        return count;
        
    }
    public int goodNodes(TreeNode root) {      
       return Helper(root , 0 , INTMAX);
      
   }
}
