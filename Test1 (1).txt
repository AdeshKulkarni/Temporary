  useEffect(() => {
    // The AbortController is used to cancel the API request if the component unmounts.
    const controller = new AbortController();
    const { signal } = controller;

    const fetchAllInitialData = async () => {
      try {
        const createOrEditPayload = {
          circle: userData.circleCode,
          QED: userData.quarterEndDate,
          reportMasterId: '4001',
        };
        // Pass the signal to your API calls
        const createOrEditResponse = await callApi(DO_CREATE_OR_EDIT_URL, createOrEditPayload, 'POST', signal);

        let currentReportId = '';
        let currentCreateEditSts = '';
        if (createOrEditResponse) {
          [currentReportId, currentCreateEditSts] = createOrEditResponse.split('~');
        } else {
          throw new Error('Invalid response from doCreateOrEdit');
        }

        setReportId(currentReportId);

        let mainFormDataResponse;
        const mainDataPayload = { ...createOrEditPayload, reportId: currentReportId };

        if (currentCreateEditSts === '10') {
          mainFormDataResponse = await callApi(INDEX_API_URL, mainDataPayload, 'POST', signal);
        } else {
          mainFormDataResponse = await callApi(EDIT_API_URL, mainDataPayload, 'POST', signal);
        }

        const newFormData = { ...initialFormData };
        const newFormErrors = {};

        tableConfig.forEach((row) => {
          const apiValue = mainFormDataResponse[String(row.dctNo)];
          newFormData[row.modelKey] = formatApiValue(apiValue, row.validationType);
          newFormErrors[row.modelKey] = '';
        });

        const payload = {
          circle: userData.circleCode,
          QED: userData.quarterEndDate,
          reportMasterId: '4001',
        };
        const responseData = await callApi(ADDITIONAL_DATA_URL, payload, 'POST', signal);

        const additionalDataMapping = {
          val248: 'valuedct45',
          val249: 'valuedct46',
          val250: 'valuedct47',
          val254: 'valuedct48',
          val255: 'valuedct49',
          val830: 'valuedct51',
        };
        for (const apiKey in additionalDataMapping) {
          const modelKey = additionalDataMapping[apiKey];
          const rowConfig = tableConfig.find((r) => r.modelKey === modelKey);
          if (rowConfig) {
            newFormData[modelKey] = formatApiValue(responseData[apiKey], rowConfig.validationType);
          }
        }

        setFormData(newFormData);
        setFormErrors(newFormErrors);
        setQedDate(userData.quarterEndDate);
      } catch (error) {
        // This is the professional way to handle request cancellations.
        // We check if the error was caused by the request being aborted.
        // The error name 'CanceledError' is common with libraries like Axios.
        if (error.name === 'CanceledError') {
          console.log('Initial data fetch was cancelled. This is expected and not an error.');
        } else {
          // Only show the snackbar for actual network or server errors.
          console.error('Failed to fetch initial data:', error);
          showSnackbar('Failed to load initial data. Please try again.', 'error');

          // Initialize form with default values on error
          const tempFormData = {};
          tableConfig.forEach((row) => {
            tempFormData[row.modelKey] = row.validationType === 'numInput' ? '0' : '0.00';
          });
          setFormData(tempFormData);
        }
      } finally {
        // Ensure isLoading is set to false even if the request was cancelled.
        if (!signal.aborted) {
           await new Promise((resolve) => setTimeout(resolve, 2000));
           setIsLoading(false);
        }
      }
    };

    fetchAllInitialData();

    // This is the cleanup function that runs when the component unmounts.
    // It aborts any ongoing API requests initiated by this effect.
    return () => {
      controller.abort();
    };
  }, [tableConfig, callApi, showSnackbar, userData]);
