import React, { useState, useEffect, useMemo } from "react";
import {
  Divider,
  Typography,
  Box,
  Stack,
  Button,
  Paper,
  MenuItem,
  TextField,
  Tabs,
  Tab,
} from "@mui/material";
import { DataGrid, GridToolbar, GridOverlay } from "@mui/x-data-grid";
import { encrypt } from "../../Security/AES-GCM256";
import axios from "axios";
import downloadFile from "../../DownloadFile/DownloadFile";

const iv = crypto.getRandomValues(new Uint8Array(12));
const ivBase64 = btoa(String.fromCharCode(...iv));
const salt = crypto.getRandomValues(new Uint8Array(16));
const saltBase64 = btoa(String.fromCharCode(...salt));

const EmpanelmentData = () => {
  const [data, setData] = useState([]);
  const [userIp, setUserIp] = useState({
    selectionMode: "Empanelment",
    datatype: "Active",
    type: "",
    subType: "",
    year: "",
  });
  const [selectionOptions, setSelectionOptions] = useState({});
  const [archiveYears, setArchiveYears] = useState([]);
  const [loadingYears, setLoadingYears] = useState(false);

  // --- Columns for "Empanelment" mode ---
  const empanelmentColumns = [
    { field: "FRN", headerName: "FRN", width: 170, flex: 1 },
    { field: "FIRM_NAME", headerName: "FIRM NAME", width: 200, flex: 1.5 },
    {
      field: "EMPANELMENT_SUB_TYPE",
      headerName: "EMPANELMENT SUB TYPE",
      width: 280,
      flex: 2,
      valueGetter: (params) => {
        const allSubTypes = Object.values(selectionOptions).flat();
        return allSubTypes.find(opt => String(opt.ASSIGNMENT_TYPE_ID) === String(params.value))?.ASSIGNMENT_SUBTYPE || params.value;
      },
    },
    { field: "EMPANELMENT_TYPE", headerName: "EMPANELMENT TYPE", width: 160, flex: 1 },
    { field: "CONTACT_PERSON", headerName: "Contact Person", width: 180, flex: 1 },
    { field: "POC_EMAIL", headerName: "Contact Email", width: 220, flex: 1 },
  ];

  // --- Columns for "Assignment" mode ---
  const assignmentColumns = [
    { field: "FRN_NO", headerName: "FRN", width: 170, flex: 1 },
    { field: "ACCOUNT_NAME", headerName: "Account Name", width: 250, flex: 1.5 },
    { field: "ACCOUNT_NUMBER", headerName: "Account Number", width: 200, flex: 1 },
    { field: "BRANCH_CODE", headerName: "Branch Code", width: 150, flex: 1 },
    {
      field: "AUDIT_DATE",
      headerName: "Audit Date",
      width: 180,
      flex: 1,
      valueGetter: (params) => {
        // **FIX:** Explicitly check for a truthy value.
        if (!params.value) return 'N/A';
        return new Date(params.value).toLocaleDateString();
      }
    },
    {
      field: "ASSIGNMENT_TYPE",
      headerName: "Assignment Sub Type",
      width: 220,
      flex: 1.5,
       valueGetter: (params) => {
        // **FIX:** Handle missing data and provide a better fallback.
        if (!params.value) return 'N/A'; // Show N/A if the ID is missing.
        const allSubTypes = Object.values(selectionOptions).flat();
        const foundOption = allSubTypes.find(opt => String(opt.ASSIGNMENT_TYPE_ID) === String(params.value));
        return foundOption ? foundOption.ASSIGNMENT_SUBTYPE : `Unknown ID: ${params.value}`;
      },
    },
  ];

  const filteredEmpanelmentRows = useMemo(() => {
    if (userIp.selectionMode !== 'Empanelment' || !Array.isArray(data)) {
        return [];
    }
    return data.filter(row => row && row.FRN);
  }, [data, userIp.selectionMode]);

  const rows = userIp.selectionMode === 'Empanelment' ? filteredEmpanelmentRows : data;

  useEffect(() => {
    fetchOptions();
  }, []);

  useEffect(() => {
    if (userIp.selectionMode === "Empanelment" && userIp.type && userIp.subType) {
      fetchEmpanelmentData();
    } else if (userIp.selectionMode === "Assignment" && userIp.subType) {
      fetchAssignmentData();
    }
  }, [userIp.selectionMode, userIp.datatype, userIp.type, userIp.subType, userIp.year]);

  useEffect(() => {
    if (userIp.selectionMode === 'Empanelment' && userIp.datatype === "Archive") {
      fetchArchiveYears();
    }
  }, [userIp.datatype, userIp.selectionMode]);

  const postEncryptedData = async (url, dataToSend) => {
    const jsonFormData = JSON.stringify(dataToSend);
    const encryptedData = await encrypt(iv, salt, jsonFormData);
    const payload = { iv: ivBase64, salt: saltBase64, data: encryptedData };
    return axios.post(url, payload, {
      headers: { Authorization: `Bearer ${localStorage.getItem("token")}` },
    });
  };

  const fetchOptions = async () => {
    try {
      const response = await axios.post("Server/IAMServices/getAssignmentType", {}, {
        headers: { Authorization: `Bearer ${localStorage.getItem("token")}` },
      });
      const result = response.data.result || [];
      const options = {};
      new Set(result.map((op) => op.ASSIGNMENT_TYPE_NAME)).forEach((typeName) => {
        options[typeName] = result.filter((op) => op.ASSIGNMENT_TYPE_NAME === typeName);
      });
      setSelectionOptions(options);
    } catch (e) { console.error("Failed to fetch selection options:", e); }
  };

  const fetchArchiveYears = async () => {
    setLoadingYears(true);
    try {
      const response = await postEncryptedData("Server/IAMServices/getArchiveYears", {});
      setArchiveYears(response.data.result || []);
    } catch (e) { console.error("Failed to fetch archive years:", e); }
    finally { setLoadingYears(false); }
  };

  const fetchEmpanelmentData = async () => {
    const dataToSend = {
      Type: userIp.datatype,
      EMPANELMENT_TYPE: userIp.type,
      EMPANELMENT_SUB_TYPE: userIp.subType,
      financial_year: userIp.year,
    };
    try {
      const response = await postEncryptedData("Server/IAMServices/getActiveFRN", dataToSend);
      setData(response.data.result || []);
    } catch (e) {
      console.error("Failed to fetch empanelment data:", e);
      setData([]);
    }
  };

  const fetchAssignmentData = async () => {
    const dataToSend = { assignmentType: userIp.subType };
    try {
      const response = await postEncryptedData("Server/IAMServices/fetchAssignData", dataToSend);
      const result = response.data.result;

      // --- DEBUGGING LINE ---
      console.log('API Response for Assignment:', result);

      if (result && Object.keys(result).length > 0) {
        setData([result]);
      } else {
        setData([]);
      }
    } catch (e) {
      console.error("Failed to fetch assignment data:", e);
      setData([]);
    }
  };

  const handleModeChange = (event) => {
    setUserIp({
      selectionMode: event.target.value,
      datatype: 'Active', type: '', subType: '', year: ''
    });
    setData([]);
  };

  const handleTabChange = (event, newValue) => {
    setUserIp((prev) => ({ ...prev, datatype: newValue, type: "", subType: "", year: "" }));
    setData([]);
  };

  const handleSelectType = (type) => {
    setUserIp((prev) => ({ ...prev, type, subType: "" }));
  };

  const handleSelectSubType = (subType) => {
    setUserIp((prev) => ({ ...prev, subType }));
  };

  const handleSelectYear = (year) => {
    setUserIp((prev) => ({ ...prev, year }));
  };

  const handleDownload = async () => {
    if (rows.length === 0) { alert("No data to download."); return; }
    try {
      const response = await postEncryptedData("Server/IAMServices/download_empanelment_data", rows);
      if (response.data.result) {
        downloadFile(response.data.result, "excel", `${userIp.selectionMode}_list`);
      }
    } catch (e) { console.error("Failed to download file:", e); }
  };

  const CustomNoRowsOverlay = () => {
    let message = "Please make a selection to view data.";
    if (userIp.selectionMode === 'Empanelment') {
      if (!userIp.type) message = "Please select a Type.";
      else if (!userIp.subType) message = "Please select a Sub Type.";
      else message = "No data found for the selected Empanelment filters.";
    } else if (userIp.selectionMode === 'Assignment') {
      if (!userIp.subType) message = "Please select a Sub Type to fetch Assignment data.";
      else message = "No Assignment data found for the selected Sub Type.";
    }
    return (
      <GridOverlay>
        <Box p={2} sx={{ textAlign: "center", color: "text.secondary" }}><Typography>{message}</Typography></Box>
      </GridOverlay>
    );
  };

  const showDataGrid = userIp.subType && (userIp.selectionMode === 'Assignment' || (userIp.selectionMode === 'Empanelment' && userIp.type));

  const getRowId = (row) => userIp.selectionMode === 'Empanelment' ? row?.FRN : row?.FRN_NO;

  return (
    <Box sx={{ width: "100%" }}>
      <Typography variant="h5" sx={{ mb: 2 }}>Data Viewer</Typography>
      <Divider sx={{ mb: 3 }} />

      <Paper sx={{ width: "100%", overflow: "hidden", p: 2 }}>
        <Stack direction="row" spacing={2} sx={{ p: 2 }} alignItems="center">
          <TextField select label="Mode" value={userIp.selectionMode} onChange={handleModeChange} sx={{ width: 200 }} size="small">
            <MenuItem value="Empanelment">Empanelment</MenuItem>
            <MenuItem value="Assignment">Assignment</MenuItem>
          </TextField>

          {userIp.selectionMode === 'Assignment' && (
            <TextField select label="Sub Type" value={userIp.subType} onChange={(e) => handleSelectSubType(e.target.value)} sx={{ width: 220 }} size="small">
              {Object.values(selectionOptions).flat().map((option) => (
                <MenuItem key={option.ASSIGNMENT_TYPE_ID} value={option.ASSIGNMENT_TYPE_ID}>
                  {option.ASSIGNMENT_SUBTYPE}
                </MenuItem>
              ))}
            </TextField>
          )}
        </Stack>

        {userIp.selectionMode === 'Empanelment' && (
          <>
            <Box sx={{ borderBottom: 1, borderColor: "divider" }}>
              <Tabs value={userIp.datatype} onChange={handleTabChange}>
                <Tab label="Active" value="Active" />
                <Tab label="Archive" value="Archive" />
              </Tabs>
            </Box>
            <Stack direction="row" spacing={2} sx={{ p: 2 }} alignItems="center">
              {userIp.datatype === "Archive" && (
                <TextField select label="Year" value={userIp.year} onChange={(e) => handleSelectYear(e.target.value)} sx={{ width: 200 }} size="small" disabled={loadingYears}>
                  {archiveYears.map((year) => <MenuItem key={year} value={year}>{year}</MenuItem>)}
                </TextField>
              )}
              <TextField select label="Type" value={userIp.type} onChange={(e) => handleSelectType(e.target.value)} sx={{ width: 200 }} size="small">
                {Object.keys(selectionOptions).map((type) => <MenuItem key={type} value={type}>{type}</MenuItem>)}
              </TextField>
              {userIp.type && (
                <TextField select label="Sub Type" value={userIp.subType} onChange={(e) => handleSelectSubType(e.target.value)} sx={{ width: 220 }} size="small" disabled={!selectionOptions[userIp.type]}>
                  {(selectionOptions[userIp.type] || []).map((option) => (
                    <MenuItem key={option.ASSIGNMENT_TYPE_ID} value={option.ASSIGNMENT_TYPE_ID}>
                      {option.ASSIGNMENT_SUBTYPE}
                    </MenuItem>
                  ))}
                </TextField>
              )}
            </Stack>
          </>
        )}

        {showDataGrid && (
          <Box sx={{ width: "100%", mt: 2 }}>
            {userIp.selectionMode === 'Empanelment' && (
              <Box sx={{ display: "flex", justifyContent: "flex-end", mb: 1 }}>
                <Button onClick={handleDownload} variant="contained" size="medium" disabled={rows.length === 0}>Download Excel</Button>
              </Box>
            )}
            <Box sx={{ height: 500, width: "100%" }}>
              <DataGrid
                rows={rows}
                columns={userIp.selectionMode === 'Empanelment' ? empanelmentColumns : assignmentColumns}
                getRowId={getRowId}
                slots={{ noRowsOverlay: CustomNoRowsOverlay, toolbar: GridToolbar }}
                pageSizeOptions={[10, 25, 50]}
                initialState={{ pagination: { paginationModel: { pageSize: 10 } } }}
                sx={{ "& .MuiDataGrid-columnHeaders": { backgroundColor: "grey.100", fontWeight: "bold" } }}
              />
            </Box>
          </Box>
        )}
      </Paper>
    </Box>
  );
};

export default EmpanelmentData;
