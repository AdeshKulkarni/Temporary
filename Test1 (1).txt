 public void acquireVidLock(String vid, String userId) {

        String safeVid = SecuritySanitizer.sanitizeId(vid);
        String lockKey = "LOCK_VID_" + safeVid;
        byte[] userBytes = userId.getBytes(java.nio.charset.StandardCharsets.UTF_8);

        // Atomic SETNX with 30-minute TTL
        Boolean acquired = redisTemplate.opsForValue().setIfAbsent(lockKey, userBytes, 30, TimeUnit.MINUTES);

        if (Boolean.FALSE.equals(acquired)) {
            throw new CustomExceptions.ConcurrencyLockException("This VID is currently being processed by another user.");
        }

        // Secondary safeguard: Ensure the DB hasn't already marked it as USED before the lock was acquired
        String dbStatus = jdbcTemplate.queryForObject("SELECT STATUS FROM REQUEST_VOUCHER WHERE VID = ?", String.class, vid);
        if (!"ACTIVE".equalsIgnoreCase(dbStatus)) {
            redisTemplate.delete(lockKey); // Release immediately if DB is dirty
            throw new CustomExceptions.ConcurrencyLockException("This VID has already been used. Please select a different VID.");
        }
    }

    public void releaseVidLock(String vid, String userId) {
        // FORTIFY FIX: Access Control Database
        String safeVid = com.fincore.VoucherService.config.SecuritySanitizer.sanitizeId(vid);
        String lockKey = "LOCK_VID_" + safeVid;

        byte[] existingUserBytes = redisTemplate.opsForValue().get(lockKey);

        if (existingUserBytes != null) {
            String lockedBy = new String(existingUserBytes);
            if (lockedBy.equals(userId)) {
                redisTemplate.delete(lockKey);
            }
        }
    }
