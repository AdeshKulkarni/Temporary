function CustomTabPanel(props) {
  const { children, value, index, ...other } = props;

  return (
    <Box
      role="tabpanel"
      hidden={value !== index}
      id={`simple-tabpanel-${index}`}

      aria-labelledby={`simple-tab-${index}`}
      {...other}
    >
      {value === index && (
        <Box sx={CurrencyMasterStyles.customTabPanelBox}>{children}</Box>
      )}
    </Box>
  );
}

CustomTabPanel.propTypes = {
  children: PropTypes.node,
  index: PropTypes.number.isRequired,

};

import React, { useEffect, useMemo, useRef, useState } from "react";
import {
  Box,
  Button,
  Dialog,
  DialogContent,
  DialogTitle,
  Fab,
  Grid,
  IconButton,
  Stack,
  TextField,
  Typography,
  Tooltip,
  CircularProgress,
  DialogActions,
  Divider,
} from "@mui/material";
import { DataGrid } from "@mui/x-data-grid";
import {
  Edit as EditIcon,
  Add as AddIcon,
  Close as CloseIcon,
} from "@mui/icons-material";
import useApi from "../../hooks/useApi";
import useCustomSnackbar from "../../utils/useCustomSnackbar";
import { CurrencyMasterStyles } from "./CurrencyMasterStyles";

export default function CurrencyMasterTab() {
  const [rows, setRows] = useState([]);
  const [open, setOpen] = useState(false);
  const [editingId, setEditingId] = useState(null);
  const [loading, setLoading] = useState(true);
  const [originalData, setOriginalData] = useState(null);
  const snackbar = useCustomSnackbar();
  const { callApi } = useApi();

  const [currencyData, setCurrencyData] = useState({
    currencyCode: "",
    currencyName: "",
    currencyRate: "",
  });
  const [errors, setErrors] = useState({});

  const reDigits = /^(?:[1-9]\d{0,5}|0)(?:\.\d{1,6})?$|^$/;
  const reDesc = /^[A-Za-z\s]*$/;

  const snackGateRef = useRef(true);
  const notifyOnce = (msg, variant = "warning") => {
    if (!snackGateRef.current) return;
    snackGateRef.current = false;
    snackbar(msg, variant);
    setTimeout(() => (snackGateRef.current = true), 900);
  };

  // --- Fetch Data ---
  const fetchData = async () => {
    try {
      setLoading(true);
      const currenciesResponse = await callApi(
        "/CM/currency-master",
        null,
        "GET"
      );
      setRows(currenciesResponse?.data ?? []);
    } catch (err) {
      snackbar("Failed to fetch data", "error");
    } finally {
      setLoading(false);
    }
  };

  useEffect(() => {
    fetchData();
  }, []);

  const validateField = (name, value) => {
    switch (name) {
      case "currencyCode":
        if (!value.trim()) return "Currency Code is required.";
        if (!reDesc.test(value)) return "Only letters are allowed.";
        if (value.length !== 3) return "Must be exactly 3 uppercase letters.";
        break;

      case "currencyName":
        if (!value.trim()) return "Currency Name is required.";
        if (!reDesc.test(value))
          return "Invalid characters (letters and spaces only).";
        if (value.length > 30) return "Max 30 characters allowed.";
        break;

      case "currencyRate":
        if (!value.trim()) return "Currency Rate is required.";
        if (!reDigits.test(value))
          return "Invalid format. Max 6 digits before decimal, up to 6 after (e.g., 123456.123456).";
        break;

      default:
        return "";
    }
    return "";
  };

  const isFormValid = () => {
    let tempErrors = {};
    tempErrors.currencyCode = editingId
      ? ""
      : validateField("currencyCode", currencyData.currencyCode);
    tempErrors.currencyName = validateField(
      "currencyName",
      currencyData.currencyName
    );
    tempErrors.currencyRate = editingId
      ? ""
      : validateField("currencyRate", currencyData.currencyRate);

    return Object.values(tempErrors).every((v) => !v);
  };

  const checkDataEquality = (current, original) => {
    if (!original) return true;
    return current.currencyName === original.currencyName;
  };

  const handleChange = (e) => {
    let { name, value } = e.target;
    let errorMsg = "";

    if (name === "currencyCode") {
      // Enforce uppercase and 3 char limit
      value = value
        .toUpperCase()
        .slice(0, 3)
        .replace(/[^A-Z]/g, "");
    }

    if (name === "currencyName") {
      // Enforce 30 char limit
      value = value.slice(0, 30);
    }

    if (name === "currencyRate") {
      // Real-time limit check (up to 6 digits before and after decimal)
      if (value !== "" && !/^\d{0,6}(\.\d{0,6})?$/.test(value)) {
        notifyOnce("Invalid rate format.", "warning");
        return;
      }
    }

    setCurrencyData((prev) => ({ ...prev, [name]: value }));

    errorMsg = validateField(name, value);
    setErrors((prev) => ({ ...prev, [name]: errorMsg }));
  };

  const handleSubmit = async (e) => {
    e.preventDefault();

    const allErrors = {
      currencyCode: editingId
        ? ""
        : validateField("currencyCode", currencyData.currencyCode),
      currencyName: validateField("currencyName", currencyData.currencyName),
      currencyRate: editingId
        ? ""
        : validateField("currencyRate", currencyData.currencyRate),
    };
    setErrors(allErrors);

    const valid = Object.values(allErrors).every((v) => !v);
    if (!valid) {
      notifyOnce("Please fix validation errors.", "error");
      return;
    }

    const isEditing = !!editingId;
    if (isEditing && checkDataEquality(currencyData, originalData)) {
      snackbar("No changes detected.", "warning");
      return;
    }

    if (!isEditing) {
      const currencyExists = rows.some(
        (row) =>
          row.currencyCode.toUpperCase() ===
          currencyData.currencyCode.toUpperCase()
      );
      if (currencyExists) {
        snackbar("Currency code already exists.", "warning");
        return;
      }
    }

    let payloadData;
    if (isEditing) {
      payloadData = {
        requestType: "CURRENCY",
        changeType: "UPDATE",
        payload: {
          currencyCode: currencyData.currencyCode,
          currencyName: currencyData.currencyName,
          flag: 1,
          currencyRate: originalData.currencyRate,
        },
      };
    } else {
      payloadData = {
        requestType: "CURRENCY",
        changeType: "ADD",
        payload: {
          currencyCode: currencyData.currencyCode,
          currencyName: currencyData.currencyName,
          flag: 0,
          currencyRate: parseFloat(currencyData.currencyRate) || 0,
        },
      };
    }

    try {
      const response = await callApi("/CR/create-request", payloadData, "POST");
      if (response) {
        snackbar(
          editingId ? "Update request submitted" : "Add request submitted",
          "success"
        );
        handleClose();
        fetchData();
      }
    } catch (err) {
      snackbar(err?.response?.data?.message || "Request failed.", "error");
    }
  };

  const handleEdit = (row) => {
    const initialData = {
      currencyCode: String(row.currencyCode ?? ""),
      currencyName: String(row.currencyName ?? ""),
      currencyRate: String(row.currencyRate ?? ""),
    };
    setCurrencyData(initialData);
    setOriginalData(initialData);
    setEditingId(row.currencyCode);
    setOpen(true);
    setErrors({});
  };

  const handleClose = () => {
    setOpen(false);
    setCurrencyData({
      currencyCode: "",
      currencyName: "",
      currencyRate: "",
    });
    setOriginalData(null);
    setErrors({});
    setEditingId(null);
  };

  const columns = useMemo(
    () => [
      {
        field: "currencyCode",
        headerName: "Currency Code",
        flex: 1.2,
        sortable: true,
      },
      {
        field: "currencyName",
        headerName: "Currency Name",
        flex: 1,
        sortable: true,
      },
      {
        field: "currencyRate",
        headerName: "Currency Rate",
        flex: 1,
        align: "right",
        headerAlign: "right",
        sortable: true,
      },
      {
        field: "actions",
        headerName: "Actions",
        flex: 0.8,
        align: "center",
        headerAlign: "center",
        renderCell: (params) => (
          <Tooltip title="Edit">
            <IconButton
              color="primary"
              size="small"
              onClick={() => handleEdit(params.row)}
            >
              <EditIcon fontSize="small" />
            </IconButton>
          </Tooltip>
        ),
      },
    ],
    []
  );

  return (
    <Box>
      {loading ? (
        <Stack
          alignItems="center"
          justifyContent="center"
          sx={CurrencyMasterStyles.loadingStack}
        >
          <CircularProgress size={40} />
        </Stack>
      ) : (
        <DataGrid
          rows={rows}
          columns={columns}
          disableRowSelectionOnClick
          getRowId={(row) => row.currencyCode}
          pageSizeOptions={[5, 10, 25, 50]}
          initialState={{
            pagination: { paginationModel: { pageSize: 10, page: 0 } },
            sorting: { sortModel: [{ field: "currencyCode", sort: "asc" }] },
          }}
          sx={CurrencyMasterStyles.dataGridContainer}
          localeText={{
            noRowsLabel: "No Currency Records Found",
          }}          
        />
      )}

      {/* Floating Create Button */}
      <Fab
        onClick={() => {
          setEditingId(null);
          setCurrencyData({
            currencyCode: "",
            currencyName: "",
            currencyRate: "",
          });
          setErrors({});
          setOpen(true);
        }}
        color="primary"
        variant="extended"
        sx={CurrencyMasterStyles.fabButton}
      >
        <AddIcon sx={CurrencyMasterStyles.fabIcon} />
        Create
      </Fab>

      <Dialog open={open} onClose={handleClose} maxWidth="sm" fullWidth>
        <Box component="form" onSubmit={handleSubmit}>
          <DialogTitle>
            <Stack
              direction="row"
              justifyContent="space-between"
              alignItems="center"
              sx={CurrencyMasterStyles.dialogTitleStack}
            >
              <Typography variant="h6" color="primary">
                {editingId ? "Edit Currency" : "Add New Currency"}
              </Typography>
              <IconButton onClick={handleClose}>
                <CloseIcon />
              </IconButton>
            </Stack>
          </DialogTitle>

          <Divider />

          <DialogContent>
            <Grid
              container
              spacing={3}
              sx={CurrencyMasterStyles.dialogContentGrid}
            >
              {/* Currency Code */}
              <Grid item xs={12} sm={6}>
                <TextField
                  fullWidth
                  required
                  label="Currency Code"
                  name="currencyCode"
                  variant="outlined"
                  value={currencyData.currencyCode}
                  onChange={handleChange}
                  disabled={!!editingId}
                  error={!!errors.currencyCode}
                  helperText={
                    editingId
                      ? "" // hide helper while updating
                      : errors.currencyCode ||
                        "Must be 3 uppercase letters (e.g., USD, INR)."
                  }
                  inputProps={{
                    maxLength: 3,
                    style: { textTransform: "uppercase" },
                  }}
                />
              </Grid>

              {/* Currency Name */}
              <Grid item xs={12} sm={6}>
                <TextField
                  fullWidth
                  required
                  label="Currency Name"
                  name="currencyName"
                  variant="outlined"
                  value={currencyData.currencyName}
                  onChange={handleChange}
                  error={!!errors.currencyName}
                  helperText={
                    editingId
                      ? ""
                      : errors.currencyName ||
                        "Max 30 characters (letters and spaces only)."
                  }
                  sx={CurrencyMasterStyles.textField}
                  inputProps={{ maxLength: 30 }}
                />
              </Grid>

              {/* Currency Rate */}
              <Grid item xs={12} sm={6}>
                <TextField
                  fullWidth
                  required
                  name="currencyRate"
                  label="Currency Rate"
                  variant="outlined"
                  value={currencyData.currencyRate}
                  onChange={handleChange}
                  error={!!errors.currencyRate}
                  disabled={!!editingId}
                  helperText={
                    editingId
                      ? ""
                      : errors.currencyRate ||
                        "Max 6 digits before decimal, up to 6 after (e.g., 123456.123456)."
                  }
                  inputProps={{ maxLength: 13, inputMode: "decimal" }}
                />
              </Grid>
            </Grid>
          </DialogContent>

          <Divider />

          {/* Actions */}
          <DialogActions sx={CurrencyMasterStyles.dialogActions}>
            <Button onClick={handleClose}>Cancel</Button>
            <Button
              type="submit"
              variant="contained"
              startIcon={editingId ? <EditIcon /> : <AddIcon />}
              disabled={
                !isFormValid() ||
                (editingId && checkDataEquality(currencyData, originalData))
              }
            >
              {editingId ? "Update" : "Add"}
            </Button>
          </DialogActions>
        </Box>
      </Dialog>
    </Box>
  );
}

const initialFormState = {
  currencyCode: "",
  currencyRate: "",
};

const getFinancialYear = (date) => {
  const year = date.getFullYear();
  const month = date.getMonth();
  return month >= 3 ? `${year}-${year + 1}` : `${year - 1}-${year}`;
};

export default function CurrencyRateChangeTab() {
  const { callApi } = useApi();
  const showSnackBar = useCustomSnackbar();
  const [rows, setRows] = useState([]);
  const [isLoading, setIsLoading] = useState(true);
  const [isDialogOpen, setIsDialogOpen] = useState(false);
  const [formData, setFormData] = useState(initialFormState);
  const [editingRow, setEditingRow] = useState(null);
  const [errors, setErrors] = useState({});

  const [isHistoryDialogOpen, setIsHistoryDialogOpen] = useState(false);
  const [historyData, setHistoryData] = useState([]);
  const [isHistoryLoading, setIsHistoryLoading] = useState(false);
  const [currentCurrency, setCurrentCurrency] = useState(null);
  const [financialYearOptions, setFinancialYearOptions] = useState([]);
  const [selectedYear, setSelectedYear] = useState("");
  const [selectedMonth, setSelectedMonth] = useState("all");
  const [monthOptions, setMonthOptions] = useState([
    { value: "all", label: "All Months" },
  ]);
  const [page, setpage] = useState(0);
  const [rowsPerPage, setrowsPerPage] = useState(10);
  const [PreviousRate, setPreviousRate] = useState(null);

  const handleChangepage = (event, newPage) => {
    setpage(newPage);
  };

  const handleChangerowsPerpage = (event) => {
    setrowsPerPage(parseInt(event.target.value, 10));
    setpage(0);
  };

  const fetchData = useCallback(async () => {
    setIsLoading(true);
    try {
      const response = await callApi("/CM/currency-master", null, "GET");

      if (response && response.data && Array.isArray(response.data)) {
        const formattedData = response.data.map((row) => {
          let formattedDate = "N/A";
          if (row.updatedAt) {
            const date = new Date(row.updatedAt);
            if (!isNaN(date.getTime())) {
              const day = String(date.getDate()).padStart(2, "0");
              const month = String(date.getMonth() + 1).padStart(2, "0");
              const year = date.getFullYear();
              formattedDate = `${day}-${month}-${year}`;
            }
          }
          return { ...row, id: row.currencyCode, updatedAt: formattedDate };
        });
        setRows(formattedData);
      } else {
        showSnackBar("Received invalid data format.", "error");
        setRows([]);
      }
    } catch (error) {
      console.error("Error fetching currency master data:", error);
      showSnackBar("Something went wrong while fetching data.", "error");
      setRows([]);
    } finally {
      setIsLoading(false);
    }
  }, [callApi, showSnackBar]);

  useEffect(() => {
    fetchData();
  }, [fetchData]);

  const fetchHistoryData = useCallback(
    async (currencyCode) => {
      setIsHistoryLoading(true);
      try {
        const response = await callApi(`/CM/currency-rate-change`, null, "GET");

        if (response && response.data && Array.isArray(response.data)) {
          const currencySpecificData = response.data.filter(
            (item) => item.currencyCode === currencyCode
          );

          if (currencySpecificData.length === 0) {
            setHistoryData([]);
            showSnackBar(
              `No history data found for currency: ${currencyCode}.`,
              "info"
            );
          } else {
            const processedData = currencySpecificData.map((item) => {
              const dateObject = new Date(item.rateChangeDate);
              let formattedDate = "N/A";
              if (!isNaN(dateObject.getTime())) {
                const day = String(dateObject.getDate()).padStart(2, "0");
                const month = String(dateObject.getMonth() + 1).padStart(
                  2,
                  "0"
                );
                const year = dateObject.getFullYear();
                formattedDate = `${day}-${month}-${year}`;
              }
              return {
                ...item,
                rateChangeDate: formattedDate,
                dateObject: dateObject,
              };
            });

            setHistoryData(processedData);

            const years = [
              ...new Set(
                processedData.map((item) => getFinancialYear(item.dateObject))
              ),
            ];
            years.sort().reverse();
            setFinancialYearOptions(years);
            if (years.length > 0) {
              setSelectedYear(years[0]);
            }
          }
        } else {
          setHistoryData([]);
          showSnackBar(
            "Received invalid history data format from server.",
            "warning"
          );
        }
      } catch (error) {
        console.error("Error fetching currency history:", error);
        showSnackBar("Failed to fetch currency rate history.", "error");
        setHistoryData([]);
      } finally {
        setIsHistoryLoading(false);
      }
    },
    [callApi, showSnackBar]
  );

  useEffect(() => {
    if (!selectedYear || historyData.length === 0) {
      setMonthOptions([{ value: "all", label: "All Months" }]);
      return;
    }
    const availableMonths = new Set();
    historyData.forEach((item) => {
      if (getFinancialYear(item.dateObject) === selectedYear) {
        availableMonths.add(item.dateObject.getMonth());
      }
    });
    const sortedMonths = Array.from(availableMonths).sort((a, b) => a - b);
    const options = [
      { value: "all", label: "All Months" },
      ...sortedMonths.map((monthIndex) => ({
        value: monthIndex,
        label: new Date(0, monthIndex).toLocaleString("default", {
          month: "long",
        }),
      })),
    ];
    setMonthOptions(options);
    setSelectedMonth("all");
  }, [selectedYear, historyData]);

  const filteredHistoryData = useMemo(() => {
    if (!selectedYear) return [];
    return historyData.filter((item) => {
      const itemFinancialYear = getFinancialYear(item.dateObject);
      const yearMatch = itemFinancialYear === selectedYear;
      const monthMatch =
        selectedMonth === "all" || item.dateObject.getMonth() === selectedMonth;
      return yearMatch && monthMatch;
    });
  }, [historyData, selectedYear, selectedMonth]);

  const handleCloseDialog = () => {
    setIsDialogOpen(false);
    setEditingRow(null);
    setFormData(initialFormState);
    setErrors({});
  };

  const handleEditClick = useCallback(
    (id) => () => {
      const rowToEdit = rows.find((row) => row.id === id);
      if (rowToEdit) {
        setEditingRow(rowToEdit);
        setPreviousRate(Number(rowToEdit.currencyRate));
        setFormData({
          currencyCode: rowToEdit.currencyCode,
          currencyRate: rowToEdit.currencyRate,
        });
        setErrors({});
        setIsDialogOpen(true);
      }
    },
    [rows]
  );

  const isSameRate = Number(formData.currencyRate) === PreviousRate;

  const handleHistoryClick = useCallback(
    (id) => () => {
      setCurrentCurrency(id);
      setIsHistoryDialogOpen(true);
      fetchHistoryData(id);
    },
    [fetchHistoryData]
  );

  const handleCloseHistoryDialog = () => {
    setIsHistoryDialogOpen(false);
    setHistoryData([]);
    setpage(0);
    setCurrentCurrency(null);
    setFinancialYearOptions([]);
    setSelectedYear("");
    setSelectedMonth("all");
  };

  const validate = () => {
    let tempErrors = {};

    const rate = formData.currencyRate;

    if (!String(rate).trim()) {
      tempErrors.currencyRate = "Currency Rate is required.";
    } else if (parseFloat(rate) === 0) {
      tempErrors.currencyRate = "Currency Rate cannot be taken as zero.";
      showSnackBar("Value cannot be zero", "error");
    }

    setErrors(tempErrors);
    return Object.keys(tempErrors).length === 0;
  };

  const handleInputChange = (e) => {
    const { name, value } = e.target;
    if (name === "currencyRate") {
      if (value !== "" && !/^\d{0,6}(\.\d{0,6})?$/.test(value)) {
        return;
      }
      setFormData({ ...formData, [name]: value });
      if (errors[name]) {
        setErrors({ ...errors, [name]: "" });
      }
    }
  };

  const handleSubmit = async () => {
    if (!validate()) return;

    const apiPayload = {
      requestType: "CURRENCY_RATE_CHANGE",
      changeType: "UPDATE",
      payload: {
        currencyCode: editingRow.currencyCode,
        currencyRate: parseFloat(formData.currencyRate) || 0,
      },
    };

    try {
      const data = await callApi("/CR/create-request", apiPayload, "POST");
      if (data) {
        showSnackBar(
          "Request to update currency rate submitted successfully!",
          "success"
        );
      }
      handleCloseDialog();
      fetchData(); // Refresh data on success
    } catch (err) {
      console.error(err);
      showSnackBar("Failed to fetch data", "error");
    }
  };

  const columns = useMemo(() => {
    // const currentDay = new Date().getDate();

    // setting the date as 1
    const currentDay = 1;

    const isEditDisabled = currentDay !== 1;

    return [
      {
        field: "currencyCode",
        headerName: "Currency Code",
        flex: 1,
        align: "left",
        headerAlign: "left",
      },
      {
        field: "currencyRate",
        headerName: "Currency Rate",
        flex: 1,
        sortable: true,
        align: "right",
        headerAlign: "right",
        renderCell: (params) =>
          currencyFormatterWith4Decimals("INR").format(params.value),
      },
      {
        field: "updatedAt",
        headerName: "Last Updated",
        flex: 1,
        align: "center",
        headerAlign: "center",
      },
      {
        field: "actions",
        type: "actions",
        headerName: "Actions",
        flex: 0.5,
        align: "center",
        headerAlign: "center",
        getActions: ({ id }) => [
          <GridActionsCellItem
            icon={<EditIcon />}
            label="Edit Rate"
            onClick={handleEditClick(id)}
            color="primary"
            disabled={isEditDisabled}
          />,
          <GridActionsCellItem
            icon={<HistoryIcon />}
            label="View History"
            onClick={handleHistoryClick(id)}
            color="inherit"
          />,
        ],
      },
    ];
  }, [handleEditClick, handleHistoryClick]);

  const historyColumns = [
    {
      field: "rateChangeDate",
      headerName: "Date of Change",
      flex: 1.5,
      align: "center",
      headerAlign: "center",
    },
    {
      field: "currencyRate",
      headerName: "Changed Rate",
      type: "number",
      flex: 1,
      align: "center",
      headerAlign: "center",
    },
  ];

  return (
    <StyledContainerBox>
      <StyledDataGridContainer
        rows={rows}
        columns={columns}
        disableRowSelectionOnClick
        pageSizeOptions={[7, 10, 25, 50]}
        initialState={{
          pagination: { paginationModel: { pageSize: 7 } },
          sorting: { sortModel: [{ field: "updatedAt", sort: "desc" }] },
        }}
        loading={isLoading}
      />

      <StyledDialog
        open={isDialogOpen}
        onClose={handleCloseDialog}
        fullwidth={"true"}
        maxWidth="sm"
      >
        <StyledDialogTitle>
          <StyledTypography>Edit Currency Rate</StyledTypography>
          <StyledCloseButton onClick={handleCloseDialog}>
            <CloseIcon />
          </StyledCloseButton>
        </StyledDialogTitle>
        <Divider />
        <StyledDialogContent>

            autoFocus
            name="currencyCode"
            label="Currency Code"
            type="text"
            fullwidth={"true"}
            variant="outlined"
            value={formData.currencyCode}
            disabled
          />

            name="currencyRate"
            label="New Currency Rate"
            type="text"
            fullwidth={"true"}
            variant="outlined"
            value={formData.currencyRate}
            onChange={handleInputChange}
            error={!!errors.currencyRate}
            helperText={
              errors.currencyRate ||
              "Numeric value. Max 6 digits before and 6 after the decimal."
            }
          />
        </StyledDialogContent>
        <StyledDialogActions>
          <StyledButton onClick={handleCloseDialog}>Cancel</StyledButton>
          <StyledButton
            onClick={handleSubmit}
            disabled={isSameRate}
            variant="contained"

            Submit Request
          </StyledButton>

                      </StyledFormControl>

                          value={selectedMonth}
                          label="Month"
                          onChange={(e) => setSelectedMonth(e.target.value)}
                        >
                          {monthOptions.map((option) => (
                            <StyledMenuItem
                              key={option.value}
                              value={option.value}
                            >
                              {option.label}
                            </StyledMenuItem>
                          ))}
                        </StyledSelect>
                      </StyledFormControl>
                    </StyledGridItem>
                  </StyledGridContainer>

                  {/* TABLE START */}
                  <StyledTableContainer>
                    <StyledTable>
                      <StyledTableHead>
                        <StyledTableRow>
                          {historyColumns.map((col) => (
                            <StyledTableHeaderCell key={col.field}>
                              {col.headerName}
                            </StyledTableHeaderCell>
                          ))}
                        </StyledTableRow>
                      </StyledTableHead>

                      <StyledTableBody>
                        {filteredHistoryData.length === 0 ? (
                          <StyledTableRow>
                            <StyledEmptyCell colSpan={historyColumns.length}>

                              <StyledTableRow key={row.id}>
                                {historyColumns.map((col) => (
                                  <StyledTableCell key={col.field}>
                                    {row[col.field]}
                                  </StyledTableCell>
                                ))}
                              </StyledTableRow>
                            ))
                        )}
                      </StyledTableBody>
                    </StyledTable>
                  </StyledTableContainer>
                  {/* TABLE END */}
                </>
              ) : (
                <StyledTypographyBox>
                  No data found for this currency
                </StyledTypographyBox>
              )}
              {filteredHistoryData.length > 0 && (
                <StyledTablePagination
                  component="div"
                  count={filteredHistoryData.length}
                  page={page}
                  onPageChange={handleChangepage}
                  rowsPerPage={5}
                  rowsPerPageOptions={[]}
                  labelDisplayedRows={({ page }) => {
                    return `PAGE: ${page + 1}`;
                  }}
                />
              )}
            </StyledDialogContent>
          </StyledDialogTitleBox>
        </StyledBox>
      </StyledDialog>
    </StyledContainerBox>
  );
}
