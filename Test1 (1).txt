package com.tcs.fincore.CommonRequestService.model;

import com.tcs.fincore.CommonRequestService.model.enums.ChangeType;
import com.tcs.fincore.CommonRequestService.model.enums.RequestStatus;
import com.tcs.fincore.CommonRequestService.model.enums.RequestType;
import jakarta.persistence.*;
import lombok.Data;
import java.time.LocalDateTime;

@Entity
@Table(name = "TMMOC")
@Data
public class TempMoc {

    // Note: JPA entities work best with a single primary key column.
    // We will add one even if it's not in the original table definition.
    @Id
    @GeneratedValue(strategy = GenerationType.IDENTITY)
    @Column(name = "REQUEST_ID")
    private Long id;

    // --- Request Tracking Fields ---
    @Column(name = "CREATOR_ID", length = 50)
    private String creatorId;

    @Column(name = "EXECUTOR_ID", length = 50)
    private String executorId;

    @Enumerated(EnumType.STRING)
    @Column(name = "REQ_TYPE", length = 30)
    private RequestType reqType;

    @Enumerated(EnumType.STRING)
    @Column(name = "CHANGE_TYPE", length = 10)
    private ChangeType changeType;
    
    @Enumerated(EnumType.STRING)
    @Column(name = "REQ_STATUS", length = 10)
    private RequestStatus reqStatus;

    @Column(name = "CREATION_DATE")
    private LocalDateTime creationDate = LocalDateTime.now();
    
    @Column(name = "EXECUTION_DATE")
    private LocalDateTime executionDate;
    
    @Column(name = "EXECUTOR_REMARKS", length = 200)
    private String executorRemarks;

    // --- MOC Data Fields ---
    @Lob // Using @Lob for the JSON payload string
    @Column(name = "PAYLOAD")
    private String payload;

    @Column(name = "TMMOC_PDATE")
    private java.sql.Date tmmocPdate;

    @Column(name = "TMMOC_BRANCH", length = 50)
    private String tmmocBranch;
    
    @Column(name = "CURRENCY", length = 3)
    private String currency;
    
    @Column(name = "TMMOC_CGL", length = 50)
    private String tmmocCgl;
    
    @Column(name = "TMMOC_AMT")
    private java.math.BigDecimal tmmocAmt;

    @Column(name = "TXN_TYPE", length = 20)
    private String txnType;

    @Column(name = "PRODUCT_TYPE", length = 20)
    private String productType;
    
    @Column(name = "REMARKS", length = 40)
    private String remarks;
    
    @Column(name = "AR_FLAG", length = 1)
    private String arFlag;

    // --- THIS IS THE UPDATED PART ---
    @Column(name = "TARGET_ID", length = 150)
    private String targetId;
}
