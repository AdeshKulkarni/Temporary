class Solution {
    public int Total(int[] nums) {
        int sum = 0;
        int n = nums.length;
        for (int i = 0; i < n; i++) {

            sum += nums[i];
        }

        return sum;
    }

    public boolean Helper(int[] nums, int k, int index, int sum, int target, int count) {

        if (sum > target || index == nums.length) {
            return false;
        }
        D.p(sum, target, count);

        if (sum == target) {
            count++;
            // sum = 0;
        }

        if (count == k) {
            return true;
        }

        sum += nums[index];

        boolean ans1 = Helper(nums, k, index + 1, sum, target, count);

        sum -= nums[index];

        boolean ans2 = Helper(nums, k, index + 1, sum, target, count);

        return ans1 || ans2;

    }

    public boolean canPartitionKSubsets(int[] nums, int k) {

        int total = Total(nums);

        if (k == 1) {
            return true;
        }

        if (total % k != 0) {
            return false;
        }

        int target = total / k;

        return Helper(nums, k, 0, 0, target, 0);

    }
}
