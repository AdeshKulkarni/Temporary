-- ==============================================================================
-- 5. TRIGGERS
-- ==============================================================================


CREATE OR REPLACE EDITIONABLE TRIGGER "REQUEST_VID_TRIGGER" 
BEFORE INSERT ON REQUEST_VOUCHER
FOR EACH ROW
DECLARE
    v_seq_number NUMBER;
    v_prefix VARCHAR2(20);
BEGIN
    v_prefix := 'VID-' || TO_CHAR(SYSDATE, 'YYMM') || '-';
    
    SELECT REQUEST_VOUCHER_SEQ.NEXTVAL
    INTO v_seq_number
    FROM DUAL;
    
    :NEW.VID := v_prefix || LPAD(v_seq_number, 5, '0');
END;
/
ALTER TRIGGER "REQUEST_VID_TRIGGER" ENABLE;
