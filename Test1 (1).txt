class Solution {
    public int minOperations(int[] nums, int x) {
        int n = nums.length;
        Map<Integer, Integer> map = new HashMap<>(); // Stores prefix sum -> index count
        int low = 0;
        int currentSum = 0;
        int finalAns = Integer.MAX_VALUE; // Initialize to max to find minimum operations
        
        // Base case: 0 sum takes 0 elements
        map.put(0, 0);

        // Phase 1: Record all prefix sums from the left
        int leftSum = 0;
        for (int i = 0; i < n; i++) {
            leftSum += nums[i];
            if (leftSum > x) break; // Positive numbers optimization
            map.put(leftSum, i + 1);
            if (leftSum == x) {
                finalAns = Math.min(finalAns, i + 1);
            }
        }

        // Phase 2: Traverse from the right (suffix sums) and check the map for the complement
        int rightSum = 0;
        for (int i = n - 1; i >= 0; i--) {
            rightSum += nums[i];
            if (rightSum > x) break;
            
            int neededFromLeft = x - rightSum;
            
            // If the map has our complement from the left side
            if (map.containsKey(neededFromLeft)) {
                int leftElements = map.get(neededFromLeft);
                int rightElements = n - i;
                
                // CRITICAL CHECK: Ensure left and right pointers/indices don't overlap!
                if (leftElements + rightElements <= n) {
                    finalAns = Math.min(finalAns, leftElements + rightElements);
                }
            }
        }

        return finalAns == Integer.MAX_VALUE ? -1 : finalAns;
    }
}
