// 1. You calculated your limits beforehand
int limitA = totalA - k;
int limitB = totalB - k;
int limitC = totalC - k;

int low = 0;
int maxMiddle = 0;

// Track what is currently INSIDE your middle window
int[] windowCounts = new int[3]; 

for (int high = 0; high < s.length(); high++) {
    // Eat the right character (expand window)
    windowCounts[s.charAt(high) - 'a']++;

    // THE VIOLATION TRIGGER (This is your while loop!)
    // If the middle window exceeds ANY of the allowed limits, it's invalid.
    while (windowCounts[0] > limitA || windowCounts[1] > limitB || windowCounts[2] > limitC) {
        // Spit out the left character (shrink window)
        windowCounts[s.charAt(low) - 'a']--;
        low++;
    }

    // Capture the valid middle window size
    maxMiddle = Math.max(maxMiddle, high - low + 1);
}

return s.length() - maxMiddle;
