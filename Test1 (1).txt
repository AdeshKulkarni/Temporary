  '&:nth-of-type(odd)': {},

// Columns
const columns = [
  { key: 'standard', label: 'Standard', editable: true },
  { key: 'subStandard', label: 'Sub-standard', editable: true },
  { key: 'doubtful', label: 'Doubtful', editable: true },
  { key: 'loss', label: 'Loss', editable: true },
  { key: 'adjustment', label: 'Adjustment', editable: true, conditional: true },
  { key: 'total', label: 'Total', editable: false },
];

const sectionMap = {
  facTotal: ['fac1', 'fac2', 'fac3'],
  secTotal: ['sec1', 'sec2', 'sec3'],
  inTotal: ['in1', 'in2', 'in3', 'in4'],
  // ## CHANGE 1: Corrected the list of rows to aggregate for the 'outTotal'. Removed 'out5' which is a header.
  outTotal: ['out1', 'out2', 'out3', 'out4'],
  advA3: ['inTotal', 'outTotal'],
};

const getFieldName = (id, key) => {
  let base = id.startsWith('fac') ? 'facility' : id.startsWith('sec') ? 'security' : 'sector';

  const suffix =
    {
      standard: 'Standard',
      subStandard: 'SubStandard',
      doubtful: 'Doubtful',
      loss: 'Loss',
      adjustment: 'Adj',
    }[key] || 'Total';

  let idx;

  if (id === 'inTotal') idx = 'a_Total';
  else if (id === 'outTotal') idx = 'b_Total';
  else if (id === 'advA3') idx = 'ab_Total';
  else if (id.startsWith('in')) idx = `a${id.replace('in', '')}`;
  else if (id === 'out1') idx = `b${id.replace('out', '')}`;
  else if (id === 'out2' || id === 'out3' || id === 'out4') idx = `${id.replace('out', '')}`; // Corrected logic for 'out4'
  else if (id.endsWith('Total')) idx = 'Total';
  else idx = id.replace(/\D/g, '');

  return `${base}_${suffix}_${idx}`;
};

const buildSavePayload = (
  values,
  userId,
  circleCode,
  quarterEndDate,
  role,
  reportName,
  save,
  status,
  reportMasterId,
  reportId,
  areMocPending
) => {
  const payload = {
    userId,
    circleCode,
    quarterEndDate,
    role,
    reportName,
    save,
    status,
    reportMasterId,
    reportId,
    areMocPending,
  };
  Object.entries(values).forEach(([id, fields]) => {
    Object.entries(fields).forEach(([key, val]) => {
      if (['standard', 'subStandard', 'doubtful', 'loss', 'adjustment'].includes(key)) {
        if (!(id.includes('header') || id === 'out5')) {
            payload[getFieldName(id, key)] = val || '0';
        }
      }
    });
  });
  return payload;
};

export default function Schedule9Table({ showAdjustment = false }) {
  const { state } = useLocation();
  const showSnackbar = useCustomSnackbar();
  const user = JSON.parse(localStorage.getItem('user'));
  const [reportObject, setReportObject] = useState(state?.report || null);
  const navigate = useNavigate();

  const { callApi } = useApi();

  const [values, setValues] = useState(
    rowDefinitions.reduce((acc, { id }) => {
      acc[id] = {
        standard: '',
        subStandard: '',
        doubtful: '',
        loss: '',
        adjustment: '',
        ysa: '',
      };
      return acc;
    }, {})
  );

  // Fetch data
  useEffect(() => {
    const fetchReport = async () => {
      try {
        const requestPayload = {
          circleCode: user.circleCode,
          quarterEndDate: user.quarterEndDate,
          role: user.capacity,
          reportName: reportObject.name,
        };

        const data = await callApi('/Maker/getSC09ReportData', requestPayload, 'POST');

        setValues((prev) => {
          const nv = { ...prev };

          const backendFieldMap = {
            standard: 'Standard',
            subStandard: 'SubStandard',
            doubtful: 'Doubtful',
            loss: 'Loss',
            adjustment: 'Adj',
          };
          rowDefinitions.forEach(({ id, type }) => {
            if (type !== 'entry') return;

            const fields = ['standard', 'subStandard', 'doubtful', 'loss'];
            if (user.capacity === '61') fields.push('adjustment');

            const base = id.startsWith('fac') ? 'facility' : id.startsWith('sec') ? 'security' : 'sector';

            // Use getFieldName to correctly map backend field names to frontend state
            const standardFieldName = getFieldName(id, 'standard');
            const index = standardFieldName.split('_').pop();


            fields.forEach((field) => {
              const suffix = backendFieldMap[field];
              const key = `${base}_${suffix}_${index}`;
              if (data[key] !== undefined) {
                 nv[id][field] = data[key];
              }
            });
          });

          return nv;
        });
      } catch (error) {
        if (error.code !== 'ERR_CANCELED') {
          showSnackbar('Failed to fetch data', 'error');
        }
      }
    };

    const fetchValidation = async () => {
      try {
        const requestPayload2 = {
          circleCode: user.circleCode,
          quarterEndDate: user.quarterEndDate,
          role: user.capacity,
        };

        const data = await callApi('/Maker/getSC09ValiadationData', requestPayload2, 'POST');
        
        setValues((prev) => {
          const nv = { ...prev };

          if (user.capacity === '51' || user.capacity === '52') {
            const normalize = (s) => s.replace(/\s|[^a-zA-Z0-9]/g, '').toLowerCase();
            ['fac1', 'fac2', 'fac3'].forEach((id) => {
              const label = rowDefinitions.find((r) => r.id === id).label;
              const key = Object.keys(data).find((k) => normalize(k) === normalize(label));
              nv[id].ysa = data[key] || '0';
            });
            nv.facTotal.ysa = ['fac1', 'fac2', 'fac3']
              .reduce((sum, id) => sum + (parseFloat(nv[id].ysa) || 0), 0)
              .toFixed(2);
          } else if (user.capacity === '61' || user.capacity === '62') {
            // ## CHANGE 2: Correctly handle the '~' separated response for capacity '61'
            // The API returns "adjustment_value~total_value". We need to populate both.
            const getValues = (code) => {
                const val = data[code] || "0~0";
                const parts = val.split('~');
                return { adj: parts[0] || '0', ysa: parts[1] || '0' };
            }

            const codeMap = {
              fac1: '090101001051',
              fac2: '090101001052',
              fac3: '090101001053',
            };

            Object.keys(codeMap).forEach((id) => {
              const { adj, ysa } = getValues(codeMap[id]);
              nv[id].ysa = ysa;
              nv[id].adjustment = adj; // Populate the adjustment field from API
            });

            nv.facTotal.ysa = ['fac1', 'fac2', 'fac3']
              .reduce((sum, id) => sum + (parseFloat(nv[id].ysa) || 0), 0)
              .toFixed(2);
          }

          return nv;
        });
      } catch (error) {
        if (error.code !== 'ERR_CANCELED') {
          showSnackbar('Failed to load validation data', 'error');
        }
      }
    };

    fetchReport();
    fetchValidation();
  }, []); // Dependencies are empty to run only on mount

  const handleChange = (id, field, val) => {
    if (!/^-?\d*\.?\d{0,2}$/.test(val)) return;
    setValues((prev) => ({
      ...prev,
      [id]: { ...prev[id], [field]: val },
    }));
  };

  const getRowData = (id) => {
    // For entry rows (not section totals)
    if (!sectionMap[id]) {
      const rowValues = values[id];
      const sum = ['standard', 'subStandard', 'doubtful', 'loss'].reduce(
        (acc, field) => acc + (parseFloat(rowValues[field]) || 0),
        0
      );
      const adj = showAdjustment ? parseFloat(rowValues.adjustment) || 0 : 0;
      const total = sum + adj;
      const ysa = parseFloat(rowValues.ysa);
      const isMismatch =
        id.startsWith('fac') && !isNaN(ysa)
          ? ysa.toFixed(2) !== total.toFixed(2)
          : false;
      return { ...rowValues, total: total.toFixed(2), isMismatch };
    }
    // For section total rows
    const agg = sectionMap[id].reduce(
      (acc, childId) => {
        const childRowData = getRowData(childId);
        Object.keys(acc).forEach((field) => {
             acc[field] += parseFloat(childRowData[field]) || 0
        });
        return acc;
      },
      { standard: 0, subStandard: 0, doubtful: 0, loss: 0, adjustment: 0 }
    );
    const total = Object.values(agg).reduce((sum, val) => sum + val, 0);
    
    return {
      standard: agg.standard.toFixed(2),
      subStandard: agg.subStandard.toFixed(2),
      doubtful: agg.doubtful.toFixed(2),
      loss: agg.loss.toFixed(2),
      adjustment: showAdjustment ? agg.adjustment.toFixed(2) : '',
      total: total.toFixed(2),
      ysa: values[id].ysa,
      isMismatch: false, // Mismatch logic is mainly for entry rows
    };
  };

  const computedRowData = useMemo(() => {
    const result = {};
    rowDefinitions.forEach((row) => {
      result[row.id] = getRowData(row.id);
    });
    return result;
  }, [values, showAdjustment]);

  const totalMismatchFlags = useMemo(() => {
    const a = computedRowData['facTotal'];
    const b = computedRowData['secTotal'];
    const c = computedRowData['advA3'];

    const mismatches = {};
    const keysToCompare = ['standard', 'subStandard', 'doubtful', 'loss'];
    if (showAdjustment) keysToCompare.push('adjustment');

    keysToCompare.forEach((key) => {
        mismatches[key] = !(a[key] === b[key] && b[key] === c[key]);
    });
    
    return mismatches;
  }, [computedRowData, showAdjustment]);

  // Validation on submit
  const validateForSubmit = useMemo(() => {
    const errs = [];
    // Row-level validation for facility totals
    ['fac1', 'fac2', 'fac3'].forEach((id) => {
      const { total, ysa } = computedRowData[id];
      // ## CHANGE 3: More reliable float comparison
      if (parseFloat(ysa || 0).toFixed(2) !== total) {
        errs.push(`Total not matching for ${rowDefinitions.find((r) => r.id === id).label}`);
      }
    });

    // Cross-section validation (A1=A2=A3)
    const aTot = computedRowData['facTotal'],
      bTot = computedRowData['secTotal'],
      cTot = computedRowData['advA3'];
    [
      ['standard', 'Standard'],
      ['subStandard', 'Sub-standard'],
      ['doubtful', 'Doubtful'],
      ['loss', 'Loss'],
    ].forEach(([key, label]) => {
      if (!(aTot[key] === bTot[key] && bTot[key] === cTot[key])) {
        errs.push(`A-1, A-2, A-3 totals differ for ${label}`);
      }
    });
    if (showAdjustment) {
      if (!(aTot.adjustment === bTot.adjustment && bTot.adjustment === cTot.adjustment)) {
        errs.push('A-1, A-2, A-3 totals differ for Adjustment');
      }
    }
    return errs;
  }, [computedRowData, showAdjustment]);

  const onSave = async () => {
    try {
      const payload = buildSavePayload(
        values, user.userId, user.circleCode, user.quarterEndDate, user.capacity,
        reportObject.name, true, reportObject.status, reportObject.reportMasterId,
        reportObject.reportId, reportObject.areMocPending
      );

      const response = await callApi('/Maker/SubmitSC09Report', payload, 'POST');
      if (response && typeof response === 'string') {
        const [_flag, newReportId, newStatus] = response.split('~');
        setReportObject((prev) => ({ ...prev, reportId: newReportId, status: newStatus }));
      }
      if (response) showSnackbar('Saved Successfully!', 'success');
      else showSnackbar('Not Saved', 'error');
    } catch (err) {
      console.error(err);
      showSnackbar('Not Saved', 'error');
    }
  };
  const onSubmit = async () => {
    try {
      if (validateForSubmit.length) return;
      const payload = buildSavePayload(
        values, user.userId, user.circleCode, user.quarterEndDate, user.capacity,
        reportObject.name, false, reportObject.status, reportObject.reportMasterId,
        reportObject.reportId, reportObject.areMocPending
      );
      const response = await callApi('/Maker/SubmitSC09Report', payload, 'POST');
      if (response && typeof response === 'string') {
        const [_flag, newReportId, newStatus] = response.split('~');
        setReportObject((prev) => ({ ...prev, reportId: newReportId, status: newStatus }));
      }
      if (response) {
        showSnackbar('Report submitted successfully!', 'success');
        setTimeout(() => navigate('../'), 1000);
      } else showSnackbar('Failed to submit report!', 'error');
    } catch (err) {
      console.error(err);
      showSnackbar('Failed to submit report!', 'error');
    }
  };

  return (
    <Box>
      <TableRow>
        <TableCell colSpan={columns.filter((c) => !c.conditional || showAdjustment).length + 2}>
          <Stack direction={'row'} spacing={2} sx={{ my: 2 }}>
            <CustomButton label={'Save'} buttonType={'save'} onClickHandler={onSave} />
            <CustomButton
              label={'Submit'}
              buttonType={'submit'}
              onClickHandler={onSubmit}
              disabled={!!validateForSubmit.length}
            />
          </Stack>
        </TableCell>
      </TableRow>
      {validateForSubmit.length > 0 && (
        <Alert severity="error" sx={{ mb: 2 }}>
          <ul style={{ margin: 0, paddingLeft: '1.2em' }}>
            {validateForSubmit.map((e, i) => ( <li key={i}>{e}</li> ))}
          </ul>
        </Alert>
      )}
      <TableContainer component={Paper} sx={{ width: '100%' }}>
        <Table sx={{ tableLayout: 'fixed' }}>
          <TableHead>
            <TableRow>
              <StyledTableCell>
                Classification of Advances (...) (A1 = A2 = A3)
              </StyledTableCell>
              {columns.map((col) =>
                  (!col.conditional || showAdjustment) && (
                    <StyledTableCell key={col.key} align="center">
                      {col.label}
                    </StyledTableCell>
                  )
              )}
            </TableRow>
          </TableHead>
          <TableBody>
            {rowDefinitions.map((row) => {
              const data = computedRowData[row.id];
              if (row.type !== 'entry') {
                return (
                  <StyledTableRow key={row.id}>
                    <StyledTableCell
                      colSpan={columns.filter((c) => !c.conditional || showAdjustment).length + 1}
                      sx={{ fontWeight: 'bold' }}
                    >
                      {row.label}
                    </StyledTableCell>
                  </StyledTableRow>
                );
              }
              return (
                <StyledTableRow key={row.id}>
                  <StyledTableCell>{row.label}</StyledTableCell>
                  {columns.map((col) =>
                      (!col.conditional || showAdjustment) && (
                        <StyledTableCell key={col.key} align="right">
                          <FormInput
                            value={data[col.key] || ''}
                            onChange={(e) => handleChange(row.id, col.key, e.target.value)}
                            readOnly={
                              !col.editable || Boolean(sectionMap[row.id]) ||
                              (user.capacity === '61' && col.key !== 'adjustment') ||
                              (user.capacity === '61' && ['fac1', 'fac2', 'fac3'].includes(row.id) && col.key === 'adjustment')
                            }
                            inputType="wholeAmountDecimal"
                            sx={{
                              ...(data.isMismatch && col.key === 'total' && { border: '1px solid red', borderRadius: 1 }),
                              ...(['facTotal', 'secTotal', 'advA3'].includes(row.id) && col.key !== 'total' && totalMismatchFlags[col.key] && { border: '1px solid red', borderRadius: 1 }),
                            }}
                          />
                        </StyledTableCell>
                      )
                  )}
                </StyledTableRow>
              );
            })}
          </TableBody>
        </Table>
      </TableContainer>
    </Box>
  );
}
