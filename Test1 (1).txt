package com.tcs.fincore.MOCService.config;

import org.springframework.context.annotation.Bean;
import org.springframework.context.annotation.Configuration;
import org.springframework.security.config.annotation.web.builders.HttpSecurity;
import org.springframework.security.config.annotation.web.configuration.EnableWebSecurity;
import org.springframework.security.config.http.SessionCreationPolicy;
import org.springframework.security.web.SecurityFilterChain;

@Configuration
@EnableWebSecurity
public class SecurityConfig {

    @Bean
    public SecurityFilterChain securityFilterChain(HttpSecurity http) throws Exception {
        http
                // 1. Disable CSRF, not needed for stateless REST APIs
                .csrf(csrf -> csrf.disable())

                // 2. Define authorization rules
                .authorizeHttpRequests(authz -> authz
                        // IMPORTANT: For now, permit all requests to solve the 401 error.
                        // In a real application,we would secure these endpoints.
                        // Example: .requestMatchers("/api/v1/admin/**").hasRole("ADMIN")
                        .requestMatchers("/**").permitAll() // Allows access to all your API endpoints
//                        .requestMatchers("/actuator/**").permitAll() // Allows access to Spring Actuator
                        .anyRequest().authenticated() // Secure any other endpoints not listed
                )

                // 3. Set session management to stateless, essential for REST APIs
                .sessionManagement(session -> session.sessionCreationPolicy(SessionCreationPolicy.STATELESS));

        // In a real application with JWTs, you would add your JWT filter here
        // .addFilterBefore(jwtAuthFilter, UsernamePasswordAuthenticationFilter.class);

        return http.build();
    }
}

@RestController
@RequestMapping("moc") 
@RequiredArgsConstructor
@Slf4j
public class MocRequestController {

    private final MocRequestService mocRequestService;

    @GetMapping("/current-posting-date")
    public String getCurrentPostingDate() {
        return LocalDate.now().format(DateTimeFormatter.ISO_LOCAL_DATE);
    }
    

    @PostMapping("/create-batch")
    @ResponseStatus(HttpStatus.CREATED)
    public List<MocRequest> createBatchRequest(
            @Valid @RequestBody BatchRequestDto batchDto,
            @RequestHeader("Authorization") String token) throws JsonProcessingException {
        
        String userId = JwtUtil.getUserIdFromToken(token);
        log.info("Received MOC batch request creation call from user: {}", userId);
        return mocRequestService.createBatchRequest(batchDto, userId);
    }

    @PatchMapping("/update-request")
    public MocRequest updateRequestStatus(
            @RequestHeader("Authorization") String token,
            @RequestBody ProcessMocRequestDto processRequestDto) throws JsonProcessingException {
        String executorId = JwtUtil.getUserIdFromToken(token);
        return mocRequestService.updateRequestStatus(processRequestDto, executorId).get();
    }

    @GetMapping("/my-requests")
    public List<MocRequest> getMyRequests() {
        log.info("Received request for 'my-requests' (now returning ALL requests).");
        return mocRequestService.getMyRequests();
    }

    @GetMapping("/pending-requests")
    public List<MocRequest> getPendingRequests(@RequestHeader("Authorization") String token) {
        return mocRequestService.getPendingRequests();
    }
    
    
    @PostMapping("/process-bulk")
    public List<MocRequest> processBulkRequests(
            @RequestHeader("Authorization") String token,
            @Valid @RequestBody BulkProcessMocRequestDto processRequestDto) throws JsonProcessingException {
        
        String executorId = JwtUtil.getUserIdFromToken(token);
        log.info("Received bulk MOC process call from user: {}", executorId);
        return mocRequestService.processBulkRequests(processRequestDto, executorId);
    }

    @GetMapping("/by-batch/{batchId}")
    public ResponseEntity<List<MocRequest>> getRequestsByBatchId(@PathVariable String batchId) {
        log.info("Received request for all MOC data by Batch ID: {}", batchId);
        List<MocRequest> requests = mocRequestService.getRequestsByBatchId(batchId);
        return ResponseEntity.ok(requests);
    }

    @DeleteMapping("/my-request/{requestId}")
    public ResponseEntity<Void> cancelMyRequest(
            @RequestHeader("Authorization") String token,
            @PathVariable Long requestId) {
        
        String userId = JwtUtil.getUserIdFromToken(token);
        log.info("Received request from user {} to cancel MOC request {}", userId, requestId);
        mocRequestService.cancelMyRequest(requestId, userId);
        return ResponseEntity.noContent().build(); 
    }

    @DeleteMapping("/my-requests/by-journal/{journalIdPrefix}")
    public ResponseEntity<Void> cancelMyRequestsByJournal(
            @RequestHeader("Authorization") String token,
            @PathVariable String journalIdPrefix) {
        
        String userId = JwtUtil.getUserIdFromToken(token);
        log.info("Received request from user {} to bulk-cancel MOC requests with journal prefix {}", userId, journalIdPrefix);
        mocRequestService.cancelMyRequestsByJournalPrefix(journalIdPrefix, userId);
        return ResponseEntity.noContent().build(); 
    }
    @DeleteMapping("/my-requests/by-batch/{batchId}")
    public ResponseEntity<Void> cancelMyRequestsByBatch(
            @RequestHeader("Authorization") String token,
            @PathVariable String batchId) throws JsonProcessingException {
        
        String userId = JwtUtil.getUserIdFromToken(token);
        log.info("Received request from user {} to bulk-cancel MOC requests with batch ID {}", userId, batchId);
        mocRequestService.cancelMyRequestsByBatchId(batchId, userId);
        return ResponseEntity.noContent().build(); 
    }

    /**
     * Cancels all PENDING requests created by the user for a list of Journal Prefixes.
     */
    @DeleteMapping("/my-requests/by-journal")
    public ResponseEntity<Void> cancelMyRequestsByJournalPrefixes(
            @RequestHeader("Authorization") String token,
            @RequestBody List<String> journalIdPrefixes) throws JsonProcessingException {
        
        String userId = JwtUtil.getUserIdFromToken(token);
        log.info("Received request from user {} to bulk-cancel MOC requests with journal prefixes: {}", userId, journalIdPrefixes.size());
        mocRequestService.cancelMyRequestsByJournalPrefixes(journalIdPrefixes, userId);
        return ResponseEntity.noContent().build(); 
    }
    @GetMapping("/status")
    public ResponseEntity<List<MocRequestStatusDto>> getMocStatusList() {
        log.info("Received request for MOC status list");
        return ResponseEntity.ok(mocRequestService.getMocRequestStatusList());
    }
}

package com.tcs.fincore.MOCService.Controllers;

import com.tcs.fincore.MOCService.Models.enums.RequestStatus;

import jakarta.persistence.AttributeConverter;
import jakarta.persistence.Converter;
import java.util.stream.Stream;

@Converter(autoApply = true) // autoApply makes it work for all RequestStatus fields
public class RequestStatusConverter implements AttributeConverter<RequestStatus, String> {

    @Override
    public String convertToDatabaseColumn(RequestStatus status) {
        if (status == null) {
            return null;
        }
        return status.getCode(); // Converts PENDING to "P"
    }

    @Override
    public RequestStatus convertToEntityAttribute(String code) {
        if (code == null) {
            return null;
        }
        // Finds the Enum constant that matches the code from the DB
        return Stream.of(RequestStatus.values())
                .filter(c -> c.getCode().equals(code))
                .findFirst()
                .orElseThrow(IllegalArgumentException::new);
    }
}

package com.tcs.fincore.MOCService.Models.enums;

import lombok.Getter;

import java.util.Arrays;
import java.util.Map;
import java.util.stream.Collectors;

@Getter
public enum AcClassification {
	  ASSET('A', "Asset"),
	    LIABILITY('L', "Liability"),
	    INCOME('I', "Income"),
	    EXPENSE('E', "Expense"),
	    MEMO('M', "Memo A/c");

	    private final char code;
	    private final String description;

	    AcClassification(char code, String description) {
	        this.code = code;
	        this.description = description;
	    }

	  
	    public static Map<String, String> getClassificationMap() {
	        return Arrays.stream(AcClassification.values())
	                .collect(Collectors.toMap(
	                        ac -> String.valueOf(ac.getCode()), 
	                        AcClassification::getDescription
	                ));
	    }

    @Override
    @Transactional
    public List<MocRequest> createBatchRequest(BatchRequestDto batchDto, String creatorId) throws JsonProcessingException {
        
        String batchId = sequenceService.getNextBatchId();
        log.info("Creating new MOC batch with Batch ID: {}", batchId);

       
        LocalDate postingDate = LocalDate.now();
        
        List<MocRequest> requestsToSave = new ArrayList<>();
        String commonRemarks = batchDto.getCommonBatchRemarks();

        Map<String, List<BatchRequestDto.MocRequestRow>> groupedRows = batchDto.getRows().stream()
                .collect(Collectors.groupingBy(row -> row.getBranch() + ":" + row.getCurrency()));
        
        log.info("Batch {} contains {} distinct Branch-Currency groups.", batchId, groupedRows.size());

        for (Map.Entry<String, List<BatchRequestDto.MocRequestRow>> entry : groupedRows.entrySet()) {
            
            String journalIdPrefix = sequenceService.getNextJournalIdPrefix();
            int journalSequence = 1;
            List<BatchRequestDto.MocRequestRow> rowsInGroup = entry.getValue();
            
            for (BatchRequestDto.MocRequestRow rowDto : rowsInGroup) {
                
                String journalId = String.format("%s-%03d", journalIdPrefix, journalSequence);
                
                CreateMocRequestDto payloadDto = new CreateMocRequestDto();
                
             
                payloadDto.setPDate(postingDate); 
              
                payloadDto.setChangeType(rowDto.getChangeType());
                payloadDto.setMasterMocId(rowDto.getMasterMocId());
                payloadDto.setBranch(rowDto.getBranch());
                payloadDto.setCurrency(rowDto.getCurrency());
                payloadDto.setCgl(rowDto.getCgl());
                payloadDto.setAmount(rowDto.getAmount()); // This is now pos/neg
                payloadDto.setProductType(rowDto.getProductType());
                payloadDto.setRemarks(rowDto.getRemarks());
                payloadDto.setArFlag(rowDto.getArFlag());
                payloadDto.setAcClassification(rowDto.getAcClassification());
                payloadDto.setBatchId(batchId); 
                payloadDto.setJournalId(journalId); 
                payloadDto.setCommonBatchRemarks(commonRemarks);
                payloadDto.setTransactionCount(journalSequence); 
                
                MocRequest request = new MocRequest();
                request.setCreatorId(creatorId);
                request.setChangeType(rowDto.getChangeType());
                request.setRequestStatus(RequestStatus.PENDING);
                request.setBatchId(batchId);
                request.setJournalId(journalId);
                request.setCommonBatchRemarks(commonRemarks);
                request.setPayload(objectMapper.writeValueAsString(payloadDto));

                requestsToSave.add(request);

                journalSequence++; 
            }
        }
        return mocRequestRepository.saveAll(requestsToSave);
    }
  
   

    @Override
    @Transactional
    public Optional<MocRequest> updateRequestStatus(ProcessMocRequestDto dto, String executorId) throws JsonProcessingException {
        MocRequest request = mocRequestRepository.findById(dto.getRequestId())
                .orElseThrow(() -> new ResourceNotFoundException("MOC Request with ID " + dto.getRequestId() + " not found."));

        if (executorId.equals(request.getCreatorId())) {
            throw new IllegalStateException("User cannot approve or reject their own MOC request.");
        }

        if (request.getRequestStatus() != RequestStatus.PENDING) {
            throw new IllegalStateException("MOC Request has already been processed.");
        }

        request.setExecutorId(executorId);
        request.setExecutorRemarks(dto.getRemarks());
        request.setExecutionDate(LocalDateTime.now());
        request.setRequestStatus(dto.getStatus());

        if (RequestStatus.ACCEPTED.equals(dto.getStatus())) {
            log.info("MOC Request {} ACCEPTED. Processing for Batch ID: {}", dto.getRequestId(), request.getBatchId());
            processApproval(request); 
        } else {
            log.info("MOC Request {} REJECTED.", dto.getRequestId());
        }

        return Optional.of(mocRequestRepository.save(request));
    }
    
    private void processApproval(MocRequest request) throws JsonProcessingException {
        CreateMocRequestDto payloadDto = objectMapper.readValue(request.getPayload(), CreateMocRequestDto.class);

        switch (request.getChangeType()) {
            case ADD:
                GlTransaction newGlTxn = new GlTransaction();
                
                newGlTxn.setBatchId(payloadDto.getBatchId());
                newGlTxn.setJournalId(payloadDto.getJournalId());
                
                if (payloadDto.getPDate() != null) {
                    newGlTxn.setPostDate(payloadDto.getPDate().atStartOfDay());
                }
                
                newGlTxn.setBranchCode(payloadDto.getBranch());
                newGlTxn.setCurrency(payloadDto.getCurrency());
                newGlTxn.setCgl(payloadDto.getCgl());
                newGlTxn.setNarration(payloadDto.getRemarks());
                
                // Store the negative value in the Credit column as requested.
                BigDecimal amount = payloadDto.getAmount();
                if (amount.signum() < 0) { // Negative amount is Credit
                    newGlTxn.setCreditAmount(amount); // <-- Stores the negative value
                    newGlTxn.setDebitAmount(BigDecimal.ZERO);
                } else { // Positive or zero amount is Debit
                    newGlTxn.setDebitAmount(amount);
                    newGlTxn.setCreditAmount(BigDecimal.ZERO);
                }
                
                newGlTxn.setTransactionCount(payloadDto.getTransactionCount());
                newGlTxn.setSourceFlag("J"); // Hardcoded as requested
                
                glTransactionRepository.save(newGlTxn);
                log.info("Successfully ADDED new GL_TRANSACTION record (Batch ID: {}, Journal ID: {}).", payloadDto.getBatchId(), payloadDto.getJournalId());
                break;
            case UPDATE:
                log.warn("UPDATE ChangeType not yet implemented for GL_TRANSACTIONS. Request {} skipped.", request.getId());
                break;
            case DELETE:
                log.warn("DELETE ChangeType not yet implemented for GL_TRANSACTIONS. Request {} skipped.", request.getId());
                break;
        }
    }

    @Override
    @Transactional
    public List<MocRequest> processBulkRequests(BulkProcessMocRequestDto dto, String executorId) throws JsonProcessingException {

        boolean hasBatchId = StringUtils.hasText(dto.getBatchId());
        boolean hasJournalPrefixes = dto.getJournalIdPrefixes() != null && !dto.getJournalIdPrefixes().isEmpty();

        if (!hasBatchId && !hasJournalPrefixes) {
            throw new IllegalArgumentException("Either batchId or a list of journalIdPrefixes must be provided.");
        }
        if (hasBatchId && hasJournalPrefixes) {
            throw new IllegalArgumentException("Provide either batchId or journalIdPrefixes, not both.");
        }

        List<MocRequest> requestsToProcess;
        if (hasBatchId) {
            log.info("Processing bulk request by Batch ID: {} for user: {}", dto.getBatchId(), executorId);
            requestsToProcess = mocRequestRepository.findByBatchIdAndRequestStatus(dto.getBatchId(), RequestStatus.PENDING);
        } else {
            log.info("Processing bulk request by Journal Prefixes: {} for user: {}", dto.getJournalIdPrefixes().size(), executorId);
            requestsToProcess = new ArrayList<>();
            for (String prefix : dto.getJournalIdPrefixes()) {
                requestsToProcess.addAll(
                    mocRequestRepository.findByJournalIdStartingWithAndRequestStatus(prefix, RequestStatus.PENDING)
                );
            }
        }

        if (requestsToProcess.isEmpty()) {
            log.warn("No pending requests found for the given criteria.");
            return Collections.emptyList();
        }

        List<MocRequest> processedRequests = new ArrayList<>();
        
        for (MocRequest request : requestsToProcess) {
            
            if (executorId.equals(request.getCreatorId())) {
                log.warn("Skipping request {}: User {} cannot approve/reject their own request.", request.getId(), executorId);
                continue; 
            }
          
            if (request.getRequestStatus() != RequestStatus.PENDING) {
                log.warn("Skipping request {}: It is not in PENDING status.", request.getId());
                continue;
            }

            request.setExecutorId(executorId);
            request.setExecutorRemarks(dto.getRemarks());
            request.setExecutionDate(LocalDateTime.now());
            request.setRequestStatus(dto.getStatus());
        
            if (RequestStatus.ACCEPTED.equals(dto.getStatus())) {
                log.info("MOC Request {} ACCEPTED. Processing...", request.getId());
                processApproval(request); 
            } else {
                log.info("MOC Request {} REJECTED.", request.getId());
            }
            processedRequests.add(request);
        }
     
        return mocRequestRepository.saveAll(processedRequests);
    }

    @Override
    @Transactional
    public void cancelMyRequest(Long requestId, String userId) {
        MocRequest request = mocRequestRepository.findById(requestId)
                .orElseThrow(() -> new ResourceNotFoundException("MOC Request with ID " + requestId + " not found."));

        if (!userId.equals(request.getCreatorId())) {
            log.warn("User {} attempted to cancel request {} owned by {}", userId, requestId, request.getCreatorId());
            throw new IllegalStateException("You are not authorized to cancel this request.");
        }

        if (request.getRequestStatus() != RequestStatus.PENDING) {
            log.warn("User {} attempted to cancel request {} which is already in status {}", userId, requestId, request.getRequestStatus());
            throw new IllegalStateException("Cannot cancel a request that has already been processed.");
        }

        mocRequestRepository.delete(request);
        log.info("User {} successfully canceled MOC request {}", userId, requestId);
    }

    }

    @Override
    @Transactional
    public void cancelMyRequestsByJournalPrefixes(List<String> journalIdPrefixes, String userId) {
        log.info("User {} attempting to bulk-cancel PENDING requests for journal prefixes: {}", userId, journalIdPrefixes.size());
        
        List<MocRequest> requestsToCancel = new ArrayList<>();
        for (String prefix : journalIdPrefixes) {
            List<MocRequest> requestsForPrefix = mocRequestRepository.findByJournalIdStartingWithAndCreatorIdAndRequestStatus(
                    prefix, userId, RequestStatus.PENDING);
            
            if (!requestsForPrefix.isEmpty()) {
                 requestsToCancel.addAll(requestsForPrefix);
            } else {
                 log.warn("No pending requests found for user {} with journal prefix {}", userId, prefix);
            }
        }

        if (requestsToCancel.isEmpty()) {
            throw new ResourceNotFoundException("No pending requests found for you with the specified journal prefixes.");
        }

        mocRequestRepository.deleteAll(requestsToCancel);
        log.info("User {} successfully bulk-canceled {} requests for {} journal prefixes.", userId, requestsToCancel.size(), journalIdPrefixes.size());
    }

package com.tcs.fincore.MOCService.Service;

import jakarta.persistence.EntityManager;
import jakarta.persistence.PersistenceContext;
import lombok.RequiredArgsConstructor;
import org.springframework.stereotype.Service;
import jakarta.transaction.Transactional;

@Service
@RequiredArgsConstructor
public class SequenceService {

    @PersistenceContext
    private EntityManager entityManager;

    /**
     * Pads the given number with leading zeros to a total length of 7.
     */
    private String padToSevenDigits(Long id) {
        if (id == null) {
            throw new IllegalStateException("Sequence query returned null");
        }
        return String.format("%07d", id);
    }

    /**
     * Gets the next BATCH_ID from the BATCH_SEQ sequence. 
     */
    @Transactional
    public String getNextBatchId() {
        Number nextId = (Number) entityManager
                .createNativeQuery("select BATCH_SEQ.nextval from dual")
                .getSingleResult();
        
        return padToSevenDigits(nextId.longValue());
    }

    /**
     * Gets the next JOURNAL_ID prefix from the JRNL_SEQ sequence.
     */
    @Transactional
    public String getNextJournalIdPrefix() {
        
      
        Number nextId = (Number) entityManager
                .createNativeQuery("select JRNL_SEQ.nextval from dual")
                .getSingleResult();
      

        return padToSevenDigits(nextId.longValue());
    }
}

        const batchId = request.batchId;

        if (!groupedBatches.has(batchId)) {
          groupedBatches.set(batchId, {
            id: batchId,
            batchId: batchId,
            creatorId: request.creatorId,
            requestStatus: request.requestStatus,
            requestDate: request.requestDate,
            commonBatchRemarks: request.commonBatchRemarks,
            totalDebit: 0,
            totalCredit: 0,
            requestCount: 0,
          });
        }

        const batchDetails = groupedBatches.get(batchId);
        batchDetails.requestCount++;

        try {
          
          const payload = JSON.parse(request.payload);
          const amount = parseFloat(payload.amount) || 0;
          
          // Negative is Credit, Positive is Debit
          if (amount < 0) { 
            batchDetails.totalCredit += Math.abs(amount);
          } else { 
            batchDetails.totalDebit += amount;
          }
         
        } catch (e) {
          console.error(`Failed to parse payload for request ${request.id}`, e);
        }
      }
      setRows(Array.from(groupedBatches.values()));
    } catch (err) {
      console.error("Failed to fetch pending requests:", err);
      const errorMsg =
        err.response?.data?.message || "Failed to load pending requests.";
      showSnackBar(errorMsg, "error");
    } finally {
      setLoading(false);
    }

              // Negative is Credit, Positive is Debit
              return {
                ...req,
                payload,
                journalPrefix: prefix,
                amount: Math.abs(amount),
                transactionType: amount < 0 ? "Credit" : "Debit",

            } catch (e) {
              console.error("Failed to parse payload for req:", req.id, e);
              return null;
            }

          .filter(Boolean);
