  if (req.user.user_role !== "1") {
    return res.status(401).json({
      message: "Unauthorized api call, request not from legal source",
    });
  }

  try {
    console.log("req.data", req.data);
    console.log("req.data", req.user);
    let data={ user: {
            template: "",
            financial_year: req.user.financial_year,
            pf_number: req.user.pf_number,
          },
          data: { firms: req.data }}        
    const response = await fetch(
      "http://10.0.26.158:8081/FRNController/generateAllPdf",
      {
        method: "POST",
        body: JSON.stringify({
          user: {
            template: "",
            financial_year: req.user.financial_year,
            pf_number: req.user.pf_number,
          },
          data: { firms: req.data },
        }),
        headers: {
          "Content-Type": "application/json",
        },
      }
    );
    const data = await response.json();
    res.json(data);
  } catch (e) {
    res.status(500).json({
      status: "Failed to bring reports list, no connection between LB and APP",
    });
  }
});
