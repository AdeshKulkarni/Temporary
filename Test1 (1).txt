package com.fincore.JournalService.Models;

import jakarta.persistence.*;
import lombok.Data;
import lombok.NoArgsConstructor;
import lombok.AllArgsConstructor;
import java.time.LocalDateTime;

@Entity
@Table(name = "JOURNAL_ACTIVITY_LOGS") // Ensure this matches your actual table name
@Data
@NoArgsConstructor
@AllArgsConstructor
public class JournalLog {

    @Id
    @GeneratedValue(strategy = GenerationType.SEQUENCE, generator = "JOURNAL_LOG_SEQ_GEN")
    @SequenceGenerator(
        name = "JOURNAL_LOG_SEQ_GEN", 
        sequenceName = "JOURNAL_LOG_SEQ", // Must match the SQL sequence name created above
        allocationSize = 1
    )
    @Column(name = "LOG_ID")
    private Long id;

    @Column(name = "USER_ID")
    private String userId;

    @Column(name = "ACTION_TYPE")
    private String actionType; 

    @Column(name = "CHANGE_TYPE")
    private String changeType; 

    @Column(name = "ACTION_TIME")
    private LocalDateTime actionTime;

    @Column(name = "REQUEST_ID")
    private Long requestId; 

    @Column(name = "IP_ADDRESS")
    private String ipAddress;

    @Lob 
    @Column(name = "NEW_VALUE")
    private String newValue;

    @Lob
    @Column(name = "OLD_VALUE")
    private String oldValue;
}
