import React, { useState, useMemo, useEffect, useCallback } from "react";
import {
  Box,
  Dialog,
  DialogActions,
  DialogContent,
  DialogTitle,
  TextField,
  Button,
  Grid,
  FormControl,
  InputLabel,
  Select,
  MenuItem,
  CircularProgress,
} from "@mui/material";
import { DataGrid, GridActionsCellItem } from "@mui/x-data-grid";
import EditIcon from "@mui/icons-material/Edit";
import HistoryIcon from "@mui/icons-material/History";
import useApi from "../../../hooks/useApi";
import useCustomSnackbar from "../../../utils/useCustomSnackbar";

const initialFormState = {
  currencyCode: "",
  currencyRate: "",
};

const getFinancialYear = (date) => {
  const year = date.getFullYear();
  const month = date.getMonth();
  return month >= 3 ? `${year}-${year + 1}` : `${year - 1}-${year}`;
};

export default function CurrencyRateChange() {
  const { callApi } = useApi();
  const showSnackBar = useCustomSnackbar();
  const [rows, setRows] = useState([]);
  const [isLoading, setIsLoading] = useState(true);
  const [isDialogOpen, setIsDialogOpen] = useState(false);
  const [formData, setFormData] = useState(initialFormState);
  const [editingRow, setEditingRow] = useState(null);
  const [errors, setErrors] = useState({});

  const [isHistoryDialogOpen, setIsHistoryDialogOpen] = useState(false);
  const [historyData, setHistoryData] = useState([]);
  const [isHistoryLoading, setIsHistoryLoading] = useState(false);
  const [currentCurrency, setCurrentCurrency] = useState(null);
  const [financialYearOptions, setFinancialYearOptions] = useState([]);
  const [selectedYear, setSelectedYear] = useState("");
  const [selectedMonth, setSelectedMonth] = useState("all");
  const [monthOptions, setMonthOptions] = useState([{ value: "all", label: "All Months" }]);

  const fetchData = useCallback(async () => {
    setIsLoading(true);
    try {
      const response = await callApi("CM/currency-master", null, "GET");
      if (response && response.data && Array.isArray(response.data)) {
        const formattedData = response.data.map((row) => {
          let formattedDate = "N/A";
          if (row.updatedAt) {
            const date = new Date(row.updatedAt);
            if (!isNaN(date.getTime())) {
              const day = String(date.getDate()).padStart(2, "0");
              const month = String(date.getMonth() + 1).padStart(2, "0");
              const year = date.getFullYear();
              formattedDate = `${day}-${month}-${year}`;
            }
          }
          return { ...row, id: row.currencyCode, updatedAt: formattedDate };
        });
        setRows(formattedData);
      } else {
        showSnackBar("Received invalid data format.", "error");
        setRows([]);
      }
    } catch (error) {
      console.error("Error fetching currency master data:", error);
      showSnackBar("Something went wrong while fetching data.", "error");
      setRows([]);
    } finally {
      setIsLoading(false);
    }
  }, [callApi, showSnackBar]);

  useEffect(() => {
    fetchData();
  }, [fetchData]);

  // --- BUG FIX: This function is now corrected to filter by currencyCode ---
  const fetchHistoryData = useCallback(
    async (currencyCode) => {
      setIsHistoryLoading(true);
      try {
        // Step 1: Fetch the entire history log from the API
        const response = await callApi(`CM/currency-rate-change`, null, "GET");

        if (response && response.data && Array.isArray(response.data)) {
          // Step 2: **CRITICAL FIX** - Filter the results for the selected currency
          const currencySpecificData = response.data.filter(
            (item) => item.currencyCode === currencyCode
          );

          if (currencySpecificData.length === 0) {
            setHistoryData([]);
            showSnackBar(`No history data found for currency: ${currencyCode}.`, "info");
          } else {
            // Step 3: Process and set ONLY the filtered data to state
            const processedData = currencySpecificData.map((item) => ({
              ...item,
              dateObject: new Date(item.rateChangeDate),
            }));

            setHistoryData(processedData);

            // The rest of the logic now correctly operates on the filtered data
            const years = [...new Set(processedData.map((item) => getFinancialYear(item.dateObject)))];
            years.sort().reverse();
            setFinancialYearOptions(years);
            if (years.length > 0) {
              setSelectedYear(years[0]);
            }
          }
        } else {
          setHistoryData([]);
          showSnackBar("Received invalid history data format from server.", "warning");
        }
      } catch (error) {
        console.error("Error fetching currency history:", error);
        showSnackBar("Failed to fetch currency rate history.", "error");
        setHistoryData([]);
      } finally {
        setIsHistoryLoading(false);
      }
    },
    [callApi, showSnackBar]
  );

  useEffect(() => {
    if (!selectedYear || historyData.length === 0) {
      setMonthOptions([{ value: "all", label: "All Months" }]);
      return;
    }
    const availableMonths = new Set();
    historyData.forEach((item) => {
      if (getFinancialYear(item.dateObject) === selectedYear) {
        availableMonths.add(item.dateObject.getMonth());
      }
    });
    const sortedMonths = Array.from(availableMonths).sort((a, b) => a - b);
    const options = [
      { value: "all", label: "All Months" },
      ...sortedMonths.map((monthIndex) => ({
        value: monthIndex,
        label: new Date(0, monthIndex).toLocaleString("default", { month: "long" }),
      })),
    ];
    setMonthOptions(options);
    setSelectedMonth("all");
  }, [selectedYear, historyData]);

  const filteredHistoryData = useMemo(() => {
    if (!selectedYear) return [];
    return historyData.filter((item) => {
      const itemFinancialYear = getFinancialYear(item.dateObject);
      const yearMatch = itemFinancialYear === selectedYear;
      const monthMatch = selectedMonth === "all" || item.dateObject.getMonth() === selectedMonth;
      return yearMatch && monthMatch;
    });
  }, [historyData, selectedYear, selectedMonth]);

  const handleCloseDialog = () => {
    setIsDialogOpen(false);
    setEditingRow(null);
    setFormData(initialFormState);
    setErrors({});
  };

  const handleEditClick = useCallback(
    (id) => () => {
      const rowToEdit = rows.find((row) => row.id === id);
      if (rowToEdit) {
        setEditingRow(rowToEdit);
        setFormData({
          currencyCode: rowToEdit.currencyCode,
          currencyRate: rowToEdit.currencyRate,
        });
        setErrors({});
        setIsDialogOpen(true);
      }
    },
    [rows]
  );

  const handleHistoryClick = useCallback(
    (id) => () => {
      setCurrentCurrency(id);
      setIsHistoryDialogOpen(true);
      fetchHistoryData(id);
    },
    [fetchHistoryData]
  );

  const handleCloseHistoryDialog = () => {
    setIsHistoryDialogOpen(false);
    setHistoryData([]);
    setCurrentCurrency(null);
    setFinancialYearOptions([]);
    setSelectedYear("");
    setSelectedMonth("all");
  };

  const validate = () => {
    let tempErrors = {};
    if (!String(formData.currencyRate).trim()) {
      tempErrors.currencyRate = "Currency Rate is required.";
    }
    setErrors(tempErrors);
    return Object.keys(tempErrors).length === 0;
  };

  const handleInputChange = (e) => {
    const { name, value } = e.target;
    if (name === "currencyRate") {
      if (value !== "" && !/^\d{0,6}(\.\d{0,6})?$/.test(value)) {
        return;
      }
      setFormData({ ...formData, [name]: value });
      if (errors[name]) {
        setErrors({ ...errors, [name]: "" });
      }
    }
  };

  const handleSubmit = async () => {
    if (!validate()) return;
    const apiPayload = {
      requestType: "CURRENCY_RATE_CHANGE",
      changeType: "UPDATE",
      payload: {
        currencyCode: editingRow.currencyCode,
        currencyRate: parseFloat(formData.currencyRate) || 0,
      },
    };
    try {
      await callApi("CR/create-request", apiPayload, "POST");
      showSnackBar(`Request to update currency rate submitted successfully!`, "success");
      handleCloseDialog();
      fetchData();
    } catch (error) {
      console.error("Error submitting request:", error);
      showSnackBar(error.response?.data?.message || `Failed to submit update request.`, "error");
    }
  };

  const columns = useMemo(
    () => [
      { field: "currencyCode", headerName: "Currency Code", flex: 1, align: "center", headerAlign: "center" },
      { field: "currencyRate", headerName: "Current Rate", type: "number", flex: 1, align: "center", headerAlign: "center" },
      { field: "updatedAt", headerName: "Last Updated", flex: 1, align: "center", headerAlign: "center" },
      {
        field: "actions", type: "actions", headerName: "Actions", width: 120, align: "center", headerAlign: "center",
        getActions: ({ id }) => [
          <GridActionsCellItem icon={<EditIcon />} label="Edit Rate" onClick={handleEditClick(id)} color="primary" />,
          <GridActionsCellItem icon={<HistoryIcon />} label="View History" onClick={handleHistoryClick(id)} color="inherit" />,
        ],
      },
    ],
    [handleEditClick, handleHistoryClick]
  );

  const historyColumns = [
    { field: "currencyRate", headerName: "Changed Rate", type: 'number', flex: 1, align: "right", headerAlign: "right" },
    {
      field: "rateChangeDate", headerName: "Date of Change", width: 250, align: "left", headerAlign: "left",
      valueFormatter: (params) => {
        if (!params.value) return "";
        const date = new Date(params.value);
        return date.toLocaleString('en-IN', { day: '2-digit', month: '2-digit', year: 'numeric', hour: '2-digit', minute: '2-digit', second: '2-digit', hour12: true });
      },
    },
  ];

  return (
    <Box sx={{ p: 3 }}>
      <DataGrid
        rows={rows}
        columns={columns}
        disableRowSelectionOnClick
        pageSizeOptions={[5, 10, 25, 50]}
        initialState={{
          pagination: { paginationModel: { pageSize: 10 } },
          sorting: { sortModel: [{ field: "updatedAt", sort: "desc" }] },
        }}
        loading={isLoading}
        autoHeight
        sx={{
          width: "100%",
          "& .MuiDataGrid-cell:focus, & .MuiDataGrid-cell:focus-within": { outline: "none" },
          "& .MuiDataGrid-row:nth-of-type(odd)": { backgroundColor: (theme) => theme.palette.action.hover },
        }}
      />

      <Dialog open={isDialogOpen} onClose={handleCloseDialog} fullWidth maxWidth="sm">
        <DialogTitle sx={{ fontWeight: "bold" }}>Edit Currency Rate</DialogTitle>
        <DialogContent>
          <TextField autoFocus margin="dense" name="currencyCode" label="Currency Code" type="text" fullWidth variant="outlined" value={formData.currencyCode} disabled />
          <TextField margin="dense" name="currencyRate" label="New Currency Rate" type="text" fullWidth variant="outlined" value={formData.currencyRate} onChange={handleInputChange} error={!!errors.currencyRate} helperText={errors.currencyRate} />
        </DialogContent>
        <DialogActions sx={{ p: "16px 24px" }}>
          <Button onClick={handleCloseDialog}>Cancel</Button>
          <Button onClick={handleSubmit} variant="contained">Submit Request</Button>
        </DialogActions>
      </Dialog>
      
      <Dialog open={isHistoryDialogOpen} onClose={handleCloseHistoryDialog} fullWidth maxWidth="md">
        <DialogTitle sx={{ fontWeight: "bold" }}>
          Rate Change History for {currentCurrency}
        </DialogTitle>
        <DialogContent>
          {isHistoryLoading ? (
             <Box sx={{ display: 'flex', justifyContent: 'center', alignItems: 'center', height: 300 }}>
               <CircularProgress />
             </Box>
          ) : (
            <>
              <Grid container spacing={2} sx={{ mb: 2, mt: 1 }}>
                <Grid item xs={12} sm={6}>
                  <FormControl fullWidth>
                    <InputLabel id="financial-year-select-label">Financial Year</InputLabel>
                    <Select
                      labelId="financial-year-select-label"
                      value={selectedYear}
                      label="Financial Year"
                      onChange={(e) => setSelectedYear(e.target.value)}
                    >
                      {financialYearOptions.map(year => (
                        <MenuItem key={year} value={year}>{year}</MenuItem>
                      ))}
                    </Select>
                  </FormControl>
                </Grid>
                <Grid item xs={12} sm={6}>
                  <FormControl fullWidth>
                    <InputLabel id="month-select-label">Month</InputLabel>
                    <Select
                      labelId="month-select-label"
                      value={selectedMonth}
                      label="Month"
                      onChange={(e) => setSelectedMonth(e.target.value)}
                    >
                      {monthOptions.map(option => (
                        <MenuItem key={option.value} value={option.value}>
                            {option.label}
                        </MenuItem>
                      ))}
                    </Select>
                  </FormControl>
                </Grid>
              </Grid>
              <Box sx={{ height: 400, width: '100%' }}>
                <DataGrid
                  rows={filteredHistoryData}
                  columns={historyColumns}
                  pageSizeOptions={[5, 10, 25]}
                  initialState={{
                     pagination: { paginationModel: { pageSize: 10 } },
                     sorting: { sortModel: [{ field: 'rateChangeDate', sort: 'desc' }] },
                  }}
                  disableRowSelectionOnClick
                  sx={{ "& .MuiDataGrid-cell:focus, & .MuiDataGrid-cell:focus-within": { outline: "none" } }}
                  localeText={{ noRowsLabel: 'No history found for the selected period' }}
                />
              </Box>
            </>
          )}
        </DialogContent>
        <DialogActions sx={{ p: "16px 24px" }}>
          <Button onClick={handleCloseHistoryDialog}>Close</Button>
        </DialogActions>
      </Dialog>
    </Box>
  );
}
