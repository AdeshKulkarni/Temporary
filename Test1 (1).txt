  // --- BACKDATED FEATURE: Reload Currencies when VID is selected ---
  useEffect(() => {
    const reloadCurrenciesForVid = async () => {
      if (!selectedVoucher) return;

      try {
        let url = "/JS/voucher/currencies";
        if (selectedVoucher.isBackDated === 1 && selectedVoucher.backDate) {
          url += `?backDate=${selectedVoucher.backDate}`;
        }

        const currencyResponse = await callApi(url, null, "GET");
        const currencyData = currencyResponse?.data || currencyResponse;

        if (Array.isArray(currencyData)) {
          let uniqueCurrencies = [...new Set(currencyData)];
          uniqueCurrencies.sort((a, b) => {
            if (a === "INR") return -1;
            if (b === "INR") return 1;
            return a.localeCompare(b);
          });
          setCurrencyOptions(uniqueCurrencies);
        }
      } catch (e) {
        console.error("Failed to reload currencies for backdate", e);
      }
    };

    reloadCurrenciesForVid();
  }, [selectedVoucher, callApi]);
