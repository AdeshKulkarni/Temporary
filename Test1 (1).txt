import React, { useState, useEffect } from 'react';

// --- Material-UI Imports ---
import {
    Box,
    Typography,
    Fab,
    Dialog,
    DialogActions,
    DialogContent,
    DialogTitle,
    TextField,
    Button,
    Container,
    Paper
} from '@mui/material';
import AddIcon from '@mui/icons-material/Add';
import EditIcon from '@mui/icons-material/Edit';
// --- CHANGED: DeleteIcon is no longer needed ---
// import DeleteIcon from '@mui/icons-material/Delete'; 
import { DataGrid, GridActionsCellItem } from '@mui/x-data-grid';


// --- CHANGED: Mock data now has 'createdAt' instead of 'rateDate' ---
const fetchCurrencyData = () => {
    return new Promise((resolve) => {
        const dataFromBackend = [
            { id: 1, currencyCode: 'USD', currencyName: 'United States Dollar', currencyRate: 1.00, createdAt: '2025-09-29T10:00:00Z' },
            { id: 2, currencyCode: 'EUR', currencyName: 'Euro', currencyRate: 0.92, createdAt: '2025-09-29T11:30:00Z' },
            { id: 3, currencyCode: 'INR', currencyName: 'Indian Rupee', currencyRate: 83.50, createdAt: '2025-09-29T12:00:00Z' },
            { id: 4, currencyCode: 'GBP', currencyName: 'British Pound', currencyRate: 0.79, createdAt: '2025-09-28T09:00:00Z' },
            { id: 5, currencyCode: 'JPY', currencyName: 'Japanese Yen', currencyRate: 149.82, createdAt: '2025-09-29T14:45:00Z' },
        ];
        setTimeout(() => resolve(dataFromBackend), 1500);
    });
};

// --- CHANGED: initialFormState updated ---
const initialFormState = {
    currencyCode: '',
    currencyName: '',
    currencyRate: '',
};

function CurrencyMaster() {
    // State Management
    const [rows, setRows] = useState([]);
    const [loading, setLoading] = useState(true);
    const [isDialogOpen, setIsDialogOpen] = useState(false);
    const [formData, setFormData] = useState(initialFormState);
    const [editingRow, setEditingRow] = useState(null);
    const [errors, setErrors] = useState({});

    // Data Fetching
    useEffect(() => {
        setLoading(true);
        fetchCurrencyData().then(data => {
            setRows(data);
            setLoading(false);
        }).catch(error => {
            console.error("Failed to fetch currency data:", error);
            setLoading(false);
        });
    }, []);

    const handleEditClick = (id) => () => {
        const rowToEdit = rows.find((row) => row.id === id);
        if (rowToEdit) {
            setEditingRow(rowToEdit);
            setFormData(rowToEdit);
            setIsDialogOpen(true);
        }
    };

    const columns = [
        { field: 'id', headerName: 'ID', width: 70 },
        { field: 'currencyCode', headerName: 'Currency Code', width: 130 },
        { field: 'currencyName', headerName: 'Currency Name', flex: 1, minWidth: 200 },
        { field: 'currencyRate', headerName: 'Currency Rate', type: 'number', width: 130, align: 'right', headerAlign: 'right' },
        // --- CHANGED: Replaced 'rateDate' with 'createdAt' ---
        {
            field: 'createdAt',
            headerName: 'Created At',
            type: 'dateTime',
            width: 180,
            valueGetter: (params) => params.value ? new Date(params.value) : null,
            renderCell: (params) => params.value ? new Date(params.value).toLocaleString() : ''
        },
        {
            field: 'actions',
            type: 'actions',
            headerName: 'Actions',
            width: 80,
            cellClassName: 'actions',
            // --- CHANGED: Removed the Delete action item ---
            getActions: ({ id }) => [
                <GridActionsCellItem icon={<EditIcon />} label="Edit" onClick={handleEditClick(id)} color="primary" />,
            ],
        },
    ];

    const handleOpenDialog = () => {
        setEditingRow(null);
        setFormData(initialFormState);
        setIsDialogOpen(true);
    };

    const handleCloseDialog = () => {
        setIsDialogOpen(false);
        setEditingRow(null);
        setFormData(initialFormState);
        setErrors({});
    };

    const handleInputChange = (e) => {
        const { name, value } = e.target;
        setFormData(prev => ({ ...prev, [name]: value }));
    };

    // --- CHANGED: All new validation logic is implemented here ---
    const validateForm = () => {
        let tempErrors = {};
        
        // --- Validation for Currency Name (applies to both Add and Edit) ---
        if (!formData.currencyName) {
            tempErrors.currencyName = "Currency Name is required.";
        } else if (formData.currencyName.length > 30) {
            tempErrors.currencyName = "Cannot exceed 30 characters.";
        }

        // --- Validations for Add mode ONLY ---
        if (!editingRow) {
            // --- Validation for Currency Code ---
            if (!formData.currencyCode) {
                tempErrors.currencyCode = "Currency Code is required.";
            } else if (!/^[A-Za-z]{3}$/.test(formData.currencyCode)) {
                tempErrors.currencyCode = "Must be exactly 3 alphabets.";
            }

            // --- Validation for Currency Rate ---
            if (!formData.currencyRate) {
                tempErrors.currencyRate = "Rate is required.";
            } else if (!/^\d{1,6}(\.\d{1,6})?$/.test(formData.currencyRate)) {
                tempErrors.currencyRate = "Invalid format. E.g., 123456.123456";
            }
        }
        
        setErrors(tempErrors);
        return Object.keys(tempErrors).length === 0;
    };

    const handleSubmit = () => {
        if (validateForm()) {
            const now = new Date().toISOString();
            if (editingRow) {
                // Edit mode
                setRows(rows.map((row) => (row.id === editingRow.id ? formData : row)));
            } else {
                // Add mode
                const newId = rows.length > 0 ? Math.max(...rows.map(r => r.id)) + 1 : 1;
                setRows([...rows, { ...formData, id: newId, createdAt: now }]);
            }
            handleCloseDialog();
        }
    };

    return (
        <Container maxWidth="xl" sx={{ mt: 4, mb: 4, display: 'flex', justifyContent: 'center', width: '100%' }}>
            <Paper elevation={3} sx={{ p: 3, borderRadius: 2, width: '80%', maxWidth: 1200 }}>
                <Typography variant="h4" component="h1" gutterBottom sx={{ fontWeight: '600', color: 'primary.main' }}>
                    Currency Master
                </Typography>
                <Box sx={{ height: '70vh', width: '100%' }}>
                    <DataGrid rows={rows} columns={columns} loading={loading} disableRowSelectionOnClick />
                </Box>
            </Paper>

            <Fab color="primary" aria-label="add" onClick={handleOpenDialog} sx={{ position: 'fixed', bottom: 30, right: 30 }}>
                <AddIcon />
            </Fab>

            <Dialog open={isDialogOpen} onClose={handleCloseDialog} fullWidth maxWidth="sm">
                <DialogTitle sx={{ fontWeight: 'bold' }}>
                    {editingRow ? 'Edit Currency' : 'Add New Currency'}
                </DialogTitle>
                <DialogContent>
                    {/* --- CHANGED: Added 'disabled' prop based on editingRow state --- */}
                    <TextField 
                        autoFocus 
                        margin="dense" 
                        name="currencyCode" 
                        label="Currency Code" 
                        type="text" 
                        fullWidth 
                        variant="outlined" 
                        value={formData.currencyCode} 
                        onChange={handleInputChange} 
                        error={!!errors.currencyCode} 
                        helperText={errors.currencyCode} 
                        disabled={!!editingRow}
                    />
                    <TextField 
                        margin="dense" 
                        name="currencyName" 
                        label="Currency Name" 
                        type="text" 
                        fullWidth 
                        variant="outlined" 
                        value={formData.currencyName} 
                        onChange={handleInputChange} 
                        error={!!errors.currencyName} 
                        helperText={errors.currencyName}
                    />
                    <TextField 
                        margin="dense" 
                        name="currencyRate" 
                        label="Currency Rate" 
                        type="number" 
                        fullWidth 
                        variant="outlined" 
                        value={formData.currencyRate} 
                        onChange={handleInputChange} 
                        error={!!errors.currencyRate} 
                        helperText={errors.currencyRate} 
                        disabled={!!editingRow}
                    />
                    {/* --- CHANGED: Removed the rateDate TextField completely --- */}
                </DialogContent>
                <DialogActions sx={{ p: '16px 24px' }}>
                    <Button onClick={handleCloseDialog}>Cancel</Button>
                    <Button onClick={handleSubmit} variant="contained">Submit</Button>
                </DialogActions>
            </Dialog>
        </Container>
    );
}

export default CurrencyMaster;
