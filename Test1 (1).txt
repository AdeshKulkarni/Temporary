class Solution {

    public int Helper(TreeNode root , int currentCount , int maxCount) {
        
        Boolean LeftToRight = false;

        if(root == null) {
            return 0;
        }

        if(LeftToRight) {
            currentCount  = Helper(root.left , currentCount , maxCount) + 1;
             D.p("left currentCount " + currentCount);
        }
        else {
            currentCount = Helper(root.right , currentCount , maxCount) + 1;
            D.p("right currentCount " + currentCount);

        }

        LeftToRight = !LeftToRight;
        
         maxCount = Math.max(maxCount , currentCount);
  

         return maxCount;       

        

    }


    public int longestZigZag(TreeNode root) {

        return Helper(root , 0 , 0) - 1;
        
    }
}

Arrays.deepToString(new Object[]{o[i + 1]}) : "[]"; sb.append(o[i]).append(": ").append(val.substring(1, val.length() - 1)).append(" | "); } System.out.println(sb); } }
