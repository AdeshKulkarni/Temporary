class Solution {
    // Rule 1: The Global "Record Book" Variable
    // We store the final answer here safely outside the recursion.
    TreeNode ans = null;

    // Rule 2: The Helper Method (Bottom-Up)
    // Returns an int representing how many targets (0, 1, or 2) it found.
    public int helper(TreeNode node, TreeNode p, TreeNode q) {
        
        // Base case: hit the bottom, found nothing.
        if (node == null) {
            return 0;
        }

        // Post-Order: Ask the left and right children FIRST.
        int left = helper(node.left, p, q);
        int right = helper(node.right, p, q);

        // Process the current node LAST. 
        // Am I one of the targets?
        int self = 0;
        if (node == p || node == q) {
            self = 1;
        }

        // Tally up everything found in this entire subtree
        int total = left + right + self;

        // THE MAGIC MOMENT: 
        // If the total is exactly 2, it means both p and q are in this subtree.
        // Because we are Bottom-Up, the very first node to hit '2' is the LOWEST manager.
        if (total == 2 && ans == null) {
            ans = node;
        }

        // Return the count to the parent node above us.
        return total;
    }

    public TreeNode lowestCommonAncestor(TreeNode root, TreeNode p, TreeNode q) {
        // Kick off the DFS. We don't care about the final int return value here.
        helper(root, p, q);
        
        // Return the node we saved in our global variable!
        return ans;
    }
}
