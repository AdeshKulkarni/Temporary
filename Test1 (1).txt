private static String sanitizeSegmentString(String value, String fieldName) {
        if (value == null || value.isEmpty() || value.contains("..")) {
            throw new IllegalArgumentException("Invalid characters in " + fieldName);
        }
        StringBuilder clean = new StringBuilder(value.length());
        for (int i = 0; i < value.length(); i++) {
            char c = value.charAt(i);
            int idx = ALLOWED_CHARS.indexOf(c);
            if (idx < 0) {
                throw new IllegalArgumentException("Invalid character in " + fieldName);
            }
            clean.append(ALLOWED_CHARS.charAt(idx));
        }
        String cleans = clean.toString();
        log.info("Key is {}", cleans);
        return cleans;
    }
