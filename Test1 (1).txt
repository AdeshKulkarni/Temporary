MERGE INTO datalakedb.gl_balance b
USING (
    SELECT
        CONCAT(
            jt.branch,
            jt.currency,
            jt.cgl
        ) AS glcc,
        SUM(
            CASE
                WHEN jt.currency = 'INR' THEN jt.amount
                ELSE jt.amount * COALESCE(cm.currency_rate, 1)
            END
        ) AS net_amount
    FROM journal_request jr
    LATERAL VIEW JSON_TUPLE(
        jr.payload,
        'branch',
        'currency',
        'cgl',
        'amount'
    ) jt AS branch, currency, cgl, amount
    LEFT JOIN currency_master cm
        ON jt.currency = cm.currency_code
    WHERE jr.req_status = 'P'
      AND jr.creator_id <> ?
      AND /* your targetClause here */
    GROUP BY
        jt.branch,
        jt.currency,
        jt.cgl
) s
ON b.glcc = s.glcc
WHEN MATCHED THEN
  UPDATE SET
    b.amount = b.amount + s.net_amount,
    b.balance_date = CURRENT_DATE()
WHEN NOT MATCHED THEN
  INSERT (glcc, amount, balance_date)
  VALUES (s.glcc, s.net_amount, CURRENT_DATE());
