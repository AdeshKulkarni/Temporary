class Employee {
    String name;
    Integer id;
    Long BaseSalary = Long.valueOf(56);

       Long CalulateSalery() {
       return BaseSalary;
    }

}

class Manager extends Employee {
    Integer bonus;


    @Override
    Long CalulateSalery() {
        return BaseSalary + bonus;
    }



}
class Intern extends Employee {
    Integer Stipend;

    @Override
    Long CalulateSalery() {
        return BaseSalary - Stipend;
    }
}
