class Solution {

    public void Helper(TreeNode root , int targetSum , int currentSum , List<List<Integer>> ans , List<Integer> current) {

        if(root == null) {
            return;
        }

        currentSum += root.val;

        current.add(root.val);

           D.p("adding element" + root.val);
           D.p("adding element list" + current);
           D.p("Sum" + currentSum);



        if(root.left == null && root.right == null) {

            if(currentSum == targetSum) {
                ans.add(new ArrayList<>(current));
              current.subList(1, current.size()).clear();   
            }
        }

         if(Math.abs(currentSum) > Math.abs(targetSum)) {
             currentSum -= root.val;
             current.remove(current.size() - 1);

             D.p("removed Element" + root.val);

        }

        Helper(root.left , targetSum , currentSum , ans , current);
        Helper(root.right , targetSum , currentSum ,  ans , current);

    } 

    public List<List<Integer>> pathSum(TreeNode root, int targetSum) {
        List<List<Integer>> ans = new ArrayList<>();
        
        Helper(root , targetSum , 0 , ans , new ArrayList<>());

        return ans;

    }
}

class D { static void p(Object... o) { StringBuilder sb = new StringBuilder(); for (int i = 0; i < o.length; i += 2) { String val = (i + 1 < o.length) ? Arrays.deepToString(new Object[]{o[i + 1]}) : "[]"; sb.append(o[i]).append(": ").append(val.substring(1, val.length() - 1)).append(" | "); } System.out.println(sb); } }
