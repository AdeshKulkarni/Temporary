package com.fincore.JournalService.Aspect;

import com.fasterxml.jackson.databind.ObjectMapper;
import com.fincore.JournalService.Dto.BatchRequestDto;
import com.fincore.JournalService.Dto.BulkProcessJournalRequestDto;
import com.fincore.JournalService.Dto.ProcessJournalRequestDto;
import com.fincore.JournalService.Models.JournalLog;
import com.fincore.JournalService.Models.JournalRequest;
import com.fincore.JournalService.Repository.JournalLogRepository;
import jakarta.servlet.http.HttpServletRequest;
import lombok.RequiredArgsConstructor;
import lombok.extern.slf4j.Slf4j;
import org.aspectj.lang.JoinPoint;
import org.aspectj.lang.annotation.AfterReturning;
import org.aspectj.lang.annotation.Aspect;
import org.springframework.stereotype.Component;
import org.springframework.web.context.request.RequestContextHolder;
import org.springframework.web.context.request.ServletRequestAttributes;

import java.time.LocalDateTime;

@Aspect
@Component
@Slf4j
@RequiredArgsConstructor
public class JournalActivityLogger {

    private final JournalLogRepository journalLogRepository;
    private final ObjectMapper objectMapper;

    // --- 1. Log Batch Creation (Standard) ---
    @AfterReturning(
            pointcut = "execution(* com.fincore.JournalService.Service.JournalRequestService.createBatchRequest(..)) && args(batchDto, creatorId, creatorRole)",
            returning = "result"
    )
    public void logCreateBatch(JoinPoint joinPoint, BatchRequestDto batchDto, String creatorId, Integer creatorRole, Object result) {
        String summary = "Created Batch with " + (batchDto.getRows() != null ? batchDto.getRows().size() : 0) + " rows.";
        // Passing null for requestId is correct here, but saveLog must handle it safely
        saveLog(creatorId, "CREATE", "BATCH_CREATION", summary, null, null);
    }

    // --- 2. Log Bulk Batch Creation (Optimized Upload) ---
    @AfterReturning(
            pointcut = "execution(* com.fincore.JournalService.Service.JournalRequestService.createBulkBatchRequest(..)) && args(batchDto, creatorId, creatorRole)",
            returning = "batchId"
    )
    public void logBulkCreate(JoinPoint joinPoint, BatchRequestDto batchDto, String creatorId, Integer creatorRole, Object batchId) {
        String summary = "Bulk Upload Created. Batch ID: " + batchId + ". Rows: " + (batchDto.getRows() != null ? batchDto.getRows().size() : 0);
        saveLog(creatorId, "CREATE", "BULK_UPLOAD", summary, null, null);
    }

    // --- 3. Log Single Status Update (Approve/Reject) ---
    @AfterReturning(
            pointcut = "execution(* com.fincore.JournalService.Service.JournalRequestService.updateRequestStatus(..)) && args(dto, executorId, executorRole)",
            returning = "result"
    )
    public void logStatusUpdate(JoinPoint joinPoint, ProcessJournalRequestDto dto, String executorId, Integer executorRole, Object result) {
        if (result instanceof JournalRequest) {
            JournalRequest req = (JournalRequest) result; // The updated entity
            String newValue = "Status: " + dto.getStatus() + ", Remarks: " + dto.getRemarks();
            // We can treat 'changeType' as the Status (ACCEPTED/REJECTED)
            saveLog(executorId, dto.getStatus().toString(), "STATUS_UPDATE", newValue, null, req.getId());
        }
    }

    // --- 4. Log Bulk Status Update (Bulk Approve/Reject) ---
    @AfterReturning(
            pointcut = "execution(* com.fincore.JournalService.Service.JournalRequestService.processBulkRequests(..)) && args(dto, executorId, executorRole)",
            returning = "result"
    )
    public void logBulkStatusUpdate(JoinPoint joinPoint, BulkProcessJournalRequestDto dto, String executorId, Integer executorRole, Object result) {
        String target = dto.getBatchId() != null ? "Batch ID: " + dto.getBatchId() : "Journals: " + dto.getJournalIdPrefixes();
        String newValue = "Bulk " + dto.getStatus() + " for " + target + ". Remarks: " + dto.getRemarks();
        
        saveLog(executorId, "BULK_" + dto.getStatus(), "BULK_PROCESS", newValue, null, null);
    }

    // --- 5. Log Single Cancellation ---
    @AfterReturning(
            pointcut = "execution(* com.fincore.JournalService.Service.JournalRequestService.cancelMyRequest(..)) && args(requestId, userId)"
    )
    public void logCancelRequest(JoinPoint joinPoint, Long requestId, String userId) {
        saveLog(userId, "DELETE", "CANCEL_REQUEST", "User canceled request ID: " + requestId, null, requestId);
    }

    // --- 6. Log Bulk Cancellation (By Batch) ---
    @AfterReturning(
            pointcut = "execution(* com.fincore.JournalService.Service.JournalRequestService.cancelMyRequestsByBatchId(..)) && args(batchId, userId)"
    )
    public void logCancelBatch(JoinPoint joinPoint, String batchId, String userId) {
        saveLog(userId, "DELETE", "CANCEL_BATCH", "User canceled all requests in Batch: " + batchId, null, null);
    }

    // --- Helper to Save to DB ---
    private void saveLog(String userId, String actionType, String changeType, String newValue, String oldValue, Long requestId) {
        try {
            JournalLog logEntry = new JournalLog();
            
            logEntry.setUserId(userId);
            
            logEntry.setActionType(actionType);
            logEntry.setChangeType(changeType);
            logEntry.setActionTime(LocalDateTime.now());
            logEntry.setRequestId(requestId);
            
            // Capture IP Address
            logEntry.setIpAddress(getClientIp());

            // Convert Objects to JSON String if they aren't already strings
            if (newValue != null && !newValue.getClass().equals(String.class)) {
                newValue = objectMapper.writeValueAsString(newValue);
            }

            logEntry.setNewValue(truncate(newValue, 250));
            logEntry.setOldValue(truncate(oldValue, 250));



            journalLogRepository.save(logEntry);
            log.info("? Audit Log Saved: {} performed {} on {}", userId, actionType, (requestId != null ? requestId : "Batch"));

        } catch (Exception e) {
            log.error("Failed to save audit log: {}", e.getMessage());
        }
    }

    private String getClientIp() {
        try {
            HttpServletRequest request = ((ServletRequestAttributes) RequestContextHolder.currentRequestAttributes()).getRequest();
            String xForwardedFor = request.getHeader("X-Forwarded-For");
            if (xForwardedFor != null && !xForwardedFor.isEmpty()) {
                return xForwardedFor.split(",")[0].trim();
            }
            return request.getRemoteAddr();
        } catch (Exception e) {
            return "UNKNOWN";
        }
    }

    private String truncate(String value, int length) {
        if (value != null && value.length() > length) {
            return value.substring(0, length - 3) + "...";
        }
        return value;
    }
}

package com.fincore.JournalService.config;

import jakarta.persistence.AttributeConverter;
import jakarta.persistence.Converter;
import java.util.stream.Stream;

import com.fincore.JournalService.Models.enums.RequestStatus;

@Converter(autoApply = true)
public class RequestStatusConverter implements AttributeConverter<RequestStatus, String> {

    @Override
    public String convertToDatabaseColumn(RequestStatus status) {
        if (status == null) {
            return null;
        }
        return status.getCode(); // Converts PENDING to "P"
    }

    @Override
    public RequestStatus convertToEntityAttribute(String code) {
        if (code == null) {
            return null;
        }
        // Finds the Enum constant that matches the code from the DB
        return Stream.of(RequestStatus.values())
                .filter(c -> c.getCode().equals(code))
                .findFirst()
                .orElseThrow(IllegalArgumentException::new);
    }
}


package com.fincore.JournalService.config;

import com.fincore.commonutilities.config.RedisConfig; // Ensures Redis Template is loaded
import com.fincore.commonutilities.jwt.JwtUtil;
import com.fincore.commonutilities.security.ContextRbacFilter;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.context.annotation.Bean;
import org.springframework.context.annotation.Configuration;
import org.springframework.context.annotation.Import;
import org.springframework.data.redis.core.StringRedisTemplate;
import org.springframework.security.config.annotation.web.builders.HttpSecurity;
import org.springframework.security.config.annotation.web.configuration.EnableWebSecurity;
import org.springframework.security.config.http.SessionCreationPolicy;
import org.springframework.security.web.SecurityFilterChain;
import org.springframework.security.web.authentication.UsernamePasswordAuthenticationFilter;

/**
 * Global security configuration for the application.
 * This class uses Spring Security's Java configuration to define security rules,
 * such as disabling CSRF, making the session stateless, and defining authorization policies.
 *
 * The `@Configuration` annotation marks this class as a source of bean definitions.
 * The `@EnableWebSecurity` annotation enables Spring Security's web security support and provides
 * the integration with the Spring MVC.
 */
@Configuration
@EnableWebSecurity
@Import({RedisConfig.class, JwtUtil.class}) // Import Redis Config from Library
public class SecurityConfig {

    @Autowired
    private StringRedisTemplate redisTemplate;

    @Autowired
    private JwtUtil jwtUtil;

    @Bean
    public SecurityFilterChain securityFilterChain(HttpSecurity http) throws Exception {
        http
                .csrf(csrf -> csrf.disable())
                .cors(cors -> {}) // Enable CORS (uses your existing WebConfig)
                .sessionManagement(s -> s.sessionCreationPolicy(SessionCreationPolicy.STATELESS))

                // LOCK DOWN ENDPOINTS
                .authorizeHttpRequests(authz -> authz
                        .requestMatchers("/**").permitAll() // Allows access to all your API endpoints
                        .requestMatchers("/actuator/**", "/auth/**", "/error").permitAll()
                        // All other requests MUST pass the ContextRbacFilter
                        .anyRequest().authenticated()
                )

                // ======== INSERTING THE GUARD HERE =======
                // This filter does two things:
                // 1. Validates JWT & Sets User Context
                // 2. Checks Redis for Permissions (e.g. "POST:/create-request|SEGMENT_CODE")
                .addFilterBefore(new ContextRbacFilter(redisTemplate, jwtUtil), UsernamePasswordAuthenticationFilter.class);

        return http.build();
    }
}

package com.fincore.JournalService.Controllers;

import com.fasterxml.jackson.core.JsonProcessingException;
import com.fincore.JournalService.Dto.*;

import com.fincore.commonutilities.jwt.JwtUtil;
import jakarta.validation.Valid;
import lombok.RequiredArgsConstructor;
import lombok.extern.slf4j.Slf4j;
import org.springframework.core.io.ByteArrayResource;
import org.springframework.core.io.Resource;
import org.springframework.http.HttpHeaders;
import org.springframework.http.HttpStatus;
import org.springframework.http.MediaType;
import org.springframework.http.ResponseEntity;
import org.springframework.web.bind.annotation.*;
import org.springframework.web.multipart.MultipartFile;

import java.io.IOException;
import java.time.LocalDate;
import java.util.List;
import java.util.Map;

@RestController
@RequestMapping("journals")
@RequiredArgsConstructor
@Slf4j
public class JournalRequestController {

    private final JournalRequestService journalRequestService;
    private final JournalBulkValidationService journalBulkValidationService;
    private final JwtUtil jwtUtil;

    @GetMapping("/download-template")
    public ResponseEntity<Resource> downloadTemplate() throws IOException {
        byte[] data = journalBulkValidationService.generateTemplateBytes();
        return ResponseEntity.ok()
                .header(HttpHeaders.CONTENT_DISPOSITION, "attachment; filename=\"Journal_Upload_Template.xlsx\"")
                .contentType(MediaType.parseMediaType("application/vnd.openxmlformats-officedocument.spreadsheetml.sheet"))
                .body(new ByteArrayResource(data));
    }

    @PostMapping(value = "/bulk-validate-init", consumes = MediaType.MULTIPART_FORM_DATA_VALUE)
    public ResponseEntity<?> initiateValidation(@RequestParam("file") MultipartFile file,
                                                @RequestParam("postingDate") String date) {
        try {
            LocalDate pDate = LocalDate.parse(date);
            String reqId = journalBulkValidationService.initiateValidation(file, pDate);
            return ResponseEntity.ok(Map.of("status", "QUEUED", "requestId", reqId));
        } catch (IOException e) {
            log.error("File processing failed", e);
            return ResponseEntity.status(HttpStatus.INTERNAL_SERVER_ERROR)
                    .body(Map.of("error", "Failed to process file: " + e.getMessage()));
        } catch (Exception e) {
            log.error("Validation init failed", e);
            return ResponseEntity.badRequest().body(Map.of("error", "Invalid Request: " + e.getMessage()));
        }
    }

    @GetMapping("/bulk-status/{requestId}")
    public ResponseEntity<BulkUploadStateDto> checkStatus(@PathVariable String requestId) {
        BulkUploadStateDto state = journalBulkValidationService.getState(requestId);
        if (state == null) return ResponseEntity.notFound().build();
        return ResponseEntity.ok(state);
    }

    @GetMapping("/download-bulk-file/{requestId}")
    public ResponseEntity<Resource> downloadFile(@PathVariable String requestId, @RequestParam String type) {
        byte[] data = journalBulkValidationService.getFileBytes(requestId, type);
        if (data == null) return ResponseEntity.notFound().build();

        String name = type.equals("ERROR") ? "Error_Report.xlsx" : "Success.csv";
        String mime = type.equals("ERROR") ? "application/vnd.openxmlformats-officedocument.spreadsheetml.sheet" : "text/csv";

        return ResponseEntity.ok()
                .header(HttpHeaders.CONTENT_DISPOSITION, "attachment; filename=\"" + name + "\"")
                .contentType(MediaType.parseMediaType(mime))
                .body(new ByteArrayResource(data));
    }

    // --- STANDARD OPERATIONS ---
    @PostMapping("/create-batch")
    @ResponseStatus(HttpStatus.CREATED)
    public List<JournalRequest> createBatchRequest(@Valid @RequestBody BatchRequestDto batchDto,
                                                   @RequestHeader("Authorization") String token) throws JsonProcessingException {
        String userId = jwtUtil.getUserIdFromToken(token);
        Integer userRole = jwtUtil.getUserRoleFromToken(token);
        return journalRequestService.createBatchRequest(batchDto, userId, userRole);
    }

    @PostMapping("/create-bulk-batch")
    @ResponseStatus(HttpStatus.CREATED)
    public ResponseEntity<Map<String, Object>> createBulkBatchRequest(@RequestBody BatchRequestDto batchDto,
                                                                      @RequestHeader("Authorization") String token) throws JsonProcessingException {
        String userId = jwtUtil.getUserIdFromToken(token);
        Integer userRole = jwtUtil.getUserRoleFromToken(token);
        String batchId = journalRequestService.createBulkBatchRequest(batchDto, userId, userRole);
        return ResponseEntity.ok(Map.of("status", "SUCCESS", "message", "Batch created successfully", "batchId", batchId));
    }

    @PatchMapping("/update-request")
    public JournalRequest updateRequestStatus(@RequestHeader("Authorization") String token,
                                              @RequestBody ProcessJournalRequestDto processRequestDto) throws JsonProcessingException {
        String executorId = jwtUtil.getUserIdFromToken(token);
        Integer executorRole = jwtUtil.getUserRoleFromToken(token);
        return journalRequestService.updateRequestStatus(processRequestDto, executorId, executorRole).get();
    }

    @GetMapping("/my-requests")
    public List<JournalRequest> getMyRequests() { return journalRequestService.getMyRequests(); }

    @GetMapping("/pending-requests")
    public List<JournalRequest> getPendingRequests(@RequestHeader("Authorization") String token) {
        String userId = jwtUtil.getUserIdFromToken(token);
        Integer userRole = jwtUtil.getUserRoleFromToken(token);
        return journalRequestService.getPendingRequests(userId, userRole);
    }

    @PostMapping("/process-bulk")
    public List<JournalRequest> processBulkRequests(@RequestHeader("Authorization") String token,
                                                    @Valid @RequestBody BulkProcessJournalRequestDto processRequestDto) throws JsonProcessingException {
        String executorId = jwtUtil.getUserIdFromToken(token);
        Integer executorRole = jwtUtil.getUserRoleFromToken(token);
        return journalRequestService.processBulkRequests(processRequestDto, executorId, executorRole);
    }

    @GetMapping("/by-batch/{batchId}")
    public ResponseEntity<List<JournalRequest>> getRequestsByBatchId(@PathVariable String batchId) {
        return ResponseEntity.ok(journalRequestService.getRequestsByBatchId(batchId));
    }

    @DeleteMapping("/my-request/{requestId}")
    public ResponseEntity<Void> cancelMyRequest(@RequestHeader("Authorization") String token, @PathVariable Long requestId) {
        journalRequestService.cancelMyRequest(requestId, jwtUtil.getUserIdFromToken(token));
        return ResponseEntity.noContent().build();
    }

    @DeleteMapping("/my-requests/by-batch/{batchId}")
    public ResponseEntity<Void> cancelMyRequestsByBatch(@RequestHeader("Authorization") String token, @PathVariable String batchId) {
        journalRequestService.cancelMyRequestsByBatchId(batchId, jwtUtil.getUserIdFromToken(token));
        return ResponseEntity.noContent().build();
    }

    @DeleteMapping("/my-requests/by-journal-list")
    public ResponseEntity<Void> cancelMyRequestsByJournalPrefixes(@RequestHeader("Authorization") String token, @RequestBody List<String> journalIdPrefixes) {
        journalRequestService.cancelMyRequestsByJournalPrefixes(journalIdPrefixes, jwtUtil.getUserIdFromToken(token));
        return ResponseEntity.noContent().build();
    }

    @GetMapping("/status")
    public ResponseEntity<List<JournalRequestStatusDto>> getJournalStatusList() {
        return ResponseEntity.ok(journalRequestService.getJournalRequestStatusList());
    }
}

@Getter
public enum AcClassification {
	  ASSET('A', "Asset"),
	    LIABILITY('L', "Liability"),
	    INCOME('I', "Income"),
	    EXPENSE('E', "Expense"),
	    MEMO('M', "Memo A/c");

	    private final char code;
	    private final String description;

	    AcClassification(char code, String description) {
	        this.code = code;
	        this.description = description;
	    }

	  
	    public static Map<String, String> getClassificationMap() {
	        return Arrays.stream(AcClassification.values())
	                .collect(Collectors.toMap(
	                        ac -> String.valueOf(ac.getCode()), 
	                        AcClassification::getDescription
	                ));
	    }
}

package com.fincore.JournalService.Service;

import lombok.RequiredArgsConstructor;
import lombok.extern.slf4j.Slf4j;
import org.springframework.data.domain.PageRequest;
import org.springframework.data.domain.Pageable;
import org.springframework.stereotype.Service;

import com.fincore.JournalService.Dto.BranchDto;
import com.fincore.JournalService.Models.BranchMaster;
import com.fincore.JournalService.Repository.BranchMasterRepository;

import java.util.List;
import java.util.Map;
import java.util.stream.Collectors;

@Service
@RequiredArgsConstructor
@Slf4j
public class BranchService {

    private final BranchMasterRepository branchMasterRepository;

    public Map<String, String> getBranchCodeAndNameMap() {
        // This method is likely cached or used rarely, leaving as is
        List<BranchMaster> branches = branchMasterRepository.findAll();
        return branches.stream()
                .collect(Collectors.toMap(BranchMaster::getCode, BranchMaster::getName));
    }

    public List<BranchDto> searchBranches(String query) {
        log.info("Searching branches for: '{}'", query);

        Pageable limit = PageRequest.of(0, 50);


        List<BranchMaster> results = branchMasterRepository
                .findByNameContainingIgnoreCaseOrCodeContainingIgnoreCase(query, query, limit);

        return results.stream()
                .map(branch -> new BranchDto(branch.getCode(), branch.getName()))
                .collect(Collectors.toList());
    }
}


package com.fincore.JournalService.Service;

import lombok.RequiredArgsConstructor;
import lombok.extern.slf4j.Slf4j;
import org.springframework.data.domain.PageRequest;
import org.springframework.data.domain.Pageable;
import org.springframework.stereotype.Service;
import org.springframework.util.StringUtils;

import com.fincore.JournalService.Dto.CglDto;
import com.fincore.JournalService.Models.CglMaster;
import com.fincore.JournalService.Repository.CglMasterRepository;

import java.util.List;
import java.util.stream.Collectors;

@Service
@RequiredArgsConstructor
@Slf4j
public class CglService {

    private final CglMasterRepository cglMasterRepository;

    private CglDto convertToDto(CglMaster cgl) {
        return new CglDto(cgl.getCglNumber(), cgl.getDescription());
    }

    public List<CglDto> searchCgls(String query) {



        if (!StringUtils.hasText(query)) {
            return List.of();
        }

        log.info("Searching CGLs for: '{}'", query);

        Pageable limit = PageRequest.of(0, 50);

        List<CglMaster> results = cglMasterRepository
                .findByCglNumberContainingOrDescriptionContainingIgnoreCase(query, query, limit);

        return results.stream()
                .map(this::convertToDto)
                .collect(Collectors.toList());
    }
}

            return !formatErrors.isEmpty() || !dbErrors.isEmpty() || !balErrors.isEmpty();
        }

        String getAllErrors() {
            List<String> all = new ArrayList<>();
            all.addAll(formatErrors);
            all.addAll(dbErrors);
            all.addAll(balErrors);
            return String.join("; ", all);
        }
    }

    public String initiateValidation(MultipartFile file, LocalDate postingDate) throws IOException {
        String requestId = UUID.randomUUID().toString();
        BulkUploadStateDto state = new BulkUploadStateDto();
        state.setRequestId(requestId);
        state.setStatus("PROCESSING");
        state.setCurrentStage(1);
        state.setMessage("Initializing Upload...");
        state.setTotalRows(0);
        saveStateToRedis(state);

        byte[] fileBytes = file.getBytes();
        String filename = file.getOriginalFilename();
        self.processAsync(requestId, fileBytes, filename, postingDate);
        return requestId;
    }

    @Async("bulkExecutor")
    public void processAsync(String requestId, byte[] fileBytes, String filename, LocalDate postingDate) {
        log.info("Starting Async Validation for ReqID: {}", requestId);
        try {
            updateState(requestId, s -> s.setMessage("Parsing File..."));

            List<ExcelRowData> parsedRows;
            boolean isCsv = filename != null && (filename.toLowerCase().endsWith(".csv") || filename.toLowerCase().endsWith(".txt"));

            if (isCsv) {
                parsedRows = parseCsvBytes(fileBytes);
            } else {
                parsedRows = parseExcelBytes(fileBytes);
            }

            updateState(requestId, s -> {
                s.setTotalRows(parsedRows.size());
                s.setMessage("Validating Formats...");
            });

            if (runFormatCheck(parsedRows)) {
                failRequest(requestId, parsedRows, "Format Validation Failed", 1);
                return;
            }
            updateState(requestId, s -> { s.setCurrentStage(2); s.setMessage("Checking Database..."); });

            if (runDbCheck(parsedRows)) {
                failRequest(requestId, parsedRows, "Database Validation Failed", 2);
                return;
            }
            updateState(requestId, s -> { s.setCurrentStage(3); s.setMessage("Checking Balances..."); });

            if (runBalanceCheck(parsedRows)) {
                failRequest(requestId, parsedRows, "Debit/Credit Balance Mismatch", 3);
                return;
            }

            completeRequest(requestId, parsedRows, postingDate, isCsv);

        } catch (Exception e) {
            log.error("Async Validation Error for {}", requestId, e);
            updateState(requestId, s -> {
                s.setStatus("ERROR");
                s.setMessage("System Error: " + e.getMessage());
                s.setHasErrorFile(false);
            });
        }
    }

    // --- PARSING LOGIC ---

    private List<ExcelRowData> parseCsvBytes(byte[] bytes) throws IOException {
        List<ExcelRowData> list = new ArrayList<>();
        try(BufferedReader br = new BufferedReader(new InputStreamReader(new ByteArrayInputStream(bytes), StandardCharsets.UTF_8))) {
            String line; int i=0;
            while((line=br.readLine())!=null) {
                if(line.trim().isEmpty()) continue;
                String[] c = line.split(",", -1);

                // HEADER SKIP
                if (i == 0) {
                    String headerCheck = c.length > 0 ? c[0].trim().toLowerCase() : "";
                    // Skip if header row (Branch/Batch) BUT NOT if it's data "01"
                    if ((headerCheck.contains("branch") || headerCheck.contains("batch")) && !headerCheck.equals("01")) {
                        i++; continue;
                    }
                }

                ExcelRowData d = new ExcelRowData();
                d.rowIndex = i++;

                if (c.length >= 14) {
                    // --- SYSTEM FORMAT (14 Cols) ---
                    d.isSystemFormat = true;
                    validateSystemColumns(d, c[0], c[1], c[2], c[3], c[11], c[12], c[13]);

                    d.branch = c[4].trim();
                    d.currency = c[5].trim();
                    d.cgl = c[6].trim();
                    parseAmount(d, c[7].trim(), c[8].trim());
                    d.remarks = c[9].trim();

                    String rawProd = c[10].trim();
                    d.productCode = rawProd.isEmpty() ? "A" : rawProd;

                } else if (c.length >= 7) {
                    // --- TEMPLATE FORMAT (7 Cols) ---
                    if (c[0].trim().equals("01")) {
                        d.formatErrors.add("Invalid System File Format: Missing Columns (Expected 14, Found " + c.length + ")");
                    } else {
                        d.branch = c[0].trim();
                        d.currency = c[1].trim();
                        d.cgl = c[2].trim();
                        parseAmount(d, c[3].trim(), c[4].trim());
                        d.remarks = c[5].trim();
                        String rawProd = c.length > 6 ? c[6].trim() : "";
                        d.productCode = rawProd.isEmpty() ? "A" : rawProd;
                    }
                } else {
                    d.formatErrors.add("Invalid CSV Format: Missing columns or delimiters");
                }
                list.add(d);
            }
        }
        return list;
    }

    private List<ExcelRowData> parseExcelBytes(byte[] bytes) throws IOException {
        try(Workbook wb = new XSSFWorkbook(new ByteArrayInputStream(bytes))) {
            return parseSheet(wb.getSheetAt(0));
        }
    }

    private List<ExcelRowData> parseSheet(Sheet sheet) {
        List<ExcelRowData> list = new ArrayList<>();
        DataFormatter fmt = new DataFormatter();
        for(Row r : sheet) {
            if(isRowEmpty(r)) continue;

            // Header Check
            if (r.getRowNum() == 0) {
                String h = fmt.formatCellValue(r.getCell(0)).toLowerCase();
                // If header text found, skip. If "01" (data), process it.
                if ((h.contains("branch") || h.contains("batch")) && !h.equals("01")) continue;
            }

            ExcelRowData d = new ExcelRowData();
            d.rowIndex = r.getRowNum();

            String col0 = parseCode(r.getCell(0), fmt);

            if (col0.equals("01")) {
                // SYSTEM FORMAT
                d.isSystemFormat = true;
                String c1 = parseCode(r.getCell(1), fmt);
                String c2 = parseCode(r.getCell(2), fmt);
                String c3 = parseCode(r.getCell(3), fmt);
                String c11 = parseCode(r.getCell(11), fmt);
                String c12 = parseCode(r.getCell(12), fmt);
                String c13 = parseCode(r.getCell(13), fmt);

                validateSystemColumns(d, col0, c1, c2, c3, c11, c12, c13);

                d.branch = parseCode(r.getCell(4), fmt);
                d.currency = fmt.formatCellValue(r.getCell(5)).trim().toUpperCase();
                d.cgl = parseCode(r.getCell(6), fmt);
                parseAmount(d, fmt.formatCellValue(r.getCell(7)), fmt.formatCellValue(r.getCell(8)));
                d.remarks = fmt.formatCellValue(r.getCell(9));

                String rawProd = parseCode(r.getCell(10), fmt);
                d.productCode = rawProd.isEmpty() ? "A" : rawProd;

            } else {
                // TEMPLATE FORMAT
                d.branch = col0;
                d.currency = fmt.formatCellValue(r.getCell(1)).trim().toUpperCase();
                d.cgl = parseCode(r.getCell(2), fmt);
                parseAmount(d, fmt.formatCellValue(r.getCell(3)), fmt.formatCellValue(r.getCell(4)));
                d.remarks = fmt.formatCellValue(r.getCell(5));
                String rawProd = parseCode(r.getCell(6), fmt);
                d.productCode = rawProd.isEmpty() ? "A" : rawProd;
            }
            list.add(d);
        }
        return list;
    }

    private void validateSystemColumns(ExcelRowData d, String c0, String cDate, String cYear, String cMonth, String c11, String c12, String c13) {
        // 1. Batch ID Check
        if (!"01".equals(c0)) {
            d.formatErrors.add("Batch ID (Col 1) must be '01'");
        }

        // 2. Date Validation (ddMMyyyy)
        if (cDate == null || cDate.trim().length() != 8) {
            d.formatErrors.add("Invalid Date Format (Expected ddMMyyyy)");
        } else {
            try {
                LocalDate date = LocalDate.parse(cDate.trim(), SYSTEM_DATE_FMT);

                // 3. Financial Year Validation
                int month = date.getMonthValue();
                int year = date.getYear();
                String expectedYear = (month >= 4) ? String.valueOf(year) : String.valueOf(year - 1);
                int fm = (month >= 4) ? (month - 3) : (month + 9);
                String expectedMonth = String.format("%02d", fm);

                if (!expectedYear.equals(cYear.trim())) d.formatErrors.add("Financial Year mismatch");
                if (!expectedMonth.equals(cMonth.trim())) d.formatErrors.add("Financial Month mismatch");

            } catch (DateTimeParseException e) {
                d.formatErrors.add("Invalid Date Value (Expected ddMMyyyy)");
            }
        }

        // 4. Fixed Columns Check
        if (!"B".equalsIgnoreCase(c11.trim())) d.formatErrors.add("Col 12 must be 'B'");
        if (!"C".equalsIgnoreCase(c12.trim())) d.formatErrors.add("Col 13 must be 'C'");
        if (!"D".equalsIgnoreCase(c13.trim())) d.formatErrors.add("Col 14 must be 'D'");
    }

    private void parseAmount(ExcelRowData d, String amountRaw, String typeRaw) {
        try {
            if (amountRaw == null) amountRaw = "";
            if (typeRaw == null) typeRaw = "";
            String clean = CLEAN_AMOUNT_REGEX.matcher(amountRaw).replaceAll("");
            if (clean.isEmpty()) { d.amount = BigDecimal.ZERO; return; }
            BigDecimal v = new BigDecimal(clean);
            boolean isCredit = typeRaw.toUpperCase().startsWith("C") || typeRaw.toUpperCase().contains("CR") || amountRaw.toUpperCase().contains("CR");
            d.amount = isCredit ? v.abs().negate() : v.abs();
            d.txnType = isCredit ? "Credit" : "Debit";
        } catch(Exception e) { d.amount = BigDecimal.ZERO; }
    }

    // --- CHECKS ---
    private boolean runFormatCheck(List<ExcelRowData> rows) {
        rows.parallelStream().forEach(d -> {
            if (d.branch == null || d.branch.isEmpty()) d.formatErrors.add("Branch is Empty");
            if (d.currency == null || d.currency.length() != 3) d.formatErrors.add("Invalid Currency");

            if (d.cgl == null || d.cgl.isEmpty()) d.formatErrors.add("CGL is Empty");
            else if (!CGL_FORMAT_REGEX.matcher(d.cgl).matches()) d.formatErrors.add("Invalid CGL Format");

            if (d.amount == null || d.amount.compareTo(BigDecimal.ZERO) == 0) d.formatErrors.add("Amount cannot be Zero");

            // --- UPDATED PRODUCT CODE CHECK ---
            // Allow "A" OR 8 digits. "A" is default.
            if (d.productCode != null && !d.productCode.isEmpty()) {
                if (!d.productCode.equals("A") && !PRODUCT_CODE_REGEX.matcher(d.productCode).matches()) {
                    d.formatErrors.add("Product Code must be 'A' or 8 digits");
                }
            }

            // INR Check
            if (d.amount != null && "INR".equalsIgnoreCase(d.currency)) {
                if (d.amount.stripTrailingZeros().scale() > 2) d.formatErrors.add("INR Amount allows max 2 decimal places");
            }
        });
        return rows.stream().anyMatch(ExcelRowData::hasErrors);
    }

    private boolean runDbCheck(List<ExcelRowData> rows) {
        Set<String> branches = rows.parallelStream().map(r -> r.branch).collect(Collectors.toSet());
        Set<String> cgls = rows.parallelStream().map(r -> r.cgl).collect(Collectors.toSet());
        Set<String> currs = rows.parallelStream().map(r -> r.currency).collect(Collectors.toSet());

        Set<String> validBranches = branchRepo.findAllById(branches).stream().map(BranchMaster::getCode).collect(Collectors.toSet());
        Set<String> validCgls = cglRepo.findAllById(cgls).stream().map(CglMaster::getCglNumber).collect(Collectors.toSet());
        Set<String> validCurrs = currencyRepo.findAllById(currs).stream().map(CurrencyMaster::getCurrencyCode).collect(Collectors.toSet());

        rows.parallelStream().forEach(d -> {
            if (!validBranches.contains(d.branch)) d.dbErrors.add("Branch Not Found: " + d.branch);
            if (!validCurrs.contains(d.currency)) d.dbErrors.add("Currency Not Found: " + d.currency);
            if (!validCgls.contains(d.cgl)) d.dbErrors.add("CGL Not Found: " + d.cgl);
        });
        return rows.stream().anyMatch(ExcelRowData::hasErrors);
    }

    private boolean runBalanceCheck(List<ExcelRowData> rows) {
        ConcurrentHashMap<String, LongAdder> balanceMap = new ConcurrentHashMap<>();
        rows.parallelStream().forEach(d -> {
            if (d.amount != null) {
                String type = d.cgl.startsWith("5") ? "MEMO" : "NORMAL";
                String key = d.branch + "_" + d.currency + "_" + type;
                long val = d.amount.multiply(new BigDecimal("10000")).longValue();
                balanceMap.computeIfAbsent(key, k -> new LongAdder()).add(val);
            }
        });
        rows.parallelStream().forEach(d -> {
            String type = d.cgl.startsWith("5") ? "MEMO" : "NORMAL";
            String key = d.branch + "_" + d.currency + "_" + type;
            LongAdder adder = balanceMap.get(key);
            if (adder != null && adder.sum() != 0) {
                BigDecimal diff = BigDecimal.valueOf(adder.sum()).divide(new BigDecimal("10000"));
                synchronized(d.balErrors) { if (d.balErrors.isEmpty()) d.balErrors.add(type + " Balance Mismatch: " + diff.toPlainString()); }
            }
        });
        return rows.stream().anyMatch(ExcelRowData::hasErrors);
    }

    private void failRequest(String reqId, List<ExcelRowData> rows, String msg, int stage) throws IOException {
        byte[] excel = generateErrorExcelFast(rows);
        saveFileToRedis(reqId, "ERROR", excel);
        updateState(reqId, s -> { s.setStatus("ERROR"); s.setMessage(msg); s.setCurrentStage(stage); s.setErrorCount(rows.stream().filter(ExcelRowData::hasErrors).count()); s.setHasErrorFile(true); });
    }

    private void completeRequest(String reqId, List<ExcelRowData> rows, LocalDate pDate, boolean isCsv) throws IOException {
        if (!isCsv) {
            byte[] csv = generateSuccessCsv(rows, pDate);
            saveFileToRedis(reqId, "SUCCESS", csv);
        }
        List<Map<String, Object>> preview = rows.stream().limit(100).map(this::mapToPreview).collect(Collectors.toList());
        String previewJson = objectMapper.writeValueAsString(preview);
        updateState(reqId, s -> { s.setCurrentStage(4); s.setStatus("SUCCESS"); s.setMessage("Validation Successful"); s.setHasSuccessFile(!isCsv); s.setPreviewDataJson(previewJson); });
    }

    private byte[] generateSuccessCsv(List<ExcelRowData> rows, LocalDate postingDate) {
        StringBuilder csv = new StringBuilder(rows.size() * 100);
        DateTimeFormatter ddMMyyyy = DateTimeFormatter.ofPattern("ddMMyyyy");
        String col2 = postingDate.format(ddMMyyyy);
        int month = postingDate.getMonthValue();
        int year = postingDate.getYear();
        String col3 = (month >= 4) ? String.valueOf(year) : String.valueOf(year - 1);
        String col4 = (month >= 4) ? String.format("%02d", month - 3) : String.format("%02d", month + 9);

        // No Header
        for (ExcelRowData row : rows) {
            String col1 = "01";
            String col5 = String.format("%5s", row.branch).replace(' ', '0');
            String col6 = (row.currency == null || row.currency.isEmpty()) ? "INR" : row.currency;
            BigDecimal absAmt = row.amount.abs();
            String plainAmt = absAmt.setScale(4, RoundingMode.HALF_UP).toPlainString();
            int dotIndex = plainAmt.indexOf('.');
            String intPart = (dotIndex == -1) ? plainAmt : plainAmt.substring(0, dotIndex);
            String decPart = (dotIndex == -1) ? "0000" : plainAmt.substring(dotIndex + 1);
            String col8 = String.format("%16s", intPart).replace(' ', '0') + "." + decPart;
            String col9 = (row.amount.compareTo(BigDecimal.ZERO) < 0) ? "Cr" : "Dr";
            String col10 = (row.remarks != null) ? row.remarks.replace(",", " ") : "";
            String col11 = (row.productCode != null && !row.productCode.isEmpty()) ? row.productCode : "A";
            csv.append(col1).append(",").append(col2).append(",").append(col3).append(",").append(col4).append(",")
                    .append(col5).append(",").append(col6).append(",").append(row.cgl).append(",").append(col8).append(",")
                    .append(col9).append(",").append(col10).append(",").append(col11).append(",B,C,D\n");
        }
        return csv.toString().getBytes(StandardCharsets.UTF_8);
    }

    private byte[] generateErrorExcelFast(List<ExcelRowData> rows) throws IOException {
        try (SXSSFWorkbook workbook = new SXSSFWorkbook(100)) {
            Sheet sheet = workbook.createSheet("Error Report");
            CellStyle errStyle = workbook.createCellStyle();
            Font f = workbook.createFont(); f.setColor(IndexedColors.RED.getIndex()); errStyle.setFont(f);
            Row h = sheet.createRow(0);
            String[] hd = {"Branch", "Currency", "CGL", "Amount", "TxnType", "Remarks", "Product", "ERRORS"};
            for(int i=0; i<hd.length; i++) h.createCell(i).setCellValue(hd[i]);
            int idx = 1;
            for(ExcelRowData d : rows) {
                if (d.hasErrors()) {
                    Row r = sheet.createRow(idx++);
                    r.createCell(0).setCellValue(d.branch);
                    r.createCell(1).setCellValue(d.currency);
                    r.createCell(2).setCellValue(d.cgl);
                    r.createCell(3).setCellValue(d.amount != null ? d.amount.abs().toString() : "");
                    r.createCell(4).setCellValue(d.txnType);
                    r.createCell(5).setCellValue(d.remarks);
                    r.createCell(6).setCellValue(d.productCode);
                    Cell c = r.createCell(7); c.setCellValue(d.getAllErrors()); c.setCellStyle(errStyle);
                }
            }
            ByteArrayOutputStream out = new ByteArrayOutputStream();
            workbook.write(out); return out.toByteArray();
        }
    }

    private String parseCode(Cell c, DataFormatter f) {
        if(c == null) return "";
        if(c.getCellType() == CellType.NUMERIC) return BigDecimal.valueOf(c.getNumericCellValue()).toPlainString().split("\\.")[0];
        return f.formatCellValue(c).trim();
    }
    private boolean isRowEmpty(Row r) {
        if(r==null) return true;
        for(int c=r.getFirstCellNum(); c<r.getLastCellNum(); c++)
            if(r.getCell(c)!=null && r.getCell(c).getCellType()!=CellType.BLANK && !r.getCell(c).toString().trim().isEmpty()) return false;
        return true;
    }
    private Map<String, Object> mapToPreview(ExcelRowData r) {
        Map<String, Object> m = new HashMap<>();
        m.put("id", r.rowIndex); m.put("branch", r.branch); m.put("currency", r.currency);
        m.put("cgl", r.cgl); m.put("amount", r.amount!=null?r.amount.abs().toString():"");
        m.put("txnType", r.txnType); m.put("remarks", r.remarks); m.put("productCode", r.productCode);
        return m;
    }
    private void saveStateToRedis(BulkUploadStateDto state) { try { redisTemplate.opsForValue().set(KEY_META + state.getRequestId(), objectMapper.writeValueAsString(state), TTL_MINUTES, TimeUnit.MINUTES); } catch (Exception e) {} }
    private void updateState(String requestId, java.util.function.Consumer<BulkUploadStateDto> updater) { try { String json = redisTemplate.opsForValue().get(KEY_META + requestId); if (json != null) { BulkUploadStateDto state = objectMapper.readValue(json, BulkUploadStateDto.class); updater.accept(state); saveStateToRedis(state); } } catch (Exception e) {} }
    private void saveFileToRedis(String requestId, String type, byte[] data) { String key = (type.equals("ERROR") ? KEY_FILE_ERR : KEY_FILE_SUC) + requestId; String base64 = Base64.getEncoder().encodeToString(data); redisTemplate.opsForValue().set(key, base64, TTL_MINUTES, TimeUnit.MINUTES); }
    public BulkUploadStateDto getState(String reqId) { try { String json = redisTemplate.opsForValue().get(KEY_META + reqId); return json != null ? objectMapper.readValue(json, BulkUploadStateDto.class) : null; } catch (Exception e) { return null; } }
    public byte[] getFileBytes(String reqId, String type) { String key = (type.equals("ERROR") ? KEY_FILE_ERR : KEY_FILE_SUC) + reqId; String base64 = redisTemplate.opsForValue().get(key); return base64 != null ? Base64.getDecoder().decode(base64) : null; }
}

                .collect(Collectors.groupingBy(row -> row.getBranch() + ":" + row.getCurrency()));

        for (Map.Entry<String, List<BatchRequestDto.JournalRequestRow>> entry : groupedRows.entrySet()) {

        }

    }

    @Override
    @Transactional
    public Optional<JournalRequest> updateRequestStatus(ProcessJournalRequestDto dto, String executorId, Integer executorRole) throws JsonProcessingException {
        log.debug("Updating single request: {}", dto.getRequestId());
        JournalRequest request = journalRequestRepository.findById(dto.getRequestId())
                .orElseThrow(() -> new ResourceNotFoundException("Journal Request with ID " + dto.getRequestId() + " not found."));

        boolean isCreator = executorId.equals(request.getCreatorId());
        boolean isSameRole = executorRole != null && executorRole.equals(request.getCreatorRole());

        if (isCreator && isSameRole) {
            log.warn("SECURITY VIOLATION: User {} (Role: {}) attempted to approve/reject their own request (ID: {}).", 
                     executorId, executorRole, request.getId());
            throw new IllegalStateException("User cannot approve or reject their own request with the same role.");
        }

        if (request.getRequestStatus() != RequestStatus.PENDING) {
            throw new IllegalStateException("Journal Request has already been processed.");
        }

        request.setExecutorId(executorId);
        request.setExecutorRemarks(dto.getRemarks());
        request.setExecutionDate(LocalDateTime.now());
        request.setRequestStatus(dto.getStatus());

        if (RequestStatus.ACCEPTED.equals(dto.getStatus())) {
            log.info("Journal Request {} ACCEPTED. Processing for Batch ID: {}", dto.getRequestId(), request.getBatchId());
            processApproval(request); 
            sendCreatorNotification(request, true);
        } else {
            log.info("Journal Request {} REJECTED.", dto.getRequestId());
            sendCreatorNotification(request, false);
        }

        return Optional.of(journalRequestRepository.save(request));
    }
    
    private void processApproval(JournalRequest request) throws JsonProcessingException {
        CreateJournalRequestDto payloadDto = objectMapper.readValue(request.getPayload(), CreateJournalRequestDto.class);
        
        if (payloadDto.getChangeType() == ChangeType.ADD) {
       
            GlTransaction newGlTxn = new GlTransaction();
            newGlTxn.setBatchId(payloadDto.getBatchId());
            newGlTxn.setJournalId(payloadDto.getJournalId());
            
            
            if (payloadDto.getPDate() != null) {
                newGlTxn.setPostDate(payloadDto.getPDate().atStartOfDay());
                newGlTxn.setTransactionDate(payloadDto.getPDate()); // Set Date explicitly
            } else {
                newGlTxn.setPostDate(LocalDateTime.now());
                newGlTxn.setTransactionDate(LocalDate.now());
            }
            
            newGlTxn.setBranchCode(payloadDto.getBranch());
            newGlTxn.setCurrency(payloadDto.getCurrency());
            newGlTxn.setCgl(payloadDto.getCgl());
            newGlTxn.setNarration(payloadDto.getRemarks());
            
            BigDecimal amount = payloadDto.getAmount();
            if (amount.signum() < 0) { 
                newGlTxn.setCreditAmount(amount);
                newGlTxn.setDebitAmount(BigDecimal.ZERO);
            } else { 
                newGlTxn.setDebitAmount(amount);
                newGlTxn.setCreditAmount(BigDecimal.ZERO);
            }
            
            newGlTxn.setTransactionCount(payloadDto.getTransactionCount());
            newGlTxn.setSourceFlag("J"); 
            
            glTransactionRepository.save(newGlTxn);
        } 
    }

    @Override
    @Transactional
    public List<JournalRequest> processBulkRequests(BulkProcessJournalRequestDto dto, String executorId, Integer executorRole) throws JsonProcessingException {
        log.info("Starting processBulkRequests for user: {}", executorId);

        boolean hasBatchId = StringUtils.hasText(dto.getBatchId());
        boolean hasJournalPrefixes = dto.getJournalIdPrefixes() != null && !dto.getJournalIdPrefixes().isEmpty();

        if (!hasBatchId && !hasJournalPrefixes) {
            throw new IllegalArgumentException("Either batchId or a list of journalIdPrefixes must be provided.");
        }

        List<JournalRequest> requestsToProcess;
        if (hasBatchId) {
            requestsToProcess = journalRequestRepository.findByBatchIdAndRequestStatus(dto.getBatchId(), RequestStatus.PENDING);
        } else {
            Specification<JournalRequest> spec = (root, query, criteriaBuilder) -> {
                List<Predicate> predicates = new ArrayList<>();
                for (String prefix : dto.getJournalIdPrefixes()) {
                    predicates.add(criteriaBuilder.like(root.get("journalId"), prefix + "%"));
                }
                Predicate journalIdOrPredicates = criteriaBuilder.or(predicates.toArray(new Predicate[0]));
                Predicate statusPredicate = criteriaBuilder.equal(root.get("requestStatus"), RequestStatus.PENDING);
                return criteriaBuilder.and(journalIdOrPredicates, statusPredicate);
            };
            requestsToProcess = journalRequestRepository.findAll(spec);
        }

        if (requestsToProcess.isEmpty()) {
            return Collections.emptyList();
        }
        
        List<JournalRequest> processedRequests = new ArrayList<>();
        List<GlTransaction> transactionsToSave = new ArrayList<>();
        
        for (JournalRequest request : requestsToProcess) {
            boolean isCreator = executorId.equals(request.getCreatorId());
            boolean isSameRole = executorRole != null && executorRole.equals(request.getCreatorRole());

            if (isCreator && isSameRole) continue;
            if (request.getRequestStatus() != RequestStatus.PENDING) continue;

            request.setExecutorId(executorId);
            request.setExecutorRemarks(dto.getRemarks());
            request.setExecutionDate(LocalDateTime.now());
            request.setRequestStatus(dto.getStatus());
        
            if (RequestStatus.ACCEPTED.equals(dto.getStatus())) {
                CreateJournalRequestDto payloadDto = objectMapper.readValue(request.getPayload(), CreateJournalRequestDto.class);
                
                if (payloadDto.getChangeType() == ChangeType.ADD) {
                    GlTransaction newGlTxn = new GlTransaction();
                    newGlTxn.setBatchId(payloadDto.getBatchId());
                    newGlTxn.setJournalId(payloadDto.getJournalId());
                    
                    // UPDATED: Explicitly setting Transaction Date
                    if (payloadDto.getPDate() != null) {
                        newGlTxn.setPostDate(payloadDto.getPDate().atStartOfDay());
                        newGlTxn.setTransactionDate(payloadDto.getPDate());
                    } else {
                        newGlTxn.setPostDate(LocalDateTime.now());
                        newGlTxn.setTransactionDate(LocalDate.now());
                    }

                    newGlTxn.setBranchCode(payloadDto.getBranch());
                    newGlTxn.setCurrency(payloadDto.getCurrency());
                    newGlTxn.setCgl(payloadDto.getCgl());
                    newGlTxn.setNarration(payloadDto.getRemarks());
                    
                    BigDecimal amount = payloadDto.getAmount();
                    if (amount.signum() < 0) { 
                        newGlTxn.setCreditAmount(amount);
                        newGlTxn.setDebitAmount(BigDecimal.ZERO);
                    } else { 
                        newGlTxn.setDebitAmount(amount);
                        newGlTxn.setCreditAmount(BigDecimal.ZERO);
                    }
                    newGlTxn.setTransactionCount(payloadDto.getTransactionCount());
                    newGlTxn.setSourceFlag("J"); 
                    
                    transactionsToSave.add(newGlTxn);
                }
            }
            processedRequests.add(request);
        }
     
        if (!transactionsToSave.isEmpty()) {
            glTransactionRepository.saveAll(transactionsToSave);
        }
        
        List<JournalRequest> savedRequests = journalRequestRepository.saveAll(processedRequests);
        
        try {
            boolean isApproved = RequestStatus.ACCEPTED.equals(dto.getStatus());
            String status = isApproved ? "approved" : "rejected";
            String message;
            String aggregateId;

            if (hasBatchId) {
                aggregateId = dto.getBatchId();
                message = String.format("Your Journal batch (ID: %s) with %d entries has been %s by user %s.",
                                        aggregateId, savedRequests.size(), status, executorId);
            } else { 
                aggregateId = dto.getJournalIdPrefixes().get(0); 
                message = String.format("Your %d Journal requests (Journals: %s,...) have been %s by user %s.",
                                        savedRequests.size(), aggregateId, status, executorId);
            }

            if (!isApproved && StringUtils.hasText(dto.getRemarks())) {
                message += " Reason: " + dto.getRemarks();
            }

            String creatorId = requestsToProcess.get(0).getCreatorId();
            notificationWriterService.createNotification(
                creatorId,              
                null,                   
                message,
                JOURNAL_POSTING_URL,    
                aggregateId,            
                EVENT_SOURCE
            );
        } catch (Exception e) {
            log.error("Failed to send creator bulk notification", e);
        }
        
        return savedRequests;
    }

    @Override
    @Transactional
    public String createBulkBatchRequest(BatchRequestDto batchDto, String creatorId, Integer creatorRole) throws JsonProcessingException {
        
        String batchId = sequenceService.getNextBatchId();
        log.info("Processing BULK Batch ID: {}", batchId);
       
        LocalDate postingDate = LocalDate.now();
        List<JournalRequest> requestsToSave = new ArrayList<>();
        
      
        String commonRemarks = (batchDto.getCommonBatchRemarks() != null && !batchDto.getCommonBatchRemarks().isEmpty()) 
                                ? batchDto.getCommonBatchRemarks() 
                                : "Bulk Upload Batch " + batchId;

        // Grouping logic (Same as original, ensures Journal IDs are sequenced per Branch/Currency)
        Map<String, List<BatchRequestDto.JournalRequestRow>> groupedRows = batchDto.getRows().stream()
                .collect(Collectors.groupingBy(row -> row.getBranch() + ":" + row.getCurrency()));

        for (Map.Entry<String, List<BatchRequestDto.JournalRequestRow>> entry : groupedRows.entrySet()) {
            
            String journalIdPrefix = sequenceService.getNextJournalIdPrefix();
            int journalSequence = 1;
            List<BatchRequestDto.JournalRequestRow> rowsInGroup = entry.getValue();
            
            for (BatchRequestDto.JournalRequestRow rowDto : rowsInGroup) {
                
                String journalId = String.format("%s-%03d", journalIdPrefix, journalSequence);
                
                CreateJournalRequestDto payloadDto = new CreateJournalRequestDto();
                payloadDto.setPDate(postingDate); 
                payloadDto.setChangeType(rowDto.getChangeType());
                payloadDto.setMasterJournalId(rowDto.getMasterJournalId());
                payloadDto.setBranch(rowDto.getBranch());
                payloadDto.setCurrency(rowDto.getCurrency());
                payloadDto.setCgl(rowDto.getCgl());
                payloadDto.setAmount(rowDto.getAmount()); 
                payloadDto.setProductType(rowDto.getProductType());
                payloadDto.setRemarks(rowDto.getRemarks());
     
                payloadDto.setArFlag(rowDto.getArFlag() != null ? rowDto.getArFlag() : "A");
                payloadDto.setAcClassification(rowDto.getAcClassification() != null ? rowDto.getAcClassification() : "A");
                
                payloadDto.setBatchId(batchId); 
                payloadDto.setJournalId(journalId); 
                payloadDto.setCommonBatchRemarks(commonRemarks);
                payloadDto.setTransactionCount(journalSequence); 
                
                JournalRequest request = new JournalRequest();
                request.setCreatorId(creatorId);
                request.setCreatorRole(creatorRole);
                request.setChangeType(rowDto.getChangeType()); // Default to ADD
                request.setRequestStatus(RequestStatus.PENDING);
                request.setBatchId(batchId);
                request.setJournalId(journalId);
                request.setCommonBatchRemarks(commonRemarks);
                request.setPayload(objectMapper.writeValueAsString(payloadDto));

                requestsToSave.add(request);
                journalSequence++; 
            }
        }

      
        journalRequestRepository.saveAll(requestsToSave);
        
     
        try {
            NotificationConfigDto config = permissionConfigService.getConfig("JOURNAL_AUTH"); 
            String message = String.format("Bulk Journal Batch %s (%d rows) by %s pending approval.", 
                                           batchId, requestsToSave.size(), creatorId);

            notificationWriterService.createNotification(
                null,                   
                config.getTargetRoles(),
                message,
                config.getTargetUrl(),  
                batchId,                
                "JournalService"
            );
        } catch (Exception e) {
            log.error("Failed to create notification for Bulk Batch: {}", batchId, e);
            
        }
   
        return batchId; // Return ID only
    }

import lombok.RequiredArgsConstructor;
import lombok.extern.slf4j.Slf4j;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Propagation;
import org.springframework.transaction.annotation.Transactional;

import com.fincore.JournalService.Models.Notifications;
import com.fincore.JournalService.Repository.NotificationRepository;

@Service
@RequiredArgsConstructor
@Slf4j
public class NotificationWriterService {

    private final NotificationRepository notificationRepository;

    @Transactional(propagation = Propagation.MANDATORY) 
    public void createNotification(String userId, String targetRole, String message, String linkUrl, String aggregateId, String eventSource) {

        if (userId == null && targetRole == null) {
            log.warn("Skipping notification creation: Both userId and targetRole are null. AggregateID: {}", aggregateId);
            return;
        }

        if (message == null || message.isBlank()) {
            log.warn("Skipping notification creation: Message is null or blank. AggregateID: {}", aggregateId);
            throw new IllegalArgumentException("Notification message cannot be null or blank.");
        }

        Notifications notification = Notifications.builder()
                .userId(userId)
                .targetRole(targetRole)
                .message(message)
                .linkUrl(linkUrl)
                .aggregateId(aggregateId)
                .eventSource(eventSource)
                .build();

        notificationRepository.save(notification);

        if (userId != null) {
            log.info("Saved 1-to-1 notification event for user: {} (AggregateID: {})", userId, aggregateId);
        } else {
            log.info("Saved 1-to-many notification event for role: {} (AggregateID: {})", targetRole, aggregateId);
        }
    }
}

import lombok.RequiredArgsConstructor;
import lombok.extern.slf4j.Slf4j;
import org.springframework.cache.annotation.Cacheable;
import org.springframework.stereotype.Service;

import com.fincore.JournalService.Dto.NotificationConfigDto;
import com.fincore.JournalService.Repository.PermissionRepository;

import java.util.List;
import java.util.stream.Collectors;

@Service
@RequiredArgsConstructor
@Slf4j
public class PermissionConfigService {

    private final PermissionRepository permissionRepository;

    @Cacheable(value = "notification_configs", key = "#requestType")
    public NotificationConfigDto getConfig(String requestType) {

        log.info("Cache Miss: Fetching DB permissions for MOC notification type: {}", requestType);

        List<Object[]> results = permissionRepository.findUrlAndRolesByRequestType(requestType);

        if (results.isEmpty()) {
            log.warn("No notification config found for request type {}. Using defaults.", requestType);
            return new NotificationConfigDto("/dashboard", null); // Default to no roles
        }

        String url = (String) results.get(0)[0];
        if (url == null) url = "/dashboard";

        String roles = results.stream()
                .map(row -> String.valueOf(row[1])) // Role ID
                .distinct()
                .collect(Collectors.joining(","));

        if (roles.isEmpty()) {
             log.warn("No roles found for request type {}. Notification will not be sent to a group.", requestType);
             return new NotificationConfigDto(url, null);
        }
        
        log.info("Fetched Notification Config for {}: URL=[{}], Roles=[{}]", requestType, url, roles);
        return new NotificationConfigDto(url, roles);
    }
}

import jakarta.persistence.EntityManager;
import jakarta.persistence.PersistenceContext;
import lombok.RequiredArgsConstructor;
import org.springframework.stereotype.Service;
import jakarta.transaction.Transactional;

@Service
@RequiredArgsConstructor
public class SequenceService {

    @PersistenceContext
    private EntityManager entityManager;

    /**
     * Pads the given number with leading zeros to a total length of 7.
     */
    private String padToSevenDigits(Long id) {
        if (id == null) {
            throw new IllegalStateException("Sequence query returned null");
        }
        return String.format("%07d", id);
    }

    /**
     * Gets the next BATCH_ID from the BATCH_SEQ sequence. 
     */
    @Transactional
    public String getNextBatchId() {
        Number nextId = (Number) entityManager
                .createNativeQuery("select BATCH_SEQ.nextval from dual")
                .getSingleResult();
        
        return padToSevenDigits(nextId.longValue());
    }

    /**
     * Gets the next JOURNAL_ID prefix from the JRNL_SEQ sequence.
     */
    @Transactional
    public String getNextJournalIdPrefix() {
        
      
        Number nextId = (Number) entityManager
                .createNativeQuery("select JRNL_SEQ.nextval from dual")
                .getSingleResult();
      

        return padToSevenDigits(nextId.longValue());
    }
}

  const [viewingBatchId, setViewingBatchId] = useState(null);
  const [viewingBatchCreatorId, setViewingBatchCreatorId] = useState(null);

  const [selectedJournalPrefixes, setSelectedJournalPrefixes] = useState([]);
  const [partialExecutorRemarks, setPartialExecutorRemarks] = useState("");
  const [isPartialSubmitting, setIsPartialSubmitting] = useState(false);

  useEffect(() => {
    let userIdFound = false;
    try {
      const authDataString =
        sessionStorage.getItem("persist:auth") ||
        localStorage.getItem("persist:auth");

      if (authDataString) {
        const authData = JSON.parse(authDataString);
        const userDataString = authData.user;

        if (userDataString) {
          const userData = JSON.parse(userDataString);
          if (userData && userData.userId) {
            setCurrentUserId(String(userData.userId));
            userIdFound = true;
          }
        }
      }
    } catch (error) {
      console.error("Failed to parse auth data from storage:", error);
      showSnackBar("Error verifying user. Please refresh the page.", "error");
    }

    if (!userIdFound) {
      console.warn(
        "Could not find 'persist:auth' data in sessionStorage or localStorage."
      );
      showSnackBar(
        "Could not verify user. Action buttons may not work.",
        "warning"
      );
    }
  }, [showSnackBar]);

  const fetchPendingBatches = useCallback(async () => {
    setLoading(true);
    try {
      const pendingRequests = await callApi(
        "/JS/journals/pending-requests",
        null,
        "GET"
      );

      if (!Array.isArray(pendingRequests) || pendingRequests.length === 0) {
        setRows([]);
        setLoading(false);
        return;
      }

      const groupedBatches = new Map();
      for (const request of pendingRequests) {
        const batchId = request.batchId;

        if (!groupedBatches.has(batchId)) {
          groupedBatches.set(batchId, {
            id: batchId,
            batchId: batchId,
            creatorId: request.creatorId,
            requestStatus: request.requestStatus,
            requestDate: request.requestDate,
            commonBatchRemarks: request.commonBatchRemarks,
            totalDebit: 0,
            totalCredit: 0,
            requestCount: 0,
          });
        }

        const batchDetails = groupedBatches.get(batchId);
        batchDetails.requestCount++;

        try {
          const payload = JSON.parse(request.payload);
          const amount = parseFloat(payload.amount) || 0;

          if (amount < 0) {
            batchDetails.totalCredit += Math.abs(amount);
          } else {
            batchDetails.totalDebit += amount;
          }
        } catch (e) {
          console.error(`Failed to parse payload for request ${request.id}`, e);
        }
      }
      setRows(Array.from(groupedBatches.values()));
    } catch (err) {
      showSnackBar("Failed to load pending requests", "error");
    } finally {
      setLoading(false);
    }
  }, [callApi, showSnackBar]);

  useEffect(() => {
    fetchPendingBatches();
  }, [fetchPendingBatches]);

  const handleOpenModal = (batch, mode) => {
    setSelectedBatch(batch);
    setModalMode(mode);
    setExecutorRemarks("");
    setIsModalOpen(true);
  };

  const handleCloseModal = () => {
    if (isSubmitting) return;
    setIsModalOpen(false);
    setSelectedBatch(null);
  };

  const handleOpenViewModal = async (batchId, creatorId) => {
    setIsViewModalOpen(true);
    setIsViewLoading(true);
    setBatchDetails([]);
    setViewingBatchId(batchId);
    setViewingBatchCreatorId(String(creatorId));

    try {
      const data = await callApi(
        `/JS/journals/by-batch/${batchId}`,
        null,
        "GET"
      );

      setTimeout(() => {
        if (Array.isArray(data)) {
          const journalGroups = new Map();

          const processedData = data
            .map((req) => {
              if (req.requestStatus !== "PENDING") {
                return null;
              }
              try {
                const payload = JSON.parse(req.payload);
                const prefix = req.journalId.split("-")[0];
                const amount = parseFloat(payload.amount) || 0;

                return {
                  ...req,
                  payload,
                  journalPrefix: prefix,
                  amount: Math.abs(amount),
                  transactionType: amount < 0 ? "Credit" : "Debit",
                };
              } catch (e) {
                console.error("Failed to parse payload for req:", req.id, e);
                return null;
              }
            })
            .filter(Boolean);

          processedData.sort((a, b) => a.journalId.localeCompare(b.journalId));

          for (const row of processedData) {
            const prefix = row.journalPrefix;
            if (!journalGroups.has(prefix)) {
              journalGroups.set(prefix, []);
            }
            journalGroups.get(prefix).push(row);
          }

          setBatchDetails(Array.from(journalGroups.values()));
        } else {
          setBatchDetails([]);
          showSnackBar("No details found for this batch.", "info");
        }
        setIsViewLoading(false);
      }, 100);
    } catch (error) {
      showSnackBar("Failed to load batch details", "error");
      setBatchDetails([]);
      setIsViewLoading(false);
    }
  };

  const handleCloseViewModal = () => {
    setIsViewModalOpen(false);
    setBatchDetails([]);
    setViewingBatchId(null);
    setViewingBatchCreatorId(null);
    setSelectedJournalPrefixes([]);
    setPartialExecutorRemarks("");
    setIsPartialSubmitting(false);
  };

  const handleSubmitAction = async () => {
    if (modalMode === "REJECTED" && !executorRemarks.trim()) {
      showSnackBar("Approver remarks are required for rejection.", "warning");
      return;
    }

    setIsSubmitting(true);
    const payload = {
      batchId: selectedBatch.batchId,
      status: modalMode,
      remarks:
        executorRemarks.trim() ||
        (modalMode === "ACCEPTED" ? "Accepted" : "Rejected"),
    };

    try {
      await callApi("/JS/journals/process-bulk", payload, "POST");
      handleCloseModal();
      fetchPendingBatches();
    } catch (error) {
      console.error("Failed to process batch:", error);
      showSnackBar("Failed to process batch", "error");
    } finally {
      setIsSubmitting(false);
    }
  };

  const handleSubmitPartialAction = async (mode) => {
    if (mode === "REJECTED" && !partialExecutorRemarks.trim()) {
      showSnackBar("Approver remarks are required for rejection.", "warning");
      return;
    }

    if (selectedJournalPrefixes.length === 0) {
      showSnackBar("Please select at least one entry to process.", "warning");
      return;
    }

    setIsPartialSubmitting(true);

    const payload = {
      journalIdPrefixes: selectedJournalPrefixes,
      status: mode,
      remarks:
        partialExecutorRemarks.trim() ||
        (mode === "ACCEPTED" ? "Accepted" : "Rejected"),
    };

    try {
      await callApi("/JS/journals/process-bulk", payload, "POST");
      handleCloseViewModal();
      fetchPendingBatches();
    } catch (error) {
      console.error("Failed to process partial batch:", error);
      showSnackBar("Failed to process partial selection", "error");
    } finally {
      setIsPartialSubmitting(false);
    }
  };

  const handleSelectAllClick = (event) => {
    if (event.target.checked) {
      const newSelecteds = batchDetails.map((group) => group[0].journalPrefix);
      setSelectedJournalPrefixes(newSelecteds);
      return;
    }
    setSelectedJournalPrefixes([]);
  };

  const handleRowCheckboxClick = (event, prefix) => {
    if (event) {
      event.stopPropagation();
    }

    const selectedIndex = selectedJournalPrefixes.indexOf(prefix);
    let newSelected = [];

    if (selectedIndex === -1) {
      newSelected = newSelected.concat(selectedJournalPrefixes, prefix);
    } else if (selectedIndex === 0) {
      newSelected = newSelected.concat(selectedJournalPrefixes.slice(1));
    } else if (selectedIndex === selectedJournalPrefixes.length - 1) {
      newSelected = newSelected.concat(selectedJournalPrefixes.slice(0, -1));
    } else if (selectedIndex > 0) {
      newSelected = newSelected.concat(
        selectedJournalPrefixes.slice(0, selectedIndex),
        selectedJournalPrefixes.slice(selectedIndex + 1)
      );
    }
    setSelectedJournalPrefixes(newSelected);
  };

  const columns = [
    {
      field: "batchId",
      headerName: "Batch ID",
      width: 125,
      align: "center",
      headerAlign: "center",
    },
    {
      field: "requestStatus",
      headerName: "Status",
      width: 125,
      align: "center",
      headerAlign: "center",
      renderCell: (params) => (
        <Chip label={params.value} color="warning" size="small" />
      ),
    },
    {
      field: "creatorId",
      headerName: "Creator",
      width: 150,
      align: "center",
      headerAlign: "center",
      renderCell: (params) => (
        <Chip icon={<AccountCircleIcon />} label={params.value} size="small" />
      ),
    },
    {
      field: "requestDate",
      headerName: "Submitted On",
      width: 200,
      align: "center",
      headerAlign: "center",
      renderCell: (params) => (
        <Box>
          {new Date(params.value).toLocaleString("en-IN", {
            dateStyle: "medium",
            timeStyle: "short",
          })}
        </Box>
      ),
    },
    {
      field: "commonBatchRemarks",
      headerName: "Creator Remarks",
      width: 200,
      align: "center",
      headerAlign: "center",
    },
    {
      field: "requestCount",
      headerName: "Entries",
      width: 100,
      align: "center",
      headerAlign: "center",
    },
    {
      field: "totalCredit",
      headerName: "Total Credit",
      width: 150,
      align: "right",
      headerAlign: "right",
      renderCell: (params) => (
        <Typography variant="body2" sx={{ mt: 1.5 }}>
          {formatCurrency(params.value)}
        </Typography>
      ),
    },
    {
      field: "totalDebit",
      headerName: "Total Debit",
      width: 150,
      align: "right",
      headerAlign: "right",
      renderCell: (params) => (
        <Typography variant="body2" sx={{ mt: 1.5 }}>
          {formatCurrency(params.value)}
        </Typography>
      ),
    },
    {
      field: "actions",
      headerName: "Actions",
      flex: 0.8,
      minWidth: 130,
      align: "center",
      headerAlign: "center",
      sortable: false,
      renderCell: (params) => {
        const isSelfCreated =
          currentUserId &&
          params.row.creatorId &&
          String(currentUserId) === String(params.row.creatorId);

        const selfCreateTooltip =
          "You cannot approve or reject your own request";

        return (
          <Stack direction="row" spacing={0.5} justifyContent="center">
            <Tooltip title="View Details">
              <IconButton
                onClick={() =>
                  handleOpenViewModal(params.row.batchId, params.row.creatorId)
                }
                color="primary"
              >
                <VisibilityIcon />
              </IconButton>
            </Tooltip>
            <Tooltip title={isSelfCreated ? selfCreateTooltip : "Accept Batch"}>
              <span>
                <IconButton
                  onClick={() => handleOpenModal(params.row, "ACCEPTED")}
                  color="success"
                  disabled={isSelfCreated}
                >
                  <CheckIcon />
                </IconButton>
              </span>
            </Tooltip>
            <Tooltip title={isSelfCreated ? selfCreateTooltip : "Reject Batch"}>
              <span>
                <IconButton
                  onClick={() => handleOpenModal(params.row, "REJECTED")}
                  color="error"
                  disabled={isSelfCreated}
                >
                  <ClearIcon />
                </IconButton>
              </span>
            </Tooltip>
          </Stack>
        );
      },
    },
  ];

  const isViewingSelfCreated =
    currentUserId &&
    viewingBatchCreatorId &&
    String(currentUserId) === String(viewingBatchCreatorId);

  return (
    <Box sx={mainContainerStyle}>
      <DataGrid
        rows={rows}
        columns={columns}
        loading={loading}
        getRowId={(row) => row.batchId}
        disableRowSelectionOnClick
        pageSizeOptions={[5, 10, 25]}
        disableColumnResize={true}
        rowHeight={52}
        initialState={{
          pagination: {
            paginationModel: { page: 0, pageSize: 10 },
          },
          sorting: { sortModel: [{ field: "requestDate", sort: "desc" }] },
        }}
        slots={{
          noRowsOverlay: CustomNoRowsOverlay,
        }}
        sx={dataGridContainerStyle}
      />

      {/* --- Batch Action Modal --- */}
      <Dialog
        open={isModalOpen}
        onClose={handleCloseModal}
        fullWidth
        maxWidth="sm"
      >
        <DialogTitle>
          {modalMode === "ACCEPTED" ? "Approve" : "Reject"} Batch
        </DialogTitle>
        <DialogContent>
          <DialogContentText>
            Are you sure you want to {modalMode.toLowerCase()} the entire batch{" "}
            <strong>{selectedBatch?.batchId}</strong>?
            {modalMode === "REJECTED" && " Remarks are required for rejection."}
          </DialogContentText>

          <TextField
            autoFocus={modalMode === "REJECTED"}
            required={modalMode === "REJECTED"}
            margin="dense"
            id="executorRemarks"
            label="Approver Remarks"
            type="text"
            fullWidth
            variant="standard"
            value={executorRemarks}
            onChange={(e) => setExecutorRemarks(e.target.value)}
            disabled={isSubmitting}
            InputProps={{
              inputProps: {
                maxLength: 200,
              },
            }}
          />
        </DialogContent>
        <DialogActions sx={dialogActionsStyle}>
          <Stack direction="row" spacing={1} justifyContent="flex-end">
            <Button onClick={handleCloseModal} disabled={isSubmitting}>
              Cancel
            </Button>
            <Button
              onClick={handleSubmitAction}
              variant="contained"
              color={modalMode === "ACCEPTED" ? "success" : "error"}
              disabled={isSubmitting}
            >
              {isSubmitting ? (
                <CircularProgress size={24} color="inherit" />
              ) : (
                `Confirm ${modalMode === "ACCEPTED" ? "Approval" : "Rejection"}`
              )}
            </Button>
          </Stack>
        </DialogActions>
      </Dialog>

      {/* --- View Details Modal --- */}
      <Dialog
        open={isViewModalOpen}
        onClose={handleCloseViewModal}
        fullWidth
        maxWidth="lg"
      >
        <DialogTitle sx={dialogTitleStyle}>
          Batch Details: {viewingBatchId}
          <IconButton
            aria-label="close"
            onClick={handleCloseViewModal}
            sx={{
              color: (theme) => theme.palette.grey[500],
            }}
          >
            <CloseIcon />
          </IconButton>
        </DialogTitle>

        <DialogContent dividers sx={{ p: 0 }}>
          {isViewLoading ? (
            <Box sx={loadingContainerStyle}>
              <CircularProgress />
            </Box>
          ) : (
            <>
              {batchDetails.length > 0 ? (
                <TableContainer component={Paper} sx={tableContainerStyle}>
                  <Table sx={{ minWidth: 650 }} size="small" stickyHeader>
                    <TableHead>
                      <TableRow>
                        <TableCell padding="checkbox" sx={tableHeaderCellStyle}>
                          <Checkbox
                            color="primary"
                            indeterminate={
                              selectedJournalPrefixes.length > 0 &&
                              selectedJournalPrefixes.length <
                                batchDetails.length
                            }
                            checked={
                              batchDetails.length > 0 &&
                              selectedJournalPrefixes.length ===
                                batchDetails.length
                            }
                            onChange={handleSelectAllClick}
                            disabled={isViewingSelfCreated}
                          />
                        </TableCell>
                        <TableCell align="center" sx={tableHeaderCellStyle}>
                          Journal ID
                        </TableCell>
                        <TableCell align="center" sx={tableHeaderCellStyle}>
                          Branch
                        </TableCell>
                        <TableCell align="center" sx={tableHeaderCellStyle}>
                          Currency
                        </TableCell>
                        <TableCell align="center" sx={tableHeaderCellStyle}>
                          CGL
                        </TableCell>
                        <TableCell align="center" sx={tableHeaderCellStyle}>
                          Amount
                        </TableCell>
                        <TableCell align="center" sx={tableHeaderCellStyle}>
                          Txn Type
                        </TableCell>
                      </TableRow>
                    </TableHead>

                    <TableBody>
                      {batchDetails.map((group, groupIndex) => {
                        const isSelected = selectedJournalPrefixes.includes(
                          group[0].journalPrefix
                        );
                        const groupRowCount = group.length;

                        return group.map((row, rowIndex) => (
                          <TableRow
                            key={row.id}
                            hover={!isViewingSelfCreated}
                            onClick={() =>
                              !isViewingSelfCreated &&
                              handleRowCheckboxClick(null, row.journalPrefix)
                            }
                            role="checkbox"
                            aria-checked={isSelected}
                            selected={isSelected}
                            sx={{
                              cursor: isViewingSelfCreated
                                ? "not-allowed"
                                : "pointer",
                              ...(rowIndex === 0 &&
                                groupIndex > 0 && {
                                  borderTop: (theme) =>
                                    `2px solid ${theme.palette.divider}`,
                                }),
                            }}
                          >
                            {rowIndex === 0 && (
                              <TableCell
                                padding="checkbox"
                                rowSpan={groupRowCount}
                                sx={stickyColumnCell}
                              >
                                <Box sx={stickyBoxStyle}>
                                  <Checkbox
                                    color="primary"
                                    checked={isSelected}
                                    disabled={isViewingSelfCreated}
                                    onClick={(e) =>
                                      !isViewingSelfCreated &&
                                      handleRowCheckboxClick(
                                        e,
                                        row.journalPrefix
                                      )
                                    }
                                  />
                                </Box>
                              </TableCell>
                            )}
                            {rowIndex === 0 && (
                              <TableCell
                                align="center"
                                rowSpan={groupRowCount}
                                sx={stickyColumnCell}
                              >
                                <Box sx={stickyBoxStyle}>
                                  {row.journalPrefix}
                                </Box>
                              </TableCell>
                            )}
                            <TableCell align="center" sx={cellBorderStyle}>
                              {row.payload.branch}
                            </TableCell>
                            <TableCell align="center" sx={cellBorderStyle}>
                              {row.payload.currency}
                            </TableCell>
                            <TableCell align="center" sx={cellBorderStyle}>
                              {row.payload.cgl}
                            </TableCell>
                            <TableCell align="center" sx={cellBorderStyle}>
                              {formatCurrency(row.amount)}
                            </TableCell>
                            <TableCell align="center" sx={cellBorderStyle}>
                              <Chip
                                label={row.transactionType}
                                color={getTxnTypeColor(row.transactionType)}
                                size="small"
                              />
                            </TableCell>
                          </TableRow>
                        ));
                      })}
                    </TableBody>
                  </Table>
                </TableContainer>
              ) : (
                <Typography sx={{ p: 3 }}>
                  No pending details found for this batch.
                </Typography>
              )}
            </>
          )}
        </DialogContent>

        <DialogActions sx={dialogActionsStyle}>
          <Grid container spacing={2} alignItems="center">
            <Grid item xs={12} md={6}>
              <TextField
                margin="dense"
                id="partialExecutorRemarks"
                label="Approver Remarks (for selected)"
                type="text"
                fullWidth
                size="small"
                value={partialExecutorRemarks}
                onChange={(e) => setPartialExecutorRemarks(e.target.value)}
                disabled={
                  isPartialSubmitting ||
                  selectedJournalPrefixes.length === 0 ||
                  isViewingSelfCreated
                }
                InputProps={{
                  inputProps: {
                    maxLength: 200,
                  },
                }}
              />
            </Grid>
            <Grid item xs={12} md={6}>
              <Stack direction="row" spacing={1} justifyContent="flex-end">
                <Button
                  variant="contained"
                  color="success"
                  onClick={() => handleSubmitPartialAction("ACCEPTED")}
                  disabled={
                    isPartialSubmitting ||
                    selectedJournalPrefixes.length === 0 ||
                    isViewingSelfCreated
                  }
                >
                  {isPartialSubmitting ? (
                    <CircularProgress size={24} color="inherit" />
                  ) : (
                    "Accept Selected"
                  )}
                </Button>
                <Button
                  variant="contained"
                  color="error"
                  onClick={() => handleSubmitPartialAction("REJECTED")}
                  disabled={
                    isPartialSubmitting ||
                    selectedJournalPrefixes.length === 0 ||
                    isViewingSelfCreated
                  }
                >
                  {isPartialSubmitting ? (
                    <CircularProgress size={24} color="inherit" />
                  ) : (
                    "Reject Selected"
                  )}
                </Button>
              </Stack>
            </Grid>
            {isViewingSelfCreated && (
              <Grid item xs={12}>
                <Typography
                  color="error"
                  variant="body2"
                  textAlign="center"
                  sx={{ pt: 1, width: "100%" }}
                >
                  You cannot approve or reject entries from your own request.
                </Typography>
              </Grid>
            )}
          </Grid>
        </DialogActions>
      </Dialog>
    </Box>
  );
}

import React, { useState, useRef, useEffect } from "react";
import {
  Box,
  Typography,
  Paper,
  TextField,
  Button,
  CircularProgress,
  Chip,
  Alert,
  Stack,
  Stepper,
  Step,
  StepLabel,
} from "@mui/material";
import {
  CloudUpload as CloudUploadIcon,
  RestartAlt as RestartAltIcon,
  Description as DescriptionIcon,
  CheckCircle as CheckCircleIcon,
  Error as ErrorIcon,
  Download as DownloadIcon,
  Send as SendIcon,
  FileDownload as FileDownloadIcon
} from "@mui/icons-material";

import { DataGrid } from "@mui/x-data-grid";

import {
  mainContainerStyle,
  uploadBoxStyle,
  controlPanelStyle,
  iconStyle,
  headerTextStyle,
  subTextStyle,
  dateinputfield,
  stepperContainerStyle,
  stepLabelStyle,
  progressContainerStyle,
  dataGridContainerStyle,
  actionButtonsContainer,
  alertStyle,
  successPaperStyle,
  dataGridHeaderBoxStyle,
  dataGridTitleStyle,
  chipStyle,
  
  postingDate
} from "./JournalBulkStyle";

import useApi from "../../hooks/useApi";
import useCustomSnackbar from "../../utils/useCustomSnackbar";

const steps = ["Format Validation", "Database Check", "DR/CR Check", "Finalizing"];

const JournalBulkUpload = () => {
  const { callApi } = useApi();
  const showSnackBar = useCustomSnackbar();

  // --- STATE ---
  const [selectedDate, setSelectedDate] = useState("");
  const [file, setFile] = useState(null);
  const [isCsvInput, setIsCsvInput] = useState(false);

  // Process Status
  const [requestId, setRequestId] = useState(null);
  const [isProcessing, setIsProcessing] = useState(false);
  const [activeStep, setActiveStep] = useState(0);
  const [failedStep, setFailedStep] = useState(-1);

  // Results
  const [resultStatus, setResultStatus] = useState(null);
  const [resultMsg, setResultMsg] = useState("");
  const [previewRows, setPreviewRows] = useState([]);
  const [totalRows, setTotalRows] = useState(0);
  const [hasSuccessFile, setHasSuccessFile] = useState(false);
  const [hasErrorFile, setHasErrorFile] = useState(false);

  // Flags
  const [autoDownloadTriggered, setAutoDownloadTriggered] = useState(false);
  const [isCreatingBatch, setIsCreatingBatch] = useState(false);

  const fileInputRef = useRef(null);

  const handleDownloadTemplate = async () => {
    try {
        let token = "";
        const getRaw = (key) => sessionStorage.getItem(key) || localStorage.getItem(key);
        const storageAuth = getRaw("persist:auth");
        if (storageAuth) { 
            try { 
                const parsed = JSON.parse(storageAuth); 
                if (parsed?.token) token = parsed.token.replace(/^"|"$/g, ""); 
            } catch (e) {} 
        }

        const response = await fetch("/JS/journals/download-template", {
            method: "GET",
            headers: { Authorization: `Bearer ${token}` }
        });

        if (!response.ok) throw new Error("Failed to download template");

        const blob = await response.blob();
        const url = window.URL.createObjectURL(blob);
        const link = document.createElement("a");
        link.href = url;
        link.setAttribute("download", "Journal_Upload_Template.xlsx");
        document.body.appendChild(link);
        link.click();
        document.body.removeChild(link);
        window.URL.revokeObjectURL(url);
    } catch (e) {
        showSnackBar("Failed to download template", "error");
    }
  };

  const handleDownloadFile = async (type) => {
    if (!requestId) return;

    try {
      let token = "";
      const getRaw = (key) => sessionStorage.getItem(key) || localStorage.getItem(key);
      const storageAuth = getRaw("persist:auth");
      if (storageAuth) { 
          try { 
              const parsed = JSON.parse(storageAuth); 
              if (parsed?.token) token = parsed.token.replace(/^"|"$/g, ""); 
          } catch (e) {} 
      }

      const response = await fetch(
        `/JS/journals/download-bulk-file/${requestId}?type=${type}`,
        {
          method: "GET",
          headers: { 
              Authorization: `Bearer ${token}`,
          },
        }
      );

      if (!response.ok) throw new Error("Server returned " + response.status);

      const blob = await response.blob();
      const url = window.URL.createObjectURL(blob);
      const link = document.createElement("a");
      link.href = url;
      link.setAttribute(
        "download",
        type === "ERROR" ? `Error_Report_${requestId}.xlsx` : `Success_Journal_${requestId}.csv`
      );
      document.body.appendChild(link);
      link.click();
      document.body.removeChild(link);
      window.URL.revokeObjectURL(url);
    } catch (error) {
      showSnackBar("Failed to download file.", "error");
    }
  };

  useEffect(() => {
    let interval;
    if (isProcessing && requestId) {
      interval = setInterval(async () => {
        try {
          const res = await callApi(`/JS/journals/bulk-status/${requestId}`, null, "GET");
          if (!res) return;

          if (res.totalRows) setTotalRows(res.totalRows);
          if (res.currentStage > 0) setActiveStep(res.currentStage - 1);

          if (res.status === "ERROR") {
            setIsProcessing(false);
            setResultStatus("ERROR");
            setResultMsg(res.message);
            
            const failIdx = res.currentStage ? res.currentStage - 1 : 0;
            setFailedStep(failIdx);
            setActiveStep(failIdx);
            setHasErrorFile(res.hasErrorFile);
            
            clearInterval(interval);
            showSnackBar(res.message, "error");

            if (res.hasErrorFile && !autoDownloadTriggered) {
                setAutoDownloadTriggered(true);
                setTimeout(() => handleDownloadFile("ERROR"), 500);
            }
          } 
          else if (res.status === "SUCCESS") {
            setIsProcessing(false);
            setResultStatus("SUCCESS");
            setActiveStep(4);
            setHasSuccessFile(res.hasSuccessFile);
            
            if (res.previewDataJson) {
               setPreviewRows(JSON.parse(res.previewDataJson));
            }
            
            clearInterval(interval);
            showSnackBar("Validation Successful!", "success");
          }
        } catch (e) {
        }
      }, 1000);
    }
    return () => clearInterval(interval);
  }, [isProcessing, requestId, autoDownloadTriggered]);

  const handleDateChange = (e) => {
    setSelectedDate(e.target.value);
    resetState();
  };

  const handleUploadClick = () => {
      if (!selectedDate) {
          showSnackBar("Please select the Posting Date first.", "warning");
          return;
      }
      if (fileInputRef.current) {
          fileInputRef.current.value = ""; 
          fileInputRef.current.click();
      }
  };

  const handleFileChange = (e) => {
    const selectedFile = e.target.files[0];
    if (selectedFile) {
        if (!selectedFile.name.toLowerCase().match(/\.(xlsx|xls|csv)$/)) {
            showSnackBar("Invalid file type.", "error");
            setFile(null); return;
        }
        setFile(selectedFile);
        setIsCsvInput(selectedFile.name.toLowerCase().endsWith(".csv"));
        resetState();
    }
  };

  const resetState = () => {
    setResultStatus(null);
    setResultMsg("");
    setHasSuccessFile(false);
    setHasErrorFile(false);
    setPreviewRows([]);
    setTotalRows(0);
    setRequestId(null);
    setActiveStep(0);
    setFailedStep(-1);
    setAutoDownloadTriggered(false);
  };

  const resetForm = () => {
    setFile(null);
    setSelectedDate("");
    resetState();
    setIsProcessing(false);
    setIsCreatingBatch(false);
    if (fileInputRef.current) fileInputRef.current.value = "";
  };

  const startValidation = async () => {
    if (!file || !selectedDate) {
      showSnackBar("Please select Date and File.", "warning");
      return;
    }
    
    resetState();
    setIsProcessing(true);

    const formData = new FormData();
    formData.append("file", file);
    formData.append("postingDate", selectedDate);

    try {
      // 1. Get Token
      let token = "";
      const getRaw = (key) => sessionStorage.getItem(key) || localStorage.getItem(key);
      const storageAuth = getRaw("persist:auth");
      if (storageAuth) {
         try {
            const parsed = JSON.parse(storageAuth);
            if (parsed?.token) token = parsed.token.replace(/^"|"$/g, "");
         } catch(e) {}
      }

      // 2. Native Fetch
      const response = await fetch("/JS/journals/bulk-validate-init", {
          method: "POST",
          headers: {
              "Authorization": `Bearer ${token}`
          },
          body: formData
      });

      if (!response.ok) {
          const errText = await response.text();
          throw new Error(errText || "Upload failed");
      }

      const res = await response.json();
      
      if (res && res.requestId) {
          setRequestId(res.requestId);
      } else {
          setIsProcessing(false);
          showSnackBar("Failed to initiate validation.", "error");
      }
    } catch (err) {
      console.error("Init Error:", err);
      setIsProcessing(false);
      showSnackBar("Upload Failed. Check console.", "error");
    }
  };

  const handleSubmitBatch = async () => {
    if (previewRows.length === 0) return;
    setIsCreatingBatch(true);
    try {
      const payloadRows = previewRows.map((row) => ({
        changeType: "ADD",
        branch: row.branch,
        currency: row.currency,
        cgl: row.cgl,
        amount: row.txnType.includes("Credit") ? "-" + row.amount : row.amount,
        productType: row.productCode,
        remarks: row.remarks,
        arFlag: "A",
        acClassification: "A",
      }));

      const finalPayload = { commonBatchRemarks: `Bulk Upload - ${selectedDate}`, rows: payloadRows };
      const response = await callApi("/JS/journals/create-bulk-batch", finalPayload, "POST");

      if (response && response.status === "SUCCESS") {
        showSnackBar(`Batch Created! ID: ${response.batchId}`, "success");
        resetForm();
      }
    } catch (e) {
      showSnackBar("Failed to create batch", "error");
    } finally {
      setIsCreatingBatch(false);
    }
  };

  // --- GRID COLUMNS ---
  const columns = [
    { field: "branch", headerName: "Branch", flex: 1, align: "center", headerAlign: "center" },
    { field: "currency", headerName: "Currency", width: 100, align: "center", headerAlign: "center" },
    { field: "cgl", headerName: "CGL", flex: 1.5, align: "center", headerAlign: "center" },
    { field: "productCode", headerName: "Product", width: 120, align: "center", headerAlign: "center", renderCell: (p) => p.value || "-" },
    { 
      field: "amount", headerName: "Amount", flex: 1.5, minWidth: 150, align: "right", headerAlign: "center",
      valueFormatter: (params) => {
        const val = params?.value || params;
        if (!val) return "0.00";
        return val; // No comma formatting
      }
    },
    { 
      field: "txnType", headerName: "Type", width: 100, align: "center", headerAlign: "center",
      renderCell: (params) => (
        <Chip 
          label={params.value} 
          color={params.value && (params.value.includes("CR") || params.value.includes("Credit")) ? "error" : "success"} 
          size="small" 
          variant="filled" 
          sx={chipStyle} 
        />
      )
    },
    { field: "remarks", headerName: "Remarks", flex: 2, align: "center", headerAlign: "center" },
  ];

  return (
    <Box sx={mainContainerStyle}>
      <Paper elevation={1} sx={controlPanelStyle}>
        
        {/* Template Download Button */}
        <Box sx={{ display: 'flex', justifyContent: 'flex-end', mb: 2 }}>
            <Button variant="text" startIcon={<FileDownloadIcon />} onClick={handleDownloadTemplate}>
                Download Template
            </Button>
        </Box>

        <Box sx={dateinputfield}>
            <TextField
                type="date"
                label="Posting Date"
                size="small"
                value={selectedDate}
                onChange={handleDateChange}
                InputLabelProps={{ shrink: true }}
                inputProps={{ max: new Date().toISOString().split("T")[0] }}
            />
        </Box>

        <Box sx={uploadBoxStyle} onClick={handleUploadClick}>
          <input type="file" hidden accept=".xlsx,.xls,.csv" onChange={handleFileChange} ref={fileInputRef} />
          {file ? (
            <>
              <DescriptionIcon sx={{ fontSize: 60, color: "success.main", mb: 1 }} />
              <Typography sx={headerTextStyle}>{file.name}</Typography>
              <Chip label="Ready" color="success" size="small" variant="outlined" sx={{ mt: 1 }} />
              <Typography sx={subTextStyle}>Click to change</Typography>
            </>
          ) : (
            <>
              <CloudUploadIcon sx={iconStyle} />
              <Typography sx={headerTextStyle}>Click to Upload</Typography>
              <Typography sx={subTextStyle}>Supported: .xlsx, .xls, .csv</Typography>
            </>
          )}
        </Box>

        {(isProcessing || resultStatus) && (
          <Box sx={stepperContainerStyle}>
            <Stepper activeStep={activeStep} alternativeLabel>
              {steps.map((label, index) => {
                const isFailed = index === failedStep;
                const isCompleted = index < activeStep || resultStatus === "SUCCESS";
                return (
                  <Step key={label}>
                    <StepLabel
                      error={isFailed}
                      sx={stepLabelStyle}
                      StepIconComponent={() => {
                        if (isFailed) return <ErrorIcon color="error" />;
                        if (isCompleted) return <CheckCircleIcon color="success" />;
                        if (index === activeStep && isProcessing) return <CircularProgress size={20} />;
                        return <Typography color="text.disabled" fontWeight="bold">{index + 1}</Typography>;
                      }}
                    >
                      {label}
                    </StepLabel>
                  </Step>
                );
              })}
            </Stepper>
            {isProcessing && (
                <Box sx={progressContainerStyle}>
                    <Typography variant="caption" color="text.secondary">
                       {totalRows > 0 ? `Rows Detected: ${totalRows.toLocaleString()}` : "Processing file..."}
                    </Typography>
                </Box>
            )}
          </Box>
        )}

        <Stack direction="row" sx={actionButtonsContainer}>
          <Button
            variant="contained"
            onClick={startValidation}
            disabled={isProcessing || !file || !selectedDate}
            size="large"
            startIcon={isProcessing ? <CircularProgress size={20} color="inherit" /> : <CheckCircleIcon />}
          >
            {isProcessing ? "Validating..." : "Validate"}
          </Button>
          <Button variant="outlined" color="error" onClick={resetForm} disabled={isProcessing} size="large" startIcon={<RestartAltIcon />}>
            Reset
          </Button>
        </Stack>

        <Box sx={{ mt: 3 }}>
          {resultStatus === "ERROR" && (
            <Alert severity="error" sx={alertStyle}>
              <Typography variant="subtitle2" fontWeight="bold">Validation Failed</Typography>
              <Typography variant="body2">{resultMsg}</Typography>
            </Alert>
          )}

          {resultStatus === "SUCCESS" && (
            <Paper variant="outlined" sx={successPaperStyle}>
              <Stack direction={{ xs: "column", sm: "row" }} justifyContent="space-between" alignItems="center" spacing={2}>
                <Box>
                  <Stack direction="row" alignItems="center" spacing={1}>
                    <CheckCircleIcon color="success" />
                    <Typography variant="subtitle1" fontWeight="bold">Validation Successful!</Typography>
                  </Stack>
                  <Typography variant="body2">{totalRows.toLocaleString()} rows verified.</Typography>
                </Box>
                <Stack direction="row" spacing={2}>
                  {!isCsvInput && hasSuccessFile && (
                    <Button
                      variant="outlined"
                      color="success"
                      startIcon={<DownloadIcon />}
                      onClick={() => handleDownloadFile("SUCCESS")}
                      sx={{ bgcolor: "background.paper" }}
                    >
                      Download CSV
                    </Button>
                  )}
                  <Button
                    variant="contained"
                    color="success"
                    startIcon={isCreatingBatch ? <CircularProgress size={20} color="inherit" /> : <SendIcon />}
                    onClick={handleSubmitBatch}
                    disabled={isCreatingBatch}
                  >
                    {isCreatingBatch ? "Creating..." : "Create Batch"}
                  </Button>
                </Stack>
              </Stack>
            </Paper>
          )}
        </Box>
      </Paper>
     
      {resultStatus === "SUCCESS" && previewRows.length > 0 && (
        <Paper elevation={1} sx={dataGridContainerStyle}>
          <Box sx={dataGridHeaderBoxStyle}>
            <Typography variant="h6" sx={dataGridTitleStyle}>Data Preview</Typography>
          </Box>
          <DataGrid
            rows={previewRows}
            columns={columns}
            initialState={{
                pagination: { paginationModel: { pageSize: 50 } },
            }}
            pageSizeOptions={[50, 100, 150]}
            disableSelectionOnClick
            density="comfortable"
            sx={{ border: "none" }}
          />
        </Paper>
      )}
    </Box>
  );
};

export default JournalBulkUpload;
