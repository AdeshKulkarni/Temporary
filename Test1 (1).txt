package com.tcs.fincore.MOCService.Models.enums;

import lombok.Getter;

import java.util.Arrays;
import java.util.Map;
import java.util.stream.Collectors;

@Getter
public enum AcClassification {
    ASSET('A', "Asset"),
    LIABILITY('L', "Liability"),
    INCOME('I', "Income"),
    EXPENSE('E', "Expense"),
    MEMO('M', "Memo A/c");

    private final char code;
    private final String description;

    AcClassification(char code, String description) {
        this.code = code;
        this.description = description;
    }

    public static Map<Character, String> getClassificationMap() {
        return Arrays.stream(AcClassification.values())
                .collect(Collectors.toMap(AcClassification::getCode, AcClassification::getDescription));
    }
}












package com.tcs.fincore.MOCService.Controllers;


import lombok.RequiredArgsConstructor;

import org.springframework.http.ResponseEntity;
import org.springframework.web.bind.annotation.GetMapping;
import org.springframework.web.bind.annotation.RequestMapping;
import org.springframework.web.bind.annotation.RequestParam;
import org.springframework.web.bind.annotation.RestController;

import com.tcs.fincore.MOCService.Dto.CglDto;
import com.tcs.fincore.MOCService.Models.enums.AcClassification;
import com.tcs.fincore.MOCService.Service.CglService;

import java.util.List;
import java.util.Map;

@RestController
// IMPORTANT: Your controller path must match what React is calling.
// Based on our previous fixes, it should be "/api/cgls"
@RequestMapping("/api/cgls") 
@RequiredArgsConstructor 
public class CglController {

	 private final CglService cglService;

	    @GetMapping("/classifications")
	    public ResponseEntity<Map<String, String>> getCglClassifications() {
	        return ResponseEntity.ok(cglService.getClassificationTypes());
	    }

	    @GetMapping("/filter")
	    public ResponseEntity<List<CglDto>> filterCglsByClassification(
	            @RequestParam("classification") AcClassification classification) {
	        return ResponseEntity.ok(cglService.getCglsByClassification(classification));
	    }

	  
	    @GetMapping("/search")
	    public ResponseEntity<List<CglDto>> searchCgls(
	            // --- FIX ---
	            // Made query optional. This allows the React code to send
	            // just a classification to get all CGLs for that Head.
	            @RequestParam(value = "query", required = false, defaultValue = "") String query,
	            // --- END FIX ---
	            @RequestParam(value = "classification", required = false) AcClassification classification) {
	        
	        return ResponseEntity.ok(cglService.searchCgls(query, classification));
	    }
}
