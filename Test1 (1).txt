import React, { useState, useMemo, useEffect } from "react";
import {
  Box,
  Fab,
  Dialog,
  DialogActions,
  DialogContent,
  DialogTitle,
  TextField,
  Button,
} from "@mui/material";
import { DataGrid, GridActionsCellItem } from "@mui/x-data-grid";
import AddIcon from "@mui/icons-material/Add";
import EditIcon from "@mui/icons-material/Edit";
import useApi from "../../../hooks/useApi"; // Assuming the path is the same
import useCustomSnackbar from "../../../utils/useCustomSnackbar"; // Assuming the path is the same

const initialFormState = {
  stateCode: "",
  stateName: "",
};

export default function StateMaster() {
  const { callApi } = useApi();
  const showSnackBar = useCustomSnackbar();
  const [rows, setRows] = useState([]);
  const [isLoading, setIsLoading] = useState(true);

  const [isDialogOpen, setIsDialogOpen] = useState(false);
  const [formData, setFormData] = useState(initialFormState);
  const [editingRow, setEditingRow] = useState(null);
  const [errors, setErrors] = useState({});

  const fetchData = async () => {
    setIsLoading(true);
    try {
      const response = await callApi("CM/states", null, "GET");
      console.log(response);

      if (response && response.data && Array.isArray(response.data)) {
        const formattedData = response.data.map((row) => {
          let formattedDate = "N/A";
          if (row.updatedAt) {
            const date = new Date(row.updatedAt);
            if (!isNaN(date.getTime())) {
              const day = String(date.getDate()).padStart(2, "0");
              const month = String(date.getMonth() + 1).padStart(2, "0");
              const year = date.getFullYear();
              formattedDate = `${day}-${month}-${year}`;
            }
          }
          return {
            ...row,
            id: row.stateCode,
            updatedAt: formattedDate,
          };
        });

        setRows(formattedData);
        if (formattedData.length === 0) {
          showSnackBar("No state data found.", "info");
        }
      } else {
        showSnackBar("Received invalid data format.", "error");
        setRows([]);
      }
    } catch (error) {
      console.error("Error fetching state data:", error);
      showSnackBar(
        "Something went wrong while fetching data. Please try again later.",
        "error"
      );
      setRows([]);
    } finally {
      setIsLoading(false);
    }
  };

  useEffect(() => {
    fetchData();
  }, []);

  const handleOpenDialog = () => {
    setEditingRow(null);
    setFormData(initialFormState);
    setErrors({});
    setIsDialogOpen(true);
  };

  const handleCloseDialog = () => {
    setIsDialogOpen(false);
    setEditingRow(null);
    setFormData(initialFormState);
    setErrors({});
  };

  const handleEditClick = (id) => () => {
    const rowToEdit = rows.find((row) => row.id === id);
    if (rowToEdit) {
      setEditingRow(rowToEdit);
      setFormData({
        stateCode: rowToEdit.stateCode,
        stateName: rowToEdit.stateName,
      });
      setErrors({});
      setIsDialogOpen(true);
    }
  };

  const validate = () => {
    let tempErrors = {};

    if (!formData.stateName?.trim()) {
      tempErrors.stateName = "State Name is required.";
    } else if (/\d/.test(formData.stateName)) {
      tempErrors.stateName = "Numbers are not allowed in State Name.";
    }

    if (!editingRow) {
      if (!formData.stateCode?.trim()) {
        tempErrors.stateCode = "State Code is required.";
      } else if (
        formData.stateCode.length > 3 ||
        formData.stateCode.length < 2
      ) {
        tempErrors.stateCode = "State Code must be 2 to 3 characters.";
      }
    }

    setErrors(tempErrors);
    return Object.keys(tempErrors).length === 0;
  };

  const handleInputChange = (e) => {
    const { name, value } = e.target;
    let processedValue = value;

    if (name === "stateCode") {
      processedValue = value
        .toUpperCase()
        .slice(0, 3)
        .replace(/[^A-Z]/g, "");
    }
    if (name === "stateName") {
      processedValue = value.slice(0, 30).replace(/[^a-zA-Z\s]/g, "");
    }

    setFormData({ ...formData, [name]: processedValue });
    if (errors[name]) {
      setErrors({ ...errors, [name]: "" });
    }
  };

  const handleSubmit = async () => {
    if (!validate()) {
      return;
    }

    const isEditing = !!editingRow;
    const changeType = isEditing ? "UPDATE" : "ADD";

    const apiPayload = {
      requestType: "STATE",
      changeType: changeType,
      payload: {
        stateCode: formData.stateCode,
        stateName: formData.stateName,
      },
    };

    try {
      await callApi("CR/create-request", apiPayload, "POST");

      showSnackBar(
        `Request to ${
          isEditing ? "update" : "add"
        } state submitted successfully!`,
        "success"
      );
      handleCloseDialog();
      fetchData();
    } catch (error) {
      console.error("Error submitting request:", error);
      showSnackBar(
        error.response?.data?.message ||
          `Failed to submit ${isEditing ? "update" : "add"} request.`,
        "error"
      );
    }
  };

  const columns = useMemo(
    () => [
      {
        field: "stateCode",
        headerName: "State Code",
        flex: 1,
        sortable: true,
        align: "center",
        headerAlign: "center",
      },
      {
        field: "stateName",
        headerName: "State Name",
        flex: 2,
        sortable: true,
        align: "center",
        headerAlign: "center",
      },
      {
        field: "updatedAt",
        headerName: "Last Updated",
        flex: 1,
        sortable: true,
        align: "center",
        headerAlign: "center",
      },
      {
        field: "actions",
        type: "actions",
        headerName: "Actions",
        width: 80,
        headerAlign: "center",
        align: "center",
        cellClassName: "actions",
        getActions: ({ id }) => [
          <GridActionsCellItem
            icon={<EditIcon />}
            label="Edit"
            onClick={handleEditClick(id)}
            color="primary"
          />,
        ],
      },
    ],
    []
  );

  return (
    <Box sx={{ p: 3 }}>
      <DataGrid
        rows={rows}
        columns={columns}
        disableRowSelectionOnClick
        pageSizeOptions={[5, 10, 25, 50]}
        initialState={{
          pagination: { paginationModel: { pageSize: 10, page: 0 } },
          sorting: { sortModel: [{ field: "updatedAt", sort: "desc" }] },
        }}
        loading={isLoading}
        autoHeight
        sx={{
          width: "100%",
          "& .MuiDataGrid-cell:focus, & .MuiDataGrid-cell:focus-within": {
            outline: "none",
          },
          "& .MuiDataGrid-row:nth-of-type(odd)": {
            backgroundColor: (theme) => theme.palette.action.hover,
          },
        }}
      />

      <Fab
        color="primary"
        aria-label="add"
        onClick={handleOpenDialog}
        sx={{ position: "fixed", bottom: 30, right: 30 }}
      >
        <AddIcon />
      </Fab>

      <Dialog
        open={isDialogOpen}
        onClose={handleCloseDialog}
        fullWidth
        maxWidth="sm"
      >
        <DialogTitle sx={{ fontWeight: "bold" }}>
          {editingRow ? "Edit State" : "Add New State"}
        </DialogTitle>
        <DialogContent>
          <TextField
            autoFocus
            margin="dense"
            name="stateCode"
            label="State Code"
            type="text"
            fullWidth
            variant="outlined"
            value={formData.stateCode}
            onChange={handleInputChange}
            error={!!errors.stateCode}
            helperText={errors.stateCode}
            disabled={!!editingRow}
            inputProps={{ maxLength: 3 }}
          />
          <TextField
            margin="dense"
            name="stateName"
            label="State Name"
            type="text"
            fullWidth
            variant="outlined"
            value={formData.stateName}
            onChange={handleInputChange}
            error={!!errors.stateName}
            helperText={errors.stateName}
            inputProps={{ maxLength: 30 }}
          />
        </DialogContent>
        <DialogActions sx={{ p: "16px 24px" }}>
          <Button onClick={handleCloseDialog}>Cancel</Button>
          <Button onClick={handleSubmit} variant="contained">
            Submit
          </Button>
        </DialogActions>
      </Dialog>
    </Box>
  );
}
