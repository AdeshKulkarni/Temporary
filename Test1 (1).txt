    public Map<String, Object> getPendingBatchesForChecker(Integer userRole, String userId, int page, int size, String batchIdFilter, String vidFilter) {

        // DYNAMIC EXECUTION ROLE QUERY:
        // Automatically uses PENDING_AT to join WORKFLOW_TRANSITIONS (w). 
        // 1. Normal users see pending batches they haven't acted on yet.
        // 2. The Execution Role (PSO) ONLY sees batches that have FAILED execution, requiring a manual Retry.
        // (Scheduled batches waiting for the system timer are hidden from this actionable screen).
        String sql = """
            SELECT 
                p.BATCH_ID, 
                p.VID_FK AS VID, 
                p.CREATED_BY AS CREATOR_ID, 
                p.CREATED_AT AS REQ_DATE, 
                p.COMMON_BATCH_REMARKS AS BATCH_REMARKS,
                p.RECORDS_COUNT AS TOTAL_ROWS, 
                p.SCHEDULED_TIME,
                w.WORKFLOW_STAGE, 
                w.WORKFLOW_CODE, 
                w.ESCALATION_TIME, 
                w.WORKFLOW_ORDER, 
                w.IS_FINAL_APPROVAL,
                p.IS_EXECUTION_SCHEDULED,
                COALESCE(u.FIRST_NAME || ' ' || u.LAST_NAME, p.CREATED_BY) AS CREATOR_NAME,
                (SELECT LAST_ACTION FROM VOUCHER_WORKFLOW WHERE BATCH_ID_FK = p.BATCH_ID ORDER BY LAST_ACTION_AT DESC FETCH FIRST 1 ROWS ONLY) AS LATEST_ACTION
            FROM VOUCHER_POSTING p
            INNER JOIN WORKFLOW_TRANSITIONS w ON p.PENDING_AT = w.WORKFLOW_ID
            LEFT JOIN USERS u ON p.CREATED_BY = u.USER_ID
            WHERE p.BATCH_STATUS = 'P'
            AND w.ALLOWED_ROLE_ID = ?
            AND (
                -- 1. STANDARD PENDING BATCH: Normal visibility (Not Execution stage)
                (w.WORKFLOW_STAGE != 'Execution'
                 AND NOT EXISTS (SELECT 1 FROM VOUCHER_WORKFLOW vw WHERE vw.BATCH_ID_FK = p.BATCH_ID AND vw.LAST_ACTION_BY = ?)
                )
                OR 
                -- 2. EXECUTION ROLE (PSO): Shows ONLY Failed executions for Retry action!
                (w.WORKFLOW_STAGE = 'Execution'
                 AND (SELECT LAST_ACTION FROM VOUCHER_WORKFLOW WHERE BATCH_ID_FK = p.BATCH_ID ORDER BY LAST_ACTION_AT DESC FETCH FIRST 1 ROWS ONLY) = 'Execution Failed' 
                )
            )
            ORDER BY p.CREATED_AT DESC
        """;

        List<Map<String, Object>> allPending = jdbcTemplate.query(sql, (rs, rowNum) -> {
            Map<String, Object> map = new HashMap<>();
            boolean isFailed = "Execution Failed".equals(rs.getString("LATEST_ACTION"));

            map.put("batchId", rs.getString("BATCH_ID"));
            map.put("vid", rs.getString("VID"));
            map.put("creatorId", rs.getString("CREATOR_ID"));
            map.put("creatorName", rs.getString("CREATOR_NAME"));
            map.put("requestDate", rs.getTimestamp("REQ_DATE"));
            map.put("commonBatchRemarks", rs.getString("BATCH_REMARKS"));
            map.put("requestCount", rs.getLong("TOTAL_ROWS"));
            map.put("escalationTime", rs.getLong("ESCALATION_TIME"));
            map.put("approvalWorkflow", rs.getString("WORKFLOW_CODE"));
            map.put("currentOrder", rs.getObject("WORKFLOW_ORDER"));
            map.put("isFinalApproval", rs.getString("IS_FINAL_APPROVAL"));

            map.put("isExecutionFailed", isFailed);
            
            if (isFailed) {
                map.put("workflowState", "Execution");
            } else {
                map.put("workflowState", rs.getString("WORKFLOW_STAGE"));
            }

            return map;
        }, userRole, userId); 

        // JAVA-SIDE FILTERING
        List<Map<String, Object>> filteredList = allPending.stream().filter(row -> {
            boolean matchesBatch = true;
            boolean matchesVid = true;
            if (batchIdFilter != null && !batchIdFilter.trim().isEmpty()) {
                String bId = (String) row.get("batchId");
                matchesBatch = bId != null && bId.toLowerCase().contains(batchIdFilter.trim().toLowerCase());
            }
            if (vidFilter != null && !vidFilter.trim().isEmpty()) {
                String vId = (String) row.get("vid");
                matchesVid = vId != null && vId.toLowerCase().contains(vidFilter.trim().toLowerCase());
            }
            return matchesBatch && matchesVid;
        }).toList();

        // JAVA-SIDE PAGINATION
        int totalElements = filteredList.size();
        int fromIndex = Math.min(page * size, totalElements);
        int toIndex = Math.min(fromIndex + size, totalElements);
        List<Map<String, Object>> pagedList = filteredList.subList(fromIndex, toIndex);

        for (Map<String, Object> row : pagedList) {
            row.put("timeSinceLastAction", calculateTimeSinceLastAction((String) row.get("batchId")));
        }

        Map<String, Object> response = new HashMap<>();
        response.put("content", pagedList);
        response.put("totalElements", totalElements);
        response.put("totalPages", (int) Math.ceil((double) totalElements / size));
        return response;
    }
