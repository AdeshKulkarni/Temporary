package com.tcs.fincore.CommonRequestService.service;

import com.tcs.fincore.CommonRequestService.dto.CglDto;
import com.tcs.fincore.CommonRequestService.model.CglMaster;
import com.tcs.fincore.CommonRequestService.model.enums.AcClassification;
import com.tcs.fincore.CommonRequestService.repository.CglMasterRepository;
import lombok.RequiredArgsConstructor;
import org.springframework.stereotype.Service;

import java.util.Arrays;
import java.util.List;
import java.util.Map;
import java.util.stream.Collectors;

@Service
@RequiredArgsConstructor
public class CglService {

    private final CglMasterRepository cglMasterRepository;

    private CglDto convertToDto(CglMaster cgl) {
        return new CglDto(cgl.getCglNumber(), cgl.getDescription());
    }

    public List<CglDto> getCglsByClassification(AcClassification classification) {
        return cglMasterRepository.findByAcClassification(classification)
                .stream()
                .map(this::convertToDto)
                .collect(Collectors.toList());
    }

    public List<CglDto> searchCglsByNumber(String query) {
        return cglMasterRepository.findByCglNumberContaining(query)
                .stream()
                .map(this::convertToDto)
                .collect(Collectors.toList());
    }

    // --- NEW METHOD ---
    // Service logic to handle the new combined search
    public List<CglDto> searchCgls(String query, AcClassification classification) {
        if (classification != null) {
            // If classification is provided, use the new combined search
            return cglMasterRepository.findByCglNumberContainingAndAcClassification(query, classification)
                    .stream()
                    .map(this::convertToDto)
                    .collect(Collectors.toList());
        } else {
            // Otherwise, fall back to the original search
            return searchCglsByNumber(query);
        }
    }
    // --- END OF NEW METHOD ---


    public Map<String, String> getClassificationTypes() {
        return Arrays.stream(AcClassification.values())
                .collect(Collectors.toMap(
                        AcClassification::name,
                        AcClassification::getDescription
                ));
    }
}
