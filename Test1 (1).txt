const formatDateTime = (dateVal) => {
  if (!dateVal) return "N/A";
  // Formats to: "19 Jun 2026, 12:58 pm"

};


const shortenString = (value) => {
  let pos = 0;
  for (let i = 0; i < 2; i++) {
    const matchPos = value.indexOf("\n", pos + 1);
    if (matchPos == -1) {
      break;
    }
    i++;
    pos = matchPos;
  }
  if (pos > 100) {
    return value.substring(0, 100);
  }
  return value.substring(0, pos);
}

const formatBigIntForDisplay = (val) => {
  if (typeof val !== "bigint") {
    val = BigInt(val || 0);
  }
  const str = val.toString().padStart(5, "0");
  const integer = str.slice(0, -4);
  const fractional = str.slice(-4);
  return `${integer || "0"}.${fractional}`;
};

const ExpandableText = ({ text, maxLength = 50 }) => {
  const [anchorEl, setAnchorEl] = useState(null);

  if (!text) return null;

  // If text is short, just show it normally
  if (text.length <= maxLength) {
    return <Typography variant="body2">{text}</Typography>;
  }

  const handleClick = (event) => {
    setAnchorEl(event.currentTarget);
  };

  const handleClose = () => {
    setAnchorEl(null);
  };

  const open = Boolean(anchorEl);

  return (
    <Box>
      <Typography variant="body2" sx={{ display: 'inline' }}>
        {text.substring(0, maxLength)}...
      </Typography>
      <Typography
        variant="body2"
        color="primary"
        sx={{ display: 'inline', cursor: 'pointer', ml: 0.5, fontWeight: 'bold' }}
        onClick={handleClick}
      >
        Show more
      </Typography>

      {/* Popover automatically prevents going off-screen */}
      <Popover
        open={open}
        anchorEl={anchorEl}
        onClose={handleClose}
        anchorOrigin={{
          vertical: 'bottom',
          horizontal: 'left',
        }}
        transformOrigin={{
          vertical: 'top',
          horizontal: 'left',
        }}
        slotProps={{
          paper: {
            sx: {
              maxWidth: { xs: '90vw', sm: 400 }, // Ensures it never exceeds screen width
              maxHeight: 300, // Adds a scrollbar if the text is massive
              p: 2,
              boxShadow: 3,
              overflowY: 'auto',
              wordWrap: 'break-word' // Forces long words to wrap instead of pushing the box wide
            }
          }
        }}
      >
        <Typography variant="body2" sx={{ whiteSpace: 'pre-wrap' }}>
          {text}
        </Typography>
      </Popover>
    </Box>
  );
};

    /**
     * The core asynchronous engine for processing massive bulk voucher uploads.
     * * PURPOSE:
     * This method safely moves up to 100,000+ rows from the Redis Cache into the Oracle Database
     * without causing OutOfMemory (OOM) errors.
     * */

    @Async("bulkExecutor")
    public void executeAsyncBatchCreation(String batchId, String journalPrefix, String requestId, String commonRemarks, String creatorId, Integer creatorRole,  String vid) {

        String lockKey = "LOCK_REQ_" + requestId;
        MDC.put("batchId", batchId);
        MDC.put("user", creatorId);

        Map<String, String> contextMap = MDC.getCopyOfContextMap();
        AtomicInteger successfulRowsCount = new AtomicInteger(0);

        log.info("ASYNC_BATCH_START | Batch: {} | Source: {}", batchId, requestId);
        long start = System.currentTimeMillis();

        try {
            // 1. Enforce hard network timeout on database locks (60 seconds)
            jdbcTemplate.setQueryTimeout(60);

            jdbcTemplate.update("DELETE FROM VOUCHER_POSTING_DATA WHERE BATCH_ID = ?", batchId);
            progressService.sendProgress(batchId, creatorId, 5, "PROCESSING", "Preparing Data...");

            int totalChunks = validationStateManager.getTotalChunks(requestId);

            if (totalChunks == 0) {
                log.error("CACHE_MISS | Batch: {} | Status: No valid rows found", batchId);
                progressService.sendProgress(batchId, creatorId, 0, "FAILED", "Data cache expired or empty.");
                return;
            }

            // Fetch the TRUE row count that the Command Service just saved to the database
            Long totalRows = jdbcTemplate.queryForObject(
                    "SELECT NVL(RECORDS_COUNT, 0) FROM VOUCHER_POSTING WHERE BATCH_ID = ?",
                    Long.class,
                    batchId
            );

            int batchSize = 5000; // Keep this just for the math in the loop offset

            List<CompletableFuture<Void>> futures = new ArrayList<>();
            AtomicReference<BigDecimal> totalDebit = new AtomicReference<>(BigDecimal.ZERO);
            AtomicReference<BigDecimal> totalCredit = new AtomicReference<>(BigDecimal.ZERO);
            AtomicInteger completedChunks = new AtomicInteger(0);

            // 2. BACKPRESSURE: Limit memory usage to exactly 3 chunks at a time.
            java.util.concurrent.Semaphore memorySemaphore = new java.util.concurrent.Semaphore(3);

            for (int i = 0; i < totalChunks; i++) {
                final int chunkIndex = i;
                final int startIdx = i * batchSize;

                memorySemaphore.acquire();

                try {
                    CompletableFuture<Void> future = CompletableFuture.supplyAsync(() -> {
                                try {
                                    return validationStateManager.getChunkFromCache(requestId, chunkIndex);
                                } catch (Exception ex) {
                                    log.error("CACHE_FETCH_FAIL | Batch: {} | Chunk: {}", batchId, chunkIndex, ex);
                                    throw new RuntimeException(ex);
                                }
                            }, chunkWorkerExecutor)
                            .thenAcceptAsync(chunk -> {
                                if (contextMap != null) MDC.setContextMap(contextMap);
                                try {
                                    self.processChunkRaw(chunk, batchId, journalPrefix, commonRemarks, creatorId, creatorRole, startIdx, totalDebit, totalCredit);
                                    successfulRowsCount.addAndGet(chunk.size());
                                } finally {
                                    MDC.clear();
                                }
                            }, chunkWorkerExecutor)
                            .thenRun(() -> {
                                int done = completedChunks.incrementAndGet();
                                int percent = 5 + (done * 90 / totalChunks);
                                progressService.sendProgress(batchId, creatorId, percent, "PROCESSING", String.format("Writing chunks: %d of %d", done, totalChunks));
                            })
                            .whenComplete((result, ex) -> {
                                // Releases the permit if the thread finishes (Success or Fail)
                                memorySemaphore.release();
                            });

                    futures.add(future);

                } catch (java.util.concurrent.RejectedExecutionException ree) {
                    // 3. LEAK PREVENTION: If the thread pool refuses to accept the task,
                    // release the permit before throwing the error up to the main catch block.
                    memorySemaphore.release();
                    throw ree;
                }
            }

            // Wait for all chunks to finish execution safely
            CompletableFuture.allOf(futures.toArray(new CompletableFuture[0])).get(45, java.util.concurrent.TimeUnit.MINUTES);

            if (successfulRowsCount.get() != totalRows) {
                throw new CustomExceptions.DatabaseOperationException(
                        "Data Integrity Failure: Expected " + totalRows + " rows but only successfully inserted " + successfulRowsCount.get() + " rows."
                );
            }

            progressService.sendProgress(batchId, creatorId, 100, "COMPLETED", "Batch Created Successfully");

            jdbcTemplate.update("""
                UPDATE VOUCHER_POSTING 
                SET BATCH_STATUS = 'P' 
                WHERE BATCH_ID = ? 
                AND BATCH_STATUS = 'S'
            """, batchId);

            log.info("BATCH_CREATION_FINISHED | Batch: {} | Rows: {}", batchId, totalRows);

            try {
                String wfSql = """
                    SELECT wt.ALLOWED_ROLE_ID, wt.WORKFLOW_STAGE 
                    FROM VOUCHER_POSTING vp 
                    JOIN WORKFLOW_TRANSITIONS wt ON vp.PENDING_AT = wt.WORKFLOW_ID 
                    WHERE vp.BATCH_ID = ? AND wt.WORKFLOW_ORDER = 2
                """;
                Map<String, Object> step2Info = jdbcTemplate.queryForMap(wfSql, batchId);
                String nextApproverRoleId = String.valueOf(step2Info.get("ALLOWED_ROLE_ID"));
                String stageName = (String) step2Info.get("WORKFLOW_STAGE");

                notificationWriterService.createNextStepApprovalNotification(batchId, nextApproverRoleId, stageName);
            } catch (org.springframework.dao.EmptyResultDataAccessException e) {
                log.error("Could not find Step 2 workflow configuration for Batch {} to send notification.", batchId);
            }

        } catch (Throwable t) {
            // 4. THE ULTIMATE CATCH-ALL: Catches Exceptions, OOM Errors, and Thread Deaths
            log.error("BATCH_FATAL_ERROR | Batch: {} | Error: {}", batchId, t.getMessage(), t);
            handleAsyncFailure(batchId, creatorId, requestId, "An internal error occurred during processing.");
        } finally {
            redisTemplate.delete(lockKey);
            MDC.clear();
        }
    }

    private void handleAsyncFailure(String batchId, String creatorId, String requestId, String message) {
        try {
            // 1. FETCH VID FIRST
            String vid = jdbcTemplate.queryForObject("SELECT VID_FK FROM VOUCHER_POSTING WHERE BATCH_ID = ?", String.class, batchId);

            // 2. REVERT TO ACTIVE IF CRASH HAPPENS
            if (vid != null) {
                workflowEngine.updateVidStatus(vid, VoucherStatus.ACTIVE);
            }
            // 3. CLEANUP ORPHANS
            jdbcTemplate.update("DELETE FROM VOUCHER_POSTING_DATA WHERE BATCH_ID = ?", batchId);

            // FIX: Set to 'Q' (Queued/Stuck) instead of 'ERROR' or 'R'
            jdbcTemplate.update("UPDATE VOUCHER_POSTING SET BATCH_STATUS = 'Q' WHERE BATCH_ID = ?", batchId);

        } catch (DataAccessException e) {
            log.error("CLEANUP_CRITICAL_FAIL | Batch: {} | Cannot record failure state", batchId);
        }
        progressService.sendProgress(batchId, creatorId, 0, "FAILED", message);
        sendNotification(creatorId, null, "Batch creation Failed for Batch ID : " + batchId, batchId);
    }



    @Transactional(propagation = Propagation.REQUIRES_NEW)
    public void processChunkRaw(List<ExcelRowData> chunk ,
                                String batchId,
                                String journalPrefix,
                                String commonRemarks,
                                String creatorId,
                                Integer creatorRole,
                                int globalOffset,
                                AtomicReference<BigDecimal> globalDebit,
                                AtomicReference<BigDecimal> globalCredit) {

        // FIX: Removed POSTING_DATE from columns.
        // 11 explicit columns + 1 sequence = 12 total targets.
        // 11 explicit '?' placeholders match the 11 explicit columns.
        String sql = "INSERT INTO VOUCHER_POSTING_DATA (" +
                "REQ_ID, REQ_STATUS, REQ_DATE, CREATOR_ID, BATCH_ID, " +
                "JOURNAL_ID, REQ_BRANCH_CODE, REQ_CURRENCY, REQ_CGL, " +
                "REQ_AMOUNT, REQ_NARRATION, REQ_PRODUCT) " +
                "VALUES (VOUCHER_POSTING_DATA_SEQ.NEXTVAL, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?)";

        try {
            jdbcTemplate.batchUpdate(sql, new org.springframework.jdbc.core.BatchPreparedStatementSetter() {
                @Override
                public void setValues(java.sql.PreparedStatement ps, int i) throws java.sql.SQLException {
                    ExcelRowData r = chunk.get(i);

                    int globalIdx = globalOffset + i + 1;
                    String jId = journalPrefix + "-" + globalIdx;

                    // 1. Case-Insensitive Check for Credit
                    BigDecimal absAmt = (r.amount != null) ? r.amount.abs() : BigDecimal.ZERO;
                    String safeTxnType = r.txnType != null ? r.txnType.trim() : "";
                    boolean isCredit = "CR".equalsIgnoreCase(safeTxnType) || "CREDIT".equalsIgnoreCase(safeTxnType);

                    // NEW LOGIC: Debit is negative (for DEBIT_AMOUNT), Credit is positive (for CREDIT_AMOUNT)
                    BigDecimal signedAmt = isCredit ? absAmt : absAmt.negate();

                    // 2. PERFECTLY MAPPED SETTERS (Shifted to 11 parameters)
                    ps.setString(1, "P");                                                           // REQ_STATUS
                    java.time.LocalDateTime istTime = java.time.LocalDateTime.now(java.time.ZoneId.of("Asia/Kolkata"));
                    ps.setTimestamp(2, java.sql.Timestamp.valueOf(istTime));         // REQ_DATE      // REQ_DATE
                    ps.setString(3, creatorId);                                                     // CREATOR_ID
                    ps.setString(4, batchId);                                                       // BATCH_ID
                    ps.setString(5, jId);                                                           // JOURNAL_ID
                    ps.setString(6, r.branch);                                                      // REQ_BRANCH_CODE
                    ps.setString(7, r.currency);                                                    // REQ_CURRENCY
                    ps.setString(8, r.cgl);                                                         // REQ_CGL
                    ps.setBigDecimal(9, signedAmt);                                                 // REQ_AMOUNT

                    // POSTING_DATE binding safely removed

                    ps.setString(10, r.remarks);                                                    // REQ_NARRATION (Shifted from 11)
                    ps.setString(11, r.productCode);                                                // REQ_PRODUCT (Shifted from 12)
                }

                @Override
                public int getBatchSize() {
                    return chunk.size();
                }
            });

            // 3. Update Thread-Safe Aggregates Case-Insensitively
            for (ExcelRowData r : chunk) {
                BigDecimal absAmt = (r.amount != null) ? r.amount.abs() : BigDecimal.ZERO;
                String safeTxnType = r.txnType != null ? r.txnType.trim() : "";

                if ("CR".equalsIgnoreCase(safeTxnType) || "CREDIT".equalsIgnoreCase(safeTxnType)) {
                    globalCredit.updateAndGet(curr -> curr.add(absAmt));
                } else {
                    globalDebit.updateAndGet(curr -> curr.add(absAmt));
                }
            }

        } catch (org.springframework.dao.DataAccessException e) {
            log.error("JDBC_BATCH_FAIL | Batch: {} | Offset: {} | Error: {}", batchId, globalOffset, e.getMessage());
            throw new CustomExceptions.DatabaseOperationException("Database integrity error during chunk insert", e);
        }

    }
    @Async("bulkExecutor")
    public void executeAsyncBatchProcessing(BulkProcessVoucherPostingDto dto, String executorId) {
        String batchId = dto.getBatchId();
        MDC.put("batchId", batchId);
        MDC.put("user", executorId);
        long start = System.currentTimeMillis();

        try {
            self.executeApprovalInTransaction(dto, executorId);
            log.info("BATCH_PROCESSING_DONE | Batch: {} | Status: {} | Duration: {}ms", batchId, dto.getStatus(), System.currentTimeMillis() - start);
        } catch (org.springframework.dao.PessimisticLockingFailureException e) {
            log.error("DATABASE_LOCK_TIMEOUT | Batch: {} | Status: GL_BALANCE row locked", batchId);
            handleAsyncApprovalError(batchId, executorId, dto.getStatus(), "The system is busy. It will retry after some time");
        } catch (DataAccessException e) {
            log.error("PROCEDURE_EXEC_FAIL | Batch: {} | Type: {}", batchId, e.getClass().getSimpleName());
            handleAsyncApprovalError(batchId, executorId, dto.getStatus(), "Database execution error The scheduler will retry again shortly");
        } catch (Exception e) {
            log.error("INTERNAL_PROCESSING_FAULT | Batch: {} | Type: {}", batchId, e.getClass().getSimpleName());
            handleAsyncApprovalError(batchId, executorId, dto.getStatus(),
                    "Internal system fault. The system will automatically attempt to execute this batch again shortly.");
        } finally {
            MDC.clear();
        }
    }

    @Transactional(propagation = Propagation.REQUIRES_NEW)
    public void executeApprovalInTransaction(BulkProcessVoucherPostingDto dto, String executorId) {
        String batchId = dto.getBatchId();


        // FIX: Fetch both Creator and VID for logging
        Map<String, Object> postingInfo = jdbcTemplate.queryForMap(
                "SELECT CREATED_BY, VID_FK FROM VOUCHER_POSTING WHERE BATCH_ID = ?", batchId);
        String vid = (String) postingInfo.get("VID_FK");

        // FIX 2: Fetch Creator from VOUCHER_POSTING
        String creatorId = jdbcTemplate.queryForObject(
                "SELECT CREATED_BY FROM VOUCHER_POSTING WHERE BATCH_ID = ?", String.class, batchId);

        if (creatorId != null && creatorId.equalsIgnoreCase(executorId)) {
            log.error("SECURITY_VIOLATION | Self-approval attempt blocked | Batch: {} | User: {}", batchId, executorId);
            handleAsyncApprovalError(batchId, executorId, dto.getStatus(), "Security Violation: You cannot approve or reject your own batch.");
            return;
        }

        progressService.sendProgress(batchId, executorId, 20, "PROCESSING", "Executing Oracle Core Business Logic...");

        if (RequestStatus.ACCEPTED.equals(dto.getStatus())) {

            List<HdfsSyncDto> syncData = jdbcTemplate.execute(
                    "{call PROCESS_JOURNAL_BATCH(?, ?, ?, ?, ?)}",
                    (CallableStatementCallback<List<HdfsSyncDto>>) cs -> {
                        cs.setString(1, batchId);
                        cs.setString(2, executorId);
                        cs.setString(3, dto.getRemarks());
                        cs.setString(4, "A");
                        cs.registerOutParameter(5, -10);
                        cs.execute();
                        List<HdfsSyncDto> list = new ArrayList<>();
                        try (ResultSet rs = (ResultSet) cs.getObject(5)) {
                            while (rs.next()) {
                                list.add(new HdfsSyncDto(
                                        rs.getString("BRANCH"), rs.getString("CURRENCY"), rs.getString("CGL"),
                                        rs.getDate("BAL_DATE") != null ? rs.getDate("BAL_DATE").toLocalDate() : null,
                                        rs.getBigDecimal("NEW_BALANCE"), rs.getBigDecimal("NEW_INR_BALANCE")
                                ));
                            }
                        }
                        return list;
                    }
            );

            jdbcTemplate.update(
                    "UPDATE VOUCHER_POSTING SET BATCH_STATUS = 'A', EXECUTED_AT = SYSTIMESTAMP, EXECUTOR_ID = ? WHERE BATCH_ID = ?",
                    executorId, batchId);


            // --- ADDED: LOG SYSTEM EXECUTION SUCCESS ---
            workflowEngine.logWorkflowAction(batchId, vid, "Execution Completed", "SYSTEM", "Automated background execution completed successfully.", null);

            try {
                hdfsSyncService.syncToDataLake(syncData);
            } catch (Exception hdfsEx) {
                log.error("HDFS_SYNC_FAIL | Batch: {} | Error: {} | Status: Background recovery queued", batchId, hdfsEx.getClass().getSimpleName());
                jdbcTemplate.update("INSERT INTO HDFS_SYNC_RETRY_QUEUE (BATCH_ID, STATUS) VALUES (?, 'P')", batchId);
            }

            // CRITICAL FIX 1: Wrap notifications so they don't roll back the successful approval
            try {
                progressService.sendProgress(batchId, executorId, 100, "COMPLETED", "Batch Approved Successfully");
                notificationWriterService.createBatchApprovedNotification(batchId, creatorId, executorId);
            } catch (Exception notifEx) {
                log.warn("NOTIFICATION_WARN | Batch: {} | Status: Approved, but failed to send UI notification", batchId);
            }

        } else if (RequestStatus.REJECTED.equals(dto.getStatus())) {

            jdbcTemplate.update(
                    "UPDATE VOUCHER_POSTING_DATA SET REQ_STATUS = 'R' WHERE BATCH_ID = ? AND REQ_STATUS = 'P'",
                    batchId);
            jdbcTemplate.update(
                    "UPDATE VOUCHER_POSTING SET BATCH_STATUS = 'R', EXECUTED_AT = SYSTIMESTAMP, EXECUTOR_ID = ? WHERE BATCH_ID = ?",
                    executorId, batchId);

            // CRITICAL FIX 2: Wrap notifications so they don't roll back the successful rejection
            try {
                progressService.sendProgress(batchId, executorId, 100, "COMPLETED", "Batch Rejected");
                notificationWriterService.createBatchRejectedNotification(batchId, creatorId, executorId, dto.getRemarks());
            } catch (Exception notifEx) {
                log.warn("NOTIFICATION_WARN | Batch: {} | Status: Rejected, but failed to send UI notification", batchId);
            }
        }
    }



    private void handleAsyncApprovalError(String batchId, String executorId, RequestStatus actionStatus, String defaultErrorMsg) {

        int retryCount = 0;
        try {
            retryCount = jdbcTemplate.queryForObject(
                    "SELECT RETRY_COUNT FROM SCHEDULED_BATCH_RETRY WHERE BATCH_ID = ?",
                    Integer.class, batchId
            );
        } catch (Exception ignored) {}

        String finalMsg;
        boolean isFinalFailure = false;

        // If we have hit the maximum retries (5), we stop retrying and send the final failure message.
        if (retryCount >= 5) {
            isFinalFailure = true;
            jdbcTemplate.update("UPDATE VOUCHER_POSTING SET BATCH_STATUS = 'P', IS_EXECUTION_SCHEDULED = 'NO' WHERE BATCH_ID = ?", batchId);
            finalMsg = "Execution failed due to an unexpected database or server timeout. Please check server health.";
        } else {
            // Exponential Backoff Logic for retries
            int delayMinutes = 5;  // Default for Attempt 1
            if (retryCount == 2) delayMinutes = 10;
            else if (retryCount == 3) delayMinutes = 20;
            else if (retryCount >= 4) delayMinutes = 40;

            // Push into Q status and set IS_EXECUTION_SCHEDULED = 'YES' so the scheduler picks it up, even if it was originally manual.
            jdbcTemplate.update(
                    "UPDATE VOUCHER_POSTING SET BATCH_STATUS = 'Q', IS_EXECUTION_SCHEDULED = 'YES', SCHEDULED_TIME = CAST(SYSTIMESTAMP AT TIME ZONE 'Asia/Kolkata' AS TIMESTAMP) + NUMTODSINTERVAL(?, 'MINUTE') WHERE BATCH_ID = ?",
                    delayMinutes, batchId
            );


            finalMsg = "Batch execution delayed due to network or database timeout. The system will auto-retry in " + delayMinutes + " minutes (Attempt " + Math.max(1, retryCount + 1) + " of 5). Please wait.";
        }

        progressService.sendProgress(batchId, executorId, 0, "FAILED", finalMsg);

        String vid = null;
        String creatorId = null;

        try {
            Map<String, Object> batchInfo = jdbcTemplate.queryForMap(
                    "SELECT VID_FK, CREATED_BY FROM VOUCHER_POSTING WHERE BATCH_ID = ?", batchId
            );
            vid = (String) batchInfo.get("VID_FK");
            creatorId = (String) batchInfo.get("CREATED_BY");
        } catch (org.springframework.dao.EmptyResultDataAccessException e) {
            log.warn("Batch info not found for Batch: {}", batchId);
        }

        workflowEngine.logWorkflowAction(batchId, vid, "Execution Failed", "SYSTEM", finalMsg, null);

        try {
            if ("SYSTEM".equalsIgnoreCase(executorId)) {
                // Background retry notifications
                if (creatorId != null) {
                    notificationWriterService.createNotification(creatorId, null, finalMsg, "/voucher-view/" + batchId, batchId, "VoucherService");
                }
            } else {
                // Manual click notifications
                String checkerMsg = String.format("Execution alert for Batch '%s': %s", batchId, finalMsg);
                notificationWriterService.createNotification(executorId, null, checkerMsg, "/voucher-view/" + batchId, batchId, "VoucherService");

                if (creatorId != null && !creatorId.equalsIgnoreCase(executorId)) {
                    String makerMsg = String.format("Execution alert for your Batch '%s': %s", batchId, finalMsg);
                    notificationWriterService.createNotification(creatorId, null, makerMsg, "/voucher-view/" + batchId, batchId, "VoucherService");
                }
            }
        } catch (Exception e) {
            log.warn("NOTIFICATION_WARN | Batch: {} | Could not send failure notifications", batchId);
        }
    }


    @Async("bulkExecutor")
    public void executeAsyncBatchCancellation(String batchId, String userId) {
        MDC.put("batchId", batchId);
        MDC.put("user", userId);
        long start = System.currentTimeMillis();

        try {
            progressService.sendProgress(batchId, userId, 20, "PROCESSING", "Purging Data Rows...");

            // 1. Fetch VID BEFORE the procedure destroys the batch records
            String vid = null;
            try {
                vid = jdbcTemplate.queryForObject("SELECT VID_FK FROM VOUCHER_POSTING WHERE BATCH_ID = ?", String.class, batchId);
            } catch (Exception ignored) {}

            // 2. Call the Oracle PL/SQL procedure to handle massive deletes safely
            jdbcTemplate.execute("{call PURGE_JOURNAL_BATCH(?, ?)}", (CallableStatementCallback<Object>) cs -> {
                cs.setString(1, batchId);
                cs.setString(2, userId);
                cs.execute();
                return null;
            });

            // 3. Explicitly release the VID back to the pool!
            if (vid != null) {
                workflowEngine.updateVidStatus(vid, VoucherStatus.ACTIVE);
            }

            logAudit(userId, "CANCEL_SUCCESS", "DELETE", "Purged Batch " + batchId);
            progressService.sendProgress(batchId, userId, 100, "COMPLETED", "Batch " + batchId + " has been deleted successfully.");
            sendNotification(userId, null, "Batch " + batchId + " has been deleted successfully.", batchId);

            log.info("BATCH_PURGE_COMPLETE | Batch: {} | Duration: {}ms", batchId, System.currentTimeMillis() - start);

        } catch (DataAccessException e) {
            log.error("PURGE_DB_ERROR | Batch: {} | Status: SQL operation failed | Cause: {}", batchId, e.getMessage());

            // This safely reverts the batch to 'P' (Pending) without crashing the thread
            jdbcTemplate.update("UPDATE VOUCHER_POSTING SET BATCH_STATUS = 'P' WHERE BATCH_ID = ?", batchId);

            progressService.sendProgress(batchId, userId, 0, "FAILED", "Delete Failed: System error or missing database procedure.");
        } finally {
            MDC.clear();
        }
    }

    }

        try {
            JournalLog l = new JournalLog();
            l.setUserId(user);
            l.setActionType(action);

            l.setNewValue(val.length() > 3900 ? val.substring(0, 3900) : val);

            voucherLogRepository.save(l);
        } catch (DataAccessException e) {
            log.error("AUDIT_LOG_FAIL | User: {} | Status: Persistence error", user);
        }
    }

    @Transactional
    public String createBatchFromCacheAsync(String requestId, String commonRemarks, String creatorId, Integer creatorRole, String vid , LocalDateTime scheduledTime) {

        // FORTIFY FIX: Clean inputs before they touch Redis or Database
        String safeRequestId = SecuritySanitizer.sanitizeId(requestId);
        String safeVid = (vid != null) ? SecuritySanitizer.sanitizeId(vid) : null;

        // Log safely
        String lockKey = "LOCK_REQ_" + safeRequestId;

        Boolean acquired;

        try {
            acquired = redisTemplate.opsForValue().setIfAbsent(lockKey, new byte[0], 60, TimeUnit.MINUTES);
        } catch (DataAccessException e) {
            log.error("REDIS_LOCK_FAIL | ReqID: {} | Status: Connection to lock service failed", safeRequestId);
            throw new CustomExceptions.DatabaseOperationException("Distributed lock service unavailable", e);
        }

        if (Boolean.FALSE.equals(acquired)) {
            log.warn("CONCURRENCY_LOCKED | ReqID: {} | Status: Processing already in progress", requestId);
            throw new CustomExceptions.ConcurrencyLockException("Batch creation is already in progress for this Request ID.");
        }

        try {
            // FIX 1: Validate safeVid existence to prevent "null" crashes
            if (safeVid == null || safeVid.trim().isEmpty()) {
                throw new CustomExceptions.InvalidDataException("VID is missing from the payload. Cannot save batch.");
            }

            // --- ATOMIC LOCK FOR MANUAL POSTING ---
            String lockSql = "SELECT STATUS FROM REQUEST_VOUCHER WHERE VID = ? FOR UPDATE";
            String vidStatus;
            try {
                vidStatus = jdbcTemplate.queryForObject(lockSql, String.class, safeVid);
            } catch (EmptyResultDataAccessException e) {
                throw new CustomExceptions.InvalidDataException("VID (" + safeVid + ") does not exist.");
            }

            if (!"ACTIVE".equals(vidStatus)) {
                throw new CustomExceptions.InvalidDataException("VID (" + safeVid + ") is currently in use or locked.");
            }

             // Immediately mark as USED (using safeVid)
            workflowEngine.updateVidStatus(safeVid, VoucherStatus.USED);

            // 1. GENERATE IDs
            String batchId = sequenceService.getNextBatchId();
            String journalPrefix = sequenceService.getNextJournalIdPrefix();

            // 2. FETCH ROW COUNT (using safeRequestId)
            byte[] metaBytes = redisTemplate.opsForValue().get("DATA_META_" + safeRequestId);
            long totalRows = 0;
            if (metaBytes != null) {
                try {
                    Map<String, Object> meta = objectMapper.readValue(metaBytes, new com.fasterxml.jackson.core.type.TypeReference<>() {});
                    totalRows = ((Number) meta.get("totalRows")).longValue();
                } catch (Exception ignored) {
                }
            }

            // 3. WORKFLOW ROUTING
            String workflowCode;
            try {
                workflowCode = jdbcTemplate.queryForObject(
                        "SELECT WORKFLOW_CODE FROM WORKFLOW_TRANSITIONS WHERE ALLOWED_ROLE_ID = ? AND WORKFLOW_ORDER = 1 FETCH FIRST 1 ROWS ONLY",
                        String.class, creatorRole
                );
            } catch (EmptyResultDataAccessException e) {
                throw new CustomExceptions.InvalidDataException("Configuration Error: No workflow mapped for Role ID " + creatorRole);
            }

            Long firstApproverWorkflowId;
            try {
                firstApproverWorkflowId = jdbcTemplate.queryForObject(
                        "SELECT WORKFLOW_ID FROM WORKFLOW_TRANSITIONS WHERE WORKFLOW_CODE = ? AND WORKFLOW_ORDER = 2",
                        Long.class, workflowCode
                );
            } catch (EmptyResultDataAccessException e) {
                throw new CustomExceptions.InvalidDataException("Configuration Error: Missing Step 2 for " + workflowCode);
            }

            // 4. SYNCHRONOUS MASTER RECORD CREATION
            VoucherPosting posting = new VoucherPosting();
            posting.setBatchId(batchId);
            posting.setVidFk(safeVid);
            posting.setCommonBatchRemarks(commonRemarks);
            posting.setCreatorRole(creatorRole);
            posting.setCreatedBy(creatorId);
            posting.setWorkflowCode(workflowCode);
            posting.setPendingAt(firstApproverWorkflowId);
            posting.setBatchStatus("S");
            posting.setRecordsCount(totalRows);

            // Save the schedule directly from Step 3 UI
            if (scheduledTime != null) {
                workflowEngine.validateScheduledTime(scheduledTime, false);
                posting.setIS_EXECUTION_SCHEDULED("YES");
                posting.setScheduledTime(scheduledTime);
            } else {
                posting.setIS_EXECUTION_SCHEDULED("NO");
                posting.setScheduledTime(null);
            }
            posting.setIsScheduleOverride("NO");
            posting.setRecordsCount(totalRows);

            voucherPostingRepository.save(posting);
            String action = (scheduledTime != null) ? "Submitted (Scheduled)" : "Submitted";
            workflowEngine.logWorkflowAction(batchId, safeVid, action, creatorId, commonRemarks, scheduledTime); // Use safeVid here
            progressService.sendProgress(batchId, creatorId, 0, "PROCESSING", "Initializing Batch Creation...");

            // 6. ASYNC HANDOFF (Pass to background thread ONLY after DB commit is successful)
            TransactionSynchronizationManager.registerSynchronization(new TransactionSynchronization() {
                @Override
                public void afterCommit() {
                    // Pass the safe variables to the async worker
                    asyncWorker.executeAsyncBatchCreation(batchId, journalPrefix, safeRequestId, commonRemarks, creatorId, creatorRole, safeVid);
                }
            });

            return batchId;


        } catch (Exception e) {
            // ==============================================================================
            // THE FIX: IF ANY VALIDATION FAILS BEFORE REACHING THE ASYNC WORKER,
            // WE MUST MANUALLY DELETE THE ORPHANED LOCK HERE SO THE USER CAN TRY AGAIN!
            // ==============================================================================
            redisTemplate.delete(lockKey);

            if (e instanceof org.springframework.dao.DataIntegrityViolationException) {
                log.error("DB_CONSTRAINT_FAIL | ReqID: {} | Error: {}", safeRequestId, ((org.springframework.dao.DataIntegrityViolationException) e).getRootCause() != null ? ((org.springframework.dao.DataIntegrityViolationException) e).getRootCause().getMessage() : e.getMessage());
                throw new CustomExceptions.InvalidDataException("Database constraint failed. Ensure the VID is not duplicated or missing.");
            } else if (e instanceof DataAccessException) {
                log.error("SEQUENCE_GEN_FAIL | ReqID: {} | Status: Failed to generate sequence", safeRequestId);
                throw new CustomExceptions.DatabaseOperationException("Failed to generate Batch ID sequence", e);
            }

            // Rethrow the 30-minute time validation error or any other exception back to the UI
            throw e;
        }
        finally {
            // --- NEW: MICROSECOND RELEASE ---
            // Free the VID instantly whether the save succeeded or crashed.
            // If it succeeded, the DB is already marked as 'USED', so it's safe to drop the Redis lock.
            if (safeVid != null) {
                releaseVidLock(safeVid, creatorId); // Release the safeVid
            }
        }
    }

    @Transactional
    public List<JournalRequest> createBatchRequest(BatchPostingDto dto, String creatorId, Integer creatorRole) {

        if (dto.getVid() == null || dto.getVid().trim().isEmpty()) {
            throw new CustomExceptions.InvalidDataException("VID is missing from the payload. Cannot save batch.");
        }

        // NOTE: The old LOCK_MANUAL_VID_ redis logic was completely removed from here.
        // The UI lock now handles concurrency safely.

        try {
            String batchId = sequenceService.getNextBatchId();
            String journalPrefix = sequenceService.getNextJournalIdPrefix();

            // Checks if the VID is already used by an active/pending batch, ignoring rejected ones
            Integer vidExists = jdbcTemplate.queryForObject(
                    "SELECT COUNT(*) FROM VOUCHER_POSTING WHERE VID_FK = ? AND BATCH_STATUS NOT IN ('R', 'REJECTED', 'ERROR')",
                    Integer.class, dto.getVid()
            );

            if (vidExists != null && vidExists > 0) {
                throw new CustomExceptions.InvalidDataException("Voucher ID (" + dto.getVid() + ") is currently in use by an active batch. Please select a fresh VID.");
            }

            List<JournalRequest> entities = new ArrayList<>();
            Map<String, BigDecimal> balancingMap = new java.util.HashMap<>();
            int rowIndex = 1;

            for (BatchPostingDto.JournalRequestRow row : dto.getRows()) {
                JournalRequest entity = new JournalRequest();
                entity.setRequestStatus(RequestStatus.PENDING);
                entity.setRequestDate(LocalDateTime.now());
                entity.setCreatorId(creatorId);
                entity.setBatchId(batchId);

                String journalId = journalPrefix + "-" + rowIndex;
                entity.setJournalId(journalId);
                entity.setBranchCode(row.getBranch());
                entity.setCurrency(row.getCurrency());

                String cgl = row.getCgl() != null ? row.getCgl().trim() : "";
                entity.setCgl(cgl);

                BigDecimal amount = row.getAmount();

                if (amount == null || amount.compareTo(BigDecimal.ZERO) == 0) {
                    throw new CustomExceptions.InvalidDataException("Transaction amount cannot be zero.");
                }

                entity.setAmount(amount);
                entity.setNarration(row.getRemarks());
                entity.setProductCode(row.getProductType());

                // --- STRICT BACKEND BALANCING RULE ---
                String cglType = cgl.startsWith("5") ? "MEMO" : "NORMAL";
                String compositeKey = row.getBranch() + " - " + row.getCurrency() + " - " + cglType;
                balancingMap.put(compositeKey, balancingMap.getOrDefault(compositeKey, BigDecimal.ZERO).add(amount));

                entities.add(entity);
                rowIndex++;
            }

            // --- ENFORCE BALANCE ---
            for (Map.Entry<String, BigDecimal> entry : balancingMap.entrySet()) {
                if (entry.getValue().compareTo(BigDecimal.ZERO) != 0) {
                    throw new CustomExceptions.InvalidDataException(
                            "Balancing rule failed for [" + entry.getKey() + "]. Total Debits and Credits must exactly match."
                    );
                }
            }

            List<JournalRequest> savedEntities = journalRequestRepository.saveAll(entities);

            // ==========================================
            // 2. DYNAMIC WORKFLOW INITIALIZATION
            // ==========================================
            String workflowCode;
            try {
                workflowCode = jdbcTemplate.queryForObject(
                        "SELECT WORKFLOW_CODE FROM WORKFLOW_TRANSITIONS WHERE ALLOWED_ROLE_ID = ? AND WORKFLOW_ORDER = 1 FETCH FIRST 1 ROWS ONLY",
                        String.class, creatorRole
                );
            } catch (EmptyResultDataAccessException e) {
                throw new CustomExceptions.InvalidDataException("Configuration Error: No workflow mapped for Role ID " + creatorRole);
            }

            Map<String, Object> step2Info;
            try {
                step2Info = jdbcTemplate.queryForMap(
                        "SELECT WORKFLOW_ID, ALLOWED_ROLE_ID, WORKFLOW_STAGE  FROM WORKFLOW_TRANSITIONS WHERE WORKFLOW_CODE = ? AND WORKFLOW_ORDER = 2",
                        workflowCode
                );
            } catch (EmptyResultDataAccessException e) {
                throw new CustomExceptions.InvalidDataException("Configuration Error: Missing Step 2 for " + workflowCode);
            }

            Long firstApproverWorkflowId = ((Number) step2Info.get("WORKFLOW_ID")).longValue();

            VoucherPosting posting = new VoucherPosting();
            posting.setBatchId(batchId);
            posting.setVidFk(dto.getVid());
            posting.setCreatedBy(creatorId);
            posting.setPendingAt(firstApproverWorkflowId);
            posting.setBatchStatus("P"); // 'P' = Pending
            posting.setWorkflowCode(workflowCode);
            posting.setCommonBatchRemarks(dto.getCommonBatchRemarks());
            posting.setCreatorRole(creatorRole);

            if (dto.getRows() != null) {
                posting.setRecordsCount((long) dto.getRows().size());
            }

            // Handle Maker Scheduling
            if (dto.getScheduledTime() != null) {
                workflowEngine.validateScheduledTime(dto.getScheduledTime() , false);
                posting.setIS_EXECUTION_SCHEDULED("YES");
                posting.setScheduledTime(dto.getScheduledTime());
                posting.setIsScheduleOverride("NO");
            } else {
                posting.setIS_EXECUTION_SCHEDULED("NO");
                posting.setIsScheduleOverride("NO");
            }

            voucherPostingRepository.save(posting);
            logAudit(creatorId, "BATCH_MANUAL", "ADD", "Manual Batch " + batchId + " Created under " + workflowCode);

            if (dto.getVid() != null && !dto.getVid().isEmpty()) {
                workflowEngine.updateVidStatus(dto.getVid(), VoucherStatus.USED);
            }

            String action = (dto.getScheduledTime() != null) ? "Submitted (Scheduled)" : "Submitted";
            workflowEngine.logWorkflowAction(batchId, dto.getVid(), action, creatorId, dto.getCommonBatchRemarks(), dto.getScheduledTime());

            return savedEntities;

        } finally {
            // --- NEW: MICROSECOND RELEASE ---
            // Drops the UI-driven lock instantly so if validation failed, the VID is freed for others.
            releaseVidLock(dto.getVid(), creatorId);
        }
    }

    public void deleteBatchRequest(String batchId, String userId) {

        // FORTIFY FIX: Clean the batch ID right at the top
        String safeBatchId = com.fincore.VoucherService.config.SecuritySanitizer.sanitizeId(batchId);

        // 1. Fetch the batch (Using the safe ID)
        VoucherPosting batch = voucherPostingRepository.findById(safeBatchId)
                .orElseThrow(() -> new CustomExceptions.InvalidDataException("Batch not found."));

        // 2. SECURITY CHECK: Only the creator can delete it
        if (!batch.getCreatedBy().equalsIgnoreCase(userId)) {
            log.warn("SECURITY_VIOLATION | User {} attempted to delete batch {} owned by {}", userId, safeBatchId, batch.getCreatedBy());
            throw new IllegalStateException("You do not have permission to delete this batch.");
        }

        // 3. STATE CHECK: Can only delete if it is still PENDING ('P')
        if (!"P".equalsIgnoreCase(batch.getBatchStatus())) {
            throw new IllegalStateException("Only Pending batches can be deleted. Current status: " + batch.getBatchStatus());
        }


        // 4. STRICT WORKFLOW ORDER CHECK: Ensure it is exactly at Order 2
        String orderSql = """
            SELECT wt.WORKFLOW_ORDER 
            FROM VOUCHER_POSTING vp 
            JOIN WORKFLOW_TRANSITIONS wt ON vp.PENDING_AT = wt.WORKFLOW_ID 
            WHERE vp.BATCH_ID = ?
        """;
        Integer currentOrder;
        try {
            // Pass the safe ID to the Database
            currentOrder = jdbcTemplate.queryForObject(orderSql, Integer.class, safeBatchId);
        } catch (EmptyResultDataAccessException e) {
            currentOrder = -1; // Failsafe
        }

        if (currentOrder == null || currentOrder != 2) {
            throw new IllegalStateException("Batch cannot be deleted as it has already been processed or moved from the initial stage.");
        }
         // 6. Hand off to the background thread to do the heavy database deletion
        // Pass the safe ID here too!
        asyncWorker.executeAsyncBatchCancellation(safeBatchId, userId);
    }

    public void cancelMyRequest(Long requestId, String userId) {
        JournalRequest request = journalRequestRepository.findById(requestId)
                .orElseThrow(() -> new RuntimeException("Request not found: " + requestId)); // Replace RuntimeException with your custom ResourceNotFoundException if you have one

        if (!request.getCreatorId().equalsIgnoreCase(userId)) {
            log.warn("UNAUTHORIZED_DELETE | User: {} | RequestOwner: {} | ReqID: {}", userId, request.getCreatorId(), requestId);
            throw new IllegalStateException("You can only cancel your own requests.");
        }

        if (request.getRequestStatus() != RequestStatus.PENDING) {
            throw new IllegalStateException("Only PENDING requests can be cancelled. Current: " + request.getRequestStatus());
        }

        journalRequestRepository.delete(request);
        log.info("REQUEST_CANCELLED | User: {} | ReqID: {}", userId, requestId);
    }

    @Transactional
    public void updateBatchSchedule(String batchId, String scheduledTime) {
        Integer exists = jdbcTemplate.queryForObject("SELECT COUNT(*) FROM VOUCHER_POSTING WHERE BATCH_ID = ?", Integer.class, batchId);
        if (exists == null || exists == 0) {
            throw new CustomExceptions.InvalidDataException("The batch is still being initialized. Please wait a few seconds.");
        }

        // --- ADD THIS LOCK ---
        VoucherPosting posting = voucherPostingRepository.findById(batchId)
                .orElseThrow(() -> new CustomExceptions.InvalidDataException("Batch not found."));

        if (posting.getVidFk() != null) {
            // This is where you flip it to USED at the final moment
            workflowEngine.updateVidStatus(posting.getVidFk(), VoucherStatus.USED);
        }


        if (scheduledTime != null && !scheduledTime.isEmpty()) {
            LocalDateTime parsedTime = LocalDateTime.parse(scheduledTime, java.time.format.DateTimeFormatter.ofPattern("yyyy-MM-dd HH:mm:ss"));
            workflowEngine.validateScheduledTime(parsedTime , false);

            jdbcTemplate.update("UPDATE VOUCHER_POSTING SET SCHEDULED_TIME = TO_TIMESTAMP(?, 'YYYY-MM-DD HH24:MI:SS'), IS_SCHEDULE_OVERRIDE = 'NO', IS_EXECUTION_SCHEDULED = 'YES' WHERE BATCH_ID = ?", scheduledTime, batchId);
            jdbcTemplate.update("UPDATE VOUCHER_WORKFLOW SET LAST_ACTION = 'Submitted (Scheduled)', SCHEDULED_TIME = TO_TIMESTAMP(?, 'YYYY-MM-DD HH24:MI:SS') WHERE BATCH_ID_FK = ? AND LAST_ACTION LIKE 'Submit%'", scheduledTime, batchId);
        }

        // SMART DOUBLE-LOCK LOGIC:
        // Check if the AsyncWorker is still writing data. (If actual rows match requested rows, it's done!)
        String checkDataLoadedSql = """
            SELECT CASE WHEN p.RECORDS_COUNT = (SELECT COUNT(*) FROM VOUCHER_POSTING_DATA d WHERE d.BATCH_ID = p.BATCH_ID) 
            THEN 1 ELSE 0 END FROM VOUCHER_POSTING p WHERE p.BATCH_ID = ?
        """;

        Integer isDataLoaded = jdbcTemplate.queryForObject(checkDataLoadedSql, Integer.class, batchId);

        if (isDataLoaded != null && isDataLoaded == 1) {
            // 1. Data is fully loaded. Safe to show to checkers!
            jdbcTemplate.update("UPDATE VOUCHER_POSTING SET BATCH_STATUS = 'P' WHERE BATCH_ID = ?", batchId);

            // 2. FETCH WORKFLOW DETAILS FOR NOTIFICATION
            // 2. DYNAMIC NOTIFICATION TRIGGER
            // This is the only place it happens now!
            try {
                // REPLACEMENT: Changed DESIGNATION to WORKFLOW_STAGE
                String wfSql = """
                    SELECT wt.ALLOWED_ROLE_ID, wt.WORKFLOW_STAGE 
                    FROM VOUCHER_POSTING vp 
                    JOIN WORKFLOW_TRANSITIONS wt ON vp.PENDING_AT = wt.WORKFLOW_ID 
                    WHERE vp.BATCH_ID = ? AND wt.WORKFLOW_ORDER = 2
                """;

                Map<String, Object> step2Info = jdbcTemplate.queryForMap(wfSql, batchId);


                String nextApproverRoleId = String.valueOf(step2Info.get("ALLOWED_ROLE_ID"));
                String stageName = (String) step2Info.get("WORKFLOW_STAGE"); // Update key here too

                // 3. SEND THE NOTIFICATION TO THE CHECKERS NOW
                notificationWriterService.createNextStepApprovalNotification(batchId, nextApproverRoleId, stageName);

            } catch (EmptyResultDataAccessException e) {
                log.error("Could not find Step 2 workflow configuration for Batch {} to send notification.", batchId);
            }
        } else {
            // Data is still loading, do not notify yet.
            // Note: Your AsyncWorker MUST also call this notification logic
            // once it finishes its final row insertion and performs the BATCH_STATUS = 'P' update.
            log.info("Schedule updated for Batch {}, data still loading. Notification deferred.", batchId);
        }

    }

    // ==================================================================================

    public void acquireVidLock(String vid, String userId) {

        // FORTIFY FIX: Access Control Database
        String safeVid = SecuritySanitizer.sanitizeId(vid);
        String lockKey = "LOCK_VID_" + safeVid;
        byte[] userBytes = userId.getBytes(java.nio.charset.StandardCharsets.UTF_8);

        // Atomic SETNX with 30-minute TTL
        Boolean acquired = redisTemplate.opsForValue().setIfAbsent(lockKey, userBytes, 30, TimeUnit.MINUTES);

        if (Boolean.FALSE.equals(acquired)) {
            throw new CustomExceptions.ConcurrencyLockException("This VID is currently being processed by another user.");
        }

        // Secondary safeguard: Ensure the DB hasn't already marked it as USED before the lock was acquired
        String dbStatus = jdbcTemplate.queryForObject("SELECT STATUS FROM REQUEST_VOUCHER WHERE VID = ?", String.class, vid);
        if (!"ACTIVE".equalsIgnoreCase(dbStatus)) {
            redisTemplate.delete(lockKey); // Release immediately if DB is dirty
            throw new CustomExceptions.ConcurrencyLockException("This VID has already been used. Please select a different VID.");
        }
    }

    public void releaseVidLock(String vid, String userId) {
        // FORTIFY FIX: Access Control Database
        String safeVid = com.fincore.VoucherService.config.SecuritySanitizer.sanitizeId(vid);
        String lockKey = "LOCK_VID_" + safeVid;

        byte[] existingUserBytes = redisTemplate.opsForValue().get(lockKey);

        if (existingUserBytes != null) {
            String lockedBy = new String(existingUserBytes);
            if (lockedBy.equals(userId)) {
                redisTemplate.delete(lockKey);
            }
        }
    }

    private String calculateTimeSinceLastAction(String batchId) {
        Optional<VoucherWorkflow> latestAction =
                workflowRepository.findTopByBatchIdFkOrderByLastActionAtDesc(batchId);

        if (latestAction.isEmpty() || latestAction.get().getLastActionAt() == null) {
            return "0h 0m";
        }

        Duration duration = Duration.between(latestAction.get().getLastActionAt(), LocalDateTime.now());
        return duration.toHours() + "h " + duration.toMinutesPart() + "m";
    }


    public Map<String, Object> getPendingBatchesForChecker(Integer userRole, String userId, int page, int size, String batchIdFilter, String vidFilter) {


        // 1. EXACT ORIGINAL SQL - COMPLETELY UNTOUCHED
        String sql = """
            SELECT 
                p.BATCH_ID, 
                p.VID_FK AS VID, 
                p.CREATED_BY AS CREATOR_ID, 
                p.CREATED_AT AS REQ_DATE, 
                p.COMMON_BATCH_REMARKS AS BATCH_REMARKS,
                p.RECORDS_COUNT AS TOTAL_ROWS, 
                p.SCHEDULED_TIME,
                w.WORKFLOW_STAGE, 
                w.WORKFLOW_CODE, 
                w.ESCALATION_TIME, 
                w.WORKFLOW_ORDER, 
                w.IS_FINAL_APPROVAL,
                COALESCE(u.FIRST_NAME || ' ' || u.LAST_NAME, p.CREATED_BY) AS CREATOR_NAME
            FROM VOUCHER_POSTING p
            INNER JOIN WORKFLOW_TRANSITIONS w ON p.PENDING_AT = w.WORKFLOW_ID
            LEFT JOIN USERS u ON p.CREATED_BY = u.USER_ID
            WHERE w.ALLOWED_ROLE_ID = ? AND p.BATCH_STATUS = 'P'
            AND NOT EXISTS (
                SELECT 1 FROM VOUCHER_WORKFLOW vw 
                WHERE vw.BATCH_ID_FK = p.BATCH_ID AND vw.LAST_ACTION_BY = ?
            )
            ORDER BY p.CREATED_AT DESC
        """;

        // Fetch all rows using original logic
        List<Map<String, Object>> allPending = jdbcTemplate.query(sql, (rs, rowNum) -> {
            Map<String, Object> map = new HashMap<>();
            map.put("batchId", rs.getString("BATCH_ID"));
            map.put("vid", rs.getString("VID"));
            map.put("creatorId", rs.getString("CREATOR_ID"));
            map.put("creatorName", rs.getString("CREATOR_NAME"));
            map.put("requestDate", rs.getTimestamp("REQ_DATE"));
            map.put("commonBatchRemarks", rs.getString("BATCH_REMARKS"));
            map.put("requestCount", rs.getLong("TOTAL_ROWS"));
            map.put("escalationTime", rs.getLong("ESCALATION_TIME"));
            map.put("workflowState", rs.getString("WORKFLOW_STAGE"));
            map.put("approvalWorkflow", rs.getString("WORKFLOW_CODE"));
            map.put("currentOrder", rs.getObject("WORKFLOW_ORDER"));
            map.put("isFinalApproval", rs.getString("IS_FINAL_APPROVAL"));
            return map;
        }, userRole, userId);

        // 2. JAVA-SIDE FILTERING (Bypasses Redaction Errors)
        List<Map<String, Object>> filteredList = allPending.stream().filter(row -> {
            boolean matchesBatch = true;
            boolean matchesVid = true;
            if (batchIdFilter != null && !batchIdFilter.trim().isEmpty()) {
                String bId = (String) row.get("batchId");
                matchesBatch = bId != null && bId.toLowerCase().contains(batchIdFilter.trim().toLowerCase());
            }
            if (vidFilter != null && !vidFilter.trim().isEmpty()) {
                String vId = (String) row.get("vid");
                matchesVid = vId != null && vId.toLowerCase().contains(vidFilter.trim().toLowerCase());
            }
            return matchesBatch && matchesVid;
        }).toList();

        // 3. JAVA-SIDE PAGINATION
        int totalElements = filteredList.size();
        int fromIndex = Math.min(page * size, totalElements);
        int toIndex = Math.min(fromIndex + size, totalElements);
        List<Map<String, Object>> pagedList = filteredList.subList(fromIndex, toIndex);

        // 4. CALCULATE TIME (Only for the paginated subset to save performance)
        for (Map<String, Object> row : pagedList) {
            row.put("timeSinceLastAction", calculateTimeSinceLastAction((String) row.get("batchId")));
        }

        Map<String, Object> response = new HashMap<>();
        response.put("content", pagedList);
        response.put("totalElements", totalElements);
        response.put("totalPages", (int) Math.ceil((double) totalElements / size));
        return response;

    }


    public Map<String, Object> getAllRequestsSummary(String userId, Integer roleId, int page, int size, String batchIdFilter, String vidFilter) {

        // 1. DYNAMIC WHERE CLAUSE BUILDER
        // Wrapped the base conditions in parentheses so AND operators apply correctly
        StringBuilder whereClause = new StringBuilder("""
            WHERE (p.CREATED_BY = ? 
               OR EXISTS (
                   SELECT 1 FROM WORKFLOW_TRANSITIONS wt 
                   WHERE wt.WORKFLOW_CODE = p.WORKFLOW_CODE 
                   AND wt.ALLOWED_ROLE_ID = ? 
                   AND wt.WORKFLOW_ORDER > 1
               ))
        """);

        List<Object> paramsList = new java.util.ArrayList<>();
        paramsList.add(userId);
        paramsList.add(roleId);

        // Append Batch ID search if provided
        if (batchIdFilter != null && !batchIdFilter.trim().isEmpty()) {
            whereClause.append(" AND p.BATCH_ID LIKE ? ");
            paramsList.add("%" + batchIdFilter.trim() + "%");
        }

        // Append VID search if provided (Case insensitive)
        if (vidFilter != null && !vidFilter.trim().isEmpty()) {
            whereClause.append(" AND LOWER(p.VID_FK) LIKE LOWER(?) ");
            paramsList.add("%" + vidFilter.trim() + "%");
        }

        // 2. FAST COUNT QUERY
        String countSql = "SELECT COUNT(*) FROM VOUCHER_POSTING p " + whereClause.toString();
        Long totalElements = jdbcTemplate.queryForObject(countSql, Long.class, paramsList.toArray());
        if (totalElements == null) totalElements = 0L;

        // 3. PAGINATED DATA QUERY
        int offset = page * size;
        String dataSql = """
            SELECT 
                p.BATCH_ID, 
                p.VID_FK,
                p.CREATED_BY AS CREATOR_ID, 
                u.FIRST_NAME,
                u.LAST_NAME,
                p.CREATED_AT AS REQ_DATE, 
                p.RECORDS_COUNT AS TOTAL_ROWS, 
                p.BATCH_STATUS, 
                w.WORKFLOW_STAGE, 
                w.WORKFLOW_CODE,
                w.WORKFLOW_ORDER
            FROM VOUCHER_POSTING p
            LEFT JOIN WORKFLOW_TRANSITIONS w ON p.PENDING_AT = w.WORKFLOW_ID
            LEFT JOIN USERS u ON p.CREATED_BY = u.USER_ID
            """ + whereClause.toString() + """
            ORDER BY p.CREATED_AT DESC, p.BATCH_ID DESC
            OFFSET ? ROWS FETCH NEXT ? ROWS ONLY
        """;

        // Add the pagination parameters to the END of the params list
        List<Object> dataParams = new java.util.ArrayList<>(paramsList);
        dataParams.add(offset);
        dataParams.add(size);

        List<Map<String, Object>> content = jdbcTemplate.query(dataSql, (rs, rowNum) -> {
            Map<String, Object> map = new java.util.HashMap<>();
            map.put("batchId", rs.getString("BATCH_ID"));
            map.put("vid", rs.getString("VID_FK"));
            map.put("creatorId", rs.getString("CREATOR_ID"));

            // JAVA-SIDE FORMATTING: Safely concatenate names in Java memory, not in the SQL engine
            String fName = rs.getString("FIRST_NAME");
            String lName = rs.getString("LAST_NAME");
            String fullName = (fName != null || lName != null)
                    ? ((fName != null ? fName : "") + " " + (lName != null ? lName : "")).trim()
                    : rs.getString("CREATOR_ID");
            map.put("creatorName", fullName);

            map.put("requestDate", rs.getTimestamp("REQ_DATE"));
            map.put("requestCount", rs.getLong("TOTAL_ROWS"));
            map.put("approvalWorkflow", rs.getString("WORKFLOW_CODE"));
            map.put("workflowState", rs.getString("WORKFLOW_STAGE"));
            map.put("currentOrder", rs.getObject("WORKFLOW_ORDER"));

            String bStatus = rs.getString("BATCH_STATUS");
            map.put("batchStatus", bStatus);

            if ("R".equals(bStatus)) {
                map.put("currentStatus", "REJECTED");
            } else if ("A".equals(bStatus) || "C".equals(bStatus)) {
                map.put("currentStatus", "POSTED");
            } else if ("Q".equals(bStatus)) {
                map.put("currentStatus", "QUEUED FOR RETRY");
            } else {
                map.put("currentStatus", rs.getString("WORKFLOW_STAGE") != null ? rs.getString("WORKFLOW_STAGE") : "PENDING");
            }

            map.put("timeSinceLastAction", calculateTimeSinceLastAction(rs.getString("BATCH_ID")));

            return map;
        }, dataParams.toArray()); // <-- Use dataParams here

        // 4. WRAP IN PAGINATION
        Map<String, Object> response = new java.util.HashMap<>();
        response.put("content", content);
        response.put("totalElements", totalElements);
        response.put("totalPages", (int) Math.ceil((double) totalElements / size));

        return response;
    }


    /**
     * Retrieves the comprehensive summary details for a specific Voucher Batch.
     * This method hydrates the "View & Action" modal in the UI, merging runtime batch data
     * with its underlying configuration blueprint (VID) in a single atomic database trip.
     */
    public Map<String, Object> getBatchSummaryDetails(String batchId) {

        log.info("BATCH_SUMMARY_FETCH | BatchID: {}", SecuritySanitizer.sanitizeForLog(batchId));


        String sql = """
            SELECT 
                -- ==========================================
                -- 1. RUNTIME BATCH DETAILS (The actual transaction submitted today)
                -- ==========================================
                vp.VID_FK AS VID,
                vp.CREATED_BY AS BATCH_CREATOR_ID,
                -- Defensive fallback: If the user was deleted from the system, return their raw ID
                COALESCE(bu.FIRST_NAME || ' ' || bu.LAST_NAME, vp.CREATED_BY) AS BATCH_CREATED_BY_NAME,
                vp.CREATED_AT AS BATCH_CREATED_AT,
                vp.CREATOR_ROLE AS CREATORS_ROLE,
                vp.SCHEDULED_TIME,
                vp.EXECUTED_AT,
                
                -- ==========================================
                -- 2. CONFIGURATION BLUEPRINT DETAILS (The VID template setup)
                -- ==========================================
                vc.CATEGORY_NAME,
                
                -- Inline Scalar Subquery: Aggregates multiple Issue Categories into a single comma-separated 
                -- string so we strictly return only 1 row for the queryForMap to consume safely.
                (
                    SELECT LISTAGG(ic.ISSUE_CATEGORY_NAME, ', ') WITHIN GROUP (ORDER BY ic.ISSUE_CATEGORY_NAME)
                    FROM REQUEST_VOUCHER_ISSUE_CATEGORY rvic
                    JOIN ISSUE_CATEGORY ic ON rvic.ISSUE_CATEGORY_ID = ic.ID
                    WHERE rvic.REQUEST_VOUCHER_ID = rv.ID
                ) AS ISSUE_CATEGORIES,
                
                rv.CREATED_BY AS VID_CREATOR_ID,
                -- Distinguishes between who created the batch vs who originally configured the VID
                COALESCE(vu.FIRST_NAME || ' ' || vu.LAST_NAME, rv.CREATED_BY) AS VID_CREATED_BY_NAME,
                rv.CREATED_AT AS VID_CREATED_AT,
                rv.DESCRIPTION
                
            FROM VOUCHER_POSTING vp
            -- Link the Batch to its Master Configuration Template
            LEFT JOIN REQUEST_VOUCHER rv ON vp.VID_FK = rv.VID
            LEFT JOIN VOUCHER_CATEGORY vc ON rv.CATEGORY_ID = vc.ID
            -- First User Join: Resolves the Maker who uploaded the file
            LEFT JOIN USERS bu ON vp.CREATED_BY = bu.USER_ID
            -- Second User Join: Resolves the Admin who created the VID
            LEFT JOIN USERS vu ON rv.CREATED_BY = vu.USER_ID
            WHERE vp.BATCH_ID = ?
        """;

        try {
            // queryForMap expects exactly 1 row. If the batchId is valid, it returns the Key-Value map.
            return jdbcTemplate.queryForMap(sql, batchId);
        } catch (EmptyResultDataAccessException e) {
            // Graceful degradation: If a bad Batch ID is passed, return an empty object
            // instead of throwing a 500 Internal Server Error, allowing the UI to handle it cleanly.
            return new HashMap<>();
        }
    }


    public List<Map<String, Object>> getWorkflowHistory(String batchId) {

        log.info("WORKFLOW_HISTORY_FETCH | BatchID: {}", SecuritySanitizer.sanitizeForLog(batchId));
        String sql = """
        SELECT 
            w.LAST_ACTION_BY, 
            COALESCE(u.FIRST_NAME || ' ' || u.LAST_NAME, w.LAST_ACTION_BY) AS ACTION_USER_NAME, 
            w.LAST_ACTION, 
            w.LAST_ACTION_AT, 
            w.REMARKS,
            w.SCHEDULED_TIME
        FROM VOUCHER_WORKFLOW w 
        LEFT JOIN USERS u ON w.LAST_ACTION_BY = u.USER_ID
        WHERE w.BATCH_ID_FK = ? 
        ORDER BY w.LAST_ACTION_AT ASC
        """;

        List<Map<String, Object>> history = jdbcTemplate.queryForList(sql, batchId);

        // ========================================================
        // THIS IS WHERE THE TIME IS CALCULATED AS WE DECIDED!
        // ========================================================
        for (int i = 0; i < history.size() - 1; i++) {
            LocalDateTime start = ((Timestamp) history.get(i).get("LAST_ACTION_AT")).toLocalDateTime();
            LocalDateTime end = ((Timestamp) history.get(i + 1).get("LAST_ACTION_AT")).toLocalDateTime();

            // Mathematically calculating the difference between Step A and Step B
            Duration d = Duration.between(start, end);
            String durationStr = d.toHours() + "h " + d.toMinutesPart() + "m";

            // Injecting the calculated time directly into the History object so React can easily read 'historyItem.stageDuration'
            history.get(i + 1).put("stageDuration", durationStr);
        }

        return history;
    }

    /**
     * Retrieves all Active Voucher Configurations that a specific role is allowed to initiate.
     * Uses a highly optimized "Fetch & Merge" pattern to prevent N+1 database queries
     * and handle 1-to-Many relationships without returning duplicate rows.
     */
    public List<Map<String, Object>> getAvailableVouchersVid(Long roleId) {

        // ==================================================================================
        // STEP 1: FETCH CORE VOUCHER DATA
        // Purpose: Get the distinct master records and basic details in a single query.
        // Note: We use LEFT JOINs so the voucher is still returned even if user/category data is missing.
        // ==================================================================================
        // ==================================================================================
        // STEP 1: FETCH CORE VOUCHER DATA
        // Purpose: Get the distinct master records and basic details in a single query.
        // Note: We use LEFT JOINs so the voucher is still returned even if user/category data is missing.
        // ==================================================================================
        String sql = """
            SELECT 
                rv.ID,
                rv.VID,
                rv.DESCRIPTION,
                c.CATEGORY_NAME,
                rv.CREATED_BY AS CREATOR_ID,
                -- Fallback: If user isn't found in USERS table, just return their raw ID
                COALESCE(u.FIRST_NAME || ' ' || u.LAST_NAME, rv.CREATED_BY) AS CREATOR_NAME,
                rv.CREATED_AT
            FROM REQUEST_VOUCHER rv
            LEFT JOIN VOUCHER_CATEGORY c ON rv.CATEGORY_ID = c.ID
            LEFT JOIN USERS u ON rv.CREATED_BY = u.USER_ID
            WHERE rv.ALLOWED_ROLE = ? AND rv.STATUS = 'ACTIVE'
            
            -- FIX: THIS ENSURES VIDS CURRENTLY ATTACHED TO LIVE BATCHES ARE COMPLETELY HIDDEN FROM THE UI
            AND NOT EXISTS (
                SELECT 1 FROM VOUCHER_POSTING vp 
                WHERE vp.VID_FK = rv.VID 
                AND vp.BATCH_STATUS NOT IN ('R', 'REJECTED', 'ERROR')
            )
            
            ORDER BY rv.CREATED_AT DESC
        """;

        // Manually mapping
        List<Map<String, Object>> vouchers = jdbcTemplate.query(sql, (rs, rowNum) -> {
            Map<String, Object> map = new HashMap<>();
            map.put("id", rs.getLong("ID"));
            map.put("vid", rs.getString("VID"));
            map.put("description", rs.getString("DESCRIPTION"));
            map.put("categoryName", rs.getString("CATEGORY_NAME"));
            map.put("createdBy", rs.getString("CREATOR_ID"));
            map.put("createdByName", rs.getString("CREATOR_NAME"));

            java.sql.Timestamp ts = rs.getTimestamp("CREATED_AT");
            map.put("createdAt", ts != null ? ts.toInstant().toString() : null);
            return map;
        }, roleId);

        // Early Exit: If no vouchers are found, return immediately to save a DB call.
        if (vouchers.isEmpty()) {
            return vouchers;
        }

        // ==================================================================================
        // STEP 2: FETCH 1-TO-MANY RELATIONSHIPS (ISSUE CATEGORIES) IN BULK
        // Purpose: Avoid the N+1 query problem. Instead of querying categories 1-by-1 in a loop,
        // we extract all the IDs and ask the database for everything at once.
        // ==================================================================================
        List<Long> voucherIds = vouchers.stream()
                .map(v -> (Long) v.get("id"))
                .toList();

        // Dynamically build the "?, ?, ?" string for the SQL IN clause based on the list size
        String inClause = String.join(",", Collections.nCopies(voucherIds.size(), "?"));

        // Oracle's LISTAGG function groups multiple rows into a single comma-separated string
        // (e.g., "Category A, Category B"). This prevents Java from having to do string manipulation.
        String issueSql = String.format("""
            SELECT 
                rvic.REQUEST_VOUCHER_ID, 
                LISTAGG(ic.ISSUE_CATEGORY_NAME, ', ') WITHIN GROUP (ORDER BY ic.ISSUE_CATEGORY_NAME) AS CATEGORIES
            FROM REQUEST_VOUCHER_ISSUE_CATEGORY rvic
            JOIN ISSUE_CATEGORY ic ON rvic.ISSUE_CATEGORY_ID = ic.ID
            WHERE rvic.REQUEST_VOUCHER_ID IN (%s)
            GROUP BY rvic.REQUEST_VOUCHER_ID
        """, inClause);

        // Store the results in a Map where Key = Voucher ID, Value = "CatA, CatB"
        Map<Long, String> issueCategoryMap = new HashMap<>();
        jdbcTemplate.query(issueSql, rs -> {
            issueCategoryMap.put(rs.getLong("REQUEST_VOUCHER_ID"), rs.getString("CATEGORIES"));
        }, voucherIds.toArray());

        // ==================================================================================
        // STEP 3: STITCH DATA TOGETHER IN MEMORY
        // Purpose: Attach the category strings back to their original voucher objects.
        // ==================================================================================
        for (Map<String, Object> voucher : vouchers) {
            Long vId = (Long) voucher.get("id");

            // If the voucher has no issue categories, default to "N/A" so the React UI
            // doesn't render an empty blank space.
            voucher.put("issueCategories", issueCategoryMap.getOrDefault(vId, "N/A"));
        }

        return vouchers;
    }

    @Transactional
    public void processWorkflowAction(ProcessVoucherRequestDto dto, String executorId, Integer executorRole) {
        String batchId = dto.getBatchId();

        // --- STRICT ACTION REMARKS ENFORCEMENT ---
        if (dto.getRemarks() == null || dto.getRemarks().trim().length() < 2 || dto.getRemarks().trim().length() > 30) {
            throw new CustomExceptions.InvalidDataException("Action remarks are mandatory and must be between 2 and 30 alphanumeric characters.");
        }
        // Strips out everything except Alphanumeric and Spaces, then reduces multiple spaces to a single space
        String sanitizedRemarks = dto.getRemarks().replaceAll("[^a-zA-Z0-9 ]", "").trim().replaceAll("\\s{2,}", " ");

        dto.setRemarks(sanitizedRemarks);

        // Gathers the Batch data, current Workflow Step configuration, and original VID creator
        // in a single  SQL query to minimize database hits.
        Map<String, Object> stepData;
        try {
            stepData = jdbcTemplate.queryForMap("""
                SELECT 
                    p.CREATED_BY AS BATCH_CREATOR, p.VID_FK, p.IS_EXECUTION_SCHEDULED, p.SCHEDULED_TIME,
                    w.WORKFLOW_CODE, w.WORKFLOW_ORDER, w.IS_FINAL_APPROVAL, 
                    w.WORKFLOW_STAGE, w.ALLOWED_ROLE_ID, w.WORKFLOW_ID,
                    rv.CREATED_BY AS VID_CREATOR
                FROM VOUCHER_POSTING p
                JOIN WORKFLOW_TRANSITIONS w ON p.PENDING_AT = w.WORKFLOW_ID
                LEFT JOIN REQUEST_VOUCHER rv ON p.VID_FK = rv.VID
                WHERE p.BATCH_ID = ?
                """, batchId);
        } catch (EmptyResultDataAccessException e) {
            throw new CustomExceptions.InvalidDataException("Batch ID not found or workflow stage invalid.");
        }

        // --- FIX: DECLARE THE VARIABLES HERE ---
        String batchCreatorId = (String) stepData.get("BATCH_CREATOR");
        String vidCreatorId = (String) stepData.get("VID_CREATOR");
        String vid = (String) stepData.get("VID_FK");

        // 3A. Segregation of Duties (Maker cannot be the Checker)
        if (executorId != null && batchCreatorId != null && executorId.trim().equalsIgnoreCase(batchCreatorId.trim())) {
            throw new CustomExceptions.InvalidDataException("Segregation of Duties violation: You cannot approve a batch you created.");
        }

        // Idempotency & Multi-Level Rule (A user can only approve a batch once per lifecycle)
        boolean hasAlreadyApproved = voucherWorkflowRepository.findByBatchIdFkOrderByLastActionAtAsc(batchId)
                .stream().anyMatch(w -> executorId.equalsIgnoreCase(w.getLastActionBy()));
        if (hasAlreadyApproved) {
            throw new CustomExceptions.InvalidDataException("You have already approved this batch.");
        }


        // Role Validation (Ensures the user holds the exact role required for this specific step)
        Integer allowedRoleId = ((Number) stepData.get("ALLOWED_ROLE_ID")).intValue();
        if (!allowedRoleId.equals(executorRole)) {
            throw new CustomExceptions.InvalidDataException("Unauthorized: Role mismatch.");
        }

        // Extract variables for routing logic
        String workflowCode = (String) stepData.get("WORKFLOW_CODE");
        Integer currentOrder = ((Number) stepData.get("WORKFLOW_ORDER")).intValue();
        String currentStageName = (String) stepData.get("WORKFLOW_STAGE");
        boolean isFinalStep = "Y".equalsIgnoreCase((String) stepData.get("IS_FINAL_APPROVAL"));

        VoucherPosting posting = voucherPostingRepository.findById(batchId).orElseThrow();

        // 4. Handle REJECTION
        if ("REJECTED".equals(dto.getStatus().toString())) {
            posting.setBatchStatus("R");
            voucherPostingRepository.save(posting);

            logWorkflowAction(batchId, vid, "Rejected (" + currentStageName + ")", executorId, dto.getRemarks(), null);

            // Update underlying tables to mark the entire batch as dead
            jdbcTemplate.update("UPDATE VOUCHER_POSTING SET BATCH_STATUS = 'REJECTED' WHERE BATCH_ID = ?", batchId);
            jdbcTemplate.update("UPDATE VOUCHER_POSTING_DATA SET REQ_STATUS = 'R', EXECUTION_DATE = SYSTIMESTAMP WHERE BATCH_ID = ?", batchId);

            // Build a "List" for notifications: Everyone who ever touched this batch needs to know it was rejected.
            String previousActorsSql = "SELECT DISTINCT LAST_ACTION_BY FROM VOUCHER_WORKFLOW WHERE BATCH_ID_FK = ? AND LAST_ACTION_BY != ?";
            List<String> previousActors = jdbcTemplate.queryForList(previousActorsSql, String.class, batchId, executorId);

            // Failsafe: Ensure the original creators are in the blast list
            if (!previousActors.contains(batchCreatorId)) {
                previousActors.add(batchCreatorId);
            }
            if (vidCreatorId != null && !previousActors.contains(vidCreatorId)) {
                previousActors.add(vidCreatorId);
            }

            // Fire the notifications
            notificationWriterService.createBatchRejectedNotificationForUsers(batchId, previousActors, executorId, dto.getRemarks());

            if (posting.getVidFk() != null && !posting.getVidFk().isEmpty()) updateVidStatus(posting.getVidFk(), VoucherStatus.ACTIVE);
            return;
        }

        // 5. Handle APPROVAL / PROGRESSION
        if (isFinalStep) {
            // Move to Execution Step (Order + 1)
            // The workflow is complete. Move the pointer to the hidden 'Execution' step.
            Long execStepId = jdbcTemplate.queryForObject("SELECT WORKFLOW_ID FROM WORKFLOW_TRANSITIONS WHERE WORKFLOW_CODE = ? AND WORKFLOW_ORDER = ?", Long.class, workflowCode, currentOrder + 1);
            posting.setPendingAt(execStepId);

            // Handle Final Approver Schedule Overrides
            boolean isOverride = Boolean.TRUE.equals(dto.getOverrideScheduleLapse());
            if (isOverride) {

                if (dto.getNewScheduledTime() != null) {
                    validateScheduledTime(dto.getNewScheduledTime(), true);
                }

                posting.setIsScheduleOverride("YES");
                posting.setIS_EXECUTION_SCHEDULED(dto.getNewScheduledTime() != null ? "YES" : "NO");
                posting.setScheduledTime(dto.getNewScheduledTime());
            }

            //Handle Expired Schedules (If Maker scheduled it for 12:00 PM but it's approved at 1:00 PM)
            else if ("YES".equalsIgnoreCase(posting.getIS_EXECUTION_SCHEDULED()) && posting.getScheduledTime() != null) {
                if (posting.getScheduledTime().isBefore(LocalDateTime.now())) {
                    throw new CustomExceptions.ScheduleLapsedException("The scheduled execution time has passed.");
                }
                else {
                    posting.setIsScheduleOverride("NO"); // Ensure it stays NO if not triggered
                }
            }

           // Execute Later (Timer Required)
            if ("YES".equalsIgnoreCase(posting.getIS_EXECUTION_SCHEDULED()) && posting.getScheduledTime() != null && posting.getScheduledTime().isAfter(LocalDateTime.now())) {
                posting.setBatchStatus("P");
                String actionLog = isOverride ? "Final Approval(Sch. Override)" : "Final Approval(Scheduled)";
                logWorkflowAction(batchId, vid, actionLog, executorId, dto.getRemarks(), posting.getScheduledTime());
                // FIX: Update VOUCHER_POSTING
                jdbcTemplate.update("UPDATE VOUCHER_POSTING SET BATCH_STATUS = 'P' WHERE BATCH_ID = ?", batchId);


            }
            // Execute Immediately (Handoff to Background Thread)
            else {
                posting.setBatchStatus("C");
                posting.setExecutedAt(LocalDateTime.now());
                String actionLog = isOverride ? "Final Approval(Imm. Override)" : "Final Approval(Posted)";
                logWorkflowAction(batchId, vid, actionLog, executorId, dto.getRemarks(), null);

                // Lock the VID permanently as completed
                if (posting.getVidFk() != null && !posting.getVidFk().isEmpty()) updateVidStatus(posting.getVidFk(), VoucherStatus.COMPLETED);

                // Dispatch the heavy DB procedure execution to the Async Worker
                BulkProcessVoucherPostingDto execDto = new BulkProcessVoucherPostingDto();
                execDto.setBatchId(batchId); execDto.setStatus(RequestStatus.ACCEPTED); execDto.setRemarks(dto.getRemarks());
                commandService.processBulkRequestsAsync(execDto, executorId, executorRole);
            }
            // --- NOTIFY BOTH CREATORS OF FINAL APPROVAL ---
            notificationWriterService.createBatchUpdateNotificationForCreators(
                    batchId, vid, batchCreatorId, vidCreatorId, "fully APPROVED", executorId
            );
        } else {
            //INTERMEDIATE APPROVAL
            // --- PERFECTED JOIN LOGIC ---
            // Joins VOUCHER_POSTING with WORKFLOW_TRANSITIONS to safely fetch the exact next routing stage
            String nextStepSql = """
                SELECT w_next.WORKFLOW_ID, w_next.ALLOWED_ROLE_ID
                FROM VOUCHER_POSTING vp
                JOIN WORKFLOW_TRANSITIONS w_curr 
                  ON vp.PENDING_AT = w_curr.WORKFLOW_ID
                JOIN WORKFLOW_TRANSITIONS w_next 
                  ON w_curr.WORKFLOW_CODE = w_next.WORKFLOW_CODE 
                 AND w_next.WORKFLOW_ORDER = (w_curr.WORKFLOW_ORDER + 1)
                WHERE vp.BATCH_ID = ?
            """;

            Map<String, Object> nextStepInfo = jdbcTemplate.queryForMap(nextStepSql, batchId);
            Long nextStepId = ((Number) nextStepInfo.get("WORKFLOW_ID")).longValue();
            String nextRoleId = String.valueOf(nextStepInfo.get("ALLOWED_ROLE_ID"));

            // Move the pointer forward
            posting.setPendingAt(nextStepId);
            logWorkflowAction(batchId, vid, "Approved (" + currentStageName + ")", executorId, dto.getRemarks(), null);


            // --- NOTIFY BOTH CREATORS OF INTERMEDIATE PROGRESS ---
            notificationWriterService.createBatchUpdateNotificationForCreators(
                    batchId, vid, batchCreatorId, vidCreatorId, "Approved (" + currentStageName + ")", executorId
            );
        }

        voucherPostingRepository.save(posting);
    }


    public void logWorkflowAction(String batchId, String vid, String action, String executorId, String remarks, LocalDateTime scheduledTime) {
        VoucherWorkflow wf = new VoucherWorkflow();
        wf.setBatchIdFk(batchId);
        wf.setVidFk(vid != null ? vid : "UNKNOWN");
        wf.setLastAction(action != null && action.length() > 30 ? action.substring(0, 30) : action);
        wf.setLastActionBy(executorId);
        wf.setLastActionAt(LocalDateTime.now());
        wf.setRemarks(remarks);
        wf.setScheduledTime(scheduledTime);
        voucherWorkflowRepository.save(wf);
    }


    public void validateScheduledTime(LocalDateTime scheduledTime, boolean isFinalApproval) {
        if (scheduledTime == null) return;

        // 1. Get "now" and strip seconds/nanos to match UI precision
        LocalDateTime now = LocalDateTime.now().withSecond(0).withNano(0);
        LocalDateTime maxTime = now.with(java.time.LocalTime.MAX);

        if (isFinalApproval) {
            // Final Approver: Can't be in the past
            if (scheduledTime.isBefore(now)) {
                throw new CustomExceptions.InvalidDataException("Scheduled time cannot be in the past.");
            }
        } else {
            // 2. APPLY 3-MINUTE NETWORK GRACE PERIOD
            // We calculate "30 minutes from now", but subtract 3 minutes
            // to allow for network latency/server processing time.
            // Effectively, we validate that it is at least 27 minutes away.
            LocalDateTime minAllowedTime = now.plusMinutes(30).minusMinutes(3);

            if (scheduledTime.isBefore(minAllowedTime)) {
                throw new CustomExceptions.InvalidDataException("Scheduled time must be at least 30 minutes from the current time.");
            }
        }

        if (scheduledTime.isAfter(maxTime)) {
            throw new CustomExceptions.InvalidDataException("Scheduled time cannot exceed today's end of day (11:59 PM).");
        }
    }


    public void updateVidStatus(String vid, VoucherStatus status) {
        // FORTIFY FIX: Access Control Database (Clean the VID before hitting the DB)
        String safeVid = SecuritySanitizer.sanitizeId(vid);

        // Your SQL and execution remain 100% untouched
        String sql = "UPDATE REQUEST_VOUCHER SET STATUS = ? WHERE VID = ?";
        jdbcTemplate.update(sql, status.name(), safeVid);
    }
