export default function JournalPostingStatus() {
  const { callApi } = useApi();
  const showSnackBar = useCustomSnackbar();

  const [loading, setLoading] = useState(true);
  const [rows, setRows] = useState([]);

  // Modal State
  const [isModalOpen, setIsModalOpen] = useState(false);
  const [isSubmitting, setIsSubmitting] = useState(false);
  const [selectedBatch, setSelectedBatch] = useState(null);

  // View Details State
  const [isViewModalOpen, setIsViewModalOpen] = useState(false);
  const [isViewLoading, setIsViewLoading] = useState(false);
  const [batchDetails, setBatchDetails] = useState([]);
  const [viewingBatchId, setViewingBatchId] = useState(null);
  const [viewingBatchStatus, setViewingBatchStatus] = useState(null);

  // Partial Action State
  const [selectedJournalPrefixes, setSelectedJournalPrefixes] = useState([]);
  const [isPartialSubmitting, setIsPartialSubmitting] = useState(false);

  // --- FETCH DATA ---
  const fetchMyBatches = useCallback(async () => {
    setLoading(true);
    try {
      const myRequests = await callApi("/JS/journals/my-requests", null, "GET");

      if (!Array.isArray(myRequests) || myRequests.length === 0) {
        setRows([]);
        setLoading(false);
        return;
      }

      const groupedBatches = new Map();
      for (const request of myRequests) {
        const batchId = request.batchId;
        if (!groupedBatches.has(batchId)) {
          groupedBatches.set(batchId, {
            id: batchId,
            batchId: batchId,
            creatorId: request.creatorId,
            requestStatus: request.requestStatus,
            requestDate: request.requestDate,
            commonBatchRemarks: request.commonBatchRemarks,
            totalDebit: 0,
            totalCredit: 0,
            requestCount: 0,
          });
        }
        
        const batchDetails = groupedBatches.get(batchId);
        batchDetails.requestCount++;
        try {
          const payload = JSON.parse(request.payload);
          const amount = parseFloat(payload.amount) || 0;
          
          // Logic: DB sends negative for credit. Sum absolute value.
          if (amount < 0) batchDetails.totalCredit += Math.abs(amount);
          else batchDetails.totalDebit += amount;
        } catch (e) { console.error(`Payload Error`, e); }
      }
      setRows(Array.from(groupedBatches.values()));
    } catch (err) {
      showSnackBar("Failed to load requests", "error");
    } finally {
      setLoading(false);
    }
  }, [callApi, showSnackBar]);

  useEffect(() => { fetchMyBatches(); }, [fetchMyBatches]);

  // --- HANDLERS ---
  const handleOpenModal = (batch) => {
    setSelectedBatch(batch);
    setIsModalOpen(true);
  };

  const handleCloseModal = () => {
    if (isSubmitting) return;
    setIsModalOpen(false);
    setSelectedBatch(null);
  };

  const handleSubmitBatchCancel = async () => {
    setIsSubmitting(true);
    try {
      await callApi(`/JS/journals/my-requests/by-batch/${selectedBatch.batchId}`, null, "DELETE");
      showSnackBar(`Batch ${selectedBatch.batchId} successfully canceled.`, "success");
      handleCloseModal();
      fetchMyBatches();
    } catch (error) {
      showSnackBar("Failed to cancel batch", "error");
    } finally {
      setIsSubmitting(false);
    }
  };

  const handleOpenViewModal = async (batchId, status) => {
    setIsViewModalOpen(true);
    setIsViewLoading(true);
    setBatchDetails([]);
    setViewingBatchId(batchId);
    setViewingBatchStatus(status);

    try {
      const data = await callApi(`/JS/journals/by-batch/${batchId}`, null, "GET");
      setTimeout(() => {
        if (Array.isArray(data)) {
          const journalGroups = new Map();
          const processedData = data.map((req) => {
              try {
                const payload = JSON.parse(req.payload);
                const prefix = req.journalId.split("-")[0];
                const amount = parseFloat(payload.amount) || 0;
                return {
                  ...req, payload, journalPrefix: prefix, amount: Math.abs(amount),
                  transactionType: amount < 0 ? "Credit" : "Debit",
                };
              } catch (e) { return null; }
            }).filter(Boolean);

          processedData.sort((a, b) => a.journalId.localeCompare(b.journalId));
          for (const row of processedData) {
            const prefix = row.journalPrefix;
            if (!journalGroups.has(prefix)) journalGroups.set(prefix, []);
            journalGroups.get(prefix).push(row);
          }
          setBatchDetails(Array.from(journalGroups.values()));
        }
        setIsViewLoading(false);
      }, 100);
    } catch (error) {
      showSnackBar("Failed to load details", "error");
      setIsViewLoading(false);
    }
  };

  const handleCloseViewModal = () => {
    setIsViewModalOpen(false);
    setBatchDetails([]);
    setViewingBatchId(null);
    setSelectedJournalPrefixes([]);
    setIsPartialSubmitting(false);
    setViewingBatchStatus(null);
    fetchMyBatches(); // Refresh on close
  };

  const handleCancelSelectedJournals = async () => {
    if (selectedJournalPrefixes.length === 0) {
      showSnackBar("Select at least one group to cancel.", "warning"); return;
    }
    setIsPartialSubmitting(true);
    try {
      await callApi("/JS/journals/my-requests/by-journal-list", selectedJournalPrefixes, "DELETE");
      showSnackBar("Selected entries canceled.", "success");
      handleCloseViewModal();
    } catch (error) {
      showSnackBar("Failed to cancel selection", "error");
      setIsPartialSubmitting(false);
    }
  };

  const handleSelectAllClick = (event) => {
    if (event.target.checked) setSelectedJournalPrefixes(batchDetails.map(g => g[0].journalPrefix));
    else setSelectedJournalPrefixes([]);
  };

  const handleRowCheckboxClick = (event, prefix) => {
    if (event) event.stopPropagation();
    const index = selectedJournalPrefixes.indexOf(prefix);
    let newSelected = [];
    if (index === -1) newSelected = [...selectedJournalPrefixes, prefix];
    else newSelected = selectedJournalPrefixes.filter(p => p !== prefix);
    setSelectedJournalPrefixes(newSelected);
  };

  // --- COLUMNS ---
  const columns = [
    { field: "batchId", headerName: "Batch ID", width: 125, align: "center", headerAlign: "center" },
    {
      field: "requestStatus", headerName: "Status", width: 125, align: "center", headerAlign: "center",

    },
    {
      field: "creatorId", headerName: "Creator", width: 150, align: "center", headerAlign: "center",
      renderCell: (params) => <Chip icon={<AccountCircleIcon />} label={params.value} size="small" />
    },
    {
      field: "requestDate", headerName: "Submitted On", width: 200, align: "center", headerAlign: "center",
      renderCell: (params) => new Date(params.value).toLocaleString("en-IN", { dateStyle: "medium", timeStyle: "short" })
    },
    { field: "commonBatchRemarks", headerName: "Creator Remarks", width: 200, align: "center", headerAlign: "center" },
    { field: "requestCount", headerName: "Entries", width: 100, align: "center", headerAlign: "center" },
    {

  return (
    <Box sx={styles.mainContainer}>
      <DataGrid
        rows={rows}
        columns={columns}
        loading={loading}
        getRowId={(row) => row.batchId}
        disableRowSelectionOnClick
        pageSizeOptions={[5, 10, 25]}
        disableColumnResize={true}
        rowHeight={52}
        initialState={{ pagination: { paginationModel: { page: 0, pageSize: 10 } }, sorting: { sortModel: [{ field: "requestDate", sort: "desc" }] } }}
        slots={{ noRowsOverlay: CustomNoRowsOverlay }}
        sx={styles.dataGridContainer}
      />

      {/* --- CANCEL CONFIRM MODAL --- */}
      <Dialog open={isModalOpen} onClose={handleCloseModal} fullWidth maxWidth="sm">
        <DialogTitle sx={{ display: "flex", alignItems: "center" }}>

        </DialogContent>
        <DialogActions sx={styles.dialogActions}>
          <Grid container spacing={2} alignItems="center" justifyContent="flex-end">
            <Grid item>
              {viewingBatchStatus === "PENDING" && (
                <Button variant="contained" color="error" onClick={handleCancelSelectedJournals} disabled={isPartialSubmitting || selectedJournalPrefixes.length === 0}>
                  Cancel Selected
                </Button>
              )}
            </Grid>
          </Grid>
        </DialogActions>
      </Dialog>
    </Box>
  );
}
