   @Override
        public RequestVid createRequest(RequestVidDto dto) {

                // Trim user inputs to avoid validation issues caused by leading/trailing
                // spaces.
                if (dto.getCustomCategory() != null) {
                        dto.setCustomCategory(dto.getCustomCategory().trim());
                }

                if (dto.getDescription() != null) {
                        dto.setDescription(dto.getDescription().trim());
                }

                if (dto.getSdReqNo() != null) {
                        dto.setSdReqNo(dto.getSdReqNo().trim());
                }

                if (dto.getRemarks() != null) {
                        dto.setRemarks(dto.getRemarks().trim());
                }

                Long categoryId = dto.getCategoryId();

                if (dto.getDescription() == null
                                || dto.getDescription().trim().isEmpty()) {

                        throw new BadRequestException(
                                        "Description cannot be empty.");
                }

                // At least one existing or custom issue category must be provided.
                if ((dto.getIssueCategoryIds() == null
                                || dto.getIssueCategoryIds().isEmpty())
                                &&
                                (dto.getCustomIssueCategories() == null
                                                || dto.getCustomIssueCategories().stream()
                                                                .allMatch(s -> s == null || s.isBlank()))) {

                        throw new BadRequestException(
                                        "Select at least one issue category.");
                }

                // Validate selected voucher category.
                if (!voucherCategoryRepository.existsById(categoryId)) {
                        throw new BadRequestException("Invalid Voucher Category selected.");
                }
                VoucherCategory voucherCategory = voucherCategoryRepository.findById(categoryId)
                                .orElseThrow(() -> new BadRequestException("Voucher category not found."));

                String categoryName = voucherCategory.getCategoryName();

                // Create a new voucher category only when "Other" is selected.
                if ("Other".equalsIgnoreCase(categoryName)
                                && (dto.getCustomCategory() == null
                                                || dto.getCustomCategory().isBlank())) {

                        throw new BadRequestException(
                                        "Custom voucher category is required.");
                }

                if ("Other".equalsIgnoreCase(categoryName)) {

                        if (dto.getCustomCategory() != null
                                        && !dto.getCustomCategory().isBlank()) {

                                if ("Other".equalsIgnoreCase(dto.getCustomCategory())) {
                                        throw new BadRequestException(
                                                        "Custom voucher category cannot be named 'Other'. ");
                                }

                                Boolean categoryExists = voucherCategoryRepository
                                                .existsByCategoryName(dto.getCustomCategory());

                                if (!categoryExists) {
                                        VoucherCategory newCategory = new VoucherCategory();
                                        newCategory.setCategoryName(dto.getCustomCategory());
                                        newCategory.setCreatedDate(LocalDateTime.now(ZoneId.of("Asia/Kolkata")));
                                        newCategory = voucherCategoryRepository.save(newCategory);
                                        categoryId = newCategory.getId();
                                } else {
                                        throw new BadRequestException("Custom voucher category already exists.");
                                }

                        }
                }
                Set<Long> finalIssueCategoryIds = new HashSet<>();

                // Validate that each selected issue category belongs to the selected voucher
                // category.
                if (dto.getIssueCategoryIds() != null) {

                        for (Long issueId : dto.getIssueCategoryIds()) {

                                IssueCategory issueCategory = issueCategoryRepository.findById(issueId)
                                                .orElseThrow(() -> new BadRequestException(
                                                                "Issue category not found : "
                                                                                + issueId));

                                if (!issueCategory.getVoucherCategoryId()
                                                .equals(categoryId)) {

                                        throw new BadRequestException(
                                                        "Issue category does not belong to the selected voucher category.");
                                }

                                finalIssueCategoryIds.add(issueId);
                        }
                }

                // Create custom issue categories if they do not already exist.
                if (dto.getCustomIssueCategories() != null) {

                        Set<String> uniqueIssueNames = new HashSet<>();

                        for (String issueName : dto.getCustomIssueCategories()) {

                                if (issueName == null
                                                || issueName.isBlank()) {

                                        continue;
                                }

                                issueName = issueName.trim().replaceAll("\\s+", " ");

                                if (issueName.length() > 100) {
                                        throw new BadRequestException(
                                                        "Issue category cannot exceed 100 characters.");
                                }

                                if ("Other".equalsIgnoreCase(issueName)) {
                                        throw new BadRequestException(
                                                        "Custom issue category cannot be named 'Other'.");
                                }

                                // Ignore duplicate custom issue category names in the same request.
                                if (!uniqueIssueNames.add(
                                                issueName.toLowerCase())) {

                                        continue;
                                }

                                Optional<IssueCategory> existing = issueCategoryRepository
                                                .findByVoucherCategoryIdAndIssueCategoryNameIgnoreCase(
                                                                categoryId,
                                                                issueName);

                                Long issueId;

                                if (existing.isPresent()) {

                                        issueId = existing.get().getId();

                                } else {

                                        IssueCategory issueCategory = new IssueCategory();

                                        issueCategory.setVoucherCategoryId(
                                                        categoryId);

                                        issueCategory.setIssueCategoryName(
                                                        issueName);

                                        issueCategory.setCreatedDate(
                                                        LocalDateTime.now(ZoneId.of("Asia/Kolkata")));

                                        issueCategory = issueCategoryRepository
                                                        .save(issueCategory);

                                        issueId = issueCategory.getId();
                                }

                                // Avoid duplicate mappings when an issue category is selected and also entered
                                // as custom.
                                if (finalIssueCategoryIds.contains(issueId)) {
                                        continue;
                                }

                                finalIssueCategoryIds.add(issueId);
                        }
                }

                String rawDesc = dto.getDescription();
                String cleanedDesc = rawDesc.trim().replaceAll("\\s+", " ");
                // Persist the voucher request.
                RequestVid requestVid = new RequestVid();
                requestVid.setCategoryId(categoryId);
                requestVid.setDescription(cleanedDesc);
                requestVid.setAllowedRole(dto.getAllowedRole());
                requestVid.setSdReqNo(dto.getSdReqNo());
                requestVid.setCreatedBy(dto.getCreatedBy());
                requestVid.setRemarks(dto.getRemarks());
                requestVid.setStatus("ACTIVE");
                requestVid.setCreatedAt(LocalDateTime.now(ZoneId.of("Asia/Kolkata")));
                RequestVid saved = requestVidRepository.saveAndFlush(requestVid);

                RequestVid ret = requestVidRepository.findById(saved.getId())
                                .orElseThrow(() -> new BadRequestException("Unable to fetch created VID."));

                log.info(
                                "Calling notification service for VID={}",
                                ret.getVid());

                // Notify the target role after successful VID creation.
                notificationWriterService.createNotification(

                                String.valueOf(ret.getAllowedRole()),

                                "New VID "
                                                + ret.getVid()
                                                + " has been created.",

                                "/voucher-enquiry/vid/" + ret.getId(),

                                ret.getId().toString(),

                                "VoucherEnquiryService");

                // Persist voucher to issue category mappings.
                for (Long issueCategoryId : finalIssueCategoryIds) {

                        RequestVoucherIssueCategory mapping = new RequestVoucherIssueCategory();

                        mapping.setRequestVoucherId(
                                        saved.getId());

                        mapping.setIssueCategoryId(
                                        issueCategoryId);

                        requestVoucherIssueCategoryRepository
                                        .save(mapping);
                }

                entityManager.clear();

                return ret;
        }
